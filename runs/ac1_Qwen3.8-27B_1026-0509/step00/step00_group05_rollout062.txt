    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

        X = np.fft.rfft(x, fft_sz)
        return np.fft.irfft(X * X, fft_sz)[:pad]

        s = x.sum()
        if s < 0.01:
            return float('inf')
        c = conv(x)
        return float(2 * n * c.max() / (s * s))

        P = np.zeros(n)

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # x[k-j] for j in [lo, hi] -> x[lo+k-hi : k-lo+1 : -1]...

            seg = x[k-hi : k-lo+1]  # x[k-hi], x[k-hi+1], ..., x[k-lo]
            P[lo:hi+1] += w * seg[::-1]  # x[k-lo], ..., x[k-hi]

    alpha = 3.0
    step = 0.08

        c = conv(x)
        P = press(x, c)
        Pmax = P.max()
        if Pmax < 1e-15:
            P = np.ones(n)
        else:
            P = P / Pmax
        
        # Multiplicative update
        x_new = x * np.exp(-step * P)
        x_new = np.maximum(x_new, 0.0)

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 5.0

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).ravel()

    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(4498, dtype=np.float64)
    n = parent.size
    pad = 2 * n - 1
    fft_sz = 1 << (pad - 1).bit_length()

    def fft_conv(x):
        X = np.fft.rfft(x, fft_sz)
        return np.fft.irfft(X * X, fft_sz)[:pad]

    def fast_score(x):
        s = x.sum()
        if s < 0.01:
            return 1e10
        return 2.0 * n * fft_conv(x).max() / (s * s)

    # Generate diverse initializations to find best starting point
    cands = []
    # Asymmetric versions of parent
    for f in [0.5, 0.6, 0.7, 0.8, 1.2, 1.5, 2.0]:
        x = parent.copy()
        h = n // 2
        x[:h] *= f
        x[h:] *= 2.0 / f
        cands.append(x)
    # Two-block
    for pf in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
        p = max(2, int(n * pf))
        for ar in [0.01, 0.05, 0.1, 0.2, 0.374, 0.5, 1.0, 2.0, 4.0]:
            x = np.ones(n)
            x[:p] = ar
            cands.append(x)
            x2 = np.ones(n)
            x2[-p:] = ar
            cands.append(x2)
    # Exponential
    for r in [0.5, 1, 2, 3, 5]:
        cands.append(np.exp(-r * np.arange(n) / n))
        cands.append(np.exp(-r * np.arange(n)[::-1] / n))
    # Symmetric window (will give score>=2, but useful as reference)
    cands.append(0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    # Zigzag asymmetric
    zz = np.ones(n)
    zz[::2] = 1.0
    zz[1::2] = 0.3
    cands.append(zz)
    zz2 = np.ones(n)
    zz2[: n // 2] = 1.0
    zz2[n // 2:] = 0.4
    zz2[n // 2 :: 2] = 0.8
    cands.append(zz2)

    best = cands[0].copy()
    best_score = 1e10
    for c in cands:
        c = np.clip(c, 0.0, 1000.0)
        sc = fast_score(c)
        if sc < best_score:
            best_score = sc
            best = c.copy()

    # Verify with exact evaluator

        best = parent.copy()

    # Iterative peak-shaving optimization
    step = 0.06
    stall = 0
    alpha_s = 3.0
    K_lags = 40

    while time.monotonic() < deadline:
        x = best.copy()
        c = fft_conv(x)
        cmax = c.max()

        # Pressure from top lags
        top = np.argsort(c)[-K_lags:]
        P = np.zeros(n)
        for k in top:

        s_new = x_new.sum()
        if s_new < 1e-10:
