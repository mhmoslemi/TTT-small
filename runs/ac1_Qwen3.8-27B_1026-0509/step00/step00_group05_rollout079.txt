            p = np.exp((c - M) / eps)
            p /= p.sum()
            g = grad_c(p, a)
            eta = 1.0 / (20.0 + 0.05 * it)
            a_new = a - eta * g
            a = proj_simplex(a_new)
            if it % 40 == 0:
                sc = score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_local = a.copy()
            if it % 300 == 0:
                eps_frac *= 0.6
        sc = score(a)
        if sc < best_sc:
            best_sc = sc
            best_local = a.copy()
        return best_local, best_sc

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    if parent.sum() > 0.01:
        inits.append(parent / parent.sum())
    inits.append(np.ones(n) / n)
    for alpha in [0.00005, 0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03]:
        a_e = np.exp(-alpha * np.arange(n, dtype=np.float64))
        inits.append(a_e / a_e.sum())
    for sl in [0.1, 0.5, 1.0, 3.0, 10.0, 30.0]:
        a_r = 1.0 + sl * np.linspace(0, 1, n)
        inits.append(a_r / a_r.sum())
    for _ in range(4):
        a_rand = rng.random(n)

    best_a = inits[0].copy()

    n_inits = len(inits)
    for i, a0 in enumerate(inits):
        if time.monotonic() > deadline:
