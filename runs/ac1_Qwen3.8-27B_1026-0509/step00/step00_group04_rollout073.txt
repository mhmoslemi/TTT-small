                    t_mix = max(0.05, t_mix * 0.7)
        except Exception:
            pass

    # Final safety: ensure a valid finite non-empty list returns
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    if out.size == 0 or not np.isfinite(out).all() or out.sum() < 1e-9:
        out = np.clip(parent, 0.0, 1000.0)
        if out.sum() < 1e-9:
            out = np.ones(n)
    return [float(x) for x in out]
