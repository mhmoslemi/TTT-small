```python
def project_simplex(v, s=1.0):
    """Project v onto the simplex {q: sum q = s, q >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    i = np.max(ind[cond])
    theta = cssv[i-1] / i
    return np.clip(v - theta, 0, None)
```

The idea:
1. Fix the current sequence `p` and its peak value `M = max_k (p*p)_k`
2. Solve: max `sum q_j` s.t. `q_j >= 0`, `(p*q)_k <= M` for all k

5. Update: `p <- (1-t)*p + t*q`

```python
    import cvxpy as cp
    
    n = len(p)

    M = c.max()
    
    # Get top n_top constraint indices
    top_idx = np.argsort(c)[::-1][:n_top]
    
    q = cp.Variable(n, nonneg=True)
    
    # Build constraints using the convolution structure
    constraints = []
    for k in top_idx:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        # (p*q)_k = sum_{j=j_lo}^{j_hi} p[k-j] * q[j]
        coeff = p[k - j_hi : k - j_lo - 1 : -1]  # p[k-j_hi], p[k-j_hi+1], ..., p[k-j_lo]
        constraints.append(coeff @ q <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(q)), constraints)
    prob.solve(solver=cp.CBC, time_limit=30)
    
    if q.value is None:
        return p  # Fallback
    return q.value / q.value.sum()
```
