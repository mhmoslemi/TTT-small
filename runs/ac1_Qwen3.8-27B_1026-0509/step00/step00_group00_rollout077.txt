import numpy as np
import time


def _acorr(a):
    n = a.size
    A = np.fft.rfft(a, 2 * n)
    return np.fft.irfft(np.abs(A) ** 2, 2 * n)


def _score(a):
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return np.inf
    return 2.0 * n * _acorr(a).max() / (S * S)


def _top_lag_grad(a, K):
    n = a.size
    c = _acorr(a)
    idx = np.argpartition(-c, min(K, 2 * n - 1))[:K]
    g = np.zeros(n)
    for m in idx:
        L = (n - 1) + int(m)
        if L < 0 or L > 2 * n - 2:
            continue
        lo = max(0, L - n + 1)
        hi = min(n - 1, L)
        if hi < lo:
            continue
        src = a[L - hi:L - lo + 1]
        g[lo:hi + 1] += 2.0 * src[::-1]
    return g


def _polish(a0, deadline, rng, n):
    a = np.clip(a0.copy(), 0.0, None)
    a = a + 1e-6
    sc = _score(a)
    best = a.copy()
    best_sc = sc
    lr = 1e-3
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # projected subgradient step on the peak autocorrelation
        g = _top_lag_grad(a, 250)
        gnorm = np.linalg.norm(g)
        if gnorm > 0:
            a2 = a - lr * g / (1.0 + lr * a.mean())
            a2 = np.clip(a2, 1e-8, None)
            s2 = _score(a2)
            if np.isfinite(s2) and s2 < sc:
                a = a2
                sc = s2
                lr *= 1.15
            else:
                lr *= 0.65
                if lr < 1e-6:
                    lr = 1e-3
        # occasional annealed perturbation to escape plateaus
        if it % 25 == 0:
            b = a.copy()
            nblock = rng.integers(max(2, n // 40), max(4, n // 6))
            st = rng.integers(0, n - nblock)
            b[st:st + nblock] *= np.exp(rng.normal(0.0, 0.18, nblock))
            if rng.random() < 0.2:
                b *= np.exp(rng.normal(0.0, 0.10))
            b = np.clip(b, 1e-8, None)
            sb = _score(b)
            if np.isfinite(sb) and (sb < sc or rng.random() < min(0.3, sc * 0.05)):

        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        if not np.isfinite(sc):
            a = best.copy()
            sc = best_sc
    return best, best_sc

    inits.append((1.0 - x) ** 0.5)
    inits.append((1.0 + np.cos(np.pi * x)) * 0.5)
    inits.append(np.where(x < 0.5, 1.0, 0.5)).astype(float)
    inits.append(np.where(x < 0.4, 1.0, 0.3).astype(float) * (1.0 - 0.3 * x))
    inits.append(1.0 / (1.0 + 8.0 * x))

    global_best = None
    global_sc = np.inf
    per = max(20.0, (deadline - time.monotonic()) - 10.0) / max(1, len(inits))
    sub_deadline = time.monotonic() + per
    for init in inits:
        init = np.clip(init, 1e-4, None)
        a, sc = _polish(init, min(sub_deadline, deadline), rng, n)
        if np.isfinite(sc) and sc < global_sc:
            global_sc = sc
            global_best = a.copy()
        if time.monotonic() > deadline - 8.0:
            break

    # global polish on the best shape
    if global_best is not None and time.monotonic() < deadline - 6.0:
        a, sc = _polish(global_best, deadline - 6.0, rng, n)
        if np.isfinite(sc) and sc < global_sc:
            global_sc = sc
            global_best = a.copy()

    # exact validation / selection
    a = np.clip(global_best if global_best is not None else np.ones(n), 0.0, 1000.0)
    exact = evaluate_sequence([float(v) for v in a])
    best_seq, best_exact = [float(v) for v in a], exact
    if fallback is not None:
        ef = evaluate_sequence([float(v) for v in fallback])
        if np.isfinite(ef) and (not np.isfinite(exact) or ef < exact):
            best_seq, best_exact = [float(v) for v in fallback], ef

    if not np.isfinite(best_exact):
        # guaranteed-valid minimal background
        best_seq = [1.0] * 64
        best_exact = evaluate_sequence(best_seq)

    return best_seq
