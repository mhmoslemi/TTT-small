            r = rng.random()
            if r < 0.3:
                f = f * np.exp(rng.normal(0, 0.15, n))
            elif r < 0.6:
                p = rng.uniform(0.3, 3.0)
                f = np.power(np.arange(n) + 1, p)
            elif r < 0.8:
                w = int(rng.integers(n // 10, n // 3))
                st = int(rng.integers(0, n - w))
                f2 = f.copy()
                f2[st:st + w] *= np.exp(rng.normal(0, 0.5, w))
                f = f2
            else:
                f = 0.5 * f + 0.5 * (x + 1)
            f = np.clip(f, 0.0, 1000.0)
            no_improve = 0
            continue

        lo, hi = max(0, k - n + 1), min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros(n)
        grad[idx] = f[k - idx]

        g = 2.0 * n * (grad * s - 2.0 * max_b) / (s ** 3)
        f_new = np.maximum(f - lr * g, 0.0)
        f_new = np.minimum(f_new, 1000.0)
        ns = 2.0 * n * fast_conv(f_new).max() / (f_new.sum() ** 2)

        if ns < cur - 1e-15:
            f = f_new
            lr = min(lr * 1.03, 10.0)
        else:
            lr *= 0.8
            if lr < 1e-14:
                break

    final = [float(v) for v in best]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs):
        final = [float(v) for v in np.arange(1, n + 1)]
    return final
