import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def _fast_score(a, n):
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    m = float(np.max(np.convolve(a, a)))
    return float(2.0 * n * m / (s * s))


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + (budget_s - 5.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(256, dtype=np.float64)
    n = int(parent.size)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # Fixed banded structure of the convolution operator: column j (value f[j])
    # touches rows j .. j+n-1.  Precompute indices once.
    jcol = np.arange(n, dtype=np.int32)
    cols_pre = np.repeat(jcol, n)
    rows_pre = np.tile(jcol, n) + np.repeat(jcol, n)
    shape = (2 * n - 1, n)

    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline - 75.0:
        iter_count += 1
        f = best
        s_f = float(np.sum(f))
        if s_f < 0.01:
            break

        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M < 1e-10:
            break

        # Normalize to unit sum for LP well-scaling (problem is scale invariant).
        f_norm = f / s_f
        M_norm = M / (s_f * s_f)

        vals = np.repeat(f_norm, n)
        A = coo_matrix((vals, (rows_pre, cols_pre)), shape=shape,
                       dtype=np.float64).tocsr()

        remaining = deadline - 110.0 - time.monotonic()
        if remaining < 8.0:
            break
        lp_time = min(150.0, max(6.0, remaining))

        result = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M_norm),
                         bounds=[(0.0, None)] * n, method='highs',
                         options={'time_limit': lp_time})

        improved = False
        if result.status == 0 and result.x is not None:
            g = np.maximum(result.x, 0.0)
            s_g = float(np.sum(g))
            if s_g > 1e-9:
                g = (s_f / s_g) * g
                g = np.clip(g, 0.0, 1000.0)

                best_t = None
                best_s = best_score
                for t in (1.0, 0.7, 0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02):
                    nf = np.clip((1.0 - t) * f + t * g, 0.0, 1000.0)
                    sc = _fast_score(nf, n)
                    if np.isfinite(sc) and sc < best_s:
                        best_s, best_t = sc, t
                if best_t is not None:
                    best = np.clip((1.0 - best_t) * f + best_t * g, 0.0, 1000.0)
                    best_score = best_s
                    no_improve = 0
                    improved = True

        if not improved:
            no_improve += 1

            # Subgradient step on F = 2n*max(conv)/sum^2 as a cheap fallback.
            kstar = int(np.argmax(conv_f))
            grad = np.zeros(n)
            jlo = max(0, kstar - n + 1)
            jhi = min(n - 1, kstar)
            grad[jlo:jhi + 1] = 2.0 * f[kstar - np.arange(jlo, jhi + 1)]
            full_grad = (2.0 * n / (s_f * s_f)) * grad \
                - (4.0 * n * M / (s_f ** 3)) * np.ones(n)
            gn = float(np.linalg.norm(full_grad))
            if np.isfinite(gn) and gn > 1e-12:
                for lr in (0.02, 0.05, 0.1, 0.25, 0.5):
                    nf = np.clip(f - lr * full_grad / gn * s_f, 0.0, 1000.0)
                    sc = _fast_score(nf, n)
                    if np.isfinite(sc) and sc < best_score:
                        best, best_score, no_improve = nf, sc, 0
                        improved = True
                        break

            if not improved and no_improve % 4 == 0:
                for _ in range(4):
                    sig = 0.04 + 0.012 * min(no_improve, 20)
                    pf = best * np.exp(rng.normal(0.0, sig, n))
                    pf = np.clip(pf, 0.0, 1000.0)
                    sc = _fast_score(pf, n)
                    if np.isfinite(sc) and sc < best_score:
                        best, best_score, no_improve = pf, sc, 0
                        break

            if no_improve > 60:
                break

    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        fb = np.ones(min(n, 128), dtype=np.float64)
        best, final_score = fb, float(evaluate_sequence(fb.tolist()))
    if not np.isfinite(final_score):
        return [1.0]
    return [float(x) for x in best]
