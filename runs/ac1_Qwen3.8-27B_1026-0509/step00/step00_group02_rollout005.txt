import numpy as np

def solve_lp(f, M, K=20):

    conv = np.convolve(f, f, mode='full')
    # Pick top K peaks
    peak_idx = np.argsort(conv)[::-1][:K]

    constraints = [g >= 0]

        lo = max(0, k - (n-1))

        a = f[k-hi:k-lo+1]
        constraints.append(a @ g[lo:hi+1] <= M)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or another fast solver
    return g.value

import numpy as np
import time
import cvxpy as cp

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0:
        parent = np.ones(256, dtype=np.float64)

    f = parent.astype(np.float64).copy()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = f.size
    sum_f = f.sum()
    stagnant = 0

    def eval_score(x):
        return float(evaluate_sequence(x.tolist()))

    while time.monotonic() < deadline:
        conv = np.convolve(f, f, mode='full')
        M = conv.max()
        cur_score = 2.0 * n * M / (sum_f ** 2)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = f.copy()

        K = min(30 + int(10 * (time.monotonic() - (deadline - budget_s))), 200, 2 * n - 1)
        peak_idx = np.argpartition(conv, -K)[-K:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        g = None
        try:
            g_var = cp.Variable(n)
            constraints = [g_var >= 0]
            for k in peak_idx:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                a = f[k - hi:k - lo + 1]
                if a.sum() > 1e-15:
                    constraints.append(a @ g_var[lo:hi + 1] <= M)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            prob.solve(solver=cp.GLOP, verbose=False)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
        except Exception:
            pass

        improved = False
        if g is not None and g.sum() > 1e-10:
            g = g / g.sum() * sum_f
            for t in (0.3, 0.15, 0.5, 0.7):
                f_new = np.clip((1.0 - t) * f + t * g, 0.0, 1000.0)
                if f_new.sum() < 0.01:
                    continue
                f_new = f_new / f_new.sum() * sum_f
                s = eval_score(f_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best = f_new.copy()
                    f = f_new.copy()
                    improved = True
                    break

        if not improved:
            stagnant += 1
            if stagnant > 30:
                f = f * np.exp(0.05 * rng.normal(0, 1, n))
                f = np.clip(f, 0.0, 1000.0)
                if f.sum() < 0.01:
                    f = best.copy()
                stagnant = 0
        else:
            stagnant = 0

        if time.monotonic() > deadline:
            break

    final_score = eval_score(best)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    return [float(x) for x in best]
