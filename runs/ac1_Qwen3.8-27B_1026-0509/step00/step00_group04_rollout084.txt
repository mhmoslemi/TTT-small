```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = t0 + total - 6.0
    rng = np.random.default_rng(seed)

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        p = None
    if p is None or p.ndim != 1 or p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(3809, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = int(p.size)

    rows_all = 2 * n - 1
    lo_arr = np.empty(rows_all, dtype=np.int64)
    hi_arr = np.empty(rows_all, dtype=np.int64)
    for k in range(rows_all):
        lo_arr[k] = max(0, k - n + 1)
        hi_arr[k] = min(k, n - 1)

    best_score = float("inf")
    best_seq = None

    def norm(x, floor=0.0):
        x = np.asarray(x, dtype=np.float64)
        if x.size == 0:
            return x
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s = x.sum(dtype=np.float64)
        if not np.isfinite(s) or s <= 1e-300:
            return np.ones(x.size, dtype=np.float64) / x.size
        x = x / s
        if floor > 0.0:
            x = np.maximum(x, floor)
            s = x.sum(dtype=np.float64)
            if s > 0:
                x = x / s
        return x

    def store(c, sc):
        nonlocal best_score, best_seq
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = np.array(c, dtype=np.float64, copy=True)

    def score_conv(c, floor=1e-15):
        c = norm(c, floor)
        if c.size != n:
            res = norm(p, floor)
            b = np.convolve(res, res)
            mx = float(np.max(b))
            sc = 2.0 * n * mx / (mx and (res.sum()**2) or 1.0)
            store(res, 2.0 * n * mx)
            return res, b, 2.0 * n * mx
        b = np.convolve(c, c)
        mx = float(np.max(b))
        sc = 2.0 * n * mx
        store(c, sc)
        return c, b, float(sc)

    def add_candidate(a):
        c = norm(a, 1e-15)
        if c.size != n:
            return
        b = np.convolve(c, c)
        mx = float(np.max(b))
        sc = 2.0 * n * mx
        store(c, sc)

    init_stop = t0 + min(75.0, 0.25 * total)

    add_candidate(p)

            a2 = a * np.exp(-lam * (x / xn))
            add_candidate(a2)
            add_candidate(a2[::-1])

        add_candidate(a)
        add_candidate(a[::-1])

    if best_seq is None:

        cur, b, cur_score = score_conv(a0, 1e-15)
        cur = cur.copy()
        eta = 0.09
        no_improve = 0

                store(cur, cur_score)

            if cur_score < 1.45:

            top_count = T
            if top_count >= b.size:
                top = np.arange(b.size, dtype=np.int64)
            else:
                k2 = min(b.size, 2 * T)
                cand_top = np.argpartition(b, -k2)[-k2:]
                if rng.random() < 0.35:
                    cand_top = rng.permutation(cand_top)
                top = cand_top[:top_count]

            w = b[top] / max(cur_score / (2.0 * n), 1e-300)
            w = np.power(np.maximum(w, 0.0), 2.0)

                ww = float(w[idx])

            rows = np.unique(np.concatenate((base_rows, top_rows)))

            if rows.size < 4:

            R = rows.size
            indptr = np.empty(R + 1, dtype=np.int64)
            indptr[0] = 0

                indptr[r + 1] = indptr[r] + cnt

            if total_nnz <= 0:

            data = np.empty(total_nnz, dtype=np.float64)

                hi = int(hi_arr[k])

                idx += cnt

            A = csr_matrix((data, cols, indptr), shape=(R, n))
            remaining = stop - time.monotonic()
            tl = int(max(4, min(90, remaining - 25)))
            if tl < 5:
                break

            res = None
            try:

                    cvec,
                    A_ub=A,
                    b_ub=np.ones(R, dtype=np.float64),
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True, "output_flag": False, "time_limit": tl},
                )
            except Exception:
                try:
                    res = linprog(
                        cvec,
                        A_ub=A,
                        b_ub=np.ones(R, dtype=np.float64),
                        bounds=bounds,
                        method="highs",
                        options={"output_flag": False, "time_limit": tl},
                    )
                except Exception:
                    res = None

            g = None
            if res is not None:
                xv = getattr(res, "x", None)
                if xv is not None and np.all(np.isfinite(xv)):
                    xv = np.asarray(xv, dtype=np.float64)
                    xv = np.maximum(xv, 0.0)
                    ss = xv.sum()
                    if ss > 1e-15:
                        g = xv / ss

            if g is None:
                no_improve += 1
                if no_improve >= 3:
                    break
                continue

            improved = False
            for sym in (False, True):
                gg = g if not sym else 0.5 * (g + g[::-1])
                gg = norm(gg, 1e-15)
                peak = gg.max() * n
                blend = 0.45 if peak > 25.0 else 0.18 if peak > 6.0 else 0.05

                for t in (0.15, 0.30, 0.50, 0.75, 1.0):
                    h = (1.0 - t) * cur + t * gg2
                    hh, _, hsc = score_conv(h, 1e-14)
                    if hsc < best_score - 1e-12:
                        improved = True
                        cur = hh.copy()
                        cur_score = hsc

            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 3:
                    break

        if best_seq is not None and cur_score < best_score:
            store(cur, cur_score)
        return best_seq.copy() if best_seq is not None else flat.copy()

    if best_score > 1.50 and time.monotonic() < deadline - 80.0:
        lp_stop = min(deadline - 45.0, t0 + max(total * 0.68, 350.0))
        best_seq = lp_search(best_seq, lp_stop)

    final_local_stop = max(time.monotonic() + 15.0, min(deadline - 10.0, t0 + total * 0.85))
    best_seq = local_search(best_seq, final_local_stop)

        best_seq = p.copy()

    out = [float(v) for v in norm(best_seq, 0.0)]
    if len(out) == 0:

    try:

        if not np.isfinite(fs):

            fs = float(evaluate_sequence(out))
    except Exception:
        out = [float(v) for v in norm(p, 0.0)]

    if any((v < 0.0 or v > 1000.0 or not np.isfinite(v)) for v in out):
        out = [1.0]

    return out
```
