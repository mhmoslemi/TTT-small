import numpy as np
import time
from scipy.optimize import differential_evolution, minimize


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        return np.inf
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    mb = float(np.max(b))
    return float(2.0 * n * mb / (s * s))


def _pwlin(N, theta, K):
    th = np.asarray(theta, dtype=np.float64).ravel()
    if th.size < 2 * K:
        th = np.concatenate([th, np.zeros(2 * K - th.size)])
    l = th[:K]
    h = th[K:2 * K]
    l = np.exp(l - np.max(l)) + 1e-12
    l /= l.sum()
    h = np.exp(h - np.max(h)) + 1e-12
    bounds = np.concatenate(([0.0], np.cumsum(l)))
    centers = 0.5 * (bounds[:-1] + bounds[1:])
    for i in range(1, K):
        if centers[i] <= centers[i - 1]:
            centers[i] = centers[i - 1] + 1e-10
    x = np.linspace(0.0, 1.0, N)
    f = np.interp(x, centers, h)
    f = np.maximum(f, 1e-14)
    s = f.sum()
    if s <= 0.0:
        return np.ones(N) / N
    return f / s


def _logbasis(N, theta, modes):
    th = np.asarray(theta, dtype=np.float64).ravel()
    need = 2 + 2 * modes
    if th.size < need:
        th = np.concatenate([th, np.zeros(need - th.size)])
    x = np.linspace(0.0, 1.0, N)
    y = x - 0.5
    sig = th[0] * y + th[1] * (y * y - 1.0 / 6.0)
    p = 2
    for m in range(1, modes + 1):
        t = 2.0 * np.pi * m * x
        sig += th[p] * np.cos(t) + th[p + 1] * np.sin(t)
        p += 2
    sig = np.clip(sig, -6.0, 6.0)
    f = np.exp(sig)
    f = np.maximum(f, 1e-12)
    s = f.sum()
    if s <= 0.0:
        return np.ones(N) / N
    return f / s


def _downsample(a, N):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = a[np.isfinite(a)]
    if a.size == 0:
        return np.ones(N) / N
    if a.size == N:
        return a.copy()
    idx = np.linspace(0.0, a.size - 1, N)
    return np.interp(idx, np.arange(a.size), a)


def _smooth_optimize(N, seq, maxfun, temp):
    f = np.asarray(seq, dtype=np.float64).ravel()
    if f.size != N or N < 8:
        f = _downsample(f, N)
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s <= 0.0:
        return None
    f = f / s
    f = np.maximum(f, 1e-12)
    f = f / f.sum()
    local_best = f.copy()
    local_score = _score_arr(local_best)

    L = 2 * N - 1
    P = 1 << max(1, (3 * N - 2).bit_length())
    tau = max(temp, 1e-3) / N

    def purpose(z):
        z = np.asarray(z, dtype=np.float64)
        z = z - np.max(z)
        f = np.exp(z)
        fs = f.sum()
        if fs <= 0.0 or not np.isfinite(fs):
            return 1e100, np.zeros_like(z)
        f = f / fs
        F = np.fft.rfft(f, P)
        b = np.fft.irfft(F * F, P).real[:L]
        b = np.maximum(b, 0.0)
        m = float(np.max(b))
        ex = np.exp((b - m) / tau)
        Z = float(ex.sum())
        if Z <= 0.0 or not np.isfinite(Z):
            return 1e100, np.zeros_like(z)
        w = ex / Z
        obj = 2.0 * N * (m + tau * np.log(Z))
        W = np.fft.rfft(w[::-1], P)
        c = np.fft.irfft(F * W, P).real
        corr = c[L - N:L][::-1]
        grad_f = 4.0 * N * corr
        gz = grad_f - f * float(np.dot(f, grad_f))
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            gz = np.zeros_like(z)
        if not np.isfinite(obj):
            obj = 1e100
        return obj, gz

    z0 = np.log(f) - np.mean(np.log(f))
    try:
        res = minimize(
            purpose,
            z0,
            method="L-BFGS-B",
            options={"maxiter": max(10, min(200, int(maxfun))),
                     "maxfun": max(20, int(maxfun)),
                     "ftol": 1e-9,
                     "gtol": 1e-6,
                     "maxls": 20}
        )
        if np.all(np.isfinite(res.x)):
            z = res.x - np.max(res.x)
            f = np.exp(z)
            fs = f.sum()
            if fs > 0.0:
                f = f / fs
                sc = _score_arr(f)
                if np.isfinite(sc) and sc < local_score:
                    local_best = f
                    local_score = sc
    except Exception:
        pass
    return local_best
