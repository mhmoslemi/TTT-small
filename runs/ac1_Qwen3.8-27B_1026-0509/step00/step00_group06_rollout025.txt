    best_score = get_score(best)
    for c in candidates[1:]:
        s = get_score(c)
        if np.isfinite(s) and s < best_score:
            best = c.copy()
            best_score = s

    a = best.copy()

    # Gradient-based optimization to minimize smooth max of autocorrelation
    t = 50.0
    lr = 0.01
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    step = 0
    no_improve = 0
    best_phase_score = best_score

    while time.monotonic() < deadline:
        step += 1
        b = conv2(a)
        grad = smooth_max_grad(a, b, t)

        m_vec = 0.85 * m_vec + 0.15 * grad
        v_vec = 0.999 * v_vec + 0.001 * grad * grad
        b1 = 1 - 0.85 ** step
        b2 = 1 - 0.999 ** step
        a = a - lr * (m_vec / b1) / (np.sqrt(v_vec / b2) + 1e-8)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best.copy()
        else:
            a /= s

        if step % 30 == 0:
            sc = get_score(a)
            if np.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc
                no_improve = 0
            else:
                no_improve += 30

            if no_improve > 2000:
                # Perturb and adjust parameters
                a = a + 0.03 * rng.normal(size=n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                t = min(t * 1.4, 400.0)
                lr = min(lr * 1.15, 0.04)
                no_improve = 0

            if step % 500 == 0:
                t = min(t * 1.05, 500.0)
                lr = max(lr * 0.95, 0.003)

            if no_improve > 5000:
                # Try a fresh cosine-like restart
                f = rng.uniform(0.5, 4.0) / (2 * (n - 1))
                ph = rng.uniform(0, 2 * np.pi)
                a_new = 1.0 + np.cos(2 * np.pi * f * np.arange(n) + ph)
                a_new = np.maximum(a_new, 0.0)
                if a_new.sum() > 1e-6:
                    a = a_new / a_new.sum()
                    sc2 = get_score(a)
                    if np.isfinite(sc2) and sc2 < best_score:
                        best = a.copy()
                        best_score = sc2
                t = 30.0
                lr = 0.01
                m_vec = np.zeros(n)
                v_vec = np.zeros(n)
                no_improve = 0

    # Final validation
    fs = get_score(best)
    if not np.isfinite(fs):
        return [1.0 / n] * n
    return [float(x) for x in best]
