import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):

def smoothed_max(b, eps):
    """Compute (1/eps) * log(sum exp(eps * b)) and its weights."""
    mb = b.max()
    e = np.exp(eps * (b - mb))
    Z = e.sum()
    F = (mb + np.log(Z)) / eps
    p = e / Z
