def propose_candidate(seed=42, budget_s=1500, **kwargs):
    n = len(height_sequence_1)

    fft_n = 2 * n  # sufficient for full conv of length-n with itself
    
    def convolve_full(a):
        """Full linear convolution of a with itself via FFT."""
        A = np.fft.rfft(a, fft_n)
        return np.fft.irfft(A * A, fft_n)[:2*n-1]
    
    def subgradient(a, conv):
        """Compute subgradient of max_k conv_k w.r.t. a."""
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            indices = np.arange(lo, hi + 1)
            grad[indices] += 2 * a[k - indices]
            if k % 2 == 0:
                mid = k // 2
                if lo <= mid <= hi:
                    grad[mid] -= a[mid]
        grad /= len(active)
        return grad
    
    # Initializations to try
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) * (S / n))
    
    # 2. Cosine window
    t = np.arange(n) / n
    w = np.cos(np.pi * t) ** 2  # Raised cosine
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum() * S)
    
    # 3. Random
    r = np.abs(rng.normal(0, 1, n)) + 0.1
    inits.append(r / r.sum() * S)
    
    # 4. Two-level alternating
    a2 = np.ones(n)
    a2[::2] = 2.0
    inits.append(a2 / a2.sum() * S)
    
    # 5. Parent
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    if parent.sum() > 0:
        inits.append(parent / parent.sum() * S)
    
    # 6. Linear ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum() * S)
    
    # Optimize each initialization
    per_init_time = (budget_s - 20) / len(inits)
    
    for init_a in inits:
        if time.monotonic() > deadline - 30:
            break
        
        a = init_a.copy()
        a_sum = a.sum()
        if a_sum <= 0:
            continue
        a = a * S / a_sum
        
        init_deadline = time.monotonic() + per_init_time
        
        # Subgradient descent
        step_base = 0.01
        momentum = 0.9
        vel = np.zeros(n)
        
        for step in range(100000):
            if time.monotonic() > init_deadline or time.monotonic() > deadline - 30:
                break
            
            conv = convolve_full(a)
            max_val = conv.max()
            
            # Evaluate periodically
            if step % 50 == 0:
                score = max_val / 1.0  # AC1 = 2n * max_val / S^2 = 2n * max_val / (2n) = max_val
                # Wait: AC1 = 2*n*max_b/sum_a^2 = 2*n*max_val/S^2 = 2*n*max_val/(2*n) = max_val
                # So AC1 = max_val when sum_a = S = sqrt(2n)!
                # So we just need to minimize max_val!
                if max_val < best_score:
                    best_score = max_val
                    best = a.copy()
            
            if step % 500 == 0:
                # Full evaluation
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best = a.copy()
            
            # Compute subgradient
            grad = subgradient(a, conv)
            
            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            eta = step_base / (1 + 0.1 * step * 0.01)
            
            # Nesterov momentum
            vel = momentum * vel + (1 - momentum) * grad
            a_new = a - eta * vel
            
            # Project: clip to nonnegative, rescale to sum S
            a_new = np.maximum(a_new, 0)
            a_new_sum = a_new.sum()
            if a_new_sum > 0:
                a_new = a_new * S / a_new_sum
            else:
                a_new = np.ones(n) * S / n
                break
            
            a = a_new
        
        # Final check
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
    
    return [float(x) for x in best]
