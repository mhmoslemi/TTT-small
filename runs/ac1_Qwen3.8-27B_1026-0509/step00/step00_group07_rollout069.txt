import numpy as np
import time

try:
    import cvxpy as cp
    _HAS_CP = True

    _HAS_CP = False


def _fft_conv(a):
    a = np.ascontiguousarray(a, dtype=np.float64)
    n = a.size
    nout = 2 * n - 1
    P = 1
    while P < nout:
        P <<= 1
    A = np.fft.rfft(a, P)
    c = np.fft.irfft(A * A, P)[:nout]
    return c


def _grad_direction(a, p):
    # g[i] = 2 * sum_k p[k] a[k-i] (valid k-i in [0,n-1]), i in [0,n-1]
    n = a.size
    Q = 1
    while Q < (2 * n - 1) + n - 1:
        Q <<= 1
    G = np.fft.rfft(a, Q)
    F = np.fft.rfft(p, Q)
    full = np.fft.irfft(G * F, Q)[:n]
    return 2.0 * full


def _norm1(a):
    s = a.sum()
    if s <= 1e-12:
        return None
    return a / s


def _score_fast(a):
    b = _fft_conv(a)
    return float(2 * a.size * b.max())


def _two_level(n):
    m = n // 2
    a = np.ones(n)
    a[m:] = 0.5
    if n - 2 * m == 1:
        a[-1] = 0.75
    return _norm1(np.clip(a, 0.0, 1000.0))


def _block_perturb(a, rng, scale=0.15):
    n = a.size
    c = a.copy()
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))

    c[s:s + w] *= float(np.exp(rng.normal(0.0, scale)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    r = _norm1(np.clip(c, 0.0, 1000.0))
    return r


def _lp_direction(f, M, rng, max_active=None):

        lo = max(0, k - (m - 1))
        hi = min(m - 1, k)
        A[row, lo:hi + 1] = f[k - (hi - lo):k - 0][::-1]
    var = cp.Variable(m)
    cons = [var >= 0, A @ var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(var)), cons)

                g0 = np.clip(var.value, 0.0, None)
                if g0.sum() > 1e-12:
                    return g0
        except Exception:
            continue
    return None
