def eval_fast(a):
    """Fast evaluation: 2n * max(conv) / sum^2"""
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * len(a) * b.max() / (s * s)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Water filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
