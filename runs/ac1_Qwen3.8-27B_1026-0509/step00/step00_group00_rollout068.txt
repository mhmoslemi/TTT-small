import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def _score(a):
    return float(evaluate_sequence([float(x) for x in a]))


def _fftconv(a, b):
    la, lb = len(a), len(b)
    sz = 1
    while sz < la + lb - 1:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:la + lb - 1]


def _build_conv_mat(f):
    n = len(f)
    off = np.tile(np.arange(n, dtype=np.int32), n)
    col = np.repeat(np.arange(n, dtype=np.int32), n)
    row = col + off
    data = np.repeat(f.astype(np.float64), n)
    return sparse.coo_matrix((data, (row, col)), shape=(2 * n - 1, n)).tocsr()


def _perturb(base, rng, deg):
    c = base.copy()
    n = c.size
    w = int(np.clip(n * deg, 2, n // 3))
    s = int(rng.integers(0, max(1, n - w + 1)))
    c[s:s + w] *= np.exp(rng.normal(0.0, 0.15))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if f.ndim != 1 or f.size == 0 or not np.isfinite(f).all():
        f = np.ones(256)
    n = len(f)
    target = float(np.sqrt(2.0 * n))
    s0 = f.sum()
    if s0 < 1e-8:
        f = np.full(n, 1.0)
        s0 = f.sum()
    f = f * (target / s0)
    f = np.clip(f, 0.0, 1000.0)

    best = f.copy()
    best_score = _score(best)
    lp_ok = False
    stall = 0

    while time.monotonic() < deadline:
        # attempt an LP-based improvement if time allows
        if time.monotonic() < deadline - 60.0:
            try:
                ff = _fftconv(f, f)
                M = float(ff.max())
                if M > 0 and M < 1e12:
                    A = _build_conv_mat(f)
                    c = -np.ones(n)
                    b = np.full(2 * n - 1, M)
                    res = linprog(c, A_ub=A, b_ub=b,
                                  bounds=[(0.0, None)] * n,
                                  method='highs',
                                  options={'presolve': True, 'time_limit': 60.0})
                    lp_ok = True
                    if res.success and res.x is not None:
                        g = np.clip(res.x, 0.0, None)
                        gs = g.sum()
                        if gs > 1e-8 and np.isfinite(gs):
                            g = g * (target / gs)
                            g = np.clip(g, 0.0, 1000.0)
                            improved = False
                            for tt in (1.0, 0.5, 0.25, 0.75):
                                cand = (1.0 - tt) * f + tt * g
                                cs = _score(cand)
                                if np.isfinite(cs) and cs < best_score:
                                    best = cand.copy()
                                    best_score = cs
                                    f = cand.astype(np.float64) / cand.sum() * target
                                    f = np.clip(f, 0.0, 1000.0)
                                    improved = True
                                    stall = 0
                                    break
                            if not improved:
                                stall += 1
                            continue
            except Exception:
                lp_ok = False

        # local perturbation polish (cheap, O(n) memory)
        cand = _perturb(best, rng, 0.01 if lp_ok else 0.05)
        cs = _score(cand)
        if np.isfinite(cs) and cs < best_score:
            best = cand.copy()
            best_score = cs
            if stall >= 4 and n > 16:
                # occasional restart around current best to escape plateau
                best = _perturb(best, rng, 0.08)
                best_score = _score(best)
            stall = 0
        else:
            stall += 1

    final_score = _score(best)
    if not np.isfinite(final_score):
        final_score = _score(np.clip(f, 0.0, 1000.0))
    if time.monotonic() > deadline - 3.0:
        pass
    return [float(x) for x in best]
