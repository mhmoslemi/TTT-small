def proj_simplex(v, s=1):
  if v.size==1 return [s]
  u = np.sort(v)[::-1]
  cssv = np.cumsum(u)
  rho = np.nonzero(u * np.arange(1, len(v)+1) - (cssv - s) > 0)[0][-1]
  theta = (cssv[rho] - s)/ (rho+1)
  return np.maximum(v - theta, 0)

```
def _proj_simplex(v, s=1.0):
    v = np.asarray(v, float).copy()
    if v.size == 1:
        v[0]=s if np.isfinite(v[0]) else 0
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, v.size+1) - (cssv - s) > 0)[0]
    if rho.size == 0:
       return np.full(v.shape, s/v.size)
    r = rho[-1]
    theta = (cssv[r] - s)/(r+1)
    out = np.maximum(v - theta, 0.0)
    out_sum = out.sum()
    if out_sum <=0 or not finite: return uniform
    if abs(out_sum - s) > 1e-12: out *= s/out_sum
    return out
```

```python
def _convolve_full(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    if n < 256:
        c = np.convolve(x, x)

def _normalize(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return None
    p = np.clip(p, 0.0, 1000.0)

    s = float(p.sum())
    if s <= 1e-14:
        return None
    return p / s


def _score_norm(p):
    p = _normalize(p)
    if p is None:
        return float("inf")
    q = _convolve_full(p)
    return float(2.0 * p.size * float(q.max()))


def _proj_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    if n == 1:
        return np.array([s], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho = np.nonzero(u * idx - (cssv - s) > 0.0)[0]
    if rho.size == 0:
        return np.full(n, s / n, dtype=np.float64)
    r = rho[-1]
    theta = (cssv[r] - s) / float(r + 1)
    out = np.maximum(v - theta, 0.0)
    out_sum = float(out.sum())

        return np.full(n, s / n, dtype=np.float64)

    return out

        return _proj_simplex(p)
    elif k == 10:
        x = rng.lognormal(0.0, rng.uniform(0.5, 1.5), n)
    else:
        w = rng.uniform(0.1, 2.0, n)
        x = w / w.sum()

    x = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(x)) or float(x.sum()) <= 1e-12:
        x = np.ones(n, dtype=np.float64)
    return _proj_simplex(x)


def _local_gradient(p, beta, max_iter, deadline, rng=None):
    p = _normalize(p)
    if p is None:
        return p, float("inf")
    n = p.size
    best = p.copy()
    best_score = _score_norm(p)
    step = 1.0
    beta = float(beta)
    no_move = 0

    if float(np.abs(p - p.mean()).max()) < 1e-13:
        if rng is not None:
            p = _proj_simplex(p + 0.02 * rng.standard_normal(n))
        else:
            p = _proj_simplex(p + 0.02 * np.cos(2.0 * math.pi * np.arange(n) / n))

    for _ in range(max_iter):
        if time.monotonic() > deadline:
            break
        q = _convolve_full(p)
        M = float(q.max())
        if M <= 1e-18:
            break
        old_score = 2.0 * n * M

        shifted = beta * (q - M)
        w = np.exp(shifted)
        sw = float(w.sum())
        if sw <= 1e-18 or not np.isfinite(sw):
            w = np.zeros_like(q)
            w[int(np.argmax(q))] = 1.0
        else:
            w /= sw

        r = _correlate_valid(w, p)
        grad = 2.0 * r
        if not np.all(np.isfinite(grad)) or float(np.dot(grad, grad)) <= 1e-24:
            break

        old_f = float(np.dot(w, q))
        accepted = None
        accepted_q = None
        accepted_score = old_score
        accepted_s = None
        s = step

        for _ls in range(6):
            cand = _proj_simplex(p - s * grad)
            if not np.all(np.isfinite(cand)):
                break
            cq = _convolve_full(cand)
            sc = float(2.0 * n * cq.max())
            f = float(np.dot(w, cq))
            if sc < old_score - 1e-12 or (f < old_f - 1e-10 and sc <= old_score + 1e-6):
                accepted = cand
                accepted_q = cq
                accepted_score = min(sc, old_score)
                accepted_s = s
                if sc < old_score - 1e-8:
                    break
            s *= 0.5

        if accepted is None:
            no_move += 1
            if no_move > 50:
                if rng is None:
                    break
                noise = 0.03 * min(1.0, 12.0 / math.sqrt(max(1, n)))
                p = _proj_simplex(p + noise * rng.standard_normal(n))
                beta = beta * 1.4 if beta < 300.0 else max(10.0, beta * 0.5)
                step = 1.0
                no_move = 0
                continue
            step *= 0.5
            if step < 1e-8:
                break
        else:
            p = accepted
            q = accepted_q
            score = float(2.0 * n * q.max())
            if score < best_score - 1e-12:
                best = p.copy()
                best_score = score
            no_move = 0
            step = min(10.0, max(1e-6, float(accepted_s) * 1.5))

            if best_score <= (2.0 * n) / (2.0 * n - 1.0) * (1.0 + 1e-8):
                break

    sc = _score_norm(p)
    if sc < best_score:
        best = p.copy()
        best_score = sc
    return best, best_score


def _lp_refine(a, max_iter=6, deadline=None):
    a = _normalize(a)
    if a is None:
        return None, float("inf")
    n = a.size
    if n < 2 or n > 250:
        return a, _score_norm(a)

    best = a.copy()
    bestM = float(_convolve_full(best).max())

    for _ in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        M = bestM
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        try:
            res = optimize.linprog(
                -np.ones(n),
                A_ub=A,
                b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                bounds=[(0.0, None)] * n,
                method="highs"
            )
        except Exception:
            break

        if res.status != 0 or res.x is None or not np.all(np.isfinite(res.x)):
            break
        g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
        s = float(g.sum())
        if s <= 1e-12:
            break
        g /= s

        cur_best = a
        cur_bestM = bestM
        for t in (1.0, 0.75, 0.5, 0.35, 0.2, 0.1):
            f = _proj_simplex((1.0 - t) * a + t * g)
            mf = float(_convolve_full(f).max())
            if mf < cur_bestM - 1e-12:
                cur_best = f
                cur_bestM = mf
        if cur_bestM >= bestM - 1e-12:
            break
        a = cur_best
        bestM = cur_bestM
        best = a.copy()

    return best, 2.0 * n * bestM


def _de_search(n, seed, deadline):
    def obj(y):
        if not np.all(np.isfinite(y)):
            return 1e100
        z = y - float(y.max())
        p = np.exp(z)
        s = float(p.sum())
        if s <= 1e-14:
            return 1e100
        p /= s
        return float(2.0 * n * _convolve_full(p).max())

    maxiter = 120 if n <= 8 else 70 if n <= 12 else 35
    popsize = 15 if n <= 8 else 12 if n <= 12 else 9
    try:
        res = optimize.differential_evolution(
            obj,
            [(-6.0, 6.0)] * n,
            seed=seed,
            maxiter=maxiter,
            popsize=popsize,
            init="latinhypercube",
            strategy="best1bin",
            tol=1e-9,
            polish=False,
            workers=1,
            updating="immediate"
        )
        if time.monotonic() > deadline:
            return None, float("inf")
        if res.x is not None:
            p = _normalize(res.x)
            if p is not None:
                return p, _score_norm(p)
    except Exception:
        pass
    return None, float("inf")


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    deadline = t0 + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)

    best = np.ones(1, dtype=np.float64)
    best_score = float("inf")
    best_by_n = {}

    def update(p, n_key=None):
        nonlocal best, best_score
        p = _normalize(p)
        if p is None:
            return
        sc = _score_norm(p)
        if np.isfinite(sc) and sc < best_score:
            best = p.copy()
            best_score = sc
        if n_key is not None:
            key = int(n_key)
            old = best_by_n.get(key)
            if old is None:
                best_by_n[key] = (p.copy(), sc)
            else:
                if sc < old[1]:
                    best_by_n[key] = (p.copy(), sc)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = _normalize(parent)
    except Exception:
        parent = None

    if parent is not None:
        update(parent, parent.size)

    update(np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64))
    update(np.ones(10, dtype=np.float64) / 10.0)

    for n in range(5, 91, 2):
        if time.monotonic() > deadline - 30.0:
            break
        s = _legendre_sequence(n)
        if s is None:
            continue
        for beta in (0.1, 0.25, 0.4, 0.6, 0.9, 1.3, 1.8):
            if time.monotonic() > deadline - 30.0:
                break
            p = _normalize(np.exp(beta * s) + 1e-12)
            update(p, n)
            p2 = _normalize(np.maximum(1.0 + beta * s, 0.0))
            update(p2, n)

    lengths = [4, 5, 6, 7, 8, 9, 10, 12, 14, 16, 20, 24, 30, 40, 60, 100, 150]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        if n <= 8:
            K, steps = 26, 260
            betas = (15.0, 55.0)
        elif n <= 20:
            K, steps = 16, 200
            betas = (20.0, 70.0)
        elif n <= 60:
            K, steps = 10, 130
            betas = (30.0, 80.0)
        else:
            K, steps = 6, 90
            betas = (50.0,)

        for s in range(K):
            if time.monotonic() > deadline:
                break
            p = _make_start(n, s, rng)
            sc = _score_norm(p)
            update(p, n)
            for beta in betas:
                if time.monotonic() > deadline:
                    break
                p2, sc2 = _local_gradient(p, beta, steps, deadline, rng)
                update(p2, n)
                if sc2 < best_score:
                    break

        if n <= 180 and (n in best_by_n) and time.monotonic() < deadline:
            p_lp, sc_lp = _lp_refine(
                best_by_n[n][0],
                max_iter=8 if n <= 80 else 4,
                deadline=deadline
            )
            update(p_lp, n)

    if time.monotonic() < deadline:
        for n in (5, 6, 7, 8, 9, 10, 12):
            if time.monotonic() > deadline:
                break
            p, sc = _de_search(n, seed + 1000 + n, deadline)
            if p is not None:
                update(p, n)
                p2, sc2 = _local_gradient(p, 80.0, 160, deadline, rng)
                update(p2, n)

    if parent is not None and time.monotonic() < deadline:
        p = parent.copy()
        if p.size <= 250 and (p.size not in best_by_n):
            best_by_n[p.size] = (p, _score_norm(p))
        else:
            if p.size in best_by_n:
                p = best_by_n[p.size][0]
        beta = 30.0 if p.size <= 200 else 20.0
        steps = 600 if p.size <= 100 else 350 if p.size <= 600 else 220
        p2, sc2 = _local_gradient(p, beta, steps, deadline, rng)
        update(p2, p2.size if p2 is not None else p.size)
        if p2 is not None and p2.size <= 200:
            p3, sc3 = _lp_refine(p2, max_iter=4, deadline=deadline)
            update(p3, p3.size if p3 is not None else p2.size)
        elif p2 is not None and time.monotonic() < deadline:
            p4, sc4 = _local_gradient(p2, 80.0, 200, deadline, rng)
            update(p4, p4.size if p4 is not None else p2.size)

    if time.monotonic() < deadline:
        for beta in (20.0, 60.0, 150.0):
            if time.monotonic() > deadline:
                break
            p2, sc2 = _local_gradient(best, beta, 180, deadline, rng)
            update(p2, p2.size if p2 is not None else best.size)

    if time.monotonic() < deadline:
        noise_scale = 0.025 * min(1.0, 12.0 / math.sqrt(max(1, best.size)))
        for _ in range(500):
            if time.monotonic() > deadline:
                break
            cand = _proj_simplex(best + noise_scale * rng.standard_normal(best.size))
            sc = _score_norm(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    final = _normalize(best)
    if final is None:
        final = np.array([1.0], dtype=np.float64)

    try:
        fin_score = float(evaluate_sequence(final.tolist()))
        if not np.isfinite(fin_score):
            final = np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64)
    except Exception:
        pass

    return [float(x) for x in final]
```
