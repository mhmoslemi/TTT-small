```python
import numpy as np
import time

def fft_convolve(x, y):
    """Linear convolution via FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()  # next power of 2
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    return np.fft.irfft(X * Y, size)[:n]

def score_fast(x):
    """Fast version of evaluate_sequence using FFT."""
    x = np.maximum(0, np.minimum(1000, x))
    S = x.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(x, x)
    max_b = c.max()
    return 2 * len(x) * max_b / (S * S)
```

```python
def coordinate_optimize(x, c, j, fft_size, n):
    """Optimize x[j] to minimize max_k c[k], given current c and x."""
    # c[k] = c_k0 + d_k * x[j]
    # where d_k = 2*x[k-j] if 0 <= k-j < n, else 0
    # and c_k0 = c[k] - 2*x[j]*x[k-j] (for valid k-j)
    # Actually, c[k] currently includes the x[j] terms. If we change x[j] to v:
    # c_new[k] = c[k] - 2*x[j]*x[k-j] + 2*v*x[k-j] = c[k] + 2*(v-x[j])*x[k-j]
    # (for valid k-j; if k=2j, it's c[k] - x[j]^2 + v^2 = c[k] - x[j]^2 + v^2)
    
    # For simplicity, handle the general case:
    # c_new[k] = c[k] + 2*(v - x[j]) * x[k-j] for k-j valid and k != 2j
    # c_new[2j] = c[2j] - x[j]^2 + v^2 (if 2j < 2n-1)
    
    # This is a piecewise-linear function of v. We want to minimize max_k c_new[k].
    # At v = x[j] (current), the max is c[k*].
    # Moving v up or down changes different c[k] values.
    
    # For now, use a simple bisection on the 1D subproblem.
    # The optimal v minimizes max_k (a_k + b_k * v) where:
    #   b_k = 2*x[k-j] (for valid k-j, k != 2j)
    #   a_k = c[k] - 2*x[j]*x[k-j] (for those k)
    #   For k = 2j: a_{2j} = c[2j] - x[j]^2, b_{2j} = 0, and we have v^2 instead.
    #   For invalid k-j: a_k = c[k], b_k = 0.
    
    # The v^2 term complicates things. For now, ignore k=2j (or handle separately).
    
    # Actually, for the 1D minimax of affine functions:
    # min_v max_k (a_k + b_k * v)
    # This is solved by the "minimax" of lines. The optimal v is where the upper envelope has a flat minimum.
    
    # Simpler approach: ternary search on v in [0, 1000], since the function is unimodal (convex).
    
    best_v = x[j]
    best_max = c.max()

    for v in [0, x[j]*0.5, x[j]*1.5, max(x[j]+1, 1)]:

        # c_new[k] = c[k] + 2*(v-x[j])*x[k-j] for valid k-j
        # This is O(n) to compute
        d = 2*(v - x[j])
        # c_new = c + d * (x shifted)
        # x[k-j] for k in [j, j+n-1] is x[0..n-1]
        c_new = c.copy()  # O(n) - but we want to avoid this allocation

        max_val = 0
        for k in range(j, j+n):
            val = c[k] + d * x[k-j]
            if val > max_val:
                max_val = val
        # Also k outside [j, j+n-1]: c_new[k] = c[k]
        # The max of those is c.max() minus the max in [j, j+n-1]... 
        # Actually, c_new[k] = c[k] for k < j or k >= j+n
        # So max_outside = max of c[k] for k not in [j, j+n-1]
        # This is O(n) to compute
        
        # For efficiency, let's just compute the full max
        # But O(n) per trial * 4 trials * n positions = 4n^2 per sweep. That's 200M for n=6962.
        # At ~10ns per op: 2s per sweep. 1500s / 2s = 750 sweeps. OK.

            best_max = max_val
            best_v = v
    
    x[j] = best_v
```

Actually, the key realization: for coordinate descent, we don't need to recompute the full c at each step. We can maintain c incrementally.

When we change x[j] from old to new:
