import numpy as np
import time
import math
from scipy.optimize import minimize


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.ones(v.size, dtype=np.float64) / v.size
    v = v - np.max(v)
    e = np.exp(v)
    s = e.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(v.size, dtype=np.float64) / v.size
    return e / s


class _ConvOp:
    __slots__ = ("n", "m", "L")

    def __init__(self, n):
        self.n = int(n)
        self.m = 2 * self.n - 1
        need = max(2, 3 * self.n - 2)
        L = 1
        while L < need:
            L <<= 1
        self.L = L

    def conv_and_a(self, a):
        A = np.fft.rfft(a, self.L)
        c = np.fft.irfft(A * A, self.L)[: self.m]
        c = np.where(c < 0.0, 0.0, c)
        return A, c

    def conv(self, a):
        return self.conv_and_a(a)[1]

            if loga.size != n or not np.all(np.isfinite(loga)):

            loga = loga - float(np.mean(loga))

            a = _softmax(loga)
            c = op.conv_and_a(a)[1]

            loga_work = loga
            if n < 64:

    base_schedule = [
