min_a max_k h_k(a) subject to constraints

where h_k(a) = conv(a,a)[k] / S^2 (normalized).

The exponential weights / multiplicative weights approach:
- Maintain p[k] (probability over lags)
- At each step: 
  - Compute a subgradient direction using p: s = sum_k p[k] * grad h_k(a) = 2 * sum_k p[k] * (shifted a by k)
  - Update a in the direction -s (with projection)
  - Update p: p[k] ∝ 1 + eta * h_k(a) (or p[k] ∝ exp(eta * h_k(a)))

This should converge to a near-optimal solution.

For n = 6121, each gradient computation is O(n) (since it's a shifted/reversed version of a). Each convolution to evaluate h_k(a) is O(n) if we only need specific k values, or O(n log n) via FFT for all k.

Let me use FFT for the convolution: np.fft.irfft(np.fft.rfft(a)**2) gives conv(a,a) in O(n log n).

For n = 6121, an FFT takes about 100μs. So we can do thousands of iterations in a second.

import numpy as np
import time

def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def _eval(a):
    return float(evaluate_sequence(a.tolist()))

def _lp_dual(a, M, k_active):
    """Solve dual LP: min M*sum(y) s.t. B'y >= 1, y >= 0."""
    m = len(k_active)
    n = len(a)
    # Build B: B[j,k] = a[k_active[k] - j] if 0 <= k_active[k]-j < n
    B = np.zeros((n, m))
    for idx, k in enumerate(k_active):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        g_lo = k - hi
        g_hi = k - lo
        B[g_lo:g_hi+1, idx] = a[lo:hi+1]
    # Dual: min M*1'y s.t. B'y >= 1, y >= 0
    # Use scipy linprog: min c'y s.t. A_ub y <= b_ub, bounds
    from scipy.optimize import linprog
    c = M * np.ones(m)
    A_ub = -B
    b_ub = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*m, method='highs')
    if res.success and res.x is not None:
        y = np.clip(res.x, 0, None)
        # Recover primal from complementary slackness
        r = B.T @ y  # >= 1
        active_j = np.where(r < 1 + 1e-3)[0]
        if len(active_j) == 0:
            active_j = np.where(r < 1 + 1e-1)[0]
        if len(active_j) == 0:
            return None
        # BFS on columns B[:, active_j]
        cols = B[:, active_j]
        nc = len(active_j)
        # Solve cols @ alpha = 1
        sol, residuals, rank, sv = np.linalg.lstsq(cols, np.ones(nc), rcond=1e-12)
        if sol is None or np.any(sol < -1e-8):
            sol = np.clip(sol, 0, None)
            if sol.sum() < 1e-10:
                sol = np.ones(nc) / nc
            S_primal = sol.sum()
            g = np.zeros(n)
            g[active_j] = sol / S_primal
            return g
        else:
            S_primal = sol.sum()
            if S_primal < 1e-10:
                return None
            g = np.zeros(n)
            g[active_j] = sol / S_primal
            return g
    return None
