        width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        candidate = best.copy()
        candidate[start:start + width] *= float(np.exp(rng.normal(0.0, 0.03)))
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                # Nudge back toward uniform
                candidate = 0.5 * best + 0.5 * mean_val
                candidate = np.clip(candidate, 0.0, 1000.0)
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = candidate
                    best_score = score
                stagnant = 0

    return [float(x) for x in best]
