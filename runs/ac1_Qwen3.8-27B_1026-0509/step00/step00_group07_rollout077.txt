        s = x.sum()
        return x / s if s > 1e-15 else np.full(n, 1.0 / n)

    # Symmetry-breaking noise

    if not np.isfinite(best_score):
        best_a = np.ones(n) / n
        best_score = 2.0

    def one_step(arr, beta, eta):
        c = fftconvolve(arr, arr, method='fft')
        mc = c.max()
        sh = np.clip(beta * (c - mc), -50, 0.0)
        w = np.exp(sh)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= ws
        g = correlate(w, arr, mode='full', method='fft')[:n]
        gc = g - g.mean()
        gs = gc.std()
        if gs > 1e-15:
            gc /= gs
        a_new = arr * np.exp(np.clip(-eta * gc, -5, 5))
        return normalize(a_new), c

    phases = [(50, 0.08, 0), (200, 0.08, 0), (1000, 0.06, 0), (5000, 0.05, 0)]

        nonlocal a, pi, stall
