import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        parent = np.ones(100)
    parent = np.clip(parent, 0, 1000)
    n = parent.size
    if n < 2:
        return [1.0, 1.0]

    L = 1 << (2 * n - 1 - 1).bit_length()

    def convolve_fft(a_vec):
        ap = np.zeros(L)
        ap[:n] = a_vec
        A = np.fft.rfft(ap)
        B = A * A
        return np.fft.irfft(B, L)[:2 * n - 1]

    def score(a_vec):
        b = convolve_fft(a_vec)
        s = a_vec.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * n * float(np.max(b)) / (s * s)

    def gradient_topk(a_vec, b, K=5):
        top_idx = np.argsort(b)[-K:]
        w = b[top_idx].astype(np.float64)
        sw = w.sum()
        if sw < 1e-30:
            return np.zeros(n)
        w /= sw
        j = np.arange(n)
        grad = np.zeros(n)
        for ii in range(K):
            kk = int(top_idx[ii])
            idx = kk - j
            valid = (idx >= 0) & (idx < n)
            grad[valid] += w[ii] * 2.0 * a_vec[idx[valid]]
        return grad

    def optimize(a, max_iters, deadline_t, base_lr):
        local_best = a.copy()
        local_score = score(a)
        lr = base_lr
        stuck = 0
        for it in range(max_iters):
            if time.monotonic() > deadline_t:
                break
            b = convolve_fft(a)
            grad = gradient_topk(a, b, K=5)
            gc = grad - grad.mean()
            a_add = a - lr * gc
            a_add = np.maximum(a_add, 1e-8)
            a_add /= a_add.sum()
            a_mult = a / np.sqrt(1.0 + lr * np.maximum(grad, 0))
            a_mult = np.maximum(a_mult, 1e-8)
            a_mult /= a_mult.sum()
            a = 0.6 * a_add + 0.4 * a_mult
            a = np.maximum(a, 1e-8)
            a /= a.sum()
            if it % 30 == 0:
                sc = score(a)
                if sc < local_score:
                    local_best = a.copy()
                    local_score = sc
                    stuck = 0
                else:
                    stuck += 1
                    if stuck > 100:
                        bs = max(5, n // 80)
                        st = int(rng.integers(0, n - bs))
                        a[st:st + bs] *= np.exp(rng.normal(0, 0.4, bs))
                        a = np.maximum(a, 1e-8)
                        a /= a.sum()
                        stuck = 0
                        lr = max(0.05, lr * 0.7)
            if it % 500 == 0:
                lr = max(0.02, min(5.0, lr * (1.002 if local_score > 1.7 else 0.998)))
        return local_best, local_score

    inits = []

    # 1. Spectral: flat magnitude, random phase

    A_mag[0] = 10.0
    phases = rng.uniform(-np.pi, np.pi, L // 2 + 1)
    A_spec = A_mag * np.cos(phases) + 1j * A_mag * np.sin(phases)
    A_spec[0] = A_mag[0]
    if A_spec.size > 1:

    ap = np.fft.irfft(A_spec, L)
    a1 = np.maximum(ap[:n], 0.01)
    a1 /= a1.sum()
    inits.append(a1)

    # 2. Cosine modulation
    k = np.arange(n)
    a2 = 1.0 + 0.4 * np.cos(2 * np.pi * k / n) + 0.1 * np.cos(4 * np.pi * k / n)
    a2 = np.maximum(a2, 0.01)
    a2 /= a2.sum()
    inits.append(a2)

    # 3. Two-frequency cosine
    a3 = 1.0 + 0.3 * np.cos(2 * np.pi * k / n) + 0.3 * np.cos(2 * np.pi * 3 * k / n + 0.5)
    a3 = np.maximum(a3, 0.01)
    a3 /= a3.sum()
    inits.append(a3)

    # 4. Exponential modulation
    a4 = np.exp(0.3 * np.sin(2 * np.pi * k / n))
    a4 /= a4.sum()
    inits.append(a4)

    # 5. Random smooth

    # Smooth it

    a5 = uniform_filter(a5, size=max(3, n // 50))
    a5 = np.maximum(a5, 0.01)
    a5 /= a5.sum()
    inits.append(a5)

    a6 = 1.0 + 0.5 * w

        a_try = best.copy()

            a_try[nb:nb + sc_] *= np.exp(rng.normal(0, 0.15, sc_))
        a_try = np.maximum(a_try, 0.0)
        ssum = a_try.sum()

        sc_tr = score(a_try)
        if sc_tr < best_score:
            best = a_try
            best_score = sc_tr

    final_list = [float(v) for v in best]
    fs = float(evaluate_sequence(final_list))
    if not np.isfinite(fs) or fs >= 2.0:
        fb = parent / parent.sum()
        final_list = [float(v) for v in fb]

    return final_list
