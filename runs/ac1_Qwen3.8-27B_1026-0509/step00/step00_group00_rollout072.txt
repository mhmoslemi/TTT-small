import numpy as np
import time


def _fft_convolve(a, b):
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    if size < n_out:
        size = 1 << (n_out.bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def _normalize(a, target):
    s = a.sum()
    if s > 1e-30:
        a = a * (target / s)
    return a


def _optima_from(a0, n, eps, lr, deadline, rng):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    target = float(n)
    a = _normalize(a, target)
    best = a.copy()
    c = _fft_convolve(a, a)
    best_sc = 2.0 * c.max() / n
    inv_n = 1.0 / n
    it = 0
    stall = 0
    a_rev_buf = np.empty(n)
    while time.monotonic() < deadline:
        a = np.clip(a, 0.0, 1000.0)
        c = _fft_convolve(a, a)
        m = c.max()
        w = np.exp((c - m) / eps)
        wsum = w.sum()
        if wsum < 1e-300:
            break
        w /= wsum
        np.copyto(a_rev_buf, a[::-1])
        cv1 = _fft_convolve(w, a_rev_buf)
        first = cv1[n - 1: 2 * n - 1]
        cv2 = _fft_convolve(a, w)
        second = cv2[:n]
        grad = first + second
        a = a - lr * grad
        a = _normalize(np.clip(a, 0.0, 1000.0), target)
        it += 1
        sc = 2.0 * c.max() * inv_n
        if sc < best_sc - 1e-9:
            best_sc = sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            a = best + rng.normal(0.0, 0.02, n)
            a = _normalize(np.clip(a, 0.0, 1000.0), target)
            stall = 0
    return best
