def make_A(a):
    n=a.size
    A=np.zeros((2*n-1,n), dtype=float)
    # row k: j=0..min(n,k+1)-1, A[k,j]=a[k-j]
    # Vectorize via for k

        A[k, :jend] = a[k-jend+1:k+1][::-1]
    return A

import numpy as np, time

def _p2(m):
    return 1 << (m-1).bit_length()

def _fft_self(x,L):
    X=np.fft.rfft(x,L)
    return np.fft.irfft(X*X,n=L)[:2*x.size-1]

def _fft_corr(w,a,Far,L):
    W=np.fft.rfft(w,L)
    n=a.size
    return np.fft.irfft(W*Far,n=L)[n-1:2*n-1]

def _fft_conv(v,Fa,L):
    V=np.fft.rfft(v,L)
    return np.fft.irfft(V*Fa,n=L)[:2*v.size-1]

def _score(a,L=None):
    a=np.asarray(a,dtype=float).ravel()

    c=_fft_self(a,L)
    M=float(c.max())
    if not np.isfinite(M): return np.inf,a
    return 2.0*a.size*M,a

import numpy as np
import time


def _p2(m):
    return 1 << (m - 1).bit_length()


def _fft_self(x, L):
    X = np.fft.rfft(x, L)
    return np.fft.irfft(X * X, n=L)[: 2 * x.size - 1]


def _fft_corr(w, a, Far, L):
    W = np.fft.rfft(w, L)
    n = a.size
    return np.fft.irfft(W * Far, n=L)[n - 1 : 2 * n - 1]


def _fft_conv(v, Fa, L):
    V = np.fft.rfft(v, L)
    return np.fft.irfft(V * Fa, n=L)[: 2 * v.size - 1]


def _score(a, L=None):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return np.inf, a
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12 or not np.isfinite(s):
        return np.inf, a
    a = a / s
    if L is None:
        L = _p2(2 * a.size - 1)
    c = _fft_self(a, L)
    M = float(c.max())
    if not np.isfinite(M):
        return np.inf, a
    return 2.0 * a.size * M, a


def _left_flat(n):
    a = np.ones(n, dtype=float)
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a / a.sum()


def _lp_dir(a, Fa, Far, L1, L2, M, deadline):
    n = a.size
    y = np.ones(2 * n - 1, dtype=float) / (2 * n - 1)
    best_mu = 0.0
    best_T = None

    for _ in range(22):
        if time.monotonic() >= deadline:
            break

        T = _fft_corr(y, a, Far, L2)
        T = np.clip(T, 0.0, None)
        mu = float(T.min())
        if mu > best_mu * (1.0 + 1e-9):
            best_mu = mu
            best_T = T.copy()

        tmax = float(T.max())
        if tmax <= mu:
            break
        scale = tmax - mu
        v = np.exp(-(T - mu) / scale)
        vs = float(v.sum())
        if vs <= 0.0 or not np.isfinite(vs):
            break
        v /= vs

        s = _fft_conv(v, Fa, L1)
        smax = float(s.max())
        if smax <= 0.0 or not np.isfinite(smax):
            break

        y *= np.exp(np.clip(s / smax, 0.0, 1.0))
        ysum = float(y.sum())
        if ysum <= 0.0 or not np.isfinite(ysum):
            break
        y /= ysum

        if mu <= best_mu * (1.0 + 1e-8):
            break

    if best_T is None or best_mu <= 0.0:
        return None

    trng = float(best_T.max() - best_T.min())
    if trng <= 1e-14:
        return [np.ones(n, dtype=float) / n]

    out = []
    for al in (25.0, 90.0):
        b = np.exp(-al * (best_T - best_T.min()) / trng)
        bs = float(b.sum())
        if bs > 0.0 and np.isfinite(bs):
            b /= bs
            out.append(b)
    return out or None


def _local_search(a, deadline, rng):
    n = a.size
    a = np.clip(np.asarray(a, dtype=float), 0.0, 1000.0)

    if s <= 1e-12 or not np.isfinite(s):

    L1 = _p2(2 * n - 1)
    L2 = _p2(3 * n - 2)

    c = _fft_self(a, L1)
    bestM = max(0.0, float(c.max()))
    best = a.copy()

    tau_vals = (0.05, 0.15, 0.40)

                x = np.exp(logx - logx.max())
                res = evalx(x)
                if res is None:

                    a, M = x, mn

                        best, bestM = a.copy(), M
                    improved = True
                    break

        if improved:

        else:

def propose_candidate(seed=42, budget_s=1500, **kwargs):
