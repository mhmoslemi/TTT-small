import time


def propose_candidate(seed=42, budget_s=1500, **kwargs):

    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    def exact_score(a):
        # a assumed sum == 1 ; matches evaluator exactly (scale invariant)
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:2 * n - 1]
        return float(2.0 * n * c.max())

    # FFT size large enough for both a*a (>=2n-1) and grad conv (>=3n-2)
    N = 1 << max(2, (3 * n - 2).bit_length())

    def conv_a(a):
        A = np.fft.rfft(a, N)
        return np.fft.irfft(A * A, N)[:2 * n - 1]

    def smooth_grad(a, c, tau):
        tc = tau * c
        tc -= tc.max()
        p = np.exp(tc)
        ps = p.sum()
        if not np.isfinite(ps) or ps <= 0.0:
            return None
        p /= ps
        flip_a = a[::-1]
        full = np.fft.irfft(np.fft.rfft(flip_a, N) * np.fft.rfft(p, N), N)
        return 2.0 * full[n - 1:n - 1 + n]

    def proj_simplex(v):
        m = v.size
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, m + 1) > (cssv - 1.0)
        if not cond.any():
            return np.full(m, 1.0 / m)
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

    a0 = parent / parent.sum()
