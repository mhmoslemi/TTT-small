```python
import numpy as np
import time
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog

def solve_lp(f, M, n):
    """Solve max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    # Build sparse A: (2n-1) x n
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        # A[k, j] = f[k-j] if 0 <= k-j < n
        j_start = max(0, k-n+1)
        j_end = min(n, k+1)
        for j in range(j_start, j_end):
            A[k, j] = f[k-j]
    A = A.tocsr()
    
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

        n = f.size

    if f.sum() > 0:
        f *= S_target / f.sum()
    
    best = f.copy()

    # Precompute sparsity pattern for convolution matrix
    total = 2 * n - 1
    # Column j has nonzeros at rows j, j+1, ..., min(j+n-1, total-1)
    n_per_col = np.minimum(n, total - np.arange(n))  # = min(n, 2n-1-j) for j=0..n-1

    cols = np.repeat(np.arange(n), counts)

    val_idx = rows - cols  # index into f

        return csr_matrix((data, (rows, cols)), shape=(total, n))

            return res.x
        return None

    while time.monotonic() < deadline:

        if time.monotonic() + 30 > deadline:
            break

        M = b.max()

        if g is None:
            patience += 1
            if patience > 3:
                break
            continue
        
        S_g = g.sum()
        if S_g < 1e-10:
            break
        
        g = g * (S_target / S_g)
        f_new = (1 - t_step) * f + t_step * g
        f_new = np.clip(f_new, 0, 1000)
