    deadline = time.monotonic() + budget_s - 3.0

    N = 1 << (2 * n - 1).bit_length()

    K = 20  # number of top peaks to consider
    beta = 50.0  # softmax sharpness
    
    no_improve_count = 0
    last_best_score = best_score
    
    while time.monotonic() < deadline:
        # Compute convolution
        F = np.fft.rfft(f, N)
        conv = np.fft.irfft(F * F, N)[:2 * n - 1]
        
        # Get top-K peaks with softmax weights
        if K >= len(conv):
            top_idx = np.arange(len(conv))
            top_vals = conv
        else:

            top_vals = conv[top_idx]

        w = np.exp(beta * (top_vals - top_vals.max()))
        w /= w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += w[i] * 2.0 * f[idx[mask]]
        
        # Gradient step
        f = f - lr * grad
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = f0 / f0.sum()
        else:
            f = f / s
        
        # Decay learning rate slowly
        lr *= 0.9998
        
        # Periodic evaluation (every ~200 iterations or so)
        if time.monotonic() > deadline - 5.0:
            break
        
        # Check score periodically
        # Use a cheap proxy: 2n * max(conv) (since sum=1, score = 2n*max(conv))
        proxy_score = 2.0 * n * np.max(conv)
        if proxy_score < last_best_score * 1.001:

            test = (f * 10).tolist()
            sc = float(evaluate_sequence(test))
