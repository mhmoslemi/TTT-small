            g[i_lo:i_hi+1] = x[k-i_lo:k-i_hi-1:-1]

        # i-k in [0, n-1] means i in [k, k+n-1], intersected with [0, n-1]
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2+1] += x[i_lo2-k:i_hi2-k+1]
        
        return g
    
    # This is still somewhat slow for n=6962 due to the slicing. Let me use FFT instead.
    # Actually, the gradient of C[k] w.r.t. x is just the "cross-correlation" of x with a delta at position k.
    # More precisely, it's 2 * (x correlated with delta_k), but with the full convolution indexing.

    # For the full convolution C[m] = sum_i x[i]x[m-i],

    # find the max, and compute the corresponding gradient.

    fft_size = 1
    while fft_size < 2*n - 1:
        fft_size *= 2
    
    def compute_conv(x):
        F = np.fft.rfft(x, fft_size)
        C = np.fft.irfft(F * F, fft_size)
        return C[:2*n-1]
    
    def compute_grad(x, k):
        """Gradient of C[k] w.r.t. x, computed in O(n)."""
        g = np.zeros(n)
        # g[i] = x[k-i] (if 0<=k-i<n) + x[i-k] (if 0<=i-k<n)
        # Part 1: x[k-i] for i where 0<=k-i<n
        #   i ranges from max(0, k-n+1) to min(n-1, k)
        #   x[k-i] for these i: x[k - (k-n+1)] to x[k - 0] = x[n-1] to x[k] (reversed)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # x[k-lo], x[k-lo-1], ..., x[k-hi]
            g[lo:hi+1] = x[hi:k-lo+1][::-1]

        #   i ranges from max(0, k) to min(n-1, k+n-1)
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            # x[lo2-k], x[lo2-k+1], ..., x[hi2-k]
            g[lo2:hi2+1] += x[lo2-k:hi2-k+1]
        
        return g
    
    # Main optimization loop
    # Normalize: work with sum(x) = n
    x = parent.copy()
    
    alpha = 0.01  # initial step size
    no_improve = 0
    
    while time.time() < deadline:

        max_C = C.max()
        score = 2.0 * n * max_C / n**2  # since sum = n, score = 2*n*max_C/n^2 = 2*max_C/n
        
        if score < best_score:
            best = x.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        k_star = int(np.argmax(C))

        g = compute_grad(x, k_star)

        g -= g.mean()

        x_new = x - alpha * g
        x_new = np.maximum(x_new, 0)
        s_new = x_new.sum()
        if s_new > 0:
            x_new = x_new * n / s_new
        else:
            x_new = x.copy()

        C_new = compute_conv(x_new)
        if C_new.max() < max_C:
            x = x_new
            alpha = min(alpha * 1.05, 0.1)  # increase step size slightly
        else:

            x = x * n / x.sum()
