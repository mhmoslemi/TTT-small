import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    t0 = time.monotonic()
    deadline = t0 + max(1.0, budget_s - 15.0)
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(3809, dtype=np.float64)
    
    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # LP-based optimization
    for iteration in range(30):
        if time.monotonic() > deadline:
            break
        iter_start = time.monotonic()
        
        # Compute current convolution and its max
        ff = np.convolve(best, best)
        M = float(np.max(ff))
        
        # Build the constraint matrix A (2n-1) x n
        # A[k, j] = best[k-j] if 0 <= k-j <= n-1
        # Column j is best placed starting at row j
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = best
        
        # LP: maximize sum(g) subject to A @ g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

            break
        
        g = res.x

        if sum_g < 1e-10:

        sum_best = float(np.sum(best))
        g_scaled = g * (sum_best / sum_g)

        new_f = (1 - t) * best + t * g_scaled

import numpy as np
import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 15.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(3809, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Precompute index structure for the convolution constraint matrix
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    diff_clipped = np.clip(diff, 0, n - 1)

    rng = np.random.default_rng(seed)
    no_improve = 0

    for iteration in range(500):
        if time.monotonic() > deadline:
            break

        # Compute current autocorrelation peak
        ff = np.convolve(best, best)
        M = float(np.max(ff))

        # Build constraint matrix A[k,j] = best[k-j] for valid indices
        A = np.where(mask, best[diff_clipped], 0.0)

        # Solve LP: maximize sum(g) subject to A@g <= M, g >= 0
        try:
            res = linprog(
                -np.ones(n),
                A_ub=A,
                b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': 60}
            )
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = res.x
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            break

        sum_best = float(np.sum(best))
        g_scaled = g * (sum_best / sum_g)

        # Try mixing with various step sizes
        improved = False
        for t in [0.5, 0.3, 0.7, 0.2, 0.1, 0.4, 0.8, 0.15, 0.6]:
            new_f = np.clip((1.0 - t) * best + t * g_scaled, 0.0, 1000.0)
            if np.sum(new_f) < 0.01:
                continue
            score = float(evaluate_sequence(new_f.tolist()))
            if np.isfinite(score) and score < best_score:
                best = new_f
                best_score = score
                improved = True
                break

        if not improved:
            no_improve += 1
            # If LP is stuck, try perturbations
            if no_improve >= 5:
                for _ in range(200):
                    if time.monotonic() > deadline:
                        break
                    width = int(rng.integers(max(1, n // 50), max(2, n // 8)))
                    start = int(rng.integers(0, n - width))
                    cand = best.copy()
                    cand[start:start + width] *= float(np.exp(rng.normal(0, 0.3)))
                    cand = np.clip(cand, 0.0, 1000.0)
                    score = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best = cand
                        best_score = score
                        no_improve = 0
                        improved = True
                        break
                if not improved and no_improve >= 20:
                    break
        else:
            no_improve = 0

    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
