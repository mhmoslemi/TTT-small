Valid: 0 <= n-1-p < n → 0 <= p <= n-1 ✓

import numpy as np
import time
from scipy import fft as spfft

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s)) - 5.0

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 1:
        parent = np.ones(100)
    n = parent.size
    N_fft = 1 << (3 * n - 1).bit_length()

    best = parent.copy().astype(np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    def run_descent(a0, max_iter, T, lr_init):
        a = np.maximum(a0, 0.0).astype(np.float64)
        S = a.sum()
        if S < 1e-10:
            return a0, np.inf
        a = a / S
        local_best = a.copy()
        a_f = spfft.rfft(a, N_fft)
        b = spfft.irfft(a_f * a_f, N_fft)[:2 * n - 1]
        local_score = 2.0 * n * b.max()
        lr = lr_init
        momentum = np.zeros(n)
        beta = 0.85

        for it in range(max_iter):
            if time.monotonic() > deadline - 3.0:
                break
            b_max = b.max()
            w = np.exp(T * (b - b_max))
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros_like(b)
                w[int(np.argmax(b))] = 1.0
            else:
                w /= ws
            w_f = spfft.rfft(w, N_fft)
            a_rev_f = spfft.rfft(a[::-1], N_fft)
            g = 2.0 * spfft.irfft(a_rev_f * w_f, N_fft)[n - 1:2 * n - 1]
            momentum = beta * momentum + (1.0 - beta) * g
            a = a - lr * momentum
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-10:
                a = local_best.copy()
                lr *= 0.5
            else:
                a /= S
            a_f = spfft.rfft(a, N_fft)
            b = spfft.irfft(a_f * a_f, N_fft)[:2 * n - 1]
            sc = 2.0 * n * b.max()
            if sc < local_score:
                local_score = sc
                local_best = a.copy()
                lr = min(lr * 1.01, 5.0)
            else:
                lr *= 0.97
            lr = max(lr, 1e-7)
        return local_best, local_score

    def local_search(a0, n_trials=200):
        a = np.maximum(a0, 0.0).copy()
        S = a.sum()
        if S < 1e-10:
            return a0, np.inf
        a /= S
        b = spfft.irfft(spfft.rfft(a, N_fft) ** 2, N_fft)[:2 * n - 1]
        sc = 2.0 * n * b.max()
        local_best = a.copy()
        local_score = sc
        w_min = max(1, n // 200)
        w_max = max(2, n // 8)
        for t in range(n_trials):
            if time.monotonic() > deadline - 3.0:
                break
            width = int(rng.integers(w_min, w_max + 1))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            cand = a.copy()
            cand[start:stop] *= np.exp(rng.normal(0.0, 0.15))
            cand = np.maximum(cand, 0.0)
            S = cand.sum()
            if S < 1e-10:
                continue
            cand /= S
            b = spfft.irfft(spfft.rfft(cand, N_fft) ** 2, N_fft)[:2 * n - 1]
            cs = 2.0 * n * b.max()
            if cs < local_score:
                local_score = cs
                local_best = cand.copy()
                a = cand.copy()
        return local_best, local_score

    base = parent / max(parent.sum(), 1e-10)
    x = np.arange(n, dtype=np.float64)
    inits = [base.copy()]
    for amp in [0.02, 0.05, 0.1, 0.2, 0.4]:
        inits.append(np.maximum(base + amp * rng.standard_normal(n), 0.0))
    for f in [1, 2, 3, 5, 8, 13]:
        for amp in [0.05, 0.1, 0.2]:
            inits.append(np.maximum(base + amp * np.sin(2 * np.pi * f * x / n), 0.0))
    for f in [1, 3, 7]:
        for amp in [0.1, 0.3]:
            inits.append(np.maximum(base + amp * np.cos(2 * np.pi * f * x / n), 0.0))
    for _ in range(5):
        inits.append(rng.random(n))
    for _ in range(3):
        r = rng.random(n)
        r[n // 3:2 * n // 3] *= 0.3
        inits.append(r)

    T_lr = [(50.0, 0.05), (200.0, 0.02), (1000.0, 0.01)]

    for a0 in inits:
        if time.monotonic() > deadline - 30.0:
            break
        a0 = np.maximum(a0, 0.0)
        S = a0.sum()
        if S < 1e-10:
            continue
        a0 = a0 / S
        for T, lr0 in T_lr:
            if time.monotonic() > deadline - 10.0:
                break
            result, sc = run_descent(a0, max_iter=8000, T=T, lr_init=lr0)

                best_score = sc2

        r, sc = local_search(best, n_trials=500)
        if sc < best_score:
            best = r.copy()
            best_score = sc

    final = [float(max(0.0, min(1000.0, v))) for v in best]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs):
        final = [float(max(0.0, min(1000.0, v))) for v in parent]
    return final
