def _fft_convolution(x, fft_size, conv_len):
    X = np.fft.rfft(x, fft_size)
    return np.fft.irfft(X * X, fft_size, n=fft_size)[:conv_len]


def _lag_gradient(x, k, n):
    g = np.zeros(n)
    j0 = max(0, k - n + 1)
    j1 = min(k, n - 1)
    if j1 < j0:
        return g
    js = np.arange(j0, j1 + 1)
    ks = k - js
    mask = js != ks
    g[js[mask]] = 2.0 * x[ks[mask]]
    if np.any(~mask):
        g[js[~mask]] = 2.0 * x[js[~mask]]
    return g


def _local_score(x, n, fft_size, conv_len):
    S = float(np.sum(x))
    if S < 0.01:
        return float("inf")
    B = float(np.max(_fft_convolution(x, fft_size, conv_len)))
    return 2.0 * n * B / (S * S)
