        # dq/db_a = sum_k w_k * dc_k/db_a
        # dc_k/db_a_j = 2*a[k-j] (if valid) -- this is the gradient w.r.t. a
        # Weighted: db_q_j = 2 * sum_k w_k * a[k-j] * [0<=k-j<n]
        #            = 2 * sum_i a[i] * w[i+j] (where w index is i+j)
        # This is a cross-correlation of w and a.
        
        # Compute weighted gradient using FFT:
        # db_q_j = 2 * sum_{i: valid} a[i] * w[i+j]
        # = 2 * (a * w)[j] in some convolution sense
        # Actually: let's define W[i] = w[i] for i in 0..2n-2
        # Then db_q_j = 2 * sum_i a[i] * W[i+j] for i+j in 0..2n-2
        # = 2 * sum_i a[i] * W[i+j] = 2 * cross_corr(W, a)[j]
        # In FFT terms: cross_corr(W, a) = ifft(fft(W) * conj(fft(a)))
        
        # Let's compute W padded to length L
        W = np.zeros(L)
        W[:2*n-1] = w
        WA = np.fft.rfft(W, L)
        aA = np.fft.rfft(a, L)
        # Cross correlation: ifft(WA * conj(aA)) gives (W ⊛ a)
        # (W ⊛ a)_j = sum_i W_i * a_{i+j} ... hmm, need to be careful
        # Actually np.correlate style: (W ⊛ a)_j = sum_i W_j+i * a_i
        # In FFT: ifft(fft(W) * conj(fft(a))) gives (W * a)[k] = sum_j W_j a_{k-j} (linear conv)
        # That's not quite what I want.
        
        # Let me just do it directly. For n=4806, a direct computation is O(n^2) which is
        # too slow for 4806. Let me use FFT.
        
        # What I want: grad_a[j] = 2 * sum_{i: 0<=i<n, 0<=i+j<2n-1} a[i] * w[i+j]
        # Let me define: for j = 0,...,n-1:
        #   grad_a[j] = 2 * sum_{i=0}^{n-1} a[i] * w[i+j]
        # (where w[m] = 0 if m >= 2n-1, which it already is)
        
        # This is 2 * (a cross-correlated with w, offset by j)
        # = 2 * np.convolve(w, a[::-1], mode='full')[n-1+j : 2n-1][::1][:n]... 
        # This is getting confusing. Let me use a different approach.
        
        # Note: grad_a[j] = 2 * sum_i a[i] * w[i+j]
        # Define b[i] = a[i] for i in 0..n-1, 0 otherwise
        # Define c[m] = w[m] for m in 0..2n-2, 0 otherwise
        # Then grad_a[j] = 2 * sum_i b[i] * c[i+j]
        # This is 2 * (b ⊛ c)[j] where (b ⊛ c)[j] = sum_i b[i] c[i+j] (cross-correlation)
        # = 2 * ifft(fft(b) * conj(fft(c)))... no
        # Cross-correlation (b ⊛ c)[j] = sum_i b[i] conj(c[i+j]) for complex
        # For real: = sum_i b[i] c[i+j]
        # In FFT: (b ⊛ c) = ifft(conj(fft(b)) * fft(c))
        # Wait, let me verify: ifft(conj(B) * C)[j] = sum_k B_k^* C_k exp(2πikj/L)/L
        # = sum_k (sum_i b_i exp(-2πiki/L))^* (sum_m c_m exp(-2πikm/L)) exp(2πikj/L) / L
        # = sum_i sum_m b_i c_m exp(-2πiki/L) exp(2πikm/L) exp(2πikj/L) / L
        # = sum_i sum_m b_i c_m delta_{i, m+j} (circular, for large L it's linear)
        # = sum_i b_i c_{i+j}
        # Yes! So (b ⊛ c)[j] = ifft(conj(fft(b)) * fft(c))[j]
        
        b_padded = np.zeros(L)
        b_padded[:n] = a
        B = np.fft.rfft(b_padded, L)
        Cspec = np.fft.rfft(W, L)  # W already has zeros after 2n-1
        cross = np.fft.irfft(np.conj(B) * Cspec, L)[:n]
        grad_a = 2.0 * cross
        
        # Also the -2q/S term from the S in the denominator
        # df/db_a_j = (2n/S^2)*(grad_a_j) + (2n*q_smooth)*(-2/S^3)
        # = (2n/S^3) * (S * grad_a_j - 2 * q_smooth)
        grad_fa = (2.0 * n / S**3) * (S * grad_a - 2.0 * q_smooth)
