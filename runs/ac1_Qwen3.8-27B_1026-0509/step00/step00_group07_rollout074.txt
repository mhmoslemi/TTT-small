def _lp_step(a, M):
    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(L):
        lo = max(0, j - n + 1)
        hi = min(n, j + 1)
        for i in range(lo, hi):
            A[j, i] = a[j - i]
    c = -np.ones(n)
    b_ub = np.full(L, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    return res.x
