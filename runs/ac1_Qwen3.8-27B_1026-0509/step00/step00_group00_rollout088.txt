def _fft_conv(a,b):
    need = a.size + b.size - 1
    size = 1
    while size < need: size <<=1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)[:need]
    out[out < 0] = 0
    return out

import numpy as np
import time
import math

try:
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog
except Exception:
    toeplitz = None
    linprog = None


def _fft_conv(a, b):
    need = a.size + b.size - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)[:need]
    out[out < 0.0] = 0.0
    return out


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if a.size == 0:
        return np.array([1.0], dtype=np.float64) * target
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(a.size, target / a.size, dtype=np.float64)
    return a * (target / s)


def _softmax_y(y):
    z = y - float(np.max(y))
    e = np.exp(z)
    s = float(np.sum(e))
    if not math.isfinite(s) or s <= 0.0:
        return np.full_like(y, 1.0 / y.size, dtype=np.float64)
    return e / s


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not math.isfinite(s) or s < 0.01:
        return np.inf
    n = a.size
    if n <= 1024:
        c = np.convolve(a, a)
    else:
        c = _fft_conv(a, a)
    mc = float(np.max(c))
    if not math.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def _make_random(n, style, rng):
    n = int(max(2, n))
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    t = x / float(n - 1)
    style = int(style) % 5
    if style == 0:
        a = np.exp(rng.normal(0.0, 0.75, n))
    elif style == 1:
        a = np.exp(3.0 * (t - 0.5))
    elif style == 2:
        f = 1.0 + 8.0 * float(rng.random())
        ph = 2.0 * math.pi * float(rng.random())
        a = np.exp(1.0 * np.sin(f * t + ph) + 0.6 * rng.normal(0.0, 1.0, n))
    elif style == 3:

    else:

        a += 0.1 * rng.random(n)
        a *= (0.3 + 1.4 * t)
    return _norm(a, 1.0)


def _mirror_search(a0, t_end, rng):
    a = _norm(a0, 1.0)
    n = a.size
    if n < 3:
        return a.copy(), _score_arr(a)

    S = math.sqrt(2.0 * n)
    a = _norm(a, S)
    y = np.log(np.maximum(a, 1e-300))
    y = y - float(np.mean(y))

    best = a.copy()
    best_score = _score_arr(a)
    last_imp = 0
    step = 0
    eps = 0.18
    eta = 0.62
    K = int(min(190, max(16, n // 32)))

    while time.monotonic() < t_end - 0.75:
        c = _fft_conv(a, a)
        maxc = float(np.max(c))
        s = float(np.sum(a))
        sc = 2.0 * n * maxc / (s * s) if s > 1e-300 else np.inf

        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best = a.copy()
            last_imp = step

        if not math.isfinite(maxc) or maxc <= 1e-16:
            break

        if step > 0 and step % 120 == 0:
            eps = max(0.004, 0.90 * eps)
            eta = max(0.10, 0.985 * eta)
            K = min(230, K + 2)

        if step - last_imp > 750:
            y = y + rng.normal(0.0, 0.28, n)
            y = y - float(np.mean(y))
            y = np.clip(y, -45.0, 45.0)
            eps = min(0.28, 1.7 * eps)
            eta = min(1.30, eta + 0.22)
            last_imp = step

        mx = maxc
        Kloc = min(K, c.size)
        if Kloc >= c.size:
            idx = np.arange(c.size, dtype=np.int64)
        else:
            idx = np.argpartition(c, c.size - Kloc)[-Kloc:]

        vals = c[idx] - mx
        w = np.exp(vals / max(eps, 1e-12))
        ws = float(np.sum(w))
        if not math.isfinite(ws) or ws <= 0.0:
            w = np.ones(Kloc, dtype=np.float64) / float(Kloc)
        else:
            w = w / ws

        g = np.zeros(n, dtype=np.float64)
        aa = a
        for k, wk in zip(idx, w):
            if wk <= 1e-9:
                continue
            lo = int(k) - n + 1
            if lo < 0:
                lo = 0
            hi = int(k)
            if hi >= n:
                hi = n - 1
            if hi < lo:
                continue
            g[lo:hi + 1] += wk * aa[k - hi:k - lo + 1][::-1]

        H = g - float(np.mean(g))
        if not np.all(np.isfinite(H)):

        rms = math.sqrt(float(np.mean(H * H)))

            rms = 1.0

        y = y - eta * (H / rms)
        y = y - float(np.mean(y))
        y = np.clip(y, -50.0, 50.0)
        a = _softmax_y(y) * S
        step += 1

    return best, best_score


def _lp_refine(a0, t_end):
    if toeplitz is None or linprog is None:
        return a0, _score_arr(a0)

    a = np.asarray(a0, dtype=np.float64)
    if a.size < 32:
        return a, _score_arr(a)

    if a.size > 2048:
        m = 2048
        old_x = np.linspace(0.0, 1.0, a.size)
        new_x = np.linspace(0.0, 1.0, m)
        a = np.interp(new_x, old_x, a)

    n = a.size
    S = math.sqrt(2.0 * n)
    a = np.clip(a, 1e-8, 1000.0)
    a = _norm(a, S)

    best = a.copy()
    best_score = _score_arr(a)
    blend = 0.42
    stuck = 0

    for _ in range(1000):
        if time.monotonic() > t_end - 4.0:
            break

        c = _fft_conv(a, a) if n > 1024 else np.convolve(a, a)
        M = float(np.max(c))
        s = float(np.sum(a))
        sc = 2.0 * n * M / (s * s)
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        if not math.isfinite(M) or M <= 1e-14:
            break

        first_col = np.empty(2 * n - 1, dtype=np.float64)
        first_col[:n] = a
        first_col[n:] = 0.0
        first_row = np.zeros(n, dtype=np.float64)
        first_row[0] = a[0]

        try:
            A = toeplitz(first_col, first_row)
        except Exception:
            break

        rhs = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

            res = linprog(
                -np.ones(n, dtype=np.float64),
                A_ub=A,
                b_ub=rhs,
                bounds=bounds,
                method="highs",
                options={"time_limit": remaining, "presolve": True},
            )
        except Exception:
            try:
                res = linprog(
                    -np.ones(n, dtype=np.float64),
                    A_ub=A,
                    b_ub=rhs,
                    bounds=bounds,
                    method="highs",
                )
            except Exception:
                break

        if res is None or not getattr(res, "success", False) or res.x is None:
            break

        b = np.asarray(res.x, dtype=np.float64)
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.clip(b, 0.0, None)
        bs = float(np.sum(b))
        if not math.isfinite(bs) or bs <= 1e-12:
            break

        g = np.clip(g, 0.0, 1000.0)
        sg = float(np.sum(g))
        if not math.isfinite(sg) or sg <= 1e-12:

        cs = float(np.sum(cand))
        if not math.isfinite(cs) or cs <= 1e-12:
            break
        cand = cand * (s / cs)
        sc2 = _score_arr(cand)
        if math.isfinite(sc2) and sc2 < best_score - 1e-13:
            best_score = sc2
            best = cand.copy()
            stuck = 0

        a = cand
        if stuck > 12:
            blend = max(0.06, 0.72 * blend)
            stuck = 0
        else:
            blend = max(0.06, 0.985 * blend)

    return best, best_score
