                        local_best_score = s_local
                        no_improve = 0
                        lr *= 0.99
                elif no_improve == 100:
                    s_local = fast_score(a)
                    if s_local < local_best_score:
                        local_best = a.copy()
                        local_best_score = s_local
                        no_improve = 0
                elif no_improve == 150:
                    s_local = fast_score(a)
                    if s_local < local_best_score:
                        local_best = a.copy()
                        local_best_score = s_local
                        no_improve = 0
                        lr *= 0.9
            if no_improve == 0:
                lr *= 0.9995

            # occasional score check
            if int(time.monotonic() * 8) % 7 == 0:
                s_chk = fast_score(a)
                if s_chk < local_best_score:
                    local_best = a.copy()
                    local_best_score = s_chk

        return local_best, local_best_score

    # Phase 1: coarse optimization from parent
    t1 = optimize(parent, time_budget=500, lr_init=5.0/n, eps_init=10.0/n, top_k=64)
    if t1[1] < best_score:
        best = t1[0].copy()
        best_score = t1[1]

    # Phase 2: finer optimization
    t2 = optimize(best, time_budget=400, lr_init=2.0/n, eps_init=3.0/n, top_k=128)
    if t2[1] < best_score:
        best = t2[0].copy()
        best_score = t2[1]

    # Phase 3: try chirp init
    if time.monotonic() < deadline - 100:
        f = rng.integers(1, n // 4)
        a_chirp = 1.0 + 0.7 * np.cos(2.0 * np.pi * np.arange(n) ** 2 / n * f / max(1, n // 10))
        a_chirp = np.maximum(a_chirp, 0.0)
        a_chirp /= a_chirp.sum()
        t3 = optimize(a_chirp, time_budget=min(200, deadline - time.monotonic() - 20), lr_init=3.0/n, eps_init=5.0/n, top_k=64)
        if t3[1] < best_score:
            best = t3[0].copy()
            best_score = t3[1]

    # Phase 4: try random init
    if time.monotonic() < deadline - 50:
        a_rand = rng.uniform(0.5, 1.5, n)
        a_rand /= a_rand.sum()
        t4 = optimize(a_rand, time_budget=min(100, deadline - time.monotonic() - 15), lr_init=3.0/n, eps_init=5.0/n, top_k=48)
        if t4[1] < best_score:
            best = t4[0].copy()
            best_score = t4[1]

    # Phase 5: final fine-tuning
    if time.monotonic() < deadline - 20:
        t5 = optimize(best, time_budget=min(100, deadline - time.monotonic() - 12), lr_init=0.5/n, eps_init=1.0/n, top_k=256)
        if t5[1] < best_score:
            best = t5[0].copy()
            best_score = t5[1]

    # Final validation with exact evaluator
    final_list = [float(x) for x in best]
    final_score = float(evaluate_sequence(final_list))
    if not np.isfinite(final_score) or final_score > best_score:
        final_score = best_score
        final_list = [float(x) for x in parent]
        final_score = float(evaluate_sequence(final_list))

    return final_list
