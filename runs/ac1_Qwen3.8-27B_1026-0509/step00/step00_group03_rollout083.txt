    def mirror_step(p, grad, eta):
        gmax=float(np.max(grad))
        if not np.isfinite(gmax): return p.copy()

        p = np.clip(p, 0.0, 1000.0)
        tot = float(p.sum())
        if not np.isfinite(tot) or tot <= 1e-12:
            return np.ones(p.size, dtype=np.float64) / max(1, p.size)
        p /= tot
        return p

    def score(p):
        p = np.asarray(p, dtype=np.float64)
        tot = float(p.sum())
        if p.size == 0 or not np.isfinite(tot) or tot <= 1e-12:
            return float("inf")
        c = np.convolve(p, p)
        if c.size == 0:
            return float("inf")
        return float(2.0 * p.size * float(c.max()) / (tot * tot))

    base = norm(parent)
    state = {"seq": base.copy(), "score": score(base)}

    def set_best(p, sc):
        if np.isfinite(sc) and sc < state["score"]:
            state["score"] = float(sc)
            state["seq"] = norm(p)

    def add_start(p):
        p = norm(p)
        sc = score(p)
        if np.isfinite(sc):
            starts.append((sc, p))

    starts = [(state["score"], base.copy())]

    if time.monotonic() < deadline - 20.0:
        for th in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
            f = np.exp(th * (x - 0.5))
            g = np.exp(-th * (x - 0.5))
            for h in (f, g):
                add_start(h)
                if n > 0:
                    add_start(base * h)

    if time.monotonic() < deadline - 80.0:
        s2 = s * s
        s3 = s2 * s
        s5 = s3 * s2
        for a in (-0.65, -0.35, 0.0, 0.35, 0.65):
            for b in (-0.25, 0.0, 0.25):
                for c in (-0.15, 0.0, 0.15):
                    v = 1.0 + a * s + b * s3 + c * s5
                    add_start(v)
                    if n > 0:
                        add_start(base * v)

    rng = np.random.default_rng(seed)

    def smooth_random(scale=0.25):
        v = rng.normal(size=n)
        if n > 20:
            F = np.fft.rfft(v)
            cut = max(2, n // 18)
            F[cut:] = 0.0
            v = np.fft.irfft(F, n)
        v -= float(v.mean())
        return np.exp(scale * v)

    if time.monotonic() < deadline - 40.0:
        for _ in range(4):
            r = smooth_random()
            add_start(r)
            if n > 0:
                add_start(base * r)

    starts.sort(key=lambda z: z[0])
    if starts:
        set_best(starts[0][1], starts[0][0])

    if _HAVE_SCIPY and time.monotonic() < deadline - 120.0:
        s2 = s * s
        s3 = s2 * s
        s4 = s2 * s2

        def poly_eval(theta):
            a, b, c, d = (float(v) for v in theta)
            v = 1.0 + a * s + b * s2 + c * s3 + d * s4
            v = np.maximum(v, 0.0)
            tot = float(v.sum())
            if tot <= 1e-9:
                return 1e6
            v /= tot
            sc = score(v)
            return 1e6 if not np.isfinite(sc) else sc

        inits = (
            (0.35, -0.05, 0.08, 0.0),
            (0.55, -0.12, 0.02, 0.0),
            (-0.55, -0.12, -0.02, 0.0),
            (0.25, 0.15, 0.12, 0.05),
        )
        for z0 in inits:
            if time.monotonic() >= deadline - 80.0:
                break
            try:
                res = minimize(poly_eval, np.asarray(z0, float),
                               method="Nelder-Mead",
                               options={"maxiter": 90, "xatol": 1e-5, "fatol": 1e-8})
                if res.fun < 1e5:
                    a, b, c, d = map(float, res.x)
                    v = 1.0 + a * s + b * s2 + c * s3 + d * s4
                    add_start(v)
            except Exception:
                pass

    starts.sort(key=lambda z: z[0])
    if starts:
        set_best(starts[0][1], starts[0][0])

    def mirror_step(p, grad, eta):
        gmax = float(np.max(grad))
        if not np.isfinite(gmax):
            return p.copy()
        logz = np.log(np.maximum(p, FLOOR)) + eta * (gmax - grad)
        zmax = float(np.max(logz))
        if not np.isfinite(zmax):
            return p.copy()
        z = np.exp(logz - zmax)
        zsum = float(z.sum())
        if not np.isfinite(zsum) or zsum <= FLOOR:
            return p.copy()
        z /= zsum
        return z

    def perturb(p, sigma=0.22):
        v = smooth_random(sigma)
        q = np.maximum(p, FLOOR) * v
        return norm(q)

    def consider_known(p, sc):
        if np.isfinite(sc) and sc < state["score"]:
            state["score"] = float(sc)
            state["seq"] = norm(p)

    def run_mirror(p0, limit):
        if limit <= time.monotonic():
            return
        p = norm(p0)
        sc = score(p)
        if not np.isfinite(sc):
            p = state["seq"].copy()
            sc = state["score"]
        if np.isfinite(sc):
            consider_known(p, sc)
            cur = p.copy()
            cur_sc = sc
        else:
            return

        eta = max(50.0, 0.5 * float(n))
        stuck = 0
        last_mult = 1.0

        while time.monotonic() < limit:
            c = np.convolve(cur, cur)
            if c.size == 0:
                break
            maxc = float(c.max())
            if maxc <= 0.0:
                break

            beta = min(25000.0, max(2000.0, 8.0 / maxc))
            w = np.exp(beta * (c - maxc))
            ws = float(w.sum())
            if not np.isfinite(ws) or ws <= FLOOR:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w /= ws

            cross = np.convolve(w, cur[::-1])
            grad = 2.0 * cross[n - 1:2 * n - 1]
            if grad.size != n or not np.all(np.isfinite(grad)):
                stuck += 1
                if stuck > 100:
                    break
                continue

            if float(np.max(grad) - np.min(grad)) <= 1e-14:
                q = perturb(cur, 0.25)
                qc = score(q)
                if np.isfinite(qc) and qc < cur_sc:
                    cur, cur_sc = q, qc
                    consider_known(cur, cur_sc)
                    stuck = 0
                    continue
                stuck += 1
                if stuck > 100:
                    break
                continue

            improved = False
            for mult in (1.0, 0.5, 0.25, 0.125, 0.0625):
                q = mirror_step(cur, grad, eta * mult)
                qc = score(q)
                if np.isfinite(qc) and qc < cur_sc:
                    cur, cur_sc = q, qc
                    consider_known(cur, cur_sc)
                    improved = True
                    last_mult = mult
                    break

            if improved:
                if last_mult >= 1.0:
                    eta = min(50.0 * float(n), eta * 1.05)
                elif last_mult >= 0.5:
                    eta = min(50.0 * float(n), eta * 1.01)
                else:
                    eta = max(0.05 * float(n), eta * 0.8)
                stuck = 0
            else:
                eta = max(0.05 * float(n), eta * 0.7)
                stuck += 1
                if stuck == 12:
                    escaped = False
                    for th in (0.35, -0.35, 0.7, -0.7):
                        f = np.exp(th * (x - 0.5))
                        q = norm(0.6 * cur + 0.4 * cur * f)
                        qc = score(q)
                        if np.isfinite(qc) and qc < cur_sc:
                            cur, cur_sc = q, qc
                            consider_known(cur, cur_sc)
                            escaped = True
                            break
                    if not escaped:
                        q = perturb(cur, 0.30)
                        qc = score(q)
                        if np.isfinite(qc) and qc < cur_sc:
                            cur, cur_sc = q, qc
                            consider_known(cur, cur_sc)
                    stuck = 0
                if stuck > 350:
                    break

    def lp_polish(p0, limit):
        if not _HAVE_SCIPY or limit <= time.monotonic():
            return
        p = norm(p0)
        sc = score(p)
        if not np.isfinite(sc):
            p = state["seq"].copy()
            sc = state["score"]
        if not np.isfinite(sc):
            return
        consider_known(p, sc)
        cur = p.copy()
        cur_sc = sc

        K = 80
        S_cap = 1.4
        stalls = 0
        ones = np.ones(n, dtype=np.float64)

        while time.monotonic() < limit and stalls < 6:
            c = np.convolve(cur, cur)
            if c.size == 0:
                break
            M = float(c.max())
            if M <= 0.0:
                break

            K = max(40, min(300, K))
            if K < c.size:
                top = np.argpartition(c, K)[:K]
            else:
                top = np.arange(c.size)

            rows, cols, data, bubs = [], [], [], []
            for k in top:
                ki = int(k)
                start = max(0, ki - n + 1)
                end = min(n - 1, ki)
                if start > end:
                    continue
                vals = cur[ki - end:ki - start + 1][::-1]
                nz = vals > FLOOR
                if not np.any(nz):
                    continue
                idx = np.nonzero(nz)[0]
                row = len(bubs)
                bubs.append(M)
                rows.extend([row] * idx.size)
                cols.extend(start + idx)
                data.extend(vals[nz])

            if not bubs:
                break

            row = len(bubs)
            bubs.append(S_cap)
            rows.extend([row] * n)
            cols.extend(range(n))
            data.extend(ones)

            try:
                A = sparse.csr_matrix((data, (rows, cols)), shape=(len(bubs), n))
                rem = max(1.0, limit - time.monotonic())
                res = linprog(
                    -ones,
                    A_ub=A,
                    b_ub=np.asarray(bubs, dtype=np.float64),
                    bounds=[(0.0, None)] * n,
                    method="highs",
                    options={"presolve": True, "time_limit": min(25.0, rem)},
                )
            except Exception:
                break

            if res is None or res.status != 0 or res.x is None:
                break

            b = np.asarray(res.x, dtype=np.float64)
            sb = float(b.sum())
            if not np.isfinite(sb) or sb <= 1e-12:
                stalls += 1
                K = min(300, K + 50)
                S_cap = min(15.0, S_cap * 1.6)
                continue

            g = b / sb
            best_t = 0.0
            best_cand = cur
            best_sc = cur_sc
            for t in (0.05, 0.1, 0.2, 0.35, 0.55, 0.8, 1.0):
                cand = norm((1.0 - t) * cur + t * g)
                cs = score(cand)
                if np.isfinite(cs) and cs < best_sc:
                    best_sc, best_t, best_cand = cs, t, cand

            if best_t > 0.0:
                cur, cur_sc = best_cand, best_sc
                consider_known(cur, cur_sc)
                stalls = 0
                S_cap = min(15.0, S_cap * 1.7 + 0.2)
                K = min(300, K + 40)
            else:
                stalls += 1
                K = min(300, K + 70)
                S_cap = min(15.0, S_cap * 1.5 + 0.2)

    if budget > 900.0:
        slices = (420.0, 170.0, 170.0, 130.0, 110.0, 90.0)
    elif budget > 450.0:
        slices = (150.0, 90.0, 80.0, 70.0)
    elif budget > 180.0:
        slices = (70.0, 45.0, 30.0)
    else:
        slices = (20.0)

    for i, (sc, pstart) in enumerate(starts[:len(slices)]):
        if time.monotonic() >= deadline - 5.0:
            break
        limit = min(deadline - 3.0, time.monotonic() + slices[i])
        run_mirror(pstart, limit)

    if _HAVE_SCIPY and budget > 700.0 and time.monotonic() < deadline - 160.0:
        lp_polish(state["seq"].copy(), min(deadline - 60.0, time.monotonic() + 240.0))

    if time.monotonic() < deadline - 1.0:
        run_mirror(state["seq"].copy(), deadline - 1.0)

    def finalize(a):
        a = np.asarray(a, dtype=np.float64).copy()
        if a.size == 0:
            a = np.ones(1, dtype=np.float64)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        tot = float(a.sum())
        if not np.isfinite(tot) or tot <= 1e-12:
            a = np.ones_like(a)
        a /= a.sum()
        return [float(v) for v in a]

    candidates = [finalize(state["seq"]), finalize(parent), [1.0 / max(1, n)] * n]
    best_list = None
    best_final = float("inf")
    for lst in candidates:
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < best_final:
            best_final = float(sc)
            best_list = lst

    if best_list is None:
        best_list = [1.0]
    return best_list
