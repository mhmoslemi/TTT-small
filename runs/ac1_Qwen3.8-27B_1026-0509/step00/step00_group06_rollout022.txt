import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    deadline = time.monotonic() + budget_s - 20
    
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    n = len(parent)
    target_sum = np.sqrt(2.0 * n)
    
    def normalize(x):
        s = x.sum()
        if s < 1e-10:
            return np.ones(n) * target_sum / n
        return np.clip(x * (target_sum / s), 0.0, 1000.0)
    
    a = normalize(parent)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Parameters
    T = 5.0
    T_min = 0.05
    T_decay = 0.999
    base_lr = 1.0
    eval_every = 3
    
    iteration = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        max_c = c.max()
        
        # Softmax weights
        T_eff = max(T, 1e-3)
        c_shift = c - max_c
        w = np.exp(c_shift / T_eff)
        w_sum = w.sum()
        w = w / w_sum
        
        # Gradient
        S1 = np.convolve(a[::-1], w)[n-1:2*n-1]
        cs = np.concatenate([[0.0], np.cumsum(w)])
        W = cs[n:2*n] - cs[0:n]
        grad = (S1 + W * a) / T_eff
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            step = min(base_lr / gnorm, 0.5)
        else:
            step = base_lr
        
        # Multiplicative update
        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-10)
        a_new = normalize(a_new)
        
        iteration += 1
        
        # Evaluate every few iterations
        if iteration % eval_every == 0:
            score = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a_new.copy()
                best_score = score
                no_improve = 0
            else:
                no_improve += 1
        
        a = a_new
        T = max(T * T_decay, T_min)
        
        # Random restart if stuck
        if no_improve > 100:
            a = best * np.exp(rng.normal(0, 0.5, n))

    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
