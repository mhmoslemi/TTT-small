        inits = [np.ones(n)]

    best_a = inits[0].copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    for a0 in inits:
        if time.monotonic() > deadline:
            break
        a = iter_refine(a0.copy())
        a = block_refine(a)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a = iter_refine(best_a.copy(), max_it=200000)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

    if not np.isfinite(best_score):
        best_a = np.ones(n)
    return [float(v) for v in best_a]
