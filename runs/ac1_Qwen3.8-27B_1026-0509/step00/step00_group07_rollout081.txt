def _conv_fft_len(n):
    return 1 << max(1, int(2 * n - 1)).bit_length()


def _work_fft_len(n):
    return 1 << max(1, int(3 * n - 2)).bit_length()


def _fft_conv(a, L=None):
    n = a.size
    if L is None:
        L = _conv_fft_len(n)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    return c[:2 * n - 1]


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(np.sum(a))
    n = a.size
    if n == 0 or s <= 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m) or m <= 0:
        return np.inf
    return float(2.0 * n * m / (s * s))


def _from_log(loga):
    loga = np.asarray(loga, dtype=np.float64).ravel()
    if loga.size == 0:
        return np.array([1.0])
    z = loga - float(np.max(loga))
    z = np.where(np.isfinite(z), z, -30.0)
    a = np.exp(z)
    a = np.maximum(a, 1e-12)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(loga.size, dtype=np.float64) / loga.size
    return a / s


def _make_seeds(n, rng, base=None):
    out = []
    x = np.linspace(0.0, 1.0, n)
    xs = np.maximum(x, 1e-6)
    xe = np.maximum(1.0 - x, 1e-6)

    ps = np.array([-3.0, -1.75, -0.75, 0.0, 0.75, 1.75, 3.0])
    for p in ps:
        for q in ps:
            out.append(_from_log(p * np.log(xs) + q * np.log(xe)))

    us = np.array([-2.0, -1.0, 0.0, 1.0, 2.0])
    vs = np.array([-2.0, -1.0, 0.0, 1.0, 2.0])
    for u in us:
        for v in vs:
            out.append(_from_log(u * x + v * x * x))

    for _ in range(6):
        r = rng.normal(size=n)
        r = np.cumsum(r)
        r -= float(np.mean(r))

                A_ub=C,
                b_ub=np.full(m, b, dtype=np.float64),
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(1.0, end - time.monotonic())},
            )
        except Exception:
            try:
                res = linprog(
                    cobj,
                    A_ub=C,
                    b_ub=np.full(m, b, dtype=np.float64),
                    bounds=bounds,
                    method="highs",
                )
            except Exception:
                break

        if res is None or not getattr(res, "success", False):
            break
        g = np.asarray(res.x, dtype=np.float64).ravel()
        if g.size != n:
            break
        g = np.maximum(np.where(np.isfinite(g), g, 0.0), 0.0)
        gs = float(np.sum(g))
        if gs <= 1e-12:
            break
        g = g / gs

        improved = False
        for t in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125):
            h = (1.0 - t) * cur + t * g
            h = np.maximum(h, 1e-12)
            hs = float(np.sum(h))
            if hs <= 1e-12:
                continue
            h = h / hs
            sc = _score_fast(h)
            if np.isfinite(sc) and sc < best_sc - 1e-10:
                best_sc = sc
                best = h.copy()
                cur = h
                improved = True
                register(h, sc, force=False)
                break

        if not improved:
            if float(np.linalg.norm(g - cur)) < 1e-3:
                break

    register(best, best_sc, force=True)
    return best, best_sc


def _block_search(a, seconds, rng, register):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 1e-12)
    a = a / float(np.sum(a))
    best = a.copy()
    best_sc = _score_fast(best)
    end = time.monotonic() + max(0.0, float(seconds))
    n = a.size
    stall = 0

    while time.monotonic() < end:
        w = int(rng.integers(max(8, n // 250), max(10, n // 6)))
        start = int(rng.integers(0, max(1, n - w + 1)))
        cand = best.copy()
        scale = 0.18 if stall < 5000 else 0.38
        cand[start:start + w] *= np.exp(rng.normal(0.0, scale))
        if rng.random() < 0.08:
            cand *= 0.7 + 0.6 * np.linspace(0.0, 1.0, n)
        cand = np.maximum(cand, 1e-12)
        cs = float(np.sum(cand))
        if cs <= 1e-12:
            continue
        cand /= cs
        sc = _score_fast(cand)
        if np.isfinite(sc) and sc < best_sc - 1e-10:
            best_sc = sc
            best = cand.copy()
            stall = 0
            register(best, best_sc, force=False)
        else:
            stall += 1
            if stall > 12000:
                stall = 0

    register(best, best_sc, force=True)
    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start_time = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = start_time + max(0.0, budget - 5.0)
    rng = np.random.default_rng(seed)

    best_list = None
    best_exact = np.inf

    gvars = globals()
    if "height_sequence_1" in gvars:
        try:
            arr = np.asarray(gvars["height_sequence_1"], dtype=np.float64).ravel()
            arr = np.clip(np.where(np.isfinite(arr), arr, 0.0), 0.0, 1000.0)
            if arr.size > 0:
                lst = [float(v) for v in arr.tolist()]
                sc = evaluate_sequence(lst)
                if np.isfinite(sc):
                    best_list = lst
                    best_exact = float(sc)
        except Exception:
            pass

    if best_list is None:
        a = np.ones(64, dtype=np.float64)
        a /= float(np.sum(a))
        lst = [float(v) for v in a.tolist()]
        sc = evaluate_sequence(lst)
        best_list = lst
        best_exact = float(sc) if np.isfinite(sc) else 2.0

    best_fast_a = None
    best_fast_score = np.inf

    def register(a, scf=0.0, force=False):
        nonlocal best_list, best_exact, best_fast_a, best_fast_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(np.sum(arr))
        if s <= 1e-12:
            return
        arr = arr / s

        if not np.isfinite(scf):
            scf = _score_fast(arr)
        if np.isfinite(scf) and scf < best_fast_score:
            best_fast_score = float(scf)
            best_fast_a = arr.copy()

        if force or (np.isfinite(scf) and scf < best_exact - 1e-7):
            try:
                ar_clip = np.clip(arr, 0.0, 1000.0)
                lst = [float(v) for v in ar_clip.tolist()]
                sc = evaluate_sequence(lst)
                if np.isfinite(sc) and float(sc) < best_exact:
                    best_exact = float(sc)
                    best_list = lst
            except Exception:
                pass

    if time.monotonic() < deadline - 30.0:
        if budget >= 700.0:
            n = 1024
        elif budget >= 250.0:
            n = 512
        else:
            n = 256

        reserve = 160.0 if n >= 1024 else 90.0
        seeds = _make_seeds(n, rng)
        scored = []
        for a in seeds:
            sc = _score_fast(a)
            if np.isfinite(sc):
                scored.append((float(sc), a.copy()))
        scored.sort(key=lambda z: z[0])

        if not scored:
            a0 = np.ones(n, dtype=np.float64) / n
            register(a0, 2.0, force=True)
        else:
            register(scored[0][1], scored[0][0], force=True)

            max_starts = 8 if n >= 1024 else 12
            max_starts = min(max_starts, len(scored))
            remain = max(0.0, deadline - time.monotonic() - reserve)
            if remain > 30.0 and max_starts > 0:
                seed_time = max(8.0, remain / max_starts)
                for i in range(max_starts):
                    if time.monotonic() > deadline - reserve:
                        break
                    a_seed = scored[i][1].copy()
                    if time.monotonic() > deadline - max(10.0, seed_time):
                        break
                    a_opt, sc_opt = _optimize_seed(a_seed, seed_time, rng)
                    register(a_opt, sc_opt, force=True)

        if best_fast_a is not None and time.monotonic() < deadline - 70.0:
            cur = best_fast_a.copy()
            lp_time = min(120.0, max(5.0, deadline - time.monotonic() - 120.0))
            if cur.size <= 1024 and lp_time > 10.0:
                cur, _ = _lp_polish(cur, lp_time, rng, register)

            if time.monotonic() < deadline - 60.0:
                bl_time = min(70.0, max(5.0, deadline - time.monotonic() - 45.0))
                if bl_time > 10.0:
                    cur, _ = _block_search(cur, bl_time, rng, register)

            if time.monotonic() < deadline - 25.0:
                register(cur, _score_fast(cur), force=True)

        if best_exact > 1.503 and best_fast_a is not None and time.monotonic() < deadline - 260.0:
            n2 = 2048 if best_fast_a.size < 1024 else 4096
            if best_fast_a.size < n2:
                x1 = np.linspace(0.0, 1.0, best_fast_a.size)
                x2 = np.linspace(0.0, 1.0, n2)
                a2 = np.interp(x2, x1, best_fast_a)
                a2 += 0.012 * rng.normal(size=n2)
                a2 = np.maximum(a2, 1e-12)
                extra = _make_seeds(n2, rng, base=a2)
                ex_scored = []
                for e in extra:
                    sc = _score_fast(e)
                    if np.isfinite(sc):
                        ex_scored.append((float(sc), e))
                ex_scored.sort(key=lambda z: z[0])
                if ex_scored:
                    a2 = 0.55 * a2 / float(np.sum(a2)) + 0.45 * ex_scored[0][1]
                a2 = np.maximum(a2, 1e-12)
                a2 = a2 / float(np.sum(a2))
                secs2 = max(30.0, min(220.0, deadline - time.monotonic() - 90.0))
                if secs2 > 30.0:
                    a3, sc3 = _optimize_seed(a2, secs2, rng)
                    register(a3, sc3, force=True)
                    if time.monotonic() < deadline - 70.0:
                        a4, _ = _block_search(a3, min(45.0, deadline - time.monotonic() - 45.0), rng, register)
                        register(a4, _score_fast(a4), force=True)

    if best_fast_a is not None:
        register(best_fast_a, _score_fast(best_fast_a), force=True)

    try:
        final = np.asarray(best_list, dtype=np.float64).ravel()
        final = np.where(np.isfinite(final), final, 0.0)
        final = np.clip(final, 0.0, 1000.0)
        lst = [float(v) for v in final.tolist()]
        if not lst:
            lst = [1.0]
        if not np.isfinite(evaluate_sequence(lst)):
            lst = [1.0]
    except Exception:
        lst = [1.0]
    return lst
