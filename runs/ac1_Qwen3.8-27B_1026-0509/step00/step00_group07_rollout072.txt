                          method='highs', options={'time_limit': time_limit, 'presolve': True})
        except Exception:
            return None
        if res.status == 0 and res.fun < -1e-10:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                return g
        return None

    # Try multiple sizes
    for n in [80, 120, 200]:
        if time.monotonic() > deadline:
            break

        inits = [
            np.ones(n) / n,
            np.linspace(1.0, 0.2, n),
            np.sin(np.linspace(0, np.pi, n)) ** 2,
            np.exp(-1.5 * np.linspace(0, 1, n)),
            np.exp(-1.5 * np.linspace(0, 1, n)[::-1]),
            np.concatenate([np.linspace(0.1, 1.0, n // 2), np.linspace(1.0, 0.1, n - n // 2)]),
        ]
        for _ in range(4):
            inits.append(rng.dirichlet(np.ones(n) * 0.5))

        for f0 in inits:
            if time.monotonic() > deadline:
                break
            f = f0.copy()
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                continue
            f = f / s
            update_best(f)

            # LP-based iterations
            max_lp_iters = 80 if n <= 120 else 50
            for it in range(max_lp_iters):
                if time.monotonic() > deadline:
                    break
                g = lp_iteration(f, n, t=0.1, time_limit=4 if n <= 150 else 8)
                if g is not None:
                    t = max(0.05, 0.3 * (1.0 - it / max_lp_iters))
                    f_new = (1 - t) * f + t * g
                    f_new = np.maximum(f_new, 0)
                    sn = f_new.sum()
                    if sn > 1e-10:
                        f_new = f_new / sn
                        old = best_score
                        update_best(f_new)
                        if best_score < old:
                            f = f_new
                        elif rng.random() < 0.05:
                            f = f_new
                if it % 25 == 24:
                    f = f + 0.03 * rng.standard_normal(n)
                    f = np.maximum(f, 0)
                    f = f / f.sum()

            # Local search
            n_ls = 300 if n <= 120 else 150
            for it in range(n_ls):
                if time.monotonic() > deadline:
                    break
                w = max(1, n // 40)
                st = rng.integers(0, n - w)
                f_new = f.copy()
                f_new[st:st + w] *= np.exp(rng.normal(0, 0.25))
                if rng.random() < 0.3:
                    f_new = 0.7 * f_new + 0.3 * f_new[::-1]
                f_new = np.maximum(f_new, 0)
                sn = f_new.sum()
                if sn > 1e-10:
                    f_new = f_new / sn
                    old = best_score
                    update_best(f_new)
                    if best_score < old:
                        f = f_new

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100, dtype=np.float64)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs):
        result = [1.0] * 100

    return result
