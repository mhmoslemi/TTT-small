def _block_lp(a, nb, L):
    try:
        from scipy.optimize import linprog
    except Exception:
        return None
    n = a.size
    nb = int(min(max(nb, 8), max(8, n)))
    edges = np.linspace(0, n, nb+1, dtype=int)
    # ensure increasing
    for j in range(nb):
       if edges[j+1] <= edges[j]: edges[j+1] = edges[j] + 1
    if edges[-1] != n: edges[-1] = n
    lens = np.diff(edges)
    # if any zero? set
    if np.any(lens <= 0): return None
    x0 = np.empty(nb)
    for j in range(nb): x0[j]=a[edges[j]:edges[j+1]].mean()
    prefix = np.empty(n+1); prefix[0]=0; prefix[1:]=np.cumsum(a)
    ks = np.arange(2*n-1)
    S = np.empty((2*n-1, nb), dtype=np.float64)
    for j in range(nb):
       b0=edges[j]; b1=edges[j+1]-1
       low = ks - b1
       low = np.where(low < 0, 0, low)
       high = ks - b0
       high = np.where(high > n-1, n-1, high)
       valid = low <= high
       col = np.zeros(2*n-1, dtype=np.float64)
       col[valid] = prefix[high[valid]+1] - prefix[low[valid]]
       # scaled by n
       S[:,j] = n * col
    dmax, _ = _score_info(a, L, None)
    M1 = float(np.max(S @ x0))
    M = max(dmax, M1) * 1.001 + 1e-18
    cobj = -lens.astype(float)
    try:
       res = linprog(cobj, A_ub=S, b_ub=np.full(2*n-1, M), bounds=(0.0, 1000.0), method='highs')
       if res is None or not getattr(res, 'success', False): return None
       x = np.asarray(res.x, dtype=np.float64)
       if x.size != nb or not np.all(np.isfinite(x)): return None
       x = np.clip(x, 0.0, 1000.0)
       g = np.repeat(x, lens)
       g = np.maximum(g, 0.0)
       s = float(g.sum())

import numpy as np
import time
import math


def _next_power_of_2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _normalize(a, floor=1e-12):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, floor, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-6:
        return None
    return a / s


def _asym(a):
    return float(np.mean((a - a[::-1]) ** 2))


def _score_info(a, L, q=None):
    n = a.size
    M = 2 * n - 1
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    d = n * c[:M]
    dmax = float(np.max(d))
    if q is None:
        return dmax, fa
    dpos = np.maximum(d, 1e-300)
    logd = np.log(dpos)
    lm = float(np.max(logd))
    z = np.clip(q * (logd - lm), -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s > 0.0 and math.isfinite(s):
        w = e / s
        logS = q * lm + math.log(s)
    else:
        w = np.full(M, 1.0 / M)
        logS = math.log(M)
    return dmax, w, logS, fa


def _dmax_only(a, L):
    return _score_info(a, L, None)[0]


def _corr_w(w, fa, L, n):
    fw = np.fft.rfft(w[::-1], L)
    conv = np.fft.irfft(fa * fw, L)
    return conv[n - 1:2 * n - 1][::-1]


def _block_lp(a, nb, L):
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = a.size
    nb = int(min(max(8, int(nb)), max(1, n)))
    edges = np.linspace(0, n, nb + 1, dtype=int)
    for j in range(nb):
        if edges[j + 1] <= edges[j]:
            edges[j + 1] = edges[j] + 1
    if edges[-1] != n:
        edges[-1] = n
    lens = np.diff(edges)
    if np.any(lens <= 0):
        return None

    x0 = np.empty(nb, dtype=np.float64)
    for j in range(nb):
        x0[j] = a[edges[j]:edges[j + 1]].mean()

    prefix = np.empty(n + 1, dtype=np.float64)
    prefix[0] = 0.0
    prefix[1:] = np.cumsum(a)

    ks = np.arange(2 * n - 1, dtype=np.int64)
    S = np.empty((2 * n - 1, nb), dtype=np.float64)
    for j in range(nb):
        b0 = edges[j]
        b1 = edges[j + 1] - 1
        low = np.maximum(ks - b1, 0)
        high = np.minimum(ks - b0, n - 1)
        valid = low <= high
        col = np.zeros(2 * n - 1, dtype=np.float64)
        col[valid] = prefix[high[valid] + 1] - prefix[low[valid]]
        S[:, j] = n * col

    dmax, _ = _score_info(a, L, None)
    M1 = float(np.max(S @ x0))
    M = max(dmax, M1) * 1.001 + 1e-18

    try:
        res = linprog(
            -lens.astype(float),
            A_ub=S,
            b_ub=np.full(2 * n - 1, M),
            bounds=(0.0, 1000.0),
            method="highs",
        )
        if res is None or not getattr(res, "success", False):
            return None
        x = np.asarray(res.x, dtype=np.float64)
        if x.size != nb or not np.all(np.isfinite(x)):
            return None
        x = np.clip(x, 0.0, 1000.0)
        g = np.repeat(x, lens)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s < 1e-8:
            return None
        return g / s
    except Exception:
        return None


def _local(a0, L, stop_time, best, best_score, rng, q_schedule):
    a = _normalize(a0)
    if a is None:
        return best, best_score
    n = a.size

    it = 0
    bear = 0
    eta = 0.25
    lp_calls = 0

    if time.monotonic() >= stop_time:
        return best, best_score

    m = len(q_schedule)
    frac = np.cumsum(np.ones(m, dtype=float) / m)
    base = time.monotonic()
    total = max(30.0, stop_time - base)

    for idx, q in enumerate(q_schedule):
        phase_stop = min(float(stop_time), base + total * float(frac[idx]))
        while time.monotonic() < phase_stop:
            dmax, w, logS, fa = _score_info(a, L, int(q))

            if it % 300 == 0:
                try:
                    sc = float(evaluate_sequence(a.tolist()))
                    if math.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best = a.copy()
                except Exception:
                    pass

            y = _corr_w(w, fa, L, n)
            g = 2.0 * n * y
            g = g - float(np.dot(a, g))
            scale = float(np.std(g))
            if scale < 1e-14:
                scale = float(np.max(np.abs(g)))
            if scale < 1e-14:
                break
            gn = np.clip(g / scale, -3.0, 3.0)

            accepted = False
            chosen_eta = None
            e = eta
            for _ in range(4):
                arg = np.clip(-e * gn, -20.0, 20.0)
                a_new = a * np.exp(arg)
                a_new = np.maximum(a_new, 1e-12)
                snew = float(a_new.sum())
                if not math.isfinite(snew) or snew < 1e-10:
                    e *= 0.5
                    continue
                a_new /= snew
                dnew, lSnew, _, _ = _score_info(a_new, L, int(q))
                if dnew < dmax * (1.0 - 1e-9) or (
                    abs(dnew - dmax) < 1e-12 and lSnew < logS
                ):
                    a = a_new
                    dmax = dnew
                    logS = lSnew
                    accepted = True
                    chosen_eta = e
                    break
                e *= 0.5

            it += 1
            if accepted:
                if chosen_eta == eta:
                    eta = min(eta * 1.2, 1.0)
                else:
                    eta = max(eta * 0.85, 1e-3)
                bear = 0
            else:
                bear += 1
                eta = max(eta * 0.6, 1e-3)
                if bear > 40 and time.monotonic() + 5.0 < stop_time:
                    bias = 0.05 * (np.linspace(0.0, 1.0, n) - 0.5)
                    if rng.random() < 0.5:
                        bias = -bias
                    a = a * np.exp(rng.normal(0.0, 0.08, n) + bias)
                    a = _normalize(a)
                    if a is None:
                        break
                    bear = 0

                if bear > 150 and lp_calls < 4 and time.monotonic() + 80.0 < stop_time:
                    gdir = _block_lp(a, 128, L)
                    if gdir is not None:
                        for tm in (0.25, 0.5, 0.8):
                            cand = (1.0 - tm) * a + tm * gdir
                            cand = _normalize(cand)
                            if cand is not None:
                                dnew, _, _, _ = _score_info(cand, L, int(q))
                                if dnew < dmax * (1.0 - 1e-9):
                                    a = cand
                                    dmax = dnew
                                    break
                    lp_calls += 1
                    bear = 0

        try:
            sc = float(evaluate_sequence(a.tolist()))
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best = a.copy()
        except Exception:
            pass

    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    budget = float(budget_s) if budget_s is not None else 1500.0
    start = time.monotonic()
    hard_stop = start + max(60.0, budget) - 10.0

    n = None
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1 or p.size == 0:
            raise ValueError
        parent = _normalize(p)
        if parent is not None:
            n = parent.size
    except Exception:
        pass

    if parent is None or n is None:
        n = 2535
        parent = _normalize(np.ones(n, dtype=np.float64))

    L = _next_power_of_2(3 * n - 2)
    t = np.linspace(0.0, 1.0, n)
    pool = []

    def add(a):
        an = _normalize(a)
        if an is None:
            return
        try:
            sc = 2.0 * _dmax_only(an, L)
        except Exception:
            sc = math.inf
        if math.isfinite(sc):
            pool.append((sc, an))

    add(parent)
    add(np.ones(n))

    for g in (0.2, 0.35, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 1.8, 2.2, 2.8):
        add(np.exp(g * t))
        add(1.0 / (1.0 + np.exp(-g * (t - 0.5))))
        add(1.0 / (1.0 + np.exp(-g * (t - 0.35))))
        add(np.exp(g * (t - 0.5)))

    for aa, bb in (
        (0.5, 0.5), (0.8, 0.2), (0.2, 0.8),
        (1.5, 0.5), (0.5, 1.5), (1.0, 0.3), (0.3, 1.0),
    ):
        add((t + 1e-6) ** aa * (1.0 - t + 1e-6) ** bb)

    for _ in range(160):
        g = rng.uniform(0.1, 2.4)
        amp = rng.uniform(0.0, 1.8)
        f = rng.uniform(0.6, 5.0)
        ph = rng.uniform(0.0, 2.0 * math.pi)
        s = np.exp(g * t + amp * np.sin(2.0 * math.pi * f * t + ph))
        add(s)

    for _ in range(80):
        s = 1.0
        for _f in range(3):
            f = rng.uniform(0.5, 4.0)
            amp = rng.uniform(0.1, 0.9)
            ph = rng.uniform(0.0, 2.0 * math.pi)
            if rng.random() < 0.5:
                s += amp * np.sin(2.0 * math.pi * f * t + ph)
            else:
                s += amp * np.cos(2.0 * math.pi * f * t + ph)
        s = np.maximum(s, 0.05)
        p = rng.uniform(0.35, 3.0)
        add(s ** p)

    if not pool:
        out = _normalize(np.ones(n))
        return [float(x) for x in out]

    pool.sort(key=lambda z: z[0])

    top_exact = []
    for sc, a in pool[:14]:
        try:
            esc = float(evaluate_sequence(a.tolist()))
        except Exception:
            esc = math.inf
        top_exact.append((esc if math.isfinite(esc) else sc * 1.000001, a))
    top_exact.sort(key=lambda z: z[0])

    best = top_exact[0][1].copy()
    best_score = top_exact[0][0]
    if not math.isfinite(best_score):
        best_score = 2.0 * _dmax_only(best, L)

    main_stop = hard_stop - 240.0
    if time.monotonic() < main_stop:
        best, best_score = _local(
            best, L, main_stop, best, best_score, rng,
            [8, 16, 32, 64, 128],
        )

    if time.monotonic() < hard_stop - 170.0:
        alt = None
        for _, a in pool:
            if np.linalg.norm(a - best) > 1e-6 and _asym(a) > 1e-14:
                alt = a
                break
        if alt is None and len(pool) > 1:
            alt = pool[1][1]
        if alt is not None:
            stop2 = min(hard_stop - 150.0, time.monotonic() + 90.0)
            if stop2 > time.monotonic() + 20.0:
                best, best_score = _local(
                    alt, L, stop2, best, best_score, rng,
                    [16, 32, 64, 128],
                )

    lp_done = 0
    while time.monotonic() < hard_stop - 80.0 and lp_done < 8:
        if time.monotonic() + 70.0 > hard_stop:
            break
        gdir = _block_lp(best, 128 if lp_done % 2 == 0 else 256, L)
        lp_done += 1
        if gdir is None:
            break
        improved = False
        for tm in (0.2, 0.4, 0.6, 0.8):
            cand = (1.0 - tm) * best + tm * gdir
            cand = _normalize(cand)
            if cand is None:
                continue
            try:
                sc = float(evaluate_sequence(cand.tolist()))
            except Exception:
                sc = math.inf
            if math.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                improved = True
                break
        if not improved and lp_done >= 4:
            break

    if time.monotonic() < hard_stop - 120.0 and best_score > 1.9:
        stop3 = min(hard_stop - 40.0, time.monotonic() + 80.0)
        if stop3 > time.monotonic() + 20.0:
            best, best_score = _local(
                best, L, stop3, best, best_score, rng,
                [32, 64, 128, 256],
            )

    try:
        sc = float(evaluate_sequence(best.tolist()))
        if math.isfinite(sc):
            best_score = sc
        else:
            raise ValueError
    except Exception:
        best = _normalize(np.ones(n))
        best_score = 2.0

    best = np.asarray(best, dtype=np.float64)
    best = np.clip(best, 0.0, 1000.0)
    if best.size == 0 or not np.all(np.isfinite(best)):
        best = np.ones(max(1, n), dtype=np.float64)

    return [float(x) for x in best]
