import numpy as np
import time


def _fft_size(n):
    s = 1
    m = 2 * n - 1
    while s < m:
        s <<= 1
    return s


def _score_fft(a, fsz):
    A = np.fft.rfft(a, fsz)
    b = np.fft.irfft(A * A, fsz)[:2 * len(a) - 1]
    return 2.0 * len(a) * float(b.max())


def _init_starts(n, rng):
    starts = []
    if 'height_sequence_1' in globals():
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if p.ndim == 1 and p.size == n and np.isfinite(p).all() and p.sum() > 0.01:
            starts.append(p / p.sum())
    base = np.ones(n) / n
    starts.append(base)
    r = rng.uniform(0.02, 0.20)
    ramp = np.exp(r * (np.linspace(0, 1, n) - 0.5))
    starts.append(ramp / ramp.sum())
    saw = np.linspace(0.6, 1.4, n)
    starts.append(saw / saw.sum())
    # two-block asymmetric
    blk = np.empty(n)
    blk[:n // 2] = 1.0
    blk[n // 2:] = 0.05
    blk = np.maximum(blk, 1e-6)
    starts.append(blk / blk.sum())
    seen = []
    out = []
    for s in starts:
        s = np.maximum(np.asarray(s, dtype=np.float64), 0.0)
        ss = s.sum()
        if ss < 0.01:
            continue
        s /= ss
        if all(not np.allclose(s, o, atol=1e-3) for o in seen):
            seen.append(s)
            out.append(s)
    if not out:
        out.append(base)
    return out


def _optimize(a, fsz, deadline, rng, seed):
    n = len(a)
    best = a.copy()
    best_proxy = _score_fft(a, fsz)
    eps_best = np.inf
    a = a.copy()
    step = 0.10
    mom = np.zeros(n)
    beta = 0.7
    noimp = 0
    local_rng = np.random.default_rng(seed + 1)
    it = 0
    last_exact = 0.0
    while time.monotonic() < deadline:
        A = np.fft.rfft(a, fsz)
        b = np.fft.irfft(A * A, fsz)[:2 * n - 1]
        k = int(np.argmax(b))
        bmax = b[k]
        # subgradient over near-tied top values
        thr = bmax - 1e-9 * max(1.0, bmax)
        idx = np.flatnonzero(b >= thr)
        idx = idx[-min(len(idx), 4):]
        mirror = np.zeros(n)
        for kstar in idx:
            j_lo = max(0, kstar - n + 1)
            j_hi = min(n - 1, kstar)
            if j_hi >= j_lo:
                mirror[j_lo:j_hi + 1] += a[kstar - j_hi:kstar - j_lo + 1][::-1]
        mirror /= max(1, len(idx))
        grad = mirror - 2.0 * bmax
        mom = beta * mom + grad
        a -= step * mom
        a = np.maximum(a, 0.0)
        ssum = a.sum()
        if ssum < 1e-12:
            a = best.copy()
            mom = np.zeros(n)
            continue
        a /= ssum
        it += 1
        if it % 40 == 0:
            proxy = _score_fft(a, fsz)
            if proxy < best_proxy - 1e-9:
                best_proxy = proxy
                best = a.copy()
                noimp = 0
                step = min(0.25, step * 1.03)
            else:
                noimp += 1
                if noimp > 60:
                    step *= 0.55
                    noimp = 0
                    # exploration kick
                    if local_rng.random() < 0.5:
                        m = max(2, n // 60)
                        st = int(local_rng.integers(0, n - m))
                        a[st:st + m] *= np.exp(local_rng.normal(0, 0.15, m - st + 1 - st))
                    else:
                        a += local_rng.normal(0, 1e-3, n) * np.linspace(0.5, 1.5, n)
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                    mom = np.zeros(n)
        if time.monotonic() - last_exact > 0.8:
            last_exact = time.monotonic()
            sc = evaluate_sequence(best.tolist())
            if np.isfinite(sc) and sc < eps_best:
                eps_best = sc
    sc = evaluate_sequence(best.tolist())
    return best, (sc if np.isfinite(sc) else np.inf)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 6.0
    rng = np.random.default_rng(seed)

    n = 4806
    if 'height_sequence_1' in globals():
        p = np.asarray(height_sequence_1)
        if p.ndim == 1 and p.size > 0:
            n = int(p.size)
    fsz = _fft_size(n)

    starts = _init_starts(n, rng)
    starts.sort(key=lambda s: _score_fft(s, fsz))

    best_seq = starts[0]
    best_score = np.inf
    per = max(20.0, (deadline - t0) / max(1, len(starts)))

    for i, s0 in enumerate(starts):
        s0 = np.maximum(s0, 0.0)
        s0 /= s0.sum()
        phase_deadline = min(deadline, t0 + per * (i + 1))
        # skip if essentially no time left for a meaningful run
        if time.monotonic() > phase_deadline - 5.0 and i > 0:
            break
        seq, sc = _optimize(s0, fsz, phase_deadline, rng, i)
        if np.isfinite(sc) and (not np.isfinite(best_score) or sc < best_score):
            best_score = sc
            best_seq = seq
        if time.monotonic() > deadline - 8.0:
            break

    # final validation with authoritative evaluator
    final = evaluate_sequence([float(x) for x in best_seq])
    if np.isfinite(final):
        return [float(x) for x in best_seq]

    # fallback: the parent / first start
    fb = [float(x) for x in starts[0]]
    if np.isfinite(evaluate_sequence(fb)):
        return fb
    ones = [0.5] * n
    return ones
