            s = float(n)

        c = np.real(ifft(F * F))[:L]
        c[c < 0.0] = 0.0
        return c

            if wk <= 0.0:
                continue
            i_lo = kk - n + 1
            if i_lo < 0:
                i_lo = 0
            i_hi = kk
            if i_hi >= n:
                i_hi = n - 1
            seg = x[kk - i_hi:kk - i_lo + 1][::-1]
            g[i_lo:i_hi + 1] += (2.0 * wk / peak) * seg
        return g

    idx_n = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    quick_stop = min(deadline, t0 + 25.0)

    def quick_add(x):
        if time.monotonic() >= quick_stop:
            return
        x = norm_x(x)
        c = autocorr(x)
        consider(x, c, 2.0 * n * float(c.max()))

    quick_add(np.ones(n))

    ramp_u = idx_n / max(1, n - 1) - 0.5
    if np.max(np.abs(ramp_u)) > 0:
        ramp_u *= (2.0 / np.max(np.abs(ramp_u)))
    for am in (0.2, 0.35, 0.5, 0.65, 0.8, 0.95, 1.1):
        if time.monotonic() >= quick_stop:
            break
        quick_add(1.0 + am * ramp_u)

    for f in (1, 2, 3, 5, 8, 13, 21, 34, 55):
        u = np.sin((2.0 * np.pi * f / n) * (idx_n - center))
        for am in (0.2, 0.35, 0.5, 0.65, 0.8, 0.95, 1.1):
            if time.monotonic() >= quick_stop:
                break
            quick_add(1.0 + am * u)

    for am in (0.4, 0.8, 1.2):
        z = rng.standard_normal(n)
        u = np.tanh(z - z[::-1])
        quick_add(1.0 + am * u)

    if deadline - time.monotonic() > 80 and n > 1:
        B = min(64, n)
        edges = np.linspace(0, n, B + 1, dtype=int)
        lengths = edges[1:] - edges[:-1]
        cm = 0.5 * (B - 1)

        def make_x_from_h(h):
            return norm_x(np.repeat(h, lengths))

        starts = []

        def add_start(h):
            h = np.asarray(h, dtype=np.float64)
            h = np.clip(h, 1e-8, None)
            if h.size == B and np.all(np.isfinite(h)) and float(h.sum()) > 0:
                starts.append(h)

        arB = np.arange(B, dtype=np.float64)
        ramp_b = arB / max(1, B - 1) - 0.5
        ramp_b *= 2.0 / max(np.max(np.abs(ramp_b)), 1.0)
        add_start(np.ones(B))
        for _ in range(8):
            add_start(rng.lognormal(0.0, 0.45, size=B))
        add_start(1.0 + 0.5 * ramp_b)
        add_start(1.0 - 0.5 * ramp_b)
        for f in (1, 2, 3, 5, 8):
            u = np.sin((2.0 * np.pi * f / B) * (arB - cm))
            add_start(1.0 + 0.5 * u)
            add_start(1.0 + 0.75 * u)
        z = rng.standard_normal(B)
        u = np.tanh(z - z[::-1])
        add_start(1.0 + 0.7 * u)

        scored = []
        for h in starts:
            if time.monotonic() >= deadline:
                break
            x = make_x_from_h(h)
            s = 2.0 * n * float(autocorr(x).max())
            scored.append((s, h))
        scored.sort(key=lambda qq: qq[0])
        top = [h for sh, h in scored[:4]]

        if top:
            if cand_score < 1.50:
                coarse_budget = min(deadline - time.monotonic() - 20.0,
                                    max(20.0, 0.07 * budget))
            else:
                coarse_budget = min(deadline - time.monotonic() - 20.0,
                                    max(40.0, 0.20 * budget))
            if coarse_budget > 20.0:
                per = coarse_budget / float(len(top))

                def coarse_local(h_init, stop_time):
                    h = np.asarray(h_init, dtype=np.float64).copy()
                    h = np.clip(h, 1e-12, None)
                    x = make_x_from_h(h)
                    c = autocorr(x)
                    cur = float(c.max())
                    best_h = h.copy()
                    best_x = x.copy()
                    best_c = cur
                    eta = 0.6
                    stall = 0

                    while time.monotonic() < stop_time:
                        if cur < best_c:
                            best_c = cur
                            best_h = h.copy()
                            best_x = x.copy()

                        g = top_gradient(c, x, 16)
                        bm = np.empty(B, dtype=np.float64)
                        for i in range(B):
                            bm[i] = g[edges[i]:edges[i + 1]].mean()
                        ctr = float((bm * lengths).sum() / max(1, lengths.sum()))
                        D = bm - ctr
                        sd = float(np.std(D))
                        if sd < 1e-14:
                            break

                        improved = False
                        a = eta
                        for _ in range(4):
                            if time.monotonic() >= stop_time:
                                break
                            hn = h * np.exp(-a * D / sd)
                            hn = np.clip(hn, 1e-12, None)
                            xn = make_x_from_h(hn)
                            cn = autocorr(xn)
                            sn = float(cn.max())
                            if sn < cur * (1.0 - 1e-8):
                                h = hn
                                x = xn
                                c = cn
                                cur = sn
                                improved = True
                                break
                            a *= 0.5

                        if improved:
                            stall = 0
                            eta = min(1.5, eta * 1.02)
                            if cur < best_c:
                                best_c = cur
                                best_h = h.copy()
                                best_x = x.copy()
                        else:
                            stall += 1
                            eta = max(0.05, eta * 0.85)
                            if stall > 18:
                                stall = 0
                                eta = 0.6
                                h = best_h.copy()
                                h *= np.exp(rng.normal(0.0, 0.45, size=B))
                                if rng.random() < 0.45:
                                    u = (arB - cm) / max(1.0, np.sqrt(B))
                                    h = h * (1.0 + 0.35 * u)
                                h = np.clip(h, 1e-12, None)
                                x = make_x_from_h(h)
                                c = autocorr(x)
                                cur = float(c.max())

                    return best_x, 2.0 * n * best_c

                for h_init in top:
                    stop = min(deadline, time.monotonic() + per)
                    if stop <= time.monotonic():
                        break
                    xb, sc = coarse_local(h_init, stop)
                    consider(xb, None, sc)

    if deadline - time.monotonic() > 20:
        def full_local(x_init, stop_time):
            x = norm_x(x_init).copy()
            c = autocorr(x)
            cur = float(c.max())
            best_x = x.copy()
            best_c = cur
            eta = 0.6
            stall = 0
            strength = 0.25
            K = 24 if n <= 20000 else 12

            while time.monotonic() < stop_time:
                if cur < best_c:
                    best_c = cur
                    best_x = x.copy()

                g = top_gradient(c, x, K)
                ctr = float(np.sum(x * g))
                D = g - ctr
                sd = float(np.std(D))
                if sd < 1e-14:
                    stall += 1
                else:
                    improved = False
                    a = eta
                    for _ in range(4):
                        if time.monotonic() >= stop_time:
                            break
                        xn = x * np.exp(-a * D / sd)
                        xn = np.clip(xn, 1e-12, None)
                        s = float(xn.sum())
                        if s < 1e-15:
                            break
                        xn /= s
                        cn = autocorr(xn)
                        sn = float(cn.max())
                        if sn < cur * (1.0 - 1e-8):
                            x = xn
                            c = cn
                            cur = sn
                            improved = True
                            break
                        a *= 0.5

                    if improved:
                        stall = 0
                        eta = min(1.5, eta * 1.02)
                        if cur < best_c:
                            best_c = cur
                            best_x = x.copy()
                        continue

                    stall += 1
                    eta = max(0.05, eta * 0.85)

                if stall > 22:
                    stall = 0
                    eta = 0.6
                    strength = min(1.35, strength * 1.12)

                    y = best_x.copy()
                    z = rng.standard_normal(n)
                    u = np.tanh(z - z[::-1])
                    y = y + (strength / float(n)) * u

                    if strength > 0.8 or rng.random() < 0.25:
                        amp = 0.7 + rng.random() * 0.9
                        alt = norm_x(1.0 + amp * u)
                        w = rng.uniform(0.15, 0.55)
                        y = (1.0 - w) * y + w * alt

                    count = max(1, n // 300)
                    idx = rng.integers(0, n, size=count)
                    y[idx] *= np.exp(rng.normal(0.0, 0.2 + 0.25 * strength, size=count))

                    y = np.clip(y, 1e-12, None)
                    s = float(y.sum())
                    if s > 1e-15:
                        x = y / s
                    else:
                        x = best_x.copy()
                    c = autocorr(x)
                    cur = float(c.max())

                    if cur > best_c * 1.8:
                        x = best_x.copy()
                        c = autocorr(x)
                        cur = best_c
                        strength = max(0.15, strength * 0.65)

            return best_x, 2.0 * n * best_c

        xb, sc = full_local(cand_best_x.copy(), deadline)
        consider(xb, None, sc)

    ex = float(evaluate_sequence(cand_best_x.tolist()))
    if np.isfinite(ex) and ex < global_best_score:
        global_best_score = ex
        global_best_x = cand_best_x.copy()

    if len(global_best_x) == 0 or not np.isfinite(global_best_x).all():
        global_best_x = parent.copy()
        if len(global_best_x) == 0 or global_best_x.sum() < 0.01:
            return [1.0]

    return [float(v) for v in global_best_x]
