import numpy as np
import time
import math

try:
    from scipy.optimize import linprog as _linprog
except Exception:
    _linprog = None


def _binom_decay(n):
    a = np.empty(n, dtype=np.float64)
    v = 1.0
    for i in range(n):
        a[i] = v
        v = v * (2 * i + 1) / (2 * i + 2)
    return a


def _build_A(b):
    n = b.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[k, lo:hi + 1] = b[k - lo:k - hi - 1:-1]
    return A


def _solve_lp(b):
    n = b.size
    A = _build_A(b)
    c = np.zeros(n + 1)
    c[n] = 1.0
    if _linprog is not None:

        A_eq[0, :n] = 1.0
        b_eq = np.ones(1)
        bounds = [(0.0, None)] * n + [(0.0, None)]
        res = _linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            return np.asarray(res.x[:n], dtype=np.float64)
        return None
    import cvxpy as cp
    a = cp.Variable(n, nonneg=True)

        sel = b[k - lo:k - hi - 1:-1]

    cons.append(cp.sum(a) == 1)
