        # Active set: top few

        k_active = np.where(active)[0]

        sg = np.zeros(n)

            # (f*f)_k = sum_j f_j f_{k-j}, d/d f_j = f_{k-j}

        sg /= len(k_active)

        f_new = np.maximum(f_new, 0)

import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 5.0
    
    try:
        f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).astype(np.float64).copy()
    except:

    if f.size < 2 or not np.isfinite(f).all() or f.sum() < 0.01:
        f = np.ones(512.0)
    
    n = f.size
    f = f / f.sum()
    
    # Perturb
    perturb = rng.normal(0, 0.01, n)
    f = f + perturb
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    
    # FFT setup
    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1
    
    def autoconv(f):
        F = np.fft.rfft(f, fft_n)
        c = np.fft.irfft(F * F, fft_n)
        return c[:2*n-1]
    
    def simplex_project(v):
        """Project v onto the probability simplex."""
        v = np.maximum(v, 0)
        s = v.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return v / s
    
    # Main loop
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        b = autoconv(f)
        M = b.max()
        
        # Active set: top 10% of lags
        thresh = M * 0.9

            je = min(k, n - 1)
            if js <= je:

        f_new = simplex_project(f_new)
        
        # Occasionally add random kick
        if it % 500 == 499:
            f_new += rng.normal(0, 0.005, n)
            f_new = simplex_project(f_new)
        
        f = f_new

            score = float(evaluate_sequence(f.tolist()))

                best_f = f.copy()
