import numpy as np
import time
import cvxpy as cp
from scipy import sparse


def _build_conv_matrix(f):
    n = len(f)
    m = 2 * n - 1
    widths = np.empty(m, dtype=np.int64)
    rows = []
    cols = []
    vals = []
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        widths[k] = je - js + 1
        j = np.arange(js, je + 1)
        rows.append(np.full(je - js + 1, k, dtype=np.int64))
        cols.append(j)
        vals.append(f[k - j])
    C = sparse.coo_matrix(
        (np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
        shape=(m, n)).tocsr()
    return C


def _solve_lp(f, M):
    n = len(f)
    C = _build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [C @ g <= M])
    for solver in (cp.ECOS, cp.GLPK, cp.CBC, cp.PDLP):
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and np.all(np.isfinite(g.value)):
                return np.maximum(np.asarray(g.value, dtype=float), 0.0)
        except Exception:
            continue
    return None
