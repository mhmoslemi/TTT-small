def _proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = v.copy()
    # Scale to sum s after clipping
    v[v < 0] = 0
    total = v.sum()
    if total <= 1e-15:
        return np.full_like(v, s / len(v))
    if total == s:
        return v
    # Find threshold
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find rho such that sum(max(v - tau, 0)) = s
    # rho = number of positive (v_i - tau)
    w = np.arange(1, len(v) + 1)
    tau = (cumsum - s) / w
    # Find the valid tau
    mask = (v_sorted - tau >= 0) & (tau <= v_sorted)

    # Actually, let me use the standard algorithm

def _fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def _proj_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = v.copy()
    v[v < 0] = 0
    if v.sum() <= 1e-15:
        return np.full(n, 1.0/n)
    v_sort = np.sort(v)[::-1]
    cs = np.cumsum(v_sort)
    w = np.arange(1, n+1, dtype=float)
    tau = (cs - 1.0) / w
    # Find kink point
    V = v_sort
    onsite = V * w  # not quite right
    # Standard algorithm: find rho where V[rho-1] >= tau[rho-1] > V[rho]
    # Simpler: binary search

    idx = np.where((v_sort - tau) >= 0)[0]
    if len(idx) == 0:

    rho = idx[-1] + 1
    t = (v_sort[:rho].sum() - 1.0) / rho

    """Compute a*a convolution using precomputed FFT size L."""
