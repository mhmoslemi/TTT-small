import time

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        parent = np.ones(256)
    if parent.ndim != 1 or parent.size < 4:
        parent = np.ones(64)
    n = parent.size
    s = parent.sum()
    if s < 0.01:
        return [1.0] * 64
    parent_n = parent / s

    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1

    def fft_conv(x):
        fx = np.fft.rfft(x, fft_n)
        return np.fft.irfft(fx * fx, fft_n)[:2 * n - 1]

    def fast_score_normalized(z):
        c = fft_conv(z)
        return float(c.max())

    def subgrad(z, active):
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = k - np.arange(lo, hi + 1)
            g[lo:hi + 1] += 2.0 * z[idx]
        return g

    def optimize(z0, iters, lr0):
        z = z0.copy()
        lr = lr0
        bz = z.copy()
        bs = fast_score_normalized(bz)
        for it in range(iters):
            if time.monotonic() > deadline:
                break
            conv = fft_conv(z)
            mc = conv.max()
            sc = float(mc)
            if sc < bs:
                bs = sc
                bz = z.copy()
            thresh = mc * 0.998
            act = np.where(conv >= thresh)[0]
            if len(act) > 20:
                act = np.argsort(conv)[-20:]
            g = subgrad(z, act)
            z = z * np.exp(-lr * g)
            z = np.maximum(z, 0.0)
            sz = z.sum()
            if sz < 1e-15:
                z = bz.copy()
                continue
            z /= sz
            if it % 500 == 499:

                    lr *= 0.6
                elif sc < bs * 0.99:
                    lr = min(lr * 1.3, 2.0)
            if it % 5000 == 4999:
                z *= np.exp(rng.normal(0, 0.03, n))
                z = np.maximum(z, 0.0)
                zz = z.sum()
                if zz > 1e-15:
                    z /= zz
        return bz, bs

    candidates = []
    candidates.append(parent_n.copy())
