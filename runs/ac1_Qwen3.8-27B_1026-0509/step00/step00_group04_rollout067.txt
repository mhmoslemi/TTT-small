import numpy as np
import time


def _fft_pow2(k):
    n = 1
    while n < k:
        n <<= 1
    return n


def _surrogate_and_weights(a, T, N1):
    """Return smoothed objective value and softmax weights over auto-convolution."""
    fa = np.fft.rfft(a, N1)
    q = np.fft.irfft(fa * fa, N1)[:2 * a.size - 1]
    qmax = float(q.max())
    t = T * (q - qmax)
    e = np.exp(t)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(q)
        w[int(np.argmax(q))] = 1.0
    else:
        w = e / s
    val = qmax + float(np.log(s)) / T
    return val, w

    fa = np.fft.rfft(a, N2)
    corr = np.fft.irfft(fw * np.conj(fa), N2)[:a.size]
    return 2.0 * corr
