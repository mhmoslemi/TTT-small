import numpy as np

try:
    from scipy.optimize import minimize, linprog
except Exception:
    minimize = linprog = None


def _normalize(p):
    p = np.asarray(p, dtype=float).ravel()
    if p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return p / s


def _score_unit(p):
    p = _normalize(p)
    if p is None:
        return float("inf")
    c = np.convolve(p, p)
    return float(2.0 * p.size * np.max(c))


def _sqrt_seq(n):
    b = np.ones(n, dtype=float)
    for t in range(1, n):
        b[t] = b[t - 1] * (2.0 * t - 1.0) / (2.0 * t)
    a = b[::-1]
    return _normalize(a)


def _rows(n):
    rows = []
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        js = np.arange(j0, j1 + 1, dtype=int)
        rows.append((js, k - js))
    return rows


def _softmax(z):
    z = np.asarray(z, dtype=float)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 1e-300:
        return np.full(z.shape, 1.0 / z.size, dtype=float)
    return e / s


def _project_simplex(v):
    v = np.asarray(v, dtype=float).ravel()
    if v.size != 1:
        return v
    w = np.sort(v)[::-1]
    cssum = np.cumsum(w)
    idx = np.arange(1, v.size + 1)
    rho = np.nonzero(w * idx > (cssum - 1.0))[0]
    if rho.size == 0:
        rho = np.array([0])
    else:
        rho = rho[-1]
    theta = (cssum[rho] - 1.0) / float(rho + 1)
    return np.maximum(v - theta, 0.0)


def _weighted_grad(rows, p, w):
    n = p.size
    g = np.zeros(n, dtype=float)
    for k, wk in enumerate(w):
        if wk <= 1e-15:
            continue
        js, ip = rows[k]
        g[js] += wk * p[ip]
    return 2.0 * g


def _obj_jac(rows, n, beta, z):
    p = _softmax(z)
    c = np.convolve(p, p)
    m = float(np.max(c))
    ex = np.exp(beta * (c - m))
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        return float("inf"), np.zeros(n, dtype=float)
    val = m + float(np.log(s))
    w = ex / s
    gp = _weighted_grad(rows, p, w)
    grad_z = p * (beta * gp - beta * float(np.dot(p, gp)))
    if not np.isfinite(val) or not np.all(np.isfinite(grad_z)):
        return float("inf"), np.zeros(n, dtype=float)
    return val, grad_z


def _pg_polish(p, rows, n, deadline, consider, maxit=200, seed=1):
    p = _normalize(p)
    if p is None:
        return None
    rng = np.random.default_rng(seed)
    for it in range(maxit):
        if time.monotonic() > deadline:
            break
        c = np.convolve(p, p)
        M = float(np.max(c))
        tol = max(1e-12, 1e-5 * M)
        active = c >= (M - tol)
        if not np.any(active):
            active = c >= (M - 1e-3)
        w = np.zeros(c.size, dtype=float)
        w[active] = 1.0 / float(np.count_nonzero(active))
        gp = _weighted_grad(rows, p, w)
        norm = float(np.linalg.norm(gp))
        if norm < 1e-14:
            break
        scale = 1.0 / max(norm, 1e-14)
        improved = False
        for st in (0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0) * scale:
            q = _project_simplex(p - st * gp)
            q = _normalize(q)
            if q is None:
                continue
            if np.max(np.abs(q - p)) < 1e-15:
                continue
            MQ = float(2.0 * n * np.max(np.convolve(q, q)))
            if MQ < M - 1e-12:
                p = q
                M = MQ
                consider(p)
                improved = True
                break
        if not improved:
            q = p + (0.05 * float(np.mean(p))) * rng.standard_normal(n)
            q = _project_simplex(q)
            q = _normalize(q)
            if q is not None:
                MQ = float(2.0 * n * np.max(np.convolve(q, q)))
                if MQ < M - 1e-12:
                    p = q
                    M = MQ
                    consider(p)
                    improved = True
            if not improved:
                break
    return p


def _lp_polish(p, rows, n, deadline, consider, maxit=60):
    if linprog is None:
        return p
    p = _normalize(p)
    if p is None:
        return None
    p = p + 1e-9
    p = _normalize(p)
    ones = np.ones(n, dtype=float)
    L = 2 * n - 1
    rng = np.random.default_rng(12345)
    stall = 0
    for _ in range(maxit):
        if time.monotonic() > deadline:
            break
        c = np.convolve(p, p)
        M = float(np.max(c))
        A = np.zeros((L, n), dtype=float)
        for k, (js, ip) in enumerate(rows):
            A[k, js] = p[ip]
        try:
            res = linprog(
                -ones,
                A_ub=A,
                b_ub=np.full(L, M, dtype=float),
                bounds=[(0.0, None)] * n,
                method="highs",
            )
        except Exception:
            break
        if not res.success:
            break
        b = np.asarray(res.x, dtype=float)
        b = np.clip(np.where(np.isfinite(b), b, 0.0), 0.0, None)
        Sb = float(np.sum(b))
        if not np.isfinite(Sb) or Sb <= 1.0 + 1e-8:
            if stall > 5:
                p = _normalize(p + 0.02 * float(np.mean(p)) * rng.standard_normal(n))
                stall += 1
            else:
                stall += 1
            if stall > 15:
                break
            continue
        g = b / Sb
        best = (M, p)
        for tau in (0.1, 0.2, 0.5, 1.0):
            q = (1.0 - tau) * p + tau * g
            q = _normalize(q)
            if q is None:
                continue
            MQ = float(2.0 * n * np.max(np.convolve(q, q)))
            if MQ < best[0] - 1e-12:
                best = (MQ, q)
        if best[1] is p:
            stall += 1
            if stall > 5:
                p = _normalize(p + 0.02 * float(np.mean(p)) * rng.standard_normal(n))
                continue
            if stall > 15:
                break
            continue
        p = best[1]
        M = best[0]
        consider(p)
        stall = 0
    return p
