            no_improve = 0
            beta = min(beta * 1.003, 300.0)
            lr *= 0.999
        else:
            no_improve += 1
            lr *= 0.9
            if lr < 0.02:
                x = best_x.copy()
                lr = 0.4
                x *= np.exp(rng.normal(0, 0.08, n))
                x = np.maximum(x, 0.0)
                x /= x.sum()

        if no_improve > 5000:
            no_improve = 0
            lr = 0.4
            # Explore: try multiple random perturbations
            found = False
            for _ in range(30):
                std = 0.1 + 0.3 * rng.random()
                xp = best_x * np.exp(rng.normal(0, std, n))
                xp = np.maximum(xp, 0.0)
                snp = xp.sum()
                if snp < 1e-10:
                    continue
                xp /= snp
                sp = 2.0 * n * conv_sq(xp).max() / (snp * snp)
                if sp < best_score:
                    x = xp.copy()
                    best_x = xp.copy()
                    best_score = sp
                    found = True
                    break
            if not found:
                x = best_x.copy()

    # Final validation with exact evaluator
    final_score = float(evaluate_sequence(best_x.tolist()))
    if not np.isfinite(final_score):
        return [float(v) for v in parent / s_p]
    return [float(v) for v in best_x]
