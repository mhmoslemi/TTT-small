def build_conv_matrix(a, n):
    # a length n, shape (2n-1, n), row k, col j = a[k-j] if 0<=k-j<n
    # vectorized preallocated arrays
    m = 2*n - 1
    nn = n*n
    rows = np.empty(nn, dtype=np.int32)
    cols = np.empty(nn, dtype=np.int32)
    vals = np.empty(nn, dtype=np.float64)
    for d in range(n):
        rows[d*n:(d+1)*n] = d + np.arange(n)
        cols[d*n:(d+1)*n] = np.arange(n)
        vals[d*n:(d+1)*n] = a[d]
    return scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(m,n))

def _conv2(x, y):
    m = x.size + y.size - 1
    N = _next_pow2(m)
    return np.fft.irfft(np.fft.rfft(x, N) * np.fft.rfft(y, N), N)[:m]

def _score_c(a, c):
    s = a.sum()
    if s <= 1e-12:
        return float('inf')
    return float(2.0 * a.size * c.max() / (s * s))

def _score(a):
    return _score_c(a, _conv_same(a))

def _param_a(n, q, modes):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    th = 2.0 * np.pi * x
    b = np.ones(n, dtype=np.float64)
    for m, c in zip(modes, q):
        if c != 0.0:
            b += float(c) * np.sin(m * th)
    b = np.maximum(b, 1e-10)
    b /= b.sum()
    return b

def _upsample(a, nf):
    no = a.size
    if no == nf:
        return a.copy()
    if no == 1:
        return np.full(nf, 1.0/nf)
    b0 = a * no
    x0 = (np.arange(no, dtype=np.float64) + 0.5) / no
    x1 = (np.arange(nf, dtype=np.float64) + 0.5) / nf
    b1 = np.interp(x1, x0, b0, left=b0[0], right=b0[-1])
    b1 = np.maximum(b1, 1e-12)
    b1 /= b1.sum()
    return b1

            if L > K:
                part = np.argpartition(-w, K - 1)[:K]
                order = part[np.argsort(-w[part])]
            else:
                order = np.arange(L)
            wp = w[order]
            wp /= wp.sum()

            G = np.zeros(a_work.size, dtype=np.float64)
            nn = a_work.size
            for kk, wk in zip(order, wp):
                kk = int(kk)
                lo = kk if kk < nn else nn - 1
                hi = min(kk, nn - 1)
                if lo > hi or wk <= 0.0:
                    continue
                start = kk - lo
                stop = kk - hi - 1
                if stop < 0:
                    seg = a_work[start::-1]
                else:
                    seg = a_work[start:stop:-1]
                G[lo:hi + 1] += (2.0 * wk) * seg

            G -= G.mean()
            gmax = G.max() - G.min()
            if gmax <= 1e-14:
                break

            eta = scale / gmax
            loga = np.log(a_work + 1e-300) - eta * G
            loga -= loga.max()
            an = np.exp(loga)
            san = an.sum()
            if not np.isfinite(san) or san <= 0:
                scale *= 0.5
                continue
            an /= san
            sc = _score(an)

            if sc < cur * (1.0 - 1e-10):
                a_work = an
                consider(a_work, sc)
                scale = min(1.2, scale * 1.05)
                no_acc = 0
            else:
                accepted = False
                for lam in (0.5, 0.25, 0.125, 0.0625):
                    am = a_work + lam * (an - a_work)
                    am = np.maximum(am, 0.0)
                    sam = am.sum()
                    if sam <= 0:
                        continue
                    am /= sam
                    s2 = _score(am)
                    if s2 < cur * (1.0 - 1e-10):
                        a_work = am
                        consider(a_work, s2)
                        scale = min(1.2, scale * 1.02)
                        accepted = True
                        no_acc = 0
                        break
                if not accepted:
                    scale *= 0.6
                    no_acc += 1
                    if no_acc > 300:
                        a_work *= np.exp(rng.normal(0.0, 0.05, a_work.size))
                        a_work = np.maximum(a_work, 1e-300)
                        a_work /= a_work.sum()
                        scale = 0.5
                        no_acc = 0

        consider(a_work, _score(a_work))

    cands = []
    if best_a is not None:
        cands.append(best_a)
    for _, arr in sorted(pool, key=lambda z: z[0]):
        cands.append(arr)
        if len(cands) >= 9:
            break

    result = None
    result_score = float("inf")
    for arr in cands:
        if time.monotonic() >= hard_end:
            break
        if arr is None or arr.size == 0:
            continue
        try:
            lst = [float(x) for x in np.clip(arr, 0.0, 1000.0)]
            sc = evaluate_sequence(lst)
            if not np.isfinite(sc):
                sc = _score(np.asarray(lst, dtype=np.float64))
            if np.isfinite(sc) and sc < result_score:
                result_score = float(sc)
                result = lst
        except Exception:
            pass

    if result is None:
        if best_a is not None:
            result = [float(x) for x in best_a]
        else:
            result = [1.0]

    if not result or any(not np.isfinite(v) for v in result):
        result = [1.0]

    return result
