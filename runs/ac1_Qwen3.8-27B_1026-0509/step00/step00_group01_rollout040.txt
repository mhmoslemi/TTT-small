import time
import cvxpy as cp

def solve_lp(a, M, S):
    """Solve the LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0, sum(b) <= S"""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix using Toeplitz structure
    # A[k,j] = a[k-j] if 0 <= k-j < n
    # convolve(a, b)[k] = sum_j a[k-j] * b[j] = (A @ b)[k]
    
    # Instead of building the full matrix, use cvxpy's convolve... 
    # Actually, cvxpy doesn't have a built-in convolve constraint.
    # Let me build the matrix explicitly.
    
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    
    constraints = [A @ b <= M, cp.sum(b) <= S]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    
    if b.value is None:
        return None
    return np.clip(b.value, 0, 1000)
