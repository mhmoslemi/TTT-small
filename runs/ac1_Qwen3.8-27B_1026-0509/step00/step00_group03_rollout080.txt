def propose_candidate(seed=42, budget_s=1500, **kwargs):

    def conv_full(a):
        n=a.size
        if n < 128:
            return np.convolve(a,a)
        return fftconvolve(a,a)

    def direct_score(a):
        a=np.asarray(a,float)
        if a.size==0: return np.inf
        if not np.all(np.isfinite(a)): return np.inf
        a=np.clip(a,0,1000)
        s=a.sum()
        if s <= 0.01: return np.inf
        # scale-invariant; no need normalize for score, but if sum huge okay
        c=np.convolve(a,a)
        return float(2.0*a.size*c.max()/(s*s))

    def quick_score(a):
        a=np.asarray(a,float)
        if a.size==0: return np.inf
        s=a.sum()
        if s <= 0.01: return np.inf
        c=conv_full(a)
        return float(2.0*a.size*c.max()/(s*s))

    def normalize_x(a):
        a=np.asarray(a,float).copy()
        if a.size==0: return a
        a=np.clip(a,0,1000)
        s=a.sum()
        if s < 1e-12:
            a.fill(1.0)
        else:
            a *= (a.size / s)
            a=np.clip(a,0,1000)
        return a

    def consider(a):

            return

    def smooth_and_grad(c,x,eps):
       cmax=float(c.max())
       z=(c-cmax)/eps
       ew=np.exp(z)
       wsum=ew.sum()
       w=ew/wsum
       smooth=cmax+eps*np.log(wsum)
       if x.size >= 128:
           try: grad=correlate(w,x,mode='valid',method='fft')
           except Exception: grad=correlate(w,x,mode='valid')
       else: grad=correlate(w,x,mode='valid')
       return float(smooth), grad

            return np.inf
