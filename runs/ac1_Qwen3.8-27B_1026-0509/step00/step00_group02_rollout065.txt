import numpy as np
import time

def _project_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - s
    rho_idx = np.nonzero(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_idx) == 0:
        return np.full(n, s / n)
    theta = (cv[rho_idx[-1]] + s) / (rho_idx[-1] + 1)
    return np.maximum(v - theta, 0)

def _compute_convo(a, fft_len):
    A = np.fft.rfft(a, fft_len)
    return np.fft.irfft(A * A, fft_len)[:]
