def fft_convolve(f):
    """Compute self-convolution via FFT."""
    n = len(f)
    size = 2*n - 1
    # Use next power of 2 for FFT efficiency
    nfft = 1
    while nfft < size:
        nfft *= 2
    F = np.fft.rfft(f, nfft)
    B = np.fft.irfft(F * F, nfft)[:size]
    return B
