import scipy.sparse as sp

            break
        
        # Compute M
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())
        if M < 1e-30:
            break
        
        # Build sparse conv matrix
        row_ptr = np.zeros(2*n, dtype=np.int64)
        col_parts = []
        data_parts = []

            js = max(0, k - n + 1)
            je = min(k, n - 1)
            cnt = je - js + 1

                solved = True
                break
