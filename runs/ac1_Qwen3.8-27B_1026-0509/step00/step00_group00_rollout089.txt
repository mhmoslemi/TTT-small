import numpy as np

TARGET = 1.50274


def _next_pow2(x):
    x = max(1, int(x))
    return 1 << (x - 1).bit_length()


def _base_unnorm(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return a


def _score_surrogate(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-15:
        return float("inf")
    n = int(a.size)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    m = float(c.max())
    if not math.isfinite(m) or m <= 0.0:
        return float("inf")
    return float(2.0 * n * m / (s * s))


def _safe_log(a):
    return np.log(np.maximum(a, 1e-300))


def _initial_vector(n, current_best, seed):

        dot = float(np.dot(a, grad_a))
        grad_v = a * (grad_a - dot)

        if not np.isfinite(grad_v).all():
            v = v_init + 0.04 * rng.normal(0.0, 1.0, n)
            mom[:] = 0.0
            ms[:] = 0.0
            continue

        mom = beta1 * mom + (1.0 - beta1) * grad_v
        ms = beta2 * ms + (1.0 - beta2) * (grad_v * grad_v)

        if it < 2000:
            lr = 0.035
        elif it < 10000:
            lr = 0.018
        else:
            lr = 0.008

        v = v - lr * mom / (np.sqrt(ms) + eps)
        v = v - float(np.mean(v))
        v = np.clip(v, -30.0, 30.0)

        if it - last_improve > 2500:
            resets += 1
            if resets > 12:
                break
            if a_best is not None:
                v = _safe_log(a_best)
            else:
                v = v_init.copy()
            sigma = 0.03 + 0.02 * min(3, resets)
            v = v + sigma * rng.normal(0.0, 1.0, n)
            v = v - float(np.mean(v))
            mom[:] = 0.0
            ms[:] = 0.0
            last_improve = it

        it += 1

    return a_best, s_best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    hard_deadline = t0 + max(30.0, float(budget_s) - 12.0)

    best = None
    best_score = float("inf")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best = p.copy()
            best_score = _score_surrogate(best)
    except Exception:
        best = None
        best_score = float("inf")

    if not math.isfinite(best_score) or best is None:
        try:
            n0 = 4096
            base = _base_unnorm(n0)
            best = base / base.sum()
            best_score = _score_surrogate(best)
        except Exception:
            best = np.array([1.0, 1.0], dtype=np.float64)
            best_score = _score_surrogate(best)

    if time.monotonic() < hard_deadline - 30.0:
        total = max(30.0, float(budget_s) - 25.0)
        plans = (
            (2048, 0.10),
            (4096, 0.50),
            (8192, 0.40),
        )
        for n, frac in plans:
            if time.monotonic() >= hard_deadline - 25.0:
                break
            alloc = min(frac * total, hard_deadline - time.monotonic() - 25.0)
            if alloc < 12.0:
                continue
            stop = time.monotonic() + alloc
            try:
                v0 = _initial_vector(n, best, seed + n)
                cand, score = _optimize(n, v0, stop, seed + n)
                if cand is not None:
                    cand = np.asarray(cand, dtype=np.float64)
                    cand = np.where(np.isfinite(cand), cand, 0.0)
                    cand = np.maximum(cand, 0.0)
                    sc = float(cand.sum())
                    if math.isfinite(sc) and sc > 0.01:
                        cand = cand / sc
                        exact_s = _score_surrogate(cand)
                        if math.isfinite(exact_s) and exact_s < best_score:
                            best = cand
                            best_score = exact_s
            except Exception:
                pass
            if best_score < TARGET - 1e-5:
                break

    try:
        out = [float(x) for x in np.asarray(best, dtype=np.float64)]
        if not out or any((not math.isfinite(x)) for x in out):
            raise ValueError
        final_score = evaluate_sequence(out)
        if math.isfinite(final_score):
            return out
    except Exception:
        pass

    fb = _base_unnorm(1024)
    fb = fb / fb.sum()
    out = [float(x) for x in fb]
    evaluate_sequence(out)
    return out
