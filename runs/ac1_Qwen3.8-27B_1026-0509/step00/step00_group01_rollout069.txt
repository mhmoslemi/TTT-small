import time
import numpy as np
from scipy.sparse import toeplitz as sparse_toeplitz
from scipy.optimize import linprog


def _build_conv_matrix(a):
    n = a.size
    c = np.concatenate([a, np.zeros(n - 1)])
    r = np.zeros(n)
    r[0] = a[0]
    return sparse_toeplitz(c, r)
