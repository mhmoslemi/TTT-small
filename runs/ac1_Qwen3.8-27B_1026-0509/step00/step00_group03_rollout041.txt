            break

        sg = g.sum()
        if sg < 1e-10:
            break
        g /= sg  # normalize to sum=1
        
        # Line search
        best_local = float(evaluate_sequence(a.tolist()))
        a_new = a
        for t in (0.5, 0.8, 1.0):
            trial = (1 - t) * a + t * g
            trial = np.clip(trial, 1e-12, None)
            trial /= trial.sum()
            s = float(evaluate_sequence(trial.tolist()))
            if s < best_local:
                best_local = s
                a_new = trial.copy()

        if best_local < best_score:
            best = a.copy()
            best_score = best_local
    
    # Final validation
    fs = float(evaluate_sequence(best.tolist()))
    if fs < best_score:
        best_score = fs
    return [float(x) for x in best]
