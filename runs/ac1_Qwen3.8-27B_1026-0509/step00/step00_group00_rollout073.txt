def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

    rng = np.random.default_rng(seed)
    start_t = time.monotonic()
    budget = max(1.0, float(budget_s) - 6.0)
    deadline = start_t + budget

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None
    if (parent is None or parent.ndim != 1 or parent.size == 0
            or not np.isfinite(parent).all()):
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0).astype(np.float64).copy()
    n = int(parent.size)

    F = _next_pow2(3 * n - 1)

    def conv(v):
        V = np.fft.rfft(v, F)
        return np.fft.irfft(V * V, F)[:2 * n - 1]

    def grad_from_w(w, v):
        Wr = np.fft.rfft(w[::-1], F)
        A = np.fft.rfft(v, F)
        c = np.fft.irfft(A * Wr, F)
        return c[n - 1:2 * n - 1][::-1]

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    S = float(best.sum())
    alpha = 0.35
    sigma = 0.4
    tau_frac = 0.6

    def normalize(v):
        v = np.clip(v, 0.0, 1000.0)
        s = v.sum()
        if s <= 1e-12:
            return None
        v = v * (S / s)
        return np.clip(v, 0.0, 1000.0)

    a = best.copy()
    stagnant = 0
    cycle = 0

    while time.monotonic() < deadline:
        # (re)initialize start from best with asymmetric perturbation when needed
        if cycle == 0 or stagnant > 60:
            noise = rng.normal(0.0, 1.0, n)
            bias = (rng.random() - 0.5) * 2.0 * sigma
            a = normalize(best * np.exp(sigma * noise + bias * np.linspace(-1, 1, n)))
            if a is None:
                a = best.copy()
            stagnant = 0

        c = np.maximum(c, 0.0)
        cmax = float(c.max())
        cmin = float(c.min())
        rng_max = max(cmax - cmin, 1e-12)

    if not np.isfinite(final_score):
        final = np.ones(n, dtype=np.float64)
