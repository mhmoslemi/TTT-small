    """Solve the LP: max sum(g) s.t. convolve(current, g)_j <= M for j in constraint_indices, g >= 0"""
    m = len(constraint_indices)
    if m == 0:
        return None
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for r, j in enumerate(constraint_indices):
        i_start = max(0, j - n + 1)
        i_end = min(n, j + 1)
        for i in range(i_start, i_end):
            val = current[j - i]
            if val > 0:
                rows.append(r)
                cols.append(i)
                vals.append(val)
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    b = np.full(m, M)
    
    # Minimize -sum(g) s.t. A@g <= b, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
