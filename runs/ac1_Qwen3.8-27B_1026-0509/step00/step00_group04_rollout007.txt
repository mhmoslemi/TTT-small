def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 5.0
    
    # Use parent as initial
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    n = parent.size
    best = parent.copy().astype(np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    def score_of(a):
        s = a.sum()
        if s < 1e-10:
            return np.inf
        b = np.convolve(a, a)
        return 2.0 * n * b.max() / (s * s)
    
    def optimize(a0, lr0, n_iter):
        a = a0.copy()
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return a, np.inf
        a = a / s  # normalize to sum=1
        best_local = a.copy()
        best_local_score = score_of(a)
        lr = lr0
        for it in range(n_iter):
            if time.monotonic() > deadline:
                break
            b = np.convolve(a, a)
            M = b.max()
            # Find all lags at max
            active = np.where(b == M)[0]
            # Subgradient of M
            g = np.zeros_like(a)
            for k in active:
                # d/da[j] (a*a)[k] = 2*a[k-j] if 0<=k-j<n
                for j in range(min(n, k+1)):
                    kj = k - j
                    if 0 <= kj < n:

            if len(active) > 1:

            # Actually score = 2n*M (since S=1), so grad_score = 2n * grad_M
            # But we have the nonnegativity constraint
            a_new = a - lr * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a
                s_new = 1
            a_new = a_new / s_new
            sc = score_of(a_new)
            if sc < best_local_score:
                best_local = a_new.copy()
                best_local_score = sc
                lr = min(lr * 1.01, 1.0)  # slight increase
            else:
                lr *= 0.95
            a = a_new
        return best_local, best_local_score

    inits = [parent / parent.sum(), parent / parent.sum() + rng.normal(0, 0.01, n)]

        inits.append(r / r.sum())
