import numpy as np
import time


def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - 1.0))[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n)
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)

    n = 2535
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            parent = np.clip(p, 0.0, 1000.0)
            if parent.sum() > 1e-8:
                n = parent.size
    except Exception:
        parent = None

    # FFT working sizes
    L1 = 1
    while L1 < 2 * n - 1:
        L1 <<= 1
    L2 = 1
    while L2 < (2 * n - 1) + n:
        L2 <<= 1

    def conv_full(a):
        A = np.fft.rfft(a, L1)
        return np.fft.irfft(A * A, L1)[:2 * n - 1]

    def corr_pa(p, a):
        P = np.fft.rfft(p, L2)
        A = np.fft.rfft(a, L2)
        return np.fft.irfft(P * np.conj(A), L2)

    def surrogate(a, c):
        s = a.sum()
        if s < 1e-12:
            return np.inf
        return 2.0 * n * c.max() / (s * s)

    best_a = np.ones(n)
    best_a /= n

    if not np.isfinite(best_score):
        best_score = np.inf

    def try_accept(a):

        sc = surrogate(a, conv_full(a))
        return sc

    def run(phi0, tau, eta0, max_iters):
        nonlocal best_a, best_score
        a = np.asarray(phi0, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1e6)

            return

            grad = 2.0 * h
            eta = eta0 / np.sqrt(float(t))
            a = _project_simplex(a - eta * grad)

            if t % 33 == 0:
