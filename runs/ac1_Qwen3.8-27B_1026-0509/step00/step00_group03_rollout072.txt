        optimize(f_init, max_iters=max_it)
        try_update(f_init)

    if best is None or best.size == 0:
        best = np.ones(10)

    final_s = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_s):
        best = np.ones(10)

    return [float(x) for x in best]
