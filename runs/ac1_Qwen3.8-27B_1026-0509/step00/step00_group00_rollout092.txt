def logsumexp(c, tau):
    cm = float(c.max())
    if not np.isfinite(cm): return np.inf
    if tau <= 0: return cm
    with np.errstate(over='ignore', under='ignore'):
        s = np.exp(tau * (c - cm)).sum()
    if not np.isfinite(s) or s <= 0:
        return cm
    return cm + float(np.log(s)) / tau

    import time

                parent = p / p.sum()
                n = int(p.size)

        parent = None

        parent = np.ones(n, dtype=float) / n

    NFFT = 1

        NFFT <<= 1
    idx = np.arange(n)
    XMAX = 25.0

    def conv(a):
        F = np.fft.rfft(a, n=NFFT)
        c = np.fft.irfft(F * F, n=NFFT)[:m]
        if c.min() < 0.0:
            c = np.maximum(c, 0.0)
        return c

    def softmax(x):
        y = x - np.max(x)
        y = np.clip(y, -30.0, 0.0)
        e = np.exp(y)
        s = e.sum()
        if not np.isfinite(s) or s <= 0:
            return np.full(n, 1.0 / n, dtype=float)
        return e / s

    def logsumexp(c, tau):
        cm = float(c.max())
        if not np.isfinite(cm):
            return np.inf
        if tau <= 0:
            return cm
        with np.errstate(over="ignore", under="ignore", invalid="ignore"):
            s = np.exp(tau * (c - cm)).sum()
        if not np.isfinite(s) or s <= 0:
            return cm
        return cm + float(np.log(s)) / tau

    def make_weights(c, tau, top_count=None):
        cm = float(c.max())
        if not np.isfinite(cm) or cm <= 0:
            return np.ones(m, dtype=float) / m

        if top_count is not None:
            k = int(top_count)
            if k <= 0:
                w = np.ones(m, dtype=float) / m
            elif k >= m:
                w = np.ones(m, dtype=float) / m
            else:
                part = np.argpartition(c, -k)[-k:]
                w = np.zeros(m, dtype=float)
                w[part] = 1.0 / k
        else:
            with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                e = np.exp(tau * (c - cm))
            s = e.sum()
            if not np.isfinite(s) or s <= 0:
                w = np.zeros(m, dtype=float)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
