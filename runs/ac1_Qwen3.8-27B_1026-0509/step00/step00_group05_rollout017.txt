        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f_new.copy()
            best_score = score
            no_improve = 0
            if t_mix < 0.25:
                t_mix = min(0.25, t_mix * 1.15)
        else:
            no_improve += 1
            t_mix = max(0.02, t_mix * 0.75)
            if no_improve > max_no_improve:
                f = best.copy()
                n_pert = max(1, n // 50)
                idx = rng.integers(0, n, size=n_pert)
                f[idx] *= np.exp(rng.normal(0, 0.3, size=n_pert))
                f = np.clip(f, 0, 1000)
                s = np.sum(f)
                if s > 1e-10:
                    f = f * target_sum / s
                no_improve = 0
                K = min(400, K + 50)
                t_mix = 0.1

        f = f_new

    # Final polish: small random perturbations on best
    remaining = deadline - time.monotonic()
    if remaining > 5:
        for _ in range(int(remaining * 20)):
            if time.monotonic() > deadline - 5:
                break
            f_try = best.copy()
            n_p = max(1, n // 500)
            idx = rng.integers(0, n, size=n_p)
            f_try[idx] *= np.exp(rng.normal(0, 0.05, size=n_p))
            f_try = np.clip(f_try, 0, 1000)
            score = float(evaluate_sequence(f_try.tolist()))
            if np.isfinite(score) and score < best_score:
                best = f_try.copy()
                best_score = score

    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        return [1.0] * n
    return [float(x) for x in best]
