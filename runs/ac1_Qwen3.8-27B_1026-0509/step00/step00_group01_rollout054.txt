import numpy as np

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # Constraint: (f*g)[k] <= M for all k
    # (f*g)[k] = sum_j f[j] * g[k-j]
    # This is a matrix-vector product: A @ g <= M * ones
    # where A[k,j] = f[k-j] if 0 <= k-j < n else 0

    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    return g.value
