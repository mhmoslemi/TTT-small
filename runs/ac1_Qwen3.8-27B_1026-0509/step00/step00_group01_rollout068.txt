import numpy as np
import time


def _proj_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, u.size + 1) > (cssv - 1))[0]
    if idx.size == 0:
        m = u.mean()
        return np.clip(u - m, 0.0, None)
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smax_grad(B, a, beta, N):
    bs = B - B.max()
    p = np.exp(beta * bs)
    p /= p.sum()
    af = a[::-1]
    c = np.fft.irfft(np.fft.rfft(af, N) * np.fft.rfft(p, N), N)
    g = c[a.size - 1:a.size - 1 + a.size]
    return (2.0 / beta) * g


def _normalize(a, n):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 1e-15:
        return np.full(n, 1.0 / n)
    return a / s
