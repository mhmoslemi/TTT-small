def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.real(ifft(fft(a, size) * fft(b, size))[:n])

def _proj_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > cssv - 1)[0]
    if rho.size == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _optimize_small(n, budget_s, rng):

        grad[valid & (2*indices == k_star)] = 2 * p[shifted[valid & (2*indices == k_star)]]
