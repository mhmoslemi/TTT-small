import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):

        s = n

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    def get_conv(a):
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)
        return c[:2*n-1]
    
    lr = 0.05
    noise_scale = 0.01
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        c = get_conv(a)
        peak_val = c.max()
        
        # Find top peaks (within 5% of max)
        threshold = peak_val * 0.95
        peak_mask = c >= threshold
        peak_indices = np.where(peak_mask)[0]
        
        # Compute gradient at peak positions
        g = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1)
            km = k - j
            g[j] += 2.0 * a[km]
            if k % 2 == 0:

                if lo <= mid <= hi:
                    g[mid] -= a[mid]  # correct the double count
        
        # Gradient descent
        a = a - lr * g / peak_val  # normalize gradient by peak value
        a = np.maximum(a, 0.0)
        
        # Add small noise for exploration
        a += noise_scale * rng.standard_normal(n) * a.sum() / n
        a = np.maximum(a, 0.0)
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            a = best.copy()
            continue
        a = a / s
        
        # Evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score

- Number of distinct peak positions: up to 2n-1
- If perfectly spread: max = 1/(2n-1) → score = 2n/(2n-1) ≈ 1
- Realistically with m random bumps: the pairs cluster, so max > average
