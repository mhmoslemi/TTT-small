```python
import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where (A @ g) = f * g (full)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    A = sp.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A.tocsr()

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0. Returns optimal g."""
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(objective, constraints)
    prob.solve()
    if g.value is None:
        return None
    return np.maximum(g.value, 0)
```

```python
def project_simplex(v, s=1):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Scale: project onto {x >= 0, sum(x) = s}
    # Equivalent to: project v/s onto {x >= 0, sum(x) = 1}, then multiply by s.
    # Algorithm:
    # 1. sort v in descending order
    # 2. find the largest k such that sum of top k elements - k * lambda > 0
    #    where lambda = (sum of top k elements - s) / k
    # 3. projection = max(v - lambda, 0)
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is O(n log n) due to the sort. For n=3809, that's fast.
