        archive = [(sc, arr.copy()) for sc, arr in temp[:8]]
    else:
        best_score = 2.0
        best_a = np.ones(n, dtype=np.float64) / n
        archive = []

    def add_archive(arr, sc):
        if arr is None:
            return
        arr = norm(arr)
        archive.append((sc, arr))
        if len(archive) > 12:
            archive.sort(key=lambda z: z[0])
            del archive[8:]

    bounds_lp = [(0.0, None)] * (n + 1)
    a = best_a.copy()
    cur = float(best_score)

    K_mirror = 48
    K_lp = 32
    lp_interval = 40
    max_lp = 250
    lp_count = 0
    iter_no = 0
    stagnant = 0
    global_stagn = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        c = conv(a)
        cur = float(2.0 * n * np.max(c))
        idx = top_idx(c, K_mirror)
        cands = []

        def push(b):
            if len(cands) >= 10:
                return
            b = np.asarray(b, dtype=np.float64)
            b = np.maximum(b, 0.0)
            s = float(b.sum())
            if np.isfinite(s) and s > 0.0:
                cands.append(b / s)

        if idx.size:
            vals = np.asarray(c[idx], dtype=np.float64)
            mx, mn = float(vals.max()), float(vals.min())
            if mx - mn > 1e-300:
                w = 0.5 + 0.5 * (vals - mn) / (mx - mn)
            else:
                w = np.ones(idx.size, dtype=np.float64)
            w = w / w.sum()
            g = np.zeros(n, dtype=np.float64)
            for r, k in enumerate(idx):
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if hi >= lo:
                    g[lo:hi + 1] += w[r] * 2.0 * a[k - hi:k - lo + 1][::-1]
            gmax = float(g.max())
            if gmax > 1e-300:
                gg = g / gmax
                for eta in (0.05, 0.15, 0.5, 1.5):
                    push(a * np.exp(-eta * gg))

        if (iter_no % lp_interval == 0) and (lp_count < max_lp) and (rem > 90.0):
            lpidx = top_idx(c, K_lp)
            sdir = lp_dir(a, c, lpidx)
            if sdir is not None:
                for lam in (0.2, 0.5, 1.0):
                    push((1.0 - lam) * a + lam * sdir)
            lp_count += 1

        if iter_no % 20 == 0 or stagnant >= 6:
            strength = 0.12 + 0.18 * min(1.0, float(stagnant) / 50.0)
            push(rand_perturb(a, strength))

        improved = False
        best_cand = None
        best_cand_score = cur
        for b in cands:
            if time.monotonic() > deadline - 2.0:
                break
            sc = float(2.0 * n * np.max(conv(b)))
            if sc < best_cand_score - 1e-14:
                best_cand_score = sc
                best_cand = b

        if best_cand is not None:
            a = best_cand
            cur = best_cand_score
            improved = True

        if improved:
            stagnant = 0
            if cur < best_score - 1e-12:
                best_score = cur
                best_a = a.copy()
                add_archive(best_a, best_score)
                global_stagn = 0
            else:
                global_stagn += 1
        else:
            stagnant += 1
            global_stagn += 1
            if stagnant >= 15 or global_stagn >= 150:
                strength = 0.20 + 0.50 * min(1.0, float(global_stagn) / 300.0)
                src = best_a
                if archive and rng.random() < 0.5:
                    src = archive[int(rng.integers(0, len(archive)))][1]
                if rng.random() < 0.25:
                    a = random_start()
                else:
                    a = rand_perturb(src, strength)
                c = conv(a)
                cur = float(2.0 * n * np.max(c))
                stagnant = 0
                global_stagn = max(0, global_stagn - 20)

        if iter_no % 500 == 499 and K_mirror < 128:
            K_mirror += 16
        iter_no += 1

    final_candidates = [best_a.copy()]
    try:
        final_candidates.append(norm(p))
    except Exception:
        pass
    for _, arr in archive[:3]:
        final_candidates.append(arr.copy())
        if len(final_candidates) >= 4:
            break

    final_arr = best_a
    best_exact = np.inf
    for arr in final_candidates:
        if time.monotonic() >= t0 + budget - 1.0:
            break
        arr = np.clip(np.asarray(arr, dtype=np.float64), 0.0, 1000.0)
        s = float(arr.sum())
        if not np.isfinite(s) or s < 0.01 or not np.all(np.isfinite(arr)):
            continue
        arr = arr / s
        sc = evaluate_sequence([float(x) for x in arr])
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            final_arr = arr

    if not np.isfinite(best_exact):
        arr = np.clip(np.asarray(final_arr, dtype=np.float64), 0.0, 1000.0)
        if arr.size == 0 or not np.all(np.isfinite(arr)) or arr.sum() < 0.01:
            arr = np.ones(6962, dtype=np.float64)
        final_arr = arr / arr.sum()

    return [float(x) for x in final_arr]
