import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix


def _fft_plan(n):
    conv_len = 2 * n - 1
    fft_size = 1
    while fft_size < conv_len:
        fft_size <<= 1
    return conv_len, fft_size


def _score(seq):
    try:
        s = float(evaluate_sequence([float(x) for x in seq]))
        return s if np.isfinite(s) else np.inf
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1500, **kwargs):

        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    conv_len, fft_size = _fft_plan(n)

    f = parent / parent.sum()
    best = f.copy()
    best_score = _score(best)

        s = a.sum()
        return a / s if s > 0 else a

        all_rows = np.concatenate([np.arange(s, s + l) for s, l in zip(col_starts, col_lengths)])

            continue
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best = cand

    return [float(x) for x in best]
