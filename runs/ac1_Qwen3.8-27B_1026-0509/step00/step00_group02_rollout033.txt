```python
def lp_refine(a, M, max_iter=50):
    """Solve LP with constraint generation."""
    n = len(a)
    active_lags = set()
    
    # Start with the lag(s) where conv(a,a) is maximum
    c = conv_full(a)
    active_lags.add(int(np.argmax(c)))
    
    for _ in range(max_iter):
        # Build constraint matrix from active lags
        lags = sorted(active_lags)
        A = np.zeros((len(lags), n))
        for idx, k in enumerate(lags):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            A[idx, j_lo:j_hi+1] = a[k - j_hi:k - j_lo + 1][::-1]
        
        # Solve LP
        b = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b)), [A @ b <= M])
        try:
            prob.solve(solver=cp.GLOP)
        except:
            prob.solve(solver=cp.CLIP)
        
        if b.value is None:
            return None
        b_val = np.maximum(b.value, 0)
        
        # Check if any constraint is violated
        conv_ab = np.convolve(a, b_val)  # This is O(n^2)!
        # Or use FFT
        nfft = 1
        while nfft < 2*n - 1:
            nfft <<= 1
        A_f = np.fft.rfft(a, n=nfft)
        B_f = np.fft.rfft(b_val, n=nfft)
        conv_ab = np.fft.irfft(A_f * B_f, n=nfft)[:2*n-1]
        
        violated = [k for k in range(len(conv_ab)) if conv_ab[k] > M * (1 + 1e-8)]
        if not violated:
            return b_val
        # Add the most violated constraint
        k_new = max(violated, key=lambda k: conv_ab[k])
        active_lags.add(k_new)
    
    return np.maximum(b_val, 0)  # Return last solution even if not fully feasible
```

For n=3: Can we find a=[a0,a1,a2] with sum=1 such that conv(a,a) is as flat as possible?
conv = [a0^2, 2a0a1, a1^2+2a0a2, 2a1a2, a2^2]
We want all 5 elements equal to 1/5.
a0^2 = 1/5, so a0 = 1/sqrt(5)
a2^2 = 1/5, so a2 = 1/sqrt(5)
2a0a1 = 1/5, so a1 = 1/(10a0) = sqrt(5)/10
Check: a1^2 + 2a0a2 = 5/100 + 2*1/5 = 1/20 + 2/5 = 1/20 + 8/20 = 9/20 ≠ 1/5

So it's not achievable for n=3. The flat autocorrelation is only achievable for certain (n, alphabet) combinations (related to Barker codes and related structures).

    while nfft < 2 * n - 1:
        nfft <<= 1
    nfft3 = 1
    while nfft3 < 3 * n - 2:
        nfft3 <<= 1

        X = np.fft.rfft(x, n=nfft)
        return np.fft.irfft(X * X, n=nfft)[:2 * len(x) - 1]

    def conv_cross(x, y):
        X = np.fft.rfft(x, n=nfft)
        Y = np.fft.rfft(y, n=nfft)
        return np.fft.irfft(X * Y, n=nfft)[:len(x) + len(y) - 1]

    def score(x):

            return float(evaluate_sequence([float(v) for v in x]))

            return float('inf')

    a0 = parent / parent.sum()
    best = a0.copy()
    best_score = score(best)

    def lp_solve(a, M, max_gen=80, time_limit=10.0):
        n_loc = len(a)
        c = conv_self(a)
        active = [int(np.argmax(c))]
        b_val = None
        t0 = time.monotonic()
        for _ in range(max_gen):
            if time.monotonic() - t0 > time_limit:
                break
            lags = sorted(set(active))
            A = np.zeros((len(lags), n_loc))
            for i, k in enumerate(lags):
                j_lo = max(0, k - n_loc + 1)
                j_hi = min(n_loc - 1, k)
                A[i, j_lo:j_hi + 1] = a[k - j_lo:k - j_hi + 1][::-1]
            b = cp.Variable(n_loc, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(b)), [A @ b <= M])
            solved = False
            for solver in [cp.GLOP, cp.CLIP, cp.ECOS]:

                    prob.solve(solver=solver)
                    solved = True
                    break

                    continue
            if not solved or b.value is None:

            viol = cross > M * (1 + 1e-7)

        return b_val if b_val is not None else None

        nloc = len(a)
        m = np.zeros(nloc)
        v = np.zeros(nloc)
        t_count = 0

            w = np.exp(alpha * (c - c_max))
            w /= w.sum()

            S = a.sum()
            M = c_max
            grad = 2.0 * nloc * (S * g - 2.0 * M) / (S ** 3)

            best_score = s

    # Final local search on best

        M = conv_self(a).max()
        while time.monotonic() < deadline - 10:
            b = lp_solve(a, M, max_gen=50, time_limit=5.0)

                break
            S_b = b.sum()
            S_a = a.sum()
            if S_b <= S_a * (1 + 1e-5):

    final_s = score(best)
