import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def _norm_nonneg(x):
    x = np.clip(np.asarray(x, dtype=np.float64), 0.0, 1000.0)
    s = x.sum()
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones_like(x, dtype=np.float64) / x.size
    return x / s


def _exactly_score(x):
    v = evaluate_sequence(np.asarray(x, dtype=np.float64).tolist())
    return v if np.isfinite(v) else np.inf


def _build_conv_mat(f):
    n = f.size
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_idx = col_idx.copy() + np.tile(np.arange(n, dtype=np.int64), n)
    vals = np.tile(f, n)
    return sparse.coo_array((vals, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()


def _solve_lp(f):
    n = f.size
    A = _build_conv_mat(f)
    c = np.zeros(n + 1); c[-1] = 1.0
    ones = np.ones(2 * n - 1)
    neg_col = sparse.csc_array((ones, (np.arange(2 * n - 1), np.zeros(2 * n - 1, dtype=np.int64))),
                               shape=(2 * n - 1, 1))
    A_ub = sparse.hstack([A, -neg_col]).tocsr()
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res is None or not res.success or res.x is None:
        return None
    g = np.clip(res.x[:n], 0.0, None)
    gs = g.sum()
    if gs <= 1e-15:
        return None
    return g / gs


def _block_perturb(base, rng):
    c = base.copy()

    w = int(rng.integers(max(1, n // 400), max(2, n // 20)))

    c[st:stp] *= float(np.exp(rng.normal(0.0, 0.15)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, 1000.0)
