def _project_simplex(v, target):
    """Project v onto {x>=0, sum(x)=target} via bisection on threshold."""
    n = v.size
    lo = 0.0
    hi = v.max() - target / n + 1.0
    if hi < 0: hi = v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.maximum(v - mid, 0.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.maximum(v - tau, 0.0)
