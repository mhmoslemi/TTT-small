    """Solve: max sum(g) s.t. conv(f,g)[k] <= M for k in top_k, 0<=g<=1000."""

    # For each k in top_k: sum_i f_i * g_{k-i} <= M

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0, budget_s - 8)

    parent = np.ones(256, dtype=np.float64)
    if 'height_sequence_1' in globals():
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
    n = parent.size
    if n < 4:
        return parent.tolist()

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    conv_len = 2 * n - 1
    N = 1
    while N < conv_len:
        N *= 2

    def conv_self(x):
        fx = fft(x, N)
        return np.real(ifft(fx * fx, N)[:conv_len])

    def norm1(x):
        s = x.sum()
        return x / s if s > 1e-12 else x

    def eval_fast(a):
        """Approximate score using FFT."""
        s = a.sum()
        if s < 1e-12:
            return np.inf
        c = conv_self(a)
        return 2 * n * c.max() / (s * s)

    def lp_direction(a, K=80):
        """Find LP direction: max sum(g) s.t. conv(a,g)[k]<=M at top-K lags."""
        conv = conv_self(a)
        M = conv.max()
        if M < 1e-15:
            return None
        top_k = np.argsort(conv)[-K:]
        c_obj = -np.ones(n)
        A = np.zeros((K, n))
        for idx, k in enumerate(top_k):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                A[idx, js:je + 1] = a[k - np.arange(js, je + 1)]
        b_ub = np.full(K, M)
        bounds = [(0.0, 1000.0)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 5, 'presolve': True})
            if res.success and res.x is not None and res.x.sum() > 0.01:
                return np.clip(res.x, 0, 1000.0)
        except Exception:
            pass
        return None

    def gradient_step(a, conv, k_active, step):
        """Projected gradient step on simplex to reduce active convolution peaks."""
        grad = np.zeros(n)
        for k in k_active:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                grad[js:je + 1] += 2.0 * a[k - np.arange(js, je + 1)]
        gmax = grad.max()
        if gmax < 1e-15:
            return a
        a_new = a - step * grad / gmax
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            return a
        return a_new / s

    def run_opt(a_init, time_limit, tag=""):
        nonlocal best, best_score
        a = norm1(a_init.astype(np.float64).copy())
        t_start = time.monotonic()
        K = min(80, conv_len)
        t_mix = 0.4
        it = 0
        prev_val = conv_self(a).max()

        while time.monotonic() < deadline and time.monotonic() - t_start < time_limit:
            conv = conv_self(a)
            val = conv.max()
            # Adaptive K: top lags near the peak
            frac = 0.1 if it < 200 else 0.05
            thresh = val * (1 - frac)
            active_all = np.where(conv >= thresh)[0]
            K_eff = min(K, len(active_all))
            if K_eff < 5:
                K_eff = 5
                active_all = np.argsort(conv)[-K_eff:]

            # Try LP
            g = lp_direction(a, K=K_eff)
            if g is not None and g.sum() > 0.01:
                g_n = g / g.sum()
                a_cand = (1 - t_mix) * a + t_mix * g_n
                a_cand = np.maximum(a_cand, 0)
                sc = eval_fast(a_cand)
                if np.isfinite(sc) and sc < eval_fast(a) * 1.02:
                    a = a_cand / a_cand.sum()
                else:
                    # Gradient fallback
                    k_active = np.argsort(conv)[-min(10, K_eff):]
                    a = gradient_step(a, conv, k_active, 0.02)
            else:
                k_active = np.argsort(conv)[-min(10, K_eff):]
                a = gradient_step(a, conv, k_active, 0.01)

            it += 1
            if it % 30 == 0:
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = a.copy()
            if time.monotonic() > deadline:
                break
        # Final check
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
        return a

    # Phase 1: Optimize from parent
    a_parent = parent / parent.sum()
    run_opt(a_parent, budget_s * 0.5, "parent")

    # Phase 2: Try alternative initializations
    remaining = deadline - time.monotonic()
    if remaining > 30:
        inits = []
        # Skewed: exponential decay
        x = np.arange(n, dtype=np.float64)
        for alpha in [1, 2, 4, 8]:
            inits.append(np.exp(-alpha * x / n))
        # Bimodal: peaks at 1/4 and 3/4
        inits.append(np.exp(-((x - n * 0.25) / (n * 0.15)) ** 2) +
                       np.exp(-((x - n * 0.75) / (n * 0.15)) ** 2))
        # Step: first half
        inits.append(np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]))
        # Random
        inits.append(rng.random(n))
        # Parent perturbed
        inits.append(parent / parent.sum() * np.exp(rng.normal(0, 0.1, n)))

        per_init = remaining / (len(inits) + 1)
        for idx, init in enumerate(inits[:6]):
            if time.monotonic() > deadline - 10:
                break
            if init.sum() < 0.01:
                continue
            run_opt(init, max(10, per_init), f"init{idx}")

    # Final validation
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        final_sc = float(evaluate_sequence(parent.tolist()))
        best = parent

    return [float(x) for x in best]
