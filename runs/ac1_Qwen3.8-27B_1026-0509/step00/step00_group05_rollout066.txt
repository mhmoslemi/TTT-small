import numpy as np
import time
from scipy.optimize import linprog


def _build_conv_matrix(x):
    n = x.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi + 1] = x[k - np.arange(lo, hi + 1)]
    return A


def _lp_direction(x, time_limit):
    n = x.size
    M = float(np.max(np.convolve(x, x)))
    A = _build_conv_matrix(x)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    b = np.asarray(res.x, dtype=np.float64)
    sb = b.sum()
    if sb < 1e-12:
        return None
    return b / sb
