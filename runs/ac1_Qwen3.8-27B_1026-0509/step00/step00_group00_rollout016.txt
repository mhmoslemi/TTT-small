        return 2.0 * n * c.max() / (a.sum() ** 2)

        s0 = fscore(a0)

    all_rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    all_cols = np.concatenate([np.full(n, j) for j in range(n)])
    total_nnz = n * n

    for it in range(2000):
        if time.monotonic() > deadline - 40:
            break

        if S < 1e-12:
            a = best.copy()
            continue

        M = conv.max()
        if M < 1e-15:
            break

        vals = np.tile(an, n)
        A = csc_matrix((vals, (all_rows, all_cols)), shape=(2 * n - 1, n))

        # Solve LP
        b = cp.Variable(n)

        try:
            t0 = time.monotonic()
            prob.solve(solver=cp.CLARABEL, verbose=False)
            if time.monotonic() - t0 > 200:
