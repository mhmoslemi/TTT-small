import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 6.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(257, dtype=np.float64)
    n = int(parent.size)

    f = parent.astype(np.float64)
    f = f / f.sum()

    def fast_score(a):
        s = a.sum()
        if s < 1e-10:
            return np.inf
        return float(2.0 * n * np.convolve(a, a).max() / (s * s))

    def exact_score(a):
        return float(evaluate_sequence([float(x) for x in a]))

    best = f.copy()
    best_score = exact_score(best)

    def project_simplex(v):
        v = np.asarray(v, dtype=np.float64).ravel()
        m = v.size
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        cand = u * np.arange(1, m + 1) > (cssv - 1.0)
        idx = np.nonzero(cand)[0]
        rho = int(idx[-1]) if idx.size else m - 1
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w / w.sum()

    def subgrad(f, eta):
        conv = np.convolve(f, f)
        M = conv.max()
        active = np.where(conv >= 0.995 * M)[0]
        grad = np.zeros(n)
        for k in active:
            sh = k - np.arange(n)
            ok = (sh >= 0) & (sh < n)
            grad[ok] += f[sh[ok]]
        grad *= 2.0 / max(1, active.size)
        return project_simplex(f - eta * grad)

    def mwa(f, eta):
        conv = np.convolve(f, f)
        k = int(np.argmax(conv))
        sh = k - np.arange(n)
        ok = (sh >= 0) & (sh < n)
        g = np.zeros(n)
        g[ok] = 2.0 * f[sh[ok]]
        fn = f * np.exp(-eta * g)
        fn = np.where(fn > 1e-18, fn, 1e-18)
        return fn / fn.sum()

    def build_C(lags, f):
        ri, ci, da = [], [], []
        for li, k in enumerate(lags):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            ci.extend(range(lo, hi + 1))
            ri.extend([li] * (hi - lo + 1))
            da.extend(f[k - hi:k - lo + 1][::-1].tolist())
        if not da:
            return None
        return sp.csc_matrix((da, (ri, ci)), shape=(len(lags), n))

    def solve_lp(prob):
        for slv in (cp.ECOS, cp.PDLP, cp.CBC, cp.GLOP):
            try:
                prob.solve(solver=slv, verbose=False)
                if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                    return True
            except Exception:
                continue
        return False

    def lp_escape(f, hard_stop):
        conv = np.convolve(f, f)
        M = float(conv.max())
        kstar = int(np.argmax(conv))
        lags = [kstar]
        g_sol = None
        for _ in range(24):
            if time.monotonic() > hard_stop:
                break
            C = build_C(lags, f)
            if C is None:
                break
            gv = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(gv)), [C @ gv <= M])
            if not solve_lp(prob):
                break
            if gv.value is None:
                break
            g_sol = np.clip(np.asarray(gv.value).ravel(), 0.0, None)
            if g_sol.sum() < 1e-10:
                break
            convfg = np.convolve(f, g_sol)
            viol = np.where(convfg > M * 1.0005)[0].tolist()
            vset = set(lags)
            viol = [k for k in viol if k not in vset]
            if not viol:
                break
            if len(viol) > 250:
                vals = np.array([convfg[k] for k in viol])
                top = np.argsort(vals)[-250:]
                viol = [viol[i] for i in top]
            lags.extend(viol)
        if g_sol is not None and g_sol.sum() > 1e-10:
            return g_sol / g_sol.sum()
        return None

    def blend_line(f, d, base_obj):
        best_t, best_f, best_o = 0.0, f.copy(), base_obj
        for t in (0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6):
            fn = (1.0 - t) * f + t * d
            fn = np.maximum(fn, 0.0)
            s = fn.sum()
            if s < 1e-10:
                continue
            fn = fn / s
            o = fast_score(fn)
            if o < best_o - 1e-12:
                best_o, best_t, best_f = o, t, fn.copy()
        return best_f, best_o, best_t

    t0 = time.monotonic()
    eta = 1.0 / n
    step = 0
    last_improve = 0
    lp_every = 8

    while time.monotonic() < deadline - 25.0:
        hard = min(deadline - 60.0, time.monotonic() + 40.0)
        if step % lp_every == 0:
            g = lp_escape(f, hard)
            if g is not None:
                base = fast_score(f)
                fn, o, _ = blend_line(f, g, base)
                if o < base:
                    f = fn
                    cur = exact_score(f)
                    if cur < best_score - 1e-9:
                        best_score, best = cur, f.copy()
                        last_improve = step
                eta = max(eta * 0.97, 0.25 / n)
        else:
            f = subgrad(f, eta)
            if step % 16 == 0:
                o = fast_score(f)
                if o < exact_score(best) - 2e-4:
                    cur = exact_score(f)
                    if cur < best_score - 1e-9:
                        best_score, best = cur, f.copy()
                        last_improve = step

        step += 1
        if (step - last_improve) > 400 and step > 200:
            f = project_simplex(f + 0.004 * rng.standard_normal(n))
            eta = max(eta, 0.5 / n)

    # final light polish with multiplicative updates
    hard = deadline - 12.0
    for _ in range(40):
        if time.monotonic() > hard:
            break
        d = mwa(best, 0.02)
        fn, o, _ = blend_line(best, d, fast_score(best))
        if o < fast_score(best):
            best = fn

    best = np.clip(best, 0.0, 999.0)
    if best.sum() < 0.01:
        best = parent
    else:
        best = best / best.sum()
    try:
        if exact_score(best) < best_score:
            best_score = exact_score(best)
    except Exception:
        pass
    return [float(x) for x in best]
