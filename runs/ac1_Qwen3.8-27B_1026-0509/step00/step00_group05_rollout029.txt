```python
def fast_convolve(x, y):
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()  # next power of 2
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    return np.fft.irfft(X * Y, N)[:n]
```
