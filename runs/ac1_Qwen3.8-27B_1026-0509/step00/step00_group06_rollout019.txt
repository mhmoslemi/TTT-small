    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    
    if n < 2:
        return [1.0]
    
    # FFT sizes
    N_self = 1 << (2 * n - 1).bit_length()
    N_cross = 1 << (3 * n - 2).bit_length()
    
    # Verify parent score
    best_a = parent.copy().astype(np.float64)
    best_score = float(evaluate_sequence(best_a.tolist()))
    
    # Generate multiple initializations
    inits = []
    
    # Init 1: perturbed parent
    a1 = parent.astype(np.float64).copy()
    a1 = a1 / (a1.sum() + 1e-30)
    noise = rng.normal(0, 0.5, n)
    a1 = a1 + noise * np.std(a1)
    a1 = np.maximum(a1, 0)
    a1 = a1 / a1.sum()
    inits.append(a1)
    
    # Init 2: random nonnegative
    a2 = rng.uniform(0.1, 1.0, n)
    a2 = a2 / a2.sum()
    inits.append(a2)
    
    # Init 3: random with structure (low-freq perturbation)
    a3 = np.ones(n) / n
    k = rng.integers(1, max(2, n // 4))
    freq = rng.uniform(1, 5)
    x = np.arange(n)
    a3 = a3 * (1 + 0.4 * np.cos(2 * np.pi * freq * x / n + rng.uniform(0, 6.28)))
    a3 = np.maximum(a3, 0)
    a3 = a3 / a3.sum()
    inits.append(a3)
    
    # Init 4: chirp-like
    a4 = 1.0 + 0.5 * np.sin(np.pi * np.arange(n) * 7.0 / n)
    a4 = np.maximum(a4, 0)
    a4 = a4 / a4.sum()
    inits.append(a4)
    
    # Allocate time per initialization (leave margin)
    n_inits = len(inits)
    per_init_time = (float(budget_s) - 20.0) / n_inits
    
    for i, a_init in enumerate(inits):
        init_deadline = time.monotonic() + per_init_time
        if init_deadline > total_deadline:
            init_deadline = total_deadline
        if time.monotonic() >= init_deadline:
            break
        
        # Try a few eps values for the first init
        eps_vals = [5.0, 20.0] if i == 0 else [20.0]
        for eps_val in eps_vals:
            if time.monotonic() >= init_deadline:
                break
            cand, score = _optimize_from(
                a_init, n, N_self, N_cross, init_deadline, rng,
                init_eps=eps_val, init_lr=0.05
            )
            if score < best_score:
                best_score = score
                best_a = cand
    
    # Final local refinement on best
    if time.monotonic() < total_deadline - 5.0:
        a_refine = best_a.copy()
        a_refine = a_refine / a_refine.sum()
        cand, score = _optimize_from(
            a_refine, n, N_self, N_cross, total_deadline, rng,
            init_eps=50.0, init_lr=0.02
        )
        if score < best_score:
            best_score = score
            best_a = cand
    
    # Final validation
    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score):
        final_score = float(evaluate_sequence(parent.tolist()))
        best_a = parent
    
    return [float(x) for x in best_a]
