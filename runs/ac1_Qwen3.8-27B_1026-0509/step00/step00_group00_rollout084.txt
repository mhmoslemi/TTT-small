    a = np.ones(N) / N
    a += 1e-6 * np.random.default_rng(42).random(N) # small random init to break symmetry
    a = a / np.sum(a)
    best_score = np.inf
    best_a = a.copy()
    
    fft_len = 1
    while fft_len < 2 * N - 1:
        fft_len *= 2
    
    for i in range(iters):
        # Compute outer product and inverse FFT
        A = np.fft.rfft(a, fft_len)
        C = np.fft.irfft(A * A, fft_len)[:2*N-1]
        
        k_star = np.argmax(C)
        
        # Compute gradient
        idx = k_star - np.arange(N)
        valid = (idx >= 0) & (idx < N)
        grad = np.zeros(N)
        grad[valid] = 2 * a[idx[valid]]
        
        # Update
        a = a * np.exp(-eta * grad / C[k_star]) # normalized grad for stability
        a += 1e-8
        a = a / np.sum(a)
        
        # Check score
        score = 2 * N * C[k_star] # since we are in loop, we already have C!
        if score < best_score:
            best_score = score
            best_a = a.copy()
            
    return best_score, best_a
