import numpy as np
import time
from scipy import sparse
import cvxpy as cp


def _conv_matrix(x, rows, n):
    """Sparse converse-Toeplitz A with A[k,j]=x[k-j] (0<=k-j<n), only given rows."""
    cols_all = [np.arange(max(0, k - n + 1), min(n - 1, k) + 1, dtype=np.int32) for k in rows]
    rows_all = [np.full(len(c), k, dtype=np.int32) for c, k in zip(cols_all, rows)]
    vals_all = [x[k - c] for c, k in zip(cols_all, rows)]
    A = sparse.csc_matrix(
        (np.concatenate(vals_all), (np.concatenate(rows_all), np.concatenate(cols_all))),
        shape=(len(rows), n),
    )
    return A


def _solve_lp(A, M, n, deadline):
    g = cp.Variable(n, nonneg=True)
    cons = [A @ g <= M * np.ones(len(A.indices)), g <= 1000.0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
    solvers = [cp.CBC, cp.PDLP, cp.GLPK, cp.ECOS, cp.CVXOPT]
    for s in solvers:
        if time.monotonic() > deadline:
            return None
        try:
            t0 = time.monotonic()
            prob.solve(solver=s, timeout=max(2.0, min(30.0, deadline - t0 - 1.0)))
            if g.value is not None and np.all(np.isfinite(g.value)):
                return np.clip(g.value, 0.0, 1000.0)
        except Exception:
            continue
    return None
