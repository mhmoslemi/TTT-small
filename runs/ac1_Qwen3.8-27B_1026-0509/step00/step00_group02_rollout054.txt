import time
import scipy.fft

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    conv_len = 2 * n - 1
    fft_size = scipy.fft.next_fast_len(conv_len)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    def compute_conv(f):
        F = scipy.fft.rfft(f, fft_size)
        return scipy.fft.irfft(F * F, fft_size)[:conv_len]

    def quick_score(f):
        s = np.sum(f)
        if s < 0.01:
            return np.inf
        c = compute_conv(f)
        return 2.0 * n * float(np.max(c)) / (s * s)

    def try_update(f_trial):
        nonlocal best, best_score
        f_trial = np.clip(f_trial, 0.0, 1000.0)
        s = np.sum(f_trial)
        if s < 0.01:
            return False
        sc = quick_score(f_trial)
        if np.isfinite(sc) and sc < best_score:
            exact = float(evaluate_sequence(f_trial.astype(float).tolist()))
            if exact < best_score:
                best = f_trial.copy()
                best_score = exact
                return True
        return False

    t = np.linspace(0.0, 1.0, n)

    # Phase 2: iterative multiplicative gradient descent on max-conv
    f = best.copy()
    alpha = 0.15

        if p_max < 1e-15:

        S_new = np.sum(f_new)
