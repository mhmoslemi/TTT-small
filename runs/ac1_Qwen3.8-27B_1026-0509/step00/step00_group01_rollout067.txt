import numpy as np
import time


def props(v, s):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (css - s))[0]
    if len(rho) == 0:
        th = (v.sum() - s) / n
    else:
        th = (css[rho[-1]] - s) / (rho[-1] + 1)
    return np.maximum(v - th, 0.0)


def _smooth_opt(a, deadline, beta, base_lr, decay, iters):
    n = a.size
    s = a.sum()
    fc = 2 * n - 1
    N1 = 1 << max(1, (fc - 1).bit_length())
    N2 = 1 << max(1, (3 * n - 2 - 1).bit_length())
    w = np.zeros(N2)
    w[:n] = 1.0
    W = np.fft.rfft(w, N2)

        if time.monotonic() > deadline:
            break

    return a
