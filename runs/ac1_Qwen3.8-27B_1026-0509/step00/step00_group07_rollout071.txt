import numpy as np
import time


def _fft_len(n):
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    return m


def _conv_a(a, N):
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)


def _corr(w, a, N):
    # (w corr a)[m] = sum_k w_k a_{k-m}  for m=0..n-1 (a zero-padded outside [0,n))
    Fw = np.fft.rfft(w, N)
    Fa = np.fft.rfft(a, N)
    return np.fft.irfft(Fw * np.conj(Fa), N)
