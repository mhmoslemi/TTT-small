import time

    import cvxpy as cp

def _score(x):
    n = x.size
    b = np.convolve(x, x)
    mb = float(b.max())
    s = float(x.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * mb / (s * s))


def _bg(x):
    b = np.convolve(x, x)
    k = int(np.argmax(b))
    n = x.size
    idx = k - np.arange(n)
    g = np.zeros_like(x)
    ok = (idx >= 0) & (idx < n)
    g[ok] = 2.0 * x[idx[ok]]
    return float(b.max()), k, g


def _subgrad(x, rng, steps, deadline):
    y = np.clip(x, 0.0, None).astype(np.float64)
    sm = y.sum()
    if sm <= 1e-12:
        return x
    y = y / sm
    esc = 0.12
    for _ in range(steps):
        if time.monotonic() > deadline:
            break
        mb, k, g = _bg(y)
        ng = float(np.max(g))
        if ng <= 0.0:
            break
        for f in (1.0, 0.5, 0.25, 0.125, 0.0625):
            lr = esc * f
            y2 = np.clip(y - lr / (ng * ng + 1e-12) * g, 0.0, None)
            s2 = y2.sum()
            if s2 <= 1e-12:
                continue
            y2 = y2 / s2
            if _bg(y2)[0] <= mb + 1e-10:
                y = y2
                esc = min(esc * 1.04, 5.0)
                break
    return y


def _block(x, rng, deadline):
    n = x.size
    c = x.copy()
    w = int(rng.integers(max(2, n // 400), n // 6))
    s = int(rng.integers(0, n - w))
    c[s:s + w] *= np.exp(rng.normal(0.0, 0.25, w))
    if rng.random() < 0.3:
        c[s:s + w] += rng.uniform(0.0, 1.0) * c.mean()
    return np.clip(c, 0.0, 1000.0)


def _initializations(n, s_target, rng):
    out = []
    for a in (0.5, 1.0, 2.0, 3.0, 5.0):
        x = np.exp(-a * np.linspace(0.0, 1.0, n))
        out.append(x / x.sum() * s_target)
    for a in (0.5, 1.5, 3.0):
        x = 1.0 + a * np.cos(np.linspace(0.0, 1.6 * np.pi, n))
        x = np.clip(x, 0.05, None)
        out.append(x / x.sum() * s_target)
    for a in (0.8, 2.0, 4.0):
        t = np.linspace(0.0, 1.0, n)
        x = (1.0 - t) ** a + 0.02
        out.append(x / x.sum() * s_target)
    out.append(rng.random(n) * 0.8 + 0.2)
    out.append(rng.permutation(n).astype(np.float64) / n + 0.05)
    return [np.clip(v / v.sum() * s_target, 0.0, 1000.0) for v in out]


def _lp_jump(x, deadline):
    if not _HAVE_LP:
        return None
    if time.monotonic() > deadline - 120.0:
        return None
    n = x.size
    if n * n > 2.6e7:
        return None
    b = np.convolve(x, x)
    M = float(b.max())
    rows = (np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)).astype(np.int64)
    cols = np.repeat(np.arange(n), n).astype(np.int64)
    data = np.tile(x.astype(np.float64), n)
    A = csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M * cp.ones(2 * n - 1)])
    for s in cp.installed_solvers():
        if time.monotonic() > deadline - 90.0:

        try:

            g2 = np.clip(np.asarray(g.value, dtype=np.float64), 0.0, None)
            if g2.sum() <= 1e-9:
                continue
            return g2 / g2.sum() * x.sum()
        except Exception:
            continue
    return None
