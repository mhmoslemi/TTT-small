import numpy as np
import time


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _conv2(x, y, L):
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    return np.fft.irfft(X * Y, L)


def _maxb(a):
    n = a.size
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)
    return b[:2 * n - 1]
