import numpy as np

def _next_pow2(m):
    return 1 << int(np.ceil(np.log2(max(2, m)))) if m > 1 else 2


def _fft_autocorr(a):
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    ac = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return ac


def _fft_score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * _fft_autocorr(a).max() / (s * s))


def _lse_grad(a, beta, N2, N3):

    ac = np.fft.irfft(fa * fa, N2)[:2 * n - 1]
    m = ac.max()
    e = np.exp(beta * (ac - m))
    p = e / e.sum()

    g = np.fft.irfft(f, N3)[n - 1:2 * n - 1]
    return ac, g
