import time

try:

_LCACHE = {}


def _fft_len(n):
    n = int(n)
    if n <= 1:
        return 1
    if n not in _LCACHE:
        _LCACHE[n] = 1 << (2 * n - 2).bit_length()
    return _LCACHE[n]


def _fft_conv(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n <= 1:
        return np.array([float(x[0] * x[0]) if n else 0.0], dtype=float)
    L = _fft_len(n)
    fx = np.fft.rfft(x, L)
    return np.fft.irfft(fx * fx, n=L)[:2 * n - 1]


def _fast_score(p):
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return np.inf
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not np.isfinite(s) or s < 1e-8:
        return np.inf
    q = _fft_conv(p)
    m = float(q.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * p.size * max(0.0, m) / (s * s))


def _normalize(p, floor=1e-300):
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return np.ones(1, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, floor, 1000.0)
    s = float(p.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.ones(p.size, dtype=np.float64) / float(p.size)
    return p / s


def _p_from_w(w):
    w = np.asarray(w, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    if w.size == 0:
        return np.ones(1, dtype=np.float64)
    w = w - float(w.max())
    w = np.clip(w, -700.0, 700.0)
    p = np.exp(w)
    s = float(p.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    return p / s


def _exact_score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return np.inf

        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:

def _heuristic_p(n, rng):
    out = []
    idx = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1)

    out.append(np.ones(n, dtype=np.float64) / float(n))

    for _ in range(6):
        w = rng.normal(0.0, 1.0, n)
        w -= w[::-1]
        out.append(_p_from_w(w * (0.2 + 1.2 * rng.random())))

    for _ in range(4):
        out.append(_p_from_w(rng.normal(0.0, 0.7, n)))

    if n > 2:
        theta = 2.0 * np.pi * idx / float(n - 1)
        for m in (1, 2, 3, 4, 5):
            for a in (0.5, 1.0, 1.6):
                out.append(_p_from_w(a * np.sin(m * theta)))
                out.append(_p_from_w(a * np.sin(m * theta) + 0.4 * a * np.sin((m + 1) * theta)))
        for a in (0.4, 0.8, 1.2):
            out.append(_p_from_w(a * (idx - c) / max(1.0, float(n))))
    return out


def _choose_freqs(n):
    if n <= 4:
        return []
    base = list(range(1, 17))
    base += [18, 21, 25, 30, 36, 43, 51, 61, 73, 87, 104, 124, 147, 174, 206, 243, 289]
    res = [m for m in dict.fromkeys(base) if 0 < m < n // 2]
    return res[:16]


def _make_basis(n):
    freqs = _choose_freqs(n)
    if not freqs:
        return np.zeros((n, 0), dtype=np.float64)
    theta = 2.0 * np.pi * np.arange(n, dtype=np.float64) / float(n - 1)
    cols = []
    for i, m in enumerate(freqs):
        mt = m * theta
        cols.append(np.sin(mt))
        if i < 6:
            cols.append(np.cos(mt))
    return np.column_stack(cols if cols else [np.ones(n, dtype=np.float64)])


def _p_from_a(B, a):
    if B.size == 0 or a.size == 0:
        return np.ones(B.shape[0], dtype=np.float64) / max(1, B.shape[0])
    w = B.dot(a)
    return _p_from_w(w)


def _regress(B, p):
    if B.size == 0 or B.shape[1] == 0:
        return np.zeros(0, dtype=np.float64)
    logp = np.log(np.clip(p, 1e-300, 1000.0))
    logp -= float(logp.mean())
    try:
        return np.linalg.lstsq(B, logp, rcond=None)[0]
    except Exception:
        return np.zeros(B.shape[1], dtype=np.float64)


def _param_obj(B, mu):
    def fun(a):
        w = B.dot(a)
        w = w - float(w.max())
        p = np.exp(np.clip(w, -700.0, 700.0))
        s = float(p.sum())
        if not np.isfinite(s) or s < 1e-12:
            return 1e100, np.zeros(a.size, dtype=np.float64)
        p /= s
        q = _fft_conv(p)
        q = np.maximum(q, 0.0)
        m = float(q.max())
        if m <= 1e-300:
            return 0.0, np.zeros(a.size, dtype=np.float64)
        mu_use = float(mu) if mu > 0.0 else m / 25.0
        qt = q / mu_use
        qm = float(qt.max())
        val = mu_use * (qm + np.log(float(np.sum(np.exp(qt - qm)))))
        wt = np.exp(qt - qm)
        ws = float(wt.sum())
        if ws <= 0.0:
            wt = np.full(q.size, 1.0 / q.size)
        else:
            wt /= ws
        gp = 2.0 * np.correlate(wt, p, mode="valid")
        g = p * (gp - float(np.dot(p, gp)))
        grad = B.T.dot(g)
        return float(val), np.clip(grad, -1e6, 1e6)
    return fun


def _local_refine(p0, deadline, rng):
    p0 = _normalize(p0)
    n = p0.size
    best = p0.copy()
    best_sc = _fast_score(best)
    cur = best.copy()

    step = 0.9
    temp_factor = 30.0
    bad = 0
    it = 0

    while time.monotonic() < deadline:
        if bad > 500:
            break
        q = _fft_conv(cur)
        q = np.maximum(q, 0.0)
        m = float(q.max())
        if not np.isfinite(m) or m <= 1e-300:
            break

        tau = max(m / temp_factor, 1e-15)
        ex = np.exp((q - m) / tau)
        es = float(ex.sum())
        if not np.isfinite(es) or es <= 0.0:
            break
        ex /= es

        gp = 2.0 * np.correlate(ex, cur, mode="valid")
        gp -= float(np.dot(cur, gp))
        scale = float(np.max(np.abs(gp)))

        improved = False
        if scale > 1e-14:
            for alpha in (step, 0.5 * step, 0.25 * step, 0.1 * step, 0.02, 0.005):
                fac = np.exp(-alpha * gp / scale)
                pn = cur * fac
                s = float(pn.sum())
                if not np.isfinite(s) or s < 1e-12:
                    continue
                pn /= s
                sc = _fast_score(pn)
                if np.isfinite(sc) and sc < best_sc - 1e-12:
                    best = pn.copy()
                    best_sc = sc
                    cur = pn.copy()
                    improved = True
                    bad = 0
                    step = min(1.3, step * 1.05)
                    break

        if not improved:
            bad += 1
            step = max(0.03, step * 0.75)
            temp_factor = max(3.0, temp_factor * 0.97)

            if bad % 25 == 0:
                pert = 0.01 + 0.08 * rng.random()
                cur = _normalize(best * (1.0 + pert * rng.normal(0.0, 1.0, n)))
                sc = _fast_score(cur)
                if np.isfinite(sc) and sc < best_sc:
                    best = cur.copy()
                    best_sc = sc

                if n > 1:
                    w = rng.normal(0.0, 1.0, n)
                    w -= w[::-1]
                    tilt = _p_from_w(0.6 * w)
                    mix = 0.35 + 0.5 * rng.random()
                    mp = _normalize((1.0 - mix) * best + mix * tilt)
                    sc = _fast_score(mp)
                    if np.isfinite(sc) and sc < best_sc:
                        best = mp.copy()
                        best_sc = sc
                        cur = best.copy()

        it += 1

    return _normalize(best), best_sc

                best = pn.copy()
                best_sc = sc
                improved = True
                break

                sc = _fast_score(pn)

                    best = pn.copy()
                    best_sc = sc
                    improved = True
                    break

            if not improved:

                sc = _fast_score(pn)
                if np.isfinite(sc) and sc < best_sc:
                    best = pn.copy()
                    best_sc = sc
                else:
                    break

    return _normalize(best), _fast_score(best)


def _full_log_optimize(p0, deadline, mu_factor=30):
    p0 = _normalize(p0)
    n = p0.size
    if n < 3 or time.monotonic() >= deadline:
        return p0, _fast_score(p0)

    base_sc = _fast_score(p0)
    mu = max((base_sc / (2.0 * n)) / float(mu_factor), 1e-14)

    def fun(a):
        w = np.empty(n, dtype=np.float64)
        w[0] = 0.0
        w[1:] = a
        w -= float(w.mean())
        w = w - float(w.max())
        p = np.exp(np.clip(w, -700.0, 700.0))
        s = float(p.sum())
        if not np.isfinite(s) or s < 1e-12:
            return 1e100, np.zeros(a.size, dtype=np.float64)
        p /= s
        q = _fft_conv(p)
        q = np.maximum(q, 0.0)
        m = float(q.max())
        if m <= 1e-300:
            return 0.0, np.zeros(a.size, dtype=np.float64)
        qt = q / mu
        qm = float(qt.max())
        val = mu * (qm + np.log(float(np.sum(np.exp(qt - qm)))))
        wt = np.exp(qt - qm)
        ws = float(wt.sum())
        if ws <= 0.0:
            wt = np.full(q.size, 1.0 / q.size)
        else:
            wt /= ws
        gp = 2.0 * np.correlate(wt, p, mode="valid")
        g = p * (gp - float(np.dot(p, gp)))
        return float(val), np.clip(g[1:], -1e6, 1e6)

    a0 = np.zeros(n - 1, dtype=np.float64)
    if p0[0] > 1e-100:
        a0 = np.log(p0[1:] / p0[0])
    a0 = np.clip(a0, -3.0, 3.0)
    a0 -= float(a0.mean())

    maxiter = 25 if n > 1500 else 70
    try:
        res = minimize(fun, a0, jac=True, method="L-BFGS-B",
                       bounds=[(-4.0, 4.0)] * (n - 1),
                       options={"maxiter": int(maxiter), "maxls": 20, "gtol": 1e-6, "ftol": 1e-9})
    except Exception:
        res = None

    if res is not None and getattr(res, "x", None) is not None:
        w = np.empty(n, dtype=np.float64)
        w[0] = 0.0
        w[1:] = np.asarray(res.x, dtype=np.float64)
        w -= float(w.mean())
        w = w - float(w.max())
        p = np.exp(np.clip(w, -700.0, 700.0))
        p = _normalize(p)
        return p, _fast_score(p)
    return p0, base_sc


def _de_optimize(n, time_alloc, rng):
    n = int(n)
    if n < 2:
        return np.ones(1, dtype=np.float64), 2.0
    start = time.monotonic()
    deadline = min(start + max(0.0, float(time_alloc)), start + 1.0)

    def fun(a):
        if time.monotonic() > deadline:
            return 1e100
        p = _p_from_w(a)
        sc = _fast_score(p)
        return sc if np.isfinite(sc) else 1e100

    best_p = None
    best_sc = np.inf

    if differential_evolution is None:
        for _ in range(4000):
            if time.monotonic() >= deadline - 2.0:
                break
            a = rng.normal(0.0, 1.0, n)
            p = _p_from_w(a)
            sc = _fast_score(p)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p.copy()
        if best_p is not None and time.monotonic() < deadline - 1.0:
            best_p, best_sc = _local_refine(best_p, deadline, rng)
        return best_p if best_p is not None else np.ones(n) / n, best_sc

    try:
        maxiter = int(max(60, min(800, time_alloc * 2.0)))
        res = differential_evolution(fun, [(-3.0, 3.0)] * n,
                                     seed=int(rng.integers(0, 2**31 - 1)),
                                     maxiter=maxiter, popsize=18,
                                     tol=1e-5, polish=False)
        a = np.asarray(res.x, dtype=np.float64)
        p = _p_from_w(a)
        sc = _fast_score(p)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = p.copy()
    except Exception:
        pass

    if best_p is None:
        best_p = np.ones(n, dtype=np.float64) / n
        best_sc = _fast_score(best_p)

    if time.monotonic() < deadline - 1.0:
        best_p, best_sc = _local_refine(best_p, deadline, rng)
    return best_p, best_sc


def _optimize_length(n, time_alloc, rng, initial=None, global_deadline=None, on_candidate=None):
    n = int(n)
    if n < 2:
        p = np.ones(1, dtype=np.float64)
        if on_candidate:
            on_candidate(p)
        return p, 2.0

    start = time.monotonic()
    alloc = max(0.0, float(time_alloc))
    deadline = min(start + alloc, float(global_deadline) if global_deadline is not None else start + alloc)

    best_box = [np.inf, np.ones(n, dtype=np.float64) / float(n)]
    cands = []

    def add(p):
        p = np.asarray(p, dtype=np.float64)
        if p.size != n or not np.all(np.isfinite(p)):
            return
        p = _normalize(p)
        sc = _fast_score(p)
        if not np.isfinite(sc):
            return
        if sc < best_box[0] - 1e-14:
            best_box[0] = sc
            best_box[1] = p.copy()
        if len(cands) < 32:
            cands.append((sc, p.copy()))
        else:
            idx = max(range(len(cands)), key=lambda i: cands[i][0])
            if sc < cands[idx][0] - 1e-14:
                cands[idx] = (sc, p.copy())

    for h in _heuristic_p(n, rng):
        add(h)
    if initial is not None:
        add(initial)

    B = _make_basis(n)
    if B.size and time.monotonic() < deadline - 10.0:
        a_reg = _regress(B, best_box[1])
        spawns = []
        if a_reg.size and float(np.linalg.norm(a_reg)) > 1e-7:
            spawns.append((np.clip(a_reg, -2.5, 2.5), (30.0, 12.0)))

        rand_n = int(min(8, max(2, alloc // 40)))
        for _ in range(rand_n):
            a = rng.normal(0.0, 0.8, B.shape[1])
            spawns.append((np.clip(a, -2.5, 2.5), (25.0,)))

        if B.shape[1] > 0:
            for j in range(min(4, B.shape[1])):
                for val in (0.7, -0.7, 1.2, -1.2):
                    a = np.zeros(B.shape[1], dtype=np.float64)
                    a[j] = val
                    spawns.append((a, (20.0,)))

        for a0, mu_factors in spawns[:14]:
            if time.monotonic() >= deadline - 20.0:
                break
            add(_p_from_a(B, a0))
            for mf in mu_factors:
                if time.monotonic() >= deadline - 20.0:
                    break
                mu = max((best_box[0] / (2.0 * n)) / float(mf), 1e-13)
                try:
                    res = minimize(_param_obj(B, mu), a0, jac=True,
                                   method="L-BFGS-B",
                                   bounds=[(-2.5, 2.5)] * B.shape[1],
                                   options={"maxiter": 70, "maxls": 20, "gtol": 1e-6, "ftol": 1e-9})
                    if res is not None and getattr(res, "x", None) is not None:
                        add(_p_from_a(B, np.clip(res.x, -2.5, 2.5)))
                        add(_p_from_a(B, np.clip(0.5 * res.x, -2.5, 2.5)))
                except Exception:
                    pass

    if cands and time.monotonic() < deadline - 20.0:
        cands.sort(key=lambda t: t[0])
        top = []
        for sc, p in cands:
            if len(top) >= 4:
                break
            if all(float(np.linalg.norm(p - q)) > 0.015 for _, q in top):
                top.append((sc, p))
        for _, p in top:
            if time.monotonic() >= deadline - 15.0:
                break
            lt = min(240.0, max(10.0, (deadline - time.monotonic()) * 0.4))
            if lt < 5.0:
                break
            p2, _ = _local_refine(p, time.monotonic() + lt, rng)
            add(p2)

    if time.monotonic() < deadline - 20.0:
        p3, _ = _lp_improve(best_box[1], min(deadline - 18.0, time.monotonic() + 90.0), rng,
                            K=min(100, max(30, n // 10)))
        add(p3)

    if n <= 1024 and time.monotonic() < deadline - 30.0:
        p4, _ = _full_log_optimize(best_box[1], min(deadline - 25.0, time.monotonic() + 80.0), mu_factor=20)
        add(p4)

    if time.monotonic() < deadline - 5.0:
        p5, _ = _local_refine(best_box[1], deadline, rng)
        add(p5)

    if on_candidate is not None:
        cands.sort(key=lambda t: t[0])
        seen = []
        cnt = 0
        for _, p in cands:
            if cnt >= 3:
                break
            if all(float(np.linalg.norm(p - q)) > 0.02 for q in seen):
                on_candidate(p)
                seen.append(p)
                cnt += 1
        on_candidate(best_box[1])

    return _normalize(best_box[1]), best_box[0]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1.0

    t0 = time.monotonic()
    deadline = t0 + max(0.0, budget - 12.0)
    inner_deadline = deadline - max(5.0, 0.02 * budget)
    rng = np.random.default_rng(seed_int)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            parent = None
    except Exception:
        parent = None

    best_seq = None
    best_exact = np.inf

    def update_best(p):
        nonlocal best_seq, best_exact
        p = _normalize(p)
        sc = _exact_score(p)
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_seq = p.copy()

    if parent is not None:
        update_best(parent)
    if best_seq is None:
        update_best(np.ones(1, dtype=np.float64))

    if differential_evolution is not None and inner_deadline - time.monotonic() > 90.0:
        de_n = 64 if parent is not None and parent.size > 0 else 128
        de_time = min(120.0, max(20.0, 0.12 * (inner_deadline - time.monotonic())))
        try:
            pd, _ = _de_optimize(de_n, de_time, rng)
            if pd is not None:
                update_best(pd)
        except Exception:
            pass

    if (parent is not None and parent.size > 0 and parent.size <= 5000
            and inner_deadline - time.monotonic() > 60.0):
        n_parent = int(parent.size)
        rem = max(0.0, inner_deadline - time.monotonic())
        n_small = int(max(64, min(512, n_parent // 10)))
        small_time = 0.25 * rem
        if small_time > 20.0:
            try:
                ps, _ = _optimize_length(n_small, small_time, rng, initial=None,
                                         global_deadline=inner_deadline, on_candidate=update_best)
                update_best(ps)
            except Exception:
                pass

        rem = max(0.0, inner_deadline - time.monotonic())
        if rem > 20.0:
            try:
                pp, _ = _optimize_length(n_parent, rem, rng, initial=parent,
                                         global_deadline=inner_deadline, on_candidate=update_best)
                update_best(pp)
            except Exception:
                pass
    elif inner_deadline - time.monotonic() > 60.0:
        rem = max(0.0, inner_deadline - time.monotonic())
        init = best_seq if (best_seq is not None and best_seq.size == 512) else None
        try:
            p1, _ = _optimize_length(512, 0.5 * rem, rng, initial=init,
                                     global_deadline=inner_deadline, on_candidate=update_best)
            update_best(p1)
        except Exception:
            pass
        rem = max(0.0, inner_deadline - time.monotonic())
        if rem > 30.0:
            init = best_seq if (best_seq is not None and best_seq.size == 512) else None
            try:
                p2, _ = _optimize_length(512, rem, rng, initial=init,
                                         global_deadline=inner_deadline, on_candidate=update_best)
                update_best(p2)
            except Exception:
                pass

    if (best_seq is not None and best_seq.size <= 5000
            and deadline - time.monotonic() > 45.0):
        try:
            pf, _ = _local_refine(best_seq, deadline - 8.0, rng)
            update_best(pf)
        except Exception:
            pass

    if (best_seq is not None and best_seq.size <= 5000
            and deadline - time.monotonic() > 45.0):
        try:
            pl, _ = _lp_improve(best_seq, min(deadline - 5.0, time.monotonic() + 60.0), rng,
                                K=min(80, max(40, best_seq.size // 10)))
            update_best(pl)
        except Exception:
            pass

    update_best(best_seq if best_seq is not None else np.ones(1, dtype=np.float64))

    if best_seq is None or best_seq.size == 0 or not np.all(np.isfinite(best_seq)):
        best_seq = np.ones(1, dtype=np.float64)

    out = [float(x) for x in best_seq]
    if _exact_score(out) != np.inf:
        return out
    return [1.0]
