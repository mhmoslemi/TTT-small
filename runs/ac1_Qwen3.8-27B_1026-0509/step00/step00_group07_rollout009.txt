import time

def _optimize(a, n, deadline, n_iter=None):
    """Flatten autconvolution via subgradient descent."""
    best_score = np.inf
    best_a = a.copy()
    
    if n_iter is None:
        n_iter = 50000 if n <= 64 else 10000 if n <= 512 else 3000 if n <= 4096 else 1000
    
    pad = 1
    while pad < 2*n - 1:
        pad *= 2
    
    lr = 0.5
    stale = 0
    
    for it in range(n_iter):
        if time.monotonic() > deadline - 3:
            break
        
        A = np.fft.rfft(a, pad)
        B = np.fft.irfft(A * A, pad)[:2*n-1]
        
        k_star = int(np.argmax(B))
        max_b = B[k_star]
        score = 2.0 * n * max_b  # S=1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
        
        # Subgradient update
        j_min = max(0, k_star - n + 1)
        j_max = min(n-1, k_star)
        idx = np.arange(j_min, j_max + 1)
        a[idx] -= lr * a[k_star - idx]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:

            s = 1.0
        a /= s

            lr *= 0.7
            stale = 0
    
    return best_a, best_score
