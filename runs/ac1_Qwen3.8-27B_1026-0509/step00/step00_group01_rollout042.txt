```python
def minimize_ac1(a, n_iter=10000, tau0=1.0, tau_mult=1.1, eta0=0.1):
    n = len(a)
    a = a / sum(a)  # normalize to sum = 1
    a_rev = a[::-1].copy()
    
    for i in range(n_iter):
        tau = tau0 * (tau_mult ** i)
        
        # Compute convolution
        c = np.convolve(a, a)  # or use FFT
        c_max = c.max()
        
        # Softmax weights (numerically stable)
        log_w = tau * c - tau * c_max
        w = np.exp(log_w)
        w /= w.sum()
        
        # Gradient: 2 * (w convolved with a_rev), appropriate slice
        # w has length 2n-1, a_rev has length n
        # conv = np.convolve(w, a_rev) has length 3n-2
        # We want elements (n-1) to (2n-2)
        conv_grad = np.convolve(w, a_rev)
        grad = 2.0 * conv_grad[n-1:2*n-1]
        
        # Gradient step with line search or fixed step
        eta = eta0 / (1 + i * 0.001)
        a_new = a - eta * grad
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-10:
            break
        
        a = a_new
    
    return a
```

The projection onto the simplex: given x, find the closest point to x in {y ≥ 0, sum(y) = 1} (in Euclidean distance). This is: y = max(x - τ, 0) where τ is chosen so that sum(y) = 1.

The standard algorithm (by Duchi et al.):
1. Sort x in decreasing order: x_(1) ≥ x_(2) ≥ ... ≥ x_(n)
2. Find the largest k such that x_(k) - (sum_{i=1}^k x_(i))/(k) + 1/k > 0
3. τ = (sum_{i=1}^k x_(i) - 1)/k
4. y = max(x - τ, 0)

This is O(n log n) due to sorting.

```python
import time
from scipy.fft import fft, ifft

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Get parent sequence
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    
    # Normalize to sum = 1
    s = parent.sum()
    if s < 0.01:
        parent = np.ones(n)
        s = n
    a = parent / s
    
    best = a.copy()
    best_score = float(evaluate_sequence((best * 100).tolist()))  # scale up for safety
    
    # FFT sizes
    L_conv = 1
    while L_conv < 2*n - 1:
        L_conv *= 2
    L_grad = 1
    while L_grad < 3*n - 2:
        L_grad *= 2
    
    def smooth_step(a, tau, eta):
        n = len(a)
        a_rev = a[::-1]
        
        # Compute c = a * a using FFT
        A = fft(a, L_conv)
        C = ifft(A * A)
        c = C[:2*n-1].real
        
        # Softmax weights (numerically stable)
        c_max = c.max()
        log_w = tau * (c - c_max)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= w_sum
        
        # Gradient: 2 * (w * a_rev)[n-1 : 2n-1]
        W = fft(w, L_grad)
        AR = fft(a_rev, L_grad)
        GRAD = ifft(W * AR)
        grad = 2.0 * GRAD[n-1:2*n-1].real
        
        # Gradient step
        a_new = a - eta * grad
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        
        return a_new, c.max()
    
    def project_simplex(v):
        # Project v onto {x >= 0, sum(x) = 1}
        n = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.where(u * np.arange(1, n+1) + 1 > cssv)[0][-1]
        theta = (cssv[rho-1] - 1) / rho
        w = np.maximum(v - theta, 0)
        return w

    tau = 5.0
    tau_mult = 1.05
    eta = 0.5
    no_improve = 0
    
    while time.monotonic() < deadline:
        a, c_max = smooth_step(a, tau, eta)
        
        # Evaluate
        score = float(evaluate_sequence((a * 100).tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
```
