The matrix C has the structure:

```python
def build_conv_matrix(a):
    n = len(a)
    nnz_per_col = min(n, 2*n-1)  # actually varies per column
    # Column j: n nonzeros at rows j to j+n-1
    # Total nonzeros: sum over j of min(n, 2n-1-j) = sum_{j=0}^{n-1} min(n, 2n-1-j)
    # = sum_{j=0}^{n-1} n for j <= n-1 (since 2n-1-j >= n for j <= n-1)
    # = n^2
    # Wait: for column j, rows are j, j+1, ..., j+n-1. The last one is j+n-1.
    # For this to be <= 2n-2: j+n-1 <= 2n-2, i.e., j <= n-1. Always true.
    # So each column has exactly n nonzeros. Total: n^2.
    
    rows = np.concatenate([np.arange(j, j+n) for j in range(n)])
    cols = np.concatenate([np.full(n, j) for j in range(n)])
    vals = np.repeat(a, n)  # a[j] repeated n times for column j
    
    C = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C
```

```python
import numpy as np
import time
from scipy.sparse import csc_matrix
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(10, budget_s) - 5
    
    # Choose n for the LP-based optimization
    n = 1000
    
    # Initialize: raised cosine
    x = np.arange(n) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.clip(a, 1e-10, None)  # keep positive for numerical stability
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Ensure valid
    if not np.isfinite(best_score) or best_score > 2.0:
        a = np.ones(n)
        best = a.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    
    blend = 0.5
    
    for it in range(500):
        if time.monotonic() > deadline - 10:
            break
        
        # Compute M
        c = np.convolve(a, a)
        M = c.max()
        S = a.sum()
        
        # Build sparse constraint matrix
        rows = np.concatenate([np.arange(j, j+n) for j in range(n)])
        cols = np.concatenate([np.full(n, j, dtype=int) for j in range(n)])
        vals = np.repeat(a, n)
        C = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # Solve LP: max sum(b) s.t. C@b <= M, b >= 0
        t_lp_start = time.monotonic()
        try:
            res = linprog(-np.ones(n), A_ub=C, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': deadline - time.monotonic() - 15})
        except:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.maximum(res.x, 0)
        if b.sum() < 1e-10:
            break
        
        # Rescale to match sum
        g = b * (S / b.sum())
        
        # Blend
        a = (1 - blend) * a + blend * g
        
        # Evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
        
        # Decrease blend
        blend = max(0.1, blend * 0.99)
    
    # Fine-tune with local perturbations
    while time.monotonic() < deadline - 2:
        # Random perturbation
```
