```python
def objective_and_grad(v, epsilon, p, b):
    """Compute smooth objective and its gradient w.r.t. p, then convert to w.r.t. v."""
    # v is the unconstrained parameter, p = softmax(v)
    # b = convolve(p, p)
    bmax = b.max()
    exp_terms = np.exp((b - bmax) / epsilon)
    log_sum = np.log(exp_terms.sum())
    obj = bmax + epsilon * log_sum  # this is the smooth max = (1/eps)*log-sum-exp scaled
    
    w = exp_terms / exp_terms.sum()  # softmax weights over lags
    
    # Gradient w.r.t. p: grad_p[j] = 2 * sum_k w[k] * p[k-j] for valid k-j
    # = 2 * cross_correlation(p, w)[j]
    # Using FFT: conv(p_rev, w)[n-1+j]
    p_rev = p[::-1]
    # conv_len for p (length n) and w (length 2n-1): need n + (2n-1) - 1 = 3n-2
    g_fft_len = next_fast_len(3*n)
    P_rev = fft(p_rev, g_fft_len)
    W = fft(w, g_fft_len)
    conv_result = np.real(ifft(P_rev * W))
    grad_p = 2.0 * conv_result[n-1:2*n-1]
    
    # Convert to gradient w.r.t. v (softmax chain rule):
    # grad_v = p * (grad_p - sum(p * grad_p))
    mean_gp = np.sum(p * grad_p)
    grad_v = p * (grad_p - mean_gp)
    
    return obj, grad_v
```

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def _softmax(v):
    v = v - v.max()
    e = np.exp(v)
    return e / e.sum()

def _do_conv(x, conv_len, b_len):
    X = fft(x, conv_len)
    return np.real(ifft(X * X)[:b_len])

def _get_score(x, n, conv_len, b_len):
    b = _do_conv(x, conv_len, b_len)
    s = x.sum()
    return 2.0 * n * b.max() / (s * s)

def _topk_grad(p, b, K, n):
    b_len = b.size
    if K >= b_len:
        top_idx = np.arange(b_len)
        w = 1.0 / b_len
    else:
        top_idx = np.argpartition(b, -K)[-K:]
        top_idx = top_idx[np.argsort(b[top_idx])[::-1]]
        w = 1.0 / K
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] += p[k - idx] * w
    return 2.0 * g

        b = _do_conv(p, conv_len, b_len)
        g = _topk_grad(p, b, cur_K, n)
        mean_g = np.dot(p, g)

        v_hat = vel / (1 - b2 ** t)
        v -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        p_new = _softmax(v)
        s_new = _get_score(p_new, n, conv_len, b_len)
        if s_new < best_local_s:
            best_local = p_new.copy()
            best_local_s = s_new
            no_imp = 0
            if t % 500 == 0:
                cur_K = max(5, cur_K - 3)
        else:
            no_imp += 1

        if no_imp > 3000:
            v = np.log(np.maximum(best_local, 1e-30))
            v -= v.max()
            v += 0.03 * rng.standard_normal(n)
            m[:] = 0
            vel[:] = 0
            t = 0
            no_imp = 0
            cur_K = K
            lr = min(0.5, lr * 1.1)

    return best_local, best_local_s

def _polish(p_best, n, conv_len, b_len, rng, deadline):
    best = p_best.copy()
    best_s = _get_score(best, n, conv_len, b_len)
    width = max(1, n // 100)
    while time.monotonic() < deadline:
        candidate = best.copy()
        start = rng.integers(0, n - width)
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.05))
        candidate = np.clip(candidate, 0, 1000)
        if candidate.sum() < 0.01:
            continue
        candidate /= candidate.sum()
        s = _get_score(candidate, n, conv_len, b_len)
        if s < best_s:
            best = candidate
            best_s = s
    return best, best_s

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0
    rng = np.random.default_rng(seed)

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        p = np.clip(p, 0.0, 1000.0)
        n = p.size
    except Exception:
        n = 6733
        p = np.ones(n, dtype=np.float64)
    if p.size != n or n < 10 or p.sum() < 0.01:
        p = np.ones(n, dtype=np.float64)
    p = p / p.sum()

    conv_len = next_fast_len(2 * n - 1)
    b_len = 2 * n - 1

    best_p = p.copy()
    best_s = _get_score(p, n, conv_len, b_len)

    j = np.arange(n)
    inits = [
        p.copy(),
        np.ones(n) / n,
        _softmax(0.3 * rng.standard_normal(n)),
        _softmax(0.1 * np.cos(2 * np.pi * j / n)),
    ]
    bump = np.cos(np.pi * (j / n - 0.5)) ** 2
    inits.append(np.maximum(bump, 1e-10) / np.maximum(bump, 1e-10).sum())

    phase1_time = (budget_s - 20.0) * 0.7
    per_init = phase1_time / len(inits)

    for ii, ip in enumerate(inits):
        if time.monotonic() >= t0 + phase1_time:
            break
        end_i = min(t0 + (ii + 1) * per_init, t0 + phase1_time)
        rp, rs = _optimize(ip, n, conv_len, b_len, rng, end_i, K=30, lr=0.1)
        if rs < best_s:
            best_p = rp
            best_s = rs

    # Phase 2: refine with smaller K
    remaining = deadline - time.monotonic()
    if remaining > 30:
        end2 = t0 + phase1_time + remaining * 0.5
        rp, rs = _optimize(best_p, n, conv_len, b_len, rng, end2, K=10, lr=0.05)
        if rs < best_s:
            best_p = rp
            best_s = rs

    # Phase 3: polish with local perturbations
    remaining = deadline - time.monotonic()
    if remaining > 15:
        polish_deadline = t0 + budget_s - 12.0
        rp, rs = _polish(best_p, n, conv_len, b_len, rng, polish_deadline)
        if rs < best_s:
            best_p = rp
            best_s = rs

    result = [float(x) for x in best_p]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [float(x) for x in p]
        fs = evaluate_sequence(result)
    return result
