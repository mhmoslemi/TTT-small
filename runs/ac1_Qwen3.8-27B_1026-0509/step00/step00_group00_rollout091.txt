def _norm1(a):
    """Return a finite, nonnegative, sum-1 array, or None."""
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s


def _score_general(a):
    """FFT score for arbitrary-length nonnegative a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    need = 2 * n - 1
    sz = 1 << (need - 1).bit_length() if need > 1 else 1
    fa = np.fft.rfft(a, sz)
    c = np.fft.irfft(fa * fa.conj(), sz)[:need]
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return 2.0 * n * m / (s * s)
