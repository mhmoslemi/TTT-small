import numpy as np
import time


def _direct_score(a):
    b = np.convolve(a, a)
    S = a.sum()
    if S < 1e-9:
        return float("inf")
    return float(2.0 * a.size * b.max() / (S * S))


def _fam(n, kind, rng):
    i = np.arange(n, dtype=np.float64)
    if kind == "ramp":
        a = i + 1.0
    elif kind == "ramp0":
        a = i + 0.3
    elif kind == "dramp":
        a = (n - i) + 1.0
    elif kind == "exp":
        beta = rng.uniform(0.25, 0.7)
        a = np.exp(beta * i / n)
        a = a + a[::-1] * rng.uniform(0.0, 0.3)
    elif kind == "poly":
        p = rng.uniform(0.5, 1.6)
        a = (i + 1.0) ** p
    elif kind == "half":
        a = 1.0
        m = n // 2
        a[m:] *= rng.uniform(0.2, 0.8)
    else:
        a = 1.0 + 0.5 * rng.normal(size=n)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-9:
        a = np.ones(n)
    return a
