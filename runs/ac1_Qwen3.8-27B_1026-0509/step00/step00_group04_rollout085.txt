def _npl(m):
    L = 1
    if m > 1:
        L = 1 << (int(m) - 1).bit_length()
    return max(2, L)


def _norm(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s <= 1e-300 or p.size == 0:
        return np.ones(max(1, p.size), dtype=np.float64) / max(1, p.size)
    return p / s


def _fft_score(p, n, Lc):
    f = np.fft.rfft(p, Lc)
    c = np.fft.irfft(f * f, Lc)
    c = c[: 2 * n - 1]
    np.maximum(c, 0.0, c)
    return float(2.0 * n * c.max()), c


def _grad_fft(r, p, n, Lg):
    R = 2 * n - 1
    conv = np.fft.irfft(
        np.fft.rfft(r[::-1], Lg) * np.fft.rfft(p, Lg),
        Lg,
    )
    v = conv[R - 1 - np.arange(n)]
    np.maximum(v, 0.0, v)
    return 2.0 * v

            if np.isfinite(sc) and sc < global_score:
                global_p = p_best
                global_score = sc
            if global_score <= 1.50274 * 1.0005:
                break

    if time.monotonic() < deadline - 25:
        rem = deadline - time.monotonic()
        phase = min(deadline - 12.0, time.monotonic() + max(20.0, 0.65 * rem))
        rng = np.random.default_rng(seed * 7 + 991)
        p_best, _ = _optimize(global_p.copy(), n, TARGET_FACTOR, phase, rng)
        sc = _eval_scaled(p_best, 10.0)
        if np.isfinite(sc) and sc < global_score:
            global_p = p_best
            global_score = sc

    final_seq = [float(x) for x in global_p]
    fs = _eval_scaled(global_p, 1.0)
    if not np.isfinite(fs):
        final_seq = [float(x) for x in scored[0][1]]
        fs = _eval_scaled(scored[0][1], 1.0)

    if not np.isfinite(fs) or not final_seq:
        fallback = np.ones(n, dtype=np.float64) / n
        final_seq = [float(x) for x in fallback]

    return final_seq
