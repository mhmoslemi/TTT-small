def _proj_simplex(v):
    n = v.size
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    rho = np.arange(1, n + 1)
    idx = np.nonzero(w * rho > (cs[-1] - 1.0))[0]
    if idx.size == 0:
        theta = (cs[-1] - 1.0) / n

        theta = (cs[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _logsumexp_c(C, mu):
    cm = C.max()
    return float(cm + mu * np.log(np.sum(np.exp((C - cm) / mu))))


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_end = time.monotonic() + max(1.0, float(budget_s) - 4.0)

    # ---- initialization (non-constant to break symmetry) ----
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        a = np.ones(6121)
    if a.size == 0 or not np.isfinite(a).all():
        a = np.ones(6121)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if np.ptp(a) < 1e-12 * max(1.0, a.max()):      # uniform-ish -> force non-constant arch
        i = np.arange(n, dtype=np.float64)
        a = 1.0 - (((i / max(1, n - 1)) - 0.5) * 2.0) ** 2
        a = a + 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-9:
        a = np.ones(n)
    a = a / a.sum()

    C0 = np.convolve(a, a)
    cmax0 = float(C0.max())
    best = a.copy()
    best_score = 2.0 * n * cmax0

    # ================= PHASE 1 : FISTA on log-sum-exp max =================
    mu = 0.05 * cmax0
    a_prev = a.copy()
    a_tprev = a.copy()
    t = 1.0
    L = 8.0 / max(mu, 1e-12)
    f_cur = None
    n_fista = 0
    while time.monotonic() < t_end * 0.42 and n_fista < 6000:
        C = np.convolve(a, a)
        f_cur = _logsumexp_c(C, mu)
        p = np.exp((C - C.max()) / mu)
        p /= p.sum()
        g = np.convolve(a, p[::-1], "full")[2 * n - 2:3 * n - 2]  # ~ grad (up to scale)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_prev + ((t - 1.0) / t_new) * (a_prev - a_tprev)

        # backtracking line search
        L_local = L
        for _ in range(8):
            cand = _proj_simplex(y - g / L_local)
            if not np.isfinite(cand).all() or cand.sum() <= 0:
                L_local *= 2.0
                continue
            Cn = np.convolve(cand, cand)
            fn = _logsumexp_c(Cn, mu)
            if fn <= f_cur - 0.5 * L_local * np.dot(cand - a, cand - a) + 1e-12:
                break
            L_local *= 2.0
        L = L_local
        a_tprev = a_prev
        a_prev = cand
        a = cand
        t = t_new
        n_fista += 1

        # score check (cheap-ish, every few iters)
        if n_fista % 25 == 0:
            sc = 2.0 * n * float(np.convolve(a, a).max())
            if sc < best_score:
                best_score = sc
                best = a.copy()
        # anneal mu
        if n_fista % 150 == 0 and mu > 1e-3 * cmax0:
            mu *= 0.75

    # ================= PHASE 2 : cutting-plane LP polish =================
    C = np.convolve(a, a)
    M = float(C.max())
    order = np.argsort(C)[::-1]
    active = [int(k) for k in order[:8]]
    B = 1.0
    Nvars = n

    def make_row(k):
        row = np.zeros(n)
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        row[j0:j1 + 1] = a[k - j1:k - j0 + 1: -1]
        return row

    step_traject = [0.3, 0.25, 0.2, 0.15, 0.12, 0.1]
    it2 = 0
    while time.monotonic() < t_end * 0.95:
        A_ub = np.vstack([make_row(k) for k in active])
        b_ub = M * np.ones(len(active))
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0.0, B)] * n, method="highs",
                          options={"time_limit": max(1.0, (t_end * 0.95 - time.monotonic()) * 0.5)})
        except Exception:
            res = None
        if res is None or not res.success or res.x is None:
            break
        bv = np.asarray(res.x, dtype=np.float64)
        sb = bv.sum()
        if sb < 1e-9:
            break
        bv = bv / sb
        tt = step_traject[min(len(step_traject) - 1, it2 // 40)]
        a = (1.0 - tt) * a + tt * bv
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s <= 0:
            break
        a /= s
        it2 += 1

        C = np.convolve(a, a)
        M = float(C.max())
        sc = 2.0 * n * M
        if sc < best_score:
            best_score = sc
            best = a.copy()

        # refresh active set with current top violators of the peak
        order = np.argsort(C)[::-1]
        new_active = []
        for k in order[:14]:
            if int(k) not in active:
                new_active.append(int(k))
            if len(active) + len(new_active) >= 40:
                break
        active = [int(k) for k in order[:6]] + new_active
        active = list(dict.fromkeys(active))[:40]

    # ================= finalize =================
    # verify the saved best with the authoritative evaluator
    try:
        bs = float(evaluate_sequence([float(x) for x in best]))
    except Exception:
        bs = np.inf
    # also test the live `a`
    try:
        cs = float(evaluate_sequence([float(x) for x in a]))
        if np.isfinite(cs) and cs < bs:
            best, bs = a.copy(), cs
    except Exception:
        pass

    out = [float(x) for x in np.clip(best, 0.0, 1000.0)]
    if not out or not all(np.isfinite(x) for x in out):
        out = [1.0 / n] * n
    return out
