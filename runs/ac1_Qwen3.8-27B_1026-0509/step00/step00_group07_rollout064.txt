import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1)

    # Geometric initialization with near-optimal decay parameter
    t_init = 1.256
    r = np.exp(-t_init / n)
    a = r ** np.arange(n, dtype=np.float64)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Precompute sparsity pattern for convolution matrix A[k,j]=a[k-j]
    j_arr = np.repeat(np.arange(n, dtype=np.int32), n)
    l_arr = np.tile(np.arange(n, dtype=np.int32), n)
    k_arr = j_arr + l_arr
    del j_arr

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    for it in range(300):
        if time.monotonic() > deadline - 60:
            break

        conv = np.convolve(a, a)
        M = float(conv.max())
        a_sum = float(a.sum())
        if a_sum < 0.01 or M <= 0:
            break

        # Build sparse A matrix with current a values
        vals = a[l_arr].astype(np.float64)
        A = sp.csr_matrix((vals, (k_arr, np.repeat(np.arange(n, dtype=np.int32), n))),
                          shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)

        t0 = time.monotonic()
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            # Perturb and continue
            rng = np.random.default_rng(seed + it)
            a = np.clip(a * np.exp(rng.normal(0, 0.02, n)), 0, 1000.0)
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
            continue

        b_opt = np.maximum(res.x, 0.0)
        b_sum = float(b_opt.sum())
        if b_sum < 1e-12:
            break

        b_norm = b_opt * (a_sum / b_sum)
        b_norm = np.clip(b_norm, 0.0, 1000.0)
        b_norm = np.maximum(b_norm, 0.0)

        # Convergence check
        rel_change = float(np.abs(b_norm - a).max()) / max(1.0, float(a.max()))
        a = b_norm

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

        if rel_change < 1e-7:
            break

    # Final validation
    fs = float(evaluate_sequence(best.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    # Ensure nonempty finite list
    if len(best) == 0 or not np.isfinite(best).all():
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
