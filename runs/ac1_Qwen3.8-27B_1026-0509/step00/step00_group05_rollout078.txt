def _lbfgs(x0,n,T,maxcalls,deadline):
    if _minimize is None or T<=0: return None
    x_best=x0.copy(); sc_best=_internal_score(x0)
    calls=0
    class Stop(Exception): pass
    def fun(y):
       nonlocal calls, x_best, sc_best
       calls += 1
       if calls > maxcalls or time.monotonic() >= deadline: raise Stop
       x=_softmax(y)
       if x is None: return 1e100, np.zeros(n)
       c=_conv(x,x)
       cmax=float(c.max())
       if not math.isfinite(cmax) or cmax<=0: return 1e100, np.zeros(n)
       sc=2*n*cmax
       if sc < sc_best:
           sc_best=sc; x_best=x.copy()
       zc=(c-cmax)/T
       zc -= zc.max()
       zc = np.minimum(zc,0.0)
       e=np.exp(zc)
       se=e.sum()
       if not math.isfinite(se) or se<=0:
           # fallback top
           K=max(8, len(c)//100)
           idx=np.argpartition(c,-K)[-K:]
           w=np.zeros_like(c); w[idx]=c[idx]; s=w.sum(); w/=s if s>0 else 1
           F=sc
       else:
           F=cmax + T*math.log(se)
           w=e/se
       g=2.0*_conv(w, x[::-1])
       g=np.where(np.isfinite(g) & (g>0), g, 0.0)
       if not np.any(g>0):
           return F, np.zeros(n)
       dot=float(np.dot(g,x))
       gy=x*(g - dot*x)
       if not np.all(np.isfinite(gy)):
           return 1e100, np.zeros(n)
       return F, gy
    y0=np.log(np.maximum(x0,1e-300))
    try:
       res=_minimize(fun, y0, jac=True, method='L-BFGS', options={'maxiter': max(10, maxcalls//2), 'ftol':1e-9, 'gtol':1e-7, 'maxls':20, 'maxcor':20})
       if res.x is not None:
          xf=_softmax(res.x); if xf is not None and _internal_score(xf)<sc_best: x_best=xf
    except Stop:
       pass
    except Exception:
       pass
    return x_best

def _normalize(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    x = np.maximum(x, 1e-15)
    s = float(x.sum())
    if not math.isfinite(s) or s <= 1e-12:
        x = np.ones_like(x) * 1e-15
        s = float(x.sum())
    x = x / s
    if x.size and float(x.max()) > 1000.0:
        x = x * (1000.0 / x.max())
        x = x / x.sum()
    return x


def _conv(a, b=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64).reshape(-1)
    la, lb = a.size, b.size
    if la == 0 or lb == 0:
        return np.zeros(0, dtype=np.float64)
    target = la + lb - 1
    if la * lb <= 2_000_000:
        c = np.convolve(a, b)
        return np.maximum(c, 0.0)
    L = 1 << (target - 1).bit_length()
    fa = np.fft.rfft(a, L)
    fb = fa if b is a else np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)[:target]
    return np.maximum(c, 0.0)


def _internal_score(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    s = float(x.sum())
    if s <= 0.0 or not math.isfinite(s):
        return float("inf")
    c = _conv(x, x)
    m = float(c.max()) if c.size else 0.0
    if not math.isfinite(m):
        return float("inf")
    return 2.0 * x.size * m / (s * s)


def _conv_slice(w, x, n):
    full = _conv(w, x[::-1])
    return full[n - 1:n - 1 + n]


def _weights_top(c, K):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.ones(1, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    K = max(1, min(int(K), L))
    s = float(c.sum())
    if K >= L or s <= 0.0:
        if s > 0.0:
            return c / s
        return np.ones(L, dtype=np.float64) / float(L)
    idx = np.argpartition(c, -K)[-K:]
    w = np.zeros(L, dtype=np.float64)
    w[idx] = c[idx]
    ws = float(w.sum())
    if ws <= 0.0:
        return c / s if s > 0.0 else np.ones(L, dtype=np.float64) / float(L)
    return w / ws


def _weights_lse(c, factor=0.03):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.ones(1, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    cmax = float(c.max())
    if not math.isfinite(cmax) or cmax <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    T = float(factor) * cmax
    K = max(16, L // 20)
    if T <= 1e-300:
        return _weights_top(c, K)
    z = (c - cmax) / T
    z = z - float(z.max())
    z = np.minimum(z, 0.0)
    e = np.exp(z)
    se = float(e.sum())
    if not math.isfinite(se) or se <= 0.0:
        return _weights_top(c, K)
    return e / se


def _softmax(y):
    y = np.asarray(y, dtype=np.float64).reshape(-1)

            ph = rng.uniform(0.0, 2.0 * np.pi)
            s += amp * np.cos(2.0 * np.pi * k * u + ph)
