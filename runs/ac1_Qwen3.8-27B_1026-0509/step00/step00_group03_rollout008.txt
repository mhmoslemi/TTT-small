from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0
    
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    n = len(a)
    a = a / a.sum()
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    def conv(x):
        return np.convolve(x, x, mode='full')
    
    # Precompute the structure of the constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # We'll build this as a sparse matrix
    
    from scipy.sparse import csr_matrix
    
    t_val = 0.5  # mixing parameter
    
    while time.monotonic() < deadline:
        c = conv(a)
        M = c.max()
        
        # Build constraint matrix A (5069 x 2535)
        # A[k,j] = a[k-j] if 0 <= k-j < n
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            for j in range(max(0, k-n+1), min(n, k+1)):
                rows.append(k)
                cols.append(j)
                data.append(a[k-j])
        
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        
        # LP: max sum(b) s.t. A@b <= M, b >= 0

        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')

        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        b_norm = b / b_sum
        a_new = (1 - t_val) * a + t_val * b_norm
        
        s_new = 2 * n * conv(a_new).max() / a_new.sum()**2
        if s_new < best_score:

            best_score = s_new
        
        a = a_new
        # Decrease t over time
        t_val = max(0.1, t_val * 0.999)
    
    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    if final_score > best_score:
        # This shouldn't happen, but just in case
        best_score = final_score
    
    return [float(x) for x in best]
