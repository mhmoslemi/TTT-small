def _norm_pos(x):
    x = np.asarray(x, dtype=np.float64)
    np.nan_to_num(x, copy=False, nan=0.0, posinf=0.0, neginf=0.0)
    np.maximum(x, 0.0, out=x)
    s = x.sum()
    if not np.isfinite(s) or s <= 1e-12:
        n = x.size
        if n == 0:
            x = np.array([1.0], dtype=np.float64)
            s = 1.0
        else:
            x.fill(1.0 / x.size)
            s = 1.0
    x = x / s
    x = np.maximum(x, 1e-300)
    x /= x.sum()
    return x

def _conv(x):
    n = len(x)
    need = 2 * n - 1
    N = 1 << (need - 1).bit_length()
    X = np.fft.rfft(x, N)
    return np.fft.irfft(X * X, N)[:need]

def _score_fast(x):
    x = np.asarray(x, dtype=np.float64)
    s = x.sum()
    if not np.isfinite(s) or s <= 1e-9:
        return float("inf")
    c = _conv(x)
    m = c.max()
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * len(x) * m / (s * s))

def _top_gradient(x, c, K, rng=None):
    n = len(x)
    m = len(c)
    if K >= m:
        idx = np.arange(m)
        w = np.full(m, 1.0 / m, dtype=np.float64)
    else:
        idx = np.argpartition(c, -K)[-K:]
        ct = c[idx]
        spread = ct.max() - ct.min()
        if spread <= 1e-15:
            w = np.full(K, 1.0 / K, dtype=np.float64)
        else:
            tau = max(1e-12, 2.0 * spread)
            w = np.exp((ct - ct.max()) / tau)
            w /= w.sum()
    q = np.zeros(n, dtype=np.float64)
    for s, wt in zip(idx, w):
        si = int(s)
        lo = max(0, si - n + 1)
        hi = min(n - 1, si)
        if hi >= lo and wt != 0.0:
            q[lo:hi + 1] += wt * x[si - hi:si - lo + 1][::-1]
    f = 2.0 * q
    v = float(np.dot(x, f))
    return x * (f - v)

    y = a * np.log(te) + b * np.log(1.0 - te)
    y = np.clip(y - y.max(), -60.0, 0.0)
