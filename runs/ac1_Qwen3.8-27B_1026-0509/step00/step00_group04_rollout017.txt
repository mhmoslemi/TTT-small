import time
from scipy.sparse import coo_matrix, csr_matrix, eye as speye, vstack
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    
    # Get initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        f0 = np.clip(f0, 0.0, 1000.0)
    except:
        f0 = np.ones(256, dtype=np.float64)
    
    n = len(f0)
    best = f0.copy()
    best_score = evaluate_sequence(best.tolist())
    
    # Try U-shaped initialization
    x = np.linspace(0, 1, n)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        f_try = 1.0 + alpha * (2*x - 1)**2
        f_try = f_try / f_try.sum()
        s = evaluate_sequence(f_try.tolist())
        if np.isfinite(s) and s < best_score:
            best = f_try.copy()
            best_score = s
    
    # LP-based iteration
    def solve_lp(f, M):
        """Solve max sum(g) s.t. (f*g) <= M, g >= 0. Returns g or None."""
        # Build sparse convolution matrix
        rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
        cols = np.tile(np.arange(n), n)
        vals = np.tile(f, n)
        
        A_conv = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
        A_full = vstack([A_conv, -speye(n, format='csr')])
        b = np.concatenate([M * np.ones(2*n-1), np.zeros(n)])
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A_full, b_ub=b, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.clip(res.x, 0, 1000)
        return None
    
    t = 0.5
    for it in range(2000):
        if time.monotonic() > deadline:
            break
        
        f = best.copy()
        S = f.sum()
        if S < 0.01:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        
        g = solve_lp(f, M)
        if g is None or g.sum() < 0.01:
            t = max(0.1, t * 0.7)
            continue
        
        g = g * (S / g.sum())
        
        # Line search
        improved = False
        for t_try in [t, min(1.0, t*1.5), t*0.5, t*0.25, 0.1, 0.05]:
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, 1000)
            s = evaluate_sequence(f_new.tolist())
            if np.isfinite(s) and s < best_score:
                best = f_new.copy()
                best_score = s
                t = t_try
                improved = True
                break
        
        if not improved:
            t = max(0.05, t * 0.8)
    
    # Secondary: local search
    while time.monotonic() < deadline - 2:
        candidate = best.copy()
        width = max(2, int(n * rng.uniform(0.005, 0.05)))
        start = int(rng.integers(0, n - width))
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.15))
        if rng.random() < 0.4:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0, 1000)
        s = evaluate_sequence(candidate.tolist())
        if np.isfinite(s) and s < best_score:
            best = candidate
            best_score = s
    
    return [float(x) for x in best]
