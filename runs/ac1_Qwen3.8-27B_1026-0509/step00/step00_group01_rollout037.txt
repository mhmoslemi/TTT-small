def solve_lp_cutting_plane(a, M, max_iters=20):
    n = len(a)
    # Start with all constraints (or a subset)
    # Actually, start with just the constraint at the max lag
    c = np.convolve(a, a)
    active_lags = [np.argmax(c)]
    
    b = np.ones(n)  # initial guess
    
    for it in range(max_iters):
        # Build LP with only active constraints
        A_rows = []
        for k in active_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            row = np.zeros(n)
            for j in range(lo, hi + 1):
                row[j] = a[k - j]
            A_rows.append(row)
        
        A = np.array(A_rows)  # dense, |active_lags| x n
        
        c_obj = -np.ones(n)
        b_ub = np.full(len(active_lags), M)
        bounds = [(0.0, 1000.0)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.status != 0:
            return None
        
        b = res.x
        
        # Check full constraint
        c_ab = np.convolve(a, b)
        max_idx = np.argmax(c_ab)
        if c_ab[max_idx] > M * (1 + 1e-6):
            if max_idx not in active_lags:
                active_lags.append(max_idx)
            else:
                break  # Shouldn't happen
        else:
            break
