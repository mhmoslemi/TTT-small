import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)
    
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        parent = np.ones(256)
    
    n = parent.size
    s = parent.sum()
    if s < 0.01:
        return [1.0] * 64
    
    z0 = parent / s
    a_rev = z0[::-1].copy()
    
    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1
    
    # Extended FFT for subgradient (convolution of w [len 2n-1] with a_rev [len n])
    # Result length: (2n-1) + n - 1 = 3n-2
    fft_n_sg = 1
    while fft_n_sg < 3 * n - 2:
        fft_n_sg <<= 1
    
    def fft_conv_full(x, y=None):
        """Full linear convolution."""
        if y is None:
            y = x
        lx, ly = len(x), len(y)
        fn = 1
        while fn < lx + ly - 1:
            fn <<= 1
        X = np.fft.rfft(x, fn)
        Y = np.fft.rfft(y, fn)
        return np.fft.irfft(X * Y, fn)[:lx + ly - 1]
    
    def score(z):
        c = fft_conv_full(z)
        return float(2 * n * c.max())
    
    def compute_subgrad(z, T):
        """Compute subgradient of max(conv) using soft weights."""
        c = fft_conv_full(z)
        # Softmax weights
        c_max = c.max()
        w = np.exp((c - c_max) / T)
        w /= w.sum()
        # Subgradient: g[j] = 2 * sum_k w[k] * z[k-j]
        # = 2 * (w * z_rev)[n-1-j]
        z_rev = z[::-1]
        conv = fft_conv_full(w, z_rev)
        g = 2.0 * conv[n-1 : 2*n-1][::-1]
        return g, c
    
    # Main optimization loop
    z = z0.copy()
    best_z = z0.copy()
    best_score = score(z)
    
    lr = 0.01
    T = 0.01  # Temperature for softmax
    
    iters = 0
    stagnant = 0
    
    while time.monotonic() < deadline:
        g, c = compute_subgrad(z, T)
        
        # Multiplicative update
        z_new = z * np.exp(-lr * g)
        z_new = np.maximum(z_new, 0.0)
        sz = z_new.sum()
        if sz < 1e-15:
            lr *= 0.5
            stagnant += 1
            continue
        z_new /= sz
        
        sc = score(z_new)
        if sc < best_score:

                T = min(T * 1.5, base_scale * 0.3)
            elif stagnant > 30:
                eta *= 0.8
            elif stagnant == 0:
                eta = min(eta * 1.03, 0.5)

            T = max(T * 0.998, min_T)

        return best_z, best_score

    def gauss_smooth(x, sigma):
        ksz = int(6 * sigma) | 1
        k = np.exp(-0.5 * (np.arange(ksz) - ksz // 2) ** 2 / (sigma * sigma))
        k /= k.sum()
        return np.convolve(x, k, mode='same')

    # Build candidates
    candidates = [z0.copy()]
    for sig in [3, 5, 10, 20]:
        zs = gauss_smooth(z0, sig)
        zs = np.maximum(zs, 1e-10)
        zs /= zs.sum()
        candidates.append(zs)
    for sc in [0.02, 0.05, 0.1]:
        zp = z0 * np.exp(rng.normal(0, sc, n))
        zp = np.maximum(zp, 0)
        zp /= zp.sum()
        candidates.append(zp)
    # Power-law inits
    idx = np.arange(n, dtype=np.float64) + 1
    for p in [0.5, 1.0, 1.5]:
        z = idx ** (-p)
        z /= z.sum()
        candidates.append(z)

    # Score and sort
    scores = [get_score(c) for c in candidates]
    order = np.argsort(scores)

    total_time = max(60.0, deadline - time.monotonic() - 30.0)
    best_overall = candidates[order[0]].copy()
    best_overall_score = scores[order[0]]

    # Allocate: top 3 get 40/30/20%, rest share 10%
    n_c = len(candidates)
    for rank, idx in enumerate(order):
        remaining = deadline - time.monotonic() - 25
        if remaining < 20:
            break
        if rank == 0:
            t_alloc = 0.40 * total_time
        elif rank == 1:
            t_alloc = 0.30 * total_time
        elif rank == 2:
            t_alloc = 0.20 * total_time
        else:
            t_alloc = max(10.0, 0.10 * total_time / max(1, n_c - 3))
        t_alloc = min(t_alloc, remaining - 10)
        if t_alloc < 10:
            break
        bz, bs = optimize(candidates[idx], t_alloc)
        if bs < best_overall_score:
            best_overall_score = bs
            best_overall = bz.copy()

    # Final polish
    remaining = deadline - time.monotonic() - 25
    if remaining > 30:
        bz, bs = optimize(best_overall, remaining)
        if bs < best_overall_score:
            best_overall = bz.copy()

    result = [float(x) for x in best_overall]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [float(x) for x in z0]
        fs = evaluate_sequence(result)
        if not np.isfinite(fs):
            result = [1.0] * 64
    return result
