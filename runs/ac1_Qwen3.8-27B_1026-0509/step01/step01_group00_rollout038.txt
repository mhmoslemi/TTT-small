    except NameError:
        parent = np.ones(256)
    
    n = parent.size
    # Normalize
    s = parent.sum()
    if s < 0.01:
        return [1.0] * 64
    a = parent / s  # sum = 1
    
    # FFT setup
    fft_n = 1
    while fft_n < 2*n - 1:
        fft_n <<= 1
    
    def fft_conv(x):
        fx = np.fft.rfft(x, fft_n)
        return np.fft.irfft(fx * fx, fft_n)[:2*n-1]
    
    def score(x):
        c = fft_conv(x)
        return 2.0 * n * c.max()  # since sum(x)=1
    
    # Subgradient computation
    def subgrad(x, active):
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = k - np.arange(lo, hi + 1)
            g[lo:hi+1] += x[idx]
        return g
    
    # Simplex projection
    def proj_simplex(v, target_sum=1.0):
        # Project v onto {x >= 0, sum(x) = target_sum}
        n = len(v)
        # Binary search for threshold
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
        theta = (cssv[rho] - target_sum) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    
    best = a.copy()
    best_sc = score(a)

            break

        sc = c.max()
        if sc < best_sc:
