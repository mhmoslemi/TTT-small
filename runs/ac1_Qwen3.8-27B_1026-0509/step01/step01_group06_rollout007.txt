def _score(x):

    b = np.convolve(x, x)
    M = b.max()
    S = x.sum()
    return 2 * len(x) * M / (S * S)

    """Compute gradient of 2n*M/S^2 w.r.t. x, assuming unique max."""
    n = len(x)

    M = b.max()
    k_star = b.argmax()
    S = x.sum()

    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:

    return g, M, k_star

def _pd_step(x, eta, n_iter=100000):
    """Projected gradient descent on simplex."""
    n = len(x)
    x = x.copy().astype(np.float64)
    x = np.clip(x, 0, None)
    S = x.sum()
    if S > 0:
        x /= S
    best_x = x.copy()
    best_score = _score(x)

        b = np.convolve(x, x)
        M = b.max()
        k_star = b.argmax()
        S = x.sum()
