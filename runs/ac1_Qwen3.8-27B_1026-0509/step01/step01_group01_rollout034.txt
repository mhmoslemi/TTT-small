import numpy as np
import time
from scipy.optimize import minimize

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 40.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    fft_len = 1 << (2 * n - 1).bit_length()
    fft_len_g = 1 << (3 * n - 2).bit_length()

    def fconv(a):
        A = np.fft.rfft(a, fft_len)
        return np.fft.irfft(A * A, fft_len)[:2 * n - 1]

    def score(a):
        s = a.sum()
        if s < 1e-15:
            return np.inf
        return 2.0 * n * fconv(a).max() / (s * s)

    a0 = parent / max(parent.sum(), 1e-15)
    best_a = a0.copy()
    best_score = score(a0)

    def live():
        return time.monotonic() < deadline

    # Smooth gradient via log-sum-exp
    def lse_grad(x, tau):
        xm = x - x.max()
        a = np.exp(xm)
        a /= a.sum()
        b = fconv(a)
        tb = tau * b
        mb = tb.max()
        p = np.exp(tb - mb)
        p /= p.sum()
        val = (mb + np.log(p.sum())) / tau
        P = np.fft.rfft(p, fft_len_g)
        A = np.fft.rfft(a, fft_len_g)
        cross = np.fft.irfft(np.conj(P) * A, fft_len_g)[:n]
        g = 2.0 * cross
        ag = np.dot(a, g)
        gx = a * (g - ag)
        return val, gx

    def run_lbfgs(x0, tau, maxiter):
        cnt = [0]
        def fun(x):
            cnt[0] += 1
            if cnt[0] % 200 == 0 and not live():
                raise TimeoutError()
            v, _ = lse_grad(x, tau)
            return v
        def jac(x):
            _, g = lse_grad(x, tau)
            return g
        try:
            res = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 40, 'ftol': 1e-15})
            return res.x
        except (TimeoutError, Exception):
            return x0

    # Phase 1: L-BFGS with tau annealing
    if live():
        x0 = np.log(a0 + 1e-30)
        M0 = fconv(a0).max()
        base_tau = 10.0 / max(M0, 1e-15)
        for mult in [1, 3, 10, 30, 100]:
            if not live():
                break
            tau = base_tau * mult
            try:
                x_opt = run_lbfgs(x0, tau, maxiter=500)
                a_opt = np.exp(x_opt - x_opt.max())
                a_opt /= a_opt.sum()
                s_opt = score(a_opt)
                if s_opt < best_score:
                    best_a = a_opt.copy()
                    best_score = s_opt
                x0 = x_opt
            except Exception:
                pass

    # Phase 2: Subgradient with adaptive top-k
    def subgrad(a, lr, topk):
        b = fconv(a)
        M = b.max()
        if M < 1e-30:
            return a
        idx = np.argpartition(b, -topk)[-topk:]
        grad = np.zeros(n)
        for k in idx:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi > lo:
                grad[lo:hi] += a[k - hi + 1:k - lo + 1][::-1]
        grad /= len(idx)
        gc = grad - np.dot(a, grad)
        gnorm = np.abs(gc).max()
        if gnorm < 1e-20:
            return a
        a = a * np.exp(-lr * gc / gnorm)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        return a

    if live():
        a = best_a.copy()
        lr = 0.08
        for it in range(50000):
            if not live():
                break
            a = subgrad(a, lr, topk=300)
            if it % 300 == 0 and it > 0:
                s = score(a)
                if s < best_score:
                    best_a = a.copy()
                    best_score = s
            if it % 2000 == 0 and it > 0:
                npert = max(5, n // 80)
                idx = rng.integers(0, n, npert)
                a[idx] *= np.exp(rng.normal(0, 0.25, npert))
                a = np.maximum(a, 1e-30)
                a /= a.sum()
            lr = 0.08 * (0.99995 ** it)

    # Phase 3: Final L-BFGS polish
    if live():
        x0 = np.log(best_a + 1e-30)
        Mb = fconv(best_a).max()
        tau = 50.0 / max(Mb, 1e-15)
        try:
            x_opt = run_lbfgs(x0, tau, maxiter=600)
            a_opt = np.exp(x_opt - x_opt.max())
            a_opt /= a_opt.sum()
            s_opt = score(a_opt)
            if s_opt < best_score:
                best_a = a_opt.copy()
                best_score = s_opt
        except Exception:
            pass

    # Phase 4: Block multiplicative search
    if live():
        a = best_a.copy()
        for _ in range(8000):
            if not live():
                break
            w = int(rng.integers(max(1, n // 300), n // 15))
            pos = int(rng.integers(0, n - w))
            a2 = a.copy()
            a2[pos:pos + w] *= np.exp(float(rng.normal(0, 0.4)))
            a2 = np.maximum(a2, 0)
            s2 = a2.sum()
            if s2 > 1e-15:
                a2 /= s2
                sc2 = score(a2)
                if sc2 < best_score:
                    best_a = a2.copy()
                    best_score = sc2
                    a = a2.copy()

    # Phase 5: Final subgradient fine-tune
    if live():
        a = best_a.copy()
        for it in range(15000):
            if not live():
                break
            a = subgrad(a, 0.015, topk=150)
            if it % 500 == 0 and it > 0:
                s = score(a)
                if s < best_score:
                    best_a = a.copy()
                    best_score = s

    final = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final) or final > 1e10:
        return [float(x) for x in a0.tolist()]
    return [float(x) for x in best_a.tolist()]
