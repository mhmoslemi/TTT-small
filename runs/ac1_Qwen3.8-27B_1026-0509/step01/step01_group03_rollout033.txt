    M = b.max()
    active = np.where(b >= M - 1e-10)[0]
    # Subgradient: average of shifted a for each active k
    grad = np.zeros_like(a)
    for k in active:
        # (a*a)_k = sum_i a_i * a_{k-i}, d/d a_j = a_{k-j}
        # For j in [0, n-1], k-j must be in [0, n-1]
        js = np.arange(max(0, k-n+1), min(n, k+1))
        grad[js] += a[k - js]
    grad /= len(active)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:

    a_new /= s
    return a_new
