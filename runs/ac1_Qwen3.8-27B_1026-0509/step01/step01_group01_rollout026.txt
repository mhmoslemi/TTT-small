import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
            raise ValueError
    except Exception:
        parent = np.ones(5582, dtype=np.float64)

    n = parent.size
    fft_len = 1 << (2 * n - 1).bit_length()

    def conv(a):
        A = np.fft.rfft(a, fft_len)
        return np.fft.irfft(A * A, fft_len)[:2 * n - 1]

    def score_fast(a):
        s = a.sum()
        if s < 1e-15:
            return np.inf
        return 2.0 * n * conv(a).max() / (s * s)

    def project(a):
        np.maximum(a, 0.0, out=a)
        s = a.sum()
        if s < 1e-15:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    def compute_grad(a, top_k):
        c = conv(a)
        if top_k >= c.size:
            max_c = c.max()
            exps = np.exp(c - max_c)
            weights = exps / exps.sum()
            active = np.arange(c.size)
        else:
            top_indices = np.argpartition(c, -top_k)[-top_k:]
            top_vals = c[top_indices]
            max_v = top_vals.max()
            exps = np.exp(top_vals - max_v)
            weights = exps / exps.sum()
            active = top_indices
        grad = np.zeros(n)
        for j, w in zip(active, weights):
            j = int(j)
            lo = max(0, j - n + 1)
            hi = min(n, j + 1)
            if hi > lo:
                grad[lo:hi] += w * 2.0 * a[j - hi + 1: j - lo + 1][::-1]
        return grad

    a = parent.astype(np.float64).copy()
    project(a)
    best_a = a.copy()
    best_score = score_fast(best_a)

    def optimize(a, n_iters, lr0, decay, top_k, pert_every=0, pert_sigma=0.0):
        nonlocal best_a, best_score
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        b1, b2, eps = 0.85, 0.999, 1e-8
        for it in range(n_iters):
            if time.monotonic() > deadline:
                break
            grad = compute_grad(a, top_k)
            t += 1
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            lr = lr0 * (decay ** it)
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)
            project(a)
            if pert_every > 0 and it > 0 and it % pert_every == 0:
                npert = max(5, n // 200)
                idx = rng.integers(0, n, npert)
                a[idx] *= np.exp(rng.normal(0.0, pert_sigma, npert))
                project(a)
            if it % 100 == 0:
                s = score_fast(a)
                if s < best_score:
                    best_score = s
                    best_a = a.copy()
        return a

    # Phase 1: Main optimization
    a = optimize(a, 300000, 0.06, 0.9999, top_k=30, pert_every=800, pert_sigma=0.2)

    # Phase 2: Targeted restarts
    for r in range(50):
        if time.monotonic() > deadline - 200:
            break
        sigma = 0.1 + 0.7 * (r / 50.0)
        a = best_a * np.exp(rng.normal(0.0, sigma, n))
        project(a)
        a = optimize(a, 12000, lr0=0.03, decay=0.9995, top_k=25, pert_every=400, pert_sigma=0.1)

    # Phase 3: Fine-tuning with larger top_k
    if time.monotonic() < deadline - 100:
        a = best_a.copy()
        a = optimize(a, 150000, lr0=0.012, decay=0.99997, top_k=80, pert_every=2000, pert_sigma=0.05)

    # Phase 4: Block multiplicative search
    if time.monotonic() < deadline - 50:
        a = best_a.copy()
        for it in range(100000):
            if time.monotonic() > deadline - 30:
                break
            w = int(rng.integers(1, max(2, n // 60)))
            pos = int(rng.integers(0, n - w))
            mult = np.exp(float(rng.normal(0.0, 0.15)))
            a2 = a.copy()
            a2[pos:pos + w] *= mult
            project(a2)
            s2 = score_fast(a2)
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
                a = a2.copy()

    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score) or final_score > 1e10:
        return [float(x) for x in parent]
    return [float(x) for x in best_a]
