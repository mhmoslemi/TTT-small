        r /= r.sum()
        candidates.append(r)
    for alpha in [0.5, 0.8, 1.0, 1.3]:
        r = np.maximum(1.0 + alpha * t * (1.0 - 0.3 * t), 0.0)
        r /= r.sum()
        candidates.append(r)

    best_a = None
    best_val = 1e10
    for c in candidates:
        v = float(conv(c).max())
        if v < best_val:
            best_val = v
            best_a = c.copy()

    a_opt = best_a.copy()
    eta = 0.5
    it = 0
    i_arr = np.arange(n)

    while time.monotonic() < deadline:
        it += 1
        c = conv(a_opt)
        s = float(c.max())
        if s < best_val:
            best_val = s
            best_a = a_opt.copy()

        frac = 0.9995 if it < 5000 else (0.999 if it < 30000 else 0.9985)
        thresh = s * frac
        active = np.where(c >= thresh)[0]
        k_min, k_max = int(active[0]), int(active[-1])

        prefix = np.empty(n + 1)
        prefix[0] = 0.0
        np.cumsum(a_opt, out=prefix[1:])

        lo = np.maximum(0, k_min - i_arr)
        hi = np.minimum(n - 1, k_max - i_arr)
        valid = lo <= hi
        grad = np.zeros(n)
        grad[valid] = prefix[hi[valid] + 1] - prefix[lo[valid]]

        log_u = np.clip(-eta * grad, -50.0, 0.0)
        a_new = a_opt * np.exp(log_u)
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_opt = best_a.copy()
            eta *= 0.5
            continue
        a_opt = a_new / ss

        if it % 100 == 50:
            sc = float(conv(a_opt).max())
            if sc > s * 1.005:
                eta *= 0.7
            elif sc < s * 0.995:
                eta = min(eta * 1.15, 4.0)

        if it % 10000 == 0 and it > 0:
            eta *= 0.96

        if it % 25000 == 24999:
            scale = 0.01 + 0.01 * (it / 200000)
            a_opt *= np.exp(rng.normal(0, scale, n))
            a_opt = np.maximum(a_opt, 0.0)
            a_opt /= a_opt.sum()
            eta = max(eta, 0.3)

    result = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 64
    return result
