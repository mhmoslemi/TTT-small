```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cssv - 1), np.arange(1, n+1), 0))
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)
```

```python
from scipy.optimize import linprog
import time

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    idx = np.where(cond, np.arange(1, n+1), 0)
    rho = idx.max()
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)

def _softmax_grad(a, T):
    """Compute gradient of soft-max of conv(a,a)"""
    cc = np.convolve(a, a)
    Tcc = T * cc
    Tcc -= Tcc.max()
    p = np.exp(Tcc)
    p /= p.sum()
    grad = np.correlate(p, a, 'valid')
    return grad, cc.max()

def _lp_step(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    A = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        lo = max(0, j - n + 1)
        hi = min(n, j + 1)
        for i in range(lo, hi):
            A[j, i] = a[j - i]
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None
    if not res.success:
        return None
    return res.x
```

```python
def _ps(v):
    """Project onto probability simplex {x>=0, sum(x)=1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho = int(np.max(np.where(u * np.arange(1, n+1) > cs - 1, np.arange(1, n+1), 0)))
    if rho == 0:

    return np.maximum(v - (cs[rho-1] - 1.0) / rho, 0.0)

def _cm(a):
    """Convolution matrix: A@g = np.convolve(a,g)"""
    n = len(a)
    L = 2*n - 1
    A = np.zeros((L, n), dtype=np.float64)

        lo, hi = max(0, j-n+1), min(n, j+1)
        for i in range(lo, hi):
```
