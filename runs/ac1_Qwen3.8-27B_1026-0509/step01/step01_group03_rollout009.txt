        tl = max(0.5, min(15.0, p1_end - time.monotonic() - 5))
        f_new = lp_step(f, t, tl)
        if f_new is not None and f_new.sum() > 1e-10:
            f_new /= f_new.sum()
            sc = try_score(f_new)
            if sc <= best_score:
                f = f_new
        else:
            # Gradient fallback
            tau = 0.02
            f = grad_polish(f, tau, 5)
        t = max(0.05, t * 0.96)

    # Phase 2: Gradient polishing with tau annealing
    p2_end = t0 + 0.8 * max(1.0, float(budget_s) - 20)
    f = best.copy() if best is not None else f0
    f = np.maximum(f, 0)
    f = f / max(f.sum(), 1e-15)
    for tau in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.monotonic() > p2_end:
            break
        f = grad_polish(f, tau, 100)

    # Phase 3: More LP from best
    p3_end = t0 + 0.92 * max(1.0, float(budget_s) - 20)
    f = best.copy() if best is not None else f0
    f = np.maximum(f, 0)
    f = f / max(f.sum(), 1e-15)
    t = 0.25
    for it in range(100):
        if time.monotonic() > p3_end:
            break
        tl = max(0.5, min(10.0, p3_end - time.monotonic() - 5))
        f_new = lp_step(f, t, tl)
        if f_new is not None and f_new.sum() > 1e-10:
            f_new /= f_new.sum()
            try_score(f_new)
            f = f_new
        t = max(0.03, t * 0.94)

    # Phase 4: Final gradient polish
    f = best.copy() if best is not None else f0
    f = np.maximum(f, 0)
    f = f / max(f.sum(), 1e-15)
    for tau in [0.003, 0.001, 0.0003]:
        if time.monotonic() > deadline:
            break
        f = grad_polish(f, tau, 60)

    # Random restart exploration if time remains
    if time.monotonic() < deadline - 30:
        for _ in range(5):
            if time.monotonic() > deadline - 20:
                break
            perturbation = rng.random(n) * 0.1
            f_r = np.maximum(f0 * (1 + 0.3 * (perturbation - 0.5)), 0)
            f_r = f_r / max(f_r.sum(), 1e-15)
            try_score(f_r)
            f_r = grad_polish(f_r, 0.01, 40)

    if best is None:
        best = np.ones(n) / n

    result = [float(x) for x in np.clip(best, 0, 1000)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * n
    return result
