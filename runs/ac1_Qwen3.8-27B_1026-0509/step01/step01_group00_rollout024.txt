import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    t0 = time.monotonic()

    if s < 0.01:

        N <<= 1

    def compute_conv(z):
        fz = np.fft.rfft(z, N)
        return np.fft.irfft(fz * fz, N)[:2 * n - 1]

    def compute_subgrad(z, w):
        Fz = np.fft.fft(z, N)
        Fw = np.fft.fft(w, N)
        return 2.0 * np.fft.ifft(Fw * np.conj(Fz))[:n].real

    def optimize(z0, end_time):

        z = np.maximum(z, 0.0)
        z /= z.sum()
        bz = z.copy()
        conv = compute_conv(z)
        bs = float(conv.max())

        alpha = 60.0
        stall = 0
        it = 0

        while time.monotonic() < end_time:
            mc = float(conv.max())

                bs = mc
                bz = z.copy()
                stall = 0
            else:
                stall += 1
