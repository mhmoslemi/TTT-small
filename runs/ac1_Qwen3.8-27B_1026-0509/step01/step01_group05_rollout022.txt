        if curr < best_norm - 1e-11:
            best_norm = curr
            best = a * s
            no_improve = 0

        # Soft-max gradient over top lags
        temp = max_c * 0.04
        ntop = min(12, 2 * n - 1)
        top_idx = np.argpartition(c, -ntop)[-ntop:]
        c_top = c[top_idx]
        w = np.exp((c_top - c_top.max()) / temp)
        w /= w.sum()

        grad = np.zeros(n)
        for ii in range(ntop):
            k = int(top_idx[ii])
            wi = w[ii]
            if wi < 1e-9:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            grad[js] += wi * 2.0 * a[k - js]

        grad -= grad.mean()
        v = gamma * v + grad
        a = np.clip(a - lr * v, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = best / best.sum()
            v[:] = 0.0
        else:
            a /= s

        if no_improve > patience:
            no_improve = 0
            if lr > 0.002:
                lr *= 0.6
            elif it % 40000 < 100:
                a = best / best.sum()
                a += rng.normal(0.0, 0.003, n)
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                v[:] = 0.0
                lr = 0.008
        else:
            no_improve += 1

    # Phase 3: Coordinate descent polish
    if time.monotonic() < deadline - 20.0:
        a = best.copy()
        factors = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
        step = max(1, n // 250)
        coords = list(range(0, n, step))
        for sweep in range(40):
            if time.monotonic() > deadline - 20.0:
                break
            improved = False
            for i in coords:
                if time.monotonic() > deadline - 20.0:
                    break
                orig = a[i]
                if orig < 1e-10:
                    continue
                for f in factors:
                    a[i] = orig * f
                    sc = fast_score(a)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best = a.copy()
                        improved = True
                        break
                a[i] = orig
            if not improved:
                break

    # Final validation
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = parent.copy()
    else:
        best_score = final_sc

    return [float(x) for x in best]
