                        best_a = a.copy()
                    sigma = min(sigma * 1.08, 0.15)
                else:
                    sigma = max(sigma * 0.93, 0.003)
    return best_a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed or 42)
    t0 = time.monotonic()
    deadline = t0 + max(30.0, float(budget_s)) - 12.0

    best_score = float('inf')
    best_seq = None

    def update(a):
        nonlocal best_score, best_seq
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0:
            return
        try:
            s = evaluate_sequence(a.tolist())
        except Exception:
            return
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()

    # Build initial candidate pool
    inits = []
    try:
        ph = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if ph.ndim == 1 and ph.size > 0 and np.isfinite(ph).all() and ph.sum() > 0.01:
            inits.append(ph.copy())
    except Exception:
        pass

    for n in (35, 40, 45, 50, 60, 80, 100):
        x = np.linspace(0.0, 1.0, n)
        inits.append(0.2 + 0.8 * np.cos(np.pi * x))
        inits.append(np.exp(-3.0 * ((x - 0.5) / 0.25) ** 2) + 0.05)
        inits.append(1.0 + 0.6 * np.sin(2 * np.pi * x))
        inits.append(np.abs(rng.standard_normal(n)) + 0.05)
        inits.append(np.exp(-0.5 * ((x - 0.3) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.15) ** 2))

    # Score and sort initializers
    scored = []
    for a in inits:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-12:
            an = a / s
            scored.append((_mc(an), an))
    scored.sort(key=lambda x: x[0])

    # Allocate budget
    top_k = min(len(scored), 5)
    remaining = deadline - time.monotonic()
    time_per = max(60.0, remaining / max(top_k, 1))

    for i in range(top_k):
        if time.monotonic() > deadline - 15:
            break
        _, a = scored[i]
        dl = min(time.monotonic() + time_per, deadline - 15)
        res = _optimize(a, dl, rng)
        update(res)

    # Extra time on parent if available
    if time.monotonic() < deadline - 30:
        try:
            ph = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            ph = np.maximum(ph, 0.0)
            ph = ph / ph.sum()
            res = _optimize(ph, deadline - 8, rng)
            update(res)
        except Exception:
            pass

    # Final polish: tiny perturbations around best
    if best_seq is not None and time.monotonic() < deadline - 3:
        b = best_seq.copy()
        b = np.maximum(b, 0.0)
        bs = b.sum()
        if bs > 1e-12:
            b = b / bs
            for _ in range(5000):
                if time.monotonic() > deadline - 3:
                    break
                bp = np.maximum(b + 0.01 * rng.standard_normal(len(b)), 0.0)
                ps = bp.sum()
                if ps > 1e-12:
                    bp = bp / ps
                    update(bp)
                    if _mc(bp) < _mc(b):
                        b = bp

    if best_seq is None:
        best_seq = np.ones(6)
    out = np.clip(best_seq, 0.0, 1000.0)
    fs = evaluate_sequence(out.tolist())
    if not np.isfinite(fs):
        out = np.ones(6)
    return [float(x) for x in out]
