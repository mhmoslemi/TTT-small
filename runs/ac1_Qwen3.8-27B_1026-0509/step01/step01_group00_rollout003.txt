        if bs2 < bs:
            bs, best = bs2, bz2

    if time.monotonic() < deadline - 25:
        bz3, bs3 = run_opt(best.copy(), 100000, lr0=0.3)
        if bs3 < bs:
            bs, best = bs3, bz3

    best = np.clip(best, 0.0, 1000.0)
    res = [float(x) for x in best]
    fs = evaluate_sequence(res)
    if not np.isfinite(fs):
        res = [1.0] * 64
    return res
