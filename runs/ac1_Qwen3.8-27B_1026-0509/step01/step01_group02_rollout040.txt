```python
def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - s
    cond = u - cv / np.arange(1, n+1) > 0
    if not np.any(cond):
        return np.full(n, s/n)
    i = int(np.max(np.where(cond)[0]))
    theta = cv[i] / (i+1)
    return np.maximum(v - theta, 0.0)
```
