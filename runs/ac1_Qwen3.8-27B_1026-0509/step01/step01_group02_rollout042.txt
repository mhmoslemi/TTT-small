```python
from scipy.fft import fft, ifft
import time

def _ps(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    cond = u - cv / np.arange(1, n + 1) > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    i = int(np.max(np.where(cond)[0]) + 1)
    return np.maximum(v - cv[i - 1] / i, 0.0)

    n = 4806
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)

            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
        else:
            a = np.ones(n) / n
    except:
        a = np.ones(n) / n
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    def ac(p):
        F = fft(p, fft_size)
        return ifft(F * F).real[:2 * n - 1]
    
    best = a.copy()
    best_sc = 2.0 * n * ac(a).max()
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    mom = 0.9
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        c = ac(a)
        M = c.max()
        sc = 2.0 * n * M
        
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Gradient from top-K near-max lags
        K = 20
        if 2 * n - 1 > K:
            active = np.argpartition(c, -K)[-K:]
            active = active[c[active] >= M * 0.995]
        else:
            active = np.where(c >= M * 0.995)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        g = np.zeros(n)
        for ks in active:
            ks = int(ks)
            # First term: g[j] += a[ks-j], j in [max(0,ks-n+1), min(ks,n-1)]
            lo = ks - n + 1
            if lo < 0: lo = 0
            hi = ks
            if hi >= n: hi = n - 1
            if lo <= hi:
                g[lo:hi+1] += a[ks-hi:ks-lo+1][::-1]
            # Second term: g[j] += a[j-ks], j in [max(0,ks), min(ks+n-1,n-1)]
            lo2 = ks
            if lo2 < 0: lo2 = 0
            hi2 = ks + n - 1
            if hi2 >= n: hi2 = n - 1
            if lo2 <= hi2:
                g[lo2:hi2+1] += a[lo2-ks:hi2-ks+1]
        
        g /= len(active)

        a = _ps(a - alpha * v)
        
        it += 1
        
        if stall > 8000:

            a = _ps(best + pert)
            v[:] = 0
            mom = 0.85
            stall = 0
            it = max(0, it - it // 5)
        
        if it % 50000 == 0 and mom > 0.7:
            mom = max(0.7, mom - 0.005)
    
    return [float(x) for x in best]
```

```python
import numpy as np
from scipy.fft import fft, ifft
import time

def _ps(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    cond = u - cv / np.arange(1, n + 1) > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    i = int(np.max(np.where(cond)[0]) + 1)
```

```python
import numpy as np
from scipy.fft import fft, ifft

def _ps(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    cond = u - cv / np.arange(1, n + 1) > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    i = int(np.max(np.where(cond)[0]) + 1)
    return np.maximum(v - cv[i - 1] / i, 0.0)
```
