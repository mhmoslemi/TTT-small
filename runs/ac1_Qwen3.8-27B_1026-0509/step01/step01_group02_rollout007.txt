                    gdir = res.x / res.x.sum()
                    a = proj_simplex(0.8 * a + 0.2 * gdir)
            except:
                pass

        if stall > 4000:
            eps = rng.choice([0.3, 0.8, 1.5]) / n
            a = proj_simplex(np.maximum(a + rng.normal(0, eps, n), 0))
            alpha, mu = 0.1, 0.4
            mom = np.zeros(n)
            stall = 0

        if stall > 10000:
            a = proj_simplex(np.maximum(best + rng.normal(0, 0.5 / n, n), 0))
            alpha, mu, K = 0.2, 0.7, rng.integers(3, 12)
            mom = np.zeros(n)
            stall = 0

        it += 1

    _ = evaluate_sequence(best.tolist())
    return [float(max(0.0, min(1000.0, v))) for v in best]
