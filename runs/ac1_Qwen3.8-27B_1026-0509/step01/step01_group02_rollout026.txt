        a_new = _simplex_proj(a - lr * v)

        c_new = conv(a_new)
        sc_new = obj(c_new)

        if sc_new <= sc_cur:
            a = a_new
            c_cur = c_new
            if sc_new < best_sc:
                best_sc = sc_new
                best = a.copy()
                no_imp = 0
            else:
                no_imp += 1
            lr = min(lr * 1.003, 20.0 / n)
        else:
            lr *= 0.5
            v *= 0.3
            no_imp += 1

        if no_imp > 5000:
            scale = min(5.0 / n, 0.5 / np.sqrt(iter_count) * (1.0/n))
            a = _simplex_proj(best + rng.normal(0.0, scale, n))
            v = np.zeros(n)
            lr = 1.0 / n
            no_imp = 0
            c_cur = conv(a)

    fs = float(evaluate_sequence(best.tolist()))
    return [float(val) for val in best]
