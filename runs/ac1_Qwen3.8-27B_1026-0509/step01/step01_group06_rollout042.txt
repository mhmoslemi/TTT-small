def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _smooth_grad(a, eps=5.0):
    """Compute gradient of smooth max of a*a."""
    b = np.convolve(a, a)
    b_shifted = b - b.max()
    w = np.exp(eps * b_shifted)
    w /= w.sum()
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        lo = max(0, j)
        hi = min(2*n-1, j+n-1)
        g[j] = np.dot(w[lo:hi+1], a[lo-j:hi-j+1])
    return g

            state[1] = a.copy()

    return state

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 12.0

    parent = None
    try:
        ph = np.asarray(height_sequence_1, dtype=np.float64)
        ph = np.clip(ph, 0.0, 1000.0)
        if ph.ndim == 1 and ph.size > 0 and np.isfinite(ph).all() and ph.sum() > 0.01:
            parent = ph.copy()
    except Exception:
        pass

    best_score = float('inf')
    best_seq = np.ones(8, dtype=np.float64)

    def try_update(a):
        nonlocal best_score, best_seq
        a = np.clip(a, 0, 1000)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()

    if parent is not None:
        try_update(parent / parent.sum())

    tasks = []
    if parent is not None:
        pn = len(parent)
        pnorm = parent / parent.sum()
        tasks.append(pnorm.copy())
        for _ in range(4):
            p = pnorm * np.exp(rng.normal(0, 0.25, pn))
            p = np.clip(p, 1e-4, None)
            tasks.append(p / p.sum())

    for n in [40, 32, 36, 44, 48, 56, 28, 24, 64, 38, 42, 50]:
        if parent is not None and n == len(parent):
            continue
        idx = np.arange(n)
        inits = [
            np.clip(1 + 0.4 * np.sin(2 * np.pi * idx / n + rng.uniform(0, 6.28)), 0.05, None),
            np.clip(1 + 0.4 * np.cos(2 * np.pi * idx / n + rng.uniform(0, 6.28)), 0.05, None),
            np.exp(-0.5 * ((idx - 0.45 * n) ** 2) / (0.12 * n ** 2)) + 0.05,
            np.exp(-idx * rng.uniform(0.3, 1.2)) + 0.08,
            np.clip(1 + rng.normal(0, 0.35, n), 0.05, None),
            np.where(idx % 2 == 0, 1.0, 0.55) + rng.normal(0, 0.08, n),
        ]
        bi, bs = None, float('inf')
        for ini in inits:
            ini = np.clip(ini, 0.01, None)
            ini = ini / ini.sum()
            sc = _score(ini)
            if sc < bs:
                bs, bi = sc, ini.copy()
        if bi is not None:
            tasks.append(bi)

    # Sort by score
    scored = [(_score(t / max(t.sum(), 1e-10)), t) for t in tasks]
    scored.sort(key=lambda x: x[0])

    remaining = max(30.0, deadline - time.monotonic())
    # Give top tasks more time
    n_tasks = len(scored)
    for i, (sc, init) in enumerate(scored):
        if time.monotonic() > deadline - 8.0:
            break
        frac = 1.0 / (i + 1)
        alloc = min(remaining * frac / (1 + 0.3 * i), 300.0)
        alloc = max(alloc, 15.0)
        alloc = min(alloc, deadline - time.monotonic() - 8.0)
        if alloc < 10.0:
            break
        state = [best_score, best_seq.copy()]
        _refine(init, time.monotonic() + alloc, rng, state)
        if best_score <= 1.50274:
            break

    out = np.clip(best_seq, 0.0, 1000.0)
    if not (out.size > 0 and np.isfinite(out).all() and out.sum() > 0.01):
        out = np.ones(6, dtype=np.float64)
    fs = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(fs):
        out = np.ones(6, dtype=np.float64)
    return [float(x) for x in out]
