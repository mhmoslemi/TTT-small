```python
        shapes.append(np.exp(alpha * x * x) / np.exp(alpha))
        shapes.append(1.0 + alpha * np.exp(-20 * x * x))
    # Mixed
    for alpha in [2.0, 10.0, 50.0]:
        for beta in [0.3, 0.7, 1.0, 2.0]:
            for flip in [0, 1]:
                tt = 1.0 - t_grid if flip else t_grid
                shapes.append(1.0 + alpha * tt**beta - 0.5 * alpha * t_grid**beta)

    for s in shapes:
        if time.monotonic() > deadline:
            break
        cand = np.clip(s, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    # Phase 2: L-BFGS-B with soft-max annealing
    a = best.copy().astype(np.float64)
    a = np.clip(a, 0, 1000)
    a = a / a.sum() * n  # normalize to sum = n

    tau_schedule = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    bounds = [(0.0, 1000.0)] * n

    for tau in tau_schedule:
        if time.monotonic() > deadline - 120:
            break

        def fc(x):
            return softmax_obj(x, tau), softmax_obj_grad(x, tau)

        try:
            res = minimize(fc, a, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'maxfun': 3000, 'gtol': 1e-5})
            if res.fun < np.inf:
                a = np.clip(res.x, 0, 1000)
                if a.sum() > 0.01:
                    sc = float(evaluate_sequence(a.tolist()))
                    if sc < best_score:
                        best_score = sc
                        best = a.copy()
        except Exception:
            pass
        a = a / max(a.sum(), 1e-10) * n

    # Phase 3: subgradient refinement with true objective (top-k lags)
    a = best.copy().astype(np.float64)
    a = a / a.sum() * n
    lr = 0.05
    for it in range(80000):
        if time.monotonic() > deadline - 60:
            break
        A = rfft(a, N)
        b = np.real(irfft(A * A, N))[:2 * n - 1]
        max_b = b.max()
        S = a.sum()
        # top-k lags with softmax weights
        top_k = 10
        b_top = np.sort(b)[-top_k:]
        w_lags = np.exp(b_top / max_b - 1.0)
        w_lags /= w_lags.sum()
        top_idx = np.argsort(b)[-top_k:]
        # gradient via cross-correlation of weights with a
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w_lags
        W = rfft(w_full, N)
        g = np.real(irfft(W * np.conj(A), N))[:n]
        M_eff = float(w_lags @ b_top)
        grad = 4.0 * n / (S * S) * (g * S - M_eff)
        grad -= grad.mean()
        a = np.clip(a + lr * grad, 0, 1000)
        a = a / a.sum() * n
        lr *= 0.99995

        if it % 2000 == 0 or it == 79999:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score:
                best_score = sc
                best = a.copy()

    # Phase 4: LP-based direction (if time allows)
    if time.monotonic() < deadline - 200:
        a = best.copy().astype(np.float64)
        a = a / a.sum() * n
        A = rfft(a, N)
        b = np.real(irfft(A * A, N))[:2 * n - 1]
        M_cur = b.max()

        # Build sparse constraint matrix C (2n-1) x n: C[k,j] = a[k-j] if valid
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                js = np.arange(j_lo, j_hi + 1)
                rows.extend([k] * len(js))
                cols.extend(js.tolist())
                vals.extend(a[k - js].tolist())
        C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        from scipy.optimize import linprog
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M_cur),
                      bounds=[(0, 1000)] * n, method='highs',
                      options={'time_limit': 60})
        if res.status == 0 and res.x is not None:
            g_dir = np.clip(res.x, 0, 1000)
            if g_dir.sum() > 0.01:
                for t_mix in [0.1, 0.2, 0.3, 0.5]:
                    mix = (1 - t_mix) * a + t_mix * g_dir
                    mix = np.clip(mix, 0, 1000)
                    mix = mix / mix.sum() * n
                    sc = float(evaluate_sequence(mix.tolist()))
                    if sc < best_score:
                        best_score = sc
                        best = mix.copy()
                        a = mix.copy()

    # Phase 5: final local search + random perturbations
    for _ in range(100):
        if time.monotonic() > deadline - 10:
            break
        a = best.copy()
        rel = 0.005 + 0.01 * rng.random()
        noise = rng.normal(0, rel, n)
        a = np.clip(a + noise, 0, 1000)
        sc = float(evaluate_sequence(a.tolist()))
        if sc < best_score:
            best_score = sc
            best = a.copy()

    # Final validation
    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs) or fs > best_score:
        fs = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(fs):
            best = parent.copy()
            best_score = fs
        else:
            best = np.ones(n)
            best_score = float(evaluate_sequence(best.tolist()))

    return [float(x) for x in best]
```
