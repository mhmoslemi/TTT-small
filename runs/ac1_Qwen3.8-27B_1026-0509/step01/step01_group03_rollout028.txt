    # Initialize with binomial sequence
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1) / (2*k)
    
    # Normalize
    S = a.sum()
    a = a / S
    
    best_score = np.inf
    best_a = a.copy()
    
    lr = 0.5
    for step in range(steps):
        if time.monotonic() > deadline:
            break
        
        b = np.convolve(a, a, 'full')
        S = a.sum()
        score = 2 * n * b.max() / (S**2)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Compute soft-max weights
        w = np.exp(alpha * b / b.max())  # normalize by max to avoid overflow
        w = w / w.sum()
        
        # Gradient of soft_max with respect to a
        # ∂soft_max/∂a_j = alpha * sum_k w_k * 2*a_{k-j}
        # (zero-extended)
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            # b_k = sum_i a_i * a_{k-i}
            # ∂b_k/∂a_j = 2*a_{k-j} (zero-extended)
            # So grad[j] += w[k] * 2 * a[k-j]
            js = np.arange(n)
            valid = (k - js >= 0) & (k - js < n)
            grad[valid] += w[k] * 2 * a[k - js[valid]]
        
        # Gradient of full objective (with S constraint)
        # f = 2n * soft_max(b) / S^2
        # ∂f/∂a_j = (2n/S^2) * (∂soft_max/∂a_j) - (2n * soft_max / S^3) * 1
        soft_max_val = np.dot(w, b)
        obj_grad = (2*n/S**2) * grad - (2*n*soft_max_val/S**3) * np.ones(n)
        
        # Projected gradient step on simplex
        a = a - lr * obj_grad
        a = np.maximum(0, a)
        a = a / a.sum()
        
        # Decay learning rate
        lr *= 0.999
    
    return best_a, best_score
