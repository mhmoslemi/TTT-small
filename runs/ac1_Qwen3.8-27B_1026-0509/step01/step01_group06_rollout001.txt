        refine(a60, (deadline - time.monotonic()) * 0.30)

    # Final validation
    out = np.clip(best_seq, 0.0, 1000.0)
    if not (out.size > 0 and np.all(np.isfinite(out)) and out.sum() > 0.01):
        out = np.ones(6, dtype=np.float64)
    fs = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(fs):
        out = np.ones(6, dtype=np.float64)
    return [float(x) for x in out]
