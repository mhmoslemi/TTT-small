    """Solve max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    L = 2 * n - 1
    # Build the matrix: A[k, j] = a[k-j] if 0 <= k-j <= n-1 else 0
    # convolve(a, g)[k] = sum_j a[k-j]*g[j]
    A = np.zeros((L, n))
    for k in range(L):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    c = -np.ones(n)
    b_ub = np.full(L, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
    except:
        return None
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Take a projected subgradient step to minimize max(convolve(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    k_star = int(b.argmax())
    # Gradient of B[k_star] w.r.t. a[m] = 2*a[k_star - m] if 0 <= k_star - m < n
    grad = np.zeros(n)
    for m in range(n):
        idx = k_star - m
        if 0 <= idx < n:
            grad[m] = 2.0 * a[idx]
    new_a = a - lr * grad
    return _project_simplex(new_a)

            inits.append(1.0 + 0.7 * np.cos(2 * np.pi * f * idx / n))
        inits.append(1.0 - 0.8 * np.abs(2.0 * idx / (n - 1 + 1e-9) - 1.0))
        inits.append(1.0 + 0.8 * np.abs(2.0 * idx / (n - 1 + 1e-9) - 1.0))
        for _ in range(10):
            inits.append(np.abs(rng.normal(1.0, 0.6, n)) + 0.05)
        for _ in range(3):
            v = np.zeros(n)
            v[rng.choice(n, max(2, n // 3), replace=False)] = 1.0
            inits.append(v + 0.05)
        if parent is not None:
            if parent.size >= n:
                inits.append(parent[:n].copy())
            elif parent.size > 0:
                inits.append(np.resize(parent, n))
        for v in inits:
            v = np.clip(v, 1e-8, None)
            if v.sum() > 0:
                v = v / v.sum()
                s = _score(v)
                if np.isfinite(s):
                    cands.append((s, n, v))

    cands.sort(key=lambda x: x[0])

    for i in range(min(60, len(cands))):
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break
        s0, n, v = cands[i]
        share = min(rem * 0.3, 60.0)
        a_out, _ = _refine(v, time.monotonic() + share, rng)
        upd(a_out)
        if best_s <= 1.50274:
            break

    # Final polish on best
    if best_a is not None and time.monotonic() < deadline - 2.0:
        a_pol, _ = _refine(best_a, deadline, rng)
        upd(a_pol)

    if best_a is not None and np.isfinite(best_s):
        out = np.clip(best_a, 0.0, 1000.0)
        fs = float(evaluate_sequence(out.tolist()))
        if np.isfinite(fs) and out.sum() > 0.01:
            return [float(x) for x in out]

    if parent is not None:
        return [float(x) for x in np.clip(parent, 0.0, 1000.0)]
    return [1.0] * 6
