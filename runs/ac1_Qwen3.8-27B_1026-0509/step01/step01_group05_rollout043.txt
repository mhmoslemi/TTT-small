        ks = int(ac.argmax())
        return 2.0 * n * mb / (S * S), mb, ks, ac

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Phase 1: parametric exploration
    t = np.linspace(0.0, 1.0, n)
    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        for alpha in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0, 10.0, 50.0]:
            candidates.append(1.0 + alpha * t**p)
            candidates.append(1.0 + alpha * (1.0 - t)**p)
            candidates.append(np.exp(-alpha * (t - 0.3)**2))
            candidates.append(np.exp(-alpha * (t - 0.7)**2))
            candidates.append(1.0 + alpha * np.sin(np.pi * t)**p)
            candidates.append(np.maximum(0.0, 1.0 - alpha * np.abs(2*t - 1.0)**p))
    for cand in candidates:
        if time.monotonic() > deadline - 1300.0:
            break
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    # Phase 2: Gradient descent with Adam using top-k lag gradient
    a = best.copy().astype(np.float64)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
    t_adam = 0
    lr_schedule = 0.5
    no_improve = 0
    patience = 3000

    for it in range(1000000):
        if time.monotonic() > deadline - 60.0:
            break

        # Compute autocorrelation
        A = fft(a, fft_size)
        ac = np.real(ifft(A * A, fft_size))[:2 * n - 1]
        max_b = float(ac.max())
        S = float(a.sum())

        # Top-k lags for smoother gradient
        k_top = 8
        top_idx = np.argsort(ac)[-k_top:]
        top_vals = ac[top_idx]
        top_mean = top_vals.mean()

        # Compute gradient as weighted average of top-k lag gradients
        g = np.zeros(n)
        for kk in top_idx:
            w_k = max(0.0, ac[kk] - 0.5 * top_mean)
            if w_k <= 0:
                continue
            j_min = max(0, kk - n + 1)
            j_max = min(n - 1, kk)
            jv = np.arange(j_min, j_max + 1)
            idx = kk - jv
            g[jv] += w_k * 2.0 * a[idx]
        w_sum = w_k if 'w_k' in dir() and w_k > 0 else 1.0
        # Normalize gradient
        g /= (sum(max(0.0, ac[kk] - 0.5 * top_mean) for kk in top_idx) + 1e-12)

        # Direction: descent for score = 2n*max_b/S^2
        # d(score)/da ~ (1/S^2)*grad_b - (2*max_b/S^3)*ones
        # Direction to subtract: grad_b/S - 2*max_b/S^2 * ones (up to positive factor)
        direction = g - (2.0 * max_b / S) * 1.0

        # Adam update
        t_adam += 1
        lr = lr_schedule / (np.sqrt(t_adam) + 1.0)
        m_adam = beta1 * m_adam + (1.0 - beta1) * direction
        v_adam = beta2 * v_adam + (1.0 - beta2) * direction * direction
        m_hat = m_adam / (1.0 - beta1**t_adam)
        v_hat = v_adam / (1.0 - beta2**t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = np.clip(a + step, 0.0, 1000.0)

        # Track best (check every 20 iters to save time on evaluate_sequence)
        if it % 20 == 0:
            sc = 2.0 * n * max_b / (S * S)
            if sc < best_score - 1e-10:
                # Verify with exact evaluator
                sc_exact = float(evaluate_sequence(a.tolist()))
                if sc_exact < best_score:
                    best_score = sc_exact
                    best = a.copy()
                    no_improve = 0
                else:
                    no_improve += 20
            else:
                no_improve += 20

        # Adaptive restart
        if no_improve > patience:
            no_improve = 0
            patience = max(1000, patience // 2)
            # Perturbation
            scale = best.mean() + 1e-6
            a = best + rng.normal(0, 0.05 * scale, n)
            a = np.clip(a, 0.0, 1000.0)
            m_adam[:] = 0
            v_adam[:] = 0
            t_adam = 0
            lr_schedule = max(0.01, lr_schedule * 0.5)

    # Phase 3: Final refinement with hard subgradient
    a = best.copy()
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr_schedule = 0.1

    for it in range(200000):
        if time.monotonic() > deadline - 10.0:
            break

        A = fft(a, fft_size)
        ac = np.real(ifft(A * A, fft_size))[:2 * n - 1]
        max_b = float(ac.max())
        k_star = int(ac.argmax())
        S = float(a.sum())

        # Hard subgradient
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        jv = np.arange(j_min, j_max + 1)
        idx = k_star - jv
        g = np.zeros(n)
        g[jv] = 2.0 * a[idx]

        direction = g - (2.0 * max_b / S) * 1.0

        t_adam += 1
        lr = lr_schedule / (np.sqrt(t_adam) + 1.0)
        m_adam = 0.9 * m_adam + 0.1 * direction
        v_adam = 0.999 * v_adam + 0.001 * direction * direction
        step = lr * m_adam / (np.sqrt(v_adam / (1 - 0.999**t_adam)) + eps_adam)
        a = np.clip(a + step, 0.0, 1000.0)

        if it % 50 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score:
                best_score = sc
                best = a.copy()

    # Final validation
    final = np.clip(best, 0.0, 1000.0)
    fs = float(evaluate_sequence(final.tolist()))
    if fs < best_score:
        best_score = fs
        best = final

    return [float(x) for x in best]
