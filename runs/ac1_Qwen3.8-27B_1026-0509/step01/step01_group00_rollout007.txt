import numpy as np
import time
import cvxpy as cp


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 40.0)

    try:
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    except NameError:
        a0 = np.ones(256)
    if a0.ndim != 1 or a0.size < 4 or a0.sum() < 0.01:
        a0 = np.ones(256)
    n = a0.size
    a = a0 / a0.sum()

    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1

    def do_conv(x, y=None):
        if y is None:
            y = x
        X = np.fft.rfft(x, fft_n)
        Y = np.fft.rfft(y, fft_n)
        return np.fft.irfft(X * Y, fft_n)[:2 * n - 1]

    def get_score(arr):
        c = do_conv(arr)
        return float(2 * n * c.max())

    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_s = get_score(a)

    def subgrad_step(arr, lr, k_top=60, beta=8.0):
        c = do_conv(arr)
        cmax = float(c.max())
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        c_top = c[top_idx]
        c_sh = c_top - c_top.max()
        w = np.exp(beta * c_sh)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k_top) / k_top
        else:
            w /= ws
        g = np.zeros(n)
        for j in range(k_top):
            if w[j] < 1e-12:
                continue
            k = int(top_idx[j])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += w[j] * arr[k - np.arange(lo, hi + 1)]
        arr2 = arr * np.exp(-lr * g)
        arr2 = np.maximum(arr2, 0.0)
        s2 = arr2.sum()
        if s2 < 1e-30:
            return arr.copy(), arr, cmax
        arr2 /= s2
        return arr2, arr, cmax

        c = do_conv(arr)

            hi = min(k, n - 1)

                coeff = arr[k - np.arange(lo, hi + 1)]
                rows = np.arange(lo, hi + 1)
                expr = coeff @ b_var[rows]
                constraints.append(expr <= cmax)
            else:
                constraints.append(cp.Constant([0.0]) <= cmax)
        obj = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(obj, constraints)
        t_s = time.monotonic()
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.ECOS]:
            try:
                prob.solve(solver=solver, time_limit=max(1, int(timeout)))
                if b_var.value is not None:
                    solved = True
                    break

                continue
            if time.monotonic() - t_s > timeout:
                break
        if not solved or b_var.value is None:
            return None
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
