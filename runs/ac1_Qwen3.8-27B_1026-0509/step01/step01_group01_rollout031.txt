            g = grad / (a + 1e-20)
            g = np.clip(g, -50.0, 50.0)
            a = a * np.exp(-lr * g)
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / S
            
            # Perturbation
            if perturb_every > 0 and (it + 1) % perturb_every == 0:
                a = a * np.exp(rng.normal(0, perturb_sigma, n))
                a = np.maximum(a, 0.0)
                S = a.sum()
                if S > 1e-15:
                    a = a / S
            
            if (it + 1) % check_every == 0:
                update_best(a)
        
        update_best(a)
        return a
