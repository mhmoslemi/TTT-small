        s = np.sort(p)[::-1]

        rho = n
        for k in range(n, 0, -1):
            theta = (cumsum[k - 1] - target) / k
            if k == n or s[k - 1] > theta:
                rho = k
                break
        theta = (cumsum[rho - 1] - target) / rho
        return np.maximum(p - theta, 0)
