def _autocorr_matrix(a, n):
    """Build the matrix C such that (a * g)[k] = sum_i C[k,i] * g[i]."""
    L = 2 * n - 1
    C = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        # (a*g)[k] = sum_i a[i] * g[k-i] where 0<=i<n and 0<=k-i<n
        # => i ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            C[k, i] = a[k - i]
    return C

    return None

            trials.append((n, np.clip(rng.uniform(0.05, 1.0, n), 0.01, None)))
        x = np.arange(n)
        trials.append((n, np.clip(1.0 + 0.5*np.sin(2*np.pi*2*x/n), 0.01, None)))
        trials.append((n, np.clip(1.0 + 0.5*np.cos(2*np.pi*3*x/n), 0.01, None)))

    # Evaluate all inits first to rank
    scored = []
    for n, init in trials:
        s = float(evaluate_sequence(init.tolist()))
        if np.isfinite(s):
            scored.append((s, n, init))
        if time.monotonic() > deadline:
            break
    scored.sort(key=lambda x: x[0])

    # Run Delsarte on each, allocating more time to better initial scores
    for i, (s0, n, init) in enumerate(scored):
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        frac = remaining / max(1, len(scored) - i)
        alloc = min(120.0, max(3.0, frac))
        sub_deadline = time.monotonic() + alloc

        a = _delsarte_loop(init.copy(), min(sub_deadline, deadline), rng)
        try_seq(a)

        # Soft-max polish
        if time.monotonic() < deadline - 5.0:
            a2 = _softmax_grad_polish(a, steps=200, lr=0.05, beta=50, deadline=deadline)
            try_seq(a2)

            # Try a second round of Delsarte from polished point
            if time.monotonic() < deadline - 5.0:
                a3 = _delsarte_loop(a2.copy(), deadline, rng)
                try_seq(a3)

    # Final validation
    out = best_seq
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.ones(4, dtype=np.float64)
    fs = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(fs):
        out = np.ones(4, dtype=np.float64)
    return [float(x) for x in out]
