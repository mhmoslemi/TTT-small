def _build_A(f, n):
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, rhs):
    A = _build_A(f, n)
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= rhs])
    solvers = [cp.GLOP, cp.PDLP, cp.ECOS, cp.CVXOPT, cp.CBC]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and g.value is not None:
                gv = np.asarray(g.value).flatten()
                if np.all(np.isfinite(gv)) and gv.sum() > 1e-10:
                    return np.maximum(gv, 0)
        except:
            continue
    return None

    grad = 2.0*n*grad_sm/S**2 - 4.0*n*sm*f/S**3
    return grad
