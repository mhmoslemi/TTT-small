```python
def _lp_direction(p, M):
    """Find q that maximizes sum(q) subject to (p*q)_j <= M, q >= 0"""
    n = len(p)
    L = 2*n - 1
    A = np.zeros((L, n))
    for j in range(L):
        lo = max(0, j - n + 1)
        hi = min(n, j + 1)
        for i in range(lo, hi):
            A[j, i] = p[j - i]
    c = -np.ones(n)  # minimize -sum(q) = maximize sum(q)
    res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```
