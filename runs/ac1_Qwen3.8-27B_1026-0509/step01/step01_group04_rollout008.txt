    """Find v >= 0 minimizing 2n * max(base + 2v*c + v^2*q) / (S+v)^2"""
    
    def f(v):
        vals = base + 2.0*v*c + v*v*q
        m = vals.max()
        return 2.0 * n * m / ((S + v)**2)

    # f(0) = 2n * max(base) / S^2
    # f(large) ≈ 2n
