                    sc = _score(a)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    else:
                        a = best_a.copy()
                    stall = 0
                    tau = 0.3

    # Gradient polish
    if time.monotonic() < deadline - 5:
        a = best_a.copy()
        sc = _score(a)
        for _ in range(500):
            if time.monotonic() > deadline - 3:
                break
            a_new = _gradient_step(a, n)
            if a_new.sum() < 0.01:
                break
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new < sc:
                a = a_new
                sc = sc_new
            else:
                a_new = _gradient_step(a, n, eta=0.3 / (np.max(np.abs(a)) + 1e-10))
                if a_new.sum() > 0.01:
                    sc_new2 = _score(a_new)
                    if np.isfinite(sc_new2) and sc_new2 < sc:
                        a = a_new
                        sc = sc_new2
                    else:
                        break
                else:
                    break
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s)) - 20.0

    best_seq = None
    best_score = np.inf

    def update(a, sc):
        nonlocal best_seq, best_score
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # Primary: use height_sequence_1
    n = 400
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if init.shape[0] != n:
            init = init[:n]
        if init.sum() < 0.01:
            init = np.ones(n) / n
        sc0 = _score(init)
        update(init, sc0)
    except Exception:
        j = np.arange(n, dtype=np.float64)
        init = np.sin(np.pi * (j + 0.5) / n)
        init = init / init.sum()
        sc0 = _score(init)
        update(init, sc0)

    # LP refinement on primary init
    alloc = min(1200.0, max(1.0, float(budget_s)) * 0.55)
    dl_lp = t_start + alloc
    refined, sc_ref = _lp_refine(init, n, dl_lp, rng, max_iters=3000)
    update(refined, sc_ref)

    # Alternative initializations if time remains
    alts = []
    j = np.arange(n, dtype=np.float64)
    # Cosine tapered
    alt1 = np.cos(np.pi * j / (2 * n)) ** 2
    alt1 = np.maximum(alt1, 0.01)
    alts.append(alt1)
    # Asymmetric ramp
    alt2 = 1.0 - j / n
    alt2 = np.maximum(alt2, 0.01)
    alts.append(alt2)
    # Gaussian centered
    x = (j - n * 0.3) / (n * 0.25)
    alt3 = np.exp(-0.5 * x * x)
    alts.append(alt3)
    # Random smoothed
    r = rng.random(n)
    k = np.ones(5) / 5
    alt4 = np.convolve(r, k, mode='same')
    alt4 = np.maximum(alt4, 0.01)
    alts.append(alt4)

    remaining_checks = 3
    for alt in alts:
        if time.monotonic() > deadline - 30:
            break
        if remaining_checks <= 0:
            break
        alt_n = alt / alt.sum()
        sc_a = _score(alt_n)
        update(alt_n, sc_a)
        if sc_a < best_score + 0.05:
            sub_dl = min(time.monotonic() + 180.0, deadline - 30)
            ref2, sc2 = _lp_refine(alt_n, n, sub_dl, rng, max_iters=1500)
            update(ref2, sc2)
        remaining_checks -= 1

    # Final gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 5:
        a = best_seq.copy()
        sc = _score(a)
        for _ in range(300):
            if time.monotonic() > deadline - 3:
                break
            a_new = _gradient_step(a, n)
            if a_new.sum() < 0.01:
                break
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new < sc:
                a = a_new
                sc = sc_new
            else:
                a_new = _gradient_step(a, n, eta=0.2 / (np.max(np.abs(a)) + 1e-10))
                if a_new.sum() > 0.01:
                    sc_new2 = _score(a_new)
                    if np.isfinite(sc_new2) and sc_new2 < sc:
                        a = a_new
                        sc = sc_new2
                    else:
                        break
                else:
                    break
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    if best_seq is None:
        best_seq = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        best_score = _score(best_seq)

    final = [float(x) for x in best_seq]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs):
        fallback = np.sin(np.pi * (np.arange(400) + 0.5) / 400)
        final = [float(x) for x in fallback]
    return final
