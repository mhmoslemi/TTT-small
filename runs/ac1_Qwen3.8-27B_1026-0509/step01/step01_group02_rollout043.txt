            break

        F = fft(p, N)
        c = ifft(F * F).real[:2 * n - 1]
        sc = 2.0 * n * c.max()

        if sc < best_sc:
            best_sc, best, no_improve = sc, p.copy(), 0
        else:
            no_improve += 1

        cs = c - c.max()
        w = np.exp(np.clip(beta * cs, -100, 0))
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w /= ws

        W = np.zeros(N)
        W[:2 * n - 1] = w
        cross = ifft(fft(W) * conj(F)).real[:n]

        w_ext = np.zeros(2 * n)
        w_ext[:2 * n - 1] = w
        wc = np.empty(2 * n + 1)
        wc[0] = 0.0
        np.cumsum(w_ext, out=wc[1:])
        win = wc[n:] - wc[:n]

        g = cross + p * win
        mom = mu * mom + (1.0 - mu) * g
        p = proj(p - alpha * mom)

        beta = min(500.0, 10.0 + it * 0.02)
        if it % 5000 == 0 and it > 0:
            alpha *= 0.9
            mu = min(0.95, mu + 0.005)

        if no_improve > 10000:
            p = proj(best * (1.0 + 0.01 * rng.standard_normal(n) / np.sqrt(n)))
            mom = np.zeros(n)
            alpha = 0.5
            no_improve = 0

    rem = deadline - time.monotonic()
    if rem > 10:
        scales = np.array([0.001, 0.003, 0.01, 0.03, 0.1])
        n_att = int(min(2000, rem * 200))
        for _ in range(n_att):
            if time.monotonic() > deadline:
                break
            s = rng.choice(scales)
            q = proj(best + s * rng.standard_normal(n) / np.sqrt(n))
            Fq = fft(q, N)
            cq = ifft(Fq * Fq).real[:2 * n - 1]
            sq = 2.0 * n * cq.max()
            if sq < best_sc:
                best_sc, best = sq, q

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n) / n
    else:
        best = best / best.sum()

    fv = evaluate_sequence(best.tolist())
    if not np.isfinite(fv):
        best = np.ones(n) / n
    return [float(v) for v in best]
