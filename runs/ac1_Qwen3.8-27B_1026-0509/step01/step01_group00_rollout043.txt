            if sb > 1e-10:
                return b / sb
        return None
    
    # Main loop
    eta = 1.0
    K = 40
    iters_since_restart = 0
    max_stagnant = 3000
    
    while time.monotonic() < deadline - 30:
        c = fft_conv(a)
        ms = float(c.max())
        
        if ms < best_s:
            best_s = ms
            best_a = a.copy()
            iters_since_restart = 0
        iters_since_restart += 1
        
        g = subgrad(a, c, K)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            continue
        a /= sa
        
        # Adaptive learning rate (every 50 iters)
        if iters_since_restart % 50 == 0:
            ns = float(fft_conv(a).max())
            if ns > best_s * 1.003:
                eta *= 0.75
            elif ns < best_s:
                eta = min(eta * 1.15, 8.0)
        
        # Periodic LP jump
        if iters_since_restart % 500 == 0 and time.monotonic() < deadline - 60:
            c_cur = fft_conv(a)
            ms_cur = float(c_cur.max())
            b = lp_jump(a, c_cur, ms_cur * 1.001, K=20, max_time=20)
            if b is not None:
                t_blend = 0.3
                a_new = (1 - t_blend) * a + t_blend * b
                a_new = np.maximum(a_new, 0)
                a_new /= a_new.sum()
                ns = float(fft_conv(a_new).max())
                if ns < ms_cur:
                    a = a_new
                # If LP didn't help, just continue
        
        # Restart if stagnant
        if iters_since_restart > max_stagnant:
            noise = 1.0 + 0.03 * rng.standard_normal(n)
            a = best_a * np.maximum(noise, 0.1)
            a /= a.sum()
            iters_since_restart = 0
            eta = min(eta * 1.5, 5.0)
            K = min(K + 10, 150)
    
    result = [float(x) for x in best_a]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0/n] * n
    return result

1. The `lp_jump` function creates a sparse matrix with potentially many entries. For K=20 and n=6962, each row of the constraint matrix has about n/2 non-zeros (since each position k has about n/2 valid j's on average). So the total number of non-zeros is about 20 * 3500 = 70000. This should be fine for HiGHS.

2. The `subgrad` function loops over K positions. For K=40 and n=6962, each iteration takes about 2*n operations (creating the index array and doing the multiply-add). Total: 40 * 2 * 7000 = 560,000 operations. In numpy, this should take about 1-2ms.

3. The FFT convolution takes about n*log(n) = 7000*13 ≈ 91000 operations, or about 1ms.

So each iteration takes about 3-5ms. In 1400 seconds, that's about 300,000-500,000 iterations. The LP jump adds maybe 1-5 seconds per call, and is done every 500 iterations, so it's a minor overhead.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 25.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        parent = np.ones(256)
    if parent.ndim != 1 or parent.size < 4:
        parent = np.ones(64)
    n = parent.size
    s = parent.sum()
    if s < 0.01:
        return [1.0] * 64
    a = parent / s

    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1

    def fft_conv(x):
        fx = np.fft.rfft(x, fft_n)
        return np.fft.irfft(fx * fx, fft_n)[:2 * n - 1]

    best_a = a.copy()
    best_s = float(fft_conv(a).max())

    def subgrad(a, c, K):
        top_idx = np.argpartition(c, -K)[-K:]
        g = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            g[lo:hi + 1] += 2.0 * a[k - np.arange(lo, hi + 1)]
        return g

    def lp_jump(a, c, M, K, max_time=20):
        top_idx = np.argpartition(c, -K)[-K:]
        rows, cols, vals = [], [], []
        for ki, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            aj = a[k - js]
            nz = aj > 1e-12
            if nz.any():
                rows.extend([ki] * int(nz.sum()))
                cols.extend(js[nz].tolist())
                vals.extend(aj[nz].tolist())
        if not vals:
            return None
        A = csr_matrix((vals, (rows, cols)), shape=(K, n))
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(K, M),
                          bounds=(0, None), method='highs',
                          options={'time_limit': max_time})
        except Exception:
            return None
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            sb = b.sum()
            if sb > 1e-10:
                return b / sb
        return None

    eta = 1.0
    K = 40
    stagnant = 0

    while time.monotonic() < deadline - 30:
        c = fft_conv(a)
        ms = float(c.max())
        if ms < best_s:
            best_s = ms
            best_a = a.copy()
            stagnant = 0
        stagnant += 1

        g = subgrad(a, c, K)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            continue
        a /= sa

        if stagnant % 100 == 0:
            ns = float(fft_conv(a).max())
            if ns > best_s * 1.004:
                eta *= 0.7
            elif ns < best_s:
                eta = min(eta * 1.2, 10.0)

        if stagnant % 800 == 0 and time.monotonic() < deadline - 60:
            c_cur = fft_conv(a)
            ms_cur = float(c_cur.max())
            b = lp_jump(a, c_cur, ms_cur * 1.002, K=25, max_time=25)
            if b is not None:
                for t_blend in [0.5, 0.3, 0.15]:
                    a_try = (1 - t_blend) * a + t_blend * b
                    a_try = np.maximum(a_try, 0)
                    a_try /= a_try.sum()
                    ns_try = float(fft_conv(a_try).max())
                    if ns_try < ms_cur:
                        a = a_try
                        break

        if stagnant > 5000:
            noise = 1.0 + 0.04 * rng.standard_normal(n)
            a = best_a * np.maximum(noise, 0.05)
            a /= a.sum()
            stagnant = 0
            eta = max(eta * 0.6, 0.3)
            K = min(K + 10, 120)

    result = [float(x) for x in best_a]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
