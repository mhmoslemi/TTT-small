            a[j] = max(0, a[j] + rng.normal(0, 0.002 / np.sqrt(n)))
            a /= a.sum()
        sc = 2.0 * n * float(compute_c(a).max()) / (a.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best = a.copy()

    # Scale best back to comfortable range and validate
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.clip(a_init / max(a_init.sum(), 1e-10), 0.0, 1000.0)

    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        fallback = np.clip(a_init, 0.0, 1000.0)
        if fallback.sum() < 0.01:
            fallback = np.ones(n)
        return [float(x) for x in fallback]

    return [float(x) for x in best]
