import time
import cvxpy as cp

def _conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _score(a):
    """Compute the AC1 score."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def _lp_refine(a, deadline, max_iters=200):
    """LP-based iterative refinement."""
    n = len(a)
    a = a / a.sum()  # normalize to sum 1
    best = a.copy()
    best_score = _score(best)
    
    tau = 0.3
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = np.convolve(best, best).max()
        A = _conv_matrix(best)
        
        b = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b)), [A @ b <= M])
        
        b_val = None
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                b_val = np.asarray(b.value).flatten()
        except Exception:
            pass
        
        if b_val is None:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    b_val = np.asarray(b.value).flatten()
            except Exception:
                pass
        
        if b_val is None:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    b_val = np.asarray(b.value).flatten()
            except Exception:
                pass
        
        if b_val is None or not np.isfinite(b_val).all() or b_val.sum() < 1e-10:
            break
        
        b_val = b_val / b_val.sum()  # normalize to sum 1

        best_local_score = best_score
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
