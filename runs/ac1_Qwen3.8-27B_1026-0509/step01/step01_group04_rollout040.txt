    if sum_a < 0.01:

            if not improved:
                tau *= 0.5
                stuck += 1
                if stuck > 20 or tau < 0.003:
                    break
        return cur

    # Phase 1: LP refine parent
    lp_refine(f0, 300, deadline - 180)

    # Phase 2: Try other initializations and lengths
    configs = []
    n0 = len(f0)
    for n_try in sorted(set([n0, 200, 300, 500, 700, 1000, 1500, 2000, 3000])):
        if n_try > 5000:
            continue
        configs.append((n_try, 'ones'))
        configs.append((n_try, 'rand'))
        configs.append((n_try, 'cos'))

    for n_try, init in configs:
        if time.monotonic() > deadline - 40:
            break
        j = np.arange(n_try)
        if init == 'ones':
            f2 = np.ones(n_try)
        elif init == 'rand':
            f2 = rng.random(n_try) + 0.05
        else:
            f2 = 0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n_try) / n_try)
        try_update(f2)
        # Quick refinement
        max_it = 80 if n_try <= 500 else (40 if n_try <= 1500 else 20)
        lp_refine(f2, max_it, deadline - 40)

    # Phase 3: Uniformization heuristic on best
    if best is not None and time.monotonic() < deadline - 20:
        n_b = len(best)
        f_uni = np.full(n_b, best.sum() / n_b)
        try_update(f_uni)
        # Blend toward uniform
        for blend in [0.3, 0.5, 0.7, 0.9]:
            if time.monotonic() > deadline - 20:
                break
            f_blend = (1-blend)*best + blend*f_uni
            try_update(f_blend)

    # Phase 4: Local perturbation polish
    if best is not None and time.monotonic() < deadline - 10:
        n_b = len(best)
        attempts = 0
        while time.monotonic() < deadline - 10 and attempts < 5000:
            attempts += 1
            w = max(1, n_b // 200)
            s = int(rng.integers(0, n_b - w + 1))
            pert = best.copy()
            pert[s:s+w] *= np.exp(rng.normal(0, 0.15, w))
            try_update(pert)
            # Also try zeroing a segment
            s2 = int(rng.integers(0, n_b - w + 1))
            pert2 = best.copy()
            pert2[s2:s2+w] *= 0.5
            try_update(pert2)

    # Final validation
    if best is None:
        best = np.ones(400)
    final_list = [float(x) for x in best]
    fs = float(evaluate_sequence(final_list))
    if not np.isfinite(fs):
        final_list = [1.0] * 400
    return final_list
