    if time.monotonic() < deadline - 5:
        for _ in range(20):
            if time.monotonic() > deadline:
                break
            n = len(best_seq)
            pert = best_seq * np.exp(rng.normal(0, 0.15, n))
            pert = np.clip(pert, 1e-10, None)
            pert = pert / pert.sum()
            r = lp_refine(pert, min(20.0, (deadline - time.monotonic()) * 0.2), rng)
            try_seq(r)

    # Final validation
    out = np.clip(best_seq, 0.0, 1000.0).tolist()
    fs = evaluate_sequence(out)
    if not np.isfinite(fs) or fs <= 0:
        out = [1.0] * 6
    return out
