```python
def gradient_step(f, n, step_size):
    # Compute full convolution
    conv = np.convolve(f, f)  # length 2n-1
    k_star = np.argmax(conv)
    M = conv[k_star]
    S = f.sum()
    
    # Compute gradient
    grad = np.full(n, -4*n*M / S**3)  # default for invalid indices
    valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
    grad[valid] = 4*n*f[k_star - np.arange(n)[valid]] / S**2 - 4*n*M / S**3
    
    # Gradient descent step with projection
    f_new = f - step_size * grad
    f_new = np.maximum(f_new, 0)
    return f_new
```

The projection onto {f >= 0, sum f = 1}:
Given g (unconstrained update), find the closest point in the simplex.
Algorithm:
1. Sort g in decreasing order: g_(1) >= g_(2) >= ... >= g_(n)
2. Find the largest m such that g_(m) + 1 + (1/m) * sum_{j=1}^{m} g_(j) > 0
3. tau = (1 + sum_{j=1}^{m} g_(j)) / m
4. Projected point: max(g - tau, 0)

This is O(n log n) per step.

```python
import numpy as np
import time
import cvxpy as cp

def _score(f):
    n = len(f)
    conv = np.convolve(f, f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * conv.max() / (S * S)

def _simplex_project(g):
    """Project g onto {x >= 0, sum x = 1}."""
    # Using the standard algorithm
    u = np.sort(g)[::-1]
    n = len(u)
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(g - theta, 0)

def _build_conv_matrix(f, n):
    t = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = t - j
    valid = (d >= 0) & (d < n)
    return np.where(valid, f[np.clip(d, 0, n-1)], 0.0)

            w = np.exp(beta * c)
            w_sum = w.sum()
            w_norm = w / w_sum  # sum to 1
            
            # grad[i] = 4n * sum_k w_norm[k] * f[k-i] (valid)
            # = 4n * (B @ f)[i] where B[i,k'] = w_norm[i+k']
            B = w_norm[np.arange(n)[:, None] + np.arange(n)[None, :]]
            grad = 4.0 * n * (B @ f)
```

```python
import numpy as np
import time
import cvxpy as cp


def _score(f):
    n = len(f)
    conv = np.convolve(f, f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * conv.max() / (S * S)


def _simplex_project(g):
    u = np.sort(g)[::-1]
    n = len(u)
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(g - theta, 0)


def _build_conv_matrix(f, n):
    t = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = t - j
    valid = (d >= 0) & (d < n)
    return np.where(valid, f[np.clip(d, 0, n - 1)], 0.0)


def _soft_gd(f, n, max_iters, deadline, beta=5.0):
    f = f / f.sum()
    sc = _score(f)
    alpha = 0.5
    no_improve = 0
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        conv = np.convolve(f, f)
        c = conv - conv.max()
        w = np.exp(beta * c)
        w_norm = w / w.sum()
        B = w_norm[np.arange(n)[:, None] + np.arange(n)[None, :]]
        grad = 4.0 * n * (B @ f)
        improved = False
        for _tr in range(5):
            g = f - alpha * grad
            f_t = _simplex_project(g)
            sc_t = _score(f_t)
            if sc_t < sc:
                f, sc = f_t, sc_t
                no_improve = 0
                alpha = min(alpha * 1.02, 2.0)
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            alpha = 0.5
            if no_improve > 200:
                break
    return f, _score(f)


def _lp_step(f, n):
    f_n = f / f.sum()

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate") and g.value is not None:
                g_val = np.asarray(g.value).flatten()
                if np.isfinite(g_val).all() and g_val.sum() > 1e-10:
                    return g_val
        except Exception:
            continue
    return None
```
