More precisely, if we define $a^{(k)}$ as the vector with $(a^{(k)})_j = a_{k-j}$ for $j = \max(0, k-n+1), \ldots, \min(n-1, k)$ and 0 elsewhere, then $\nabla_a b_k = 2 a^{(k)}$.

This is very clean. The $k$-th gradient is just a reversed and shifted copy of $a$.

For the Frank-Wolfe method:
- Compute $g = \sum_{k \in K} p_k \cdot 2a^{(k)}$ where $K$ is the set of maximizing indices and $p_k$ are weights (e.g., uniform over argmax, or from softmax).
- Find $s = e_{j^*}$ where $j^* = \arg\min_j g_j$.
- Update: $a \leftarrow a + \gamma(e_{j^*} - a)$.

This is very efficient: each iteration is $O(n^2)$ (compute all $b_k$) plus $O(n)$ for the gradient and update.

                if at.sum() < 1e-10:
                    continue
                Mt = np.convolve(at, at).max()
                if Mt < best_M - 1e-12:
                    best_M, best_t = Mt, t
            if best_t > 0:
                a = (1 - best_t) * a + best_t * g
                a = np.clip(a, 0, None)
                a = a / a.sum()
                stall = 0
            else:
                stall += 1
            update_best(a)
            if stall > 30:
                a = a * np.exp(rng.normal(0, 0.4, n))
                a = np.clip(a, 1e-10, None)
                a = a / a.sum()
                stall = 0
        return a

    def smooth_polish(a, beta=30.0, max_iter=300):
        a0 = np.clip(a, 1e-10, None).copy()
        a0 = a0 / a0.sum()
        x0 = np.log(a0)

        def obj_grad(x):
            lg = x - x.max()
            e = np.exp(lg)
            aa = e / e.sum()
            b = np.convolve(aa, aa)
            bb = beta * b
            mb = bb.max()
            p = np.exp(bb - mb)
            ps = p.sum()
            f = (mb + np.log(ps)) / beta
            # grad_a = 2 * sum_k p_k * a^(k)
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                pk = p[k] / ps
                if pk < 1e-14:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                for j in range(lo, hi):
                    grad_a[j] += 2.0 * pk * aa[k - j]
            # dF/dx_i = sum_j grad_a_j (a_j delta_ji - a_j a_i)
            grad_x = grad_a * aa - aa * np.dot(grad_a, aa)
            return f, grad_x

        try:
            res = minimize(obj_grad, x0, jac=True, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'gtol': 1e-8})
            lg = res.x - res.x.max()
            an = np.exp(lg)
            an = an / an.sum()
            return np.clip(an, 1e-10, None)
        except:
            return a0

    # Generate diverse initializations
    cands = []
    if best_a is not None:
        cands.append(best_a / best_a.sum())
    for _ in range(4):
        a = np.abs(rng.normal(1.0, 0.6, n)) + 0.05
        cands.append(a / a.sum())
    # Structured
    a2 = np.ones(n)
    a2[n // 5:4 * n // 5] = 0.3 + rng.random() * 0.3
    cands.append(a2 / a2.sum())
    a3 = 1.0 + 0.5 * np.sin(2 * np.pi * rng.integers(1, 5) * np.arange(n) / n) + 0.2 * rng.standard_normal(n)
    cands.append(np.clip(a3, 0.1, None) / np.clip(a3, 0.1, None).sum())

    for a in cands:
        if time.monotonic() > deadline - 15:
            break
        a_r = lp_refine(a)
        update_best(a_r)
        if time.monotonic() < deadline - 25:
            a_p = smooth_polish(a_r)
            update_best(a_p)
        if time.monotonic() < deadline - 10:
            a_r2 = lp_refine(a_p, max_iters=300)
            update_best(a_r2)

    if best_a is not None:
        out = np.clip(best_a, 0, 1000)
        if out.size > 0 and np.isfinite(out).all() and out.sum() > 0.01:
            fs = evaluate_sequence(out.tolist())
            if np.isfinite(fs):
                return [float(x) for x in out]
    return [1.0] * n
