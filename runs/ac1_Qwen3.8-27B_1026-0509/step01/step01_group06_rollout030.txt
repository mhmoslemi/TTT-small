```python
def simplex_project(v, n):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find rho such that sum_{j<=rho} (u[j] - tau) = 1
    # where tau = (sum_{j<=rho} u[j] - 1) / rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    rho_star = np.max(rho[cond])
    tau = (cssv[rho_star-1] - 1) / rho_star
    return np.maximum(v - tau, 0)
```

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = int(np.max(rho[cond]))
    tau = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - tau, 0.0)

def _subgrad(a, lr):
    n = len(a)
    b = np.convolve(a, a)
    M = b.max()
    active = np.where(b >= M - 1e-12)[0]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        g[idx] += 2.0 * a[k - idx]
    # Line search
    for _ in range(50):
        a_new = _proj_simplex(a - lr * g / (M + 1e-15))
        M_new = np.convolve(a_new, a_new).max()
        if M_new < M:
            return a_new, M_new
        lr *= 0.5
    return a, M
