from scipy.fft import fft, ifft
import time

def _conv(p):
    n = len(p)
    s = 1 << (2 * n - 2).bit_length()
    F = fft(p, s)
    return ifft(F * F).real[:2 * n - 1]

def _proj(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    ar = np.arange(1.0, n + 1)
    m = u > (cs - s) / ar
    k = int(m.sum())
    if k == 0:
        return np.full(n, s / n)
    return np.maximum(v - (cs[k - 1] - s) / k, 0.0)
