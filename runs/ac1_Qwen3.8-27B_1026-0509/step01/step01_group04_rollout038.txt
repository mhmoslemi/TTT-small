    t = f[i]
    conv = np.convolve(f, f)
    sum_f = f.sum()

    affected = []
    for m in range(2*n-1):
        di = m - i
        if 0 <= di < n:
            if m == 2*i:
                affected.append((m, 'quad'))  # t^2 term
            else:
                affected.append((m, 'linear', f[di]))  # 2*t*x[di] term

    affected_set = set(m for m, *rest in [(a[0],) + tuple(a[1:]) for a in affected])

    mask = np.ones(2*n-1, dtype=bool)
    for m, *rest in affected:
        mask[m] = False
    unchanged_max = conv[mask].max() if mask.any() else 0
    
    def new_max(tp):
        max_val = unchanged_max
        for m, *rest in affected:
            if len(rest) == 0 or rest[0] == 'quad':
                # m = 2*i: new value is conv[m] - t^2 + tp^2
                val = conv[m] - t*t + tp*tp

                di = m - i
                val = conv[m] + 2*(tp-t)*f[di]
            max_val = max(max_val, val)
        max_val = max(max_val, 0)  # ensure non-negative
        return max_val
    
    def ac1(tp):
        new_sum = sum_f + (tp - t)
        if new_sum < 0.01:
            return np.inf
        return 2*n*new_max(tp) / (new_sum*new_sum)
    
    # Minimize ac1(tp) over tp >= 0
    # Use golden section or bounded Brent
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(ac1, bounds=(0, max(10*t, f.max()*2, 1.0)), method='bounded',
                         options={'xatol': 1e-10})
    return res.x

$g_m(x^{(0)}) + 2 \sum_k (x_k - x_k^{(0)}) x_{m-k}^{(0)} \leq M$

Simplifying: $g_m(x^{(0)}) + 2 \sum_k x_k x_{m-k}^{(0)} - 2 g_m(x^{(0)}) \leq M$

$= 2 (x * x^{(0)})_m - g_m(x^{(0)}) \leq M$

$$\min M \quad \text{s.t.} \quad 2(A_{x^{(0)}} x)_m - c_m \leq M \quad \forall m, \quad x \geq 0, \quad \sum x = 1$$
