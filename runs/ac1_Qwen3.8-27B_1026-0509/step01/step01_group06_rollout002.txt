import numpy as np
from scipy.optimize import linprog
import time


def _score_norm(a):
    """Score assuming sum(a)=1, a>=0. Returns 2n*max_b."""
    return 2.0 * len(a) * np.convolve(a, a).max()


def _lp_step(a, M):
    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        js = np.arange(i, i + L)
        valid = (js < L)
        A[valid, i] = a[js[valid] - i]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(L, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)


def _refine_grad(a, deadline, rng):
    n = len(a)
    a = np.clip(a.copy(), 1e-10, None)
    a /= a.sum()
    best_local = a.copy()
    best_sc = _score_norm(a)
    alpha = 0.05
    beta = 30.0
    stall = 0

            break
        b = np.convolve(a, a)
        M = b.max()
        b_shift = b - M
        w = np.exp(beta * b_shift)
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws
        grad = 2.0 * np.correlate(w, a, mode='valid')
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        improved = False

            step = alpha * frac * grad / gn
            a_new = a - step
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new /= s
            sc = _score_norm(a_new)
            if sc < best_sc - 1e-12:
                a = a_new
                best_local = a.copy()
                best_sc = sc
                improved = True
                break
        if improved:
            stall = 0

                alpha = 0.05

                break

def _refine_lp(a, deadline, rng):

def _make_inits(n, rng, parent):

    alt = np.where(np.arange(n) % 2 == 0, 1.4, 0.6)
    inits.append(alt / alt.sum())
    for k in (1, 2, 3, 5, 7):
        idx = np.arange(n)
        v = 1.0 + 0.85 * np.sin(2 * np.pi * k * idx / n + rng.uniform(0, 2*np.pi))
        v = np.clip(v, 0.02, None)
        inits.append(v / v.sum())
    for k in (1, 2, 3):
        idx = np.arange(n)
        v = 1.0 - 0.75 * np.cos(2 * np.pi * k * idx / n)
        v = np.clip(v, 0.02, None)
        inits.append(v / v.sum())
    for _ in range(4):
        v = np.abs(rng.normal(1.0, 0.8, n)) + 0.05
        inits.append(v / v.sum())
    # Laplacian-smoothed random
    for _ in range(2):
        v = np.abs(rng.normal(0, 1, n)) + 0.3
        for _ in range(3):
            v = (v + np.roll(v, 1)) / 2
        inits.append(v / v.sum())
    if parent is not None:
        if parent.size >= n:
            inits.append(parent[:n].copy() / parent[:n].sum())
        elif parent.size > 1:
            p = np.interp(np.linspace(0, parent.size - 1, n), np.arange(parent.size), parent)
            p = np.clip(p, 0.02, None)
            inits.append(p / p.sum())
    return inits
