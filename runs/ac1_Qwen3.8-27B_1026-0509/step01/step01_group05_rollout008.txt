    # Polish: hard-max on top peaks
    a = best.copy()
    for it in range(3000):
        if time.monotonic() > deadline - 5.0:
            break
        a = np.maximum(0, a)
        a = a / a.sum() * n
        b = autocorr(a)
        S = a.sum()
        sc = 2.0 * n * b.max() / (S * S)
        if sc < best_score:
            best_score = sc
            best = a.copy()

        topk = np.argsort(b)[-7:]
        spread = b.max() - b.min()
        wts = np.exp(-(b.max() - b[topk]) / max(spread * 0.15, 1e-10))
        wts /= wts.sum()
        w_full = np.zeros(conv_len)
        w_full[topk] = wts
        grad = wgrad(w_full, a)
        grad -= grad.mean()
        step = 0.04 * (1.0 + 0.5 * (it % 50) / 50.0)
        a = np.maximum(0, a - step * grad)

    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs) or fs > 2.0:
        return [float(x) for x in parent]
    return [float(x) for x in best]
