    best_a = a.copy()
    best_sc = score(best_a)
    
    def ipm_iter(a, M):

        # (a*c)_k = sum_i a_i * c_{k-i} for valid i
        # Constraint matrix: row k has a_i at position i where 0 <= k-i < n
        # i.e., i in [max(0, k-n+1), min(n-1, k)]

        rows = []
        cols = []
        vals = []
        for k in range(2*n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[i])
        
        A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M)
        c_obj = -np.ones(n)  # minimize -sum(c) = maximize sum(c)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')

            return a
        c = res.x
        c_sum = c.sum()
        if c_sum < 1e-15:
            return a
        return c / c_sum
    
    # IPM iterations
    alpha = 0.5

        if time.monotonic() > deadline - 60:
            break
