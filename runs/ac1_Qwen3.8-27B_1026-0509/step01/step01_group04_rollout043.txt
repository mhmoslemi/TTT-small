import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = np.inf
    
    def update_best(f, sc):
        nonlocal best_seq, best_score
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()
    
    def fast_score(f):
        """Compute the objective without the overhead of evaluate_sequence."""
        n = len(f)
        s = f.sum()
        if s < 0.01:
            return np.inf
        b = np.convolve(f, f)
        return float(2.0 * n * b.max() / (s * s))
    
    # Try multiple initializations
    candidates = []
    
    # height_sequence_1
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        if len(hs) > 0 and np.isfinite(hs).all():
            candidates.append(hs)
    except:
        pass
    
    # Ramps
    for n in [300, 400, 500, 600, 800, 1000, 1200]:
        j = np.arange(n, dtype=np.float64)
        candidates.append(j + 1.0)
    
    # Power ramps
    for n in [400, 600, 800]:
        j = np.arange(n, dtype=np.float64)
        for p in [0.5, 1.5, 2.0]:
            candidates.append((j + 1.0) ** p)
    
    # Evaluate
    for f in candidates:
        if time.monotonic() > deadline - 60:
            break
        f = np.clip(f, 0, 1000)
        sc = fast_score(f)
        update_best(f, sc)
    
    # Local optimization
    if best_seq is not None and time.monotonic() < deadline - 20:
        f = best_seq.copy()
        sc = best_score
        n = len(f)
        
        while time.monotonic() < deadline - 10:
            # Random window perturbation
            w = int(rng.integers(1, max(2, n // 80) + 1))
            s = int(rng.integers(0, n - w + 1))
            sigma = 0.05 + 0.1 * (1.0 - min(1.0, (time.monotonic() - (deadline - float(budget_s))) / float(budget_s)))
            factor = np.exp(rng.normal(0, sigma, size=w))
            new_f = f.copy()

            new_sc = fast_score(new_f)
            if np.isfinite(new_sc) and new_sc < sc:
                f = new_f
                sc = new_sc
                update_best(f, sc)
    
    # Final validation with the exact evaluator
    if best_seq is None:

        best_seq = np.arange(n, dtype=np.float64) + 1.0
    
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
