        b_conv = np.convolve(a, a)
        M = b_conv.max()
        if M < 1e-15:
            return a
        active = np.where(b_conv >= M * 0.999)[0]
        grad = np.zeros(n)
        for k in active:
            i_s = max(0, k - n + 1)
            i_e = min(n - 1, k)
            idx = np.arange(i_s, i_e + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-10:
            return a
        return a_new / sn

    def refine(a, max_time):
        n = len(a)
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 1e-10:
            return a
        a /= s
        try_score(a)

        t = 0.4
        stall = 0
        start = time.monotonic()
        prev_sc = try_score(a)

        while time.monotonic() - start < max_time:
            if time.monotonic() > deadline - 20:
                break
            tl = min(2.0, max(0.1, deadline - time.monotonic() - 20))

            b_conv = np.convolve(a, a)
            M = b_conv.max()
            A = build_A(a)
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': tl})

            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                bs = b.sum()
                if bs > 1e-10:
                    b /= bs
                    cur_sc = try_score(a)
                    best_a_local = a.copy()
                    best_sc_local = cur_sc
                    for t_try in [t, t * 0.5, 0.8, 1.0]:
                        t_try = min(1.0, max(0.01, t_try))
                        a_try = np.maximum((1 - t_try) * a + t_try * b, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        sc = float(evaluate_sequence(a_try.tolist()))
                        if np.isfinite(sc) and sc < best_sc_local:
                            best_sc_local = sc
                            best_a_local = a_try
                    a = best_a_local
                    new_sc = try_score(a)
                    if new_sc < prev_sc - 1e-8:
                        stall = 0
                        t = min(1.0, t * 1.05)
                    else:
                        stall += 1
                        t = max(0.05, t * 0.8)
                    prev_sc = new_sc
                else:
                    stall += 5
            else:
                stall += 3

            if stall > 12:
                for _ in range(30):
                    if time.monotonic() > deadline - 20:
                        break
                    a = subgrad_step(a, max(1.0, n * 0.02))
                    try_score(a)
                a = np.maximum(a + 0.01 * rng.standard_normal(n), 0)
                a /= a.sum()
                try_score(a)
                stall = 0
                t = 0.4
                prev_sc = try_score(a)

        return a

    inits = []
    try:
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if p.size > 0 and np.isfinite(p).all():
            inits.append((p.copy(), 500.0))
    except:
        pass

    n = 200
    x = np.arange(n, dtype=np.float64)
    inits.append((0.5 + 0.3 * np.cos(2 * np.pi * 3 * x / n) + 0.1, 200.0))
    inits.append((np.exp(-((x - n / 2) / (n / 5)) ** 2), 150.0))
    inits.append((rng.random(n) + 0.05, 120.0))
    v = np.zeros(n)
    v[:n // 4] = 1.0
    v[3 * n // 4:] = 1.0
    inits.append((v, 80.0))
    inits.append((0.3 + 0.4 * np.abs(np.sin(np.pi * x / n)), 80.0))

    for n_try in [300]:
        inits.append((rng.random(n_try) + 0.05, 60.0))

    for a0, mt in inits:
        if time.monotonic() > deadline - 40:
            break
        a = refine(a0, mt)

    if best is None:
        best = np.ones(100)

    if time.monotonic() < deadline - 15:
        for _ in range(2000):
            if time.monotonic() > deadline - 15:
                break
            i = rng.integers(0, best.size)
            for delta in [0.005, -0.005, 0.01, -0.01, 0.02]:
                pert = best.copy()
                pert[i] = max(0.0, pert[i] + delta)
                if pert.sum() > 0.01:
                    try_score(pert)

    result = [float(x) for x in np.clip(best, 0, 1000)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 100
    return result
