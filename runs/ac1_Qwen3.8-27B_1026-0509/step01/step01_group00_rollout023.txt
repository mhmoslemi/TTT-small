But if the max is at lag n-1 (the center), we need to decrease a_j for j near the center AND increase a_j for j near the edges (to maintain the sum). The multiplicative update handles this implicitly: it decreases where g is large and (after renormalization) the remaining mass is redistributed proportionally to the old values.

So the multiplicative update is actually fine - it automatically does the right thing by shifting mass from high-gradient to low-gradient regions.

But there's a subtlety: the multiplicative update preserves the "ratio" structure of z. If z is already quite uniform, the update will keep it fairly uniform. To make more dramatic changes, we'd need much larger steps.

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    try:

    if parent.ndim != 1 or parent.size < 4:
        parent = np.ones(64)
    n = parent.size
    s = parent.sum()
    if s < 0.01:
        return [1.0] * 64
    parent_n = parent / s

    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1
    fft_n2 = 1
    while fft_n2 < 3 * n - 2:
        fft_n2 <<= 1

    def fft_conv(x):
        fx = np.fft.rfft(x, fft_n)
        return np.fft.irfft(fx * fx, fft_n)[:2 * n - 1]

    def score(z):
        return float(fft_conv(z).max())

    def smax_grad(z, T):
        c = fft_conv(z)
        cm = c.max()
        x = T * (c - cm)
        np.clip(x, -80, 0, out=x)
        ex = np.exp(x)
        s_ex = ex.sum()
        w = ex / s_ex
        z_r = z[::-1]
        conv = np.fft.irfft(np.fft.rfft(w, fft_n2) * np.fft.rfft(z_r, fft_n2), fft_n2)
        g = 2.0 * conv[n - 1:2 * n - 1]
        return cm, g

    def optimize(z0, max_iters, T, lr_init, pert_every=5000, pert_sigma=0.01):
        z = z0.copy()
        bz = z.copy()
        bs = score(z)
        lr = lr_init
        iters_since_improve = 0
