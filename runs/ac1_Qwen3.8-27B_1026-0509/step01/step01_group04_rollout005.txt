```python
def _gradient_polish(f, S, n, max_iters, deadline, rng):
    best_f = f.copy()
    best_sc = float(evaluate_sequence(f.tolist()))
    step = 0.01 * S / n
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        ff = np.convolve(f, f)
        t_star = int(np.argmax(ff))
        
        # Gradient
        grad = np.zeros(n)
        for k in range(n):
            idx = t_star - k
            if 0 <= idx < n:
                grad[k] += f[idx]
                if idx != k:
                    grad[k] += f[idx]
        
        # Transfer mass from high-grad to low-grad
        order = np.argsort(grad)
        num_move = max(1, n // 20)
        src = order[-num_move:]  # expensive
        dst = order[:num_move]   # cheap
        
        total_src = f[src].sum()
        if total_src < 1e-10:
            continue
        
        # Move a fraction
        frac = step
        move = min(frac, total_src/2)
        f[src] -= move / len(src)
        f[dst] += move / len(dst)
        f = np.clip(f, 0, None)
        
        sc = float(evaluate_sequence(f.tolist()))
        if sc < best_sc:
            best_f = f.copy()
            best_sc = sc
        else:
            f = best_f.copy()
            step *= 0.9
    
    return best_f, best_sc
```
