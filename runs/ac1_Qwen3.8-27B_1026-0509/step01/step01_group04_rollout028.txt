            best_x = x.copy()
    
    def coord_descent(x, deadline, max_sweeps=100000):
        n = len(x)
        x = x.copy().astype(np.float64)
        ac = np.convolve(x, x)
        S = x.sum()
        
        for sweep in range(max_sweeps):
            if time.monotonic() > deadline:
                break
            max_change = 0.0
            for k in range(n):
                old = x[k]
                m_s = k
                m_e = min(k + n - 1, 2 * n - 2)
                
                # Unchanged max
                if m_s == 0 and m_e == 2 * n - 2:
                    max_unch = -1e30
                elif m_s == 0:
                    max_unch = ac[m_e + 1:].max()
                elif m_e == 2 * n - 2:
                    max_unch = ac[:m_s].max()
                else:
                    max_unch = max(ac[:m_s].max(), ac[m_e + 1:].max())
                
                m_range = np.arange(m_s, m_e + 1)
                j_idx = m_range - k
                is_q = (m_range == 2 * k)
                lins = 2.0 * x[j_idx].copy()
                lins[is_q] = 0.0
                quads = is_q.astype(np.float64)
                consts = ac[m_s:m_e + 1] - lins * old - quads * old * old
                
                # Minimize max(consts + lins*v + quads*v^2, max_unch) over v >= 0
                lo, hi = 0.0, max(2.0 * old + 1.0, 1.0)
                for _ in range(50):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = max((consts + lins * m1 + quads * m1 * m1).max(), max_unch)
                    f2 = max((consts + lins * m2 + quads * m2 * m2).max(), max_unch)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                new = max(0.0, (lo + hi) / 2.0)
                
                if abs(new - old) > 1e-14:
                    dc = new - old
                    ac[m_s:m_e + 1] += lins * dc + quads * (new * new - old * old)
                    x[k] = new
                    max_change = max(max_change, abs(new - old))
            
            # Renormalize to keep sum constant (scale invariance)
            # Actually, let's not renormalize - just track the score
            sc = 2.0 * n * ac.max() / (x.sum()**2)
            if sc < best_score:
                best_score = sc
                best_x = x.copy()
            if max_change < 1e-14:
                break
        return x, 2.0 * n * ac.max() / (x.sum()**2)
