                conv = np.convolve(f, f)
                sf, a = f.sum(), f[i]
                unc_max_lo = conv[:i].max() if i > 0 else 0.0
                unc_max_hi = conv[i + n:].max() if i + n < 2 * n - 1 else 0.0
                unc_max = max(unc_max_lo, unc_max_hi)

                def obj(t):
                    t = 0.0 if t < 0 else (1000.0 if t > 1000.0 else t)
                    s = sf + t - a
                    if s < 0.01:
                        return 1e10
                    d = t - a
                    if abs(d) < 1e-14:
                        return 2.0 * n * conv.max() / (s * s)
                    vals = conv[i:i + n].copy()
                    vals += 2.0 * d * f
                    vals[i] = conv[2 * i] + t * t - a * a
                    mb = max(unc_max, vals.max())
                    return 2.0 * n * mb / (s * s)

                try:
                    r = minimize_scalar(obj, bounds=(0.0, 1000.0), method='bounded',
                                        options={'maxiter': 50, 'xatol': 1e-9})
                    nv = float(np.clip(r.x, 0.0, 1000.0))
                    if abs(nv - a) > 1e-10:
                        f[i] = nv
                        ch = True
                except Exception:
                    pass
            sc = float(evaluate_sequence(f.tolist()))
            if np.isfinite(sc) and sc < bs:
                bs = sc
                bf = f.copy()
            if not ch:
                break
        return bf, bs

    best_f = get_dtype_f()
    best_f = best_f / max(best_f.sum(), 1e-10)
    best_s = float(evaluate_sequence(best_f.tolist()))

    # Phase 1: Thorough CD on parent
    f1, s1 = cd(best_f, 30, dl - 200)
    if s1 < best_s:
        best_f, best_s = f1, s1

    # Phase 2: Diversified initializations
    if time.monotonic() < dl - 150:
        inits = [
            (400, lambda j: np.sin(np.pi * (j + 0.5) / len(j))),
            (500, lambda j: np.sin(np.pi * (j + 0.5) / len(j))),
            (600, lambda j: np.sin(np.pi * (j + 0.5) / len(j))),
            (350, lambda j: np.sin(np.pi * (j + 0.5) / len(j))),
            (700, lambda j: np.sin(np.pi * (j + 0.5) / len(j))),
            (400, lambda j: 1.0 - np.abs(2.0 * j / len(j) - 1.0)),
            (400, lambda j: np.exp(-5.0 * (j / (len(j) - 1) - 0.5) ** 2)),
            (500, lambda j: np.exp(-5.0 * (j / (len(j) - 1) - 0.5) ** 2)),
            (400, lambda j: np.cos(np.pi * (j + 0.5) / (2 * len(j))) ** 2),
            (600, lambda j: np.cos(np.pi * (j + 0.5) / (2 * len(j))) ** 2),
        ]
        for n, fn in inits:
            if time.monotonic() > dl - 60:
                break
            j = np.arange(n, dtype=np.float64)
            fi = fn(j)
            ss = float(evaluate_sequence(fi.tolist()))
            if ss >= best_s:
                continue
            fi = fi / fi.sum()
            f2, s2 = cd(fi, 20, dl - 60)
            if s2 < best_s:
                best_f, best_s = f2, s2

    # Phase 3: Perturbation + refinement loop
    if time.monotonic() < dl - 30:
        for _ in range(5000):
            if time.monotonic() > dl - 25:
                break
            n = len(best_f)
            p = best_f.copy()
            w = max(3, n // 60)
            s0 = int(rng.integers(0, n - w + 1))
            p[s0:s0 + w] *= np.exp(rng.normal(0, 0.2, size=w))
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(p.tolist()))
            if np.isfinite(sc) and sc < best_s:
                best_f, best_s = p, sc
                f3, s3 = cd(p, 8, dl - 25)
                if s3 < best_s:
                    best_f, best_s = f3, s3

    # Phase 4: Final polish CD
    if time.monotonic() < dl - 5:
        f4, s4 = cd(best_f, 10, dl - 5)
        if s4 < best_s:
            best_f, best_s = f4, s4

    best_f = np.clip(best_f, 0.0, 1000.0)
    final_sc = float(evaluate_sequence([float(x) for x in best_f]))
    if not np.isfinite(final_sc):
        nf = np.sin(np.pi * (np.arange(400, dtype=np.float64) + 0.5) / 400)
        return [float(x) for x in nf]
    return [float(x) for x in best_f]
