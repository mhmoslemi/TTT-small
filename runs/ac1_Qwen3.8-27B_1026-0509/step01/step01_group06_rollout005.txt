```python
def grad_smooth_max_conv(theta, beta=50):
    a = np.exp(theta - theta.max())
    a = a / a.sum()
    n = len(a)
    c = np.convolve(a, a)
    mc = c.max()
    w = np.exp(beta * (c - mc)) / np.exp(beta * (c - mc)).sum()
    # gc_j = 2 * sum_k w_k * a_{k-j} (where valid)
    # This is 2 * (w * a_conv)(j) where w_conv is truncated
    # Actually: gc_j = 2 * sum_k w_k * a_{k-j}
    # Let's define: for each k, the contribution to gc is 2*w_k * (a shifted to align k-j)
    # gc = 2 * sum_k w_k * corr(a, delta_k)
    # = 2 * (w_conv_interacted with a)
    
    # More directly:
    # gc_j = 2 * sum_k w_k * a[(k-j) mod ...] 
    # But we need 0 <= k-j < n
    
    # Vectorized: A[k,j] = a[k-j] if valid
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.where(valid, a[diff.clip(0, n-1)], 0)  # (2n-1, n)
    gc = 2.0 * A.T @ w  # (n,)
    
    grad_theta = gc * a - a * (gc @ a)
    return grad_theta
```
