    bump = np.exp(-0.5 * ((xc - mu) / sigma) ** 2)
    inits.append(bump / bump.sum())
    # Perturbed uniform
    p = np.ones(n) + 0.1 * rng.standard_normal(n)
    p = np.maximum(p, 0.01)
    inits.append(p / p.sum())

    beta_start = 3.0
    iters_per_start = 200000

    for start_idx in range(len(inits)):
        if time.monotonic() > deadline:
            break
        a = inits[start_idx].copy()
        beta = beta_start
        eta = 1.0
        v = np.zeros(n)

        for it in range(iters_per_start):
            if time.monotonic() > deadline:
                break

            # Compute self-convolution via FFT
            A_fft = np.fft.rfft(a, N_fft)
            c = np.fft.irfft(A_fft * A_fft, N_fft)[:2 * n - 1]

            # Softmax weights
            cs = c - c.max()
            ew = np.exp(beta * cs)
            ws = ew.sum()
            if ws < 1e-300:
                w = np.zeros(2 * n - 1)
                w[np.argmax(c)] = 1.0
            else:
                w = ew / ws

            # Gradient: 2 * cross_correlation(w, a)[0:n]
            W_fft = np.fft.rfft(w, N_fft)
            corr = np.fft.irfft(W_fft * np.conj(A_fft), N_fft)
            grad = 2.0 * corr[:n]

            # Nesterov momentum
            v = 0.9 * v + grad
            a = _project_simplex(a - eta * v / (np.linalg.norm(v) + 1e-15))

            # Periodic exact evaluation
            if it % 100 == 0:
                sc = float(evaluate_sequence(a.tolist()))
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            # Adapt
            if it % 300 == 299:
                beta = min(beta * 1.04, 300.0)
                eta *= 0.995

    # Final check
    sc = float(evaluate_sequence(best_a.tolist()))
    if sc < best_score:
        best_score = sc

    return [float(x) for x in best_a]
