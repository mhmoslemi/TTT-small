    update_best(f1_t)

    # Secondary: try a couple other initializations if time allows
    if time.monotonic() < deadline - 180:
        for n_try in [350, 450, 550]:
            if time.monotonic() > deadline - 90:
                break
            j = np.arange(n_try)
            fi = np.sin(np.pi * (j + 0.5) / n_try)
            update_best(fi)
            fr, _ = _refine(fi, deadline - 90)
            fr = _trim(fr)
            update_best(fr)

    # Polish: random perturbations + coordinate tweaks
    if best_seq is not None and time.monotonic() < deadline - 5:
        nb = len(best_seq)
        for _ in range(3000):
            if time.monotonic() > deadline - 5:
                break
            # Block perturbation
            p = best_seq.copy()
            w = max(1, min(8, nb // 80))
            s = int(rng.integers(0, nb - w + 1))
            p[s:s + w] *= np.exp(rng.normal(0, 0.15, size=w))
            p = np.clip(p, 0, None)
            if p.sum() > 0.01:
                update_best(p)
            # Single coordinate
            i = int(rng.integers(0, nb))
            for mult in [0.4, 0.7, 1.3, 1.6, 0.2]:
                p2 = best_seq.copy()
                p2[i] *= mult
                p2 = np.clip(p2, 0, None)
                if p2.sum() > 0.01:
                    update_best(p2)
            # Try trimming the current best
            if _ % 50 == 49:
                tb = _trim(best_seq, 1e-7)
                if len(tb) < nb:
                    update_best(tb)

    if best_seq is None:
        j = np.arange(400)
        best_seq = np.sin(np.pi * (j + 0.5) / 400)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_sc = _score(result)
    if not np.isfinite(final_sc):
        j = np.arange(400)
        result = [float(np.sin(np.pi * (x + 0.5) / 400)) for x in range(400)]

    return result
