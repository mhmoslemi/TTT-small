import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 20.0)
    
    # Get parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        parent = None
    
    if parent is not None and parent.ndim == 1 and parent.size >= 4:
        n = parent.size
        z0 = parent / parent.sum()
    else:
        n = 512
        z0 = np.ones(n) / n
    
    # FFT setup
    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1
    
    def full_conv(z):
        fz = np.fft.rfft(z, fft_n)
        return np.fft.irfft(fz * fz, fft_n)[:2*n-1]
    
    def score(z):
        c = full_conv(z)
        return float(c.max())
    
    def gradient(z, active_lags, weights):
        """Compute weighted gradient of sum_k w_k * (z*z)[k]."""
        g = np.zeros(n)
        for k, w in zip(active_lags, weights):
            if w < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                ms = k - je
                me = k - js
                g[js:je+1] += 2.0 * w * z[ms:me+1][::-1]
        return g
    
    def optimize(z_init, max_iters, lr_init, p_active=10, threshold_frac=0.98):
        z = z_init.copy()
        z = np.maximum(z, 1e-15)
        z /= z.sum()
        lr = lr_init
        best_z = z.copy()
        best_score = score(z)
        stall = 0
        
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            # Compute convolution and find active lags
            c = full_conv(z)
            cx = c.max()
            if cx < best_score:
                best_score = float(cx)
                best_z = z.copy()
                stall = 0
            else:
                stall += 1
            
            # Find active set
            thresh = cx * threshold_frac
            active = np.where(c >= thresh)[0]
            if len(active) > p_active:
                # Keep top p_active
                top_vals = c[active]
                idx = np.argsort(top_vals)[-p_active:]
                active = active[idx]

            active_vals = c[active]
            weights = (active_vals / cx) ** 4  # Sharpen the weighting

            g = gradient(z, active, weights)
            
            # Scale gradient for stability
            g_norm = np.linalg.norm(g)
            if g_norm > 0:
                g = g / g_norm
            
            # Multiplicative update
            z = z * np.exp(-lr * g)
            z = np.maximum(z, 1e-15)
            z /= z.sum()
            
            # Adapt learning rate
            if it % 100 == 99:
                cur_score = score(z)
                if cur_score > best_score * 1.02:
                    lr *= 0.7
                elif cur_score < best_score * 0.99:
                    lr = min(lr * 1.2, 5.0)
            
            # Periodic noise for exploration
            if it % 2000 == 1999:
                noise = rng.normal(0, 0.05, n)
                z = z * np.exp(noise)
                z = np.maximum(z, 1e-15)
                z /= z.sum()
            
            # Early stopping if stalled too long

                break

    candidates = [z0.copy()]

            je = min(n - 1, k)
            if js <= je:
                ms = k - je
                me = k - js
                g[js:je + 1] += 2.0 * w * z[ms:me + 1][::-1]
        return g

    def optimize(z_init, max_iters, lr_start, lr_end, p_top=12, thresh_frac=0.93):
        z = np.maximum(z_init.astype(np.float64), 1e-15)
        z /= z.sum()
        bz = z.copy()
        bs = get_score(z)

        for it in range(max_iters):
            if time.monotonic() > deadline:
                break

            frac = min(1.0, it / max(1, max_iters))
            lr = lr_start * (lr_end / lr_start) ** frac

            c = full_conv(z)
            cx = c.max()

            if cx < bs:
                bs = float(cx)
                bz = z.copy()

            thresh = cx * thresh_frac
            act = np.where(c >= thresh)[0]
            if len(act) > p_top:
                order = np.argsort(c[act])[-p_top:]
                act = act[order]

            av = c[act]
            wts = (av / cx) ** 10

            g = compute_grad(z, act, wts)
            gmax = np.abs(g).max()
            if gmax < 1e-15:
                break
            g /= gmax

            z = z * np.exp(-lr * g)
            z = np.maximum(z, 1e-15)
            sm = z.sum()
            if sm < 1e-12:
                z = bz.copy()
                continue
            z /= sm

            if it % 5000 == 4999:
                z = z * np.exp(rng.normal(0, 0.02, n))
                z = np.maximum(z, 1e-15)
                z /= z.sum()

        return bz, bs

    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    cands = []

    z = np.maximum(parent, 1e-15)
    cands.append(z / z.sum())

    z = 0.5 * (1.0 + np.cos(np.pi * idx / (n - 1)))
    cands.append(z / z.sum())

    for sf in [0.15, 0.25, 0.35, 0.5, 0.7]:
        z = np.exp(-0.5 * (idx - center) ** 2 / (sf * n) ** 2)
        cands.append(z / z.sum())

    d = np.abs(idx - center) + 1.0
    for p in [-0.5, -1.0, -1.5, 0.5, 1.0, 1.5, 2.0]:
        z = d ** p
        cands.append(z / z.sum())

    for fr in [1, 2, 3, 5, 7]:
        z = 1.0 + 0.4 * np.cos(2.0 * np.pi * fr * idx / n)
        z = np.maximum(z, 0.01)
        cands.append(z / z.sum())

    for fr in [2, 3, 5]:
        z = 1.0 + 0.3 * np.sin(np.pi * idx / n) * np.cos(2.0 * np.pi * fr * idx / n)
        z = np.maximum(z, 0.01)
        cands.append(z / z.sum())

    best_z = cands[0].copy()
    best_sc = get_score(cands[0])
    for c in cands[1:]:
        sc = get_score(c)
        if sc < best_sc:
            best_sc = sc
            best_z = c.copy()

    scored = [(get_score(c), i) for i, c in enumerate(cands)]
    scored.sort()

    for sc_init, i in scored:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        iters = int(min(60000, remaining * 400))
        if iters < 500:
            break
        bz, bs = optimize(cands[i].copy(), iters, 1.5, 0.08, p_top=12, thresh_frac=0.93)
        if bs < best_sc:
            best_sc = bs
            best_z = bz.copy()

    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        iters = int(min(100000, remaining * 400))
        bz, bs = optimize(best_z.copy(), iters, 0.6, 0.03, p_top=8, thresh_frac=0.90)
        if bs < best_sc:
            best_sc = bs
            best_z = bz.copy()

    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        iters = int(min(50000, remaining * 400))
        z_pert = best_z * np.exp(rng.normal(0, 0.01, n))
        z_pert = np.maximum(z_pert, 1e-15)
        z_pert /= z_pert.sum()
        bz, bs = optimize(z_pert, iters, 0.4, 0.02, p_top=6, thresh_frac=0.88)
        if bs < best_sc:
            best_sc = bs
            best_z = bz.copy()

    best_z = np.clip(best_z, 0.0, 1000.0)
    if best_z.sum() < 0.01:
        best_z = np.ones(64)
    result = [float(x) for x in best_z]

    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 64

    return result
