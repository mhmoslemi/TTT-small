```python
import numpy as np
from scipy.optimize import linprog

def lp_direction(f, M, top_k=100):
    """Find LP descent direction."""
    n = len(f)
    # Compute autocorrelation
    ac = np.convolve(f, f, mode='full')  # length 2n-1
    # Find top-k peaks
    peaks = np.argsort(ac)[-top_k:]
    # Build constraint matrix: A[row, i] = f[peaks[row] - i] if valid
    A = np.zeros((top_k, n))
    for row, k in enumerate(peaks):
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                A[row, i] = f[idx]
    # LP: max sum(g) s.t. A*g <= M, g >= 0
    # In linprog: min -sum(g) s.t. A*g <= M, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(top_k, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        g = res.x
        return g
    return None
```
