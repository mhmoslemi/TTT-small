def lp_cutting_plane(f, M, max_rounds=100, max_active=200):
    # Prolm: max 1^T b s.t. (f*b)_j <= M for all j, b >= 0
    # Use cutting plane: maintain active constraint set A
    n = len(f)
    
    # Initial active set: where (f*f)_j is largest
    ff = fft_conv(f)
    A = set(np.argsort(ff)[-max_active:])
    
    for round in range(max_rounds):
        # Solve LP with current active set
        # max 1^T b s.t. (f*b)_j <= M for j in A, b >= 0
        # 
        # (f*b)_j = sum_k f_k b_{j-k} <= M
        # 
        # This is a standard LP. Use linprog.
        # Variables: b_0, ..., b_{n-1} (nonnegative)
        # Constraints: for each j in A: sum_k f_k b_{j-k} <= M
        # Objective: max sum(b)
        
        # Build constraint matrix (sparse)
        A_list = sorted(A)
        a = len(A_list)
        
        rows = []
        cols = []
        data = []
        for i, j in enumerate(A_list):
            # (f*b)_j = sum_{k: 0<=j-k<n} f_k b_{j-k}
            # = sum_{m: 0<=m<n, 0<=j-m<n} f_{j-m} b_m
            # For variable b_m: coefficient is f_{j-m} if 0 <= j-m < n
            lo = max(0, j - n + 1)
            hi = min(j, n - 1)
            for m in range(lo, hi + 1):
                k = j - m
                if 0 <= k < n and f[k] > 0:
                    rows.append(i)
                    cols.append(m)
                    data.append(f[k])
        
        A_ub = csr_matrix((data, (rows, cols)), shape=(a, n))
        b_ub = np.full(a, M)
        c_obj = -np.ones(n)  # minimize -sum(b)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            break
        
        b = res.x
        
        # Check for violated constraints
        fb = fft_conv(f, b)  # This gives f*b
        violated = np.where(fb > M * (1 + 1e-6))[0]
        violated = [v for v in violated if v not in A]
        
        if len(violated) == 0:
            # LP is optimal
            break
        
        # Add violated constraints (up to max_active total)
        # Prioritize most violated
        violated_sorted = sorted(violated, key=lambda v: fb[v], reverse=True)
        for v in violated_sorted:
            if len(A) >= max_active:
                # Remove least important constraint? Or just stop.
                break
            A.add(v)
    
    # Normalize b
    bs = b.sum()
    if bs < 0.01:
        return f
    b = b / bs
    return b
