    # Solve: max sum(g) s.t. (a*g)[k] <= M for k in active_constraints, g >= 0
    # active_constraints: array of k values
    n = len(a)
    # Constraint matrix: len(active_constraints) x n
    # C[k_idx, j] = a[j] if 0 <= k - j < n
    from scipy.sparse import lil_matrix
    m = len(active_constraints)
    C = lil_matrix((m, n))
    for idx, k in enumerate(active_constraints):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        C[idx, lo:hi] = a[lo:hi]
    C = C.tocsc()
    
    # Use cvxpy
    g = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(g))
    cons = [C @ g <= M]
    prob = cp.Problem(obj, cons)
    prob.solve(solver=cp.GLOP)
    return g.value
