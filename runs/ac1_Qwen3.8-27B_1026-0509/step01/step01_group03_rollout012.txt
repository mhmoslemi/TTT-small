min max_k conv(a,b)_k s.t. sum(b) = 1, b >= 0

This is a minimax LP (or equivalently, a LP with an epigraph variable):
min t
s.t. conv(a,b)_k <= t for all k
     sum(b) = 1
     b >= 0

This directly minimizes the "worst-case" autocorrelation in the direction a. The solution b* is the direction that, when mixed with a, most effectively reduces the peak.

This LP has 2n-1 + 1 = 2n constraints (plus the equality) and n+1 variables (b_0,...,b_{n-1}, t). For n=200, that's 400 constraints and 201 variables. Very manageable!

def minimax_lp(a, n, deadline):
    """Solve min t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0."""
    M = np.convolve(a, a).max()
    # conv(a,b)_k = sum_i a_i * b_{k-i} (with appropriate bounds)
    # This is linear in b.
    # Build the constraint matrix: for each k, the coefficients of b_j in conv(a,b)_k
    # (A b)_k = sum_j A_{k,j} b_j where A_{k,j} = a_j if k-j is in [0,n-1]... 
    # Actually: conv(a,b)_k = sum_i a_i b_{k-i} where the sum is over valid i.
    # For fixed k: (A b)_k = sum_j a_{k-j} b_j where k-j is in [0, n-1], i.e., j in [k-(n-1), k] intersect [0, n-1]
    
    # Build sparse A
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A b - t <= 0, i.e., [A, -I] [b; t] <= 0
    A_ineq = lil_matrix((2*n-1, n+1))
    A_ineq[:, :n] = A
    A_ineq[:, n] = -1.0  # -t
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    b_ub = np.zeros(2*n-1)
    
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, t free
    
    from scipy.optimize import linprog
    tl = max(1.0, deadline - time.monotonic() - 20)
    res = linprog(c, A_ub=np.asarray(A_ineq.tocsc()), b_ub=b_ub,
                  A_eq=A_eq, b_eq=[1.0], bounds=bounds,
                  method='highs', options={'time_limit': tl})
    
    if res.success:
        return res.x[:n], res.x[n]
    return a.copy(), M
