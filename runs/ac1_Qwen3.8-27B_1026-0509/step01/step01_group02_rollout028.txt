import numpy as np
from scipy.fft import fft, ifft
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    deadline = time.monotonic() + budget_s - 12.0

    try:
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = len(a0)
    except:
        n = 4806
        a0 = np.ones(n) / n

    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 1e-10:
        a0 = np.ones(n) / n
    a0 /= a0.sum()

    corr_size = 1 << (2 * n - 1).bit_length()
    grad_size = 1 << (3 * n - 2).bit_length()

    def compute_corr(x):
        F = fft(x, corr_size)
        return ifft(F * F).real[:2 * n - 1]

    def project_simplex(v):
        u = np.sort(v)[::-1]
        cv = np.cumsum(u) - 1.0
        cond = u - cv / np.arange(1, n + 1) > 0
        if not np.any(cond):
            return np.full(n, 1.0 / n)
        i = int(np.max(np.where(cond)[0])) + 1
        return np.maximum(v - cv[i - 1] / i, 0.0)

    rng = np.random.default_rng(seed)
    candidates = [a0]
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 2.5, 3.0]:
        c = np.arange(1, n + 1, dtype=float) ** p
        candidates.append(c)
    x = np.linspace(0, 1, n)
    candidates.append(np.sin(np.pi * x) ** 2)
    candidates.append(np.exp(-4 * (x - 0.5) ** 2))
    candidates.append(np.exp(-4 * x ** 2))
    candidates.append(np.exp(-4 * (1 - x) ** 2))
    candidates.append(1.0 / (1.0 + 20 * x * (1 - x)))

    best, best_sc = None, np.inf
    for c in candidates:
        c = np.maximum(c, 0.0)
        if c.sum() < 1e-10:
            continue
        c /= c.sum()
        sc = 2.0 * n * compute_corr(c).max()
        if sc < best_sc:
            best_sc, best = sc, c.copy()

    a = best.copy()
    corr = compute_corr(a)

    alpha = 0.5 / n
    vel = np.zeros(n)
    moment = 0.8
    no_improve = 0
    best_sc_prev = best_sc

    while time.monotonic() < deadline:
        cmax = corr.max()
        current_sc = 2.0 * n * cmax

        s_val = 5.0 / cmax
        w = np.exp(s_val * (corr - cmax))
        w /= w.sum()

        w_flip = w[::-1]
        c = ifft(fft(a, grad_size) * fft(w_flip, grad_size)).real[:3 * n - 2]
        grad = 4.0 * n * c[2 * n - 2 : n - 2 : -1]

        vel = moment * vel + (1.0 - moment) * grad
        a_new = project_simplex(a - alpha * vel)

        corr_new = compute_corr(a_new)
        sc_new = 2.0 * n * corr_new.max()

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best = a_new.copy()
            no_improve = 0
            alpha = min(alpha * 1.01, 3.0 / n)
            a = a_new
            corr = corr_new
        elif sc_new <= current_sc * 1.0005:
            a = a_new
            corr = corr_new
            no_improve += 1
        else:
            alpha *= 0.7
            no_improve += 1

        if no_improve > 12000:
            pert = rng.normal(0, 2.0 / n, n)
            a = project_simplex(best + pert)
            corr = compute_corr(a)
            vel[:] = 0.0
            alpha = 0.4 / n
            moment = 0.7
            no_improve = 0

        if no_improve > 40000:
            pert = rng.normal(0, 5.0 / n, n)
            a = project_simplex(best + pert)
            corr = compute_corr(a)
            vel[:] = 0.0
            alpha = 0.5 / n
            moment = 0.75
            no_improve = 0

        if time.monotonic() > deadline - 5.0:
            break

    return [float(v) for v in best]
