Now, the temperature $\tau$ controls the "sharpness" of the weights:
- $\tau$ small → weights concentrated on argmax → like subgradient.
- $\tau$ large → weights more spread → smoother gradient.

A good strategy: start with a moderate $\tau$ (e.g., $\tau = 0.1 \cdot b_{\max}$) and anneal it down.

        return np.real(ifft(A * A, fft_size))[:conv_len]
    
    # Main optimization
    a = parent.copy().astype(np.float64)
    a /= a.sum()
    
    best = a.copy()
    best_b = get_conv(best)
    best_score = float(2.0 * n * best_b.max())
    
    v = np.zeros(n)
    momentum = 0.9

    tau = 0.05  # temperature for soft weights (relative to bmax)

        if time.monotonic() > deadline - 30:
            break
        
        b = get_conv(a)
        bmax = float(b.max())
        score = 2.0 * n * bmax

            best_score = score
            best = a.copy()

        tau_eff = max(1e-10, tau * bmax)
        log_w = b / tau_eff - (b.max() / tau_eff)  # for numerical stability
        w = np.exp(log_w)
        w /= w.sum()
        
        # Compute smoothed gradient: g_j = 2 * sum_m w[j+m] * a[m]
        W = fft(w, fft_size2)
        A = fft(a, fft_size2)
        g = 2.0 * np.real(ifft(W * np.conj(A), fft_size2))[:n]

        eta = 0.5 * bmax / (np.linalg.norm(g) + 1e-15)
        # Cap the step
        eta = min(eta, 0.1)
        
        # Momentum update
        v = momentum * v - eta * g
        a = a + v
        
        # Project to simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a /= s

            tau *= 0.95
        
        # Periodic restart
        if it % 10000 == 5000:
            noise = rng.normal(0, 0.001, n)
            a = np.maximum(a + noise, 0)
            a /= a.sum()
            v = np.zeros(n)

    for it in range(200000):
        if time.monotonic() > deadline - 15:
            break
        
        b = get_conv(a)
        bmax = float(b.max())
        score = 2.0 * n * bmax

            best_score = score
            best = a.copy()

        i_hi = min(n-1, kstar)
        g = np.zeros(n)
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] = 2.0 * a[kstar - idx]

        final_score = float(evaluate_sequence(best.tolist()))
