def _lp_step(f, M, solver=cp.CLARABEL):
    n = len(f)
    # Build constraint matrix: A @ g <= M
    # (f*g)[k] = sum_i f[max(0,k-n+1):min(n,k+1)] * g[...]
    # Actually: (f*g)[k] = sum_i f[i]*g[k-i] for valid i, k-i
    
    # A @ g where A[k, i] = f[k-i] if 0 <= k-i < n else 0
    # But this is the same as: for each k, the k-th row of A is a shifted f
    
    # Build A as a banded matrix
    # Row k of A: A[k, k-i] = f[i] for i in [max(0,k-n+1), min(n-1,k)]
    
    A_data = []
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_data.append((k, i, f[i]))  # Wait, A[k,i] should be f[k-i]
            # (f*g)[k] = sum_i f[i] * g[k-i]
            # So A[k, j] = f[k-j] where j = k-i, i.e., A[k, k-i] = f[i]
            # For valid j: 0 <= j < n and 0 <= k-j < n
            # So A[k, j] = f[k-j] for j in [max(0, k-n+1), min(n-1, k)]
    
    # This is getting complicated. Let me use the conv matrix approach.
    
    t_idx = np.arange(2*n-1)[:, None]
    k_idx = np.arange(n)[None, :]
    d = t_idx - k_idx
    valid = (d >= 0) & (d < n)
    A = np.where(valid, f[np.clip(d, 0, n-1)], 0.0)
    
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
    prob.solve(solver=solver, verbose=False)
    
    if prob.status in ("optimal", "optimal_inaccurate") and g.value is not None:
        return np.asarray(g.value).flatten()
    return None
