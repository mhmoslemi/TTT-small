    def conv(a, b):
        return ifft(fft(a, ffsize) * fft(b, ffsize)).real[:2*n-1]

    def proj_simplex(v, s=1.0):
        nv = len(v)
        u = np.sort(v)[::-1]
        cv = np.cumsum(u) - s
        cond = u - cv / np.arange(1, nv+1) > 0
        if not np.any(cond):
            return np.full(nv, s/nv)
        i = int(np.max(np.where(cond)[0]) + 1)
        return np.maximum(v - cv[i-1]/i, 0.0)

    def eval_score(f):
        c = conv(f, f)
        return 2.0 * n * c.max()

    rng = np.random.default_rng(seed)

    # Try several initializations
    x = np.linspace(0.0, 1.0, n)
    inits = [f_init.copy()]
    inits.append(np.sin(np.pi * x**0.7))
    inits.append(np.exp(-2.5*(x-0.4)**2))
    inits.append(1.0/(1.0+20.0*x))
    inits.append(np.concatenate([np.ones(n//2)/n, np.ones(n-n//2)/(4*n)]))
    inits.append(np.concatenate([np.ones(n//2)/(4*n), np.ones(n-n//2)/n]))

    best, best_sc = None, 1e18
    for ini in inits:
        ini = np.maximum(ini, 0.0)
        s = ini.sum()
        if s < 1e-10:
            continue
        ini = ini / s
        sc = eval_score(ini)
        if sc < best_sc:
            best_sc, best = sc, ini.copy()

    # Main optimization: smooth gradient with backtracking
    f = best.copy()
    tau = 30.0
    alpha = 0.5
    v_mom = np.zeros(n)
    mom = 0.7

    it = 0
    stall_count = 0

    while time.monotonic() < deadline:
        # Compute convolution
        c = conv(f, f)
        cmax = c.max()
        sc = 2.0 * n * cmax
        if sc < best_sc:
            best_sc = sc
            best = f.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Smooth gradient via softmax
        shifted = c - cmax
        e = np.exp(tau * shifted)
        p = e / e.sum()

        # Gradient: 2 * cross-correlate(p, f) = 2 * (f_rev * p)[n-1:]
        f_rev = f[::-1]
        g = 2.0 * ifft(fft(f_rev, ffsize) * fft(p, ffsize)).real[n-1:2*n-1]

        # Momentum update
        v_mom = mom * v_mom + (1.0 - mom) * g

        # Backtracking line search
        improved = False
        for _ in range(20):
            f_new = proj_simplex(f - alpha * v_mom)
            sc_new = eval_score(f_new)
            if sc_new < sc:
                f = f_new
                improved = True
                break
            alpha *= 0.6

        if improved:
            alpha = min(alpha * 1.05, 20.0)
        else:
            alpha = 0.1
            v_mom = np.zeros(n)

        # Adapt tau
        if it % 500 == 250:
            tau = min(tau * 1.15, 500.0)

        # Restart if stalled
        if stall_count > 300:
            noise_scale = 0.05 * best_sc / n
            f = proj_simplex(best + rng.normal(0, noise_scale, n))
            v_mom = np.zeros(n)
            alpha = 0.3
            tau = max(tau * 0.5, 10.0)
            stall_count = 0

        # Periodic large jump exploration
        if it % 2000 == 1500:
            f = proj_simplex(best + rng.normal(0, 0.2/n, n))
            v_mom = np.zeros(n)
            alpha = 0.3

        it += 1

    # Final validation with exact evaluator
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        final_sc = float(evaluate_sequence(f_init.tolist()))
        best = f_init

    return [float(v) for v in best]
