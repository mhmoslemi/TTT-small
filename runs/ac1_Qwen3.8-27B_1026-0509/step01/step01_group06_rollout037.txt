```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def _score(a):
    """Compute the AC1 score for a nonnegative array with sum=1."""
    b = np.convolve(a, a)
    return 2.0 * len(a) * b.max()

def _lp_direction(a, M):
    """Find LP direction that maximizes sum(g) s.t. a*g <= M, g>=0."""
    n = len(a)
    L = 2 * n - 1
    # Build constraint matrix: (a*g)[k] = sum_i a[i]*g[k-i] <= M
    A = np.zeros((L, n))
    for k in range(L):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:

    return None
```

```python
                cn = np.convolve(x, x)
                mc = cn.max()
                p = np.exp(beta * (cn - mc))
                p /= p.sum()
                grad = np.zeros(n)
                for k in range(n2):
                    if p[k] < 1e-14:
                        continue
                    jlo = max(0, k - n + 1)
                    jhi = min(n - 1, k)
                    grad[jlo:jhi + 1] += 2.0 * p[k] * x[k - jlo:k - jhi - 1:-1]
                return grad

            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                     'jac': lambda x: np.ones(n)}]
            bounds = [(1e-15, None)] * n
            try:
                res = minimize(obj_fn, a, jac=grad_fn, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 40, 'ftol': 1e-14})
                if res.x is not None:
                    a_new = np.maximum(res.x, 1e-10)
                    a_new = a_new / a_new.sum()
                    s_new = score(a_new)
                    if s_new < score(a) + 1e-12:
                        a = a_new
                    if s_new < best_local_s:
                        best_local_s, best_local = s_new, a_new.copy()
            except Exception:
                pass
        return best_local, best_local_s

    # Phase 1: LP + smooth on parent
    if parent is not None:
        a1 = lp_refine(parent / parent.sum(), max_iters=12)
        s1 = score(a1)
        if s1 < best_s:
            best_s, best_a = s1, a1.copy()

        a2, s2 = smooth_opt(a1, deadline - 15)
        if s2 < best_s:
            best_s, best_a = s2, a2.copy()

        time_left = deadline - time.monotonic()
        if time_left > 20:
            a3 = lp_refine(a2, max_iters=10)
            s3 = score(a3)
            if s3 < best_s:
                best_s, best_a = s3, a3.copy()
            time_left = deadline - time.monotonic()
            if time_left > 15:
                a4, s4 = smooth_opt(a3, deadline - 10)
                if s4 < best_s:
                    best_s, best_a = s4, a4.copy()

    # Phase 2: Try other lengths
    for n in [35, 45, 50, 55]:
        if time.monotonic() > deadline - 30:
            break
        if parent is not None:
            init = parent[:n].copy() if len(parent) >= n else np.resize(parent, n)
        else:
            init = np.ones(n)
        a_t = lp_refine(np.maximum(init, 1e-8) / np.maximum(init, 1e-8).sum(), max_iters=8)
        s_t = score(a_t)
        if s_t < best_s:
            best_s, best_a = s_t, a_t.copy()
        time_left = deadline - time.monotonic()
        if time_left > 15:
            a_sm, s_sm = smooth_opt(a_t, deadline - 10)
            if s_sm < best_s:
                best_s, best_a = s_sm, a_sm.copy()

    # Phase 3: Perturbation restarts on best
    for _ in range(4):
        if time.monotonic() > deadline - 20:
            break
        a_p = best_a * np.exp(rng.normal(0, 0.25, len(best_a)))
        a_p = np.maximum(a_p, 1e-10)
        a_p = a_p / a_p.sum()
        a_r = lp_refine(a_p, max_iters=8)
        s_r = score(a_r)
        if s_r < best_s:
            best_s, best_a = s_r, a_r.copy()
        time_left = deadline - time.monotonic()
        if time_left > 15:
            a_sm, s_sm = smooth_opt(a_r, deadline - 5)
            if s_sm < best_s:
                best_s, best_a = s_sm, a_sm.copy()

    # Final validation
    out = np.clip(best_a, 0.0, 1000.0)
    if not (out.size > 0 and np.isfinite(out).all() and out.sum() > 0.01):
        out = np.ones(40, dtype=np.float64)
    fs = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(fs):
        out = np.ones(40, dtype=np.float64)
    return [float(x) for x in out]
```
