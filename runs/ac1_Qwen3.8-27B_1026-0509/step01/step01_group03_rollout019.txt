    js = np.maximum(0, k_vals - n + 1)
    je = np.minimum(n - 1, k_vals)

    # cols: for each k, the range is [js[k], je[k]]
    # Total number of elements = sum(cnt) = sum over k of (je[k] - js[k] + 1)
    # This equals sum over k of min(k+1, n, 2n-1-k)
    # For k = 0 to n-1: k+1 elements
    # For k = n to 2n-2: 2n-1-k elements
    # Total = n(n+1)/2 + (n-1)n/2 = n^2
    col_idx = np.concatenate([np.arange(s, e+1) for s, e in zip(js, je)])
    return rows, col_idx

def make_A(a, rows, col_idx):
    return csr_matrix((a[col_idx], (rows, col_idx)), shape=(rows.max()+1, len(a)))
