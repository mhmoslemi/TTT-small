import time
import cvxpy as cp

def _make_A(a, n):
    t = np.arange(2*n-1)[:, None]
    k = np.arange(n)[None, :]
    d = t - k
    valid = (d >= 0) & (d < n)
    return np.where(valid, a[np.clip(d, 0, n-1)], 0.0)

    return float(evaluate_sequence(a.tolist()))

def _lp_step(a, n, solver_list):
