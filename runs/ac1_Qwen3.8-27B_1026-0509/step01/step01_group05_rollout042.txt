def solve_dual_lp(f, M, n):
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0]
    for k in range(2*n - 1):
        # (f*g)_k = sum_j f[j] * g[k-j] for valid j
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # constraint: sum_{j=j_min}^{j_max} f[j] * g[k-j] <= M

        constraints.append(expr <= M)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
