import scipy.optimize as so
import scipy.sparse as sp

def lp_direction(a, M):
    # Find b >= 0 maximizing sum(b) subject to (a*b)_k <= M for all k
    # Primal: max c^T b s.t. A b <= M * 1, b >= 0
    # where A[k,j] = a[k-j] if 0 <= k-j < n
    
    n = len(a)
    # Build constraint matrix A (sparse)
    # Row k, col j: a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_lo = max(0, k-n+1)
        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Solve: max sum(b) s.t. A b <= M, b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    res = so.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as so

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 30.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
            raise ValueError("bad")
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    fft_len = 1 << (2 * n - 1).bit_length()
    fft_len2 = 1 << (3 * n - 1).bit_length()

    def fconv(x):
        X = np.fft.rfft(x, fft_len)
        return np.fft.irfft(X * X, fft_len)[:2 * n - 1]

    def sc(a):
        s = a.sum()
        if s < 1e-15:
            return 1e18
        return 2.0 * n * fconv(a).max() / (s * s)

    a = np.maximum(parent, 0.0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_sc = sc(a)

    def grad_step(a, lr, tol):
        c = fconv(a)
        M = c.max()
        if M < 1e-30:
            return a
        w = (c >= M * (1.0 - tol)).astype(np.float64)
        ws = w.sum()
        if ws < 1:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            ws = 1.0
        w /= ws
        A = np.fft.rfft(a, fft_len2)
        W = np.fft.rfft(w, fft_len2)
        g = np.fft.irfft(np.conj(A) * W, fft_len2)[:n]
        g -= g.mean()
        gm = np.abs(g).max()
        if gm < 1e-18:
            return a
        a2 = a * np.exp(-lr * g / gm)
        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s < 1e-15:
            return a
        return a2 / s

    def solve_lp(a, M, tl=20):
        diags = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
        A_mat = sp.diags(diags, list(range(n)), shape=(2 * n - 1, n), format='csc')
        try:
            res = so.linprog(-np.ones(n), A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                             bounds=[(0, None)] * n, method='highs',
                             options={'time_limit': tl, 'presolve': True})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0.0)
                sb = b.sum()
                if sb > 1e-15:
                    return b / sb
        except Exception:
            pass
        return None

    # Phase 1: Main subgradient descent with adaptive tolerance
    lr0 = 0.2
    for it in range(80000):
        if time.monotonic() > deadline - 250:
            break
        tol = max(0.001, 0.02 * np.exp(-it / 5000.0))
        lr = lr0 / np.sqrt(1.0 + it * 0.003)
        a = grad_step(a, lr, tol)
        if it % 300 == 0:
            s = sc(a)
            if s < best_sc:
                best_a = a.copy()
                best_sc = s

    if time.monotonic() < deadline - 100:
        a = best_a.copy()
        while time.monotonic() < deadline - 100:

            M = c.max()
            b = solve_lp(a, M, tl=min(20, max(5, deadline - 100 - time.monotonic())))
            if b is None:
                break
            found = False
            for t in [0.5, 0.35, 0.2, 0.1]:
                a_new = (1 - t) * a + t * b
                if fconv(a_new).max() < M * (1 - 1e-6):
                    a = a_new
                    found = True
                    break
            if not found:
                a = 0.4 * a + 0.6 * b
            s = sc(a)
            if s < best_sc:
                best_a = a.copy()
                best_sc = s
            # Subgradient burst after LP step
            nburst = min(1500, max(200, int((deadline - 100 - time.monotonic()) / 0.05)))
            for _ in range(nburst):
                if time.monotonic() > deadline - 100:
                    break
                a = grad_step(a, 0.06, 0.003)

    # Phase 3: Diversified restarts
    for r in range(20):
        if time.monotonic() > deadline - 25:
            break
        amp = 0.1 + 0.4 * r / 20.0
        mode = r % 3
        if mode == 0:
            a = best_a * np.exp(rng.normal(0, amp, n))
        elif mode == 1:
            a = best_a.copy()
            w = max(3, n // 50)
            pos = int(rng.integers(0, n - w))
            a[pos:pos + w] *= np.exp(rng.normal(0, 0.8, w))
            a = np.maximum(a, 0)
        else:
            a = best_a * (1 + amp * rng.standard_normal(n) * 0.5)
            a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a /= s
        lr0 = 0.12
        for it in range(12000):
            if time.monotonic() > deadline - 25:
                break
            tol = max(0.001, 0.01 * np.exp(-it / 2000.0))
            lr = lr0 / np.sqrt(1.0 + it * 0.005)
            a = grad_step(a, lr, tol)
            if it % 400 == 0:
                s2 = sc(a)
                if s2 < best_sc:
                    best_a = a.copy()
                    best_sc = s2

    # Phase 4: Fine-tuning with tight tolerance

        a = best_a.copy()

            if it % 800 == 0:
                s = sc(a)
                if s < best_sc:
                    best_a = a.copy()
                    best_sc = s

    # Final validation
    final = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final) or final > 1e10:
        return [float(x) for x in list(parent)]
    return [float(x) for x in list(best_a)]
