def _build_conv_matrix(f, n):

    k = np.arange(n)[None, :]
    d = t - k
    valid = (d >= 0) & (d < n)
    return np.where(valid, f[np.clip(d, 0, n-1)], 0.0)

import numpy as np
import time
import cvxpy as cp


def _conv_matrix(f, n):
    t = np.arange(2 * n - 1)[:, None]
    k = np.arange(n)[None, :]
    d = t - k
    valid = (d >= 0) & (d < n)
    return np.where(valid, f[np.clip(d, 0, n - 1)], 0.0)


def _lp_refine(f, n, deadline, max_iters=200):
    best = f.copy()
    s_t = np.sqrt(2.0 * n)
    best = best * (s_t / max(best.sum(), 1e-12))
    best_score = evaluate_sequence(best.tolist())
    if not np.isfinite(best_score):
        return best, best_score
    tau = 0.5
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        conv = np.convolve(best, best)
        M = conv.max()
        if M < 1e-15:
            break
        A = _conv_matrix(best, n)
        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
        g_val = None
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    gv = np.asarray(g.value).flatten()
                    if gv is not None and np.isfinite(gv).all() and gv.sum() > 1e-10:
                        g_val = gv
                        break
            except Exception:
                continue
        if g_val is None:
            break
        g_n = g_val * (s_t / g_val.sum())
        new_f = np.clip((1 - tau) * best + tau * g_n, 0, None)
        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = new_f
            best_score = sc
            tau = min(1.0, tau * 1.05)
        else:
            tau *= 0.8
            if tau < 0.01:
                break
        tau = max(tau, 0.05)
    return best, best_score


def _polish(f, routine, deadline):
    n = len(f)
    best = f.copy()
    best_sc = evaluate_sequence(best.tolist())
    for _ in range(3000):
        if time.monotonic() > deadline:
            break
        w = max(3, n // 40)
        s = int(routine.integers(0, n - w + 1))
        pert = best.copy()
        pert[s:s + w] *= np.exp(routine.normal(0, 0.08, size=w))
        pert = np.clip(pert, 0, None)
        if pert.sum() < 0.01:
            continue
        sc = evaluate_sequence(pert.tolist())
        if np.isfinite(sc) and sc < best_sc - 1e-10:
            best = pert
            best_sc = sc
    return best, best_sc
