def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    fft_c = 1 << (2 * n - 1).bit_length()
    fft_cc = 1 << (3 * n - 1).bit_length()

    def fconv(a):
        A = np.fft.rfft(a, fft_c)
        return np.fft.irfft(A * A, fft_c)[:2 * n - 1]

    a = parent.copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = 2.0 * n * fconv(a).max()

    def subgrad_direct(a, active):
        grad = np.zeros(n)
        for i in active:
            j_lo = max(0, i - n + 1)
            j_hi = min(n - 1, i)
            if j_hi >= j_lo:
                grad[j_lo:j_hi + 1] += a[i - j_hi:i - j_lo + 1][::-1]
        return 2.0 * grad / len(active)

    def subgrad_fft(a, active):
        A_ind = np.zeros(fft_cc)
        A_ind[active] = 1.0
        a_pad = np.zeros(fft_cc)
        a_pad[:n] = a
        FA = np.fft.rfft(A_ind, fft_cc)
        Fa = np.fft.rfft(a_pad, fft_cc)
        c = np.fft.irfft(FA * np.conj(Fa), fft_cc)[:n]
        return 2.0 * c / len(active)

    def mirror_step(a, eta, conv):
        M = conv.max()
        if M < 1e-30:
            return a.copy(), M
        active = np.flatnonzero(conv >= M * (1.0 - 1e-10))
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) < 200:
            grad = subgrad_direct(a, active)
        else:
            grad = subgrad_fft(a, active)
        gc = grad - grad.mean()
        a_new = a * np.exp(-eta * gc)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        return a_new, M

    def smooth_grad(a, conv, p):
        W = conv ** (p - 1)
        W_pad = np.zeros(fft_cc)
        W_pad[:2 * n - 1] = W
        a_pad = np.zeros(fft_cc)
        a_pad[:n] = a
        FW = np.fft.rfft(W_pad, fft_cc)
        Fa = np.fft.rfft(a_pad, fft_cc)
        c = np.fft.irfft(FW * np.conj(Fa), fft_cc)[:n]
        return 2.0 * p * c

                    best_score = sc
