                beta = min(beta * 1.001, 300.0)
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    lr = max(lr * 0.5, 0.001)
                    stall = 0
                if stall > 400:
                    a = ba * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 1e-30)
                    a /= a.sum()
                    lr = 0.1
                    stall = 0
        return ba, bs

    # Phase 1: main softmax mirror descent
    t_p1 = time.monotonic() + 0.55 * (deadline - t0)
    best_a, best_sc = fast_opt(a0, t_p1)

    # Phase 2: perturbed restarts
    rem = deadline - time.monotonic()
    for i in range(4):
        if rem < 30:
            break
        tr = rem / (5 - i) - 5
        if tr < 10:
            break
        p = best_a * np.exp(rng.normal(0, 0.06 / (i + 1), n))
        p = np.maximum(p, 1e-30)
        p /= p.sum()
        a2, s2 = fast_opt(p, time.monotonic() + tr)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        rem = deadline - time.monotonic()

    # Phase 3: exact-subgradient polishing
    if time.monotonic() < deadline - 15:
        lr = 0.004
        stall = 0
        while time.monotonic() < deadline - 15:
            c = conv(best_a)
            mc = c.max()
            sc = 2.0 * n * mc
            if sc < best_sc:
                best_sc = sc
            idx = np.argsort(c)[-50:]
            g = np.zeros(n)
            for k in idx:
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                pos = k - np.arange(lo, hi + 1)
                g[lo:hi + 1] += 2.0 * best_a[pos]
            st = np.clip(lr * g, -15, 15)
            na = best_a * np.exp(-st)
            na = np.maximum(na, 1e-30)
            na /= na.sum()
            ns = 2.0 * n * conv(na).max()
            if ns <= best_sc:
                best_a = na
                best_sc = ns
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    lr = max(lr * 0.5, 1e-6)
                    stall = 0
                if lr < 2e-6:
                    break

    result = [float(x) for x in best_a]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs) or fs > 5:
        result = [float(x) for x in a0]
        fs = evaluate_sequence(result)
    return result
