    n = len(v)

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

def _simplex_project(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)

import numpy as np
import time

def _simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return proj / s

def _grad_conv_max(a, k):
    """Exact gradient of c_k = conv(A,a)[k] w.r.t. a, where A=a."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    g = np.zeros(n)
    g[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    return g

        for k in range(L):

            g += p[k] * _grad_conv_max(a, k)
        g *= 2.0 * n  # scale for the full objective

        improved = False

                cmax = c_new.max()
                improved = True

                tau = max(0.01, tau * 2)
                lr = 1.0 / np.sqrt(n)
                stall = 0

def _lbfgs_polish(a0, budget, rng):
    from scipy.optimize import minimize
    
    n = len(a0)
    tau = 0.01
    
    def fun(x):
        # Work on the log scale to maintain positivity
        # x are log-values, a = exp(x) / sum(exp(x))
        a = np.exp(x - x.max())
        a /= a.sum()
        c = np.convolve(a, a)
        c_shift = c - c.max()
        return 2.0 * n * tau * np.log(np.sum(np.exp(c_shift / tau)))
    
    def fun_and_grad(x):
        a = np.exp(x - x.max())
        a /= a.sum()
        c = np.convolve(a, a)
        c_shift = c - c.max()
        exp_c = np.exp(c_shift / tau)
        p = exp_c / exp_c.sum()
        obj = 2.0 * n * tau * np.log(exp_c.sum())
        
        # Gradient w.r.t. x: df/dx[j] = df/da[j] * da/dx[j]
        # a = softmax(x), so da/dx[j] = a[j] * (delta_jk - sum_m a[m]... )
        # Actually da[j]/dx[k] = a[j] if j==k, -a[j]*a[k] otherwise (Jacobian of softmax)
        # So df/dx[k] = sum_j (df/da[j]) * (a[j]*delta_jk - a[j]*a[k])
        #              = a[k] * df/da[k] - a[k] * sum_j a[j] * df/da[j]
        
        g_a = np.zeros(n)
        L = 2*n - 1
        for k in range(L):
            if p[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g_a[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
        g_a *= 2.0 * n  # scale
        
        grad_x = a * (g_a - np.dot(a, g_a))
        return obj, grad_x
    
    x0 = np.log(np.maximum(a0, 1e-10))
    
    try:
        res = minimize(fun_and_grad, x0, method='L-BFGS-B',
                      options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
        a_opt = np.exp(res.x - res.x.max())
        a_opt /= a_opt.sum()
        return a_opt
    except Exception:
        return a0
