For inactive constraints (a_i > 0, so ν_i = 0): sum_{k active} λ_k * a_{k-i} = μ

This means that the "autocorrelation" of the active set (weighted by λ) is constant for all i where a_i > 0. This is a strong structural condition.

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)
    
    best = None
    best_score = np.inf
    
    def try_score(a):
        nonlocal best, best_score
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            return np.inf
        s = evaluate_sequence(a.tolist())
        if np.isfinite(s) and s < best_score:
            best = a.copy()
            best_score = s
        return s
    
    def grad_smooth_max(a, eps):
        """Returns (smoothed objective, gradient) for f(a) = 2n*soft_max(b)/S^2"""
        n = len(a)
        b = np.convolve(a, a)
        S = a.sum()
        if S < 1e-10:
            return np.inf, None
        # Soft max of b
        bm = b.max()
        eb = np.exp(eps * (b - bm))
        p = eb / eb.sum()
        sm = bm + np.log(eb.sum()) / eps
        
        # grad_sm[i] = sum_j a[j] * p[i+j]
        # Using convolution trick
        pr = p[::-1]  # reverse p
        fc = np.convolve(a, pr)  # full, length n + (2n-1) - 1 = 3n-2
        # grad_sm[i] = fc[2n-2-i]
        grad_sm = fc[n-1:2*n-1][::-1]
        
        obj = 2.0 * n * sm / S**2
        grad = 2.0 * n * (grad_sm / S**2 - 2.0 * sm / S**3)
        return obj, grad
    
    def adam_optimize(a0, n_steps=20000, eps0=5.0, eps_min=0.5):
        a = a0.copy()
        S = a.sum()
        if S > 1e-10:
            a = a / S
        n = len(a)
        m = np.zeros(n)
        v = np.zeros(n)
        b1, b2, lr = 0.85, 0.999, 0.02
        t = 0
        eps = eps0
        last_true = np.inf
        for it in range(n_steps):
            if time.monotonic() > deadline - 10:
                break
            eps = max(eps_min, eps0 * (eps_min/eps0)**(it/n_steps))
            obj, grad = grad_smooth_max(a, eps)
            if grad is None:
                break
            t += 1
            m = b1*m + (1-b1)*grad
            v = b2*v + (1-b2)*grad**2
            mh = m / (1 - b1**t)
            vh = v / (1 - b2**t)
            a = a - lr * mh / (np.sqrt(vh) + 1e-8)
            a = np.maximum(a, 0)
            if it % 200 == 0:
                s = try_score(a)
                if s < best_score and a.sum() > 0.01:
                    last_true = s
        return a
    
    def lp_refine(a0, n_iters=500):
        a = a0.copy()
        S = a.sum()
        if S > 1e-10:
            a = a / S
        n = len(a)
        
        # Build sparse structure for convolution with a as first argument
        # (a * b)_k = sum_i a[i] * b[k-i] for valid i
        # For fixed a, this is linear in b
        # A_ub row k: sum_i a[i] * b[k-i] <= M
        # The coefficient of b_j in row k is a[k-j] (if valid)
        
        # Build the matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0
        # This is a Toeplitz matrix
        from scipy.linalg import toeplitz
        # First column: A[0,j] = a[0-j] = a[-j], which is 0 for j>0, a[0] for j=0
        # First row: A[k,0] = a[k] for k=0..n-1, then 0
        col0 = np.zeros(2*n-1)
        col0[0] = a[0]
        row0 = np.zeros(2*n-1)
        row0[:n] = a[:n]
        # Actually, A[k,j] = a[k-j] if 0<=k-j<=n-1
        # For j=0: A[k,0] = a[k] for k=0..n-1, 0 for k>=n
        # For k=0: A[0,j] = a[-j] = 0 for j>0, a[0] for j=0
        # So first column is [a[0], 0, 0, ..., 0]
        # First row is [a[0], a[1], ..., a[n-1], 0, ..., 0]
        # Wait, that's not right for a Toeplitz matrix.
        
        # Let me just build it directly as sparse
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            for j in range(n):
                d = k - j
                if 0 <= d < n:
                    rows.append(k)
                    cols.append(j)
                    data.append(a[d])
        A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        
        c_lp = -np.ones(n)
        bounds = [(0, None)] * n
        t_blend = 0.3
        stale = 0
        
        for it in range(n_iters):
            if time.monotonic() > deadline - 10:
                break
            b_curr = np.convolve(a, a)
            M = b_curr.max()
            if M < 1e-15:
                break
            res = linprog(c_lp, A_ub=A_ub, b_ub=np.full(2*n-1, M),
                         bounds=bounds, method='highs',
                         options={'time_limit': min(5, deadline - time.monotonic() - 5)})
            if res.success and res.x is not None:
                b_opt = np.maximum(res.x, 0)
                bs = b_opt.sum()
                if bs > 1e-10:
                    b_opt /= bs
                    a_new = np.maximum((1-t_blend)*a + t_blend*b_opt, 0)
                    sn = a_new.sum()
                    if sn > 1e-10:
                        a_new /= sn
                        s = try_score(a_new)
                        if s <= best_score:
                            a = a_new
                            stale = 0
                        else:
                            stale += 1
                            if stale > 10:
                                t_blend = max(0.05, t_blend * 0.7)
                                stale = 0
                    # Rebuild A_ub with new a
                    data_new = []
                    rows_new = []
                    cols_new = []
                    for k in range(2*n-1):
                        for j in range(n):
                            d = k - j
                            if 0 <= d < n:
                                rows_new.append(k)
                                cols_new.append(j)
                                data_new.append(a_new[d] if sn > 1e-10 else 0)
                    # Only rebuild if we accepted
                    if a is a_new:  # this check doesn't work due to reference
                        pass
                    A_ub = csr_matrix((data_new, (rows_new, cols_new)), shape=(2*n-1, n))
            else:
                # Fallback: gradient step
                break
        
        return a
    
    # Initializations
    inits = []
    try:
        h = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if h.size > 0 and np.isfinite(h).all():
            inits.append(h.copy())
    except:
        pass
    
    n_base = 200
    inits.append(rng.random(n_base) + 0.01)
    
    # Try a structured initialization: low autocorrelation
    # Use a chirp or similar
    x = np.arange(n_base)
    inits.append(0.5 + 0.4*np.cos(2*np.pi*x**2/(2*n_base)) + 0.1)
    
    # Binary-like
    inits.append((rng.random(n_base) > 0.5).astype(float) + 0.05)
    
    for a0 in inits:
        if time.monotonic() > deadline - 60:
            break
        a0 = np.maximum(a0.astype(np.float64), 0)
        if a0.size < 2 or a0.sum() < 0.01:
            continue
        try_score(a0)
        
        # Phase 1: Gradient optimization
        a_grad = adam_optimize(a0, n_steps=15000, eps0=5.0, eps_min=0.3)
        try_score(a_grad)
        
        # Phase 2: LP refinement
        if time.monotonic() < deadline - 30:
            a_lp = lp_refine(a_grad, n_iters=200)
            try_score(a_lp)
    
    if best is None:
        best = np.ones(100)
    
    result = [float(x) for x in np.clip(best, 0, 1000)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 100
    return result
```

```python
            M = np.convolve(a, a).max()
            if M < 1e-15:
                break
            tl = max(0.5, min(4.0, deadline - time.monotonic() - 5))
            res = linprog(c_lp, A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=bnds, method='highs',
                          options={'time_limit': tl})
            if not (res.success and res.x is not None):
                stale += 1
                if stale > 20:
                    break
                continue
            b_opt = np.maximum(res.x, 0)
            bs = b_opt.sum()
            if bs < 1e-10:
                stale += 1
                continue
            b_opt /= bs
            a_new = np.maximum((1 - t_blend) * a + t_blend * b_opt, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                stale += 1
                continue
            a_new /= sn
            s = try_score(a_new)
            if s <= best_score + 1e-12:
                a = a_new
                A = build_A(a)
                stale = 0
                if it % 30 == 0:
                    t_blend = max(0.08, t_blend * 0.92)
            else:
                stale += 1
                if stale > 12:
                    t_blend = max(0.05, t_blend * 0.5)
                    stale = 0
        try_score(a)
        return a

    inits = []
    try:
        h = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if h.size > 0 and np.isfinite(h).all() and h.sum() > 0.01:
            inits.append(h.copy())
    except:
        pass

    n_base = 200
    inits.append(rng.random(n_base) + 0.01)
    half = rng.random(n_base // 2 + 1) + 0.01
    sym = np.concatenate([half, half[:-1][::-1]])[:n_base]
    inits.append(sym)

    for a0 in inits:
        if time.monotonic() > deadline - 100:
            break
        a0 = np.maximum(a0, 0).astype(np.float64)
        if a0.size < 2 or a0.sum() < 0.1:
            continue
        try_score(a0)
        a_g = opt_grad(a0, n_steps=25000)
        if time.monotonic() < deadline - 50:
            a_l = opt_lp(a_g, max_iters=350)
        if time.monotonic() < deadline - 50:
            a_g2 = opt_grad(a_l, n_steps=10000)
            if time.monotonic() < deadline - 30:
                opt_lp(a_g2, max_iters=150)

    if best is None:
        best = np.ones(100)
    result = [float(x) for x in np.clip(best, 0, 1000)]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0] * 100
    return result
```
