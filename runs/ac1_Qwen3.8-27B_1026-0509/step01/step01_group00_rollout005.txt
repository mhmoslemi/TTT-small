    if time.monotonic() < deadline - 25:
        for _ in range(3):
            a_p = np.maximum(best * np.exp(rng.normal(0, 0.03, n)), 1e-30)
            a_p /= a_p.sum()
            run_opt(a_p, [(3000, 0.01, 5), (3000, 0.005, 3), (3000, 0.002, 2)],
                    noise_every=2000, noise_std=0.005)
            if time.monotonic() > deadline - 10:
                break

    best = np.clip(best, 0.0, 1000.0)
    result = [float(x) for x in best]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [float(x) for x in a0]
        fs = evaluate_sequence(result)
    return result
