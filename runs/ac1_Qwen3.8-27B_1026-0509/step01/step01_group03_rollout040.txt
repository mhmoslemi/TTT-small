            grad -= grad.mean()
            cur_sc = 2.0 * n * bm
            improved = False
            for _ in range(15):
                a_try = np.maximum(a - lr * grad, 0)
                ns = a_try.sum()
                if ns < 1e-10:
                    lr *= 0.5
                    continue
                a_try /= ns
                sc_try = peak_score(a_try)
                if sc_try < cur_sc * (1 - 1e-7):
                    a = a_try
                    if sc_try < best_sc:
                        best_a = a.copy()
                        best_sc = sc_try
                    improved = True
                    break
                lr *= 0.5
            it += 1
            if improved:
                stale = 0
                beta = min(3000.0, beta * 1.08)
                lr = min(5.0, lr * 1.04)
            else:
                stale += 1
                lr = max(1e-4, lr * 0.5)
                if stale % 12 == 0:
                    a = np.maximum(a + 0.02 * rng.standard_normal(n), 0)
                    a = normalize(a)
        return best_a

    def lp_refine(a, time_limit, max_iters=200):
        a = normalize(np.maximum(a.astype(np.float64), 0))
        n = len(a)
        c_lp = -np.ones(n)
        bnd = [(0, None)] * n
        t_bl = 0.25
        best_a = a.copy()
        best_sc = peak_score(a)
        for _ in range(max_iters):
            if time.monotonic() > time_limit:
                break
            b_conv = np.convolve(a, a, mode='full')
            M = b_conv.max()
            if M < 1e-15:
                break
            k_vals = np.arange(2 * n - 1)
            js = np.maximum(0, k_vals - n + 1)
            je = np.minimum(n - 1, k_vals)
            cnt = je - js + 1
            F_rows = np.repeat(k_vals, cnt)
            F_cols = np.concatenate([np.arange(js[k], je[k]+1) for k in range(2*n-1)])
            F_data = a[F_rows - F_cols]
            A_ub = csr_matrix((F_data, (F_rows, F_cols)), shape=(2*n-1, n))
            tl = max(0.3, min(30.0, time_limit - time.monotonic() - 5))
            if tl < 0.3:
                break
            try:
                res = linprog(c_lp, A_ub=A_ub, b_ub=np.full(2*n-1, M),
                              bounds=bnd, method='highs', options={'time_limit': tl})
                if not (res.success and res.x is not None):
                    break
                b_opt = np.maximum(res.x, 0)
                bs = b_opt.sum()
                if bs < 1e-10:
                    break
                b_opt /= bs
                a_new = np.maximum((1-t_bl)*a + t_bl*b_opt, 0)
                if a_new.sum() < 1e-10:
                    break
                a_new /= a_new.sum()
                sc = peak_score(a_new)
                if sc < best_sc:
                    best_a = a_new.copy()
                    best_sc = sc
                a = a_new
                t_bl = max(0.03, t_bl * 0.93)
            except:
                break
        return best_a

    # Initializations
    inits = []
    try:
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if p.size > 0 and np.isfinite(p).all():
            inits.append(p.copy())
    except:
        pass
    n0 = 200
    x = np.arange(n0, dtype=np.float64)
    inits.append(rng.random(n0) + 0.02)
    v = np.full(n0, 0.01); v[:n0//3] = 1.0; v[2*n0//3:] = 1.0
    inits.append(v)
    inits.append(0.5 + 0.5*np.cos(2*np.pi*x/n0) + 0.01)
    inits.append(rng.random(300) + 0.02)
    inits.append(rng.random(400) + 0.02)

    # Time allocation: give more to first (parent) init
    total_time = max(60.0, deadline - t0)
    weights = [0.35, 0.20, 0.12, 0.12, 0.10, 0.11]
    while len(weights) < len(inits):
        weights.append(0.05)

    for i, (a0, w) in enumerate(zip(inits, weights)):
        if time.monotonic() > deadline - 25:
            break
        alloc = total_time * w
        a = np.maximum(a0.astype(np.float64), 0)
        n = len(a)
        if n < 2:
            continue
        a = normalize(a)
        try_score(a)

        # Phase 1: Gradient descent
        t1 = min(time.monotonic() + alloc * 0.45, deadline - 30)
        if time.monotonic() < t1:
            a = grad_desc(a, t1)
            try_score(a)

        # Phase 2: LP refinement
        if n <= 500 and time.monotonic() < deadline - 30:
            t2 = min(time.monotonic() + alloc * 0.40, deadline - 25)
            if time.monotonic() < t2:
                a = lp_refine(a, t2)
                try_score(a)

        # Phase 3: Gradient polish with higher beta
        t3 = min(time.monotonic() + alloc * 0.15, deadline - 20)
        if time.monotonic() < t3:
            a = grad_desc(a, t3, beta_init=200.0)
            try_score(a)

    if best is None:
        best = np.ones(100)
    result = [float(v) for v in np.clip(best, 0, 1000)]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0] * 100
    return result
