import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)
    
    best = None
    best_score = np.inf
    
    def try_score(a):
        nonlocal best, best_score
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best = a.copy()
            best_score = s
        return s
    
    def simplex_project(v, target=1.0):
        mu = np.sort(v)[::-1]
        cumsum = np.cumsum(mu)
        rho = np.nonzero(mu * np.arange(1, len(mu)+1) > (cumsum - target))[0]
        if len(rho) == 0:
            return np.full(len(v), target/len(v))
        rho = rho[-1]
        tau = (cumsum[rho] - target) / (rho + 1.0)
        return np.maximum(v - tau, 0)
    
    def full_conv(a, b=None):
        if b is None:
            b = a
        if len(a) * len(b) < 2e6:
            return np.convolve(a, b, mode='full')
        else:
            return fftconvolve(a, b)
    
    def lp_step(f, F_rows, F_cols, c_lp, bnd, t_mix):
        n = len(f)
        u = full_conv(f)
        M = u.max()
        if M < 1e-15:
            return f, True
        F_data = f[F_rows - F_cols]
        A_ub = csr_matrix((F_data, (F_rows, F_cols)), shape=(2*n-1, n))
        remaining = deadline - time.monotonic() - 30
        if remaining < 5:
            return f, True
        tl = min(30.0, remaining)
        res = linprog(c_lp, A_ub=A_ub, b_ub=np.full(2*n-1, M),
                     bounds=bnd, method='highs', options={'time_limit': tl})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                f_new = simplex_project((1-t_mix)*f + t_mix*g, 1.0)
                return f_new, False
        return f, True
    
    def gradient_polish(f, max_iters=300):
        n = len(f)
        beta = 50.0
        lr = 0.1
        prev_obj = np.inf
        for it in range(max_iters):

                break

            conv = np.fft.irfft(np.fft.rfft(f[::-1], N_fft) * np.fft.rfft(w, N_fft), N_fft)
            grad = 2.0 * conv[n-1:2*n-1]
            # Backtracking line search
            improved = False
            for _ in range(30):
                f_new = simplex_project(f - lr * grad, 1.0)
                u_new = full_conv(f_new)
                obj_new = u_new.max()
                if obj_new < prev_obj - 1e-12:
                    f = f_new
                    prev_obj = obj_new
                    improved = True
                    break
                lr *= 0.5
            if not improved:

                if lr < 1e-8:
                    beta = min(500, beta * 1.5)
                    lr = 0.1
                else:

            # Adaptive beta

            if spread > 0.3:
                beta = min(300.0, beta * 1.2)
            else:
                beta = max(10.0, beta * 0.9)
            sc = try_score(f)
        return f
    
    def run_lp_iterations(f, n_iters=200):
        n = len(f)
        # Build sparse structure for LP

            off += cnt[k]

        stale = 0

        for it in range(n_iters):
            if time.monotonic() > deadline - 30:

            if done:

            if sc < prev_sc - 1e-6:
                stale = 0
            else:
                stale += 1
            prev_sc = sc
            if stale % 10 == 0:
                t_mix = max(0.05, t_mix * 0.7)
        return f

    inits = []
    try:

        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            inits.append(p / p.sum())
    except:
        pass

    for n in [200, 400, 800]:
        if n == 200 and len(inits) > 0:
            continue
        # Skewed distribution
        x = np.linspace(0, 1, n)
        f_init = (1.0 + 0.8 * x) ** 2
        f_init /= f_init.sum()
        inits.append(f_init)
        # Another skew
        f_init2 = (1.0 + 1.0 * np.sin(np.pi * x)) ** 2
        f_init2 /= f_init2.sum()
        inits.append(f_init2)
    
    # Main optimization loop
    for idx, f0 in enumerate(inits):
        if time.monotonic() > deadline - 60:
            break
        f = f0.astype(np.float64).copy()
        sc = try_score(f)
        # LP phase
        f = run_lp_iterations(f, n_iters=150)
        sc = try_score(f)
        # Gradient polish
        f = gradient_polish(f, max_iters=200)
        sc = try_score(f)
    
    # If time remains, try to extend the best sequence
    if best is not None and time.monotonic() < deadline - 100:
        # Try zero-padding to a larger length
        n_old = len(best)
        for n_new in [n_old * 2, n_old * 3]:
            if time.monotonic() > deadline - 60:
                break
            if n_new > 2000:
                break
            f_ext = np.zeros(n_new)
            f_ext[:n_old] = best
            f_ext = f_ext / f_ext.sum()
            sc = try_score(f_ext)
            if sc < best_score:
                f = f_ext
                f = run_lp_iterations(f, n_iters=50)
                f = gradient_polish(f, max_iters=100)
                try_score(f)
    
    if best is None:
        result = [1.0] * 10
    else:
        result = [float(x) for x in np.clip(best, 0, 1000)]
    
    fs = evaluate_sequence(result)
    if not np.isfinite(fs) or fs >= best_score - 0.01:
        best_clip = np.clip(best, 0, 1000) if best is not None else np.ones(10)
        result = [float(x) for x in best_clip]
    
    return result
