def _simplex_project(v, target_sum=1.0):

    # Standard simplex projection
    n = len(v)

    if s > 0:

                if s < best_score - 1e-12:
                    best_score = s
                    best_overall = c.copy()
                    a = c
                    break

    final = np.clip(best_overall, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = final * (1.0 / final.sum() if final.sum() > 0 else 1.0)
    fs = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(fs) or fs > 2.5:
        final = np.ones(n, dtype=np.float64)
    return [float(v) for v in final]
