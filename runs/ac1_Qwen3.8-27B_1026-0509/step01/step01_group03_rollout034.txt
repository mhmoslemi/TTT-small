def lp_min_peak(a):
    """Solve LP to find b minimizing max_k (a*b)_k subject to sum(b)=1, b>=0.
    Returns (b, M_star)."""
    n = len(a)
    Fr, Fc = build_F(n)
    Fd = a[Fr - Fc]
    
    # Variables: b_0, ..., b_{n-1}, M'
    # Constraints: (a*b)_k - M' <= 0 for k = 0, ..., 2n-2
    #              sum b = 1
    #              b >= 0
    # Objective: minimize M'
    
    # c = [0, ..., 0, 1] (minimize M')
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub for (a*b)_k <= M'
    # Row k: [a_{k-j} for valid j] ... 1 (coefficient of M' is -1 for <= form)
    # Actually: (a*b)_k - M' <= 0
    # So the row is [a_{k-0}, ..., a_{k-(n-1)}, -1] with appropriate zeros
    
    # Let me build A_ub as a dense matrix for small n, sparse for large n
    # A_ub shape: (2n-1, n+1)
    
    A_ub = csr_matrix((
        np.concatenate([Fd, np.full(2*n-1, -1.0)]),
        np.concatenate([Fc, np.full(2*n-1, n)]),
        (np.concatenate([Fr, Fr]), np.concatenate([Fc, np.full(2*n-1, n)])),
    ), shape=(2*n-1, n+1))
    
    # Equality: sum b = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, M' free
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 30})
    if res.success:
        b = np.maximum(res.x[:n], 0)
        M_star = res.x[-1]

            return np.inf
        return 2.0 * n * c.max() / (S * S)

            js = max(0, k - n + 1)
            je = min(n - 1, k)

            Fc.extend(range(js, je + 1))
        return np.array(Fr), np.array(Fc)
