        return 2.0 * len(f) * M / (S * S)
    
    def lp_refine(f, deadline, max_iters=500):

        best = f.copy()
        best_sc = score(best)
        tau = 0.25
        no_improve = 0
        S = best.sum()

            M = y.max()

            A = np.where(valid, best[np.clip(d, 0, n-1)], 0.0)

                        break

                    continue
            
            if g_val is None or not np.isfinite(g_val).all() or g_val.sum() < 1e-10:
                # Try gradient step instead
                break
            
            g_val = np.maximum(g_val, 0)
            g_val = g_val * (S / g_val.sum())
            
            new_f = (1 - tau) * best + tau * g_val
            new_f = np.clip(new_f, 0, None)
            
            sc = score(new_f)
            if np.isfinite(sc) and sc < best_sc:
                best = new_f.copy()
                best_sc = sc
                no_improve = 0
            else:
                no_improve += 1
                tau *= 0.85
                if no_improve > 15:
                    break
            
            tau = min(tau * 1.02, 0.4)
            tau = max(tau, 0.03)
        
        return best, best_sc

        n = len(f)

        no_improve = 0

                hi1 = min(n - 1, t)
                if lo1 <= hi1:
                    g[lo1:hi1+1] += best[t - np.arange(lo1, hi1 + 1)]
                lo2 = max(0, t)
                hi2 = min(n - 1, t + n - 1)
                if lo2 <= hi2:
                    g[lo2:hi2+1] += best[np.arange(lo2, hi2 + 1) - t]
            g /= len(argmax)
            
            # Project gradient to sum-zero subspace
            g = g - g.mean()
            
            # Gradient descent step
            new_f = best - lr * g
            new_f = np.maximum(new_f, 0)
            if new_f.sum() < 1e-10:
                break
            new_f = new_f / new_f.sum()
            
            sc = score(new_f)
            if np.isfinite(sc) and sc < best_sc * (1 - 1e-6):
                best = new_f.copy()
                best_sc = sc
                no_improve = 0
                lr = min(lr * 1.01, 0.1)
            else:
                no_improve += 1
                lr *= 0.9
                if no_improve > 50:
                    break
        
        return best, best_sc
    
    # --- Main search ---
    
    # Try height_sequence_1 if available

        elif kind == 'tri':
            f = 1.0 - np.abs(2.0 * j / n - 1.0)
        elif kind == 'rand':
            f = rng.random(n) + 0.1
        else:
            f = np.ones(n)
        f = np.maximum(f, 0)
        if f.sum() > 0.01:
            inits.append(f)
    
    # Sort by initial score (best first)
    scored_inits = []
    for f in inits:
        sc = score(f)
        if np.isfinite(sc):
            scored_inits.append((sc, f))
    scored_inits.sort(key=lambda x: x[0])
    
    allocated = 30  # seconds for setup
    per_init_time = max(10, (budget_s - 60) / max(1, len(scored_inits)))
    
    for sc0, f0 in scored_inits:
        if time.monotonic() > deadline:
            break
        
        f0 = f0 / f0.sum()  # normalize
        update(f0, score(f0))
        
        # Phase 1: LP refinement
        rem = deadline - time.monotonic()
        if rem > 30:
            f_lp, sc_lp = lp_refine(f0, time.monotonic() + rem * 0.6)
            update(f_lp, sc_lp)
        
        # Phase 2: Gradient refinement
        if time.monotonic() < deadline - 30:
            f_grad, sc_grad = grad_refine(f_lp, time.monotonic() + (deadline - time.monotonic()) * 0.4)
            update(f_grad, sc_grad)

        if time.monotonic() < deadline - 20:
            f_lp2, sc_lp2 = lp_refine(f_grad, deadline - 20)

    if best_seq is not None and time.monotonic() < deadline - 10:

Actually, let me think about this differently. The problem is:
minimize M = max_t sum_k f[k]*f[t-k]
subject to sum f = 1, f >= 0

This is equivalent to: minimize M subject to all (f*f)[t] <= M, sum f = 1, f >= 0.

The constraint (f*f)[t] <= M is a quadratic constraint (not linear in f). So this is a quadratic program (QCP), not an LP!

The parent's LP is a relaxation: it fixes f and finds g such that f*g <= M (linear in g). Then it mixes f and g. This is a fixed-point iteration.

The direct quadratic formulation would be:
minimize M
s.t. f^T T_t f <= M for all t
     1^T f = 1
     f >= 0

where T_t is the shift matrix. This is a SOCP (second-order cone program) if we relax the quadratic constraints... actually, f^T T_t f <= M is a non-convex constraint (T_t is not PSD in general).

Wait, for t = n-1 (the middle of the full convolution): (f*f)[n-1] = sum_k f[k]*f[n-1-k]. If f is symmetric, this is sum f[k]^2 = ||f||^2. The constraint ||f||^2 <= M is convex (a Euclidean ball).

For other t, the constraint is a general quadratic form. Not necessarily convex.

So the direct approach is non-convex. The LP relaxation (parent's approach) is a valid heuristic.
