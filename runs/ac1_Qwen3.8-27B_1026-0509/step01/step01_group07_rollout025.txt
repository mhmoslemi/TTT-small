import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2

The computation is:
1. b = conv(a, a) via FFT: O(N log N)
2. w = softmax(β * b): O(n)
3. (w ⊛ a) via FFT: O(N log N)
4. grad = 4n * (S*(w⊛a) - M_soft) / S^3: O(n)

Total: O(N log N) per iteration, where N ≈ 2^13 = 8192 for n≈7000. Very fast!

import time
from scipy.fft import rfft, irfft

def _cross_corr(w, a):
    """Compute cross-correlation (w ⊛ a)[i] = sum_j w[j]*a[j-i] for i=0..n-1."""
    n = len(a)
    N = 1 << (2*n-1 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    result = irfft(np.conjugate(A) * W, N)
    return result[:n]
