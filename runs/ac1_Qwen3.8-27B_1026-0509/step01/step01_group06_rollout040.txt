def _grad_conv_smooth(a, w):
    """Gradient of sum_k w[k]*b[k] where b=conv(a,a), w given.
    Returns gradient w.r.t. a (unnormalized)."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        # a[j] appears in b[k] for k in [j, j+n-1] (clipped to [0, 2n-2])
        k_lo = max(0, j)

                grad[j] += w[k] * 2.0 * a[j]
            else:
                grad[j] += w[k] * (a[diff] + a[j])

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

Wait, the output length for 'valid' mode is max(len(a), len(v)) - min(len(a), len(v)) + 1.

Wait, I need to be more careful. From numpy source:
For `np.correlate(a, v, mode='valid')`:
- If len(a) >= len(v): output_len = len(a) - len(v) + 1
- c_k = sum_{n=0}^{len(v)-1} a[n+k] * v[n] for k=0,...,len(a)-len(v)

So `np.correlate(w, a, 'valid')` where len(w)=2n-1, len(a)=n:
output_len = (2n-1) - n + 1 = n
c_k = sum_{n=0}^{n-1} w[n+k] * a[n] for k=0,...,n-1

This is exactly what I want! grad[j] = 2 * c_j = 2 * correlate(w, a, 'valid')[j].

So: `grad = 2.0 * np.correlate(w, a, mode='valid')`
