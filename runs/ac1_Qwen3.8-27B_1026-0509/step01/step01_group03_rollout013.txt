We want to minimize max_k conv(a,a)_k subject to sum(a)=1, a>=0.

Let's denote the optimal value as M*. Then the score is 2n*M*.

At the optimum, by KKT:
- There exist multipliers λ_k >= 0 such that sum_k λ_k = 1 (normalization)
- For all k with λ_k > 0: conv(a,a)_k = M*
- For all i with a_i > 0: sum_k λ_k * ∂_i conv(a,a)_k = μ (Lagrange multiplier for sum constraint)
- ∂_i conv(a,a)_k = a_{k-i} + a_{i-k} (with out-of-range = 0)

sum_k λ_k * (a_{k-i} + a_{i-k}) = μ for all i with a_i > 0
sum_k λ_k * (a_{k-i} + a_{i-k}) >= μ for all i with a_i = 0

This is a system of equations/inequalities that characterizes the optimum.

The LP iteration essentially finds a sequence b that is "dual" to a, and the fixed point satisfies these conditions.

    deadline = t0 + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

        nonlocal best, best_score
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best = a.copy()
            best_score = s
        return s
    
    def normalize(a):
        s = a.sum()
        return a / s if s > 1e-15 else a
    
    def conv_full(a):
        return np.convolve(a, a, mode='full')
    
    def score_quick(a):
        c = conv_full(a)
        return 2.0 * len(a) * c.max() / (a.sum()**2)
    
    def project(v):
        """Project onto {x >= 0, sum(x) = 1}."""
        n = len(v)
        vs = np.sort(v)[::-1]
        cs = np.cumsum(vs)
        for k in range(n, 0, -1):
            t = (cs[k-1] - 1.0) / k
            if k == n or vs[k-1] >= t:
                break
        x = np.maximum(v - t, 0)
        return x / x.sum()
    
    def build_A(n, a):
        rows = np.empty(2*n*n - n, dtype=np.int32)  # overestimate
        cols = np.empty_like(rows)
        data = np.empty(len(rows))
        idx = 0
        for k in range(2*n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            cnt = hi - lo + 1
            rows[idx:idx+cnt] = k
            cols[idx:idx+cnt] = np.arange(lo, hi+1)
            data[idx:idx+cnt] = a[k - np.arange(lo, hi+1)]
            idx += cnt
        return coo_matrix((data[:idx], (rows[:idx], cols[:idx])), shape=(2*n-1, n)).tocsr()
    
    def lp_solve(a, M, time_limit):
        n = len(a)
        A = build_A(n, a)
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),

                     options={'time_limit': time_limit})
