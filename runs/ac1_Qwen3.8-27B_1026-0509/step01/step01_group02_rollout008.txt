    M = 1 << (3 * n - 2).bit_length()

    def fc(a):
        f = fft(a, M)
        return ifft(f * f).real[:2 * n - 1]

    def gs(a, w):
        fa = fft(a, M)
        fw = fft(w[::-1], M)
        return 2.0 * ifft(fa * fw).real[n - 1:2 * n - 1][::-1]

    def sc(a):
        s = a.sum()
        return 2.0 * n * fc(a).max() / (s * s) if s > 0.01 else 1e18

    def pj(v):
        u = np.sort(v)[::-1]
        cv = np.cumsum(u)
        c = u - cv / np.arange(1, n + 1) > 0
        if not c.any():
            return np.full(n, 1.0 / n)
        k = int(np.where(c)[0].max() + 1)
        return np.maximum(v - (cv[k - 1] - 1.0) / k, 0.0)

    a0 = np.maximum(p0, 0.0) if p0 is not None else np.ones(n)
    a0 /= a0.sum()
