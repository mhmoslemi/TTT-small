            val = np.exp(float(rng.normal(0.0, 0.4)))
            a2[pos:pos + w] *= val
            s2 = a2.sum()
            if s2 > 1e-15:
                a2 /= s2
            sv = 2.0 * n * fast_conv(a2).max()
            if sv < best_score:
                best_score = sv
                best_a = a2.copy()
                a = a2.copy()

    # Phase 4: Fine-tune from best
    if time.monotonic() < deadline - 20:
        a = best_a.copy()
        inner_loop(a, 20000, 0.05, 0.003, 0.995, 0.97, 5000, 0.02, 20)

    # Final validation
    final = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final) or final > 1e10:
        return [float(x) for x in (np.ones(n) / n)]
    return [float(x) for x in best_a]
