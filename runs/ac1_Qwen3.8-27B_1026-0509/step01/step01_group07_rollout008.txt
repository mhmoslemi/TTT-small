        """Evaluate using numpy convolution (exact same as evaluator)."""
        c = np.convolve(f, f)
        M = np.max(c)

            return np.inf

    def fast_conv(f):
        N = 1 << int(np.ceil(np.log2(2 * len(f))))  # next power of 2
        F = np.fft.rfft(f, N)
        return np.fft.irfft(F * F, N)[:2*len(f)-1]

    while time.monotonic() < phase1_deadline:

        start = int(rng.integers(0, n - width))
        f_trial = f.copy()
        # Multiply block by random factor
        f_trial[start:start+width] *= rng.uniform(0.1, 2.0)
        f_trial = np.clip(f_trial, 0, 1000)
        
        s = eval_fast(f_trial)
        if s < best_score:
            best_score = s
            best_f = f_trial.copy()
            f = f_trial.copy()

    alpha = 0.01  # step size
    best_alpha = alpha
    
    while time.monotonic() < deadline:
        # Compute autocorrelation
        c = fast_conv(f)
        max_c = np.max(c)
        S = np.sum(f)
        current_score = 2.0 * n * max_c / (S * S)
        
        if current_score < best_score:
            best_score = current_score
            best_f = f.copy()
        
        # Find peak lag
        peak_lag = int(np.argmax(c))
        
        # Subgradient: dg(peak_lag)/df[j] = 2*f[peak_lag - j]
        sg = np.zeros(n)
        # f[peak_lag - j] for valid j
        lo = max(0, peak_lag - (n-1))
        hi = min(n-1, peak_lag)
        for j in range(lo, hi+1):
            sg[j] = 2.0 * f[peak_lag - j]
        
        # Normalize
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-10:
            break
        
        # Step
        f_new = f - alpha * sg / sg_norm
        f_new = np.clip(f_new, 0, 1000)
        
        # Evaluate
        s_new = eval_fast(f_new)
        
        if s_new < current_score:
            f = f_new
            # Try to increase step size
            alpha = min(alpha * 1.1, 0.5)
        else:
            # Decrease step size
            alpha = max(alpha * 0.5, 0.001)
            # Try random perturbation instead
            if rng.random() < 0.3:
                f_new = f + rng.normal(0, 0.01 * f.max(), n)
                f_new = np.clip(f_new, 0, 1000)
                s_r = eval_fast(f_new)
                if s_r < current_score:
                    f = f_new
                    best_score = min(best_score, s_r)
                    best_f = f.copy()
    
    # Final validation
    final_score = evaluate_sequence(best_f.tolist())
    if final_score < best_score:
        best_score = final_score
    else:
        # Make sure best_f is still valid
        best_score = final_score
    
    return [float(x) for x in best_f]
