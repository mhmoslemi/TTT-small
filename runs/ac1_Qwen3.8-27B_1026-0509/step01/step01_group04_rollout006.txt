import time
import cvxpy as cp

def _convolve_matrix(f, n):
    """Build (2n-1) x n matrix A where A[t,k] = f[|t-k|]... no."""
    # A[t,k] = f[t-k] if 0<=t-k<n and 0<=k<n, else 0
    # Then (A @ g)[t] = sum_k A[t,k]*g[k] = sum_k f[t-k]*g[k] = (f*g)[t]
    A = np.zeros((2*n-1, n))
    for t in range(2*n-1):
        for k in range(n):
            d = t - k
            if 0 <= d < n:
                A[t, k] = f[d]

    valid = (d >= 0) & (d < n)
    return np.where(valid, f[np.clip(d, 0, n-1)], 0.0)

    g = cp.Variable(n, nonneg=True)

def _build_A(f, n):
    t = np.arange(2*n-1)[:, None]
    k = np.arange(n)[None, :]
    d = t - k
    valid = (d >= 0) & (d < n)
    return np.where(valid, f[np.clip(d, 0, n-1)], 0.0)

def _lp_blend(f, n, M, deadline):
    """LP: max sum(g) s.t. A@G <= M, g >= 0. Return g or None."""
    A = _build_A(f, n)
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                v = np.asarray(g.value).flatten()
                if v is not None and np.isfinite(v).all() and v.sum() > 1e-12:

            pass
