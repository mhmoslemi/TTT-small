            if time.monotonic() > deadline - 30:
                break
            a = lbfgs_opt(a, p, maxiter=600)
            try_update(a)
        if time.monotonic() < deadline - 10:
            a = lp_iter(a, max_iters=200)
            try_update(a)

    if best_seq is not None and time.monotonic() < deadline - 5:
        a_final = lp_iter(best_seq, max_iters=500)
        try_update(a_final)

        # Additional L-BFGS from the LP-polished result with high p
        if time.monotonic() < deadline - 5:
            for p in [50, 100, 200]:
                if time.monotonic() > deadline - 5:
                    break
                a2 = lbfgs_opt(a_final, p, maxiter=500)
                try_update(a2)
                a_final = a2

    if best_seq is not None:
        out = np.clip(best_seq, 0, 1000)
        fs = evaluate_sequence(out.tolist())
        if np.isfinite(fs):
            return [float(x) for x in out]
    return [1.0] * n
