from scipy.fft import fft, ifft
import time

def _conv(a):
    n = len(a)
    sz = 1 << (2 * n - 1).bit_length()
    fa = fft(a, sz)
    return ifft(fa * fa).real[:2 * n - 1]

def _proj(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    c = u - cv / np.arange(1, n + 1) > 0
    if not np.any(c):
        return np.full(n, 1.0 / n)
    i = int(np.max(np.where(c)[0]) + 1)
    return np.maximum(v - cv[i - 1] / i, 0.0)
