def _solve_lp_step(x):
    """Solve the linearized minimax LP to get a new x."""
    n = len(x)
    c0 = np.convolve(x, x)  # length 2n-1
    L = 2 * n - 1
    
    # Variables: x_0, ..., x_{n-1}, t
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: 2*(x0*x)_k - t <= c0[k]  for k = 0, ..., L-1
    # Coefficient of x_j in (x0*x)_k: x0[k-j] if 0 <= k-j <= n-1
    A_ub = np.zeros((L, n + 1))
    for k in range(L):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = 2.0 * x[idx]
        A_ub[k, n] = -1.0
    
    b_ub = c0.copy()
    
    # Equality: sum x_i = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # x_i >= 0, t free

                  bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return None
    return res.x[:n]

$$\min_t \quad t$$
$$\text{s.t.} \quad \text{tr}(A_k X) \le t \quad \forall k$$
$$\mathbf{1}^T x = 1$$
$$X = xx^T \text{ (relaxed to } X \succeq 0, \; X_{ii} = x_i \text{ for } i)$$
$$x \ge 0$$

This is getting complicated. Let me just stick with the LP linearization approach, which is simpler and proven to work.

from scipy.optimize import linprog
import time

def _lp_step(x0):
    """Solve linearized minimax LP. Returns new x or None."""
    n = len(x0)
    c0 = np.convolve(x0, x0)
    L = 2 * n - 1
    A = np.zeros((L, n + 1))
    for k in range(L):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = 2.0 * x0[k - j]
        A[k, n] = -1.0
    c_obj = np.zeros(n + 1); c_obj[n] = 1.0
    Ae = np.zeros((1, n + 1)); Ae[0, :n] = 1.0
    try:
        r = linprog(c_obj, A_ub=A, b_ub=c0, A_eq=Ae, b_eq=[1.0],
                    bounds=[(0, None)] * n + [(0, None)],
                    method='highs')
    except Exception:
        return None
    return r.x[:n] if r.success and r.x is not None else None

        if nx.sum() < 1e-12: break

                    best, best_p = x.copy(), float(np.convolve(x, x).max())
                    stall = 0
    return best
