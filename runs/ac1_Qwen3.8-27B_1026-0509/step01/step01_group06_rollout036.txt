```python
import numpy as np
from scipy.optimize import linprog
import time

def _lp_refine(f, n):
    """Solve LP to find g minimizing max_k (f*g)(k) subject to sum(g)=1, g>=0.
    Returns (g, M_prime)."""
    # Build A matrix: A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # LP: min M' s.t. A@x_g - M' <= 0, sum(x_g) = 1, x_g >= 0, M' >= 0
    # Variables: [g_0, ..., g_{n-1}, M']
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M'
    
    # Inequality constraints: A @ g <= M'  =>  A @ g - M' <= 0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(L)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        M_prime = res.x[n]
        return g, M_prime
    return None, None
```

            if np.isfinite(sc):
                best_sc = sc
                best_seq = parent.copy()
    except Exception:
        pass

    def make_inits(n):
        lst = []
        idx = np.arange(n, dtype=np.float64)
        nm1 = max(n - 1, 1)
        # Asymmetric shapes
        lst.append(np.ones(n))
        lst.append(1.0 - 0.5 * idx / nm1)
        lst.append(0.5 + 0.5 * idx / nm1)
        lst.append(1.0 - 0.4 * (idx / nm1) ** 2)
        lst.append(1.0 - 0.3 * (1 - idx / nm1) ** 2)
        # Sine shapes
        for k in (1, 2, 3):
            lst.append(1.0 + 0.3 * np.sin(2 * np.pi * k * idx / n + rng.uniform(0, 1)))
            lst.append(1.0 + 0.3 * np.cos(2 * np.pi * k * idx / n))
        # Random
        for _ in range(8):
            lst.append(np.abs(rng.normal(1.0, 0.35, n)) + 0.03)
        # Parent-based
        if parent is not None:
            if parent.size >= n:
                lst.append(parent[:n].astype(np.float64).copy())
            elif parent.size > 0:
                lst.append(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent))
        return lst

    # Collect and score all initializations
    lengths = [40, 36, 38, 34, 42, 32, 44]
    trials = []
    for n in lengths:
        if time.monotonic() > deadline - 10:
            break
        for p in make_inits(n):
            p = np.clip(p, 0.01, None)
            sc = evaluate_sequence(p.tolist())
            if np.isfinite(sc):
                trials.append((sc, n, p.copy()))

    trials.sort(key=lambda x: x[0])

    # Refine top trials with time budget
    n_refined = 0
    for i, (sc, n, p) in enumerate(trials):
        if time.monotonic() > deadline - 8:
            break
        if best_sc < 1e9 and sc > best_sc * 1.04:
            continue
        rem = deadline - time.monotonic()
        if rem < 8:
            break
        # Allocate time: generous for top trials, less for later
        if i < 3:
            ta = min(300.0, rem * 0.35)
        elif i < 8:
            ta = min(150.0, rem * 0.2)
        else:
            ta = min(60.0, rem * 0.1)
        ta = max(ta, 5.0)
        if rem - ta < 8:
            ta = rem - 10
            if ta < 3:
                break
        fb, sb = _refine(p, time.monotonic() + ta, rng, use_hot=(i % 2 == 1))
        if sb < best_sc:
            best_sc = sb
            best_seq = fb.copy()
        n_refined += 1
        if best_sc <= 1.50274:
            break

    # Final validation
    out = np.clip(best_seq.astype(np.float64), 0, 1000.0)
    if not (out.size > 0 and np.isfinite(out).all() and out.sum() > 0.01):
        out = np.ones(6)
    fs = evaluate_sequence(out.tolist())
    if not np.isfinite(fs):
        out = np.ones(6)
    return [float(x) for x in out]
