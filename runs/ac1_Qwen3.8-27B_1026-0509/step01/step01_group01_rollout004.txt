    a: length n
    p: length 2n-1 (softmax weights)
    L: FFT length >= 2n-1
    Returns: grad of length n
    """

    a_pad = np.zeros(L)
    a_pad[:n] = a
    p_pad = np.zeros(L)
    p_pad[:2*n-1] = p
    A = np.fft.rfft(a_pad)
    P = np.fft.rfft(p_pad)
