    """Compute the AC1 score for sequence a."""
    a = np.clip(a, 0, 1000)

    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (a.sum()**2)

def _lp_direction(a, M):
    """Solve LP to find best direction g with a*g <= M, g >= 0, maximizing sum(g)."""
    n = len(a)
