import numpy as np
from scipy.optimize import linprog
import time

def _score(a):
    n = len(a)
    b = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * float(b.max()) / (s * s)

def _conv_mat(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(m):
        lo = max(0, j - n + 1)
        hi = min(n, j + 1)
        if lo < hi:
            A[j, lo:hi] = a[j - np.arange(lo, hi)]
    return A

def _lp_step(a, M):
    n = len(a)
    A = _conv_mat(a)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0.0, None)] * n, method='highs')
    except Exception:
        return None
    if res.x is not None:
        return np.clip(res.x, 0.0, None)
    return None

def _refine(a0, deadline, rng):

    stall = 0
    perturbs = 0

    while time.monotonic() < deadline - 1.0:
        b = np.convolve(a, a)
        M = float(b.max())
        g = _lp_step(a, M)

        moved = False
        if g is not None:
            gs = float(g.sum())
            if gs > 1.0 + 1e-7:
                h = g / gs
                best_t = 0.0

                    sc = _score(cand)
                    if sc < best_local:

                if best_t > 0.0:

                a = np.clip(a, 1e-8, None)
                a /= a.sum()
                stall = 0

    return best, best_sc
