        S = max(float(best.sum()), 1e-10)
        b = np.convolve(best, best)
        local_best = 2.0 * n * b.max() / (S * S)

        b_len = 2 * n - 1

                break

            any_improved = False

                x_i = best[i]
                S_rest = S - x_i
                if S_rest < 1e-8 and x_i < 1e-8:
                    continue

                t_end = i + n  # always <= 2n-1 since i < n

                if i > 0:
                    C = b[:i].max()
                else:
                    C = 0.0

                bx = best  # reference, not copy

                    val = (aff + (v - x_i) * bx).max()

                    S_new = S_rest + v

                        return 1e10
                    return two_n * new_M / (S_new * S_new)

                                      options={'xatol': 1e-10, 'maxiter': 80})

                    continue

                    try:

                        if prob.status in ("optimal", "optimal_inaccurate"):
                            g_val = np.asarray(g_var.value).flatten()
                            if np.isfinite(g_val).all() and g_val.sum() > 1e-10:
                                break
                            g_val = None
                    except Exception:
                        continue
                
                if g_val is None:
                    break
                
                g_norm = g_val * (S / g_val.sum())
                new_f = 0.6 * f + 0.4 * g_norm
                new_f = np.clip(new_f, 0.0, None)
                
                sc = float(evaluate_sequence(new_f.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = new_f.copy()
                    f = new_f.copy()
        except Exception:
            pass

    x_work = best_seq.copy()
    n = len(x_work)

    update_best(x_cd)

        update_best(x_cd2)

    if not np.isfinite(final_score):
