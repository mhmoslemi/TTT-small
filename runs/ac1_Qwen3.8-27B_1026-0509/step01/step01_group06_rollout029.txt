def _convolve_max(a):
    b = np.convolve(a, a)
    return b.max(), b.argmax()

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}"""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    # First, project onto {x >= 0, sum(x) <= target_sum}
    # Then adjust to hit equality
    # Standard algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cssv - target_sum))
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _gradient_conv_max(a):
    """Compute subgradient of max_k convolve(a,a)[k]"""
    n = len(a)
    b = np.convolve(a, a)
    kmax = b.max()
    # Find all lags achieving the max
    active = np.where(b >= kmax * (1 - 1e-10))[0]
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(active)
    return b, kmax, grad

    # We project onto the simplex, which handles the constraint

def _project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)

        return np.full(n, s / n)

    theta = (cssv[rho] - s) / (rho + 1.0)
