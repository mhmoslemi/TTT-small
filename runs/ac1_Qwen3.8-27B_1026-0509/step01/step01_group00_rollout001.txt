        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        parent = np.ones(256)
    if parent.ndim != 1 or parent.size < 4:
        parent = np.ones(256)
    n = parent.size
    s = parent.sum()
    if s < 0.01:
        return [1.0] * 64
    p0 = parent / s
    
    fft_n = 1
    while fft_n < 2 * n - 1:
        fft_n <<= 1
    
    def conv(z):
        fx = np.fft.rfft(z, fft_n)
        return np.fft.irfft(fx * fx, fft_n)[:2 * n - 1]
    
    def maxc(z):
        return float(conv(z).max())
    
    def score(z):
        sz = z.sum()
        if sz < 1e-15:
            return float('inf')
        return 2.0 * n * maxc(z) / (sz * sz)
    
    def grad_maxc(z, K=25):
        c = conv(z)
        mc = c.max()
        # Soft selection: weight by (c_k / mc)^alpha for c_k > threshold
        alpha = 20.0
        w = np.where(c > mc * 0.99, (c / mc) ** alpha, 0.0)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= wsum
        
        g = np.zeros(n)
        # Use FFT-based approach for gradient
        # c_k = sum_i z[i] z[k-i] (full conv)
        # d(c_k)/d(z[j]) = z[k-j] (if valid) + z[j-k] (if valid)
        # This is like a "cross-correlation" of z with the weighted lag profile
        # Let's compute it efficiently:
        # For each active lag k with weight w_k:
        #   contribution to g[j] from lag k: w_k * (z[k-j] if 0<=k-j<n else 0) + w_k * (z[j-k] if 0<=j-k<n else 0)
        # This is: g = sum_k w_k * (shift_k(z) + shift_{-k}(z)) (with boundary handling)
        # In FFT terms: shift_k(z) is related to the inverse DFT of z*hatted * e^{-2πik/n}
        # But for linear (not cyclic) convolution, we need to be careful.
        
        # Simpler: for small K, just loop
        active = np.where(w > 1e-10)[0]
        for k in active:
            k = int(k)
            if k == 0:
                g += 2.0 * w[k] * z
            else:
                # z[k-j] for j in [0, min(k, n-1)]: valid when 0 <= k-j < n
                # j ranges from max(0, k-n+1) to min(k, n-1)
                # z[j-k] for j in [k, n-1]: valid when 0 <= j-k < n
                #   i.e., j ranges from k to n-1, and j-k ranges from 0 to n-1-k
                if k < n:
                    # First part: j in [max(0,k-n+1), k], contribution z[k-j]
                    lo = max(0, k - n + 1)
                    hi = min(k, n - 1)
                    if lo <= hi:
                        g[lo:hi+1] += w[k] * z[k - np.arange(lo, hi + 1)]
                    # Second part: j in [k, n-1], contribution z[j-k]
                    lo2 = k
                    hi2 = n - 1
                    if lo2 <= hi2:
                        g[lo2:hi2+1] += w[k] * z[np.arange(lo2, hi2 + 1) - k]
        
        return g
    
    def optimize(z0, max_iters, lr_init=0.3, phase='main'):
        z = z0.copy()
        z = np.maximum(z, 1e-16)
        z /= z.sum()
        
        best_z = z.copy()
        best_mc = maxc(z)
        lr = lr_init
        prev_mc = best_mc
        prev_prev_mc = best_mc
        
        for it in range(max_iters):
            if time.monotonic() > deadline - 5:
                break
            
            c = conv(z)
            mc = float(c.max())
            
            if mc < best_mc - 1e-9:
                best_mc = mc
                best_z = z.copy()
            
            g = grad_maxc(z)
            
            # Barzilai-Borwein inspired step
            if it > 0 and abs(prev_mc - mc) > 1e-12:
                # Not a true BB step for non-smooth, but adaptive
                pass
            if it > 1:
                if mc < prev_mc and prev_mc < prev_prev_mc:
                    lr = min(lr * 1.1, 1.5)  # accelerating
                elif mc > prev_mc:
                    lr *= 0.8  # decelerating
            prev_prev_mc = prev_mc
            prev_mc = mc
            
            # Multiplicative update
            z_new = z * np.exp(-lr * g)
            z_new = np.maximum(z_new, 1e-16)
            sz = z_new.sum()
            if sz < 1e-10:
                lr *= 0.5
                continue
            z = z_new / sz
            
            # Periodic perturbation
            if it % 3000 == 2999:
                # Random Fourier perturbation
                mode = np.zeros(fft_n // 2 + 1, dtype=complex)
                k_max = n // 2
                ks = rng.integers(1, k_max + 1, size=min(5, k_max))
                for k in ks:
                    mode[k] = (rng.normal() + 1j * rng.normal()) * 0.03
                    mode[-k if -k >= 0 else fft_n - k] = np.conj(mode[k]) if k < fft_n//2 else 0
                pert = np.fft.irfft(mode, fft_n)[:n]
                z *= np.exp(pert)
                z = np.maximum(z, 1e-16)
                z /= z.sum()
        
        return best_z, best_mc
    
    # Generate candidates
    cands = [p0.copy()]
    
    # Various structured initializations
    idx_f = np.arange(n, dtype=np.float64) / n
    
    # Exponential
    for r in np.exp(np.linspace(-0.5, 0.5, 5)):
        z = r ** np.arange(n, dtype=np.float64)
        z = np.maximum(z, 0.01)
        z /= z.sum()
        cands.append(z.copy())
    
    # Gaussian bumps
    for mu in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            z = np.exp(-0.5 * ((idx_f - mu) / sigma) ** 2)
            z = np.maximum(z, 0.01)
            z /= z.sum()
            cands.append(z.copy())
    
    # Cosine modulated
    for fr in [1, 2, 3, 7, 11]:
        for amp in [0.3, 0.6, 0.9]:
            z = 1.0 + amp * np.cos(2 * np.pi * fr * idx_f)
            z = np.maximum(z, 0.01)
            z /= z.sum()
            cands.append(z.copy())
    
    # Two bumps
    for d in [0.15, 0.2, 0.25, 0.3]:
        z = np.exp(-((idx_f - 0.5 + d) / 0.1) ** 2) + np.exp(-((idx_f - 0.5 - d) / 0.1) ** 2)
        z = np.maximum(z, 0.01)
        z /= z.sum()
        cands.append(z.copy())
    
    # Random
    for _ in range(3):
        z = np.abs(rng.standard_normal(n)) + 0.1
        z /= z.sum()
        cands.append(z.copy())
    
    # Evaluate
    best = cands[0].copy()
    best_mc = maxc(cands[0])
    for c in cands[1:]:
        mc = maxc(c)
        if mc < best_mc:
            best_mc = mc
            best = c.copy()
    
    # Optimize
    nc = len(cands)
    # Allocate time: give more to top candidates
    sorted_idx = np.argsort([maxc(c) for c in cands])
    
    for rank, ci in enumerate(sorted_idx):
        if time.monotonic() > deadline - 30:
            break
        base_it = max(500, int(30_000_000 // nc))
        if rank < 3:
            iters = base_it * 2
        elif rank < nc // 2:
            iters = base_it
        else:
            iters = max(200, base_it // 2)
        
        bz, bmc = optimize(cands[ci], iters, lr_init=0.3)
        if bmc < best_mc:
            best_mc = bmc
            best = bz.copy()
    
    # Final polish
    if time.monotonic() < deadline - 20:
        bz, bmc = optimize(best, 100000, lr_init=0.05)
        if bmc < best_mc:
            best_mc = bmc
            best = bz.copy()
    
    best = np.clip(best, 0.0, 1000.0)
    result = [float(x) for x in best]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs) or fs >= 2.0:
        # Fallback
        result = [1.0] * n
        fs = evaluate_sequence(result)
    return result
