Let me also think about the LP refinement more carefully. The LP is:
Maximize $\sum g$ subject to $g \ge 0$ and for all $t$: $\sum_k A[t,k] g[k] \le M$

where $M = \max_t (f*f)_t$ and $A[t,k]$ is the convolution matrix entry.

But wait - $(f*g)_t = \sum_k f[\max(0, t-k)] f[\min(n-1, t-k)]$... no, that's not right either.

Actually, $(f*g)_t = \sum_k f[k] g[t-k]$ (cross-correlation/convolution). The constraint is $(f*g)_t \le M$ for all $t$.

So the LP matrix $B$ has entries $B[t,k] = f[t-k]$ if $0 \le t-k < n$, else 0. This is a Toeplitz-like matrix (actually a "cross-correlation" matrix of $f$).

For the LP, I can build this matrix efficiently:
```python
def _build_cross_corr_matrix(f, n):
    """B[t,k] = f[t-k] if 0<=t-k<n else 0, where t in 0..2n-2, k in 0..n-1"""
    B = np.zeros((2*n-1, n))
    for k in range(n):
        # B[t,k] = f[t-k] for t in k..k+n-1
        B[k:k+n, k] = f
    return B
```

This is O(n^2) which is fine for n=400 (160K operations).

The LP has:
- Variables: $g \in \mathbb{R}^n$, $g \ge 0$

For n=400: 400 variables, 799 constraints. This is a small LP that should solve in milliseconds with CLARABEL or similar.

            idx = rng.choice(n, size=frac, replace=False)
            f_p = f.copy()
            f_p[idx] *= np.exp(rng.normal(0, 0.4, size=frac))
            f_p = np.maximum(f_p, 0)
            s = f_p.sum()
            if s > 1e-15:
                f_p /= s
                sc = float(evaluate_sequence(f_p.tolist()))
                if sc < best_sc:
                    f = f_p
                    no_improve = 0

    return best, best_sc


def _lp_refine(f, n, deadline):
    c = np.convolve(f, f)
    M = float(c.max())
    B = np.zeros((2 * n - 1, n))
    for k in range(n):
        B[k:k + n, k] = f
    var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(var)), [B @ var <= M])
    val = None
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                val = np.asarray(var.value).flatten()
                break
        except Exception:
            continue
    if val is None or not np.isfinite(val).all() or val.sum() < 1e-10:
        return f.copy(), float(evaluate_sequence(f.tolist()))
    g = val / val.sum()
    best_f = f.copy()
    best_sc = float(evaluate_sequence(f.tolist()))
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 0.95]:
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new /= s
        sc = float(evaluate_sequence(f_new.tolist()))
        if sc < best_sc:
            best_sc = sc
            best_f = f_new
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = np.inf

    def update_best(f):
        nonlocal best_seq, best_score
        if f is None or len(f) == 0:
            return
        f = np.asarray(f, dtype=np.float64)
        f = np.clip(f, 0.0, 1000.0)
        if f.sum() < 0.01:
            return
        sc = float(evaluate_sequence(f.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()

    inits = []
    try:
        h1 = np.array(height_sequence_1, dtype=np.float64)
        inits.append(np.maximum(h1, 0))
    except Exception:
        pass
    for n in [300, 400, 500, 600, 800]:
        j = np.arange(n, dtype=np.float64)
        inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * j / n)))
    for n in [400, 600]:
        inits.append(np.hamming(n))
    inits.append(np.ones(400))

    valid = []
    for f0 in inits:
        f0 = np.maximum(f0, 0.0)
        if f0.sum() >= 0.01:
            valid.append(f0 / f0.sum())

    n_init = len(valid)
    remaining = deadline - time.monotonic()
    time_per = max(3.0, remaining / max(1, n_init) - 1.0)

    for i, f0 in enumerate(valid):
        if time.monotonic() > deadline - 20:
            break
        update_best(f0)
        t_end = min(time.monotonic() + time_per, deadline - 5)
        if i >= n_init - 2:
            t_end = deadline - 5
        f_opt, sc = _optimize(f0, t_end, rng)
        update_best(f_opt)

    if best_seq is None:
        n = 400
        j = np.arange(n)
        f0 = np.sin(np.pi * (j + 0.5) / n)
        update_best(f0)
        f_opt, sc = _optimize(f0 / f0.sum(), deadline - 5, rng)
        update_best(f_opt)

    # Final polish on best
    if time.monotonic() < deadline - 8 and best_seq is not None:
        f = best_seq.copy() / best_seq.sum()
        f_opt, sc = _optimize(f, deadline - 2, rng)
        update_best(f_opt)

    final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    if not np.isfinite(final_sc):
        n = 400
        j = np.arange(n)
        best_seq = np.sin(np.pi * (j + 0.5) / n)

    return [float(x) for x in best_seq]
