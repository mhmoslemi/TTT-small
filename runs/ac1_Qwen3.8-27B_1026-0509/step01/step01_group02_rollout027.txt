def _ps(v):
    """Project onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    cond = u - cv / np.arange(1, n + 1) > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    i = int(np.max(np.where(cond)[0]) + 1)
    return np.maximum(v - cv[i - 1] / i, 0.0)
