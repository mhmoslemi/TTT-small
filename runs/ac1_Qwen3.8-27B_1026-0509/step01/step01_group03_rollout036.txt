        v = v.copy()

        lo, hi = v.min() - 1, v.max() + 1

            if s > target_sum:
                lo = mid
            else:
                hi = mid

    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A[k, j] = a[k-j] for j in [max(0,k-n+1), min(k, n-1)]

                      options={'time_limit': max(1.0, tl), 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-10:
                return b / bs, False
        return a, True

    def polish(a, max_iters=80):
        n = len(a)
        beta = 30.0
        lr = 0.015
        for _ in range(max_iters):
            if time.monotonic() > deadline - 15:
                break
            conv_a = np.convolve(a, a)
            cm = conv_a.max()
            if cm < 1e-15:
                break
            w = np.exp(np.clip((conv_a - cm) * beta / cm, -50, 0))
            w /= w.sum()
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-9:
                    continue
                s = max(0, k - n + 1)
                e = min(n - 1, k)
                grad[s:e + 1] += w[k] * 2.0 * a[k - np.arange(s, e + 1)]
            a = proj_simplex(a - lr * grad, 1.0)
            try_score(a)
            lr *= 0.97
            beta = min(500.0, beta * 1.1)
        return a

    inits = []
    try:
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if p.size > 0 and np.isfinite(p).all():
            inits.append(p.copy())
    except:
        pass
    inits.append(rng.random(200) * 0.5 + 0.01)
    x = np.arange(200)
    inits.append(np.clip(0.6 + 0.35 * np.sin(2 * np.pi * 2 * x / 200), 0.01, None))
    inits.append(np.clip(0.5 + 0.3 * np.cos(2 * np.pi * 5 * x / 200), 0.01, None))

    for a0 in inits:
        if time.monotonic() > deadline - 50:
            break
        a = np.maximum(a0.astype(np.float64), 0)
        ss = a.sum()
        if ss < 0.01:
            continue
        a /= ss
        try_score(a)

        t = 0.5
        prev_sc = best_score
        stall = 0
        while time.monotonic() < deadline - 40:
            tl = deadline - time.monotonic() - 35
            if tl < 5:
                break
            b, done = do_lp(a, tl)
            if done:
                break
            a = proj_simplex((1 - t) * a + t * b, 1.0)
            sc = try_score(a)
            if sc < prev_sc - 1e-7:
                stall = 0
                prev_sc = sc
            else:
                stall += 1
            if stall >= 20:
                t = max(0.05, t * 0.6)
                stall = 0
            t = max(0.05, t * 0.98)

        a = polish(a, 60)

    if best is not None and time.monotonic() < deadline - 40:
        a = best.copy()
        a /= a.sum()
        t = 0.3
        while time.monotonic() < deadline - 25:
            tl = deadline - time.monotonic() - 22
            if tl < 5:
                break
            b, done = do_lp(a, tl)
            if done:
                break
            a = proj_simplex((1 - t) * a + t * b, 1.0)
            try_score(a)
            t = max(0.03, t * 0.97)
        polish(a, 40)

    if best is None:
        best = np.ones(100)
    result = [float(v) for v in np.clip(best, 0, 1000)]
    if len(result) == 0 or not np.isfinite(evaluate_sequence(result)):
        result = [1.0] * 100
    return result
