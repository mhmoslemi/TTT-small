def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}"""
    if v.sum() < s:
        # Already in the right sum but might have negatives
        # Clip and renormalize approximately
        v = np.maximum(v, 0)
        if v.sum() < 1e-15:
            return np.full_like(v, s / len(v))
        return v / v.sum() * s
    n = len(v)
    # Standard projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
