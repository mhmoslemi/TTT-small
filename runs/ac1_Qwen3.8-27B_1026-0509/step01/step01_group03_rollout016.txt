        prev_M = np.inf
        for _ in range(50000):
            if time.monotonic() > deadline:
                break
            conv_a = np.convolve(a, a)
            M = conv_a.max()
            if M >= prev_M - 1e-12:
                stall += 1
            else:
                stall = 0
                prev_M = M
            if stall > 15:
                pert = 0.08 * rng.random(n) * a
                a = normalize(np.maximum(a + pert, 0))
                stall = 0
                continue
            b = solve_lp(a, M, rows, cols)
            if b is None:
                break
            bs = b.sum()
            if bs < 1e-12:
                break
            b /= bs
            if np.max(np.abs(b - a)) < 1e-9:
                break
            best_t = 0.0
            best_Mn = M
            for t_try in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:
                a_try = normalize(np.maximum((1 - t_try) * a + t_try * b, 0))
                Mn = np.convolve(a_try, a_try).max()
                if Mn < best_Mn - 1e-14:
                    best_Mn = Mn
                    best_t = t_try
            if best_t > 0:
                a = normalize(np.maximum((1 - best_t) * a + best_t * b, 0))
            else:
                pert = 0.05 * rng.random(n) * a
                a = normalize(np.maximum(a + pert, 0))
        return a

    # Gather initializations
    inits = []
    try:
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            inits.append(('parent', p.copy()))
    except:
        pass

    for n in [200, 150, 100, 250, 300]:
        inits.append(('rand', rng.random(n) + 0.01))
    # Smooth init
    n = 200
    x = rng.random(n)
    for _ in range(5):
        x = np.convolve(x, np.ones(7) / 7, mode='same') + 0.005
    inits.append(('smooth', x))

    # Allocate time: parent gets 60%, others share remaining
    total_budget = deadline - t0
    parent_budget = total_budget * 0.55
    other_budget = total_budget * 0.40

    for tag, a0 in inits:
        if time.monotonic() > deadline - 20:
            break
        a = np.maximum(a0.astype(np.float64), 0)
        n = len(a)
        if n < 2 or n > 500:
            continue
        a = normalize(a)
        try_score(a)

        # Time allocation
        if tag == 'parent':
            sub_deadline = min(deadline - 20, t0 + t0 and (deadline - t0) * 0.7)
        else:
            sub_deadline = deadline - 30
            if time.monotonic() > deadline - 30:
                break

        rows, cols = build_struct(n)
        a = run_lp_opt(a, rows, cols, sub_deadline, rng)
        try_score(a)

    # Final polish on best
    if best is not None and time.monotonic() < deadline - 10:
        a = best.copy()
        n = len(a)
        a = normalize(a)
        rows, cols = build_struct(n)
        a = run_lp_opt(a, rows, cols, deadline - 10, rng)
        try_score(a)

    if best is None:
        best = np.ones(100)

    result = [float(x) for x in np.clip(best, 0, 1000)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 100
    return result
