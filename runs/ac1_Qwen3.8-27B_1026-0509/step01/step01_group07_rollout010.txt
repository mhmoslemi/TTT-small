            break
        dat = build_data(f)
        A = csr_matrix((dat, col_idx, row_ptr), shape=(total_rows, n))
        tl = max(2.0, deadline - time.monotonic() - 3.0)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=M * np.ones(total_rows),
                         method='highs', options={'time_limit': tl})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        g = np.maximum(res.x, 0)
        sg = g.sum()
        if sg < 1e-12:
            break
        g /= sg
        accepted = False
        for t in [0.5, 0.3, 0.15, 0.05, 0.01]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            ss = f_new.sum()
            if ss < 1e-15:
                continue
            f_new /= ss
            sc = evaluate_sequence(f_new.tolist())
            if np.isfinite(sc) and sc < best_score:
                f = f_new
                best_f = f_new.copy()
                best_score = sc
                accepted = True
                break
        if not accepted:
            if time.monotonic() > deadline - 20:
                break
            rng = np.random.default_rng()
            mask = rng.random(n) < 0.1
            f[mask] *= np.exp(rng.normal(0, 0.3, mask.sum()))
            f = np.maximum(f, 0)
            f /= f.sum()

    final = float(evaluate_sequence(best_f.tolist()))
    if np.isfinite(final):
        best_score = min(best_score, final)
        if final < 2.0:
            pass
    return [float(x) for x in best_f]
