```python
import numpy as np
import time
import cvxpy as cp

def _fft_conv(x, y=None):
    if y is None:
        y = x
    n = len(x) + len(y) - 1
    m = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(x, m) * np.fft.rfft(y, m), m)[:n]

def _proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.searchsorted(cv - u * np.arange(1, n+1), s, side='left')
    if rho >= n:
        rho = n - 1
    theta = (cv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)

def _optimize(x0, max_iters=50000, deadline=None):
    n = len(x0)
    x = x0.copy()
    x = x / x.sum()
    best = x.copy()
    best_sc = float(evaluate_sequence(x.tolist()))
    lr = 0.01
    stale = 0
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        c = _fft_conv(x)
        cmax = c.max()
        active = np.nonzero(c >= cmax * (1 - 1e-4))[0]
        
        # Subgradient
        g = np.zeros(n)
        for t in active:
            idx = np.arange(n)
            j = t - idx
            valid = (j >= 0) & (j < n)
            gv = np.where(valid, 2.0 * x[np.clip(j, 0, n-1)], 0.0)
            # Fix diagonal
            if t % 2 == 0:
                mid = t // 2
                if 0 <= mid < n:
                    gv[mid] = x[mid]  # instead of 2*x[mid]
            g += gv
        g /= len(active)
        
        x = _proj_simplex(x - lr * g, 1.0)
        
        # Adaptive lr
        if (it + 1) % 200 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if sc < best_sc:
                best_sc = sc
                best = x.copy()
                stale = 0
                lr = max(lr * 1.05, 0.001)
            else:
                stale += 1
                lr *= 0.9
                if stale > 10:
                    # Try a random perturbation
                    x = _proj_simplex(x + 0.01 * np.random.randn(n), 1.0)
                    stale = 0
                if lr < 1e-5:
                    break
        
    sc = float(evaluate_sequence(x.tolist()))
    if sc < best_sc:
        best_sc = sc
        best = x.copy()
    return best, best_sc

def _lp_polish(x, max_iters=20, deadline=None):
    """LP-based refinement."""
    n = len(x)
    best = x.copy()
    best_sc = float(evaluate_sequence(x.tolist()))
    tau = 0.3
    
    for _ in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        M = _fft_conv(best).max()
        # Build constraint: conv(best, g)[t] <= M for all t
        # A[t, i] = best[t-i] if valid
        # Solve: max sum(g) s.t. A@g <= M, g >= 0
        g = cp.Variable(n, nonneg=True)
        # Use matrix-free: the constraint is conv(best, g) <= M
        # We need to build A explicitly for the solver
        A = np.zeros((2*n-1, n))
        for t in range(2*n-1):
            i_min = max(0, t - (n-1))
            i_max = min(n-1, t)
            for i in range(i_min, i_max+1):
                A[t, i] = best[t-i]
        
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
        solved = False
        for solver in [cp.PDP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    gv = np.asarray(g.value).flatten()
                    if np.isfinite(gv).all() and gv.sum() > 1e-10:
                        solved = True
                        break
            except:
                continue
        
        if not solved:
            break
        
        gn = gv * (best.sum() / gv.sum())
        new_x = (1 - tau) * best + tau * gn
        new_x = _proj_simplex(new_x, 1.0)
        
        sc = float(evaluate_sequence(new_x.tolist()))
        if sc < best_sc:
            best = new_x
            best_sc = sc
            tau = min(tau * 1.1, 0.5)
        else:
            tau *= 0.7
            if tau < 0.05:
                break
    
    return best, best_sc
```

Now, the simplex projection. The standard algorithm:
1. Sort v in descending order: u[0] >= u[1] >= ... >= u[n-1]
2. Find largest rho such that u[rho] > (sum(u[0:rho+1]) - s) / (rho+1)
3. theta = (sum(u[0:rho+1]) - s) / (rho+1)
4. Projected x = max(v - theta, 0)

This is O(n log n) due to sorting. For 100K iterations with n=3000, that's 100K * 3000 * log(3000) ≈ 100K * 3000 * 12 ≈ 3.6 * 10^9 operations. That might be too slow in Python.

- If v is already mostly in the simplex (small negative values), we can just clip and rescale
- For a more general approach, use the bisection method on theta

Bisection on theta: find theta such that sum(max(v - theta, 0)) = s. This is O(n * log(precision)) per iteration, with about 60 iterations for double precision. Total: O(n * 60) = O(n).
