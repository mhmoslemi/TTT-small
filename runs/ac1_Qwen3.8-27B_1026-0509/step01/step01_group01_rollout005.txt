import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 30.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
            raise ValueError
    except Exception:
        parent = np.ones(5582, dtype=np.float64)

    n = parent.size
    fft_len = 1 << (2 * n - 1).bit_length()

    def autocorr(a):
        A = np.fft.rfft(a, fft_len)
        return np.fft.irfft(A * A, fft_len)[:2 * n - 1]

    def score(a):
        c = autocorr(a)
        s = a.sum()
        if s < 1e-15:
            return 1e15
        return float(2.0 * n * c.max() / (s * s))

    def softmax_gradient_step(a, T, lr):
        c = autocorr(a)
        c_max = c.max()
        w = np.exp((c - c_max) / T)
        W = w.sum()
        if W < 1e-30:
            return a
        w /= W
        AF = np.fft.rfft(a[::-1], fft_len)
        WF = np.fft.rfft(w, fft_len)
        conv = np.fft.irfft(AF * WF, fft_len)
        grad = conv[n - 1:2 * n - 1]
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a_new / s

    def subgrad_step(a, lr):
        c = autocorr(a)
        M = c.max()
        if M < 1e-30:
            return a
        thresh = M * (1.0 - 1e-10)
        active = np.flatnonzero(c >= thresh)
        if len(active) > 300:
            active = np.argpartition(c, -300)[-300:]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi > lo:
                grad[lo:hi] += a[k - hi + 1:k - lo + 1][::-1]
        grad /= float(len(active))
        grad -= grad.mean()
        gn = np.abs(grad).max()
        if gn < 1e-18:
            return a
        a_new = a - lr * grad / gn
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a_new / s

    def run_smooth(a, n_iter, T0, T1, lr0, lr1):
        nonlocal best_a, best_score
        for it in range(n_iter):
            if time.monotonic() > deadline - 10:
                break
            fr = it / max(1, n_iter - 1)
            T = T0 * (T1 / T0) ** fr
            lr = lr0 * (lr1 / lr0) ** fr
            a = softmax_gradient_step(a, T, lr)
            if it % 1000 == 0:
                sc = score(a)
                if sc < best_score:
                    best_a = a.copy()
                    best_score = sc
        sc = score(a)
        if sc < best_score:
            best_a = a.copy()
            best_score = sc
        return a

    def run_subgrad(a, n_iter, lr0, decay):
        nonlocal best_a, best_score
        lr = lr0
        for it in range(n_iter):
            if time.monotonic() > deadline - 10:
                break
            a = subgrad_step(a, lr)
            lr = lr0 * (decay ** it)
            if it % 500 == 0:
                sc = score(a)
                if sc < best_score:
                    best_a = a.copy()
                    best_score = sc
        sc = score(a)
        if sc < best_score:
            best_a = a.copy()
            best_score = sc
        return a

    a = parent / parent.sum()
    best_a = a.copy()
    best_score = score(best_a)

    # Phase 1: Coarse (high T)
    a = run_smooth(a, 20000, 1.0, 0.2, 0.4, 0.08)
    # Phase 2: Medium
    a = run_smooth(a, 20000, 0.2, 0.03, 0.08, 0.015)
    # Phase 3: Fine
    a = run_smooth(a, 20000, 0.03, 0.003, 0.015, 0.003)
    # Phase 4: Subgradient refinement
    a = run_subgrad(a, 40000, 0.08, 0.99995)

    # Phase 5: Restart with perturbation
    for r in range(40):
        if time.monotonic() > deadline - 40:
            break
        amp = 0.15 + 0.55 * (r / 40.0)
        a = best_a * np.exp(rng.normal(0.0, amp, n))
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / max(s, 1e-15)
        a = run_smooth(a, 6000, 0.05, 0.003, 0.05, 0.005)
        a = run_subgrad(a, 8000, 0.05, 0.9999)

    # Phase 6: Final fine-tune
    if time.monotonic() < deadline - 20:
        a = best_a.copy()
        a = run_subgrad(a, 30000, 0.02, 0.99999)

    result = [float(x) for x in best_a]
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs) or fs > 10:
        result = [float(x) for x in parent]

    return result
