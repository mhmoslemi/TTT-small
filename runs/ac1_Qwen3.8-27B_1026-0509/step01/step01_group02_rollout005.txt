def _smooth_grad(a, c, eps):
    n = len(a)
    # c = conv(a, a), length 2n-1
    # p = softmax(c / eps)
    c_scaled = c / eps
    c_scaled -= c_scaled.max()  # numerical stability
    p = np.exp(c_scaled)
    p /= p.sum()
    
    # g[j] = sum_k p_k * a[k-j] (cross-correlation of p and a)
    size = 1 << (3*n).bit_length()
    P = fft(p, size)
    A = fft(a, size)
    g = ifft(P * np.conj(A)).real[:n]
    return 2.0 * g
