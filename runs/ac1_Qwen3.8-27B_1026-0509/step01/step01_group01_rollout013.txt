    def do_conv(a):
        A = np.fft.rfft(a, fft_len)
        return np.fft.irfft(A * A, fft_len)[:conv_len]

    def do_grad(a, p):
        r = a[::-1]
        P = np.fft.rfft(p, grad_len)
        R = np.fft.rfft(r, grad_len)
        res = np.fft.irfft(P * R, grad_len)[:conv_len + n - 1]
        return res[n - 1:2 * n - 1]

    def score(a):
        s = a.sum()
        if s < 1e-15:
            return np.inf
        return 2.0 * n * do_conv(a).max() / (s * s)

    def time_ok(marg=0):
        return time.monotonic() < deadline - marg

    a = parent / parent.sum()
    best_a = a.copy()
    best_score = score(a)

    def optimize(a, max_iters, lr_init, beta_init, score_every=500):
        nonlocal best_a, best_score
        lr = lr_init
        beta = beta_init
        for it in range(max_iters):
            if not time_ok(25):
                break
            c = do_conv(a)
            m = c.max()
            if m < 1e-30:
                break
            scaled = beta * (c - m)
            scaled = np.maximum(scaled, -50.0)
            p = np.exp(scaled)
            ps = p.sum()
            if ps < 1e-30:
                break
            p /= ps
            grad = do_grad(a, p)
            g = grad - grad.mean()
            gmax = np.abs(g).max()
            if gmax < 1e-20:
                break
            g /= gmax
            a = a * np.exp(-lr * g)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a /= s
            if it % 200 == 0:
                lr = lr_init * (0.995 ** (it // 200))
                beta = min(beta_init + 0.05 * (it // 200), 300.0)
            if it % score_every == 0:
                sc = score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        return a

    # Phase 1: Main subgradient descent
    a = optimize(a, 600000, 0.12, 20.0)

    # Phase 2: Restart from best with varied beta
    for rb in [40.0, 80.0, 150.0, 250.0]:
        if not time_ok(30):
            break
        a = best_a.copy()
        # Recompute with sharper softmax
        c = do_conv(a)
        m = c.max()
        p = np.exp(np.maximum(rb * (c - m), -50.0))
        p /= p.sum()
        grad = do_grad(a, p)
        g = grad - grad.mean()
        gm = np.abs(g).max()
        if gm > 1e-20:
            g /= gm
            a = a * np.exp(-0.05 * g)
            a = np.maximum(a, 0)
            a /= a.sum()
        optimize(a, 10000, 0.06, rb, score_every=200)

    # Phase 3: Perturbation restarts from best
    for restart in range(80):
        if not time_ok(20):
            break
        sigma = 0.1 + 0.5 * (restart / 80.0)
        a = best_a * np.exp(rng.normal(0.0, sigma, n))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a /= s
        optimize(a, 4000, 0.08, 60.0, score_every=300)

    # Phase 4: Block-wise local search
    if time_ok(30):
        a = best_a.copy()
        for bs in [8, 32, 128, 512]:
            if not time_ok(10):
                break
            ntries = max(100, 20000 // bs)
            for _ in range(ntries):
                if not time_ok(10):
                    break
                start = int(rng.integers(0, max(1, n - bs)))
                a2 = a.copy()
                a2[start:start + bs] *= np.exp(float(rng.normal(0.0, 0.4)))
                a2 = np.maximum(a2, 0.0)
                s = a2.sum()
                if s < 1e-15:
                    continue
                a2 /= s
                sc = score(a2)
                if sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    a = a2.copy()

    # Phase 5: Fine-tune
    if time_ok(20):
        a = best_a.copy()
        optimize(a, 20000, 0.02, 200.0, score_every=500)

    final_list = [float(x) for x in best_a]
    fs = float(evaluate_sequence(final_list))
    if not np.isfinite(fs) or fs > 1e10:
        return [float(x) for x in parent.tolist()]
    return final_list
