        g = np.convolve(a, a, mode='full')
        return 2.0 * n * g.max() / (a.sum()**2)

        n = len(a)
        g = np.convolve(a, a, mode='full')
        g_max = g.max()
        # Find all k that achieve (or nearly achieve) the max
        tol = 1e-10 * g_max + 1e-30
        K_star = np.where(g >= g_max - tol)[0]
        # Subgradient: average of gradients at K_star
        # grad_k = 2 * shift(f, k) where (shift(f,k)_j = f_{k-j} if valid)
        # = 2 * correlate_indicator
        grad = np.zeros(n)
        for k in K_star:
            # grad_k[j] = 2 * f[k-j] if 0 <= k-j < n

            hi = min(k, n - 1)

        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best = a.copy()
            best_score = s
        return s

    def norm(a):
        s = a.sum()
        return a / s if s > 1e-15 else a

    def grad_and_obj(a, beta):
        n = len(a)
        g = np.convolve(a, a, mode='full')
        obj_val = 2.0 * n * g.max() / (a.sum() ** 2)
        gm = g.max()
        p = np.exp(beta * (g - gm))
        ps = p.sum()
        if ps < 1e-300:
            return np.zeros(n), obj_val
        p /= ps
        grad = 4.0 * n * np.correlate(p, a, mode='valid')
        return grad, obj_val

    def proj_step(a, grad, eta):
        na = a - eta * grad
        lo, hi = na.min() - 1.0, na.max() + 1.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            s = np.maximum(na - mid, 0.0).sum()
            if s > 1.0:
                lo = mid
            else:
                hi = mid
        na = np.maximum(na - (lo + hi) * 0.5, 0.0)
        s = na.sum()
        return na / s if s > 1e-15 else a

    def build_A(a):
        n = len(a)
        k_vals = np.arange(2 * n - 1)
        js = np.maximum(0, k_vals - n + 1)
        je = np.minimum(n - 1, k_vals)
        cnt = je - js + 1
        F_rows = np.repeat(k_vals, cnt)
        F_cols = np.concatenate([np.arange(js[k], je[k] + 1) for k in range(2 * n - 1)])
        F_data = a[F_rows - F_cols]
        return csr_matrix((F_data, (F_rows, F_cols)), shape=(2 * n - 1, n))

    def lp_dir(a, dl):
        n = len(a)
        if n > 3000:
            return None
        g = np.convolve(a, a, mode='full')
        M = g.max()
        if M < 1e-15:
            return None
        A = build_A(a)
        tl = max(1.0, dl - time.monotonic() - 30)
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': tl})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-10:
                return b / bs
        return None

    def optimize(a, dl):
        a = norm(a.astype(np.float64).copy())
        n = len(a)
        if n < 2:
            return a
        try_score(a)

        # LP phase
        n_lp = min(35, max(5, int(7000.0 / n)))
        for it in range(n_lp):
            if time.monotonic() > dl - 30:
                break
            b = lp_dir(a, dl)
            if b is None:
                break
            sc0, _ = grad_and_obj(a, 1.0)
            improved = False
            for t in [0.25, 0.35, 0.5, 0.15, 0.6]:
                a_new = norm(np.maximum((1 - t) * a + t * b, 0))
                sc_new, _ = grad_and_obj(a_new, 1.0)
                if sc_new < sc0:
                    a = a_new
                    improved = True
                    break
            if not improved:
                break
            try_score(a)

        # Gradient phase with annealing
        beta = 10.0
        eta = 0.05 / n
        stall = 0
        for it in range(1500):
            if time.monotonic() > dl - 30:
                break
            grad, sc_cur = grad_and_obj(a, beta)
            gm = np.abs(grad).max()
            if gm < 1e-14:
                break
            improved = False
            for _ in range(25):
                a_new = proj_step(a, grad, eta)
                _, sc_new = grad_and_obj(a_new, beta)
                if sc_new < sc_cur - 1e-15:
                    a = a_new
                    improved = True
                    break
                eta *= 0.5
            if improved:
                try_score(a)
                stall = 0
            else:
                eta *= 0.5
                stall += 1
                if eta < 1e-14:
                    break
                if stall > 40:
                    # Perturb and reset
                    a = norm(np.maximum(a + 0.01 * rng.standard_normal(n), 0))
                    eta = 0.05 / n
                    stall = 0
            if it % 25 == 0:
                beta = min(3000.0, beta * 1.4)

        # Final LP polish
        for it in range(15):
            if time.monotonic() > dl - 30:
                break
            b = lp_dir(a, dl)
            if b is None:
                break
            sc0, _ = grad_and_obj(a, 1.0)
            for t in [0.1, 0.15, 0.2]:
                a_new = norm(np.maximum((1 - t) * a + t * b, 0))
                sc_new, _ = grad_and_obj(a_new, 1.0)
                if sc_new < sc0:
                    a = a_new
                    break
            try_score(a)

        a = norm(a)
        try_score(a)
        return a

    # Build initializations
    inits = []
    try:
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if p.size >= 2 and np.isfinite(p).all() and p.sum() > 0.01:
            inits.append(p.copy())
    except:
        pass
    if not inits:
        inits.append(np.ones(200))

    n0 = len(inits[0])
    inits.append(rng.random(n0) + 0.05)
    x = np.arange(n0, dtype=np.float64)
    inits.append(np.exp(-0.5 * ((x - n0 / 2) / (n0 / 4)) ** 2) + 0.01)
    inits.append(np.ones(n0))
    # Larger length
    inits.append(np.ones(400))
    inits.append(np.ones(600) * 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(600) / 600 * 3) + 0.1)

    # Optimize each
    for a0 in inits:
        if time.monotonic() > deadline - 30:
            break
        optimize(a0, deadline)

    if best is None:
        best = np.ones(100)

    result = [float(v) for v in np.clip(best, 0, 1000)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 100
    return result
