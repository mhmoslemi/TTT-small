        t2 = min(deadline - time.monotonic() - 40, 300.0)
        for restart in range(5):
            if time.monotonic() > deadline - 40:
                break
            pert = best_a.copy()
            w = max(3, len(pert) // 50)
            s = int(rng.integers(0, len(pert) - w + 1))
            pert[s:s+w] *= np.exp(rng.normal(0, 0.3, size=w))
            pert = np.clip(pert, 0.0, 1000.0)
            a_p = coord_desc(pert, 80, deadline - 40)
            update_best(a_p)

    # Tertiary: try different lengths
    if time.monotonic() < deadline - 60:
        for n_try in [350, 450, 500]:
            if time.monotonic() > deadline - 50:
                break
            jt = np.arange(n_try)
            a_t = np.sin(np.pi * (jt + 0.5) / n_try)
            a_tr = coord_desc(a_t, 100, deadline - 40)
            update_best(a_tr)

    if best_a is None:
        best_a = a0
        update_best(best_a)

    fs = float(evaluate_sequence([float(x) for x in best_a]))
    if not np.isfinite(fs):
        best_a = np.sin(np.pi * (np.arange(400) + 0.5) / 400)

    return [float(x) for x in best_a]
