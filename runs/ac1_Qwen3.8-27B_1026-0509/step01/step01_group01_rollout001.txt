import time
import scipy.sparse as sp

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    deadline = time.monotonic() + max(0.5, float(budget_s) - 20.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    S0 = parent.sum()
    if S0 < 1e-10:
        parent = np.ones(n)
        S0 = n
    a0 = parent / S0

    fft_len = 1 << (2 * n - 1).bit_length()

    def fconv(x):
        X = np.fft.rfft(x, fft_len)
        return np.fft.irfft(X * X, fft_len)[:2 * n - 1]

    def cross_conv(f, g):
        F = np.fft.rfft(f, fft_len)
        G = np.fft.rfft(g, fft_len)
        return np.fft.irfft(F * G, fft_len)[:2 * n - 1]

    def score(a):
        s = a.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * n * fconv(a).max() / (s * s)

    best_a = a0.copy()
    best_score = score(best_a)

    def solve_lp(f, M, lags):
        K = len(lags)
        rows = []
        cols = []
        data = []
        for idx, k in enumerate(lags):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi <= lo:
                continue
            vals = f[k - hi + 1:k - lo + 1][::-1]
            mask = vals > 1e-18
            if not np.any(mask):
                continue
            jj = np.arange(lo, hi)[mask]
            rows.extend([idx] * len(jj))
            cols.extend(jj.tolist())
            data.extend(vals[mask].tolist())
        if not rows:
            return None
        A = sp.csr_matrix((data, (rows, cols)), shape=(K, n))
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=[(0, None)] * n,
                          method='highs', options={'time_limit': 30})
        except Exception:
            return None
        if res.status != 0 or res.x is None:
            return None
        return res.x

    def lp_step(a, K=200):
        conv = fconv(a)
        M = conv.max()
        if M < 1e-30:
            return a
        act = np.argpartition(conv, -K)[-K:]
        act = act[np.argsort(conv[act])[::-1]]
        b = solve_lp(a, M, act)
        if b is None or b.sum() < 1e-10:
            return a
        g = b / b.sum()
        fg = cross_conv(a, g)
        viol = np.where(fg > M * 1.0005 + 1e-18)[0]
        if len(viol) > 0 and len(viol) + K < 2 * n:
            all_l = np.unique(np.concatenate([act, viol]))
            b2 = solve_lp(a, M, all_l)
            if b2 is not None and b2.sum() > 1e-10:
                g = b2 / b2.sum()
        best_t = 0.05
        best_s = score(a)

            if s < best_s:

                best_t = t
