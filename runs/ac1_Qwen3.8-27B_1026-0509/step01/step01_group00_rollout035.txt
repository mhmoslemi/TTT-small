        if mc_new < mc:
            a = a_new
            lr = min(lr * 1.01, 3.0)
        else:
            lr *= 0.9
            if lr < 0.005:
                a = best_a.copy()
                lr = 0.5

    result = [float(x) for x in best_a]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 64
    return result
