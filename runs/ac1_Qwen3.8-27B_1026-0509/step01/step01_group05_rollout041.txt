import time

    fft_size = 1

        ac = np.real(ifft(A * A, fft_size))[:2 * n - 1]

        np.maximum(ac, 0, out=ac)

        if S < 0.01:

        return 2.0 * n * ac.max() / (S * S)

        exp_t = np.exp(t * (ac - max_ac))

        v_hat = v / (1 - beta2 ** (it + 1))

        ac = np.real(ifft(A * A, fft_size))[:2 * n - 1]
        np.maximum(ac, 0, out=ac)
        S = a.sum()
        if S < 0.01:
            return np.inf
        return 2.0 * n * ac.max() / (S * S)
    
    best = a_parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Quick parametric exploration
    t_grid = np.linspace(0, 1, n)
    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        for alpha in [1, 5, 20, 50, 100, 500, 1000]:
            for flip in [0, 1]:
                tt = 1 - t_grid if flip else t_grid
                cand = 1 + alpha * tt**p
                candidates.append(cand)
        # Gaussian-like
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(np.exp(-0.5*((t_grid-0.5)/sigma)**2))
            candidates.append(np.exp(-0.5*((t_grid-0.3)/sigma)**2))
            candidates.append(np.exp(-0.5*((t_grid-0.7)/sigma)**2))
    
    for cand in candidates:
        if time.monotonic() > deadline - 60:
            break
        cand = np.clip(cand, 0, 1000)
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    
    # Main optimization: LSE + Adam
    a = best.copy().astype(np.float64)
    a = a / a.sum() * n  # normalize to mean 1
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.3
    t_seq = 5.0  # LSE temperature
    no_improve = 0
    
    it = 0
    while time.monotonic() < deadline - 60.0:
        it += 1
        
        A = fft(a, fft_size)
        ac = np.real(ifft(A * A, fft_size))[:2 * n - 1]
        np.maximum(ac, 0, out=ac)
        max_ac = ac.max()
        
        # LSE weights
        shifted = t_seq * (ac - max_ac)
        shifted = np.clip(shifted, -100, 0)
        exp_t = np.exp(shifted)
        w = exp_t / exp_t.sum()
        
        # Gradient: 2 * conv(a_rev, w)[n-1:2n-1]
        Wf = fft(w, fft_size)
        Ar = fft(a[::-1], fft_size)
        conv_val = np.real(ifft(Ar * Wf))
        grad = 2.0 * conv_val[n-1:2*n-1]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S > 0.01:
            a = a_new / S * n
        
        # Track best
        surrogate = 2.0 * max_ac / n  # = 2n * max_ac / n^2 when sum=n
        if surrogate < best_score - 1e-12:
            best_score = surrogate
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Annealing
        if it % 1000 == 0:
            t_seq = min(t_seq * 1.2, 500.0)
            if no_improve > 3000:
                lr = max(lr * 0.7, 0.01)
        
        # Restart if stuck
        if no_improve > 10000:
            a = best + rng.normal(0, 0.02 * best.mean(), n)
            a = np.maximum(a, 0.0)
            a = a / a.sum() * n
            m[:] = 0
            v[:] = 0
            lr = 0.1
            no_improve = 0
    
    # LP refinement
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a = best.copy()
        for _ in range(5):
            if time.monotonic() > deadline - 15:
                break
            try:
                from scipy.optimize import linprog
                A_arr = fft(a, fft_size)
                ac = np.real(ifft(A_arr*A_arr, fft_size))[:2*n-1]
                np.maximum(ac, 0, out=ac)
                M = ac.max()
                
                # Build constraint matrix
                A_mat = np.zeros((2*n-1, n))
                for k in range(2*n-1):
                    js = slice(max(0, k-n+1), min(n-1, k)+1)
                    A_mat[k, js] = a[k - np.arange(js.start, js.stop)]
                
                res = linprog(-np.ones(n), A_ub=A_mat, 
                             b_ub=np.full(2*n-1, M),
                             bounds=[(0, None)]*n, method='highs')
                if res.success and res.x is not None:
                    g = res.x
                    Sg = g.sum()
                    if Sg > 0.01:
                        g = g / Sg * n
                        for bt in [0.1, 0.05, 0.02, 0.01]:
                            a_try = (1-bt)*a + bt*g
                            a_try = np.maximum(a_try, 0)
                            sc = fast_score(a_try)
                            if sc < best_score:
                                best_score = sc
                                best = a_try.copy()
                                a = a_try.copy()
                                break
            except:
                break
    
    # Final validation
    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs):
        return [float(x) for x in a_parent]
    return [float(x) for x
