from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    n = parent.size

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2

    def compute_conv(a):
        A = fft(a, fft_size)
        return np.real(ifft(A * A, fft_size))[:2*n-1]

    def max_corr(a):
        return float(compute_conv(a).max())

    f = parent / parent.sum()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Precompute LP constraint structure (fixed sparsity pattern)
    row_parts, col_parts = [], []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        cnt = i_hi - i_lo + 1
        row_parts.append(np.full(cnt, k, dtype=np.int32))
        col_parts.append(np.arange(i_lo, i_hi + 1, dtype=np.int32))
    row_arr = np.concatenate(row_parts)
    col_arr = np.concatenate(col_parts)
    idx_arr = row_arr - col_arr

    def make_A(fv):
        return csr_matrix((fv[idx_arr], (row_arr, col_arr)), shape=(2*n-1, n))
