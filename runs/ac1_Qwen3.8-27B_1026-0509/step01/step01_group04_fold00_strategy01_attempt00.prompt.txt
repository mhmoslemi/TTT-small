```python
def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

```

```python
def _build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix where A[t,k] = f[t-k] if valid else 0."""
    t_idx = np.arange(2 * n - 1)[:, None]
    k_idx = np.arange(n)[None, :]
    d = t_idx - k_idx
    valid = (d >= 0) & (d < n)
    return np.where(valid, f[np.clip(d, 0, n - 1)], 0.0)


def _lp_refine(f, n, max_iters, deadline, rng):
    """Iterative LP-based autocorrelation minimization."""
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    tau = 0.35

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = _build_conv_matrix(best, n)
        M = np.convolve(best, best).max()

        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])

        g_val = None
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    g_val = np.asarray(g.value).flatten()
                    break
            except Exception:
                continue

        if g_val is None or not np.isfinite(g_val).all() or g_val.sum() < 1e-10:
            break

        g_norm = g_val * (best.sum() / g_val.sum())
        new_f = (1.0 - tau) * best + tau * g_norm
        new_f = np.clip(new_f, 0.0, None)

        sc = float(evaluate_sequence(new_f.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = new_f
            best_score = sc
        else:
            tau *= 0.7
            if tau < 0.02:
                break
        tau = max(tau * 0.97, 0.05)

    return best, best_score
```
