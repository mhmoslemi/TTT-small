import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = np.ones(100, dtype=np.float64)
    
    def try_score(a):
        nonlocal best_score, best_seq
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()
        return s
    
    def fast_conv(a):
        return np.convolve(a, a)
    
    def score_of(a):
        n = len(a)
        s = a.sum()
        if s < 1e-15:
            return np.inf
        c = fast_conv(a)
        return 2.0 * n * c.max() / (s * s)
    
    def build_A(a):
        n = len(a)
        m = 2 * n - 1
        A = np.zeros((m, n))
        for j in range(n):
            e = min(j + n, m)
            A[j:e, j] = a[:e-j]
        return A
    
    def lp_step(a):
        n = len(a)
        m = 2 * n - 1
        ca = fast_conv(a)
        M = ca.max()
        if M < 1e-15:
            return None
        A = build_A(a)
        tl = max(0.5, min(10.0, deadline - time.monotonic() - 15))
        if tl <= 0.1:
            return None

            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(m, M),
                          bounds=[(0, None)]*n, method='highs',

            return None

            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-10:

        best_a = a

            at = at / st
            sc = score_of(at)
            if sc < best_sc:

                best_a = at
