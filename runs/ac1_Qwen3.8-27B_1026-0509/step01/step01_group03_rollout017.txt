    b = np.convolve(a, a)
    M = b.max()
    k_star = np.argmax(b)
    # Subgradient
    g = np.zeros_like(a)
    # g[j] = 2 * a[k_star - j] if 0 <= k_star - j < n
    js = np.arange(len(a))
    shifted = k_star - js
    valid = (shifted >= 0) & (shifted < len(a))
    g[valid] = 2 * a[shifted[valid]]
    # Step
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s
    return a_new
