import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 25.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    s = parent.sum()
    a0 = parent / s if s > 1e-15 else np.ones(n) / n

    fft_len = 1 << (2 * n - 1).bit_length()

    def autocorr(a):
        A = np.fft.rfft(a, fft_len)
        return np.fft.irfft(A * A, fft_len)[:2 * n - 1]

    def score(a):
        return 2.0 * n * autocorr(a).max()

    best_a = a0.copy()
    best_score = score(best_a)

    def run_opt(a, n_iter, beta0, beta1, lr0, lr1, mom):
        nonlocal best_a, best_score
        a = a.copy()
        v = np.zeros_like(a)
        for it in range(n_iter):
            if time.monotonic() > deadline:
                break
            frac = it / max(n_iter, 1)
            beta = beta0 * (beta1 / max(beta0, 1e-10)) ** frac
            lr = lr0 * (lr1 / max(lr0, 1e-10)) ** frac

            A = np.fft.rfft(a, fft_len)
            c = np.fft.irfft(A * A, fft_len)[:2 * n - 1]
            cmax = c.max()
            w = np.exp(beta * (c - cmax))
            wt = w.sum()
            if wt < 1e-30:
                continue
            w /= wt

            W = np.fft.rfft(w, fft_len)
            grad = 2.0 * np.fft.irfft(W * np.conj(A), fft_len)[:n]
            grad -= grad.mean()

            gn = np.abs(grad).max()
            if gn > 1e-30:
                v = mom * v + grad / gn
                vn = np.abs(v).max()
                if vn > 1e-30:
                    a = a - lr * v / vn
            else:
                v *= 0.9

            np.maximum(a, 0.0, out=a)
            sn = a.sum()
            if sn < 1e-15:
                a[:] = a0
            else:
                a /= sn

            if it % 500 == 0 or it == n_iter - 1:
                s2 = score(a)
                if s2 < best_score:
                    best_a = a.copy()
                    best_score = s2
        return a

    # Phase 1: Main optimization
    a = run_opt(a0, 120000, 3.0, 60.0, 0.12, 0.008, 0.9)

    # Phase 2: Restarts with perturbation
    n_restarts = 40
    for r in range(n_restarts):
        if time.monotonic() > deadline - 120:
            break
        amp = 0.12 + 0.35 * (r / n_restarts)
        a = best_a * np.exp(rng.normal(0.0, amp, n))
        a = np.maximum(a, 0.0)
        a /= max(a.sum(), 1e-15)
        a = run_opt(a, 8000, 15.0, 200.0, 0.06, 0.003, 0.92)

    if time.monotonic() < deadline - 60:
        a = best_a.copy()
        run_opt(a, 80000, 150.0, 1500.0, 0.004, 0.0003, 0.96)

    # Phase 4: Final polish

        run_opt(a, 40000, 500.0, 5000.0, 0.002, 0.0001, 0.97)

        return [float(x) for x in parent]
    return [float(x) for x in best_a]
