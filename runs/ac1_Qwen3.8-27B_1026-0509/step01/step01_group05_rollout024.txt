            no_improve = 0

            no_improve += 1

        k_top = min(3 + no_improve // 300, 20)
        grad = subgrad(b, a, k_top)

        v = gamma * v - lr * grad
        a_new = a + v
        a_new = np.clip(a_new, 1e-15, None)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        else:
            a_new = best.copy()
        a = a_new

        if no_improve > 3000:
            lr *= 0.75
            no_improve = 0
        elif no_improve > 12000:
            a = best.copy() + rng.normal(0, 0.08 / np.sqrt(n), n)
            a = np.clip(a, 1e-15, None)
            a /= a.sum()
            v = np.zeros(n)
            lr = 0.06
            no_improve = 0
        elif no_improve == 0 and lr < 0.4:
            lr *= 1.01

    # Phase 2: Fine perturbation refinement
    a = best.copy()
    scale = 0.005
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        a_trial = a + rng.normal(0, scale / np.sqrt(n), n)
        a_trial = np.clip(a_trial, 1e-15, None)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s
        ns = norm_score(a_trial)
        if ns < best_ns:
            best_ns = ns
            best = a_trial.copy()
            a = a_trial.copy()
            scale = max(scale * 1.01, 1e-6)
        else:
            scale = max(scale * 0.999, 1e-6)

    # Final: try blending best with uniform
    for w in np.linspace(0, 0.3, 31):
        if time.monotonic() > deadline:
            break
        cand = (1 - w) * best + w / n
        cand = np.clip(cand, 1e-15, None)
        cand /= cand.sum()
        ns = norm_score(cand)
        if ns < best_ns:
            best_ns, best = ns, cand.copy()

    final_eval = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_eval):
        fallback = np.clip(parent, 0, 1000)
        return [float(x) for x in fallback]

    return [float(x) for x in best]
