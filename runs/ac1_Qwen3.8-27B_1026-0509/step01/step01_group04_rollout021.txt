import time
import cvxpy as cp

def _conv_mat(f, n):
    """Build (2n-1) x n matrix for convolution with f."""
    t = np.arange(2*n-1)[:, None]
    k = np.arange(n)[None, :]
    d = t - k
    valid = (d >= 0) & (d < n)
    idx = np.clip(d, 0, n-1)
    A = np.zeros((2*n-1, n))
    A[valid] = f[idx[valid]]
    return A

def _lp_step(f, n, M, deadline):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    A = _conv_mat(f, n)
    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [A @ b <= M])
    b_val = None
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate") and b.value is not None:
                b_val = np.asarray(b.value).flatten()
                if b_val.sum() > 1e-10:

    return b_val
