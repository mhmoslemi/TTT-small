from scipy.optimize import linprog, minimize
from scipy.special import logsumexp
import time

def _conv_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    c = np.convolve(a, a)
    # Smooth max
    sm = tau * logsumexp(c / tau)
    # Weights
    w = np.exp(c / tau - logsumexp(c / tau))
    # Gradient of smooth max w.r.t. a
    # d(sm)/da_j = sum_k w_k * 2 * a[k-j] (if 0<=k-j<n)
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            grad[j] += 2 * wk * a[k - j]
    return sm, grad

from scipy.special import logsumexp
import time

def _build_conv_matrix(a, n):
    L = 2 * n - 1
    A = np.zeros((L, n))
    for k in range(L):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = a[k - j]
    return A

def _lp_step(a, M):
    n = len(a)
    A = _build_conv_matrix(a, n)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True})

    sm = tau * lm
