def convolve_fft(a):
    n = a.shape[0]
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)

def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}"""

    cssv = np.cumsum(u)
