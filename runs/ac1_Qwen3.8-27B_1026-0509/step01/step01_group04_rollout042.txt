                bseg = bfull[klo : khi + 1].copy()

                def _obj(t, ai=ai, aki=aki, bseg=bseg, um=um,
                         two_i=two_i, is_2i=is_2i, klo=klo, S=S, n=n):
                    tS = S - ai + t
                    if tS < 1e-10:
                        return 1e10
                    bch = bseg + 2.0 * (t - ai) * aki
                    if is_2i:
                        bch[two_i - klo] = bfull[two_i] + t * t - ai * ai
                    mx = bch.max()
                    if um > mx:
                        mx = um
                    return 2.0 * n * mx / (tS * tS)

                cur_val = _obj(ai)
                hi_bound = max(10.0, 5.0 * S / n)
                hi_bound = min(hi_bound, 1000.0)
                res = minimize_scalar(_obj, bounds=(0.0, hi_bound), method='bounded',
                                      options={'xatol': xtol, 'maxiter': 30})
                if res.success and res.fun < cur_val - 1e-14:

                    delta = t_new - ai
                    for k in range(klo, khi + 1):
                        if k == two_i:
                            bfull[k] = t_new * t_new + (bfull[k] - ai * ai - 2 * ai * a[k - i] + 2 * ai * a[k - i])
                            bfull[k] = (bfull[k] - 2 * ai * a[k - i]) + t_new * t_new

    def _cd2(a, n, deadline, rng, max_sweeps, xtol=1e-7):
        """Cleaner CD with full recompute of bfull each sweep."""
        a = a.astype(np.float64).copy()
        for sweep in range(max_sweeps):
            if time.monotonic() > deadline:
                break
            bfull = np.convolve(a, a)
            S = a.sum()
            if S < 1e-10:
                break
            order = rng.permutation(n)
            nimp = 0
            for i in order:
                ai = a[i]
                klo = i
                khi = min(2 * n - 2, i + n - 1)
                um = 0.0
                if klo > 0:
                    um = bfull[:klo].max()
                if khi + 1 < 2 * n - 1:
                    um = max(um, bfull[khi + 1:].max())
                aki = a[klo - i : khi - i + 1].copy()
                two_i = 2 * i
                is_2i = (klo <= two_i <= khi)

                def _obj(t, ai=ai, aki=aki, um=um, two_i=two_i,
                         is_2i=is_2i, klo=klo, bfull=bfull, S=S, n=n):
                    tS = S - ai + t
                    if tS < 1e-10:
                        return 1e10
                    bch = bfull[klo:khi + 1] + 2.0 * (t - ai) * aki
                    if is_2i:
                        bch = bch.copy()
                        bch[two_i - klo] = bfull[two_i] + t * t - ai * ai
                    mx = bch.max()
                    if um > mx:
                        mx = um
                    return 2.0 * n * mx / (tS * tS)

                cur_val = _obj(ai)
                hi = min(1000.0, max(10.0, 4.0 * S / n))
                res = minimize_scalar(_obj, bounds=(0.0, hi), method='bounded',
                                      options={'xatol': xtol, 'maxiter': 25})
                if res.success and res.fun < cur_val - 1e-13:
                    a[i] = float(res.x)
                    nimp += 1
            if nimp == 0:
                break
        return a

    def _lp_refine(a, n, deadline, rng, n_steps=10):
        """LP-based refinement."""
        a = a.copy()
        sc = float(evaluate_sequence(a.tolist()))
        tau = 0.3
        for step in range(n_steps):
            if time.monotonic() > deadline:
                break
            b = np.convolve(a, a)
            M = b.max()
            # Build constraint matrix: (a * g)[t] = sum_k a[t-k] g_k
            A = np.zeros((2 * n - 1, n))
            for t in range(2 * n - 1):
                lo = max(0, t - n + 1)
                hi = min(n - 1, t)
                A[t, lo:hi + 1] = a[t - hi : t - lo + 1][::-1]
            g = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M, cp.sum(g) >= 0.01])
            g_val = None
            for slv in [cp.CLARABEL, cp.ECOS, cp.SCS]:
                try:
                    prob.solve(solver=slv, verbose=False, max_time=15)
                    if prob.status in ("optimal", "optimal_inaccurate") and g.value is not None:
                        gv = np.asarray(g.value).flatten()
                        if np.isfinite(gv).all() and gv.sum() > 1e-6:
                            g_val = gv
                            break
                except Exception:
                    continue
            if g_val is None:
                break
            g_val = g_val * (a.sum() / g_val.sum())
            new_a = (1 - tau) * a + tau * g_val
            new_a = np.clip(new_a, 0.0, None)
            new_sc = float(evaluate_sequence(new_a.tolist()))
            if np.isfinite(new_sc) and new_sc < sc:
                a = new_a
                sc = new_sc
                tau = min(tau * 1.1, 0.5)
            else:
                tau *= 0.7
                if tau < 0.05:
                    break
        return a, sc

    # Phase 1: Coordinate descent from initial
    a1 = _cd2(a0, n, deadline - 120, rng, max_sweeps=300)
    sc1 = float(evaluate_sequence(a1.tolist()))
    if sc1 < best_score:
        best, best_score = a1.copy(), sc1

    # Phase 2: LP refinement on best
    if time.monotonic() < deadline - 100:
        a2, sc2 = _lp_refine(best, n, deadline - 60, rng, n_steps=15)
        if sc2 < best_score:
            best, best_score = a2.copy(), sc2

    # Phase 3: More CD on best
    if time.monotonic() < deadline - 50:
        a3 = _cd2(best, n, deadline - 40, rng, max_sweeps=200, xtol=1e-8)
        sc3 = float(evaluate_sequence(a3.tolist()))
        if sc3 < best_score:
            best, best_score = a3.copy(), sc3

    # Phase 4: Try alternative initializations
    if time.monotonic() < deadline - 80:
        j = np.arange(n)
        alts = [
            np.sin(np.pi * (j + 0.5) / n) ** 2,
            0.5 + 0.5 * np.cos(2 * np.pi * j / n),
        ]
        for alt in alts:
            if time.monotonic() > deadline - 50:
                break
            a4 = _cd2(alt, n, deadline - 50, rng, max_sweeps=150)
            sc4 = float(evaluate_sequence(a4.tolist()))
            if sc4 < best_score:
                best, best_score = a4.copy(), sc4

    # Phase 5: Final polish
    if time.monotonic() < deadline - 30:
        a5 = _cd2(best, n, deadline - 20, rng, max_sweeps=100, xtol=1e-9)
        sc5 = float(evaluate_sequence(a5.tolist()))
        if sc5 < best_score:
            best, best_score = a5.copy(), sc5

    # Final validation
    fs = float(evaluate_sequence([float(x) for x in best]))
    if not np.isfinite(fs):
        j = np.arange(n)
        best = np.sin(np.pi * (j + 0.5) / n)

    return [float(x) for x in best]
