        a = a * np.exp(np.clip(-lr * g, -10, 10))
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum < 1e-20:
            a = best_a.copy()
            c = do_conv(a)
            continue
        a /= a_sum

        c = do_conv(a)
        sc = 2.0 * n * c.max()

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            last_improve = it

        tau = max(tau * 0.99995, 0.005)
        it += 1

        if it % 1000 == 0:
            if sc > best_score * 1.02:
                lr *= 0.8
            else:
                lr = min(lr * 1.05, 300.0)

        if it - last_improve > 50000 and it % 10000 == 0:
            s = 0.03 + 0.03 * min(1.0, (it - last_improve) / 100000.0)
            a = best_a * np.exp(rng.normal(0, s, n))
            a = np.maximum(a, 1e-30)
            a /= a.sum()
            c = do_conv(a)
            tau = max(tau, 0.05)
            lr = 50.0

    result = [float(x) for x in best_a]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * 64
    return result
