        parent = np.ones(64)
    if parent.ndim != 1 or parent.size < 4:
        parent = np.ones(64)
    n = parent.size
    s = parent.sum()
    if s < 0.01:
        return [1.0] * 64
    a0 = parent / s

    fft1 = 1
    while fft1 < 2 * n - 1:
        fft1 <<= 1
    fft2 = 1
    while fft2 < 3 * n - 2:
        fft2 <<= 1

    def conv_self(a):
        fa = np.fft.rfft(a, fft1)
        return np.fft.irfft(fa * fa, fft1)[:2 * n - 1]

    def sm_grad(c, a, tau):
        cm = c.max()
        ex = np.exp(tau * (c - cm))
        w = ex / ex.sum()
        wr = w[::-1]
        conv = np.fft.irfft(
            np.fft.rfft(a, fft2) * np.fft.rfft(wr, fft2), fft2
        )[: 3 * n - 2]
        return 2.0 * conv[n - 1 : 2 * n - 1][::-1]

    def optimize(a_init, t_end, tau, eta, tau_max, ci=50):
        a = a_init.copy()
        best = a.copy()
        bsc = float(conv_self(a).max())
        it = 0
        while time.monotonic() < t_end:
            c = conv_self(a)
            sc = c.max()
            if sc < bsc:
                bsc, best = float(sc), a.copy()
            g = sm_grad(c, a, tau)
            a = a * np.exp(np.clip(-eta * g, -500, 500))
            a = np.maximum(a, 0)
            sz = a.sum()
            if sz < 1e-30:
                a, eta = best.copy(), max(eta * 0.5, 0.005)
                continue
            a /= sz
            it += 1
            if it % (ci * 5) == 0:
                s2 = float(conv_self(a).max())
                if s2 > bsc * 1.03:
                    eta = max(eta * 0.7, 0.005)
                elif s2 < bsc * 0.99:
                    eta = min(eta * 1.08, 3.0)
            if it % (ci * 25) == 0:
                tau = min(tau * 1.35, tau_max)
            if it % (ci * 100) == 0:
                a = np.maximum(a * np.exp(rng.normal(0, 0.04, n)), 0)
                if a.sum() > 1e-30:
                    a /= a.sum()
        sc = float(conv_self(a).max())
        if sc < bsc:
            bsc, best = sc, a.copy()
        return best, bsc

    cands = [a0.copy()]
    idx = np.arange(n, dtype=np.float64)
    for r in [0.99, 1.01, 1.02]:
        z = r ** idx
        z /= z.sum()
        cands.append(z)
    z = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * idx / n)
    z = np.maximum(z, 0.01)
    z /= z.sum()
    cands.append(z.copy())
    z = a0 * np.exp(rng.normal(0, 0.05, n))
    z = np.maximum(z, 0)
    z /= z.sum()
    cands.append(z.copy())

    ba = cands[0].copy()
    bs = float(conv_self(ba).max())
    for c in cands[1:]:
        sc = float(conv_self(c).max())
        if sc < bs:
            ba, bs = c.copy(), sc

    span = deadline - t0
    t1 = t0 + 0.4 * span
    a, sc = optimize(ba, t1, 10, 0.5, 100)
    if sc < bs:
        ba, bs = a, sc

    t2 = t0 + 0.75 * span
    a, sc = optimize(ba, t2, 50, 0.3, 500)
    if sc < bs:
        ba, bs = a, sc

    a, sc = optimize(ba, deadline, 200, 0.1, 2000)
    if sc < bs:
        ba, bs = a, sc

    ba = np.clip(ba, 0.0, 1000.0)
    res = [float(x) for x in ba]
    if not np.isfinite(evaluate_sequence(res)):
        res = [1.0] * 64
    return res
