```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_idx = pid // H
    h_idx = pid % H

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer bases for this (batch, h)
    left_base = left_ptr + batch_idx * stride_l_batch + h_idx * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_batch + h_idx * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_batch + h_idx * stride_o_h

    # initial pointers for tile
    a_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < N

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # store result
    c_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)
```

"""
TriMul (outgoing) forward pass with a custom Triton kernel.
The heavy O(N³) reduction out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication L @ Rᵀ for each hidden
dimension d.  The kernel works on FP16 data (Tensor‑Core friendly) and
accumulates in FP32.  All other operations (LayerNorm, Linear, gating)
are performed in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N, H,                                 # batch, seq_len, hidden_dim
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    # 1️⃣ identify which (batch, hidden) we are computing
    pid = tl.program_id(0)                # 0‑th axis = batch * hidden
    batch_idx = pid // H
    h_idx = pid % H

    # 2️⃣ tile indices for the output matrix (M×N)
    pid_m = tl.program_id(1)              # row‑tile id
    pid_n = tl.program_id(2)              # col‑tile id

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # 3️⃣ base pointers for the current (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_l_batch + h_idx * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_batch + h_idx * stride_r_h
    out_base   = out_ptr   + batch_idx * stride_o_batch + h_idx * stride_o_h

    # 4️⃣ pointers to the current K‑tile
    a_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # 5️⃣ iterate over K dimension (blocked)
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < N

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                            # FP32 accumulation

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # 6️⃣ write output tile (cast back to FP16)
    c_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weight_dict  : dict of model parameters (torch.Tensors)
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1️⃣ LayerNorm on the input (dim -> dim)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------ #
    # 2️⃣ Linear projections and optional masking
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask_unsq = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1], broadcast over hidden_dim
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------ #
    # 3️⃣ Gating (sigmoid of Linear) – same dim -> hidden_dim
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4️⃣ Cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------ #
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 5️⃣ Triton batched GEMM: out = left @ rightᵀ  (per hidden dimension)
    # ------------------------------------------------------------------ #
    B, N, _, H = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16)  # [B,N,N,H] FP16

    # strides of the tensors (in elements, not bytes)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_batch, stride_o_i, stride_o_k, stride_o_h = out_fp16.stride()

    # parameters for the kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    NUM_WARPS = 8

    total_batch = B * H
    grid = (
        total_batch,                     # batch * hidden
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_batch,
        stride_r_i,          # j‑stride == i‑stride (N*H)
        stride_r_k,
        stride_r_h,
        stride_o_batch,
        stride_o_i,
        stride_o_k,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=NUM_WARPS,
    )

    # ------------------------------------------------------------------ #
    # 6️⃣ Convert back to float32
    # ------------------------------------------------------------------ #
    out = out_fp16.to(torch.float32)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------ #
    # 7️⃣ Output LayerNorm, gating and final Linear
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,hidden_dim]  (float32)

    out = out * out_gate               # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
