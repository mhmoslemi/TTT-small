@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_row_stride, a_col_stride, a_d_stride,
    b_batch_stride, b_row_stride, b_col_stride, b_d_stride,
    c_batch_stride, c_row_stride, c_col_stride, c_d_stride,
    B, N, hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # combined batch*hidden_dim index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch = pid0 // hidden_dim
    d = pid0 % hidden_dim

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute pointer base for batch+d
    a_base = a_ptr + batch * a_batch_stride + d * a_d_stride
    b_base = b_ptr + batch * b_batch_stride + d * b_d_stride
    c_base = c_ptr + batch * c_batch_stride + d * c_d_stride

    # Compute pointers for tile
    a_ptrs = a_base + offs_m[:, None] * a_row_stride + offs_k[None, :] * a_col_stride
    # B is transposed: rows=k, cols=j
    b_ptrs = b_base + offs_k[:, None] * b_row_stride + offs_n[None, :] * b_col_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over k blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < N

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

        # advance pointers by BLOCK_K stride
        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # write back to c
    c_ptrs = c_base + offs_m[:, None] * c_row_stride + offs_n[None, :] * c_col_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

"""
TriMul (outgoing) forward pass with a custom Triton kernel for the heavy
batched matrix multiplication:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

All cheap operations (LayerNorms, Linear projections, gating, final
Linear) are performed in PyTorch.  The core O(N³·C) reduction is executed
as a batched half‑precision GEMM using a Triton kernel that works directly
on the original memory layout (no permute / copy of the channel dimension).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
#   left  : (B, N, N, D)   (float16)
#   right : (B, N, N, D)   (float16) – accessed transposed on the last two
#   out   : (B, N, N, D)   (float16)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    a_batch_stride, a_row_stride, a_col_stride, a_d_stride,
    b_batch_stride, b_row_stride, b_col_stride, b_d_stride,
    c_batch_stride, c_row_stride, c_col_stride, c_d_stride,
    B, N, D,                                 # dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)      # combined batch*channel index
    pid_m = tl.program_id(1)     # block row (i)
    pid_n = tl.program_id(2)     # block col (j)

    # recover batch and channel indices
    batch = pid0 // D
    d = pid0 % D

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, channel)
    a_base = a_ptr + batch * a_batch_stride + d * a_d_stride
    b_base = b_ptr + batch * b_batch_stride + d * b_d_stride
    c_base = c_ptr + batch * c_batch_stride + d * c_d_stride

    # tile pointers
    a_ptrs = a_base + offs_m[:, None] * a_row_stride + offs_k[None, :] * a_col_stride
    # B is accessed transposed: rows=k, cols=j
    b_ptrs = b_base + offs_k[:, None] * b_row_stride + offs_n[None, :] * b_col_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < N

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                           # (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # write result
    c_ptrs = c_base + offs_m[:, None] * c_row_stride + offs_n[None, :] * c_col_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    right = right * right_gate

        left_h,
        right_h,

        a_batch_stride,
        a_row_stride,
        a_col_stride,
        a_d_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        b_d_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        c_d_stride,
        B,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 7) Convert back to float32 and apply the output LayerNorm + gating + linear
    # --------------------------------------------------------------------- #
    out = out_h.to(torch.float32)

        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])     # (B, N, N, dim)

    return out
```
