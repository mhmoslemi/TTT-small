        batchcnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),

        left_flat,
        right_flat,

        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, hidden_dim, N, N] then permute to [B,N,N,hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # 6. Output layernorm, out gate, final linear
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

    stride_cm, stride_cn,

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
Triton implementation of the outgoing TriMul operation (AlphaFold3).

The heavy O(N³·H) computation
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed with a custom Triton kernel that treats each (batch, hidden) pair
as an independent GEMM: left (N×N) @ rightᵀ (N×N).  The kernel works in
fp16 on Tensor‑cores while accumulating in fp32 for numerical stability.
All remaining Linear/LayerNorm/Gating steps are executed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M, K)
    stride_bk, stride_bn,          # Bᵀ: (K, N)  (original B is (N, K))
    stride_cm, stride_cn,          # C: (M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_offset in range(0, K, BLOCK_K):
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # fp16 dot -> fp32 accumulation
        acc += tl.dot(a, b)

    # write back
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (torch.Tensors)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional masking
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Outgoing TriMul via Triton GEMM (left @ rightᵀ per channel)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left: [B, N, N, hidden_dim]

    # Move hidden dimension to batch axis and make data contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Cast to fp16 for Tensor‑core acceleration
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    batchcnt = B * hidden_dim
    left_flat = left_fp16.view(batchcnt, N, N)   # [B*H, N, N]
    right_flat = right_fp16.view(batchcnt, N, N) # [B*H, N, N]

    out_flat = torch.empty((batchcnt, N, N), dtype=torch.float32, device=device)

    # Strides (all tensors are contiguous)
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # col stride (K)
    stride_bn = right_flat.stride(1)   # row stride (N)

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Blocking parameters – tuned for modern Tensor‑core GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batchcnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _triton_outgoing_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                       # M
        N,                       # N
        N,                       # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to original dim

    return out
