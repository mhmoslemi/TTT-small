"""
TriMul (outgoing) forward pass with a Triton kernel.

The heavy O(N³) part is the batched matrix multiplication:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

For each batch element and hidden channel this is a GEMM:
    out = left @ rightᵀ      with left/right ∈ ℝ^{N×N}
The kernel tiles the GEMM in three dimensions (M, N, K) using
tl.dot for fast FP32 accumulation.

All other operations (LayerNorms, Linear projections, sigmoid gates,
masking) are performed with PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M=N=K=seq_len)
    stride_am, stride_ak,              # A: (batch, M, K)
    stride_bk, stride_bn,              # Bᵀ: (batch, K, N)   (original B is (batch, N, K))
    stride_cm, stride_cn,              # C: (batch, M, N)
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with B transposed (A @ Bᵀ)."""
    pid_b = tl.program_id(0)   # batch*hidden_dim index
    pid_m = tl.program_id(1)   # block-row index
    pid_n = tl.program_id(2)   # block-col index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ tile (K x N) – note the swapped strides
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write back the result
    c = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"]
            - weights: dict of learned parameters (see function body for keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load model parameters
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating signals
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul: batched GEMM left @ rightᵀ
    # -----------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) and flatten batch*hidden into a single GEMM batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N_seq, _ = left_perm.shape
    batch_cnt = B * H

    left_flat = left_perm.view(batch_cnt, N_seq, N_seq)   # [batch_cnt, M=N, K=N]
    right_flat = right_perm.view(batch_cnt, N_seq, N_seq) # [batch_cnt, N, K]

    out_flat = torch.empty_like(left_flat)  # [batch_cnt, N, N]

    # Strides for the flattened tensors (contiguous layout)
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # original K stride (will be treated as K)
    stride_bn = right_flat.stride(1)   # original N stride (will be treated as N)

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)   # row stride (M)
    stride_cn = out_flat.stride(2)   # col stride (N)

    # Tiling parameters – these work well on modern NVIDIA GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_cnt,
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N_seq,               # M
        N_seq,               # N
        N_seq,               # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)     # back to original dim

    return out
