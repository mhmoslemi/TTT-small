"""
TriMul (outgoing) forward pass with a Triton kernel for the heavy
pairwise reduction. All cheap operations (LayerNorm, Linear, gating)
are performed with PyTorch; the O(N³) sum over the sequence dimension
is expressed as a batched matrix multiplication of shape
(B*hidden, N, N) × (B*hidden, N, N)ᵀ and computed in float16 with a
custom Triton kernel that tiles the matrices (BLOCK_M × BLOCK_N ×
BLOCK_K). The result is cast back to float32 and passed through the
output LayerNorm, gating, and final Linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : (batch, M, K)   (float16)
#   B : (batch, N, K)   (float16)  – we pass Bᵀ (shape batch, K, N)
#   C : (batch, M, N)   (float16)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    a_batch_stride, a_row_stride, a_col_stride,   # strides for A
    b_batch_stride, b_row_stride, b_col_stride,   # strides for Bᵀ
    c_batch_stride, c_row_stride, c_col_stride,   # strides for C
    M, N, K,                                 # matrix sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch id
    pid_m = tl.program_id(1)       # block row id
    pid_n = tl.program_id(2)       # block col id

    # ------------------------------------------------------------------- #
    # Offsets inside the current block
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the block for A and Bᵀ
    a_ptrs = a_ptr + pid_batch * a_batch_stride \
                     + offs_m[:, None] * a_row_stride \
                     + offs_k[None, :] * a_col_stride

    b_ptrs = b_ptr + pid_batch * b_batch_stride \
                     + offs_k[:, None] * b_row_stride \
                     + offs_n[None, :] * b_col_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                           # (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # Store result
    c_ptrs = c_ptr + pid_batch * c_batch_stride \
                     + offs_m[:, None] * c_row_stride \
                     + offs_n[None, :] * c_col_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul update.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm (dim -> dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32

    # ------------------------------------------------------------------- #
    # 2) Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------- #
    if mask is None:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask.to(dtype=x.dtype, device=device)
    mask_unsq = mask.unsqueeze(-1)                     # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid of Linear) – shape [B,N,N,hidden]
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Cast to float16 for the O(N³) reduction
    # ------------------------------------------------------------------- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N) -> view as (B*hidden, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_h = left_h.view(batch_dim, N, N)              # [B*H, N, N]
    right_h = right_h.view(batch_dim, N, N)

    # Bᵀ for the matmul: (batch, K, N) where K == N
    right_h_T = right_h.transpose(1, 2).contiguous()   # [B*H, N, N] (K, N)

    # Output buffer
    out_h = torch.empty_like(left_h)                    # float16

    # ------------------------------------------------------------------- #
    # 6) Triton batched matmul: out_h = left_h @ right_h_T
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    a_batch_stride, a_row_stride, a_col_stride = left_h.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_h_T.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_h.stride()

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_h,
        right_h_T,
        out_h,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 7) Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------- #
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------- #
    # 8) Output LayerNorm over hidden dimension, gating, final Linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # float32

    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
