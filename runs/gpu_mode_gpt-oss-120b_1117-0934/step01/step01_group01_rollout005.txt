"""
TriMul (outgoing) forward pass with a Triton kernel.

The heavy O(N³) operation
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
is implemented as a batched GEMM where each hidden channel (dim_out) is
treated as an independent batch element.  The kernel loads `left` and the
transpose of `right` in FP16, accumulates in FP32 using tensor‑core `dot`,
and writes the result in FP32.  All other operations (LayerNorm, Linear,
gating) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,               # A: (batch, M, K)
    stride_bk, stride_bn,               # Bᵀ: (batch, K, N)  (original B is (batch, N, K))
    stride_cm, stride_cn,               # C: (batch, M, N)
    stride_ab, stride_bb, stride_cb,    # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B ∈ FP16, C ∈ FP32)."""
    pid_b = tl.program_id(0)       # batch * hidden_dim index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Tile over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K) in FP16
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Load Bᵀ tile (K x N) in FP16
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Tensor‑core dot (FP16 inputs → FP32 output)
        acc += tl.dot(a, b)

    # Store result (FP32)
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model weights (all torch.float32 tensors)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load model parameters
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (applied after projection)
    # -----------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM via Triton
    #    Compute: out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
    # -----------------------------------------------------------------
    # Cast to FP16 for the GEMM (inputs), keep accumulator FP32
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # Move hidden dimension to batch axis and make tensors contiguous
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    batch_cnt = B * hidden_dim
    left_flat = left_perm.view(batch_cnt, N, N)   # [B*H, N, N]  FP16
    right_flat = right_perm.view(batch_cnt, N, N)  # [B*H, N, N]  FP16

    # Output buffer (FP32)
    out_flat = torch.empty((batch_cnt, N, N), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_ab = left_flat.stride(0)   # batch stride
    stride_am = left_flat.stride(1)   # M stride
    stride_ak = left_flat.stride(2)   # K stride

    stride_bb = right_flat.stride(0)  # batch stride
    stride_bk = right_flat.stride(2)  # K stride (after transpose)
    stride_bn = right_flat.stride(1)  # N stride (after transpose)

    stride_cb = out_flat.stride(0)    # batch stride
    stride_cm = out_flat.stride(1)    # M stride
    stride_cn = out_flat.stride(2)    # N stride

    # Tiling parameters – tuned for typical GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _gemm_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to original dim

    return out
