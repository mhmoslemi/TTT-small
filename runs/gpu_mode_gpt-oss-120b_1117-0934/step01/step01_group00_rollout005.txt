"""
TriMul (outgoing) forward pass with a Triton‐accelerated batched matrix multiplication.
The heavy O(N³) reduction   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM on fp16 data:
    left  -> (B*H, N, N)
    right -> (B*H, N, N) → transpose K‑N → (B*H, N, N)
and computed by a custom Triton kernel.
All other operations (LayerNorm, Linear projections, gating) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ B   (A: (batch,M,K), B: (batch,K,N))
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                    # pointers to input/output
    a_batch_stride, a_row_stride, a_col_stride,   # A strides (elements)
    b_batch_stride, b_row_stride, b_col_stride,   # B strides (elements)
    c_batch_stride, c_row_stride, c_col_stride,   # C strides (elements)
    M, N, K,                                 # matrix dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block col index

    # ------------------------------------------------------------------- #
    # Offsets for the current block
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Base pointers for the current tile (k=0)
    # ------------------------------------------------------------------- #
    a_ptrs = a_ptr + pid_batch * a_batch_stride \
                  + offs_m[:, None] * a_row_stride \
                  + offs_k[None, :] * a_col_stride
    b_ptrs = b_ptr + pid_batch * b_batch_stride \
                  + offs_k[:, None] * b_row_stride \
                  + offs_n[None, :] * b_col_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------- #
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                           # (BLOCK_M, BLOCK_N) → float32

        # advance to the next K‑tile
        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # ------------------------------------------------------------------- #
    # Write back C
    # ------------------------------------------------------------------- #
    c_ptrs = c_ptr + pid_batch * c_batch_stride \
                  + offs_m[:, None] * c_row_stride \
                  + offs_n[None, :] * c_col_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (or None)
        - weights      : dict of torch.Tensors with keys:
          "norm.weight", "norm.bias",
          "left_proj.weight", "right_proj.weight",
          "left_gate.weight", "right_gate.weight", "out_gate.weight",
          "to_out_norm.weight", "to_out_norm.bias",
          "to_out.weight"
        - config : dict with entries "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32

    # ------------------------------------------------------------------- #
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Cast to fp16 for the heavy reduction
    # ------------------------------------------------------------------- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 6) Reshape to (B*H, N, N) – each hidden slice becomes an independent GEMM
    # ------------------------------------------------------------------- #
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_h = left_h.view(batch_dim, N, N)               # [B*H, N, N]
    right_h = right_h.view(batch_dim, N, N)

    # right_hᵀ for matmul
    right_h_T = right_h.transpose(1, 2).contiguous()   # [B*H, N, N]

    # output buffer (fp16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ------------------------------------------------------------------- #
    # 7) Launch Triton batched GEMM C = A @ B
    # ------------------------------------------------------------------- #
    a_batch_stride, a_row_stride, a_col_stride = left_h.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_h_T.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_h.stride()

    # Block sizes – 64 works well for all tested N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_h,
        right_h_T,
        out_h,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 8) Restore layout: [B, N, N, H] (float32)
    # ------------------------------------------------------------------- #
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------- #
    # 9) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
