Thus final Triton kernel:

```
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offs within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k axis
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load tiles
        a = tl.load(
            a_ptr + pid_b * stride_ab
                + offs_m[:, None] * stride_am
                + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16
        )
        b = tl.load(
            b_ptr + pid_b * stride_bb
                + cur_k[:, None] * stride_bk
                + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16
        )
        # Dot product accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_cb \
                + offs_m[:, None] * stride_cm \
                + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _tri_mul_gemm_kernel(...):
    ...

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # cast weights
    norm_weight_h = weights["norm.weight"].to(torch.float16)
    norm_bias_h = weights["norm.bias"].to(torch.float16)
    left_proj_weight_h = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight_h = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight_h = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight_h = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight_h = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Input half conversion and layernorm
    x_half = F.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_weight_h,
        bias=norm_bias_h,
        eps=eps,
    )  # [B,N,N,dim] half

    # Projections
    left = F.linear(x_half, left_proj_weight_h)   # [B,N,N,hidden]
    right = F.linear(x_half, right_proj_weight_h)

    # Optional mask
    if mask is not None:
        # mask may be bool, convert to half
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # broadcast
        left = left * mask_h
        right = right * mask_h

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_half, left_gate_weight_h))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_weight_h))
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_weight_h))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul: reshape to (B*H,N,N)
    L = left.permute(0, 3, 1, 2).contiguous()
    R = right.permute(0, 3, 1, 2).contiguous()
    Bsize, H, N, _ = L.shape
    batch_cnt = Bsize * H

    L_flat = L.view(batch_cnt, N, N)
    R_flat = R.view(batch_cnt, N, N)

    # Allocate output buffer (half)
    out_half = torch.empty((batch_cnt, N, N), dtype=torch.float16, device=device)

    # Strides
    stride_am = L_flat.stride(1)
    stride_ak = L_flat.stride(2)
    stride_ab = L_flat.stride(0)

    stride_bk = R_flat.stride(2)  # swapped for transpose
    stride_bn = R_flat.stride(1)
    stride_bb = R_flat.stride(0)

    stride_cm = out_half.stride(1)
    stride_cn = out_half.stride(2)
    stride_cb = out_half.stride(0)

    # Tiling parameters (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_gemm_kernel[grid](
        L_flat,
        R_flat,
        out_half,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B,N,N,hidden]
    out = out_half.view(Bsize, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # Cast to float32 for following ops

    # Output normalization and projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_weight)

    return out

"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

The heavy O(N³) contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiply:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T .
All other operations (LayerNorm, linear projections, gating) are performed with
torch for simplicity.  The tensors are kept in float16 inside the kernel to
reduce memory traffic; the final output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # Bᵀ: (batch, K, N)  (original B is (batch, N, K))
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with K‑dimension reduction."""
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load B tile (K x N) – note the swapped strides for the transpose
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Accumulate in FP32 to preserve accuracy
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write result back in FP16
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask:        torch.Tensor of shape [B, N, N] or None
            weights:     dict of model parameters (float32 tensors)
            config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Cast model weights to half‑precision where appropriate
    # -----------------------------------------------------------------
    norm_w_h = weights["norm.weight"].to(torch.float16, non_blocking=True)
    norm_b_h = weights["norm.bias"].to(torch.float16, non_blocking=True)

    left_proj_w_h   = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    right_proj_w_h  = weights["right_proj.weight"].to(torch.float16, non_blocking=True)

    left_gate_w_h   = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_gate_w_h  = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_w_h    = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    to_out_norm_w = weights["to_out_norm.weight"]   # keep FP32 for final norm
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (half‑precision)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_w_h,
        bias=norm_b_h,
        eps=eps,
    )  # [B, N, N, dim]   fp16

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w_h)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x, right_proj_w_h)  # [B, N, N, hidden_dim]  fp16

    # -----------------------------------------------------------------
    # 3. Optional mask (applied after projection)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 4. Gating signals (sigmoid, half‑precision)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w_h))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w_h))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w_h))

    # Apply per‑channel gates
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ  (half‑precision kernel)
    # -----------------------------------------------------------------
    # Bring hidden dimension to the batch axis: [B, H, N, N]
    L = left.permute(0, 3, 1, 2).contiguous()
    R = right.permute(0, 3, 1, 2).contiguous()
    Bsz, H, N, _ = L.shape
    batch_cnt = Bsz * H

    # Flatten to (batch_cnt, N, N)
    L_flat = L.view(batch_cnt, N, N)
    R_flat = R.view(batch_cnt, N, N)

    # Output buffer (half‑precision)
    out_fp16 = torch.empty((batch_cnt, N, N), dtype=torch.float16, device=device)

    # Strides (all tensors are contiguous)
    stride_am = L_flat.stride(1)          # row stride (M)
    stride_ak = L_flat.stride(2)          # col stride (K)
    stride_ab = L_flat.stride(0)          # batch stride

    # For Bᵀ we swap the inner two strides
    stride_bk = R_flat.stride(2)          # original col stride
    stride_bn = R_flat.stride(1)          # original row stride
    stride_bb = R_flat.stride(0)          # batch stride

    stride_cm = out_fp16.stride(1)        # row stride (M)
    stride_cn = out_fp16.stride(2)        # col stride (N)
    stride_cb = out_fp16.stride(0)        # batch stride

    # Tile sizes – tuned for typical GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_gemm_kernel[grid](
        L_flat,
        R_flat,
        out_fp16,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # improve occupancy
    )

    # Reshape back to [B, N, N, hidden_dim] and cast to FP32 for the remaining ops
    out = out_fp16.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)  # back to original dim

    return out
