@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          a_batch_stride, a_row_stride, a_col_stride,
                          b_batch_stride, b_row_stride, b_col_stride,
                          c_batch_stride, c_row_stride, c_col_stride,
                          M, N, K,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    # propose program IDs: pid0 = batch id, pid1 = block row, pid2 = block col.
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointers for A and B
    a_ptrs = a_ptr + pid_batch * a_batch_stride + offs_m[:, None] * a_row_stride + offs_k[None, :] * a_col_stride
    b_ptrs = b_ptr + pid_batch * b_batch_stride + offs_k[:, None] * b_row_stride + offs_n[None, :] * b_col_stride

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute number of tiles in K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # compute offsets for K block
        cur_k = k * BLOCK_K
        # compute mask for valid K
        k_valid = (cur_k + offs_k) < K
        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))
        acc += tl.dot(a, b)
        # Move pointers to next K block
        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # store result
    c_ptrs = c_ptr + pid_batch * c_batch_stride + offs_m[:, None] * c_row_stride + offs_n[None, :] * c_col_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

"""
TriMul (outgoing) forward pass with a Triton kernel for the heavy
batch‑matrix‑multiplication (∑ₖ left[b,i,k,d] * right[b,j,k,d]).
All LayerNorms, Linear projections, and gating are performed in PyTorch.
The core O(N³) reduction is executed as a batched GEMM (A @ Bᵀ) using a
custom Triton kernel that works on half‑precision data and accumulates
in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (batch, M, K)   (float16)
#   B : (batch, N, K)   (float16)  → we pass Bᵀ (shape batch, K, N)
#   C : (batch, M, N)   (float16, later cast to float32)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    a_batch_stride, a_row_stride, a_col_stride,   # A strides
    b_batch_stride, b_row_stride, b_col_stride,   # Bᵀ strides
    c_batch_stride, c_row_stride, c_col_stride,   # C strides
    M, N, K,                                 # matrix dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch (combined B*hidden_dim)
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block column index

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current block
    a_ptrs = a_ptr + pid_batch * a_batch_stride \
                     + offs_m[:, None] * a_row_stride \
                     + offs_k[None, :] * a_col_stride
    b_ptrs = b_ptr + pid_batch * b_batch_stride \
                     + offs_k[:, None] * b_row_stride \
                     + offs_n[None, :] * b_col_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # Store the result
    c_ptrs = c_ptr + pid_batch * c_batch_stride \
                     + offs_m[:, None] * c_row_stride \
                     + offs_n[None, :] * c_col_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass for the TriMul (outgoing) update.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------- #
    # 1) Input LayerNorm
    # ----------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------------------- #
    # 2) Linear projections (dim → hidden_dim)
    # ----------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask (if any)
    mask_unsq = mask.unsqueeze(-1).to(x.dtype)       # [B, N, N, 1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ----------------------------------------------------------------- #
    # 3) Gating (sigmoid of Linear)
    # ----------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 4) Cast to half‑precision and reshape for batched GEMM
    # ----------------------------------------------------------------- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Bring the hidden dimension into the batch dimension: [B, H, N, N]
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_h = left_h.view(batch_dim, N, N)      # [B*H, N, N]
    right_h = right_h.view(batch_dim, N, N)    # [B*H, N, N]

    # Right matrix needs to be transposed for A @ Bᵀ
    right_h_T = right_h.transpose(1, 2).contiguous()   # [B*H, N, N]

    # Output buffer for the batched matmul
    out_h = torch.empty_like(left_h)   # half‑precision buffer

    # ----------------------------------------------------------------- #
    # 5) Triton batched GEMM: out_h = left_h @ (right_h)ᵀ
    # ----------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    a_batch_stride, a_row_stride, a_col_stride = left_h.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_h_T.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_h.stride()

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_h,
        right_h_T,
        out_h,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        N,          # M
        N,          # N
        N,          # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------- #
    # 6) Restore the original shape: [B, N, N, hidden_dim] (float32)
    # ----------------------------------------------------------------- #
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ----------------------------------------------------------------- #
    # 7) Output LayerNorm + gating + final Linear (hidden_dim → dim)
    # ----------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # float32

    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
