def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)  # shape [B,N,N,dim]

    left = F.linear(x, left_proj_weight)  # [B,N,N,hidden_dim]

    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))

    right = right * right_gate

    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    right_h = right_perm.to(torch.float16)

    batch_cnt = B * H

    right_flat = right_h.view(batch_cnt, N, N)

"""
TriMul (outgoing) forward pass.

The heavy O(N³) part is a batched matrix multiplication:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
which can be written as a batch of GEMMs  C = A @ Bᵀ
with A, B ∈ ℝ^{batch·hidden, N, N}.  The GEMM is performed by a
custom Triton kernel that works in fp16 (tensor‑core) arithmetic while
accumulating in fp32 for numerical stability.  All other operations
(LayerNorm, Linear, gating, mask handling) are done with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A (batch, M, K)
    stride_bk, stride_bn,          # Bᵀ (batch, K, N)  (original B is (batch, N, K))
    stride_cm, stride_cn,          # C (batch, M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ using fp16 tensor‑cores.

    Each program instance computes one (BLOCK_M × BLOCK_N) tile of a
    particular batch (pid_b).  The innermost reduction over K is tiled
    with size BLOCK_K.
    """
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # Offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the selected batch
    a_base = a_ptr + pid_b * stride_ab
    b_base = b_ptr + pid_b * stride_bb
    c_base = c_ptr + pid_b * stride_cb

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load Bᵀ tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Tensor‑core dot (fp16 inputs) → cast to fp32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # Store result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load model weights (all are float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (applied after projection)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates (float32)
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Batched GEMM: out = left @ rightᵀ   (heavy O(N³) part)
    # -----------------------------------------------------------------
    # Bring the hidden dimension into the batch axis and make tensors contiguous.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Cast to half for tensor‑core computation.
    left_h = left_perm.to(torch.float16)
    right_h = right_perm.to(torch.float16)

    B, H, N, _ = left_h.shape
    batch_cnt = B * H

    # View as (batch_cnt, N, N)
    left_flat = left_h.view(batch_cnt, N, N)
    right_flat = right_h.view(batch_cnt, N, N)

    # Output buffer (fp32 accumulator)
    out_flat = torch.empty((batch_cnt, N, N), dtype=torch.float32, device=device)

    # Strides (int64)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_ab = left_flat.stride(0)

    # Bᵀ strides (swap K and N)
    stride_bk = right_flat.stride(2)   # K stride after transpose
    stride_bn = right_flat.stride(1)   # N stride after transpose
    stride_bb = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    stride_cb = out_flat.stride(0)

    # Tiling parameters – tuned for fp16 tensor‑cores.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _triton_gemm_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                      # M
        N,                      # N
        N,                      # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to original dim

    return out
