@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_m, stride_left_k,
    stride_right_k, stride_right_n,
    stride_out_m, stride_out_n,
    stride_batch_left,
    stride_batch_right,
    stride_batch_out,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)  # batch*hidden index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointer for this batch
    a_ptrs = left_ptr + pid_b * stride_batch_left + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    b_ptrs = right_ptr + pid_b * stride_batch_right + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)  # dot will cast to float16 maybe but addition to float32?

        # increment pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # store
    c_ptrs = out_ptr + pid_b * stride_batch_out + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Get weights:
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps) # shape B N N dim

    # 2) Projections
    left = torch.nn.functional.linear(x, left_proj_weight) # shape B N N hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # 3) Mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype) # shape B N N 1
        left = left * mask_
        right = right * mask_

    # 4) Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel (to reduce memory)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden dim to batch axis: [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_cnt = B * H

    left_flat = left_perm.view(batch_cnt, N, N)
    right_flat = right_perm.view(batch_cnt, N, N)

    # Output buffer in float32
    out_flat = torch.empty((batch_cnt, N, N), dtype=torch.float32, device=device)

    # Strides
    stride_left_batch = left_flat.stride(0)
    stride_left_m = left_flat.stride(1)  # row
    stride_left_k = left_flat.stride(2)  # column
    stride_right_batch = right_flat.stride(0)
    stride_right_k = right_flat.stride(2)  # original column
    stride_right_n = right_flat.stride(1)  # original row
    stride_out_batch = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    # Choose block sizes based on N: For small N maybe smaller blocks.
    # Use heuristics: BLOCK_M = 128 if N>=128 else 64, similarly for BLOCK_N.
    # BLOCK_K = 32.
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32

    # Launch grid
    grid = (batch_cnt, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,  # M
        N,  # N
        N,  # K
        stride_left_m,
        stride_left_k,
        stride_right_k,
        stride_right_n,
        stride_out_m,
        stride_out_n,
        stride_left_batch,
        stride_right_batch,
        stride_out_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,hidden_dim (float32)

    # 5) Output normalization and projection
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward pass.

The heavy O(N³) computation
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed with a custom Triton kernel that computes a batched
matrix multiplication L @ Rᵀ for each hidden dimension d (and batch b).
All other operations (LayerNorm, Linear projections, gating) are done
with PyTorch.  The kernel works on FP16 inputs and accumulates in FP32
to keep the final precision while reducing memory bandwidth.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M, K)
    stride_bk, stride_bn,          # Bᵀ: (K, N)  (original B is (N, K))
    stride_cm, stride_cn,          # C: (M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)   # row block
    pid_n = tl.program_id(2)   # col block

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this tile (k starts at 0)
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)   # accumulation in FP32

        # advance to next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write the result
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask: torch.Tensor [B, N, N] (or None)
            weights: dict of model parameters (see keys below)
            config: dict with "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (over the feature dimension)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)       # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)     # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3) Optional mask (applied after projection)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Batched GEMM: out = left @ rightᵀ for each hidden channel
    #    - Convert to FP16 to reduce memory traffic
    #    - Fuse hidden dimension into the batch axis for independent GEMMs
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # bring hidden dim to batch axis
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape
    batch_cnt = B * H

    left_flat = left.view(batch_cnt, N, N)
    right_flat = right.view(batch_cnt, N, N)

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty((batch_cnt, N, N), dtype=torch.float32, device=device)

    # Strides (all tensors are contiguous)
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # K dimension (original column)
    stride_bn = right_flat.stride(1)   # N dimension (original row)

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # -----------------------------------------------------------------
    # Choose tile sizes based on N (heuristic)
    # -----------------------------------------------------------------
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to original dim

    return out
```
