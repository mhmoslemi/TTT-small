"""
TriMul (outgoing) forward pass with a Triton kernel.

The heavy O(N³) operation is:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

We reshape the tensors so that each (batch, hidden_dim) pair becomes an
independent matrix‑multiply: for each pair we compute
    out = left @ rightᵀ
where left/right have shape (N, N).  The kernel tiles the M/N dimensions
(=sequence length) and streams over K using tl.dot.  All other ops
(LayerNorm, Linear, gating) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_trimul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,        # A: (batch, M, K)
    stride_bk, stride_bn,        # Bᵀ: (batch, K, N)  (original B is (batch, N, K))
    stride_cm, stride_cn,        # C: (batch, M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # row block index
    pid_n = tl.program_id(2)   # col block index

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator for the block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            mask         : torch.Tensor [B, N, N] or None
            weights      : dict of torch.Tensors with model parameters
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) -> left / right
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ for each hidden channel
    # -----------------------------------------------------------------
    # Bring hidden dimension to the batch axis: [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_cnt = B * H

    # Collapse batch and hidden into a single GEMM batch dimension
    left_flat = left_perm.view(batch_cnt, N, N)    # [B*H, N, N]
    right_flat = right_perm.view(batch_cnt, N, N)  # [B*H, N, N]

    # Output buffer
    out_flat = torch.empty_like(left_flat)         # [B*H, N, N]

    # Strides (all tensors are contiguous)
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # K dimension (original column)
    stride_bn = right_flat.stride(1)   # N dimension (original row)

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Tiling parameters – chosen to fit in shared memory and work well on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _triton_trimul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                 # M, N, K (all equal to seq_len)
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to original dim

    return out
