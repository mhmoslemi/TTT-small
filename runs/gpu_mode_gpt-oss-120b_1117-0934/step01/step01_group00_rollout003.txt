"""
TriMul (outgoing) forward pass with an optimized Triton kernel.
All heavy LayerNorm/Linear/Gating are performed in PyTorch.
The core O(N³) reduction is computed as a batched GEMM:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
We avoid any memory copies by using the original strided layout and
transposing the right operand only virtually via swapped strides.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                         # pointers
    a_batch_stride, a_i_stride, a_k_stride, a_hidden_stride,
    b_batch_stride, b_k_stride, b_i_stride, b_hidden_stride,
    c_batch_stride, c_i_stride, c_j_stride, c_hidden_stride,
    M, N, K,                                     # matrix dims
    H: tl.constexpr,                             # hidden_dim
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------------------
    # decode batch index (b, d) from a flat batch dimension
    # -------------------------------------------------------------
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b_idx = pid_batch // H                # batch index (0 .. B-1)
    d_idx = pid_batch % H                 # hidden‑dim index (0 .. H‑1)

    # base pointers for this (b, d) pair
    a_base = a_ptr + b_idx * a_batch_stride + d_idx * a_hidden_stride
    b_base = b_ptr + b_idx * b_batch_stride + d_idx * b_hidden_stride
    c_base = c_ptr + b_idx * c_batch_stride + d_idx * c_hidden_stride

    # -------------------------------------------------------------
    # tile offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the beginning of the current tile
    a_ptrs = a_base + offs_m[:, None] * a_i_stride + offs_k[None, :] * a_k_stride
    # note the swapped strides: rows correspond to k, cols to j
    b_ptrs = b_base + offs_k[:, None] * b_k_stride + offs_n[None, :] * b_i_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                           # (BLOCK_M, BLOCK_N)

        # advance to the next k‑tile
        a_ptrs += BLOCK_K * a_k_stride
        b_ptrs += BLOCK_K * b_k_stride

    # -------------------------------------------------------------
    # write out
    # -------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * c_i_stride + offs_n[None, :] * c_j_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass for the TriMul (outgoing) update.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (dim -> dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (dim -> hidden_dim) and optional mask
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)      # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 3) Gating (sigmoid of Linear) – same dim -> hidden_dim
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Cast to half‑precision for the heavy reduction
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # output buffer (same layout as left/right)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 5) Triton batched GEMM: out_h = left_h @ (right_h)ᵀ  (no copy!)
    # -----------------------------------------------------------------
    # strides for left/right tensors (batch, i, k, hidden)
    a_batch_stride, a_i_stride, a_k_stride, a_hidden_stride = left_h.stride()
    b_batch_stride, b_i_stride, b_k_stride, b_hidden_stride = right_h.stride()
    c_batch_stride, c_i_stride, c_j_stride, c_hidden_stride = out_h.stride()

    # Virtual transposition of the right operand: rows ↔ columns
    # After transposition: row stride = original k‑stride, col stride = original i‑stride
    b_row_stride = a_k_stride          # (original k‑stride)
    b_col_stride = a_i_stride          # (original i‑stride)

    # Choose block sizes (tuned for typical N up to 1024)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = BLOCK_N = 96
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        B * hidden_dim,               # flattened (batch, hidden) dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        a_batch_stride, a_i_stride, a_k_stride, a_hidden_stride,
        b_batch_stride, b_row_stride, b_col_stride, b_hidden_stride,
        c_batch_stride, c_i_stride, c_j_stride, c_hidden_stride,
        N, N, N,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Convert back to fp32 and apply output LayerNorm + gating + final Linear
    # -----------------------------------------------------------------
    out = out_h.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim]

    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
