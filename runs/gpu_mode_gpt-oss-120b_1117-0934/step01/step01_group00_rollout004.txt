"""
TriMul (outgoing) forward pass with a custom Triton kernel for the
core reduction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
The reduction is implemented as a batched GEMM (A @ Bᵀ) where each hidden
channel is treated as an independent batch element.  All cheap operations
(LayerNorm, linear projections, gating) are executed in PyTorch; the
heavy O(N³) work is off‑loaded to the Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # base pointers
    left_batch_stride, left_row_stride, left_col_stride,   # left strides
    right_batch_stride, right_row_stride, right_col_stride, # right strides
    out_batch_stride, out_row_stride, out_col_stride,       # out strides
    M, N, K,                                 # matrix dims (all = seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for a batch of matrices.
    left : (batch, M, K)
    right: (batch, K, N)  (already transposed, i.e. original rightᵀ)
    out  : (batch, M, N)
    """
    pid_batch = tl.program_id(0)   # batch index (B * hidden_dim)
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block col index

    #--- block offsets -------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    #--- compute pointer bases ------------------------------------------------
    left_ptrs = left_ptr \
        + pid_batch * left_batch_stride \
        + offs_m[:, None] * left_row_stride \
        + offs_k[None, :] * left_col_stride

    right_ptrs = right_ptr \
        + pid_batch * right_batch_stride \
        + offs_k[:, None] * right_row_stride \
        + offs_n[None, :] * right_col_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #--- iterate over K -------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(left_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(right_ptrs, mask=b_mask, other=0.0) # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N) in float32

        # advance K‑tiles
        left_ptrs += BLOCK_K * left_col_stride
        right_ptrs += BLOCK_K * right_row_stride

    #--- write result ---------------------------------------------------------
    out_ptrs = out_ptr \
        + pid_batch * out_batch_stride \
        + offs_m[:, None] * out_row_stride \
        + offs_n[None, :] * out_col_stride

    out_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int/float) or None
            - weights: dict of torch.Tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with "dim" and "hidden_dim" (int), optional "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm (dim -> dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # --------------------------------------------------------------------- #
    # 2) Linear projections (dim -> hidden_dim)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------------------- #
    # 3) Gating (sigmoid(linear(x))) – same dim -> hidden_dim
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 4) Cast to float16 and reshape for batched GEMM
    # --------------------------------------------------------------------- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # bring hidden dimension forward: [B, hidden, N, N]
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # merge batch and hidden for batched matmul
    batch_hidden = B * hidden_dim
    left_h = left_h.view(batch_hidden, N, N)       # (B*H, N, N)
    right_h = right_h.view(batch_hidden, N, N)     # (B*H, N, N)

    # transpose right for A @ Bᵀ
    right_h_T = right_h.transpose(1, 2).contiguous()   # (B*H, N, N)

    # output buffer
    out_h = torch.empty_like(left_h)  # float16, shape (B*H, N, N)

    # --------------------------------------------------------------------- #
    # 5) Triton batched GEMM: out_h = left_h @ right_h_T
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (in elements) for Triton; need to convert to int
    a_batch_stride, a_row_stride, a_col_stride = left_h.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_h_T.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_h.stride()

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h_T,
        out_h,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 6) Restore original shape: [B, N, N, hidden_dim] (float32)
    # --------------------------------------------------------------------- #
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------------- #
    # 7) Output LayerNorm, gating and final Linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden]

    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
