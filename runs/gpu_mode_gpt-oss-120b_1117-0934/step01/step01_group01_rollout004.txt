"""
TriMul (outgoing) forward pass with a Triton kernel for the O(N³) update.

The main computation is:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
for each batch b and hidden channel d.  This is a batched GEMM where for
each (b, d) we multiply a (N, N) matrix `left` with the transpose of a
(N, N) matrix `right`.  The kernel tiles the multiplication in blocks
and uses `tl.dot` for the inner K‑tile.  All other operations
(LayerNorm, Linear projections, gating, final projection) are performed
in PyTorch.

The kernel expects tensors in contiguous (batch_cnt, N, N) layout where
`batch_cnt = B * hidden_dim`.  For each (b, d) the kernel computes the
matrix product left @ rightᵀ and writes the result to `out`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch_cnt, M, K)
    stride_bk, stride_bn,          # Bᵀ: (batch_cnt, K, N)   (original B is (batch_cnt, N, K))
    stride_cm, stride_cn,          # C: (batch_cnt, M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiply C = A @ Bᵀ.
    A has shape (batch_cnt, M, K)
    B has shape (batch_cnt, N, K)  (we read it as transposed)
    C has shape (batch_cnt, M, N)
    """
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load tile of A (M x K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load tile of Bᵀ (K x N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of torch.Tensors containing model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load weight tensors (they are already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm over the feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (applied after projection)
    # -----------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; ensure correct dtype and broadcast
        mask_ = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ for each hidden channel
    #    We reshape (B, hidden_dim, N, N) → (B*hidden_dim, N, N) and launch a
    #    Triton kernel where each (b, h) pair is a separate GEMM.
    # -----------------------------------------------------------------
    # Bring the hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_cnt = B * H

    # View as (batch_cnt, N, N)
    left_flat = left_perm.view(batch_cnt, N, N)
    right_flat = right_perm.view(batch_cnt, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)  # [batch_cnt, N, N]

    # Strides (since tensors are contiguous)
    stride_ab = left_flat.stride(0)   # batch stride
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # column stride (K)

    stride_bb = right_flat.stride(0)  # batch stride
    # For transposed access we swap the last two strides
    stride_bk = right_flat.stride(2)  # original column stride -> K stride
    stride_bn = right_flat.stride(1)  # original row stride -> N stride

    stride_cb = out_flat.stride(0)    # batch stride
    stride_cm = out_flat.stride(1)    # row stride (M)
    stride_cn = out_flat.stride(2)    # column stride (N)

    # Tile sizes (tuned for typical GPUs; works for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_cnt,                     # program_id(0)
        triton.cdiv(N, BLOCK_M),      # program_id(1)
        triton.cdiv(N, BLOCK_N),      # program_id(2)
    )

    _triton_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                  # M
        N,                  # N
        N,                  # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
