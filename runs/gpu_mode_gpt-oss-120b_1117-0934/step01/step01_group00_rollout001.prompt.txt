import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

TriMul (outgoing) forward pass with a Triton kernel for the heavy

All other operations (LayerNorm, Linear projections, gating) are performed

batched GEMM using a custom Triton kernel.

import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (batch, M, K)   (float16)
#   B : (batch, N, K)   (float16)  --> we pass Bᵀ (shape batch, K, N)
#   C : (batch, M, N)   (float16)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    a_batch_stride, a_row_stride, a_col_stride,   # A strides
    b_batch_stride, b_row_stride, b_col_stride,   # Bᵀ strides
    c_batch_stride, c_row_stride, c_col_stride,   # C strides
    M, N, K,                                 # matrix dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block col index

    # ---- compute offsets within the block ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- pointers to the start of the current block ----
    a_ptrs = a_ptr + pid_batch * a_batch_stride \
                     + offs_m[:, None] * a_row_stride \
                     + offs_k[None, :] * a_col_stride
    b_ptrs = b_ptr + pid_batch * b_batch_stride \
                     + offs_k[:, None] * b_row_stride \
                     + offs_n[None, :] * b_col_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over K dimension ----
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a_mask = mask_m[:, None] & k_valid[None, :]
        b_mask = k_valid[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) -> float32

        # advance pointers to next k‑tile
        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # ---- write the result ----
    c_ptrs = c_ptr + pid_batch * c_batch_stride \
                     + offs_m[:, None] * c_row_stride \
                     + offs_n[None, :] * c_col_stride
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

def custom_kernel(data):
    """
    Forward pass for the TriMul (outgoing) update.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (0/1)
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # 1) LayerNorm on the input (dim -> dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32

    # --------------------------------------------------------------------- #
    # 2) Linear projections (dim -> hidden_dim) and mask
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    mask_unsq = mask.unsqueeze(-1).to(x.dtype)       # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # --------------------------------------------------------------------- #
    # 3) Gating (sigmoid of Linear) – same dim -> hidden_dim
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 4) Cast to float16 for the heavy reduction
    # --------------------------------------------------------------------- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (batch*hidden_dim, N, N) where each "batch" is a separate GEMM
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_h = left_h.view(batch_dim, N, N)   # [B*H, N, N]
    right_h = right_h.view(batch_dim, N, N)

    # rightᵀ for matmul
    right_h_T = right_h.transpose(1, 2).contiguous()   # [B*H, N, N]

    # output buffer
    out_h = torch.empty_like(left_h)  # float16, same shape

    # --------------------------------------------------------------------- #
    # 5) Triton batched matmul: out_h = left_h @ (right_h)ᵀ
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    a_batch_stride, a_row_stride, a_col_stride = left_h.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_h_T.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_h.stride()

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_h,
        right_h_T,
        out_h,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 6) Restore original shape: [B, N, N, hidden_dim] (float32)
    # --------------------------------------------------------------------- #
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------------- #
    # 7) Output LayerNorm (hidden_dim) + gating + final Linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # float32

    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
