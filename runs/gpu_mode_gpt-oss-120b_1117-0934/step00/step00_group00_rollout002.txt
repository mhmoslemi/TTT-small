@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    seq_len: tl.constexpr, hidden_dim: tl.constexpr,
    stride_la_batch, stride_la_i, stride_la_k, stride_la_h,
    stride_rb_batch, stride_rb_i, stride_rb_k, stride_rb_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden indices
    bh = pid
    batch_idx = bh // hidden_dim
    hidden_idx = bh % hidden_dim

    # base pointers for this (batch, hidden)
    a_ptr = left_ptr + batch_idx * stride_la_batch + hidden_idx * stride_la_h
    b_ptr = right_ptr + batch_idx * stride_rb_batch + hidden_idx * stride_rb_h
    c_ptr = out_ptr + batch_idx * stride_out_batch + hidden_idx * stride_out_h

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in blocks
    for start_k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        cur_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < seq_len

        # load A block (offs_i x cur_k)
        a = tl.load(a_ptr + offs_i[:, None] * stride_la_i + cur_k[None, :] * stride_la_k,
                    mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B block (cur_k x offs_j) But B we need B_T (k x j) => use stride_rb_k for rows, stride_rb_i for columns
        b = tl.load(b_ptr + cur_k[:, None] * stride_rb_k + offs_j[None, :] * stride_rb_i,
                    mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # store
    c = acc.to(tl.float32) # already float32
    tl.store(c_ptr + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j,
             c, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast input to float32 for layer norm
    x = input_tensor

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if not nomask
    if not nomask:
        mask_unsqueezed = mask.to(x.dtype).unsqueeze(-1)  # broadcast
        left = left * mask_unsqueezed
        right = right * mask_unsqueezed

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # At this point we have left/right shape [bs,N,N,hidden] and dtype float32.
    # Cast to float16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output tensor
    out = torch.empty_like(left_fp16, dtype=torch.float32)  # shape [bs,N,N,hidden]

    # Get stride info in elements
    stride_la_batch, stride_la_i, stride_la_k, stride_la_h = left_fp16.stride()
    stride_rb_batch, stride_rb_i, stride_rb_k, stride_rb_h = right_fp16.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid dims
    batch_hidden = left_fp16.shape[0] * hidden_dim  # bs * hidden
    seq_len = left_fp16.shape[1]  # N
    grid = (batch_hidden,
            (seq_len + BLOCK_M - 1) // BLOCK_M,
            (seq_len + BLOCK_N - 1) // BLOCK_N)

    # Launch kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        seq_len,
        hidden_dim,
        stride_la_batch, stride_la_i, stride_la_k, stride_la_h,
        stride_rb_batch, stride_rb_i, stride_rb_k, stride_rb_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Now out shape [bs,N,N,hidden] float32

    # Apply to_out_norm layer norm on hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Final linear
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    # Ensure dtype float32
    out_final = out_final.to(torch.float32)

    return out_final

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, hidden_dim,
    stride_ah, stride_ai, stride_ak, stride_aH,
    stride_bh, stride_bi, stride_bk, stride_bH,
    stride_ch, stride_ci, stride_cj, stride_cH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute block indices
    pid_batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Derive batch and hidden indices
    batch_idx = pid_batch_hidden // hidden_dim
    hidden_idx = pid_batch_hidden % hidden_dim

    # Base pointers for this batch and hidden
    a_base = a_ptr + batch_idx * stride_ah + hidden_idx * stride_aH
    b_base = b_ptr + batch_idx * stride_bh + hidden_idx * stride_bH
    c_base = c_ptr + batch_idx * stride_ch + hidden_idx * stride_cH

    # Offsets within blocks
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < M

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    num_k_blocks = tl.cdiv(M, BLOCK_K)
    for k_iter in range(num_k_blocks):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < M

        # Load A block
        a = tl.load(a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak,
                    mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B block (transposed: k x j)
        b = tl.load(b_base + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bi,
                    mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product accumulate
        acc += tl.dot(a, b)

    # Store result
    c = acc  # float32
    tl.store(c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj,
             c, mask=mask_i[:, None] & mask_j[:, None])

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, hidden_dim,
    stride_ah, stride_ai, stride_ak, stride_aH,
    stride_bh, stride_bi, stride_bk, stride_bH,
    stride_ch, stride_ci, stride_cj, stride_cH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b, h, i, j] = sum_k left[b, i, k, h] * right[b, j, k, h]
    (equivalent to einsum(... i k d, ... j k d -> ... i j d)).

    Arguments:
        a_ptr, b_ptr, c_ptr: pointers to the left, right and output tensors.
        M: sequence length (N).
        hidden_dim: size of the hidden dimension.
        stride_*: strides for each tensor in element units.
        BLOCK_M, BLOCK_N, BLOCK_K: block sizes.
    """
    pid_batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Derive batch and hidden indices from the first program ID
    batch_idx = pid_batch_hidden // hidden_dim
    hidden_idx = pid_batch_hidden % hidden_dim

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_ah + hidden_idx * stride_aH
    b_base = b_ptr + batch_idx * stride_bh + hidden_idx * stride_bH
    c_base = c_ptr + batch_idx * stride_ch + hidden_idx * stride_cH

    # Offsets within the output block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < M

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑blocks
    num_k = tl.cdiv(M, BLOCK_K)
    for k_iter in range(num_k):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < M

        # Load a block of A (i × k)
        a = tl.load(
            a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of B transposed (k × j).  We read B[j, k] using swapped strides.
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Write the result
    tl.store(
        c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul block.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [bs, N, N, dim]
            - mask: torch.Tensor of shape [bs, N, N]
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        output tensor of shape [bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on the last dimension -----
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections -----
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating -----
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to half for the heavy kernel -----
    left_h = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    batch, seq_len, _, _ = left_h.shape  # batch = bs

    # Output tensor (float32)
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # ----- Strides (in elements) -----
    stride_la_batch, stride_la_i, stride_la_k, stride_la_h = left_h.stride()
    stride_rb_batch, stride_rb_i, stride_rb_k, stride_rb_h = right_h.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # ----- Kernel launch parameters -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    batch_hidden = batch * hidden_dim
    grid = (
        batch_hidden,
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        seq_len,
        hidden_dim,
        stride_la_batch,
        stride_la_i,
        stride_la_k,
        stride_la_h,
        stride_rb_batch,
        stride_rb_i,
        stride_rb_k,
        stride_rb_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- LayerNorm on hidden dimension -----
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ----- Apply output gate -----
    out_norm = out_norm * out_gate

    # ----- Final linear projection back to dim -----
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final.to(torch.float32)

"""
Triton‑accelerated forward pass for the “outgoing” TriMul block.

The heavy part of TriMul is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    out[b,d] = left[b,d] @ right[b,d].transpose(-2, -1)

This implementation:
  * Performs LayerNorm, linear projections, masking and gating in PyTorch.
  * Casts the projected tensors to float16 and computes the
    N×N matrix multiplication for each (batch, hidden) slice with a
    custom Triton kernel (block‑wise dot products, accumulating in float32).
  * Applies the final LayerNorm, output gate and linear projection in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, hidden_dim,
    stride_ah, stride_ai, stride_ak, stride_aH,
    stride_bh, stride_bi, stride_bk, stride_bH,
    stride_ch, stride_ci, stride_cj, stride_cH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h].

    Arguments:
        a_ptr, b_ptr, c_ptr: base pointers for left, right and output tensors.
        M: sequence length (N).
        hidden_dim: size of the hidden dimension.
        stride_*: strides (in elements) for each tensor dimension.
        BLOCK_M, BLOCK_N, BLOCK_K: tile sizes.
    """
    pid_bn = tl.program_id(0)   # batch * hidden
    pid_i  = tl.program_id(1)   # block row
    pid_j  = tl.program_id(2)   # block column

    # batch and hidden index
    batch_idx = pid_bn // hidden_dim
    hidden_idx = pid_bn % hidden_dim

    # base pointer for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_ah + hidden_idx * stride_aH
    b_base = b_ptr + batch_idx * stride_bh + hidden_idx * stride_bH
    c_base = c_ptr + batch_idx * stride_ch + hidden_idx * stride_cH

    # offsets inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < M

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(M, BLOCK_K)
    for k_iter in range(num_k):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < M

        # A block: (i, k)
        a = tl.load(
            a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block (transposed): we need B[j, k] → load as (k, j)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store result (float32)
    tl.store(
        c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block with a Triton kernel for the
    N×N reduction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [bs, N, N, dim]
            - mask: torch.Tensor [bs, N, N] (may be ignored if config["nomask"])
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm over the last dimension -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask (broadcast over hidden_dim) -----
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gates (sigmoid of linear layers) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to half for the heavy kernel -----
    left_h = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    bs, seq_len, _, _ = left_h.shape  # shape = [bs, N, N, hidden]

    # Output tensor for the reduction (float32)
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # ----- Strides (in element units) -----
    stride_la_batch, stride_la_i, stride_la_k, stride_la_h = left_h.stride()
    stride_rb_batch, stride_rb_i, stride_rb_k, stride_rb_h = right_h.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # ----- Triton launch configuration -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    batch_hidden = bs * hidden_dim
    grid = (
        batch_hidden,
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        seq_len,
        hidden_dim,
        stride_la_batch,
        stride_la_i,
        stride_la_k,
        stride_la_h,
        stride_rb_batch,
        stride_rb_i,
        stride_rb_k,
        stride_rb_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- LayerNorm over the hidden dimension -----
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ----- Apply output gate and final linear projection -----
    out_norm = out_norm * out_gate
    final = F.linear(out_norm, weights["to_out.weight"])

    return final.to(torch.float32)
