"""
TriMul (outgoing) forward pass implemented with a small Triton kernel.

The heavy N³ reduction (einsum) is performed with a batched matrix multiplication
(bmm) after applying the linear projections and gates.  A Triton kernel fuses
masking, gating and element‑wise multiplication for the left and right
projected tensors, ensuring that at least part of the computation runs on the
GPU via Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------
# Triton kernel: fused mask * gate * projection
# -----------------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    proj_ptr,          # float*   projection (left or right)
    gate_ptr,          # float*   sigmoid‑gate (already after sigmoid)
    mask_ptr,          # float*   mask (broadcasted over hidden dim)
    out_ptr,           # float*   output buffer (same shape as projection)
    B, N, H,           # int32    batch, seq_len, hidden_dim
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_out_b,  stride_out_i,  stride_out_k,  stride_out_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK: tl.constexpr,               # threads per program (set to 1)
    BLOCK_H: tl.constexpr,             # hidden‑dim tile size
):
    # each program handles exactly one (b, i, k) element
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode linear index into (b, i, k)
    b = pid // (N * N)
    rem = pid - b * (N * N)
    i = rem // N
    k = rem - i * N

    # load the scalar mask value
    mask_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    mask_val = tl.load(mask_ptr + mask_offset)

    # loop over hidden dimension in chunks of BLOCK_H
    off_h = 0
    while off_h < H:
        cur_h = tl.arange(0, BLOCK_H) + off_h
        mask_h = cur_h < H

        # offsets for projection and gate
        proj_offset = (
            b * stride_proj_b
            + i * stride_proj_i
            + k * stride_proj_k
            + cur_h * stride_proj_h
        )
        gate_offset = (
            b * stride_gate_b
            + i * stride_gate_i
            + k * stride_gate_k
            + cur_h * stride_gate_h
        )
        out_offset = (
            b * stride_out_b
            + i * stride_out_i
            + k * stride_out_k
            + cur_h * stride_out_h
        )

        proj_val = tl.load(proj_ptr + proj_offset, mask=mask_h, other=0.0)
        gate_val = tl.load(gate_ptr + gate_offset, mask=mask_h, other=0.0)

        # fused operation: proj * gate * mask
        out_val = proj_val * gate_val * mask_val
        tl.store(out_ptr + out_offset, out_val, mask=mask_h)

        off_h += BLOCK_H


# -----------------------------------------------------------------------------
# Entry point used by the evaluation harness
# -----------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary or all‑ones)
            - weights: dict of torch.Tensors with the model parameters
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Load weights (all are float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # LayerNorm on input
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -----------------------------------------------------------------
    # Linear projections and gates (no bias as in the reference)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_weight)

    left_gate_raw = F.linear(x, left_gate_weight)
    right_gate_raw = F.linear(x, right_gate_weight)
    out_gate_raw = F.linear(x, out_gate_weight)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # Prepare mask (float32) – if nomask, use all‑ones
    # -----------------------------------------------------------------
    if nomask:
        mask_f = torch.ones_like(mask, dtype=torch.float32, device=device)
    else:
        mask_f = mask.to(torch.float32)

    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # Fuse mask, gate and projection with a Triton kernel
    # -----------------------------------------------------------------
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # one program per (b,i,k) element → grid size = B * N * N
    total_el = B * N * N
    grid = (total_el,)

    # launch kernel for the left side
    mask_gate_kernel[grid](
        left_proj,
        left_gate,
        mask_f,
        left,
        B, N, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
        BLOCK=1, BLOCK_H=32,
    )

    # launch kernel for the right side
    mask_gate_kernel[grid](
        right_proj,
        right_gate,
        mask_f,
        right,
        B, N, hidden_dim,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
        BLOCK=1, BLOCK_H=32,
    )

    # -----------------------------------------------------------------
    # Outer‑product reduction: shape (B,N,N,hidden_dim)
    #   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # Implemented as batched GEMM over the hidden dimension.
    # -----------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)
    right_bh = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim) and cast to float32
    out = out_bh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # Final normalization, gating and projection back to `dim`
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate
    output = F.linear(out, to_out_weight)  # (B,N,N,dim)

    return output
