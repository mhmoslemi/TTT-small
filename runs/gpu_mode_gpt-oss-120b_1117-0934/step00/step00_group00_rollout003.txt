def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_...,
    BLOCK_H: tl.constexpr):

    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # compute base offset for left/right per b,i,j across h dimension start
    offset_left = b*stride_l_b + i*stride_l_i + j*stride_l_j
    offset_right = b*stride_r_b + i*stride_r_i + j*stride_r_j
    offset_left_gate = b*stride_gb_b + i*stride_gb_i + j*stride_gb_j
    offset_right_gate = b*stride_ga_b + i*stride_ga_i + j*stride_ga_j
    offset_mask = b*stride_m_b + i*stride_m_i + j*stride_m_j

    # load mask scalar broadcast across H
    mask_val = tl.load(mask_ptr + offset_mask)  # shape scalar
    # compute mask expansion vector later via broadcast

    # Offsets for H dimension: we need to iterate across hidden dimension in blocks.
    h_offsets = tl.arange(0, BLOCK_H)
    valid_h = h_offsets < H
    # compute per-element offsets
    offs_left = offset_left + h_offsets * stride_l_h
    offs_right = offset_right + h_offsets * stride_r_h
    offs_left_gate = offset_left_gate + h_offsets * stride_gb_h

    left = tl.load(left_ptr + offs_left, mask=valid_h, other=0.0)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Flatten mask to float
    mask = mask_tensor.to(device=device, dtype=input_tensor.dtype)

    # LayerNorm on input
    norm_weight = weights["norm.weight"].to(device, dtype=input_tensor.dtype)
    norm_bias = weights["norm.bias"].to(device, dtype=input_tensor.dtype)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(device, dtype=input_tensor.dtype) # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype=input_tensor.dtype)
    left_gate_weight = weights["left_gate.weight"].to(device, dtype=input_tensor.dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype=input_tensor.dtype)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype=input_tensor.dtype)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=input_tensor.dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=input_tensor.dtype)
    to_out_weight = weights["to_out.weight"].to(device, dtype=input_tensor.dtype)

    # Compute linear before gating
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()
    mask = mask.contiguous()

    # Allocate gated tensors
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Launch Triton kernel to apply mask and gates
    B, N, _, _ = left.shape
    # Actually left shape (B,N,N,hidden_dim)
    # We'll pass stride values.
    # Determine strides
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h = left_gate.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()
    BLOCK_H = 64 # adjust for hidden_dim

    grid = (B, N, N)
    mask_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        left_gated, right_gated,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_j, stride_left_h,
        stride_right_b, stride_right_i, stride_right_j, stride_right_h,
        stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
        stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        BLOCK_H=BLOCK_H,
    )
    # Now we have left_gated, right_gated

    # Compute batched GEMM
    # Permute for GEMM
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)
    # Flatten batch*H
    left_mat = left_perm.view(-1, N, N)  # shape (B*H, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Cast to float16 for speed, accumulate in float32
    left_mat_fp16 = left_mat.to(torch.float16)
    right_mat_fp16 = right_mat.to(torch.float16)

    out_mat_fp16 = torch.bmm(left_mat_fp16, right_mat_fp16)  # (B*H,N,N)
    out_mat = out_mat_fp16.to(torch.float32)

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # (B,N,N,H)

    # Layer norm on hidden dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out gate
    out = out * out_gate

    # Final linear
    output = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return output
```

def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # bounds check (if dimension may be >N e.g., B become small)
    mask = tl.load(mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j, mask=True, other=0.0).to(tl.float32)

    right = tl.load(right_ptr + right_off, mask=valid_h, other=0.0)

def custom_kernel(data):
    import torch, triton, triton.language as tl
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # make contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask_tensor.to(dtype).contiguous()

    # apply mask & gates via Triton
    B, N, _, _ = left.shape

    # Strides
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h = left_gate.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    BLOCK_H = 64
    grid = (B, N, N, triton.cdiv(hidden_dim, BLOCK_H))
    apply_mask_gate_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask,
        left, right,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_j, stride_left_h,
        stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
        stride_right_b, stride_right_i, stride_right_j, stride_right_h,
        stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        BLOCK_H=BLOCK_H,
    )

    # Now left/right are masked and gated

    # Batched GEMM: left @ right^T
    # permute for better memory pattern
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,2,1).contiguous()

    # reshape batch*hidden
    B, H, N, N = left_perm.shape
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)

    # Cast to fp16 for speed, compute in fp16 and cast back
    left_mat_f16 = left_mat.to(torch.float16)
    right_mat_f16 = right_mat.to(torch.float16)

    out_mat_f16 = torch.bmm(left_mat_f16, right_mat_f16)  # (B*H,N,N)
    out_mat = out_mat_f16.to(torch.float32)

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)

    # output layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # multiply by out_gate
    out = out * out_gate

    # final linear
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

def apply_mask_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,

    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr,

    j = tl.program_id(2)

```python
import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr):
    """
    In-place masking and gating.
    Computes:
        out_left = left * left_gate * mask
        out_right = right * right_gate * mask
    where mask is broadcast over the hidden dimension H.
    """
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)
    pid_h = tl.program_id(3)

    # starting index in hidden dimension for this block
    h_start = pid_h * BLOCK_H
    h = h_start + tl.arange(0, BLOCK_H)

    # mask scalar (per (b,i,j))
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset)

    # base offsets (no hidden dim)
    off_left_base = b * stride_left_b + i * stride_left_i + j * stride_left_j
    off_lgate_base = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j
    off_right_base = b * stride_right_b + i * stride_right_i + j * stride_right_j
    off_rgate_base = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j

    # guard for threads that exceed H
    mask_h = h < H

    # load values
    left = tl.load(left_ptr + off_left_base + h * stride_left_h, mask=mask_h, other=0.0)
    left_gate = tl.load(left_gate_ptr + off_lgate_base + h * stride_lgate_h, mask=mask_h, other=0.0)
    right = tl.load(right_ptr + off_right_base + h * stride_right_h, mask=mask_h, other=0.0)
    right_gate = tl.load(right_gate_ptr + off_rgate_base + h * stride_rgate_h, mask=mask_h, other=0.0)

    # compute gated and masked values
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # store results (can overwrite original tensors)

    tl.store(out_right_ptr + off_right_base + h * stride_right_h, right_out, mask=mask_h)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.
    The implementation follows the reference PyTorch code:
    1. Layer‑normalize the input.
    2. Linear projections to obtain left/right tensors and their gates.
    3. Apply a binary mask and gate scalings (performed in a fused Triton kernel).
    4. Batched matrix multiplication L @ Rᵀ per hidden channel (FP16 GEMM via torch.bmm).
    5. Layer‑normalize the hidden result, multiply by an output gate, and linearly project back.
    All computations are carried out in float32, with the heavy matrix multiplication
    performed in float16 for speed.
    """
    import torch
    import triton
    import triton.language as tl

    # unpack data
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------
    # 1. Input layer‑norm
    # -----------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # -----------------------------
    # 2. Linear projections + gates
    # -----------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)

    left = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Ensure tensors are contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # -----------------------------
    # 3. Mask & gate fusion (Triton)
    # -----------------------------
    # mask may be boolean; cast to float same as data dtype
    mask = mask_tensor.to(dtype=dtype).contiguous()

    B, N, _, _ = left.shape  # (B,N,N,hidden_dim)

    # stride information (Python ints)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h = left_gate.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    BLOCK_H = 64  # tuning parameter; must divide hidden_dim reasonably

    grid = (
        B,
        N,
        N,
        triton.cdiv(hidden_dim, BLOCK_H),
    )
    apply_mask_gate_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask,
        left, right,                # in‑place write‑back
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_j, stride_left_h,
        stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
        stride_right_b, stride_right_i, stride_right_j, stride_right_h,
        stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        BLOCK_H=BLOCK_H,
    )
    # -------------------------------------------------
    # 4. Batched GEMM: out = left @ rightᵀ per hidden channel
    # -------------------------------------------------
    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,H) -> (B,H,N,N) and transpose the two N axes
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # reshape to (B*H, N, N)
    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)

    # half‑precision GEMM for speed
    left_mat_f16 = left_mat.to(torch.float16)
    right_mat_f16 = right_mat.to(torch.float16)

    out_mat_f16 = torch.bmm(left_mat_f16, right_mat_f16)  # (B*H,N,N)
    out_mat = out_mat_f16.to(torch.float32)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H_, N_, N_).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 5. Output normalization, gate, final linear
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)
    output = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    return output
```

"""
TriMul “outgoing” forward pass with a fused Triton kernel for mask & gating.

The algorithm follows the reference PyTorch implementation:
1. Layer‑normalize the input tensor.
2. Linear projections → left, right and their sigmoid gates.
3. Apply binary mask and gate scalings in a single Triton kernel.
4. Batched matrix multiplication L @ Rᵀ per hidden channel (FP32 GEMM via torch.bmm).
5. Layer‑normalize the hidden result, multiply by the output gate, and project back.
All tensors are kept in float32; the heavy matrix multiplication is done in
torch‑cuBLAS, while the element‑wise mask/gate fusion is accelerated with Triton.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr):
    """
    out_left  = left  * left_gate  * mask
    out_right = right * right_gate * mask
    The mask is broadcast over the hidden dimension H.
    """
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)
    pid_h = tl.program_id(3)

    # hidden‑dimension slice handled by this block
    h_start = pid_h * BLOCK_H
    h = h_start + tl.arange(0, BLOCK_H)

    # scalar mask for the (b,i,j) pair
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset)

    # base offsets (no hidden dimension)
    base_left = b * stride_left_b + i * stride_left_i + j * stride_left_j
    base_lgate = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j
    base_right = b * stride_right_b + i * stride_right_i + j * stride_right_j
    base_rgate = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j

    # guard for threads that exceed H
    mask_h = h < H

    # load tensors
    left = tl.load(left_ptr + base_left + h * stride_left_h,
                   mask=mask_h, other=0.0)
    left_gate = tl.load(left_gate_ptr + base_lgate + h * stride_lgate_h,
                        mask=mask_h, other=0.0)
    right = tl.load(right_ptr + base_right + h * stride_right_h,
                    mask=mask_h, other=0.0)
    right_gate = tl.load(right_gate_ptr + base_rgate + h * stride_rgate_h,
                         mask=mask_h, other=0.0)

    # apply mask and gates
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # store (in‑place write‑back)
    tl.store(out_left_ptr + base_left + h * stride_left_h,
             left_out, mask=mask_h)
    tl.store(out_right_ptr + base_right + h * stride_right_h,
             right_out, mask=mask_h)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing version).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              input_tensor  – shape [B, N, N, dim]
              mask_tensor   – shape [B, N, N] (bool/float)
              weights_dict  – model weights as described in the prompt
              config_dict   – contains "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    import torch
    import triton
    import triton.language as tl

    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections and gates
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)

    left = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # make contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask_tensor.to(dtype=dtype).contiguous()   # broadcastable mask

    B, N, _, _ = left.shape

    # -----------------------------------------------------------------
    # 3. Fuse mask & gate using Triton
    # -----------------------------------------------------------------
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h = left_gate.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    BLOCK_H = 64
    grid = (
        B,
        N,
        N,
        triton.cdiv(hidden_dim, BLOCK_H),
    )
    apply_mask_gate_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask,
        left, right,                     # in‑place write‑back
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_j, stride_left_h,
        stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
        stride_right_b, stride_right_i, stride_right_j, stride_right_h,
        stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # 4. Batched GEMM: L @ Rᵀ per hidden channel
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # transpose the two N dimensions of right before the GEMM
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)

    # FP32 GEMM (torch.bmm is already heavily optimized)
    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H_, N_, N_).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)
    output = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    return output
