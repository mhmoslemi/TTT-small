def layer_norm(x, weight, bias, eps=1e-5):
    # compute mean and var across last dimension
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weight + bias
    return x_norm
```

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    batch = pid_bh // H
    h = pid_bh % H

    # Compute base offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Compute block offsets (for each k iteration)
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # Mask for out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers offset static for batch and channel
    left_batch_offset = batch * stride_left_batch + h * stride_left_h
    right_batch_offset = batch * stride_right_batch + h * stride_right_h
    out_batch_offset = batch * stride_out_batch + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # Load left block (M x K)
        left_ptrs = left_ptr + left_batch_offset + \
            (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block (K x N) - note orientation j axis is columns
        right_ptrs = right_ptr + right_batch_offset + \
            (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute partial multiplication
        acc += tl.dot(left, right)

    # Write back
    out_ptrs = out_ptr + out_batch_offset + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure tensors are contiguous
    x = input_tensor.contiguous()
    mask = mask.contiguous()

    # Load weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Gate and projection linear transforms
    # Compute all linear in float32
    # For gating we need sigmoid
    left_proj = torch.nn.functional.linear(x, left_proj_weight)         # B,N,N,H
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply masks
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left_proj * left_gate * mask_unsq
        right = right_proj * right_gate * mask_unsq
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Triton kernel for bilinear multiplication
    B, N, _, H = left.shape  # left shape B,N,N,H
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Get strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # Block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_j, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Apply out_gate after normalization
    # Normalize out across last dim (H)
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate
    # Linear projection to final dim
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    batch = pid_bh // H
    h = pid_bh % H

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute static offsets for batch and channel
    left_batch_offset = batch * stride_left_batch + h * stride_left_h
    right_batch_offset = batch * stride_right_batch + h * stride_right_h
    out_batch_offset = batch * stride_out_batch + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left_ptrs = left_ptr + left_batch_offset + \
            (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (K x N) with k as rows, j as cols
        right_ptrs = right_ptr + right_batch_offset + \
            (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

    out_ptrs = out_ptr + out_batch_offset + \
        (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])

"""
TriMul “outgoing” implementation.

The heavy O(N³) part (einsum '... i k d, ... j k d -> ... i j d') is
off‑loaded to a Triton kernel.  The kernel computes, for each batch
and hidden channel h, the matrix product

    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]

with left and right already masked and gated.  All other
operations (LayerNorms, linear projections, sigmoids) are kept in
PyTorch for simplicity.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,
    right_ptr,
    out_ptr,
    # sizes
    B, N, H,
    # strides for left: (batch, i, k, h)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right: (batch, j, k, h)
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    # strides for out: (batch, i, j, h)
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)               # block index for i dimension
    pid_j = tl.program_id(1)               # block index for j dimension
    pid_bh = tl.program_id(2)              # combined batch * hidden channel

    batch = pid_bh // H                    # batch index
    h = pid_bh % H                         # hidden channel index

    # base offsets for this block
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # static offsets for batch and channel
    left_batch_offset = batch * stride_left_batch + h * stride_left_h
    right_batch_offset = batch * stride_right_batch + h * stride_right_h
    out_batch_offset = batch * stride_out_batch + h * stride_out_h

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)  (i rows, k cols)
        left_ptrs = left_ptr + left_batch_offset + \
            (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: shape (K, N)  (k rows, j cols)
        right_ptrs = right_ptr + right_batch_offset + \
            (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate partial product
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + out_batch_offset + \
        (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (binary or float)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = torch.nn.functional.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # ---------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ---------- Apply mask and gates ----------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_unsq
        right = right_proj * right_gate * mask_unsq
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape  # H == hidden_dim

    # ---------- Triton kernel for the O(N³) multiplication ----------
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides in elements (not bytes)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        B * H,                     # batch * hidden channel
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb,
        lsi,
        lsk,
        lsh,
        rsb,
        rsj,
        rsk,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # final projection to dim

    return out
