import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(...):

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the Triangle Multiplicative Update (TriMul).
    This implementation follows the “outgoing” variant used in AlphaFold3:
    it applies a LayerNorm, linear projections, gated masking, computes
    out[i,j] = Σ_k left[i,k] * right[j,k] via a blocked matrix‑multiplication
    kernel, normalises, gates and projects back to the original feature dimension.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not config.get("nomask", True):
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --- Triton kernel for the einsum reduction ---
    # Ensure tensors are contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape  # left shape is (B,N,N,hidden_dim)
    # Allocate output tensor for the “inner” result
    out = torch.empty_like(left)

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)

    grid = (num_blocks_m * num_blocks_n, B * hidden_dim)

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_j, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # End of Triton kernel

    # Clean up large intermediates before next steps (optional)
    del left, right

    # Normalisation and output projection
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate

    output = F.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) – Triton accelerated forward pass.

The implementation follows the PyTorch reference:
1. Layer‑norm the input over the channel dimension.
2. Project to a hidden dimension with two linear layers (left/right).
3. Apply an optional pairwise mask.
4. Compute left/right gating scalars (sigmoid of linear projections) and
   multiply them into the projected tensors.
5. Perform the core “triangular multiplicative update”

       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   This is a batched matrix‑multiply across the sequence dimension and is
   implemented with a custom Triton kernel that tiles the M×N output and
   iterates over K in blocks.
6. Apply a second layer‑norm over the hidden dimension, an output gate,
   and a final linear projection back to the original channel size.

All operations after the gate are performed in float32, matching the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    B, N, D,                # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(0)        # indexes over i‑block and j‑block
    pid_bd = tl.program_id(1)     # indexes over batch * hidden

    # number of blocks in each output dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    i_block = pid // num_blocks_n
    j_block = pid % num_blocks_n

    b = pid_bd // D
    d = pid_bd % D

    i_base = i_block * BLOCK_M
    j_base = j_block * BLOCK_N

    # range vectors for this tile
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        k_valid = k_range < cur_k
        k_offset = k_start + k_range

        # ----- left block (M x K) -----
        # left[b, i, k, d]
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + d * stride_l_d
            + (i_range[:, None] * stride_l_i)
            + (k_offset[None, :] * stride_l_k)
        )
        left_mask = i_mask[:, None] & k_valid[None, :]
        left_block = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # ----- right block transposed (K x N) -----
        # right[b, j, k, d]  → we load as (k, j) to match tl.dot
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + d * stride_r_d
            + (k_offset[:, None] * stride_r_k)
            + (j_range[None, :] * stride_r_j)
        )
        right_mask = k_valid[:, None] & j_mask[None, :]
        right_block = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # matrix multiplication for this K‑tile
        acc += tl.dot(left_block, right_block)  # (M,N)

    # ----- store result -----
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + d * stride_o_d
        + (i_range[:, None] * stride_o_i)
        + (j_range[None, :] * stride_o_j)
    )
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of model weight tensors
            - config: dict containing "dim", "hidden_dim", "nomask" etc.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------- layer‑norm on the input -------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=eps,
    )

    # ------- linear projections (no bias) -------
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ------- mask (if used) -------
    if not config.get("nomask", True):
        if mask is None:
            mask = torch.ones((input_tensor.shape[0],
                               input_tensor.shape[1],
                               input_tensor.shape[2]),
                               dtype=input_tensor.dtype,
                               device=device)
        mask = mask.to(input_tensor.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------- gating scalars -------
    left_gate = F.linear(x, weights["left_gate.weight"].to(device)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(device)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(device)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel for the einsum ----------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape  # left shape = (B, N, N, hidden_dim)

    out = torch.empty_like(left)  # allocate result [B,N,N,hidden_dim]

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # block sizes – chosen to fit typical B200 SM resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)

    grid = (num_blocks_m * num_blocks_n, B * hidden_dim)

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ---------------------------------------------------

    # release large temporaries early
    del left, right

    # ------- second layer‑norm over hidden dimension -------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=eps,
    )

    # ------- output gate and final projection -------
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"].to(device))

    return output
