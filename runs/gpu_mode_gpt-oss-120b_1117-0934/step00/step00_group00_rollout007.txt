@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_j, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # compute batch and head
    batch = pid_h // hidden_dim
    head = pid_h % hidden_dim

    # compute base pointers for this batch/head
    # offsets: base_offset = batch*stride_left_b + head
    offset = batch * stride_left_b + head
    left_base = left_ptr + offset
    right_base = right_ptr + offset
    out_base = out_ptr + offset

    # define range indices
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # compute active masks for i,j
    i_valid = i0 + i_range < N
    j_valid = j0 + j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k0 = k_iter * BLOCK_K

        k_valid = k0 + k_range < N

        # Compute pointers for left (i,k)
        left_ptrs = left_base + (i0 + i_range)[:, None] * stride_left_i + (k0 + k_range)[None, :] * stride_left_k
        # Compute pointers for right_T (k,j). right stored as (j,k) so we load transposed.

"""
TriMul (outgoing) implementation with a Triton kernel for the
N³ reduction. The heavy reduction `out[b,i,j,d] = Σₖ left[b,i,k,d] *
right[b,j,k,d]` is performed by a custom Triton kernel; all other
operations (LayerNorms, Linear projections, gating) stay in PyTorch.
The final output has shape [B, N, N, dim] and dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
# One kernel instance handles a single (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_j, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)                     # block row (i)
    pid_j = tl.program_id(1)                     # block col (j)
    pid_h = tl.program_id(2)                     # batch * hidden index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Decode batch and head (hidden) from pid_h
    batch = pid_h // hidden_dim
    head = pid_h % hidden_dim

    # Base pointer for this (batch, head)
    base_offset = batch * stride_left_b + head
    left_base  = left_ptr  + base_offset
    right_base = right_ptr + base_offset
    out_base   = out_ptr   + base_offset

    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_valid = i0 + i_range < N
    j_valid = j0 + j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    NUM_K = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, NUM_K):
        k0 = k_iter * BLOCK_K
        k_valid = k0 + k_range < N

        # left[i,k] block
        left_ptrs = left_base + (i0 + i_range)[:, None] * stride_left_i + (k0 + k_range)[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=(i_valid[:, None] & k_valid[None, :]),
                            other=0.0).to(tl.float32)

        # right_T[k,j] block (right stored as [j,k])
        right_ptrs = right_base + (k0 + k_range)[:, None] * stride_right_k + (j0 + j_range)[None, :] * stride_right_j
        right_tile = tl.load(right_ptrs,
                             mask=(j_valid[None, :] & k_valid[:, None]),
                             other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    out_ptrs = out_base + (i0 + i_range)[:, None] * stride_out_i + (j0 + j_range)[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated outer‑product reduction.
    left, right: [B, N, N, H] float16 tensors (already masked/gated)
    Returns out: same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in elements (not bytes)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_j, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    tri_mul_kernel[grid](
        left, right, out,
        N,
        sL_b, sL_i, sL_k,
        sR_b, sR_j, sR_k,
        sO_b, sO_i, sO_j,
        H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry‑point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module using Triton for the
    N³ reduction. All tensors are assumed to be on the CUDA device already.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = config.get("eps", 1e-5)

    dim = config["dim"]          # input feature dimension
    hidden_dim = config["hidden_dim"]  # projection dimension H

    # ------------------------------------------------------------------
    # Load weights (already on device, ensure float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)

    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float32)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float32)

    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float32)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float32)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float32)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w.view(1, 1, 1, dim) + norm_b.view(1, 1, 1, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_w)          # [B,N,N,H]
    right = F.linear(x_norm, right_proj_w)        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided)
    # ------------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)  # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to float16 and run the Triton kernel for the reduction
    # ------------------------------------------------------------------
    left = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)
    out = tri_mul_triton(left, right)            # [B,N,N,H] float16
    out = out.to(torch.float32)                  # back to fp32 for the rest

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w.view(1, 1, 1, hidden_dim) + \
               to_out_norm_b.view(1, 1, 1, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Apply out_gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, to_out_w)     # [B,N,N,dim]

    return out_final
