import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def triton_matmul_tri(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a = tl.load(
            a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B_T block (K x N)
        b = tl.load(
            b_ptr + pid_b * stride_bb + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        acc += tl.dot(a, b)

    # Store results
    c = acc
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a Triton kernel for the
heavy O(N³) batched matrix multiplication:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorm, Linear projections, gating) are
performed with PyTorch.  The kernel computes a batched GEMM
for each hidden channel (batch * hidden_dim) using tiling and
tl.dot, handling the required transpose of the right operand.
The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # Bᵀ: (batch, K, N)   (original B is (batch, N, K))
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # block row index
    pid_n = tl.program_id(2)   # block col index

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Fast forward pass of the outgoing TriMul operator.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            mask: torch.Tensor of shape [B, N, N] (bool or float)
            weights: dict of torch.Tensors containing model parameters
            config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load weights (all are float32, bias may be None)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (applied after projection)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ for each hidden channel
    # -----------------------------------------------------------------
    # Bring hidden dimension to the batch axis for independent GEMMs
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_cnt = B * H

    left_flat = left_perm.view(batch_cnt, N, N)
    right_flat = right_perm.view(batch_cnt, N, N)

    out_flat = torch.empty_like(left_flat)  # [batch_cnt, N, N]

    # Strides for the flattened tensors (all are contiguous)
    stride_ab = left_flat.stride(0)   # batch stride
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # column stride (K)

    stride_bb = right_flat.stride(0)  # batch stride
    # For Bᵀ we swap the inner two strides:
    stride_bk = right_flat.stride(2)  # original column stride (1) becomes stride for K
    stride_bn = right_flat.stride(1)  # original row stride (N) becomes stride for N

    stride_cb = out_flat.stride(0)    # batch stride
    stride_cm = out_flat.stride(1)    # row stride (M)
    stride_cn = out_flat.stride(2)    # column stride (N)

    # Tiling parameters (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _triton_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                      # M
        N,                      # N
        N,                      # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to original dim

    return out
