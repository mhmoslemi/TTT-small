Equivalent to: left[b,i,:,d] dot right[b,j,:,d].

Thus we need to compute out[b,i,j,d] = sum_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d].

After that, apply to_out_norm, then multiply by out_gate, then a linear projection to output dimension dim.

    B, N, _, C = x.shape
    H = config["hidden_dim"]
    device = x.device

    # norm_weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # using torch.nn.functional.layer_norm
    # torch.nn.functional.layer_norm expects normalized shape = (C,)

    right = right * mask_unsq

    out_gate_weight = weights["out_gate.weight"]

    right = right * right_gate

    # Convert to bfloat16 for matmul
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Triton kernel for matmul across hidden dimension
    # output shape (B,N,N,H) bfloat16 maybe; but we accumulate in float32.

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    pid_i = tl.program_id(0) # block row
    pid_j = tl.program_id(1) # block column
    pid_g = tl.program_id(2) # batch-hidden group

    # compute batch and hidden index
    b = pid_g // H
    h = pid_g % H

    # compute final output pointer offsets

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (C).
2. Linear projections to hidden dimension (H):
   left = x_norm @ left_proj.weight
   right = x_norm @ right_proj.weight
3. Apply the pairwise mask (broadcasted) and element‑wise gates
   (sigmoid of linear transforms).  left/right are multiplied by their
   respective gates.
4. Compute the core O(N³) update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of H independent matrix multiplies
   (L_h @ R_hᵀ).  Implemented with a custom Triton kernel that
   processes blocks of (i,j) while looping over k.
5. Layer‑norm the result over the hidden dimension, multiply by the
   output gate, and finally linearly project back to the original dim.
All steps except the heavy reduction are performed with PyTorch; the
reduction is fully fused in Triton for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_mm_kernel(
    left_ptr,         # bf16 *
    right_ptr,        # bf16 *
    out_ptr,          # fp32 *
    B, N, H,          # integers
    # strides of left (B,N,N,H) tensor
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides of right (B,N,N,H) tensor
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides of out (B,N,N,H) tensor
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # block row (i dimension)
    pid_j = tl.program_id(1)          # block column (j dimension)
    pid_g = tl.program_id(2)          # batch * hidden group

    # derive batch and hidden index from the third dimension
    b = pid_g // H
    h = pid_g % H

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # offsets for the i and j axes (remain constant across k‑tiles)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # bounds checking
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: we need rows = k, cols = j  -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiply of the two blocks, accumulation in fp32
        acc += tl.dot(left, right)     # (M,K) @ (K,N) -> (M,N)

    # write results, respecting i/j bounds
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict of parameter tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (projected)
    device = input_tensor.device

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------------- #
    # 2. Linear projections
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Apply mask (if provided) and gates
    # --------------------------------------------------------------------- #
    if mask is None:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    mask_ = mask.unsqueeze(-1)                     # [B, N, N, 1]
    left = left * mask_
    right = right * mask_

    # gates (sigmoid of linear transforms on the normalized input)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 4. Core O(N³) reduction using Triton
    # --------------------------------------------------------------------- #
    # Cast to bfloat16 to reduce memory bandwidth; accumulation stays fp32
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Allocate output tensor (B, N, N, H) in fp32
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Extract strides (contiguous layout: (B,N,N,H))
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_bf.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_bf.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes – tuned for typical GPU resources
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i blocks
        triton.cdiv(N, BLOCK_N),   # j blocks
        B * H                      # one program per (batch, hidden) pair
    )

    tri_mul_mm_kernel[grid](
        left_bf,
        right_bf,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 5. Post‑processing: outer norm, out‑gate, final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
