def fuse_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,

    mask = offs < total_elements

    cur = offs
    stride_i_h = N * H   # size of a full row i (over k and hidden)
    stride_n_n_h = N * N * H  # size of a full sequence slice (i,k,d)

    rem = cur % stride_n_n_h
    i = rem // stride_i_h

    k = rem2 // H

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing TriMul (AlphaFold3).
    The heavy elementwise gating+mask is fused in a Triton kernel; the
    dense i‑k × j‑k multiplication is performed with a batched GEMM using
    PyTorch (which internally uses cuBLAS). The implementation follows the
    reference PyTorch module but avoids autograd.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm on input
    eps = 1e-5
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Ensure tensors are contiguous for the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask.to(left_proj.dtype).contiguous()

    # Allocate output buffers for gated tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Launch Triton kernel to fuse mask and gates
    total_elements = bs * N * N * hidden_dim
    BLOCK = 65536  # number of elements per program
    grid = (triton.cdiv(total_elements, BLOCK),)
    fuse_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_f,
        left,
        right,
        total_elements,
        B=bs,
        N=N,
        H=hidden_dim,
        BLOCK=BLOCK,
    )

    # Batched GEMM: for each hidden dimension we compute A @ B^T
    # reshape to (bs*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # matmul with transposed right (k dimension shared)
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))
    # reshape back to [bs, N, N, hidden_dim]
    out = out_perm.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Multiply by output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure output dtype is float32
    return out.to(torch.float32)

"""
Triton‑accelerated forward pass of the outgoing TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the input tensor `x` using the provided LN weights.
2. Apply linear projections (left/right) and compute three sigmoid gates
   (left, right, out) from the normalized tensor.
3. Fuse the mask, left‑gate and right‑gate with the projected tensors in a
   custom Triton kernel (`fuse_gate_mask_kernel`).  This kernel works on the
   flattened `[B, N, N, H]` layout and computes
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
   where `mask` is broadcast from `[B, N, N]` to the hidden dimension.
4. Perform the O(N³) i‑k × j‑k multiplication as a batched GEMM:
        out[b, h] = left_out[b, h] @ right_out[b, h].T
   implemented with `torch.bmm` (leveraging cuBLAS).
5. Layer‑normalize the result over the hidden dimension, multiply by the
   output gate, and linearly project back to the original feature dimension.
6. Return the final tensor in `float32`.

Only the elementwise gating/masking step is written in Triton; the heavy
matrix multiply re‑uses PyTorch's highly‑optimized cuBLAS kernel.
"""

import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def fuse_gate_mask_kernel(
    left_ptr,               # *[B,N,N,H] left projection
    right_ptr,              # *[B,N,N,H] right projection
    left_gate_ptr,          # *[B,N,N,H] left gate
    right_gate_ptr,         # *[B,N,N,H] right gate
    mask_ptr,               # *[B,N,N]   mask (broadcasted)
    out_left_ptr,           # *[B,N,N,H] gated left (output)
    out_right_ptr,          # *[B,N,N,H] gated right (output)
    total_elements,         # scalar total number of elements (=B*N*N*H)
    B: tl.constexpr,        # batch size
    N: tl.constexpr,        # sequence length
    H: tl.constexpr,        # hidden dimension
    BLOCK: tl.constexpr,    # elements per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # ---- index de‑composition of the flat offset ----
    # layout: ((b * N + i) * N + k) * H + d
    stride_i_h = N * H                     # N*H
    stride_b_nkh = N * N * H               # size of one batch slice
    b = offs // stride_b_nkh
    rem = offs % stride_b_nkh
    i = rem // stride_i_h
    rem2 = rem % stride_i_h
    k = rem2 // H
    # d = rem2 % H  # not needed for mask

    # compute mask offset = b * (N*N) + i * N + k
    mask_offset = b * (N * N) + i * N + k

    # load values (use mask to avoid OOB reads)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)
    m_val     = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    # fused gating + masking
    out_l = left  * left_g  * m_val
    out_r = right * right_g * m_val

    tl.store(out_left_ptr  + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using a Triton kernel for
    the mask‑gate fusion and PyTorch for the batched matrix multiplication.
    Arguments:
        data = (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (broadcastable)
            - weights: dict of torch.Tensor containing all linear/LN weights
            - config: dict with keys "dim", "hidden_dim", optionally others
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float32)          # ensure fp32 for the linear ops

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # 3) Gates (sigmoid)
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 4) Fuse mask + gates with a Triton kernel
    #    Ensure contiguous layout for the kernel
    left_proj  = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask.to(left_proj.dtype).contiguous()

    # Allocate output buffers
    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_elem = B * N * N * H
    BLOCK = 65536                     # reasonable work per program
    grid = (triton.cdiv(total_elem, BLOCK),)

    fuse_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_f,
        left,
        right,
        total_elem,
        B=B,
        N=N,
        H=H,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5) Batched GEMM: for each hidden channel compute A @ B^T
    #    reshape to (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_perm = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_perm shape [B*H, N, N]
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_perm.reshape(B, H, N, N).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
