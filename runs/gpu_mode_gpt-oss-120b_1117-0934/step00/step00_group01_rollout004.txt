def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # total blocks in M and N
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    # compute block indices
    pid_m = pid // num_n
    pid_n = pid % num_n

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"])
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"])
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"])

    # Convert mask to float and ensure contiguous
    mask_f = mask.to(x.dtype).contiguous()
    # Use Triton kernel to apply mask and gate sigmoids
    # compute left = left_proj * sigmoid(left_gate) * mask_f[..., None]
    # compute right similarly

    # Ensure tensors contiguous
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()

    # Output placeholders
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total = left_proj.numel()
    BLOCK = 65536  # can tune
    grid = (triton.cdiv(total, BLOCK),)

    # Kernel call for left
    gated_mask_kernel[grid](
        left_proj, left_gate, mask_f,
        left,
        total,
        hidden_dim,
        BLOCK=BLOCK
    )
    # Kernel call for right
    gated_mask_kernel[grid](
        right_proj, right_gate, mask_f,
        right,
        total,
        hidden_dim,
        BLOCK=BLOCK
    )
    # shapes are [B,N,N,hidden_dim]
    # Now batched matmul per hidden_dim
    left_t = left.permute(0, 3, 1, 2)   # B, H, N, N
    right_t = right.permute(0, 3, 1, 2) # B, H, N, N
    # compute out = left @ right^T per channel
    out_h = torch.matmul(left_t, right_t.transpose(-2, -1))  # B, H, N, N
    out = out_h.permute(0, 2, 3, 1)  # B, N, N, H

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # out_gate sigmoid (gate after norm)
    out_gate_sig = torch.sigmoid(out_gate)

    # Apply out_gate
    out_norm = out_norm * out_gate_sig

    # final linear

"""TriMul forward (outgoing) with a Triton kernel for gated masking.

The kernel fuses the mask application and gate sigmoid for the left/right
projected tensors:
    out = proj * sigmoid(gate) * mask
where `proj` and `gate` have shape (B, N, N, H) and `mask` has shape
(B, N, N) (broadcast over H).  The heavy pairwise multiplication
(out = Σ_k left[i,k,:] * right[j,k,:]) is performed with an efficient
batched matrix multiplication (torch.matmul) after the gating step.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def gated_mask_kernel(
    proj_ptr,               # *float32   left/right projection
    gate_ptr,               # *float32   pre‑sigmoid gate
    mask_ptr,               # *float32   mask (no last dim)
    out_ptr,                # *float32   output buffer
    total_elements,         # i64        total number of scalar elements in proj/gate
    hidden_dim,             # i64        size of the feature dimension H
    BLOCK: tl.constexpr,    # compile‑time tile size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)          # [BLOCK]
    mask = offs < total_elements                # guard out‑of‑bounds

    # load proj and gate (both contiguous)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid on gate
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # each group of `hidden_dim` consecutive elements shares one mask entry
    mask_index = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_index, mask=mask, other=0.0)

    out = proj * gate_sig * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, C] (float)
            mask: torch.Tensor [B, N, N] (bool or float)
            weights: dict of torch.Tensor containing:
                norm.weight, norm.bias,
                left_proj.weight, right_proj.weight,
                left_gate.weight, right_gate.weight,
                out_gate.weight,
                to_out_norm.weight, to_out_norm.bias,
                to_out.weight
            config: dict with keys "dim" and "hidden_dim"
    Returns:
        Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gates (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(dtype))          # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"].to(dtype))        # [B,N,N,H]

    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype))          # [B,N,N,H] pre‑sigmoid
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype))        # [B,N,N,H] pre‑sigmoid
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype))            # [B,N,N,H] pre‑sigmoid

    # -----------------------------------------------------------------
    # 3) Fuse mask and gate sigmoid via a Triton kernel
    # -----------------------------------------------------------------
    # Ensure tensors are contiguous for the flat indexing used in the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask.to(dtype).contiguous()          # [B,N,N]

    # output buffers
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_elems = left_proj.numel()
    BLOCK = 65536  # a reasonable tile size for the flatten kernel
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left = left_proj * sigmoid(left_gate) * mask
    gated_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_f,
        left,
        total_elems,
        hidden_dim,
        BLOCK=BLOCK,
    )
    # right = right_proj * sigmoid(right_gate) * mask
    gated_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_f,
        right,
        total_elems,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 4) Pairwise multiplication across the k dimension:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is a batched matrix multiplication per hidden channel:
    #        out_h = left_h @ right_h.T
    # -----------------------------------------------------------------
    # reshape to (B, H, N, N) for batched GEMM
    left_t = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2) # [B, H, N, N]

    # batched matmul (uses cuBLAS, highly optimized)
    out_h = torch.matmul(left_t, right_t.transpose(-2, -1))  # [B, H, N, N]

    # back to original layout [B, N, N, H]
    out = out_h.permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, out‑gate and final linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out_gate_sig = torch.sigmoid(out_gate)  # shape [B,N,N,H]
    out_norm = out_norm * out_gate_sig

    # final projection back to `dim`
    output = F.linear(out_norm, weights["to_out.weight"].to(dtype))   # [B,N,N,dim]

    return output.to(torch.float32)
