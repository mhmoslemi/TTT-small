"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The kernel computes the core 4‑D tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where `left` and `right` have already been masked and gated.
We first perform LayerNorm, linear projections and gating in PyTorch,
then launch a Triton kernel that performs a batched matrix multiplication
for every hidden dimension `d`.  The kernel works on FP16 data and accumulates
in FP32 for numerical stability.  The final result is converted back to FP32
and passed through a second LayerNorm, an output gate and a final linear
projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_d, stride_left_i, stride_left_k,
    stride_right_b, stride_right_d, stride_right_i, stride_right_k,
    stride_out_b, stride_out_d, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)   # block row (i)
    pid_j = tl.program_id(1)   # block col (j)
    pid_bd = tl.program_id(2)  # batch * hidden_dim

    # decode batch and hidden dimension from pid_bd
    b = pid_bd // D
    d = pid_bd % D

    # start offsets for this block
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets within the block
    i_offset = i_start + tl.arange(0, BLOCK_M)
    j_offset = j_start + tl.arange(0, BLOCK_N)

    # compute base pointer offsets for the current (b,d)
    off_left = b * stride_left_b + d * stride_left_d
    off_right = b * stride_right_b + d * stride_right_d
    off_out = b * stride_out_b + d * stride_out_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)

        # bounds checks
        i_valid = i_offset < N
        j_valid = j_offset < N
        k_valid = k_offset < N

        # pointers for the left block (M x K)
        a_ptrs = left_ptr + off_left + (i_offset[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=tl.float16(0.0))

        # pointers for the right block (K x N) – note that we treat the K dimension as rows
        b_ptrs = right_ptr + off_right + (k_offset[:, None] * stride_right_k) + (j_offset[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=tl.float16(0.0))

        # matrix multiply + accumulate (convert to FP32)
        acc += tl.dot(a, b).to(tl.float32)

    # store result (FP16)
    mask_store = i_valid[:, None] & j_valid[None, :]
    c_ptrs = out_ptr + off_out + (i_offset[:, None] * stride_out_i) + (j_offset[None, :] * stride_out_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary)
            - weights: dict of model weight tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- 1️⃣ LayerNorm on the input ----------
    x = input_tensor.to(torch.float32)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2️⃣ Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3️⃣ Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------- 4️⃣ Apply mask (if any) ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- 5️⃣ Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- 6️⃣ Prepare tensors for the Triton kernel ----------
    # Cast to FP16 to leverage Tensor‑cores inside the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N) – the layout the kernel expects
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm = torch.empty_like(left_perm)                # FP16 output buffer

    B, D, N, _ = left_perm.shape  # B, hidden_dim, N, N

    # strides (in elements, not bytes)
    stride_left_b, stride_left_d, stride_left_i, stride_left_k = left_perm.stride()
    stride_right_b, stride_right_d, stride_right_i, stride_right_k = right_perm.stride()
    stride_out_b, stride_out_d, stride_out_i, stride_out_j = out_perm.stride()

    # ---------- 7️⃣ Launch Triton kernel ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # blocks over i dimension
        triton.cdiv(N, BLOCK_N),          # blocks over j dimension
        B * D,                            # one program per (batch, hidden_dim)
    )

    trimul_kernel[grid](
        left_perm.data_ptr(),
        right_perm.data_ptr(),
        out_perm.data_ptr(),
        B, N, D,
        stride_left_b, stride_left_d, stride_left_i, stride_left_k,
        stride_right_b, stride_right_d, stride_right_i, stride_right_k,
        stride_out_b, stride_out_d, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- 8️⃣ Post‑kernel processing ----------
    # Back to (B, N, N, hidden_dim) and FP32
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    output = F.linear(out, weights["to_out.weight"])

    return output
