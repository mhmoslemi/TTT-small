@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # compute offsets for hidden block
    offs_d = tl.arange(0, BLOCK_D)
    # compute K loop over N
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    # compute starting pointers:
    # left offset base: left_ptr + b * stride_left_b + i * stride_left_i + k * stride_left_k + d * stride_left_d
    # right offset base: right_ptr + b * stride_right_b + j * stride_right_j + k * stride_right_k + d * stride_right_d

    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left: shape (BLOCK_K, BLOCK_D) but we need to broadcast across block K? We'll load left slices for each k.
        # We can load left block: left_ptr + b*B + i*I + cur_k*K + offs_d*D.
        left_ptrs = left_ptr + pid_b*stride_left_b + pid_i*stride_left_i + cur_k[:,None]*stride_left_k + offs_d[None,:]*stride_left_d
        left_vals = tl.load(left_ptrs, mask=mask_k[:,None] & (offs_d[None,:] < H), other=0.0) # shape (BLOCK_K, BLOCK_D)

        # right: b, j, cur_k, d
        right_ptrs = right_ptr + pid_b*stride_right_b + pid_j*stride_right_j + cur_k[:,None]*stride_right_k + offs_d[None,:]*stride_right_d
        right_vals = tl.load(right_ptrs, mask=mask_k[:,None] & (offs_d[None,:] < H), other=0.0)

        # accumulate: sum over k of left_vals[k]*right_vals[k] per d.
        # This is elementwise multiply then sum across k dimension.
        # left_vals shape (BLOCK_K, BLOCK_D), right_vals same.
        # Compute product then reduce sum over BLOCK_K: we can use tl.dot on each row?? Actually tl.dot expects vectors of shape (K,) and (K,) returning scalar. But we have multiple d.
        # Could compute with tl.sum over k of left_vals*right_vals: elementwise multiplication and then reduce sum across K axis using tl.sum.
        prod = left_vals * right_vals # shape (BLOCK_K, BLOCK_D)
        # sum across k dimension:
        acc += tl.sum(prod, axis=0) # yields (BLOCK_D,)

    # after loop, store acc to out at offsets (b,i,j,d)
    offs_d = tl.arange(0, BLOCK_D)
    d_mask = offs_d < H
    out_ptrs = out_ptr + pid_b*stride_out_b + pid_i*stride_out_i + pid_j*stride_out_j + offs_d*stride_out_d
    tl.store(out_ptrs, acc.to(tl.float16) if ... else acc, mask=d_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # ensure contiguous tensors
    x = input_tensor
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # handle mask
    nominal = config.get("nomask", False)
    if nominal:
        # ignore mask or set to ones
        mask_tensor = torch.ones((B,N,N), dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask.to(x.dtype)
    # convert mask to float32 for multiplication later
    mask_tensor = mask_tensor.to(torch.float32)

    # LayerNorm on x
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5
    )
    # Projections
    # linear expects weight shape [out_features, in_features]
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Fuse mask + gates using Triton kernel
    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_flat = mask_tensor.contiguous().view(-1)  # shape B*N*N

    # allocate
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # launch kernel
    total_elements = left.numel()
    BLOCK = 1024  # adjust if needed, ensure multiple of hidden_dim? Not required.
    grid = (triton.cdiv(total_elements, BLOCK),)
    apply_mask_gate_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_flat.data_ptr(),
        left_gated.data_ptr(),
        right_gated.data_ptr(),
        total_elements,
        hidden_dim,
        BLOCK=BLOCK
    )

    # Now compute out via batched matmul per hidden channel
    # Reshape to (B*H, N, N)
    left_reshape = left_gated.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right_gated.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)
    # Compute out reshape
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1,2))  # (B*H,N,N)
    # Reshape back
    out = out_reshape.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5
    )
    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a lightweight Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension using the
   provided weights (`norm.weight`, `norm.bias`).
2. Linear projections:
   * left  = Linear(x_norm, left_proj.weight)   # → [B,N,N,H]
   * right = Linear(x_norm, right_proj.weight)  # → [B,N,N,H]
3. Gating:
   * left_gate  = sigmoid(Linear(x_norm, left_gate.weight))
   * right_gate = sigmoid(Linear(x_norm, right_gate.weight))
   * out_gate   = sigmoid(Linear(x_norm, out_gate.weight))
4. Fuse mask, left/right projections and gates with a small Triton kernel.
   The kernel computes:
       left_g = left * left_gate * mask
       right_g = right * right_gate * mask
5. Pairwise multiplicative update:
   For each hidden channel `h` the operation is a matrix‑multiply
   `out_h = left_g[...,h] @ right_g[...,h].T`.  This is implemented as a
   batched GEMM (`torch.bmm`) after reshaping.
6. Layer‑normalize the result over the hidden dimension (`to_out_norm`),
   multiply by `out_gate`, and apply the final linear projection (`to_out`).

Only step 4 is performed in Triton; the remaining heavy linear algebra uses
high‑performance PyTorch/CuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,         # const float*  left projection
    right_ptr,        # const float*  right projection
    left_gate_ptr,    # const float*  left gate
    right_gate_ptr,   # const float*  right gate
    mask_ptr,         # const float*  mask (flattened, no hidden dim)
    out_left_ptr,     # float*        output left (masked & gated)
    out_right_ptr,    # float*        output right (masked & gated)
    total_elements,   # int64         total number of elements in tensors
    hidden_dim: tl.constexpr,  # hidden dimension (c)
    BLOCK: tl.constexpr,       # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load data (float32)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # broadcast mask over hidden_dim: each hidden element shares the same mask entry
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val

    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of torch.Tensors with the model parameters
            - config: dict containing at least
                * "dim"         – input/output channel dimension D
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, whether the mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Normalisation of the input (LayerNorm over the last dimension)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Mask handling (optional)
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask_tensor = torch.ones((B, N, N), dtype=torch.float32, device=device)
    else:
        mask_tensor = mask.to(dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 5. Fuse mask, projections and gates with a Triton kernel
    # ------------------------------------------------------------------
    # Ensure all tensors are contiguous for a simple flat view
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_flat = mask_tensor.contiguous().view(-1)   # shape B*N*N

    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_elements = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    _apply_mask_gate_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_flat.data_ptr(),
        left_gated.data_ptr(),
        right_gated.data_ptr(),
        total_elements,
        hidden_dim=H,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6. Pairwise multiplicative update via batched GEMM per hidden channel
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for batched matmul
    left_reshape = left_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshape = right_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_h = left_h @ right_h.T  for each hidden slice
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_reshape.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output normalisation, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
