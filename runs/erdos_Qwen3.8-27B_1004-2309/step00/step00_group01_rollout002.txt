import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5_and_gradient(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    c5 = np.max(corr) * dx
    
    # Find active lags
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    
    # Compute subgradient (average over active lags)
    # c_k = sum_i h[i] * (1-h[i+k])
    # In "full" mode, index k in corr corresponds to lag k - (n-1)
    # corr[m] = sum_i h[i] * (1-h[i + m - (n-1)])
    # Let lag = m - (n-1)
    # c_lag = sum_i h[i] * (1-h[i+lag])
    # ∂c_lag/∂h[j] = (1-h[j+lag]) if j+lag valid - h[j-lag] if j-lag valid
    
    grad = np.zeros(n)
    for m in active:
        lag = m - (n - 1)
        # h[j] appears as h[i] when i=j: contributes (1-h[j+lag]) if 0<=j+lag<n
        # h[j] appears as h[i+lag] when i+lag=j, i=j-lag: contributes -h[j-lag] if 0<=j-lag<n
        # For the first term: j+lag must be in [0, n-1], and i=j must be in [0, n-1]
        # For the second term: j-lag must be in [0, n-1], and i=j-lag must be in [0, n-1]
        # But also i must be in [0, n-1] for the sum to include it
        # The sum in c_lag is over i where both i and i+lag are in [0, n-1]
        
        for j in range(n):
            g = 0.0
            # First term: i = j, need j+lag in [0, n-1] and j in [0, n-1]
            if 0 <= j + lag < n:
                g += (1.0 - h[j + lag])
            # Second term: i + lag = j, so i = j - lag, need j-lag in [0, n-1]
            if 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] += g
    
    grad /= len(active)
    return c5, corr, grad, active
