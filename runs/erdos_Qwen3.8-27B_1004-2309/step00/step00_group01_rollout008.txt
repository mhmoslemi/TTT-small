from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2}"""
    target = n / 2.0
    # Use bisection on water level
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0, 1)
    return proj

def evaluate_h(h):
    """Evaluate C5 for a feasible h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def binary_to_continuous(h_bin, n):
    """Convert binary to continuous by small perturbation"""
    h = h_bin.astype(float).copy()
    # Add small perturbations to 0s and 1s
    zeros = (h_bin == 0)

    h[ones] = np.random.uniform(0.9, 1.0, size=ones.sum())
    # Project to feasible set
    h = project_feasible(h, n)
    return h

import numpy as np
import time

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project onto {x in [0,1]^n : sum(x) = n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gen_binary(n, rng):
    """Generate random binary (or near-binary) with sum = n/2"""
    h = np.zeros(n)
    if n % 2 == 0:
        pos = rng.choice(n, n // 2, replace=False)
        h[pos] = 1.0
    else:
        k = (n - 1) // 2
        pos = rng.choice(n, k + 1, replace=False)
        h[pos[:k]] = 1.0
        h[pos[k]] = 0.5
    return h

def pair_swap_search(h, n, max_iters, rng):
    """Local search by swapping pairs of coordinates"""
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    h = h.copy()
    for _ in range(max_iters):
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def subgrad_refine(h, n, max_iters, lr0, rng):
    """Projected subgradient descent on the max correlation"""
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    h = h.copy()
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        t = max_idx - (n - 1)
        
        # Gradient of R(t) w.r.t. h
        grad = np.zeros(n)
        if t >= 0:
            if n - t > 0:
                grad[:n-t] += (1.0 - h[t:n])
                grad[t:] += -h[:n-t]
        else:
            s = -t
            if n - s > 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += -h[s:]
        
        lr = lr0 / (1.0 + 0.01 * it)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = evaluate_h(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_refine(h0, n, max_iter=300, alpha=30.0):
    """Scipy SLSQP with smooth max objective"""
    from scipy.optimize import minimize
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        # Smooth max: (1/alpha)*log(sum(exp(alpha*(corr-mc)))) + mc
        return (mc + np.log(np.sum(np.exp(alpha * (corr - mc)))) / alpha) * (2.0 / n)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        weights = np.exp(alpha * (corr - mc))
        weights /= np.sum(weights)
        # Gradient of smooth max w.r.t. x
        grad = np.zeros_like(x)
        for k in range(len(corr)):
            t = k - (n - 1)
            g = np.zeros(n)
            if t >= 0:
                if n - t > 0:
                    g[:n-t] += (1.0 - x[t:n])
                    g[t:] += -x[:n-t]
            else:
                s = -t
                if n - s > 0:
                    g[s:] += (1.0 - x[:n-s])
                    g[:n-s] += -x[s:]
            grad += weights[k] * g
        return grad * (2.0 / n)
    
    bounds = [(0, 1)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success:
            return res.x, evaluate_h(res.x)
        else:
            # Still try the result
            return res.x, evaluate_h(res.x)
    except:
        return h0, evaluate_h(h0)

                best_h = h1.copy()
