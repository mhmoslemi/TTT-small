    The new g_k = g_k + d * (a_k - b_k)

  For s < 0 (k < n-1): always true
  
  Combined with j <= n-1 (always):
  If k >= n-1: n-1 <= k <= j+n-1
  If k < n-1: 0 <= k <= n-2
  
  So for k < n-1: all k from 0 to n-2 contribute (h[j] as second factor)
  For k >= n-1: k from n-1 to j+n-1 contribute

Let me just implement the O(n) loop for computing the slopes. It's clean and correct:

```python
def compute_effect(h, j, j2):
    """
    Returns the vector v where v[k] = coefficient of delta in the change to
    the k-th correlation value when h[j] += delta, h[j2] -= delta.
    """
    n = len(h)
    n_full = 2 * n - 1
    v = np.zeros(n_full)
    
    for k in range(n_full):
        s = k - (n - 1)  # actual lag
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        
        # h[j] as first factor: j in [lo, hi]
        if lo <= j <= hi:
            v[k] += (1.0 - h[j + s])
        # h[j] as second factor: j-s in [lo, hi]
        if lo <= j - s <= hi:
            v[k] -= h[j - s]
        
        # h[j2] as first factor (with negative delta)
        if lo <= j2 <= hi:
            v[k] -= (1.0 - h[j2 + s])
        # h[j2] as second factor (with negative delta)
        if lo <= j2 - s <= hi:
            v[k] += h[j2 - s]
    
    return v
```

This is O(n) per pair (since the loop is over n_full = 2n-1 values, each O(1)). For n=47, that's 93 operations per pair.

- Minimize max(g + delta * v) over delta in [d_min, d_max]

def optimize_pair_swap(h, n, seed, time_budget=300):
    """
    Coordinate descent using pair swaps with exact line search.
    """
    rng = np.random.RandomState(seed)
    h = h.copy()
    g = np.correlate(h, 1.0 - h, mode="full")
    current_max = np.max(g)
    start = time.time()
    
    n_full = 2 * n - 1
    no_improve_count = 0
    max_no_improve = 5000
    
    while time.time() - start < time_budget and no_improve_count < max_no_improve:
        j = rng.randint(n)
        j2 = rng.randint(n)
        if j == j2:
            continue
        
        # Compute effect vector
        v = np.zeros(n_full)
        for k in range(n_full):
            s = k - (n - 1)
            lo = max(0, -s)
            hi = min(n - 1, n - 1 - s)
            if lo <= j <= hi:
                v[k] += (1.0 - h[j + s])
            if lo <= j - s <= hi:
                v[k] -= h[j - s]
            if lo <= j2 <= hi:
                v[k] -= (1.0 - h[j2 + s])
            if lo <= j2 - s <= hi:
                v[k] += h[j2 - s]
        
        # Determine feasible range for delta
        d_min = max(-h[j], -(1.0 - h[j2]))
        d_max = min(1.0 - h[j], h[j2])
        
        if d_max - d_min < 1e-12:
            continue
        
        # Minimize max(g + delta * v) over delta in [d_min, d_max]
        # Using ternary search (convex function)
        lo_d, hi_d = d_min, d_max
        for _ in range(40):
            m1 = lo_d + (hi_d - lo_d) / 3.0
            m2 = hi_d - (hi_d - lo_d) / 3.0
            f1 = np.max(g + m1 * v)
            f2 = np.max(g + m2 * v)
            if f1 <= f2:
                hi_d = m2
            else:
                lo_d = m1
        
        best_delta = (lo_d + hi_d) / 2.0
        new_max = np.max(g + best_delta * v)
        
        if new_max < current_max - 1e-12:
            # Apply the update
            h[j] += best_delta
            h[j2] -= best_delta
            g = g + best_delta * v
            current_max = new_max
            no_improve_count = 0
        else:
            no_improve_count += 1
    
    return h

def optimize_slsqp(h, n, time_budget=60):
    """Optimize using SLSQP with smooth objective"""
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def obj(h):
        r = np.correlate(h, 1.0 - h, mode="full") * dx
        T = 0.01
        m = np.max(r)
        return m + T * np.log(np.sum(np.exp((r - m) / T)))
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    
    result = minimize(obj, h.copy(), method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': 500, 'ftol': 1e-10})
    
    if result.fun < np.inf:
        h_new = np.clip(result.x, 0, 1)
        # Fix sum
        diff = target_sum - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        return h_new
    return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * dx)

def _compute_v(h, j, j2, n):
    n_full = 2 * n - 1
    v = np.zeros(n_full)
    k_lo = max(0, n - 1 - j)
    k_hi = min(n_full - 1, 2 * n - 2 - j)
    if k_lo <= k_hi:
        s = np.arange(k_lo, k_hi + 1) - (n - 1)
        v[k_lo:k_hi + 1] += 1.0 - h[j + s]
    if j > 0:
        k2 = np.arange(n - 1, min(n_full, n + j))
        s2 = k2 - (n - 1)
        v[k2] -= h[j - s2]
    m2 = max(0, n - 1 - j2)
    m3 = min(n_full - 1, 2 * n - 2 - j2)
    if m2 <= m3:
        sm = np.arange(m2, m3 + 1) - (n - 1)
        v[m2:m3 + 1] -= 1.0 - h[j2 + sm]
    if j2 > 0:
        k4 = np.arange(n - 1, min(n_full, n + j2))
        s4 = k4 - (n - 1)
        v[k4] += h[j2 - s4]
    return v

def pair_swap_optimize(h0, n, seed, time_budget):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    g = np.correlate(h, 1.0 - h, mode="full").copy()
    cur_max = float(np.max(g))
    start = time.time()
    stuck = 0
    max_stuck = 200000
    while time.time() - start < time_budget and stuck < max_stuck:
        j = rng.randint(n)
        j2 = rng.randint(n)
        if j == j2:
            continue
        v = _compute_v(h, j, j2, n)
        if not np.any(v):
            continue
        d_lo = max(-h[j], h[j2] - 1.0)
        d_hi = min(1.0 - h[j], h[j2])
        if d_hi - d_lo < 1e-12:
            continue
        a, b = d_lo, d_hi
        for _ in range(50):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            if np.max(g + m1 * v) <= np.max(g + m2 * v):
                b = m2
            else:
                a = m1
        d_star = (a + b) / 2.0
        new_max = float(np.max(g + d_star * v))
        if new_max < cur_max - 1e-14:
            h[j] += d_star
            h[j2] -= d_star
            g += d_star * v
            cur_max = new_max
            stuck = 0
        else:
            stuck += 1
    return h

def slsqp_optimize(h0, n, time_budget):
    target = n / 2.0
    dx = 2.0 / n
    T = 0.01
    def obj(h):
        r = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(r)
        return m + T * np.log(np.sum(np.exp((r - m) / T)))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target}]
    res = minimize(obj, h0.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 1000, 'ftol': 1e-12})
    h_new = np.clip(res.x, 0, 1)
    h_new += (target - np.sum(h_new)) / n
    return np.clip(h_new, 0, 1)

def interpolate_init(h_base, target_n):
    n = len(h_base)
    x_old = np.arange(n) / n
    x_new = np.arange(target_n) / target_n
    h_new = np.interp(x_new, x_old, np.append(h_base, h_base[0]))
    h_new = np.clip(h_new, 0, 1)
    target = target_n / 2.0
    h_new += (target - np.sum(h_new)) / target_n
    return np.clip(h_new, 0, 1)
