def soft_max_and_grad(h, temp):
    """Returns (soft_max_value, gradient) of the soft-max of c[k]."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    m = c.max()
    obj = m + temp * np.log(np.sum(np.exp((c - m) / temp)))
    
    weights = np.exp((c - m) / temp)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = weights[i]
        if wi < 1e-12:
            continue
        lag = i - n + 1
        w = wi * dx
        for p in range(n):
            g = 0.0
            if 0 <= p - lag < n:
                g += (1.0 - h[p - lag])
            if 0 <= p + lag < n:
                g -= h[p + lag]
            grad[p] += w * g
    
    return obj, grad
