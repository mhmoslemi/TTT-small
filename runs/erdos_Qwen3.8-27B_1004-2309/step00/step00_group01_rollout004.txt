    x_max = np.max(x)

    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    cv_max = np.max(cv)
    sm = cv_max + np.log(np.sum(np.exp(T * (cv - cv_max)))) / T
    pen = penalty * (np.sum(x) - n * 0.5) ** 2
    return sm + pen


def obj_max(x, n, penalty):
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    pen = penalty * (np.sum(x) - n * 0.5) ** 2
    return float(np.max(cv)) + pen


def fix_sum(x, n):
    x = np.clip(x, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-14:
                    break
                add = min(1.0 - x[i], diff)
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if -diff <= 1e-14:
                    break
                sub = min(x[i], -diff)
                x[i] -= sub
                diff += sub
    return x


def optimize_h(x0, n, maxiter, T, penalty, use_max=False):
    bounds = [(0.0, 1.0)] * n
    if use_max:
        res = minimize(obj_max, x0, method='L-BFGS-B', bounds=bounds,
                       args=(n, penalty),
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
    else:
        res = minimize(obj_pen, x0, method='L-BFGS-B', bounds=bounds,
                       args=(n, T, penalty),
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
    return fix_sum(res.x, n)


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return fix_sum(h_new, n_new)


def binary_refine(h, n):
    """Try pushing toward binary and re-checking."""
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    s = np.sum(h_bin)
    if s > n * 0.5:
        excess = int(s - n * 0.5)
        order = np.argsort(h_bin - h)
        for i in range(min(excess, n)):
            h_bin[order[i]] = 0.0
    elif s < n * 0.5:
        deficit = int(n * 0.5 - s)
        order = np.argsort(h - h_bin)
        for i in range(min(deficit, n)):
            h_bin[order[i]] = 1.0
    if abs(np.sum(h_bin) - n * 0.5) < 0.01:
        c5_bin = compute_c5(h_bin)
        if c5_bin < compute_c5(h):
            return h_bin
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(47, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)

    # Phase 1: Optimize initial with penalty annealing
    anneal_params = [(10.0, 1e3, 400), (30.0, 1e4, 350), (100.0, 5e4, 300),
                     (300.0, 1e5, 250), (500.0, 5e5, 200), (800.0, 1e6, 150)]
    for T_pen, pen_val, mit in anneal_params:
        if time.time() - t0 > budget_s * 0.3:
            break
        h_opt = optimize_h(best_h, n0, mit, T_pen, pen_val)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Try direct max objective
    if time.time() - t0 < budget_s * 0.35:
        h_opt = optimize_h(best_h, n0, 200, 0, 1e7, use_max=True)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Binary refinement
    if time.time() - t0 < budget_s * 0.4:
        h_bin = binary_refine(best_h, n0)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
            h_opt = optimize_h(best_h, n0, 300, 100.0, 1e6)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    # Phase 2: Try different n
    for n_try in [32, 40, 48, 56, 64, 80, 96, 128, 160, 200, 256]:
        if time.time() - t0 > budget_s * 0.72:
            break
        h_interp = interpolate_h(best_h, n_try)
        for T_pen, pen_val, mit in [(30.0, 1e4, 300), (100.0, 1e5, 250), (400.0, 1e6, 150)]:
            if time.time() - t0 > budget_s * 0.72:
                break
            h_opt = optimize_h(h_interp, n_try, mit, T_pen, pen_val)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
                h_interp = best_h.copy()

    # Phase 3: Perturbation / multistart
    n_best = len(best_h)
    for i in range(60):
        if time.time() - t0 > budget_s * 0.95:
            break
        scale = 0.03 + 0.25 * rng.rand()
        h_pert = best_h + rng.randn(n_best) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_opt = optimize_h(h_pert, n_best, 200, 100.0, 1e6)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        # Also try larger perturbations occasionally
        if i % 5 == 4:
            h_pert2 = best_h + rng.randn(n_best) * 0.5
            h_pert2 = np.clip(h_pert2, 0.0, 1.0)
            h_opt2 = optimize_h(h_pert2, n_best, 200, 50.0, 1e4)
            c5_2 = compute_c5(h_opt2)
            if c5_2 < best_c5:
                best_h, best_c5 = h_opt2, c5_2

    # Final: one more binary refinement + re-optimize
    if time.time() - t0 < budget_s * 0.97:
        h_bin = binary_refine(best_h, n_best)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
        h_final = optimize_h(best_h, n_best, 200, 500.0, 1e7)
        c5_f = compute_c5(h_final)
        if c5_f < best_c5:
            best_h, best_c5 = h_final, c5_f

    n_fin = len(best_h)
    best_h = fix_sum(best_h, n_fin)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_fin
