import numpy as np
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    exp_g = np.exp(beta * (g - g.max()))
    p = exp_g / exp_g.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        s = k - (n - 1)
        if s == 0:
            c = 1.0 - 2.0 * h
        elif s > 0:
            c = np.zeros(n)
            c[s:] += (1.0 - h[:n - s])
            c[:n - s] -= h[s:]
        else:
            t = -s
            c = np.zeros(n)
            c[:n - t] += (1.0 - h[t:])
            c[t:] -= h[:n - t]
        grad += pk * c
    return dx * grad

def optimize(h0, dx, target_sum, time_limit=30.0, beta_start=5.0, beta_end=200.0):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_f = evaluate_c5(h)
    t0 = time.time()
    lr = 3.0
    no_improve = 0

    while time.time() - t0 < time_limit:
        if no_improve > 400:
            break
        frac = min((time.time() - t0) / time_limit, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        grad = smooth_max_grad(h, beta, dx)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break

        f_curr = evaluate_c5(h)
        step = lr
        accepted = False
        for _ in range(30):
            h_new = project_feasible(h - step * grad, target_sum)
            f_new = evaluate_c5(h_new)
            if f_new < f_curr - 1e-13:
                h = h_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                accepted = True
                break
            step *= 0.5
        if not accepted:
            no_improve += 10
            lr = max(step, 1e-10)
            if lr < 1e-9:
                h = project_feasible(h + 0.1 * (np.random.rand(n) - 0.5), target_sum)
                lr = 2.0
                no_improve = 0
        else:
            if step >= lr * 0.9:
                lr = min(lr * 1.2, 30.0)
            else:
                lr = max(step * 1.5, 0.1)

    return best_h, best_f
