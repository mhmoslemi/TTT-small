import time

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * (2.0 / n)

def project_onto_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Water-filling
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining
        free = (h_proj > 0) & (h_proj < 1)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

            best_h = h.copy()
            patience = 0
            # Decrease step size on improvement
            alpha *= 0.95
        else:
            patience += 1
            if patience > max_patience:

            # Contribution from h[j] as the first factor: h[j] * (1 - h[j+lag])
            if 0 <= j + lag < n:

            # Contribution from h[j] as the second factor: h[j-lag] * (1 - h[j])

            # Phase 4: Small perturbations + re-optimize
            if time.time() - t0 < budget_s - 40:
                for _ in range(3):
                    h_p = perturb(h, n, rng, frac=0.08)
                    h_p = local_search(h_p, n, max_rounds=10)
                    if compute_c5(h_p) < compute_c5(h):
                        h = h_p

            try_update(h)

    return (best_h, best_c5, len(best_h))
