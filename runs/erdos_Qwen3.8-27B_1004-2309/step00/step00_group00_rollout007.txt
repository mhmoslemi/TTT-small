import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_grad_at_lag(h, s, n):
    """Compute gradient of g_s(h) * dx with respect to h[i]"""
    # g_s(h) = sum_j h[j]*(1-h[j+s]) for valid j
    # where s can be negative
    # full_corr index = s + (n-1)
    # g_s(h) = sum over j where 0<=j<n and 0<=j+s<n of h[j]*(1-h[j+s])
    # dg/dh[i] = sum over j where (i = j and 0<=j+s<n) of (1-h[j+s])
    #            + sum over j where (i = j+s and 0<=j<n) of (-h[j])
    # = (1 - h[i+s]) * [valid] - h[i-s] * [valid]
    
    grad = np.zeros(n)
    for i in range(n):
        # First term: h[i] appears as h[j] with j=i, valid if 0 <= i+s < n
        if 0 <= i + s < n:
            grad[i] += (1.0 - h[i + s])
        # Second term: h[i] appears as h[j+s] with j+s=i, j=i-s, valid if 0 <= i-s < n
        if 0 <= i - s < n:
            grad[i] -= h[i - s]
    return grad * (2.0 / n)  # multiply by dx

def solve_lp_linearization(h, n, max_iters=200):
    """Iteratively solve LP to minimize max_k g_k(h)"""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for it in range(max_iters):
        # Compute current overlaps
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        max_overlap = np.max(overlaps)
        
        if max_overlap < best_c5 - 1e-10:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Linearize all lags at current h
        # Full correlation has 2n-1 lags
        # Lag s (shift) corresponds to full_corr index s + (n-1), s = -(n-1), ..., n-1
        c = cp.Variable(n)
        t = cp.Variable()
        constraints = [c >= 0, c <= 1, cp.sum(c) == n / 2.0]
        
        for lag_idx in range(2 * n - 1):
            s = lag_idx - (n - 1)  # shift s
            # Linearize g_s at h:
            # g_s(h) ≈ g_s(h) + grad_s(h) · (c - h)
            # = g_s(h) + sum_i grad_s[i]*(c[i] - h[i])
            # = (g_s(h) - sum_i grad_s[i]*h[i]) + sum_i grad_s[i]*c[i]
            # constant + sum_i grad_s[i]*c[i]
            
            # Compute gradient
            grad = np.zeros(n)
            for i in range(n):
                if 0 <= i + s < n:
                    grad[i] += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    grad[i] -= h[i - s]
            grad *= dx
            
            g_val = corr[lag_idx] * dx  # current value of g_s
            const = g_val - np.dot(grad, h)
            
            constraints.append(t >= const + grad @ c)
        
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                break
        
        if c.value is None:
            break
        
        new_h = np.clip(c.value, 0, 1)
        # Fix sum constraint
        target = n / 2.0
        current_sum = new_h.sum()
        if abs(current_sum - target) > 1e-6:

        h_new = new_h
        if np.allclose(h_new, h, atol=1e-8):

        h = h_new
    
    return best_h, best_c5

import numpy as np
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_at_lag(h, s, n):
    grad = np.zeros(n)
    i0 = max(0, -s)
    i1 = min(n, n - s)
    if i1 > i0:
        grad[i0:i1] += (1.0 - h[i0 + s:i1 + s])
    i2 = max(0, s)
    i3 = min(n, n + s)
    if i3 > i2:
        grad[i2:i3] -= h[i2 - s:i3 - s]
    return grad

def lp_linearize(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    c = cp.Variable(n)
    t = cp.Variable()
    constraints = [c >= 0, c <= 1, cp.sum(c) == n / 2.0]
    for lag_idx in range(2 * n - 1):
        s = lag_idx - (n - 1)
        g = grad_at_lag(h, s, n) * dx
        g_val = overlaps[lag_idx]
        const = g_val - np.dot(g, h)
        constraints.append(t >= const + g @ c)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None
    if c.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.clip(c.value, 0, 1)

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        lo, hi = -1.0, 1.0

            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0, 1)
    return h

def optimize_lp(h, n, time_limit):

    best_h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(best_h)
    start = time.time()
    stale = 0
    current = best_h.copy()
    for it in range(80000):
        if time.time() - start > time_limit:

        new_h = lp_linearize(current, n, dx)
        if new_h is None:

        new_h = project_feasible(new_h, n)
        if np.allclose(new_h, current, atol=1e-10):

        c5_new = compute_c5(new_h)

            best_c5 = c5_new
            best_h = new_h.copy()
            current = new_h.copy()
            stale = 0
        else:

                    current = trial.copy()
                    improved = True

            if not improved:
