from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return np.max(corr * dx)

def compute_c5_smooth(h, alpha=100.0):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full") * dx
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    return max_corr + np.log(np.sum(np.exp(corr - max_corr))) / alpha

def repair_h(h, n_points):
    target_sum = n_points / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        h[rng.randint(n)] -= 0.5
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, UnboundLocalError):
        n0 = 45
        h0 = np.ones(n0) * 0.5

    best = repair(h0.copy(), n0)
    best_c5 = compute_c5(best)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Local optimization from initial at n0
    for alpha in [30, 80, 200, 500]:
        if time_left() < 40:
            break
        obj = SmoothObj(n0, alpha, 500.0)
        h_new = try_lbfgsb(best, n0, obj, 4000)
        c5_v = compute_c5(h_new)
        if c5_v < best_c5:
            best_c5 = c5_v
            best = h_new

    # Phase 2: Binary-like restarts at n0
    for i in range(15):
        if time_left() < 30:
            break
        h_bin = make_binary_init(n0, rng)
        obj = SmoothObj(n0, 200, 500.0)
        h_new = try_lbfgsb(h_bin, n0, obj, 2000)
        c5_v = compute_c5(h_new)
        if c5_v < best_c5:
            best_c5 = c5_v
            best = h_new

    # Phase 3: Try larger n via interpolation
    for n_try in [60, 80, 100, 120]:
        if time_left() < 80:
            break
        x_old = np.linspace(0, 2, len(best))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best)
        h_interp = repair(h_interp, n_try)
        for alpha in [100, 400]:
            if time_left() < 30:
                break
            obj = SmoothObj(n_try, alpha, 500.0)
            h_new = try_lbfgsb(h_interp, n_try, obj, 2000)
            c5_v = compute_c5(h_new)
            if c5_v < best_c5:
                best_c5 = c5_v
                best = h_new

    # Phase 4: Random perturbations + local opt at current best n
    n_cur = len(best)
    for i in range(50):
        if time_left() < 20:
            break
        scale = 0.08 + 0.15 * (i / 50.0)
        h_pert = best + rng.normal(0, scale, n_cur)
        h_pert = repair(h_pert, n_cur)
        obj = SmoothObj(n_cur, 200, 500.0)
        h_new = try_lbfgsb(h_pert, n_cur, obj, 1500)
        c5_v = compute_c5(h_new)
        if c5_v < best_c5:
            best_c5 = c5_v
            best = h_new

    # Phase 5: Differential evolution (coarse global)
    if time_left() > 150:
        n_de = min(50, n_cur)
        obj_de = SmoothObj(n_de, 80, 200.0)
        bnds_de = [(0.0, 1.0)] * n_de
        try:
            res = differential_evolution(obj_de, bnds_de, seed=seed,
                                         maxiter=50, popsize=12,
                                         tol=1e-5, mutation=(0.4, 1.2),
                                         recombination=0.85, polish=False)
            h_de = repair(res.x, n_de)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best = h_de
        except Exception:
            pass

    # Phase 6: Nelder-Mead polish on best
    if time_left() > 30:
        n_f = len(best)
        obj_nm = DirectObj(n_f, 500.0)
        h_nm = try_nm(best, n_f, obj_nm, 5000)
        c5_nm = compute_c5(h_nm)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best = h_nm

    # Phase 7: Final high-alpha refinement
    n_f = len(best)
    for alpha in [500, 1000, 2000, 5000]:
        if time_left() < 10:
            break
        obj = SmoothObj(n_f, alpha, 2000.0)
        h_new = try_lbfgsb(best, n_f, obj, 5000)
        c5_v = compute_c5(h_new)
        if c5_v < best_c5:
            best_c5 = c5_v
            best = h_new

    # Final repair and evaluation
    best = repair(best, len(best))
    best_c5 = compute_c5(best)
    return best.tolist(), best_c5, len(best)
