def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def make_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
    else:
        idx = np.argsort(h)
    for i in idx:
        if diff > 0:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
        else:
            a = min(-diff, h[i])
            h[i] -= a
            diff += a
        if abs(diff) < 1e-12:
            break
    return h

def upsample(h, new_n):
    return np.interp(np.arange(new_n, dtype=float) / new_n,
                     np.arange(len(h), dtype=float) / len(h), h)
