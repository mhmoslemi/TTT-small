def project_onto_simplex_box(x, n):
    """Project x onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    
    def _sum_for_lambda(lam):
        return np.sum(np.clip(x + lam, 0, 1))
    
    # Bisection
    lo, hi = -np.max(x) - 1, np.max(1 - x) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        if _sum_for_lambda(mid) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2, 0, 1)
