    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y + mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    return result

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, full) * dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_for_lag(h, k, n):
    """Gradient of correlation at lag k w.r.t. h."""
    grad = np.zeros(n)
    if k >= 0:
        # corr_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j] = (1-h[j-k])*[j>=k] - h[j+k]*[j<=n-1-k]
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        kp = -k
        # corr = sum_{i=0}^{n-1-kp} h[i]*(1-h[i+kp])
        # d/dh[j] = (1-h[j+kp])*[j<=n-1-kp] - h[j-kp]*[j>=kp]
        grad[:n-kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n-kp]
    return grad

def gradient_optimize(h, n, duration_s, lr0=0.08, momentum=0.5):
    """Gradient descent to minimize max correlation."""
    start = time.time()
    h = project_feasible(h, n).astype(float).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = lr0
    
    while time.time() - start < duration_s:
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Use top-5 lags for smoother gradient
        top = np.argsort(corr)[::-1][:5]
        grad = np.zeros(n)
        for rank, idx in enumerate(top):
            weight = 1.0 / (rank + 1.0)
            k = idx - (n - 1)
            grad += weight * grad_for_lag(h, k, n)
        # Normalize
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad /= grad_norm
        
        vel = momentum * vel - lr * grad
        h_new = project_feasible(h + vel, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
            lr = min(lr * 1.005, 0.5)
        else:
            lr *= 0.995
            if lr < 0.005:
                h = best + 0.15 * np.random.randn(n)
                h = project_feasible(h, n)
                vel = np.zeros(n)
                lr = lr0
    
    return best, best_c5
