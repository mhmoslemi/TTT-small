    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    best_h = None
    best_c5 = 1e9

    def elapsed():
        return time.time() - t0

    # Use initial if available
    init = None
    try:
        init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if init is not None and len(init) > 0:
        init = project_feasible(init, len(init))
        c5 = compute_c5(init)
        if c5 < best_c5:
            best_c5 = c5
            best_h = init.copy()

    sizes = [44, 46, 48, 50, 52, 56, 60, 64]

    for n in sizes:
        if elapsed() > budget_s - 40:
            break
        for attempt in range(2):
            if elapsed() > budget_s - 30:
                break
            rng = np.random.RandomState(seed + n * 10 + attempt)
            h0 = make_binary_start(n, rng)

            # Binary SA
            t_alloc = min(35, (budget_s - 30) * 0.3)
            t_end = time.time() + t_alloc
            hb, cb = binary_search_sa(h0, seed + n * 7 + attempt * 13, t_end)
            hb = project_feasible(hb, n)
            cb = compute_c5(hb)
            if cb < best_c5:
                best_c5 = cb
                best_h = hb.copy()

            # Continuous refinement
            if elapsed() > budget_s - 25:
                break
            t_alloc2 = min(25, (budget_s - 25) * 0.35)
            t_end2 = time.time() + t_alloc2
            hc, cc = continuous_opt(hb, seed + 5000 + n * 3 + attempt, t_end2)
            hc = project_feasible(hc, n)
            cc = compute_c5(hc)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

    # If we have an initial, also refine it continuously
    if init is not None and len(init) > 0:
        if elapsed() > budget_s - 30:
            pass
        else:
            t_end = time.time() + min(30, budget_s - 30 - elapsed())
            hc, cc = continuous_opt(init, seed + 9999, t_end)
            hc = project_feasible(hc, len(hc))
            cc = compute_c5(hc)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

    # Final refinement of best solution
    if best_h is not None:
        t_rem = budget_s - elapsed() - 2
        if t_rem > 8:
            hr, cr = continuous_opt(best_h, seed + 77777, time.time() + t_rem)
            hr = project_feasible(hr, len(hr))
            cr = compute_c5(hr)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()

    if best_h is None:
        n = 50
        best_h = make_binary_start(n, np.random.RandomState(seed))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
