                candidate = project_feasible(result.x, _TARGET_SUM)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_h = candidate
                    best_c5 = c5_val
            except Exception:
                pass

    # Phase 3: Projected subgradient with adaptive step
    elapsed = time.time() - start_time
    if elapsed < budget_s - 15:
        h = best_h.copy()
        lr = 0.003
        stale = 0
        for it in range(300000):
            if time.time() - start_time > budget_s - 10:
                break
            if it > 0 and it % 500 == 0:
                c5_cur = compute_c5(h)
                if c5_cur < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_cur
                    stale = 0
                else:
                    stale += 1
                if stale > 20:
                    lr = max(0.00005, lr * 0.7)
                if stale > 80:
                    break

            lags = c5_lags(h)
            # Average subgradient over top-k active lags
            k_top = min(5, len(lags))
            top_idx = np.argsort(lags)[-k_top:]
            grad = np.zeros(n)
            for idx in top_idx:
                lag = int(idx) - (n - 1)
                grad += subgrad(h, lag)
            grad /= k_top

            h = h - lr * grad
            h = project_feasible(h, _TARGET_SUM)

    # Final: ensure exact feasibility
    best_h = project_feasible(best_h, _TARGET_SUM)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
