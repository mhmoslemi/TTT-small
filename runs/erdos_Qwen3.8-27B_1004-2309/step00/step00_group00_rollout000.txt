def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = np.sum(room)
        if total < diff:
            h = 1.0 - np.zeros_like(h)
            deficit = diff - total
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx] = 1.0
        else:
            h = h + (diff / total) * room
    else:
        room = h
        total = np.sum(room)
        if total < -diff:
            h = np.zeros_like(h)
        else:
            h = h + (diff / total) * room
    return np.clip(h, 0.0, 1.0)
