        T_sm = 0.02

        for _ in range(500000):
            if time.time() - start_time > budget_s - 15:
                break
            ov = get_overlaps(h_cur)
            c5 = float(np.max(ov))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()

            _, w = smooth_max_weights(ov, T_sm)
            g = weighted_grad(h_cur, w)
            vel = mu * vel + lr * g
            h_new = project_feasible(h_cur - vel, target_sum)
            c5_new = float(np.max(get_overlaps(h_new)))
            if c5_new <= c5 + 1e-10:
                h_cur = h_new
                lr = min(lr * 1.01, 0.05)
            else:
                lr = max(lr * 0.9, 0.0003)
                vel *= 0.5

    # Phase 3: Perturbation restarts
    time_remaining = budget_s - (time.time() - start_time) - 10
    if time_remaining > 40:
        n_tries = min(6, int(time_remaining // 60))
        for trial in range(n_tries):
            if time.time() - start_time > budget_s - 15:
                break
            scale = 0.04 / (1 + trial * 0.3)
            h_pert = project_feasible(best_h + scale * rng.standard_normal(n), target_sum)
            lr_p = 0.003
            vel_p = np.zeros(n)
            for _ in range(300000):
                if time.time() - start_time > budget_s - 12:
                    break
                ov = get_overlaps(h_pert)
                c5 = float(np.max(ov))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_pert.copy()
                _, w = smooth_max_weights(ov, 0.02)
                g = weighted_grad(h_pert, w)
                vel_p = 0.8 * vel_p + lr_p * g
                h_new = project_feasible(h_pert - vel_p, target_sum)
                c5_new = float(np.max(get_overlaps(h_new)))
                if c5_new <= c5 + 1e-10:
                    h_pert = h_new
                    lr_p = min(lr_p * 1.01, 0.03)
                else:
                    lr_p = max(lr_p * 0.9, 0.0003)
                    vel_p *= 0.5

    # Final feasibility check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target_sum)
    best_c5 = float(np.max(get_overlaps(best_h)))

    return best_h, best_c5, n
