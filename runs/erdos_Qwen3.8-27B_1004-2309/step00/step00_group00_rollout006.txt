    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx

    grad = np.zeros(n)
    for c in range(2 * n - 1):
        if weights[c] < 1e-12:
            continue
        shift = c - (n - 1)
        w = weights[c] * dx
        
        # (1-h) shifted by -shift
        # result[j] = 1-h[j-shift] for 0 <= j-shift < n
        lo = max(0, shift)
        hi = min(n, n + shift)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[lo - shift:hi - shift])
        
        # -h shifted by shift
        # result[j] = -h[j+shift] for 0 <= j+shift < n
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:

    return smax, grad

For binary local search, the idea is:

- Try swapping: change h[i] from 0 to 1 and h[j] from 1 to 0 (or vice versa)
- Accept if the max overlap decreases
- This is a 2-bit flip local search

The overlap computation for binary h is fast:
overlap(s) = number of i such that h[i+s]=1 and h[i]=0 (for the linear correlation)

Actually, for binary h:
h[i]*(1-h[i+s]) = h[i] * (1-h[i+s]) which is 1 iff h[i]=1 and h[i+s]=0
1-h[i] = 1 iff h[i]=0
So h[i+s]*(1-h[i]) = 1 iff h[i+s]=1 and h[i]=0

So overlap(s) = |{i: 0≤i<n, 0≤i+s<n, h[i+s]=1, h[i]=0}| = count of "1 followed by 0 at distance s" (in the appropriate direction).

Alternatively: overlap(s) = |{i: h[i+s]=1, h[i]=0}| = (number of 1s in positions where i+s is valid) - (number of positions where both h[i]=1 and h[i+s]=1)

import numpy as np
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(overlaps))

def project_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        if float(np.sum(np.clip(h - lo, 0.0, 1.0))) < target:
            break
        lo -= 1.0
    for _ in range(50):
        if float(np.sum(np.clip(h - hi, 0.0, 1.0))) > target:
            break
        hi += 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(h - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(overlaps))
    shifted = np.clip(alpha * (overlaps - m), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = float(np.sum(exp_s))
    w = exp_s / total
    smax = m + np.log(total) / alpha
    grad = np.zeros(n)
    for c in range(2 * n - 1):
        if w[c] < 1e-14:
            continue
        s = c - (n - 1)
        wc = w[c] * dx
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            grad[lo1:hi1] += wc * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] += wc * (-h[lo2 + s:hi2 + s])
    return smax, grad

def optimize_smooth(h, target, alpha, max_iters, deadline):
    h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        smax, grad = smooth_max_and_grad(h, alpha)
        lr = 0.5
        improved = False
        for _ in range(50):
            h_trial = project_sum(h - lr * grad, target)
            smax_trial, _ = smooth_max_and_grad(h_trial, alpha)
            if smax_trial < smax * (1 - 1e-10):
                h = h_trial
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return h

def binary_local_search(h_bin, max_iters, deadline, rng):
    n = len(h_bin)
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) < 2:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new <= best_c5 + 1e-12:
            best_c5 = c5_new
            stuck = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stuck += 1
        if stuck > 200 and it % 50 == 0:
            for _ in range(3):
                ones2 = np.where(h > 0.5)[0]
                zeros2 = np.where(h < 0.5)[0]
                i2 = int(rng.choice(ones2))
                j2 = int(rng.choice(zeros2))
                h[i2] = 0.0
                h[j2] = 1.0
                c5_2 = compute_c5(h)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                else:
                    h[i2] = 1.0
                    h[j2] = 0.0
    return h, best_c5

        target = n / 2.0
