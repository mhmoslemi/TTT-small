def compute_c5(h):
    """Compute C5 = max of full correlation times dx"""
    n = len(h)

    return float(np.max(corr) * dx)

def get_gradient_at(h, k_index):
    """Gradient of correlation at lag index k_index w.r.t. h"""
    n = len(h)
    k = k_index - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        idx = np.arange(n - k)
        grad[idx + k] += (1.0 - h[idx])
        grad[idx] -= h[idx + k]
    else:

        idx = np.arange(n - m)
        grad[idx] += (1.0 - h[idx + m])
        grad[idx + m] -= h[idx]
    return grad

    n = len(h)

import numpy as np
import time

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum} via bisection on shift."""
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    proj = np.clip(h - tau, 0.0, 1.0)
    # Fix rounding
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-proj) if diff > 0 else np.argsort(proj)
        for i in range(n):
            idx = order[i]
            if diff > 0:
                add = min(1.0 - proj[idx], diff)
                proj[idx] += add
                diff -= add
            else:
                rem = min(proj[idx], -diff)
                proj[idx] -= rem
                diff += rem
            if abs(diff) < 1e-14:
                break
    return proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_gradient_at(h, k_index):
    """Gradient of correlation at lag index k_index w.r.t. h."""
    n = len(h)
    k = k_index - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        idx = np.arange(n - k)
        grad[idx + k] += (1.0 - h[idx])
        grad[idx] -= h[idx + k]
    else:
        m = -k
        idx = np.arange(n - m)
        grad[idx] += (1.0 - h[idx + m])
        grad[idx + m] -= h[idx]
    return grad

def optimize_from(h_init, deadline, seed):
    """Optimize h from h_init until deadline."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    current_c5 = compute_c5(h)
    best_local = h.copy()
    best_local_c5 = current_c5
    alpha = 0.05
    no_improve = 0

    while time.time() < deadline:
        dx = 2.0 / n
        all_corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_max = int(np.argmax(all_corr))

        # Smoothed subgradient using top lags
        top_k = np.argsort(all_corr)[-3:][::-1]
        temps = max(0.01, 0.02 * (1 + no_improve / 200.0))
        weights = np.exp((all_corr[top_k] - all_corr[top_k[0]]) / temps)
        weights /= weights.sum()
        grad = np.zeros(n)
        for idx, w in zip(top_k, weights):
            grad += w * get_gradient_at(h, int(idx))

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = compute_c5(h_new)

        if new_c5 < current_c5 - 1e-10:
            h = h_new
            current_c5 = new_c5
            no_improve = 0
            if current_c5 < best_local_c5:
                best_local_c5 = current_c5
                best_local = h.copy()
            alpha = min(alpha * 1.003, 0.3)
        else:
            no_improve += 1
            alpha = max(alpha * 0.97, 0.002)
            if no_improve > 40:

                h_pert = h + rng.normal(0, scale, size=n)

                alpha = 0.05

                    h_rand = h + rng.normal(0, 0.35, size=n)

                    rc = compute_c5(h_rand)
                    if rc < current_c5:
                        h = h_rand
                        current_c5 = rc
                        break
                no_improve = 0
                alpha = 0.05

    return best_local, best_local_c5

def local_polish(h, max_rounds=300):
    """Coordinate-wise local search."""
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    h_best = h.copy()
    deltas = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01]

    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                h_try = h_best.copy()
                h_try[i] += d
                if h_try[i] < -0.01 or h_try[i] > 1.01:
                    continue
                h_try = project_feasible(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-11:
                    h_best = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h_best, best_c5
