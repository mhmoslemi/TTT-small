    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = 0.5 * np.ones(47)
        h_init[:23] = 1.0
        h_init[24:] = 0.0

    n = len(h_init)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP with decreasing alpha
    for alpha in [300, 100, 50, 20, 10, 5, 3, 2, 1.5]:
        if time.time() - start_time > budget_s * 0.55:
            break
        h_opt = optimize_slsqp(best_h, n, alpha, maxiter=400)
        c5 = compute_c5(h_opt, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Phase 2: Subgradient descent polishing the worst lag
    if time.time() - start_time < budget_s * 0.75:
        h_sg = subgradient_descend(best_h, n, max_iter=3000, step_start=0.005)
        c5 = compute_c5(h_sg, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sg.copy()

    # Phase 3: More SLSQP at lower alpha
    for alpha in [2, 1.5, 1.2, 1.0]:
        if time.time() - start_time > budget_s * 0.85:
            break
        h_opt = optimize_slsqp(best_h, n, alpha, maxiter=800)
        c5 = compute_c5(h_opt, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Phase 4: Try higher resolution
    for target_n in [64, 96, 128]:
        if time.time() - start_time > budget_s * 0.92:
            break
        h_interp = interpolate_h(best_h, n, target_n)
        c5_interp = compute_c5(h_interp, target_n)
        current_best = best_c5
        current_h = best_h.copy()
        current_n = n

        for alpha in [100, 50, 20, 10, 5, 3]:
            if time.time() - start_time > budget_s * 0.95:
                break
            h_opt = optimize_slsqp(h_interp, target_n, alpha, maxiter=300)
            c5 = compute_c5(h_opt, target_n)
            if c5 < current_best:
                current_best = c5
                current_h = h_opt.copy()
                current_n = target_n
                h_interp = h_opt.copy()

        if current_best < best_c5:
            best_c5 = current_best
            best_h = current_h
            n = current_n

    # Final subgradient polish
    if time.time() - start_time < budget_s * 0.97:
        n_final = len(best_h)
        h_sg = subgradient_descend(best_h, n_final, max_iter=2000, step_start=0.003)
        c5 = compute_c5(h_sg, n_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sg

    final_c5 = compute_c5(best_h, len(best_h))
    best_c5 = min(best_c5, final_c5)
    return best_h, best_c5, len(best_h)
