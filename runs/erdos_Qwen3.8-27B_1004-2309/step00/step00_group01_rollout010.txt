        else:
            hi = mid

    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corr)
    log_sum = beta * max_c + np.log(np.sum(np.exp(beta * (corr - max_c))))

import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    """Project h onto {sum(h)=target_sum, 0<=h<=1} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj_and_grad(h, beta):
    """Compute smooth log-sum-exp objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    total = np.sum(exps)
    weights = exps / total
    obj = ((beta * max_c + np.log(total)) / beta) * dx
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            w = weights[m]
            if w < 1e-15:
                continue
            if 0 <= j - k < n:
                s += w * (1.0 - h[j - k])
            if 0 <= j + k < n:
                s -= w * h[j + k]
        grad[j] = (dx / beta) * s
    return obj, grad

        h_rand = rng.rand(n_try)
        h_opt, c5 = optimize(h_rand, min(per_alloc * 0.7, budget_s * 0.85 - elapsed), seed + 10000 + n_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 3: Final refinement
    if best_h is not None:
        elapsed = time.time() - t_start
        alloc = budget_s * 0.90 - elapsed
        if alloc > 20:
            h_opt, c5 = optimize(best_h.copy(), alloc, seed + 77777)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    if best_h is None:
        n = 48
        best_h = np.ones(n) * 0.5
        best_c5 = compute_C5(best_h)

    return best_h, best_c5, len(best_h)
