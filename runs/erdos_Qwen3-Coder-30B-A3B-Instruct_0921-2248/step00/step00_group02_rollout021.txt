import numpy as np
import time
from scipy import signal

def evaluate_overlap(h, dx):
    # Compute the overlap integral C_5
    conv = np.correlate(h, 1 - h, mode="full")
    max_conv = np.max(conv) * dx
    return max_conv

def satisfy_constraints(h, n_points, dx):
    # Ensure sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    target_sum = n_points / 2
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Adjust the first entry to meet the constraint
        h[0] += diff
        # Clamp to [0, 1]
        h[0] = np.clip(h[0], 0, 1)
    return h

def perturb_h(h, n_points, dx, mode='binary'):
    # Generate a new h by doing a small perturbation
    h_new = h.copy()
    
    if mode == 'binary':
        # Pick two indices: one 1 -> 0, one 0 -> 1
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_new[i] = 0
            h_new[j] = 1
    elif mode == 'continuous':
        # adjust one point slightly
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_new[i] += delta
        h_new[i] = np.clip(h_new[i], 0, 1)
    else:
        # noise based on current values
        noise = np.random.normal(0, 0.05, n_points)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        
    # enforce sum constraint
    h_new = satisfy_constraints(h_new, n_points, dx)
    return h_new
