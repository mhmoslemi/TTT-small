import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up the optimization
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.001
    alpha_decay = 0.99
    tolerance = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5

    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        cross_corr = signal.correlate(h, 1 - h, mode='full')
        cross_corr = cross_corr * dx
        
        # Find the index of maximum overlap (k_max)
        k_max = np.argmax(np.abs(cross_corr))
        
        # Compute gradient signs (approximate gradient of overlap w.r.t. h)
        # For shift k_max, we compute d/dh_i (h_i * (1 - h_{i+k_max}))
        # This is a simplified gradient approximation
        gradient_signs = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient_signs[i] = 1 - h[j] - h[i]
        
        # Update h
        h = h - alpha * gradient_signs
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Re-normalize to maintain sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute current C5
        current_C5 = compute_C5(h, dx)
        
        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Adaptive step size
        if iteration % 50 == 0:
            alpha *= alpha_decay
            
    # Ensure final constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Compute the full cross-correlation
    cross_corr = signal.correlate(h, 1 - h, mode='full')
    cross_corr = cross_corr * dx
    # Return the maximum absolute value
    return np.max(np.abs(cross_corr))
