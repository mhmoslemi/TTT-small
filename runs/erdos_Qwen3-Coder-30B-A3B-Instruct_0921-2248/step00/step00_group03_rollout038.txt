import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy constraint
    h = h * target_sum / np.sum(h)
    
    # Set up optimization parameters
    alpha = 0.001
    alpha_decay = 0.99
    min_alpha = 1e-6
    max_iter = 10000
    tol = 1e-6
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        cross_corr = signal.correlate(h, 1 - h, mode='full')
        overlap_values = cross_corr * dx
        
        # Find worst overlap
        k_max = np.argmax(overlap_values)
        max_overlap = overlap_values[k_max]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = 1 - h[j] - h[i]
            
        # Update h
        h_new = h - alpha * gradient
        
        # Project onto [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Compute new C5
        c5_val = compute_c5(h, dx)
        
        # Update best solution
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            last_improvement = iteration
            
        # Adaptive step size
        if iteration % 50 == 0:
            alpha = max(min_alpha, alpha * alpha_decay)
            
        # Convergence check
        if iteration - last_improvement > 1000 and alpha < 1e-5:
            break
            
    return (list(best_h), best_c5, n_points)

def compute_c5(h, dx):
    cross_corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(cross_corr) * dx
