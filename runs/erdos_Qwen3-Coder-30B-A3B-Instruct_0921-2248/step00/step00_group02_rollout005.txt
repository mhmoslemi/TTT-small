import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except:
        # Create a starting point: binary with sum = n_points/2
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate(h):
        # Compute overlap integral
        conv = signal.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def is_feasible(h):
        # Check bounds and sum constraint
        return (np.all(h >= 0) and np.all(h <= 1) and 
                abs(np.sum(h) - target_sum) < 1e-10)
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Generate candidate perturbation
        new_h = h.copy()
        
        # Try different perturbation strategies
        strategy = np.random.choice(['flip', 'adjust', 'noise'])
        
        if strategy == 'flip' and n_points >= 4:
            # Flip two elements: one from 0 to 1, one from 1 to 0
            # Find 0 and 1 entries
            zeros = np.where(new_h == 0)[0]
            ones = np.where(new_h == 1)[0]
            
            if len(zeros) > 0 and len(ones) > 0:
                i0 = np.random.choice(zeros)
                i1 = np.random.choice(ones)
                new_h[i0] = 1.0
                new_h[i1] = 0.0
        elif strategy == 'adjust' and n_points > 2:
            # Adjust a single point (keep sum constraint)
            i = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.2, 0.2)
            old_value = new_h[i]
            new_h[i] = np.clip(old_value + delta, 0, 1)
            
            # Adjust another point to maintain sum
            j = np.random.randint(0, n_points)
            if i != j:
                new_h[j] = np.clip(new_h[j] - delta, 0, 1)
        else:  # noise
            # Add small noise to multiple points
            indices = np.random.choice(n_points, size=min(5, n_points//2), replace=False)
            for i in indices:
                delta = np.random.uniform(-0.1, 0.1)
                new_h[i] = np.clip(new_h[i] + delta, 0, 1)
            
            # Adjust one more point to maintain sum
            i = np.random.randint(0, n_points)
            old_sum = np.sum(new_h)
            new_h[i] = np.clip(new_h[i] + (target_sum - old_sum), 0, 1)
        
        if is_feasible(new_h):
            c5_candidate = evaluate(new_h)
            if c5_candidate < best_c5:
                best_h = new_h.copy()
                best_c5 = c5_candidate
                h = new_h.copy()
        else:
            # If infeasible, try a different approach
            h = best_h.copy()
    
    # Ensure final result is feasible
    if not is_feasible(best_h):
        # Recalculate with strict measure
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
    
    return (best_h.tolist(), best_c5, len(best_h))
