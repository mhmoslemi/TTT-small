import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    """Returns the C5 bound for the given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute full correlation of h with (1-h)
    corr = correlate(h_values, 1 - h_values, mode='full')
    # The full correlation is of length 2*n_points - 1
    # We want the maximum of the positive shifts (k = 1 to n_points-1)
    # and negative shifts (k = -n_points+1 to -1) which are
    # equivalent to shifts from 1 to n_points-1
    # The corresponding indices for positive shifts k = 1..n_points-1
    # are n_points - 1 + 1 = n_points to 2*n_points - 2
    max_val = 0.
    for k in range(1, n_points):
        idx = n_points - 1 + k
        max_val = max(max_val, corr[idx] * dx)
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points

    except NameError:
        # Default starting vector (linearly spaced with some structure)
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.1
        h0[1] = 0.3
        h0[2] = 0.6
        h0[3] = 0.8
        h0[4] = 0.9
        h0[5] = 0.8
        h0[6] = 0.5
        h0[7] = 0.2
        h0[8] = 0.1
        # Range of n points in [0,1] should sum to 1
        h0 = h0 / np.sum(h0) * (n_points/2)
        
    h_opt = h0.copy()
    best_c5 = evaluate_erdos_solution(h_opt)
    
    # Initialize lag set with coarse sample
    lag_set = np.linspace(-2, 2, 41)  # step=0.1 initially
    lag_set = np.unique(np.concatenate((lag_set, [0])))
    
    # Iterative refinement loop
    n_iter = 0
    while True:
        if time.time() - start_time > budget_s - 1.0:
            break
        n_iter += 1
        # Build constraints based on current lag set
        # Variables
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = []
        # Bound on h
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        # Sum constraint
        constraints.append(cp.sum(h) * dx == 1)
        # Constraint for each lag in lag_set
        for k in lag_set:
            k_int = int(np.round(k / dx))
            if k_int == 0:
                continue
            corr_expr = 0.0
            for i in range(n_points):
                i_shift = (i - k_int) % n_points
                corr_expr += h[i] * (1 - h[i_shift])
            constraints.append(corr_expr * dx <= C5)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_vals = h.value
                c5_val = C5.value
                if c5_val < best_c5:
                    best_c5 = c5_val
                    h_opt = h_vals.copy()
            else:
                break
        except Exception:
            break
            
        # Evaluate overlaps to find worst violations
        corr_all = correlate(h_opt, 1 - h_opt, mode='full')
        max_overs = []
        for k in range(n_points):
            idx = n_points - 1 + k
            if idx < len(corr_all):
                cur_val = corr_all[idx] * dx
                max_overs.append((cur_val, k - n_points + 1))
            else:
                break
        # We also look at negative lags
        for k in range(1, n_points):
            idx = n_points - 1 - k
            if idx >= 0:
                cur_val = corr_all[idx] * dx
                max_overs.append((cur_val, k - n_points + 1))
        
        max_overs.sort(reverse=True)
        top_k = max(1, len(max_overs) // 10)  # Top 10%
        new_lags = set()
        for val, k in max_overs[:top_k]:
            new_lags.add(k)
            
        # Add new lag samples around the problematic ones
        for k in list(new_lags):
            for dk in np.linspace(-0.1, 0.1, 5):
                new_lags.add(k + dk)
        
        new_lags = np.array(list(new_lags))
        # Filter to valid range
        new_lags = new_lags[(new_lags >= -2) & (new_lags <= 2)]
        # Keep up to 100 lags to control size
        new_lags = np.unique(np.round(new_lags / 0.05) * 0.05)[:90]
        lag_set = np.sort(np.unique(np.concatenate((lag_set, new_lags))))
        
        # Early stopping if not improving
        if n_iter > 20 and abs(corr_all[n_points-1] * dx - best_c5) < 1e-5:
            break
    
    # Final verification
    best_c5 = evaluate_erdos_solution(h_opt)
    # Clip to [0,1]
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points/2)
    final_c5 = evaluate_erdos_solution(h_opt)
    
    if final_c5 < 0.3807 and final_c5 < best_c5:
        return (h_opt.tolist(), final_c5, n_points)
    else:
        # return the initial if no improvement
        try:
            return (initial_h_values.tolist(), evaluate_erdos_solution(initial_h_values), len(initial_h_values))
        except NameError:
            # Fallback
            h_fallback = np.ones(n_points) * 0.5
            h_fallback = h_fallback / np.sum(h_fallback) * (n_points/2)
            return (h_fallback.tolist(), evaluate_erdos_solution(h_fallback), n_points)
