import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    def obj_and_grad(h_vec):
        h = h_vec.copy()

        # Compute approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)

            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps

        penalty_grad = np.zeros_like(h)
        # Bounds
        penalty_grad[h <= 0] += 1e6
        penalty_grad[h >= 1] += 1e6
        # Integral constraint
        integral = np.sum(h) * dx
        penalty_grad += 1e3 * (integral - 1.0) / n_points
        
        return c5, grad + penalty_grad
    
    # Trust region method with adaptive steps
    max_iter = 5000
    trust_radius = 0.1
    min_trust_radius = 1e-6
    
    for iter_num in range(max_iter):

        h_current = best_h.copy()

        for _ in range(10):

            trust_radius = max(min_trust_radius, trust_radius * 0.5)
    
    return (best_h.tolist(), float(best_c5), n_points)
