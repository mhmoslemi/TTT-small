import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Ensure h0 is valid
    h0 = np.clip(h0, 0, 1)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial guess to sum constraint
    h0_sum = np.sum(h0)
    if h0_sum > 0:
        h0 = h0 * target_sum / h0_sum

    # Variables
    h = cp.Variable(n_points)
    C = cp.Variable()

    # Constraints
    constraints = []
    constraints.append(cp.sum(h) == target_sum)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(C >= 0)

    # Loop over all shifts
    max_k = n_points - 1
    for k in range(-max_k, max_k + 1):
        # Compute indices for h[i] * (1 - h[i + k])
        expr = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h[i] * (1 - h[j])
        constraints.append(expr <= C / dx)

    # Objective
    objective = cp.Minimize(C)

    prob = cp.Problem(objective, constraints)
    
    # Try multiple solvers in order of preference
    solvers_to_try = ['ECOS', 'SCS', 'OSQP']
    best_h = None
    best_C = float('inf')
    
    for solver in solvers_to_try:
        if time.time() - start_time > budget_s - 5:
            break
        try:
            prob.solve(solver=solver, max_iters=500, time_limit=budget_s - (time.time() - start_time) - 5)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_val = h.value
                C_val = C.value
                if C_val < best_C and np.all(h_val >= 0) and np.all(h_val <= 1):
                    best_h = h_val.copy()
                    best_C = C_val
        except Exception:
            continue

    if best_h is None:
        # Last resort: use the initial guess as a fallback
        best_h = h0
        best_C = 0.0
        # Compute the actual overlap via correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_C = np.max(corr) * dx
    
    # Final correction to ensure sum constraint
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        actual_sum = np.sum(best_h)
        if actual_sum > 0:
            best_h = best_h * target_sum / actual_sum
    
    # Compute final C5 using correlation
    if best_h is not None:
        try:
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            final_C5 = np.max(corr) * dx
        except:
            final_C5 = best_C
    else:
        final_C5 = best_C

    return (list(best_h), float(final_C5), n_points)
