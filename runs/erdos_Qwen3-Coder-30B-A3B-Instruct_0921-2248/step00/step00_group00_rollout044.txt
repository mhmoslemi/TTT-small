import numpy as np
from scipy import correlate
import time

def evaluate_h(h):
    """Evaluate the C5 bound for a given h."""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def feasible_h(h):
    """Ensure h is feasible: in [0,1] and sum(h) = n_points/2."""
    n = len(h)
    h = np.clip(h, 0, 1)
    scale_factor = (n / 2) / np.sum(h)
    h = h * scale_factor
    return h

def mutate_h(h, mutation_rate=0.01):
    """Mutate h by random segment changes while maintaining feasibility."""
    h = h.copy()
    n = len(h)
    num_muts = max(1, int(n * mutation_rate))
    for _ in range(num_muts):
        i = np.random.randint(0, n)
        h[i] = np.random.uniform(0, 1)
    return feasible_h(h)

def crossover_h(h1, h2):
    """Uniform crossover between two h vectors."""
    n = len(h1)
    mask = np.random.rand(n) > 0.5
    h_new = np.where(mask, h1, h2)
    return feasible_h(h_new)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize a population with feasible h vectors."""
    population = []
    if initial_h is not None:
        population.append(initial_h)
    while len(population) < pop_size:
        h = np.random.uniform(0, 1, n_points)
        h = feasible_h(h)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    n = len(population)
    indices = np.random.choice(n, tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]
