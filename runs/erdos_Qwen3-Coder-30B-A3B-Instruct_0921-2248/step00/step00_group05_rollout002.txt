import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Setup and parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    
    # Initial function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create symmetric tent function
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x <= 1:
                h[i] = x
            else:
                h[i] = 2 - x
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Find maximum overlap
        conv = signal.convolve(h, 1 - h, mode='full')
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        
        # Compute gradient for the max overlap shift
        grad = np.zeros(n_points)
        shift = max_idx - (n_points - 1)
        
        # Gradient contribution for each position
        for idx in range(n_points):
            if idx + shift >= 0 and idx + shift < n_points:
                grad[idx] += (1 - h[idx + shift])
            if idx - shift >= 0 and idx - shift < n_points:
                grad[idx] -= h[idx - shift]
                
        # Update with gradient
        h = h - alpha * grad
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    conv = signal.convolve(h, 1 - h, mode='full')
    return np.max(conv) * dx
