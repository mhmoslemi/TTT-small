import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h, n_points):
    # This is the provided evaluation function, assumes h is normalized
    dx = 2.0 / n_points
    # Compute all overlap integrals for full convolution-like behavior
    conv = np.correlate(h, 1 - h, mode='full')
    # Extract the part corresponding to k in [0,2]
    mid = len(conv) // 2
    overlaps = conv[mid : mid + n_points]
    # Apply dx scaling
    overlaps *= dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction if available, otherwise use a uniform distribution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 100
        h_init = np.ones(n_points) * 0.5

    dx = 2.0 / n_points

    # Define the optimization problem
    h = cp.Variable(n_points)
    C5 = cp.Variable()

    # Set up problem constraints
    constraints = []

    # Integral constraint (sum h * dx = 1)
    constraints.append(cp.sum(h) * dx == 1)

    # Bound constraints (h[i] in [0,1])
    constraints.extend([h[i] >= 0 for i in range(n_points)])
    constraints.extend([h[i] <= 1 for i in range(n_points)])

    # Overlap constraints
    # Sample lags from [-2, 2] with a step size to make it manageable
    lag_step = 0.1
    lags = np.arange(-2, 2 + lag_step/2, lag_step)

    for k in lags:
        # Approximate circular shift using modulo
        k_int = int(np.round(k / dx))
        # Compute overlap integral for lag k
        overlap = 0.0
        for i in range(n_points):
            j = (i - k_int) % n_points
            overlap += h[i] * (1 - h[j])
        overlap *= dx
        constraints.append(overlap <= C5)

    # Objective
    objective = cp.Minimize(C5)

    # Create and solve the problem
    prob = cp.Problem(objective, constraints)

    # Start timing for solving
    time_left = budget_s - (time.time() - start_time)
    if time_left < 10:
        return h_init, evaluate_erdos_solution(h_init, n_points), n_points

    try:
        prob.solve(solver=cp.GLPK, verbose=False, time_limit=time_left)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            c5_bound = prob.value
        else:
            # fallback to initial solution if optimization fails
            h_sol = h_init
            c5_bound = evaluate_erdos_solution(h_init, n_points)
    except Exception as e:
        # fallback to initial solution
        h_sol = h_init
        c5_bound = evaluate_erdos_solution(h_init, n_points)

    # Ensure solution respects constraints
    if not np.all(h_sol >= 0) or not np.all(h_sol <= 1):
        h_sol = np.clip(h_sol, 0, 1)
    if not np.isclose(np.sum(h_sol) * dx, 1):
        h_sol = h_sol * (1 / (np.sum(h_sol) * dx)) 

    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_sol, n_points)
    
    return h_sol, final_c5, n_points
