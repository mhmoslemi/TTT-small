import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """Returns the maximum overlap integral C5 for given h"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute full cross-correlation
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The maximum overlap is at the center
    max_corr = np.max(corr)
    return max_corr * dx

        h_var = cp.Variable(n_points)
