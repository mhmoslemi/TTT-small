import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Create a reasonable starting point
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.5
        h0[-1] = 0.5
        # Make sure integral is 1
        h0 = h0 / np.sum(h0) * 2.0 / n_points
    
    dx = 2.0 / n_points
    
    # Find initial max overlap
    def compute_c5(h):
        h = np.array(h)
        overlaps = []
        for k in np.arange(0, 2, 0.1):
            k_idx = int(k / dx)
            if k_idx == 0:
                overlap = np.sum(h * (1 - h)) * dx
            else:
                h_shifted = np.roll(h, k_idx)
                overlap = np.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    # Initialize
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Adaptive constraint method
    k_steps = np.arange(0, 2, 0.1)
    k_indices = [int(k / dx) for k in k_steps if k < 2]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 50:
        iteration += 1
        # Currently using just the 1st pass optimization approach
        # Solve convex optimization problem with active constraints
        
        # Build constraints
        constraints = []
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # integral constraint
        constraints.append(cp.sum(h_vars) * dx == 1)
        
        # bounds
        constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
        constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
        
        # max overlap constraints
        for k_idx in k_indices:
            if k_idx == 0:
                overlap = cp.sum(cp.multiply(h_vars, 1 - h_vars)) * dx
            else:
                shifted_h = cp.hstack([h_vars[k_idx:], h_vars[:k_idx]])
                overlap = cp.sum(cp.multiply(h_vars, 1 - shifted_h)) * dx
            constraints.append(overlap <= C5)
        
        prob = cp.Problem(cp.Minimize(C5), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_vars.value
                new_c5 = C5.value
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
        except:
            pass
        
        # Evaluate over all lags to find worst k's
        if time.time() - start_time > budget_s - 2:
            break
            
        max_overlap = 0
        worst_k_indices = []
        for k in np.arange(0, 2, 0.05):
            k_idx = int(k / dx)
            if k_idx == 0:
                overlap = np.sum(best_h * (1 - best_h)) * dx
            else:
                h_shifted = np.roll(best_h, k_idx)
                overlap = np.sum(best_h * (1 - h_shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k_indices = [k_idx]
            elif abs(overlap - max_overlap) < 1e-6:
                worst_k_indices.append(k_idx)
                
        # Add some top k's if current set is not enough
        k_indices.extend(worst_k_indices[:3])
        k_indices = list(set(k_indices))  # Remove duplicates
        
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h, final_c5, n_points)
