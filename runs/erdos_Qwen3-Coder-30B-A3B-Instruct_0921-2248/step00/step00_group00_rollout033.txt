def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value of correlation (times dx) is our C5
    c5 = np.max(corr) * dx
    return c5
