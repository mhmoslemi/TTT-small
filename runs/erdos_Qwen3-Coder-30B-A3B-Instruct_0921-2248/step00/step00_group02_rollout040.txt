import numpy as np
import time
from scipy import signal

def evaluate_overlap(h):
    """Evaluate the maximum overlap for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h) using full mode
    conv = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral is scaled by dx
    max_overlap = np.max(conv) * dx
    return max_overlap

def perturb_h(h, indices, delta=0.05):
    """Generate a perturbed version of h by adjusting values at given indices."""
    h_new = h.copy()
    for i in indices:
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
    return h_new

    h_new = h.copy()
    h_new[idx1] = np.clip(h_new[idx1] - delta, 0, 1)
    h_new[idx2] = np.clip(h_new[idx2] + delta, 0, 1)
    return h_new
