import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Create a simple starting point
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Initial constraints
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        # Compute all correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Only the overlapping parts (max shift)
        max_k = len(corr) - n_points
        corr_vals = corr[n_points-1:n_points-1+max_k]
        return np.max(corr_vals) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    def project(h_vec):
        # Ensure h is in [0,1] and sum constraint
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Augmented Lagrangian-like method
    def augmented_lagrangian(h_vec, lambda_):
        f = objective(h_vec)
        penalty = lambda_ * (constraint_sum(h_vec))**2
        return f + penalty
    
    # Gradient descent with projection
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    t0 = time.time()
    
    # Optimization loop
    lambda_ = 1.0
    tolerance = 1e-6
    max_iter = 10000
    iter_count = 0
    
    while time.time() - t0 < budget_s * 0.95 and iter_count < max_iter:
        # Project initial point
        h = project(h)
        
        # Compute gradient numerically
        eps = 1e-5
        grad = np.zeros_like(h)
        base_val = objective(h)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            grad[i] = (objective(h_plus) - base_val) / eps
        
        # Update with gradient step
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = project(h_new)
        
        # Update lambda if necessary (simple adaptation)
        current_c5 = objective(h_new)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
            
        h = h_new
        iter_count += 1
    
    # Final project invalid values
    best_h = project(best_h)
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
