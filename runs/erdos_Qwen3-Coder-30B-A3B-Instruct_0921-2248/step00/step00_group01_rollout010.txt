import numpy as np
import random
import time

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Get the maximum value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = (h > np.median(h)).astype(float)
    else:
        # Initialize randomly
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Ensure sum is exactly 1/dx = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Binarize to get integer constraints (make sure sum is exactly n_points/2)
    # Convert to binary
    h_binary = (h > 0.5).astype(int)
    # Adjust to meet exact count
    current_sum = np.sum(h_binary)
    if current_sum > n_points // 2:
        diff = current_sum - n_points // 2
        indices = np.where(h_binary == 1)[0]
        for i in range(diff):
            idx = indices[i]
            h_binary[idx] = 0
    elif current_sum < n_points // 2:
        diff = n_points // 2 - current_sum
        indices = np.where(h_binary == 0)[0]
        for i in range(diff):
            idx = indices[i]
            h_binary[idx] = 1
            
    h = h_binary.astype(float)
    
    # Start timing
    start_time = time.time()
    best_h = h.copy()
    current_c5 = evaluate_h(h, dx)
    best_c5 = current_c5
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.9995
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        # Make a move: swap one 0 with one 1
        indices_0 = np.where(h == 0)[0]
        indices_1 = np.where(h == 1)[0]
        
        if len(indices_0) > 0 and len(indices_1) > 0:
            idx_0 = random.choice(indices_0)
            idx_1 = random.choice(indices_1)
            
            # Create new h
            new_h = h.copy()
            new_h[idx_0] = 1
            new_h[idx_1] = 0
            
            new_c5 = evaluate_h(new_h, dx)
            
            # Accept or reject
            if new_c5 < current_c5 or random.random() < np.exp((current_c5 - new_c5) / T):
                h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        # Reduce temperature
        T *= alpha
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
