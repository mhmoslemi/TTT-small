import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The correlation starts at index = len(h) - 1
    # We want the maximum over all shifts from -len(h)+1 to len(h)-1
    max_corr = np.max(corr)
    return max_corr * dx

def objective(h):
    dx = 2.0 / len(h)
    return evaluate_c5(h, dx)
