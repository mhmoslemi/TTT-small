```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral for given h"""
    # Compute cross-correlation (equivalent to np.correlate(h, 1-h, mode='full'))
    correlation = np.correlate(h, 1 - h, mode='full')
    # Find maximum value (corresponds to max_k integral)
    max_overlap = np.max(correlation)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h into [0,1]^n and ensure sum(h) = n_points/2"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust mean to target
    target_mean = 0.5
    current_mean = np.mean(h)
    if current_mean > 0:
        h = h * target_mean / current_mean
    # Ensure sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    return h

def objective_and_gradient(h, n_points, dx):
    """Objective function and its gradient"""
    # Evaluate objective
    c5 = evaluate_c5(h, dx)
    # For gradient estimation here we use approximate finite differences
    # This is a rough approximation with limited precision
    # A more precise gradient can be obtained but causes significant overhead
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    # Initialize with provided initial solution or binary construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Binary construction, then shuffled
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)

    # Store best found solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Equality constraint: sum = n_points/2 (this must be exactly
    # handled to maintain integrality constraint)
    def constraint_eq(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_eq}

    start_time = time.time()

    # Use SLSQP for constrained optimization
    # We will iterate until time runs out
    max_iter = 1000
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # SLSQP wants gradient
        # For now we use a non-gradient method
        # Thus we create a dummy objective to pass into minimize that we can evaluate manually

        res = minimize(objective_and_gradient, h, args=(n_points, dx), 
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 10, 'ftol': 1e-6, 'gtol': 1e-3})

        if res.success:
            h = res.x
            h = project_h(h, n_points)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # If optimization fails, slightly perturb the current solution
            h = np.clip(h + 0.01 * np.random.randn(n_points), 0, 1)
            h = project_h(h, n_points)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        iter_count += 1

    # Final projection to ensure
    best_h = project_h(best_h, n_points)
    best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
```
