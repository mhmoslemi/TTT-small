def evaluate_erdos_solution(h_values, n_points):
    h = np.array(h_values)
    dx = 2.0 / n_points
    # Compute all overlaps with 1-h using correlation
    corr = correlate(h, 1 - h, mode="full")
    # Extract the valid range of lags (-2 to 2)
    mid = len(corr) // 2
    valid_corr = corr[mid - n_points:mid + n_points]
    # Maximum overlap
    C5 = np.max(valid_corr) * dx
    return C5
