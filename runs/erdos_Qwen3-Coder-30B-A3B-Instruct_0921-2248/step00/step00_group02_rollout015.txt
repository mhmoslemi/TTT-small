import numpy as np
from scipy import signal
import time
import random

def evaluate_overlap(h, dx):
    # Compute the overlap integral for all shifts
    corr = signal.correlate(h, 1 - h, mode='full')
    # The full convolution includes indices from -n+1 to n-1
    # We're interested in shifts 0 to n-1 (corresponding to k=0 to k=n-1)
    # But we want to take max of overlap for valid shifts k in 0 to n-1  
    # The convolution result indices correspond to shifts:
    # i corresponds to shift k = i - (n-1) where n=len(h)
    n = len(h)
    result = corr[n-1:2*n-1]  # Valid shifts k = 0 to n-1
    return np.max(result) * dx

def compute_C5(h_values, dx):
    return evaluate_overlap(h_values, dx)

    # Check bounds and sum constraint
    return (np.all(h >= 0) and np.all(h <= 1) and
