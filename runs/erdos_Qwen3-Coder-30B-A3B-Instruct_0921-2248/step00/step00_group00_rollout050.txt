import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Compute C5 from step function h"""
    # Compute all cross-correlations
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Get the maximum value (excluding trivial lags)
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with initial construction if available, else use a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Simple construction: step function that is 0.5 over entire interval
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    dx = 2.0 / n_points
    h *= (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # SCA iterations
    for i in range(10000):  # Upper limit to prevent infinite loop
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute best lag
        corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr_val = corr[max_k]
        
        # Compute lag offset for full correlation at max_k
        lag = max_k - (n_points - 1)
        
        # Linearization around current h
        # h[i] * (1 - h[i + lag]) ≈ h0[i] * (1 - h0[i + lag]) + 
        #                          (1 - h0[i + lag]) * (h[i] - h0[i]) - 
        #                          h0[i] * (h[i + lag] - h0[i + lag])
        # This is the linearized constraint:
        # sum_{i=0}^{n-1-lag} h[i] (1 - h[i + lag]) dx <= C5
        # ≈ sum_{i=0}^{n-1-lag} [h0[i] (1 - h0[i + lag]) + (1 - h0[i + lag]) (h[i] - h0[i]) - h0[i] (h[i + lag] - h0[i + lag])] dx <= C5
        
        # Make convex problem
        # Variables
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Constraints
        constraints = []
        for k in range(2 * n_points - 1):
            offset_i = k - (n_points - 1)
            if offset_i < 0:
                valid_range = range(n_points + offset_i)
            else:
                valid_range = range(offset_i, n_points)
                
            # Get indices of h[i] and h[i+lag]
            i_indices = list(valid_range)
            k_indices = [i + offset_i for i in i_indices]
            
            # Linearized constraint for this lag
            if len(i_indices) == 0:
                continue
                
            # Build left-hand side of constraint
            lhs_expr = 0
            for j, i in enumerate(i_indices):
                k_idx = k_indices[j]
                if k_idx >= n_points or k_idx < 0:
                    continue
                h0_i = h[i]
                h0_k = h[k_idx]
                lhs_expr += h0_i * (1 - h0_k) + (1 - h0_k) * (h_var[i] - h0_i) - h0_i * (h_var[k_idx] - h0_k)
            
            constraints.append(lhs_expr * dx <= C5_var)
        
        # Feasibility constraints for h
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        constraints.append(cp.sum(h_var) * dx == 1)
        
        # Add previous constraint (to ensure current solution is feasible)
        constraints.append(cp.sum(h_var * (1 - h)) * dx <= best_C5)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    new_C5 = evaluate_erdos_solution(h)
                    if new_C5 < best_C5:
                        best_C5 = new_C5
                        best_h = h.copy()
        except:
            pass
            
    return (best_h.tolist(), best_C5, len(best_h))
