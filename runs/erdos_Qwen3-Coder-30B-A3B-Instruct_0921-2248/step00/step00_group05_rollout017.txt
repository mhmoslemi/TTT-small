import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1 - abs(i - mid) / (mid + 1)
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integral for all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        
        # Instead of reducing by one shift, we can try a batch approach
        # But for simpler implementation, we just focus on the worst
        # Compute the gradient for the most harmful shift directly
        
        # Find shifts that produce maximum overlap
        # Best is to take few high-impact shifts
        shifts = [max_corr_idx - n_points + 1]  # Most likely shift causing max
        
        # Compute gradient for these shifts
        grad = np.zeros(n_points)
        for shift in shifts:
            # Compute conv(h, 1-h) gradient at this shift
            for i in range(n_points):
                if 0 <= i + shift < n_points:
                    grad[i] += (1 - h[i + shift])
                if 0 <= i - shift < n_points:
                    grad[i] -= h[i - shift]
        
        # Gradient descent
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)  # Maintain sum constraint
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Adaptive step size
        if iteration % 100 == 0 and iteration > 0:
            alpha *= 0.95
    
    return (list(best_h), best_c5, n_points)

def compute_c5(h, dx):
    """Compute the maximum overlap integral."""
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
