import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate_h(h, dx):
    # Compute correlation and find maximum
    corr = np.correlate(h, 1-h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a symmetric tent function
        n_points = 200
        j = np.arange(n_points) - n_points / 2
        h = np.maximum(0, 1 - 2 * np.abs(j) / n_points)
        h = h / np.sum(h) * n_points / 2  # Normalize to sum = n_points/2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Iterative refinement using quadratic approximation
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 5 and iter_count < max_iter:
        iter_count += 1
        
        # Compute correlations to get max overlap
        corr = np.correlate(h, 1-h, mode="full")
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        max_k = max_corr_idx - (n_points - 1)
        
        # Design a QP problem to reduce this specific overlap
        # Vectors for shift k
        h_k = np.zeros(n_points)
        if max_k >= 0:
            h_k[max_k:] = h[:-max_k]
        else:
            h_k[:max_k] = h[-max_k:]
        
        # Objective: minimize h_k * (1 - h)
        # Using quadratic approximation around current h
        # Approximate h_k * (1 - h) ≈ h_k - h_k * h
        # But since h_k * h is quadratic (convex), we avoid it directly
        
        # Better approach: minimize \sum h_k[i] * (1 - h[i])
        # This is a linear objective: sum(h_k - h_k * h)
        
        # Create variables
        H = cp.Variable(n_points)
        # Constraint h_k * (1 - h) <= value (cannot express directly)
        # Instead, use that we want to minimize sum(h_k * (1 - h[i])) across i
        # Which is just sum(h_k) - sum(h_k * h) 
        # This is a quadratic term in h, but let's constraint the shift properly
        
        # Simplified but effective: we minimize sum(h_k[i] * (1 - h[i])) 
        # For the shift that maximizes overlap
        # Objective function
        obj = cp.Minimize(h_k.T @ (np.ones(n_points) - H))
        
        # Constraints
        constraints = [
            H >= 0,
            H <= 1,
            cp.sum(H) == n_points / 2
        ]
        
        # Create and solve problem
        prob = cp.Problem(obj, constraints)
        try:
            # Use ECOS or MOSEK solver (fast and works well for this)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = H.value
                h = new_h.copy()
                
                # Check if this improves solution
                c5 = evaluate_h(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
                # Avoid convergence issues: make sure constraints are met
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * n_points / 2
                
        except:
            pass  # If solver fails, continue with old h


    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (list(best_h), final_c5, len(best_h))
