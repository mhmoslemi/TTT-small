import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) for all lags
    corr = correlate(h, 1 - h, mode="full")
    # The result corresponds to the overlap integral for each lag
    # Note: the overlap for lag k is corr[n-1+k] * dx
    # but due to how correlate works, we need to extract from the right position
    # or compute manually to be sure
    max_overlap = 0.0
    for k in range(2*n - 1):  # full correl with n points
        # k runs from 0 to 2n-2
        # for k=0: lag=0, corr_val=overlap of h with (1-h) shifted by 0 (just convolution)
        # need to convert to actual overlap for shift k
        # if k < n: shift is positive
        # if k >= n: shift is negative (overlapping h with reversed (1-h))
        if k == 0:
            # this is the same as full convolution without shifting
            # but it's not what we want, so let's look at lags properly
            # Only compute the relative overlap for shift 0, 1, ..., n-1
            shift = 0
            overlap = 0
            for i in range(n):
                overlap += h[i] * (1 - h[(i + shift) % n]) * dx
            max_overlap = max(max_overlap, overlap)
        else:
            # how to interpret k in correlate?
            # correlate(h, 1-h) is reversed correlation
            # Taking corr as indexing from 0 = shift=0 to 2*n-2 = shift=2*n-2
            # Actually k in range 0..2*n-2
            # for k = 0: shift 0 (same alignment)
            # for k = n: shift n (first sample of h aligned with last of (1-h))
            # k = n+1: shift n+1
            # but we need more precise?
            # Let's recompute all with clearer understanding
            # We want h*1-h, so over interval 0 to 2, i.e., we shift h by k (from 0 to 2)
            # and integrate h(x)(1-h(x+k)) dx.  Here, h is sampled
            shift = k - (n - 1)  # shift index in [-n+1, n-1]
            overlap = 0
            # Time consuming but clearer
            # masking in numpy
            if shift >= 0:  # make sure it aligns correctly
                # overlap = sum_{i=0}^{n-1} h[i] * (1 - h[(i + shift) % n]) * dx
                for i in range(n):
                    overlap += h[i] * (1 - h[(i + shift) % n]) * dx
            else:
                # shift < 0 so we shift h backward, h_i comes from h_{i-shift}
                for i in range(n):
                    overlap += h[i] * (1 - h[(i - shift) % n]) * dx
            max_overlap = max(max_overlap, overlap)
    return max_overlap

def evaluate_overlap(h, dx, k):
    """Evaluate overlap integral for specific lag k"""
    n = len(h)
    # We shift by k (in units of dx, so k*dx is shift in real space)
    overlap = 0
    for i in range(n):
        # i-th element of h shifts with k places
        # Since we're working in period (with wraparound),
        # we need to handle shifts properly.
        shifted_i = (i + k) % n
        overlap += h[i] * (1 - h[shifted_i]) * dx
    return overlap

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 42  # Start with standard if available 
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        assert len(h0) == n_points, "Bad initial shape"
    else:
        # Initialize with a simple baseline, e.g., uniform distribution,
        # normalized to sum to 1, but made slightly non-uniform to improve
        # optimality by breaking symmetry.
        # Actually, let's try sin-like with some wobble.
        h0 = np.ones(n_points)
        h0 = np.concatenate([h0[:n_points//2], h0[n_points//2:] * 0.8])  # Lower values on right half
        h0 = h0 * (n_points/2.0) / np.sum(h0)
    h0 = np.clip(h0, 0, 1)
    h = h0.copy()
    h = h / np.sum(h) * n_points * dx  # ensure sum = 1 (this is fine since dx = 2/n_points)
    # However, since we are running exactly on this form: sum(h)*dx == 1
    # So in terms of intermediate sums, it means `sum(h) == n_points / 2`
    # Which is: sum(h) = 21 if n_points = 42
    target_sum = 21
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # In some versions, we might consider solving with modern LP solver eg MOSEK
    # But here we will minimize directly using gradient-like or direct search
    
    # Strategy: Use a simple iterative line-search variant directed at finding better h
    
    # The key idea is to consider small perturbations to h
    # but subject to constraints. We will treat this as solving a large LP.
    # Best solvers for this are capped at ~100 constraints
    # So we采取策略: 
    # 1. Sample a small number of lags, get their overlaps, the max is C5 estimat
    # 2. Add the worst lags (say top 10) and solve LP with those constraints
    # 3. iterate until convergence or budget expires
    
    max_iter = 100
    iterations = 0
    
    # Our constraints will be of type: sum_i h[i] * (1 - h[i+z]) * dx <= C5
    # since we want to model h(x)(1-h(x+k)) 
    
    # Method: iterative constraint refinement with LP solver
    n_lags = 100  # sample this many lags
    sample_lags = np.linspace(0, 2, n_lags, endpoint=False) # from 0 to 2 with n_lags samples
    lags_to_add = set()  # that are causing largest overlaps
    
    # Start by computing a good initial C5 estimate with a few lags
    # Sample lags from [0,2]
    quants = np.linspace(0, 2, 20, endpoint=False)  # small subset for initial estimate
    estimated_c5_values = []
    for k in quants:
        # convert k to integer lag in dx steps (shift in indices)
        k_ind = int(k / dx)
        # see overlap value
        o = evaluate_overlap(h, dx, k_ind)
        estimated_c5_values.append(o)
    initial_c5_bound = max(estimated_c5_values)
    
    # Now we start iterative approach with CVXPY
    # Let's define a fresh try each time to minimize allocations
    
    # Refinement loop
    i = 0
    last_c5 = 1e9
    last_improvement = 1e9
    
    while 1:
        now = time.time()
        if now - start_time > budget_s - 2:  # Leave 2 seconds buffer
            break
        i += 1
        # Enumerate a set of interesting lags (we'll speed this by using the ones identified)
        # The idea is to lift up our assessment of what lags are critical
        sample_n = min(100, n_lags)
        refinement_lags = []
        refinement_lags = list(range(30))  # 0..29
        # We're not going to add more than say top 20 worst lags in each step
        # to fit within LP limits.
        # We sample several lags (say)
        test_lags = list(range(0, n_lags, max(1, n_lags//10)))  # evenly distributed
        test_lags = [i for i in test_lags if i not in refinement_lags]
        
        # Make sure we test the current samples
        worst_lags = []
        worst_overlaps = []
        c5_est = 0
        for lag in test_lags:
            if lag >= n_lags:
                continue
            k_real = lag * dx
            lag_index = lag      # as integer index
            overlap = evaluate_overlap(h, dx, lag_index)
            if overlap > c5_est:
                c5_est = overlap
            worst_overlaps.append(overlap)
            worst_lags.append(lag)
        
        # Add worst constraints in terms of overlap for optimization (rounded to int)
        # We'll clip from worst to around top 20 or so before optimizing 
        # Sort deviation pairs and take top 20 augmentations
        sorted_pairs = list(zip(worst_overlaps, worst_lags))
        sorted_pairs.sort(reverse=True)
        best_indices = sorted_pairs[:min(20, len(sorted_pairs))]
        adds = [int(idx) for _, idx in best_indices]
        
        # Define a new LPC optimization problem: find h* to minimize C5 s.t.
        # h[i] >= 0, h[i] <= 1, sum(h)*dx = 1, and for additional_lags:
        # sum_i h[i] * (1 - h[i+lag]) * dx <= C5
        
        # BEGIN CVXPY OPTIMIZATION CODE
        model_h = Variable(n_points, name="h")
        C5 = Variable(name="C5")
        
        # Constraints
        constraints = []
        constraints.append(model_h >= 0)
        constraints.append(model_h <= 1)
        
        # integral constraint: sum(h)*dx = 1
        constraints.append(sum(model_h) * dx == 1)
        
        # Add lag constraints
        for lag in adds:
            raw_overlap = 0
            # Set up for lagums (directly)? we compute for all i
            # h_i * (1 - h_{i+lag%N}) * dx
            for i in range(n_points):
                shifted_i = (i + lag) % n_points
                raw_overlap += model_h[i] * (1 - model_h[shifted_i]) * dx
            constraints.append(raw_overlap <= C5)
            
        # Objective
        objective = Minimize(C5)
        
        # Solve
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                # Update h with the solution
                h_save = model_h.value 
                if h_save is not None:
                    h = np.array(h_save)
                    h = np.clip(h, 0, 1)
                    new_c5 = C5.value
                    if new_c5 is not None and new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                # Simply update end point
        except:
            pass  # not unusual with ECOS
        # Finish optimization step on this iteration
        
        # Check for early termination or floating point error etc
        if abs(last_c5 - best_c5) < 1e-5:
            # stop with no significant change
            break
        if best_c5 < 0.3807:
            break  # goal reached
        last_c5 = best_c5
        last_improvement = abs(last_c5 - best_c5)
        if (i % 5 == 0):
            # Periodic check
            pass
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
