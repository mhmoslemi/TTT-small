import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """Evaluates C5 for given h using full correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for lags in [-2, 2] with step 0.1
    lags = np.arange(-2, 2.1, 0.1)
    max_val = 0.0
    for k in lags:
        k_int = int(k / 0.1)
        if k_int < 0:
            corr = np.sum(h[:k_int] * (1 - h[-k_int:]))
        elif k_int > 0:
            corr = np.sum(h[k_int:] * (1 - h[:-k_int]))
        else:
            corr = np.sum(h * (1 - h))
        max_val = max(max_val, corr)
    return max_val * dx

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Create a simple initialization
        n_points = 100
        h0 = np.ones(n_points) * 0.5
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try several different spacings of points to avoid local minima
    n_points_list = [100, 120, 150]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        # Align h so that sum(h) * dx = 1
        h0 = np.ones(n_points) * 0.5
        h0 = h0 * (1.0 / np.sum(h0) * n_points * dx)
        
        # Sample lags
        lags = np.arange(-2, 2.1, 0.1)
        n_lags = len(lags)
        
        # Create CVXPY vars
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Constraints: h[i] in [0,1], sum(h) = 1/dx
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == 1.0/dx]
        
        # For each lag k, we want: \int h(x)(1-h(x+k)) dx <= C5
        for i, k in enumerate(lags):
            # Convert k to integer shift
            k_int = int(k / 0.1)
            if k_int < 0:
                # h[i] * (1 - h[i+k]) for all i where i+k >= 0
                corr = cp.sum(cp.multiply(h_vars[:-k_int], 1.0 - h_vars[-k_int:]))
            elif k_int > 0:
                # h[i] * (1 - h[i-k]) for all i where i-k < n_points
                corr = cp.sum(cp.multiply(h_vars[k_int:], 1.0 - h_vars[:-k_int]))
            else:
                # h[i] * (1-h[i]) for all i
                corr = cp.sum(cp.multiply(h_vars, 1.0 - h_vars))
            
            constraints.append(corr <= c5_var)
        
        # Objective function
        objective = cp.Minimize(c5_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, max_iters=5000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                c5_bound = prob.value
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_sol.copy()
        except:
            pass
            
    return (best_h, best_c5, len(best_h))
