import numpy as np
from scipy import fft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Scale to satisfy constraint
    h = h * 2.0 / h.sum()
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent in frequency domain
    def optimize_step():
        nonlocal h, best_h, best_c5
        
        # Convert to frequency domain
        H = fft.fft(h)
        # Compute the frequency domain representation of 1-h
        H_conj = np.conj(H)
        # Correlation in frequency domain
        corr_freq = H * H_conj
        # Back to time domain
        corr_time = fft.ifft(corr_freq).real
        # Max correlation (shifts)
        max_corr = np.max(corr_time)
        
        # Compute gradient
        grad_freq = np.zeros_like(H)
        # Find which shift gives max correlation
        max_idx = np.argmax(corr_time)
        # Gradient based on that shift
        H_shifted = np.roll(H, max_idx)
        # Gradient for minimizing max correlation
        grad_freq = H_shifted * H_conj
        
        # Update with gradient descent
        step_size = 0.01
        H_new = H - step_size * grad_freq
        
        # Reconstruct and project
        h_new = fft.ifft(H_new).real
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Scale to integrate to 1
        h_new = h_new * 2.0 / h_new.sum()
        
        return h_new
    
    # Iterative optimization
    max_iter = 10000
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        h_new = optimize_step()
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Using the same pattern as summarize_erdos_solution but ensuring max is accurate
    n = len(h)
    # Compute subtracted version
    h_sub = 1.0 - np.array(h)
    # Full correlation by convolution (FFT method)
    full_corr = np.correlate(h, h_sub, mode='full')
    # The max over full shifts
    max_corr = np.max(full_corr) * dx
    return max_corr
