import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h, dx):
    # Compute overlaps for all lags
    overlaps = np.correlate(h, 1-h, mode="full")
    # Normalize by dx
    overlaps *= dx
    # Return max overlap
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 42
    dx = 2.0 / n_points

    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        # Fallback initialization - uniform distribution that sums to 1
        h_init = np.ones(n_points) * (1.0 / n_points) * (n_points / 2)
        
    # Start with the current best solution
    h_current = h_init.copy()

    # Use a list of lags to test convergence
    lag_step = 0.5
    max_k = 2.0
    lags = np.arange(0.0, max_k + 1e-6, lag_step)
    
    # Initialize best solution
    best_c5 = evaluate_c5(h_current, dx)
    best_h = h_current.copy()
    
    start_t = time.time()
    iteration = 0

    # Constraint refinement loop
    while time.time() - start_t < budget_s - 1:
        iteration += 1

        # Define optimization problem
        h = Variable(n_points)
        C5 = Variable()

        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2,
            C5 >= 0
        ]

        # Add constraints for all sampled lags
        for k in lags:
            # Compute correlation value for this lag
            lag_idx = int(k / dx)
            # Shifted indices (flip and shift for convolution-like)
            shifted_indices = np.arange(n_points) - lag_idx
            # Wrap around for [] indexing
            shifted_indices = shifted_indices % n_points
            # Compute integral: sum h[i] * (1 - h[i-k]) * dx
            corr_val = 0.0
            for i in range(n_points):
                corr_val += h[i] * (1 - h[shifted_indices[i]])
            constraints.append(corr_val * dx <= C5)

        # Define and solve problem
        objective = Minimize(C5)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
        except:
            break

        if prob.status != 'optimal':
            break

        h_new = np.array(h.value).flatten()
        curr_c5 = C5.value

        # Update best solution
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_new.copy()

        # Identify worst lags (largest overlaps with current h)
        overlaps = np.zeros(len(lags))
        for i, k in enumerate(lags):
            lag_idx = int(k / dx)
            shifted_indices = np.arange(n_points) - lag_idx
            shifted_indices = shifted_indices % n_points
            corr_val = 0.0
            for j in range(n_points):
                corr_val += h_new[j] * (1 - h_new[shifted_indices[j]])
            overlaps[i] = corr_val * dx

        # Add top 10 worst lags to constraints for next iteration
        # Pick 10 most problematic lags
        np.random.shuffle(lags)
        top_lags = lags[np.argsort(overlaps)[-10:]]
        lags = np.unique(np.concatenate([lags, top_lags]))

        # Check if improvement is small enough to terminate early  
        if iteration > 5:
            # Small improvement threshold
            if np.abs(best_c5 - curr_c5) < 1e-5:
                break

        # Break early if time's about up
        if time.time() - start_t > budget_s - 1:
            break

    # Final validation
    c5_final = evaluate_c5(best_h, dx)
    # Ensure final vector correctly normalized
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)

    return (best_h.tolist(), float(c5_final), n_points)
