import numpy as np
from scipy import optimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Create symmetric tent function with peak at center
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x <= 1:
                h[i] = x
            else:
                h[i] = 2 - x
    
    # Ensure sum constraint
    h = h / np.sum(h) * target_sum
    
    def compute_C5(h_vals):
        # Compute overlap integral using full convolution
        h_conj = 1 - h_vals
        conv = np.correlate(h_vals, h_conj, mode='full')
        # Take the maximum over all shifts
        return np.max(conv) * dx
    
    def gradient_descent_step(h_vals, step_size=0.01):
        n = len(h_vals)
        h_conj = 1 - h_vals
        # compute correlation (overlap) in frequency domain for efficiency
        H = fft(h_vals)
        H_conj = fft(h_conj)
        # outer product in frequency domain
        GH = H * np.conj(H_conj)
        # inverse transform to get time domain correlation
        overlap = ifft(GH).real
        # take first n values (corresponding to shift 0 to n-1)
        corr = overlap[:n]
        # find the maximum overlap and its index
        max_idx = np.argmax(corr)
        #  compute gradient w.r.t h for that shift
        # jacobian contribution from terms contributing to max overlap
        grad = np.zeros(n)
        for k in range(n):
            # shifting h by k mod n and difference from 1-h
            shifted_h = np.roll(h_vals, -k)
            grad[k] += (1 - shifted_h[max_idx])  # d/dh_k (h_k * (1-h_{k+max_idx}))
            grad[k] -= h_vals[max_idx]  # d/dh_k (h_k * (1-h_{k+max_idx}))
        # this is an approximation of the gradient w.r.t max overlap for current max shift
        grad = -grad  # negative gradient since we minimize C5
        # update h
        h_new = h_vals - step_size * grad
        # enforce box constraints
        h_new = np.clip(h_new, 0, 1)
        # enforce sum constraint
        h_new = h_new / np.sum(h_new) * target_sum
        return h_new
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    no_improvement_count = 0
    
    # Optimization loop
    i = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # compute new h using gradient descent step
        h_new = gradient_descent_step(best_h, 0.01)
        new_C5 = compute_C5(h_new)
        if new_C5 < best_C5:
            best_h = h_new
            best_C5 = new_C5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        i += 1
        # reduce step size over time
        if i % 10 == 0:
            pass  # keep step size adaptive but not modifying it here for now
    
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, n_points)
