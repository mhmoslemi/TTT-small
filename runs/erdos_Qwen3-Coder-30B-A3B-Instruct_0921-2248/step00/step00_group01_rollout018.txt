from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Compute C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Take max over all shifts
    c5 = np.max(corr) * dx
    return c5
