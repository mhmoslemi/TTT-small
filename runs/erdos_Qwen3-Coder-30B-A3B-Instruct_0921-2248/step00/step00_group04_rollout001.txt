import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lag = 2.0
    lag_step = 0.1
    lags = np.arange(-max_lag, max_lag + lag_step/2, lag_step)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Constraints
    constraints = []
    
    # Support constraint
    constraints.append(cp.sum(h) * dx == 1.0)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints for each lag
    for k in lags:
        # Compute the overlap integral for lag k
        # This would be: \int h(x)(1 - h(x + k)) dx
        # As a discrete approximation:
        integral = 0
        for i in range(n_points):
            j = (i + int(k / dx)) % n_points
            integral += h[i] * (1 - h[j]) * dx
        constraints.append(integral <= C5)
        # Break early if time is up
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        h_opt = h.value
        C5_opt = C5.value
    except Exception:
        # Fallback to a simple heuristic if optimization fails
        h_opt = h_start.copy()
        C5_opt = evaluate_solution(h_opt, lags, dx)
        
        # Try simple optimization
        for _ in range(1000):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Gradient step with projection
            grad = np.zeros(n_points)
            for k in lags[:20]:  # Use first 20 lags for speed
                for i in range(n_points):
                    j = (i + int(k / dx)) % n_points
                    grad[i] -= (1 - h_opt[j]) * dx
                
            h_opt -= 0.01 * grad
            h_opt = np.clip(h_opt, 0, 1)
            # Re-normalize
            h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
            new_C5 = evaluate_solution(h_opt, lags, dx)
            if new_C5 < C5_opt:
                C5_opt = new_C5
                
    # Ensure final h values are valid
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    
    return (h_opt, C5_opt, n_points)

def evaluate_solution(h, lags, dx):
    max_val = 0
    n = len(h)
    for k in lags:
        integral = 0
        for i in range(n):
            j = (i + int(k / dx)) % n
            integral += h[i] * (1 - h[j]) * dx
        max_val = max(max_val, integral)
    return max_val
