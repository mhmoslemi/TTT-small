import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

        # Normalize to integral = 1
        h = h / np.sum(h) * 2
        
    # Set up constraints
    bounds = [(0, 1) for _ in range(n)]
    target_sum = n / 2
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    start_time = time.time()
        
    # Gradient-based local search with adaptive steps
    for iter_no in range(10000):

        # We compute gradients with finite differences to avoid complex derivation
        eps = 1e-5

            h_minus[i] -= eps
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Adjust to maintain integral constraint
            delta = (np.sum(h_plus) - target_sum) / n

            h_plus = np.clip(h_plus, 0, 1)

        # Adjust to meet integral constraint (maintain sum)
        adjustment = (target_sum - np.sum(h_new)) / n
        h_new += adjustment
        
        # Clamp again after adjustment
        h_new = np.clip(h_new, 0, 1)
        
        # Update if improvement
        new_c5 = evaluate_h(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
        
    return (best_h.tolist(), best_c5, len(best_h))
