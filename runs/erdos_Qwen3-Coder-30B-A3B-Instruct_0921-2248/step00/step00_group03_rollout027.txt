import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Variables
    h = cp.Variable(n_points, name="h")
    C = cp.Variable(name="C")
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum, C >= 0]
    
    # Generate all overlap constraints
    for k in range(-n_points + 1, n_points):
        # Compute indices after shift k
        indices = np.arange(n_points)
        shifted_indices = (indices + k) % n_points
        # Compute correlation term
        correlation = cp.sum(cp.multiply(h, 1 - h[shifted_indices]))
        # Add constraint
        if abs(k) < n_points:
            constraints.append(correlation <= C / dx)
            
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solver options
    solver_opts = {
        'max_iters': 200,
        'time_limit': budget_s - (time.time() - start_time) - 1
    }
    
    try:
        prob.solve(solver=cp.ECOS, **solver_opts)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_opt = C.value
        else:
            # Fallback to initial
            h_opt = h_init
            c5_opt = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    except Exception:
        # Fallback to initial if solver fails
        h_opt = h_init
        c5_opt = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    # Final validation
    if not np.all(0 <= h_opt) or not np.all(h_opt <= 1):
        h_opt = np.clip(h_opt, 0, 1)
    if abs(np.sum(h_opt) - target_sum) > 1e-6:
        h_opt = h_opt * (target_sum / np.sum(h_opt))
        
    c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    return (h_opt.tolist(), c5_bound, n_points)
