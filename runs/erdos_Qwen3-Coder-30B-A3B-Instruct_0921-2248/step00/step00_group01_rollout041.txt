import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral
    correlation = np.correlate(h, 1-h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):

        # Shuffle to get a better starting point
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        # Compute max overlap
        correlation = np.correlate(h_vec, 1-h_vec, mode="full")
        max_correlation = np.max(correlation)
        return max_correlation * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Define optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()

    # Use SLSQP for optimization
    opt_result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    if not opt_result.success:
        # fallback to simple projection method if optimization fails
        current_h = h.copy()

            current_h = current_h - 0.01 * (current_h - 0.5)  # push towards center

            current_h = current_h * target_sum / np.sum(current_h)

    if best_c5 > 0.3807:
