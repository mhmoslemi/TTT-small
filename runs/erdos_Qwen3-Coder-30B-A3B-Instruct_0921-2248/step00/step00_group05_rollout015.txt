import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(h)

        if time.time() - start_time > budget_s - 1:

        # Compute C5 and find worst overlapping shift
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Find shift k that maximizes the overlap
        corr = signal.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        
        # Compute gradient for that shift
        grad = np.zeros_like(h)
        for idx in range(n_points):

                grad[idx] = 1 - h[idx + k]
            else:
                grad[idx] = -h[idx]
        
        # Update h
        h = h - alpha * grad
        
        # Project h onto feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), int(len(best_h)))

def compute_c5(h):
    h = np.array(h)
    corr = signal.correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    return np.max(corr) * dx
