import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Create a feasible random starting point
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    if abs(np.sum(h) - target_sum) > 1e-8:
        # Adjust to satisfy sum constraint
        scale = target_sum / np.sum(h)
        h *= scale
        
    # Compute initial C5
    def compute_C5(h_vec):
        # Ensure h_vec sums to target_sum
        h_vec = np.array(h_vec)
        if abs(np.sum(h_vec) - target_sum) > 1e-8:
            scale = target_sum / np.sum(h_vec)
            h_vec *= scale
        h_vec = np.clip(h_vec, 0, 1)
        
        # Compute correlation
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # The convolution result has length 2*n_points - 1
        # We want the maximum over all shifts (k values)
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Initial value
    C5_current = compute_C5(h)
    best_h = h.copy()
    best_C5 = C5_current

    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iter_count = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 0.01:
        iter_count += 1
        # Generate neighbor by flipping intervals of length 1 or 3
        # Try different flip sizes (1 or 3)
        flip_size = np.random.choice([1, 3])
        
        # Pick a random starting index
        start_idx = np.random.randint(0, n_points - flip_size + 1)
        
        # Create candidate solution
        h_new = h.copy()
        
        # Flip values in interval [start_idx, start_idx + flip_size)
        for i in range(start_idx, start_idx + flip_size):
            h_new[i] = 1 - h_new[i]
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to satisfy integral constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-8:
            scale = target_sum / current_sum
            h_new *= scale
            
        # Compute C5 of new solution
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        if C5_new < C5_current:
            h = h_new
            C5_current = C5_new
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
        else:
            # Accept with probability exp(-(C5_new - C5_current) / T)
            delta = C5_new - C5_current
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                C5_current = C5_new
                
        # Update temperature
        T = max(T * alpha, min_T)
        
        # Occasionally restart with a random solution
        if iter_count % 5000 == 0:
            # Random restart to avoid trapping
            h = np.random.rand(n_points)
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0, 1)
            C5_current = compute_C5(h)
            if C5_current < best_C5:
                best_h = h.copy()
                best_C5 = C5_current
                
    # Final validation
    final_h = best_h.copy()
    final_C5 = compute_C5(final_h)
    
    # Ensure final solution meets constraints
    final_h = np.clip(final_h, 0, 1)
    final_h_sum = np.sum(final_h)
    if abs(final_h_sum - target_sum) > 1e-8:
        final_h *= (target_sum / final_h_sum)
    
    return (final_h.tolist(), final_C5, n_points)
