import numpy as np
import time
from scipy import signal

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h_values, 1 - h_values, mode='full')
    # Maximum value scaled by dx
    return np.max(corr) * dx

def is_feasible(h_values):
    """Check if h values satisfy constraints."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Check bounds
    if np.any(h_values < 0) or np.any(h_values > 1):
        return False
    # Check sum constraint
    if abs(np.sum(h_values) * dx - 1.0) > 1e-8:
        return False
    return True

def mutate_h(h_values, mutation_rate=0.01):
    """Mutate h values while maintaining feasibility."""
    h_new = h_values.copy()
    n_points = len(h_values)
    # Randomly choose indices to mutate
    num_mutations = max(1, int(mutation_rate * n_points))
    indices = np.random.choice(n_points, num_mutations, replace=False)
    for i in indices:
        # Add small random perturbation
        delta = np.random.normal(0, 0.05)
        h_new[i] += delta
        # Ensure bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
    # Rebalance to satisfy sum constraint
    dx = 2.0 / n_points
    current_sum = np.sum(h_new) * dx
    # Adjust all values by the same factor to meet sum = 1
    scale = 1.0 / current_sum if abs(current_sum) > 1e-10 else 1.0
    h_new *= scale
    # Ensure bounds again after scaling
    h_new = np.clip(h_new, 0, 1)
    return h_new

def crossover_h(h1, h2):
    """Perform uniform crossover between two h vectors."""
    n_points = len(h1)
    # Generate uniform random mask
    mask = np.random.rand(n_points) > 0.5
    h_new = np.where(mask, h1, h2)

    dx = 2.0 / n_points

    h_new *= scale
    # Ensure bounds
    h_new = np.clip(h_new, 0, 1)
    return h_new
