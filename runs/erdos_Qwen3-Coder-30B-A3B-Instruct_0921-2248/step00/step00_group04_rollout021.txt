        # Constraints for the linear program
        constraints = [0 <= h_var[i] for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(h_var) * dx == 1.0]  # Integral constraint

        for k_idx, k in enumerate(k_set):
            # Compute the overlap integral for lag k
            if k == 0:
                # Skip if k=0 (it's 0 but let's be safe)

            conv = []

                j = (i + lag) % n_points
                conv.append(h_var[i] * (1 - h_var[j]))

            integral_approx = cp.sum(conv) * dx
            constraints.append(integral_approx <= C5_var)

        objective = cp.Minimize(C5_var)
