        k_max_idx = np.argmax(full_corr)
        k_max = k_indices[k_max_idx] - (n_points - 1)

        # Compute gradient for h[i] due to max overlap term
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h[j]

        # Update h with small step size
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        c5 = evaluate_erdos_solution(h)[0]
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
