import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.001
    alpha_decay = 0.99
    tolerance = 1e-6
    best_c5 = float('inf')
    best_h = None
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.ones(n_points, dtype=np.float64) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute all overlaps
        # h * (1 - h[k:]) for all shifts k
        overlap_values = []
        for k in range(n_points):
            # Shift h by k positions (circular)
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted)) * dx
            overlap_values.append(overlap)
            
        # Find the largest overlap
        max_overlap = max(overlap_values)
        k_max = np.argmax(overlap_values)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Compute gradient signs
        # For overlap at k_max, gradient w.r.t. h[i] is:
        # sign(1 - h[(i+k_max)%n_points] - h[i])
        h_shifted = np.roll(h, k_max)
        gradient_signs = np.sign(1 - h_shifted - h)
        
        # Update h
        h = h - alpha * gradient_signs
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        if iteration % 50 == 0:
            alpha *= alpha_decay
            
        # Convergence check
        if iteration > 10 and abs(overlap_values[k_max] - max(overlap_values[:-1])) < tolerance:
            break
    
    if best_h is None:
        best_h = h
        
    # Final check
    overlap_values = []
    for k in range(n_points):
        h_shifted = np.roll(best_h, k)
        overlap = np.sum(best_h * (1 - h_shifted)) * dx
        overlap_values.append(overlap)
    final_c5 = max(overlap_values)
    
    return (best_h.tolist(), final_c5, n_points)
