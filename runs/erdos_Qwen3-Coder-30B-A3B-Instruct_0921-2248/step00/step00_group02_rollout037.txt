from scipy.signal import correlate
import time

        # Compute overlap with full convolution
        conv = correlate(h, 1 - h, mode='full')
        # Extract the valid overlapping parts
        valid_start = n_points - 1
        valid_end = 2 * n_points - 2
        overlaps = conv[valid_start:valid_end] * dx
        return np.max(overlaps)

        # Ensure h is within [0,1] and sum = target_sum
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust all points to make sum correct
            adjustment = (target_sum - current_sum) / n_points
            h += adjustment
            h = np.clip(h, 0, 1)
        return h

    def get_lags_to_improve(h, delta=0.1):
        # Find the indices that contribute to maximum overlap
        conv = correlate(h, 1 - h, mode='full')

        overlaps = conv[valid_start:valid_end] * dx
        max_overlap = np.max(overlaps)
        
        # Find significant lag indices
        indices = np.where(overlaps >= max_overlap - 1e-8)[0]
        return indices, max_overlap

    best_C5 = compute_C5(best_h)

    iterations = 0
    improvement_count = 0
