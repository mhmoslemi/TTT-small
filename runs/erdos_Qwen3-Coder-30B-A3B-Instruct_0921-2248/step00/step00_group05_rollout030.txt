import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the maximum overlap contributions
        max_idx = np.argmax(corr)
        # The shift index from the correlation
        shift = max_idx - (len(h) - 1)
        
        # Gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            j1 = i + shift
            j2 = i - shift
            if 0 <= j1 < n_points:
                grad[i] += (1 - h[j1])
            if 0 <= j2 < n_points:
                grad[i] += -h[j2]
                
        # Update step
        h_new = h - alpha * grad
        
        # Projection onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        h = h_new
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), int(n_points))

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
