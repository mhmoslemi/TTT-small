import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute the overlap integral for all shifts
    conv = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_h(h, dx):
    # Ensure constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * (2.0 / dx)  # scale to ensure sum = 1
    return h
