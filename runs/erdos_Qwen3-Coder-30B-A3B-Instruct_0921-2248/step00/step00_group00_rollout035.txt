import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Compute the maximum overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations
    corr = correlate(h, 1 - h, mode='full')
    # Maximum across all lags
    max_corr = np.max(corr)
    return max_corr * dx

        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)

        # For lag k, we have constraint: sum h[i]*(1 - h[i+k]) <= C5
        # Linearize h[i]*(1-h[i+k]) about current solution:
        # approx = h0[i]*(1-h0[i+k]) + (1-h0[i+k])*(h[i]-h0[i]) - h0[i]*(h[i+k]-h0[i+k])
        # Covariance matrix
        h0 = h.copy()
        c = np.zeros(n_points)
        
        # Set up optimization problem
        h_new = cp.Variable(n_points)
        C5 = cp.Variable()

        constraints = [h_new[i] >= 0 for i in range(n_points)]
        constraints.extend([h_new[i] <= 1 for i in range(n_points)])

        constraints.append(cp.sum(h_new) == n_points / 2)

        if max_lag < n_points:
            delta_k = max_lag
        else:
            delta_k = max_lag - n_points
        # For lags 0 to n_points-1-k, i ranges from 0 to n_points-1-k
        start_idx = max_lag if max_lag >= 0 else 0
        end_idx = n_points - delta_k
        
        # Linear constraint: sum of bilinear terms
        # At each position,   h[i] * (1 - h[i+delta_k])
        # Approximation around h0
        constraint_sum = 0
        for i in range(end_idx):
            coeff1 = 1 - h0[i+delta_k]  # (1 - h[i+delta_k])

    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), best_c5, n_points)
