import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
        if len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
    else:
        # Symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (n_points / 2.0)
        h = np.maximum(h, 0.0)
        
    # Ensure sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        # Compute FFTs
        H = fft(h)
        H_conj = np.conj(H)
        GH = H * H_conj
        
        # Convert back to time domain
        g = ifft(GH).real
        
        # Find maximum overlap
        max_overlap = 0
        worst_shift = 0
        for k in range(n_points):
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
                worst_shift = k
                
        # Update h to reduce worst overlap
        # Create gradient approximation
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + worst_shift) % n_points
            gradient[i] = (1 - h[j]) - h[i]
        
        # Apply gradient step
        step_size = 0.01
        h_new = h - step_size * gradient
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to meet sum constraint
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * target_sum / h_new_sum
            
        # Check improvement
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally use a random perturbation
        if no_improvement_count > 10:
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if h_new_sum > 0:
                h_new = h_new * target_sum / h_new_sum
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
                
        h = h_new.copy()
        
        if no_improvement_count > max_no_improvement:
            break
            
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    # Use correlation for maximum overlap computation
    conv = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(conv)
