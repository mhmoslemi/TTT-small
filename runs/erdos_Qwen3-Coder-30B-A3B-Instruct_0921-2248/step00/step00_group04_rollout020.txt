import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Default construction: step function with 42 points
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Normalize to unit integral
        h0 = h0 * (1.0 / np.sum(h0) * dx) / dx

    # Solver setup
    solver = "MOSEK"
    
    # Start with a coarse k-grid
    k_step = 0.5
    K = np.arange(0, 2, k_step)
    
    # Variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Initial constraints
    constraints = [0 <= h, h <= 1, sum(h) * (2.0/n_points) == 1, C5 >= 0]
    
    # Add initial constraints for sampled lags
    for k in K:
        if k == 0:
            continue
        # Compute the overlap integral for lag k
        lag_indices = np.arange(n_points) - int(k / (2.0/n_points))
        lag_indices = np.remainder(lag_indices, n_points)
        # Evaluate overlap integral using convolution
        # For lag k, we want sum_{i} h[i] * (1 - h[i+k]) * dx
        # But since h is a step function, we consider each i separately 
        # Approximate by sampling, but use constraints properly:
        # Do this exactly by using the cross-correlation: don't use np.correlate!
        # Instead we evaluate the full expression for selected k values.
        corr = h @ (1 - h[lag_indices])
        constraints.append(corr * (2.0/n_points) <= C5)

    # Objective
    objective = Minimize(C5)
    prob = Problem(objective, constraints)
    
    # Iterative optimization
    best_h = h0.copy()
    best_C5 = float('inf')
    
    # Sample a fixed set of lag values (dense enough to detect structure)
    max_lags = 50
    K_dense = np.linspace(0, 2, max_lags+1)[1:]  # Avoid 0
    
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        
        # Solve optimization problem
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status != "optimal":
                break
            C5_value = prob.value
        except:
            break
            
        if C5_value < best_C5:
            best_C5 = C5_value
            best_h = h.value.copy()
            
        # Evaluate current h on all lags to identify worst cases
        h_vec = best_h
        dx = 2.0 / n_points
        
        max_overlaps = []
        for k in K_dense:
            shift_idx = int(k / (2.0/n_points))
            h_shifted = np.roll(h_vec, -shift_idx)
            overlap = np.sum(h_vec * (1 - h_shifted)) * dx
            max_overlaps.append((overlap, k))
            
        # Sort to get top offenders
        max_overlaps.sort(reverse=True)
        worst_lags = [k for _, k in max_overlaps[:len(max_overlaps)//5]] # Add top 10% worst

        # Add additional constraints from worst lags
        if len(worst_lags) > 0:
            for k in worst_lags:
                if k == 0:
                    continue
                shift_idx = int(k / (2.0/n_points))
                h_shifted = h_vec if shift_idx == 0 else np.roll(h_vec, -shift_idx)
                corr = h @ (1 - h_shifted)
                constraints.append(corr * (2.0/n_points) <= C5)
            
            # Reconstruct problem with updated constraints
            prob = Problem(objective, constraints)
        else:
            break
            
    # Final validation
    try:
        dx = 2.0 / n_points
        final_overlap = 0.0
        for k in np.linspace(0, 2, 100):
            shift_idx = int(k / (2.0/n_points))
            h_shifted = np.roll(best_h, -shift_idx)
            overlap = np.sum(best_h * (1 - h_shifted)) * dx
            final_overlap = max(final_overlap, overlap)
        final_C5 = final_overlap
    except:
        final_C5 = best_C5

    # Return best solution found
    return (best_h, final_C5, n_points)
