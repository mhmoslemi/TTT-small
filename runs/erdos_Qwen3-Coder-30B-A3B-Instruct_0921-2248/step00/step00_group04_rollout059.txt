import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.random.random(n_points)
        h = h / np.sum(h) * target_sum

    # Normalize to satisfy constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    def evaluate(h):
        # Compute C5 using convolution
        conv = signal.correlate(h, 1 - h, mode='full')
        C5 = np.max(conv) * dx
        return C5

    def perturb(h, temperature):
        # Create a copy of h
        h_new = h.copy()
        
        # Determine amount to perturb
        epsilon = max(0.001, temperature * 0.1)
        
        # Randomly select a number of indices to adjust
        n_changes = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=n_changes, replace=False)
        
        # Adjust values
        for i in indices:
            delta = np.random.uniform(-epsilon, epsilon)
            h_new[i] += delta
        
        # Ensure all values are in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum
        current_sum = np.sum(h_new)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            # Distribute difference evenly across all positions
            h_new += diff / n_points
        
        # Clip again for precision
        h_new = np.clip(h_new, 0, 1)
        
        return h_new
    
    # Simulated Annealing
    current_h = h
    current_C5 = evaluate(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5
    T = 1.0
    cooling_rate = 0.9995
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Perturb
        new_h = perturb(current_h, T)
        new_C5 = evaluate(new_h)
        
        # Accept or reject
        delta = new_C5 - current_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_C5 = new_C5
            
            # Update best
            if current_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = current_C5
        
        # Cool down
        T *= cooling_rate
        
        # Occasionally print progress
        if iteration % 1000 == 0:
            elapsed = time.time() - start_time
            print(f"Iteration {iteration}, C5: {best_C5:.6f}, Time: {elapsed:.1f}s")
            
            if elapsed > 0.9 * budget_s:
                break
    
    # Final evaluation
    final_C5 = evaluate(best_h)
    
    return (best_h, final_C5, n_points)
