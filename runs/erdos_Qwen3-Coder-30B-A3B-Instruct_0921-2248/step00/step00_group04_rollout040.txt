import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_erdos_solution(h_values):
    """Compute the exact C5 value for the given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_rev = 1 - h
    # Compute full cross-correlation
    corr = correlate(h, h_rev, mode='full')
    # The maximum overlap is over all possible shifts
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Settings
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 20
    lag_step = 0.2
    add_lags_pct = 0.1
    
    # Initialize h from initial_h_values or construct a simple one
    try:
        h = np.array(initial_h_values)
        if len(h) == n_points:
            h = h.copy()
        else:
            # Interpolate
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h_orig)
    except:
        # Create a symmetric step function
        h = np.ones(n_points) * 0.5

    # Normalize to sum = 1 (but already normalized in the problem)
    h = h / np.sum(h) * n_points * dx
    
    # Start with a coarse set of lags
    lags = np.arange(-2, 2+lag_step, lag_step)
    lags = lags[abs(lags) <= 2.0]
    lags = lags.tolist()

    best_C5 = float('inf')
    best_h = h.copy()

    solver = 'GLPK_MI'  # fast solver

    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        try:
            # Formulate LP
            h_var = Variable(n_points)
            C5 = Variable()

            # Constraints
            constraints = [
                h_var >= 0,
                h_var <= 1,
                sum(h_var) * dx == 1,
                C5 >= 0
            ]
            
            # Add overlap constraints for current lags
            for k in lags:
                # Compute shifted indices (periodic)
                k_int = int(k / dx)
                if k_int == 0:
                    overlap_sum = sum(h_var[i] * (1 - h_var[i]) * dx for i in range(n_points))
                else:
                    # Compute convolution of h and (1 - h) shifted by k
                    shifted = np.roll(h_var, -k_int) if k_int > 0 else np.roll(h_var, k_int)
                    # penalty term introduced by shifting index
                    overlap = sum(h_var[i] * (1 - shifted[i]) * dx for i in range(n_points))
                    # Ensure k_int does not go out of range, tạm bỏ qua
                    overlap_sum = 0.0
                    for i in range(n_points):
                        idx = (i - k_int) % n_points
                        overlap_sum += h_var[i] * (1 - h_var[idx]) * dx
                constraints.append(overlap_sum <= C5)

            # Objective
            prob = Problem(Minimize(C5), constraints)

            # Solve
            prob.solve(solver=solver, verbose=False)
            
            if prob.status != 'optimal':
                break
                
            h_result = h_var.value
            C5_result = C5.value
            
            if C5_result < best_C5:
                best_C5 = C5_result
                best_h = h_result.copy()
            
            # Check all lags for violations
            corr = correlate(h_result, 1 - h_result, mode='full')
            # Normalize to dx
            corr *= dx
            max_corr = np.max(corr)
            
            # Get indices of high overlaps
            indices = np.where(corr > max_corr * (1 - add_lags_pct))[0]
            # Map indices to actual lags
            lag_indices = indices - len(h_result) + 1
            new_lags = []
            for li in lag_indices:
                lag_val = li * dx
                if lag_val not in lags:
                    new_lags.append(lag_val)
            
            # Add only a subset of new lags (to avoid too many constraints)
            if new_lags:
                lags.extend(new_lags[:int(len(new_lags)*0.5)])  # Add subset to reduce complexity
                lags = list(set(lags)) # Remove duplicates
            
        except Exception as e:
            # If there's an issue solving, fallback to simpler
            break
            
        iter_count += 1
        if (time.time() - start_time) > budget_s - 1:
            break

    # Final evaluation
    try:
        final_C5 = evaluate_erdos_solution(best_h)
    except:
        final_C5 = best_C5

    return (best_h.tolist(), final_C5, n_points)
