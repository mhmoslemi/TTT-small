import numpy as np
import time
from scipy import correlate

def evaluate_erdos_solution(h):
    """Computes the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap is over all shifts from -n+1 to n-1
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Initial construction from the parent problem."""
    # We construct a simpler feasible solution algorithmically
    n = 45
    h = np.zeros(n)
    # Make a step function: 0.5 for first 22 points, 0.5 for last 23
    h[:22] = 1.0
    h[22:] = 0.0
    # Normalize to match integral constraint
    h *= 2.0 / np.sum(h)
    return h

def make_feasible(h):
    """Ensure h is feasible: 0 <= h <= 1, sum(h) = n/2."""
    n = len(h)
    h = np.clip(h, 0, 1)
    target_sum = n / 2.0
    actual_sum = np.sum(h)
    if abs(actual_sum - target_sum) > 1e-8:
        # Adjust so that the sum is exactly target_sum
        scaling = target_sum / actual_sum
        h = h * scaling
    return h

def flip_interval(h, i, k):
    """Flip a contiguous subsequence of length k starting at i."""
    h_new = h.copy()
    end = min(i + k, len(h))
    h_new[i:end] = 1.0 - h_new[i:end]
    return h_new

def compute_c5(h):
    """Compute C5 using correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution from parent or create one
    try:
        h = initial_h_values()
    except:
        # Create a simple feasible solution
        n = 45
        h = np.ones(n) * 0.5
        h = make_feasible(h)
    n = len(h)
    dx = 2.0 / n
    
    # Start optimization
    start_time = time.time()
    
    # Parameters
    T0 = 0.1
    alpha = 0.999
    T = T0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Start with the current h
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    # MCMC-based search
    iterations = 0
    max_iterations = int(1e6)
    
    while time.time() - start_time < budget_s and iterations < max_iterations:
        # Generate a neighbor by flipping an interval
        i = np.random.randint(0, n)
        k = np.random.choice([1, 3])  # flip 1 or 3 points
        new_h = flip_interval(current_h, i, k)
        new_h = make_feasible(new_h)
        new_c5 = compute_c5(new_h)
        
        # Acceptance probability
        if new_c5 < current_c5:
            accepted = True
        else:
            delta = new_c5 - current_c5
            prob_accept = np.exp(-delta / T)
            accepted = np.random.rand() < prob_accept
        
        if accepted:
            current_h = new_h
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        T = T * alpha
        iterations += 1
        
    return best_h, best_c5, n
