from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the maximum overlap integral
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all shifts, scaled by dx
    max_overlap = np.max(corr) * dx
    return max_overlap
