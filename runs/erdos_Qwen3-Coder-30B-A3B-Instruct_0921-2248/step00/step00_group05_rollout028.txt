import numpy as np
import time
from scipy import fft
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200  # Balance accuracy and speed
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Tent function centered
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = i / mid
            else:
                h[i] = 1.0 - (i - mid) / (n_points - mid)
        h = h / np.sum(h) * target_sum
    
    # Precompute FFT of ones for efficiency
    fft_ones = fft.fft(np.ones(n_points))
    
    def compute_c5(h_vals):
        h = np.array(h_vals)
        # Compute overlap integral using correlation
        # Efficiently compute all shifts
        h_flip = np.flip(h)
        convolution = np.convolve(h, h_flip, mode='full')
        # The overlap is with (1-h), so we also need to do that
        h_reflect = 1.0 - h
        conv_h_reflect = np.convolve(h, h_reflect, mode='full')
        # Replace with (1-h) to get the actual overlap
        # We're computing 1 - h(x+k) for all k
        conv_h_reflect = np.convolve(h, np.flip(1-h), mode='full')
        # Similarly for h * (1-h): for h coeffs, convolve with ones vector
        # No, need to correct the integral properly using proper convolution
        # convolve h and (1-h) 
        conv = np.correlate(h, 1-h, mode='full')
        # Replace with the mechanical approach: full convolution of h with 1-h
        # It's: integral_h(x) (1 - h(x+k)) dx: we simulate with sums over 0..2 range
        # better to do it from chunk
        C5_vals = []
        for k in range(n_points):
            # Construct shift k such that h(i) pair with (1-h(i+k))
            if k < n_points:
                # Need h[0:n-k] * (1-h[k:n]) 
                # and h[n-k:n] * (1-h[0:k])
                # Let h_shift = h[k:] + h[:k] 
                h_shift = np.concatenate([h[k:], h[:k]])
                val = np.sum(h * (1 - h_shift))
                C5_vals.append(val * dx)
            else:
                # k >= n_points; use wrap-around
                h_shift = np.concatenate([h[k-n_points:], h[:k-n_points]])
                val = np.sum(h * (1 - h_shift))
                C5_vals.append(val * dx)
        C5 = max(C5_vals)
        return C5
    
    def evaluate_h(h_vals):
        h = np.array(h_vals)
        # Normalize to sum=1
        h = h / np.sum(h) * target_sum
        c5 = compute_c5(h)
        return c5
    
    def project_h(h_vals):
        h = np.array(h_vals)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Ensure sum is correct
        current_sum = np.sum(h)
        if np.abs(current_sum - target_sum) > 1e-6:
            h = h * target_sum / current_sum
        return h
    
    # Gradient-descent style optimization
    max_iter = 5000
    lr = 0.01
    patience = 50
    
    last_c5 = float('inf')
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        h_current = project_h(h)
        
        # Approximate gradient of overlap
        # For a very rough gradient, sample nearby h
        eps = 1e-4
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            c5_plus = evaluate_h(h_plus)
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus)
            c5_minus = evaluate_h(h_minus)
            grad[j] = (c5_plus - c5_minus) / (2*eps)
        
        # Gradient step 
        h = h - lr * grad
        h = project_h(h)
        
        c5_new = evaluate_h(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        if np.abs(c5_new - last_c5) < 1e-6:
            patience -= 1
            if patience == 0:
                break
        else:
            patience = 50
        last_c5 = c5_new
        
    
    final_h = project_h(best_h)
    final_c5 = evaluate_h(final_h)
    
    return (final_h.tolist(), round(final_c5, 10), n_points)
