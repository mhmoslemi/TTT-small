import numpy as np
from scipy.signal import correlate
import time
from math import floor

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    conv = correlate(h, 1 - h, mode='full')
    max_conv = np.max(conv)
    return max_conv * dx

def is_feasible(h, n_points):
    """Check if h satisfies constraints"""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2.0) < 1e-8)

def mutate_h(h, mut_rate, n_points):
    """Mutation operator for h"""
    h_new = h.copy()
    for i in range(n_points):
        if np.random.random() < mut_rate:
            h_new[i] = np.random.random()
    # Ensure feasibility
    h_new = np.clip(h_new, 0, 1)
    # Adjust to maintain sum constraint
    current_sum = np.sum(h_new)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale and adjust
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
        # Clamp values to maintain [0,1] bounds after scaling
        h_new = np.clip(h_new, 0, 1)
        # Final adjustment might be needed
        diff = target_sum - np.sum(h_new)
        if abs(diff) > 1e-8:
            # Adjust last element
            h_new[-1] = min(1.0, max(0.0, h_new[-1] + diff))
    return h_new

def crossover_h(h1, h2):
    """Uniform crossover between h1 and h2"""
    n = len(h1)
    mask = np.random.random(n) > 0.5
    h_new = np.where(mask, h1, h2)
    return h_new

def tournament_selection(population, fitnesses, tournament_size):
    """Select one individual using tournament selection"""
    k = min(tournament_size, len(population))
    idxs = np.random.choice(len(population), size=k, replace=False)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

    # Population-based evolutionary approach
    pop_size = 100
    max_gen = 6000
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize population
    population = [initial_h.copy()]
    for i in range(pop_size - 1):
        h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2.0) / current_sum
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Check initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    if fitnesses[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitnesses[best_idx]
    
    gen = 0
    last_improvement = 0
    patience = 500
    
    while gen < max_gen:
        if time.time() - start_time > budget_s * 0.98:
            break
        gen += 1

        # Evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in population]

        min_idx = np.argmin(fitnesses)

            best_h = population[min_idx].copy()

        # Adaptive parameters
