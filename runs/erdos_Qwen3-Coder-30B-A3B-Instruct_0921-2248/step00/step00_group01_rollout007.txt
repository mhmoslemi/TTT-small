import numpy as np
from scipy.fft import fft, ifft
import time

def evaluate_erdos_solution(h):
    """Compute the C5 value for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum of the overlapped integral
    c5 = np.max(corr) * dx
    return c5

def project_h(h, dx):
    """Project h to [0,1] and scale to integral = 1"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize to integral = 1
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * 1.0
    return h

        # Compute gradient in frequency domain
        # f(x) = h(x), g(x) = 1 - h(x)
        # We want to minimize max_k int_0^2 h(x) * (1 - h(x+k)) dx
        # This is the same as max_k int_0^2 h(x) * (1 - h(x+k)) dx
        # Gradient wrt h: g(x) - Q(x) where Q is the overlap kernel

        # Compute fft(1-h) = 1 - fft(h)
        G = 1 - H

        grad_fft = H * G
