import numpy as np
import time
from scipy import signal

def evaluate_h(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    correlation = signal.correlate(h_values, 1 - h_values, mode='full')
    # The overlap for shift k is at index k + n_points - 1
    max_overlap = np.max(correlation)
    return max_overlap * dx

            iter_count += 1
            
        # Cooling schedule
