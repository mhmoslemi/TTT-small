        # Normalize to sum to n_points / 2
        h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def correlate(h_vec):
        """Correlate h with 1-h, return all overlaps."""
        # h_vec and 1-h_vec have length n_points
        # The full cross-correlation returns 2*n_points - 1 values
        # Compute overlap for each shift
        h_extended = np.concatenate([h_vec, h_vec[:n_points-1]])
        h_rev = h_vec[::-1]
        one_minus_h = 1 - h_vec
        one_minus_h_extended = np.concatenate([one_minus_h, one_minus_h[:n_points-1]])
        # Using numpy.correlate for speed
        # modes are 'full' - produces 2*n-1 values
        corr = np.correlate(h_extended, one_minus_h_extended, mode='full')
        # Normalize to get actual overlaps
        overlaps = corr * dx
        return overlaps
    
    def c5_bound(h_vec):
        """Return the maximum overlap of h with its shifted versions."""
        overlaps = correlate(h_vec)
        return np.max(overlaps)
    
    def project_h(h_vec):
        """Project h to feasible region: [0, 1] and sum = target_sum."""
        # Ensure h is in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Fix sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def project_h_and_normalize(h_vec):
        """Project h to [0,1] and normalize sum to target_sum."""
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = c5_bound(best_h)
    
    start_time = time.time()

        overlaps = correlate(best_h)

        grad = np.zeros_like(best_h)

        # So the gradient is:
        # np.roll(1 - best_h, k)

        for i in range(n_points):

            # we're computing: d/dh_j overlap_k = (1 - h_{j+k}) 
            # So for shift k:
            # Let's compute d/dh[0, 1, 2, ..., n-1] of the k-th overlap.
            # The overlap at shift k is sum_j h_j (1 - h_{j+k})
            # d/dh_i (overlap_k) = 1 - h_{i+k}
            
            # But we need the effect of each h_i on the whole k overlap
            # d/dh_i (sum_j h_j (1 - h_{j+k})) = (1 - h_{i+k})
            # i.e. the gradient for i th component is 1-h_{i+k}
            # So the gradient = 1 - shifted
            grad = 1 - np.roll(best_h, -max_k)
        
        # Gradient for overlap at shift k
        grad = 1 - np.roll(best_h, -max_k)
        # Perturb h to reduce the overlap at shift k
        # Really, let's reflect the instance in a gradient-descent like way:
        step_size = 0.01
        new_h = best_h - step_size * grad
        new_h = project_h_and_normalize(new_h)
        
        new_c5 = c5_bound(new_h)
        if new_c5 < best_c5:
            best_h = new_h

        # Re-benchmark best
        if iteration % 100 == 0:
            best_c5 = c5_bound(best_h)
    
    # Final correction to ensure sum constraint
    best_h = project_h_and_normalize(best_h)
    final_c5 = c5_bound(best_h)
    
    # Return best solution
    return (best_h.tolist(), float(final_c5), n_points)
