import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def objective(h, dx, n_points):
    return -evaluate_c5(h, dx)

def grad_objective(h, dx, n_points):
    # Approximate gradient via finite differences
    eps = 1e-8
    grad = np.zeros_like(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        grad[i] = (evaluate_c5(h_plus, dx) - evaluate_c5(h_minus, dx)) / (2 * eps)
    return -grad

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Default construction: 40 points
        n_points = 40
        h0 = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    
    # Define constraints
    cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
    
    # Define bounds (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up optimization
    start_time = time.time()
    
    # Initial solution
    h_best = h0.copy()
    c5_best = evaluate_c5(h0, dx)
    
    # Closure to keep track of best solution
    def callback(xk):
        nonlocal h_best, c5_best
        if time.time() - start_time > budget_s * 0.95:
            return True  # Stop optimization
        c5_current = evaluate_c5(xk, dx)
        if c5_current < c5_best:
            h_best = xk.copy()
            c5_best = c5_current
        return False
    
    # Initial evaluation
    c5_best = evaluate_c5(h0, dx)
    h_best = h0.copy()
    
    # To avoid many gradient evaluations, we use a couple of inner loops
    # but respect timebudget: optimize with small steps
    
    max_iter = 500
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Create a simple optimization step with trust-constr, in case of large steps
        # but use SLSQP to respect bounds, due to constraints being linear equality
        try:
            res = minimize(
                objective, 
                h_best.copy(), 
                method='SLSQP',
                jac=grad_objective,
                bounds=bounds,
                constraints=cons,
                callback=callback,
                options={'ftol': 1e-7, 'maxiter': 10, 'disp': False}
            )
            if res.success:
                h_best = res.x
                c5_best = -res.fun
            else:
                # If not successful, step back to last good solution
                pass
        except:
            pass

        iter_count += 1

    # Final validation
    if np.abs(np.sum(h_best) - n_points / 2.0) > 1e-6:
        # Re-project to satisfy constraint if needed
        sum_h = np.sum(h_best)
        h_best = h_best * (n_points / 2.0) / sum_h
        
    # Ensure bounds
    h_best = np.clip(h_best, 0, 1)
    
    # Final computation
    c5_final = evaluate_c5(h_best, dx)
    
    return (h_best.tolist(), float(c5_final), int(n_points))
