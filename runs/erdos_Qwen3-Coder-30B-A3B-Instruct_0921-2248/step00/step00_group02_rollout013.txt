def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h"""
    n = len(h)
    dx = 2.0 / n
    # correlation with 1-h
    conv = np.correlate(h, 1 - h, mode='full')
    # select only the relevant parts (the shifts 0 to 2)
    c5 = np.max(conv) * dx
    return c5

def optimize_h_for_c(h0, C_target, dx, n_points, start_time, budget_s):

        conv = np.correlate(h_vec, 1 - h_vec, mode='full')

        return C_target - max_overlap

        # Integral = 1
        return np.sum(h_vec) * dx - 1

        lower_diff = np.sum(np.maximum(0, 0 - h_vec))
        upper_diff = np.sum(np.maximum(0, h_vec - 1))
        return lower_diff + upper_diff

    bounds = [(0, 1) for _ in range(n_points)]

    constraints = [

    h = np.clip(h, 0, 1)
