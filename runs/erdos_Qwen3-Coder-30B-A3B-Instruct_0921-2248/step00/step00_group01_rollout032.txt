import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute C5 using full correlation
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum and scale by dx
    c5 = np.max(corr) * dx
    return c5

def objective_and_gradient(h, dx, n_points):
    # Compute the max correlation (C5)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr_index = np.argmax(corr)
    c5 = corr[max_corr_index] * dx
    
    # Compute gradient of the max correlation w.r.t. h
    # For the k-th shift with max overlap:
    k = max_corr_index - (n_points - 1)
    # Gradient in full correlation:
    grad = np.zeros_like(h)
    if k >= 0:
        # h[i] contributes to correlation with (1-h[i+k]) if i+k < n
        for i in range(min(n, n - k)):
            grad[i] += (1 - h[i + k]) * dx
        # (1-h[i+k]) contributes to correlation with h[i] if i < n-k
        for i in range(min(n, n - k)):
            grad[i + k] -= h[i] * dx
    else:
        # h[i+k] contributes to (1-h[i]) if i+k >= 0
        for i in range(max(0, -k), n):
            grad[i] += (1 - h[i + k]) * dx
        # (1-h[i]) contributes to h[i+k] if i+k >= 0
        for i in range(max(0, -k), n):
            grad[i + k] -= h[i] * dx
    
    return c5, grad
