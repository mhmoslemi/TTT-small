import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Binary construction
        n_points = 70
        h = np.zeros(n_points)
        # Place 1s evenly
        ones_count = n_points // 2
        for i in range(ones_count):
            h[i * (n_points // ones_count)] = 1.0
        # Normalize
        h = h * n_points / (2 * np.sum(h))
        assert np.isclose(np.sum(h), n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Compute all overlaps
        correlation = np.correlate(h, 1 - h, mode="full")
        # Adjust for the actual range in overlap
        full_overlap = correlation * dx
        max_idx = np.argmax(full_overlap)
        max_k = max_idx - (n_points - 1)
        
        # Gradient for the worst case shift
        # overlap_k = sum(h[i] * (1 - h[(i + max_k) % n_points]))
        grad_h = np.zeros_like(h)
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad_h[i] -= (1 - h[j])
        
        # Update with gradient descent
        step_size = 0.01
        new_h = h - step_size * grad_h
        
        # Project to [0,1] and normalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))
