import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    lag_step = 0.2
    lag_samples = np.arange(-2, 2 + lag_step/2, lag_step)
    
    # Initialize h from initial_h_values if available, otherwise create simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
        h[0] = 0.4
        h[1] = 0.6
    
    best_C5 = float('inf')
    best_h = h.copy()
    
    lags = set(lag_samples)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create the LP
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) * dx == 1,
            h_var >= 0,
            h_var <= 1,
            C5_var >= 0
        ]
        
        # Add overlap constraints for current lags
        for k in lags:
            k_idx = int(round(k / dx))
            if k_idx < 0:
                # shifted h from the left
                shifted_h = np.roll(h_var, -k_idx)
            else:
                # shifted h from the right
                shifted_h = np.roll(h_var, k_idx)
                
            if k_idx >= 0:
                # left shift = pad with zeros on right
                h_shifted = np.concatenate([shifted_h[:n_points-k_idx], np.zeros(k_idx)])
                ones_minus_h = np.concatenate([1 - h_var[k_idx:], np.zeros(k_idx)])
            else:
                # right shift = pad with zeros on left
                h_shifted = np.concatenate([np.zeros(-k_idx), shifted_h[-k_idx:]])
                ones_minus_h = np.concatenate([np.zeros(-k_idx), 1 - h_var[:-k_idx]])
            
            # integral ∫ h(x)(1 - h(x+k)) dx ≈ 
            # sum(h[i]*(1-h[i+k]) for all i) * dx
            overlap_expr = cp.sum(cp.multiply(h_var, (1 - ones_minus_h))) * dx
            
            constraints.append(overlap_expr <= C5_var)
        
        # Objective
        objective = cp.Minimize(C5_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                C5_new = C5_var.value
                
                if C5_new < best_C5:
                    best_C5 = C5_new
                    best_h = h_new.copy()
                    
                # Evaluate all overlaps
                h_ref = h_new.astype(float)
                h_ref = h_ref / np.sum(h_ref) * (n_points / 2)
                h_ref = np.clip(h_ref, 0, 1)
                corr = correlate(h_ref, 1 - h_ref, mode='full')
                lags_full = np.arange(-2, 2 + dx/2, dx)
                max_corr = np.max(corr)
                
                # Find worst lags to add
                all_lags = set()
                indices = np.where(corr > 0.9 * max_corr)[0]
                for idx in indices:
                    lag_val = lags_full[idx]
                    all_lags.add(lag_val)
                
                # Add top 10% of worst lags
                if len(all_lags) > 10:
                    sorted_lags = sorted(list(all_lags), key=lambda x: abs(x))
                    lags_to_add = set()
                    for i in range(min(10, len(sorted_lags)//2)):
                        lags_to_add.add(sorted_lags[i])
                        lags_to_add.add(sorted_lags[-(i+1)])
                    lags.update(lags_to_add)
        except:
            break
    
    if best_C5 == float('inf'):
        try:
            # Fallback to initial construction
            corr = correlate(initial_h_values, 1 - np.array(initial_h_values), mode='full')
            h_ref = np.array(initial_h_values)
            dx_local = 2.0 / len(h_ref)
            best_C5 = np.max(corr) * dx_local
            best_h = np.array(initial_h_values)
        except:
            pass
    
    # Final check
    h_final = np.array(best_h)
    h_final = np.clip(h_final, 0, 1)
    
    corr = correlate(h_final, 1 - h_final, mode='full')
    lags_full = np.arange(-2, 2 + dx/2, dx)
    final_C5 = np.max(corr) * dx
    
    return (list(h_final), final_C5, n_points)
