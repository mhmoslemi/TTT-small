import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial sum is correct and clip to [0,1]
    h = np.clip(h0, 0, 1)
    scale_factor = target_sum / np.sum(h)
    h *= scale_factor
    
    # Evaluate initial C5
    def evaluate_C5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    current_C5 = evaluate_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    # Convex optimization helper
    def solve_convex_subproblem(h_current, k_max):
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize C5 (max correlation)
        C5_var = cp.Variable()
        
        # Constraints:
        # Sum constraint
        sum_constr = cp.sum(h_vars) == target_sum
        
        # Bound constraints
        bound_constrs = [h_vars >= 0, h_vars <= 1]
        
        # Linearized constraint for lag k_max
        linearized = 0
        for i in range(n_points - k_max):
            linearized += (h_current[i] * (1 - h_current[i + k_max]) + 
                          (1 - h_current[i + k_max]) * (h_vars[i] - h_current[i]) - 
                          h_current[i] * (h_vars[i + k_max] - h_current[i + k_max]))
        corr_constr = linearized * dx <= C5_var
        
        # Additional constraints for other lags (to stabilize)
        for k in range(1, n_points):
            if k == k_max:
                continue
            linearized_k = 0
            for i in range(n_points - k):
                linearized_k += (h_current[i] * (1 - h_current[i + k]) + 
                               (1 - h_current[i + k]) * (h_vars[i] - h_current[i]) - 
                               h_current[i] * (h_vars[i + k] - h_current[i + k]))
            # Just enforce these as soft constraints
            corr_k_constr = linearized_k * dx <= C5_var + 0.01 * dx  # Soft constraint
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Solve
        prob = cp.Problem(objective, [sum_constr] + bound_constrs + [corr_constr])
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_vars.value
                return new_h, prob.value
        except:
            pass
        
        return h_current, current_C5
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s - 1:
        iter_count += 1
        if iter_count > 1000:
            break
            
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Check if we can improve
        if max_corr * dx < current_C5 - 1e-8:
            # Linearize at k_max and solve convex subproblem
            new_h, new_C5 = solve_convex_subproblem(h, k_max)
            if new_C5 < current_C5:
                h = new_h
                current_C5 = new_C5
                if current_C5 < best_C5 - 1e-8:
                    best_C5 = current_C5
                    best_h = h.copy()
                last_improvement = iter_count
        else:
            # No improvement from this linearization, try another
            # Randomly select a different lag if not close to max
            if iter_count - last_improvement > 20:
                k_max = np.random.choice(range(1, n_points))
                new_h, new_C5 = solve_convex_subproblem(h, k_max)
                if new_C5 < current_C5:
                    h = new_h
                    current_C5 = new_C5
                    if current_C5 < best_C5 - 1e-8:
                        best_C5 = current_C5
                        best_h = h.copy()
                    last_improvement = iter_count
        
        # Ensure h remains feasible
        h = np.clip(h, 0, 1)
        scale_factor = target_sum / np.sum(h)
        h = h * scale_factor
        
        # Early stopping if no improvement
        if iter_count - last_improvement > 50 and iter_count > 100:
            break
    
    # Final evaluation
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
