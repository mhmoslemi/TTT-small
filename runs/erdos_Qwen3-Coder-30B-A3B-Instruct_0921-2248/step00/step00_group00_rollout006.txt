def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h"""

    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value and multiply by dx
    c5 = np.max(corr) * dx
    return c5
