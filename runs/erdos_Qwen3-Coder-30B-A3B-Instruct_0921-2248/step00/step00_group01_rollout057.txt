import numpy as np
from scipy.optimize import minimize
import time

def evaluate_C5(h, dx):
    # Compute the overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx

def project_h(h, n_points):
    # Ensure h stays within [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum(h) * dx = 1
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def objective(h, n_points, dx):
    return evaluate_C5(h, dx)

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Create a shuffled binary vector
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}
    
    best_h = h.copy()
    best_c5 = evaluate_C5(h, dx)
    
    start_time = time.time()
    
    # SLSQP optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Run optimizer
        result = minimize(
            objective,
            h.copy(),
            args=(n_points, dx),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'disp': False, 'maxiter': 10, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
            h = project_h(h, n_points)
            current_c5 = evaluate_C5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Fallback to simple projection
            h = project_h(h, n_points)
            current_c5 = evaluate_C5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final validation
    best_h = project_h(best_h, n_points)
    final_c5 = evaluate_C5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
