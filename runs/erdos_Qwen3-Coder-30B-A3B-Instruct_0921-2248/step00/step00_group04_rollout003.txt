import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lag = 2.0
    lag_step = 0.1
    lags = np.arange(-max_lag, max_lag + lag_step/2, lag_step)
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Default construction
        h_init = np.ones(n_points) * 0.5
    
    # Optimization
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) * dx == 1, C5 >= 0]
    
    # Overlap constraints for each lag
    for k in lags:
        # Shift indices (handle wrapping)
        shift = int(np.round(k / dx))
        if shift >= 0:
            h_shifted = cp.hstack([h[shift:], h[:shift]])
        else:
            h_shifted = cp.hstack([h[shift:], h[:shift]])
        
        # Compute overlap integral approximation
        overlap = cp.sum(cp.multiply(h, 1 - h_shifted)) * dx
        constraints.append(overlap <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Handle solver timeout
    prob.solve(solver=cp.GLPK, time_limit=budget_s)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_solution = h.value
        c5_bound = C5.value
    else:

        h_solution = h_init

    elapsed = time.time() - start_time
    if elapsed > budget_s:

    return h_solution, c5_bound, n_points
