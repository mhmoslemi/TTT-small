import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for a given h."""
    # Compute full convolution
    conv = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral for each k is the convolution value
    # The maximum overlap is the max of these values
    max_overlap = np.max(conv) * dx
    return max_overlap
