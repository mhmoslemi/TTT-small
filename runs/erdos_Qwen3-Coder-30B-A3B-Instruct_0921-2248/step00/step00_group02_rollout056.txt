import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlations)
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Create a simple starting point
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        # Ensure sum(h) = n_points / 2
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    best_h = h0.copy()
    best_C = evaluate_erdos_solution(h0, n_points)
    
    t0 = time.time()
    
    # Binary search on C
    left = 0.0
    right = 0.38085
    tolerance = 1e-5
    
    while right - left > tolerance and time.time() - t0 < budget_s - 1:
        mid = (left + right) / 2.0
        
        # Try to find feasible h for this C
        result = minimize(
            fun=lambda x: 0,
            x0=h0,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - n_points / 2
            },
            options={'maxiter': 200}
        )
        
        # Compute the true overlap
        h_test = result.x if result.success else h0
        C_test = evaluate_erdos_solution(h_test, n_points)
        
        # If feasible, try to improve h
        if C_test <= mid + 1e-6:
            right = mid
            if C_test < best_C:
                best_C = C_test
                best_h = h_test.copy()
        else:
            left = mid
            
    # Final refinement of best solution
    h_final = best_h.copy()
    while time.time() - t0 < budget_s - 2:
        # Try hill climbing with small perturbations
        for i in range(n_points):
            if h_final[i] > 0.01 and h_final[i] < 0.99:
                h_trial = h_final.copy()
                delta = np.random.uniform(-0.05, 0.05)
                h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
                
                # Ensure integral constraint
                sum_h = np.sum(h_trial)
                scale_factor = (n_points / 2) / sum_h
                h_trial = h_trial * scale_factor
                
                C_trial = evaluate_erdos_solution(h_trial, n_points)
                if C_trial < best_C:
                    best_C = C_trial
                    h_final = h_trial.copy()
    
    return (list(h_final), best_C, n_points)
