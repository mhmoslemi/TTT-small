import numpy as np
from scipy import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default construction: symmetric tent function
        n_points = 200
        j = np.arange(-n_points//2, n_points//2)
        h = np.maximum(0, 1 - 2 * np.abs(j) / n_points)
        h = h / np.sum(h) * n_points / 2  # Normalize to sum = n_points/2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization iterations
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        overlap = correlate(h, 1-h, mode="full") * dx
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx]
        
        # Identify worst shifts
        shift_indices = np.where(np.abs(overlap - max_overlap) < 1e-8)[0]
        
        # Formulate and solve QP
        # Variables: h
        # Objective: minimize sum_{k in worst_shifts} sum h[i]*(1-h[i+k])
        # Constraints: 0 <= h[i] <= 1, sum h[i] = n_points/2
        
        # Create optimization variables
        h_vars = cp.Variable(n_points)
        
        # Objective - sum of overlaps for worst shifts
        obj = 0
        for k in shift_indices:
            # Sum over valid i: h[i]*(1-h[i+k]) where i+k is valid
            # We adjust indices for the correct indexing
            if k < n_points:
                # This corresponds to the k-th shift in the full convolution
                idx_upper = min(n_points, n_points - k)
                obj += cp.sum(cp.multiply(h_vars[:idx_upper], (1 - h_vars[k:k+idx_upper])))
            else:
                idx_lower = k - n_points
                obj += cp.sum(cp.multiply(h_vars[idx_lower:], (1 - h_vars[:n_points-idx_lower])))
                
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Solve QP
        try:
            prob = cp.Problem(cp.Minimize(obj), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                # Project onto feasible set with constraints
                h_new = np.clip(h_new, 0, 1)
                # Rescale to match constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Check if new solution is better
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                    
        except Exception:
            pass  # Fall back to current solution
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h_values, dx):
    h = np.array(h_values)
    overlap = correlate(h, 1-h, mode="full") * dx
    return np.max(overlap)
