import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    """Evaluate the overlap integral for a given h."""
    # Compute convolution of h and (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Normalize
    conv *= dx
    # Return max value
    return np.max(conv)

def is_feasible(h, n_points):
    """Check if h satisfies the constraints."""
    dx = 2.0 / n_points
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1.0) < 1e-10)

def perturb_h(h, n_points, seed=None):
    """Generate a perturbed version of h."""
    if seed is not None:
        random.seed(seed)
    h_new = h.copy()
    dx = 2.0 / n_points
    
    # Choose a random number of swaps (1-5)
    num_swaps = random.randint(1, min(5, n_points // 4))
    
    for _ in range(num_swaps):
        # Select two random indices
        i, j = random.sample(range(n_points), 2)
        # Swap values
        h_new[i], h_new[j] = h_new[j], h_new[i]
    
    # Add some small adjustment to a random point
    idx = random.randint(0, n_points - 1)
    delta = random.uniform(-0.1, 0.1)
    old = h_new[idx]
    h_new[idx] = max(0, min(1, old + delta))
    
    # Ensure sum constraint is roughly satisfied by adjusting one point
    current_sum = np.sum(h_new) * dx
    diff = 1.0 - current_sum
    if abs(diff) > 1e-10:
        # Adjust one point to fix the sum
        adj_idx = random.randint(0, n_points - 1)
        h_new[adj_idx] = max(0, min(1, h_new[adj_idx] + diff / dx))
    
    return h_new
