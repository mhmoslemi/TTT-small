import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n = len(h)
    # Compute cross-correlation for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts 0 to n-1)
    c5 = np.max(corr[n-1:])
    return c5 * dx

    target_sum = n_points / 2.0

    current_sum = np.sum(h)

    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)

    def constraint_sum(h_vec):

    # Bound constraints
    bounds = [(0, 1) for _ in range(n_points)]

    cons = {'type': 'eq', 'fun': constraint_sum}

    best_c5 = objective(h)

    max_iter = 1000

        try:
            res = minimize(
