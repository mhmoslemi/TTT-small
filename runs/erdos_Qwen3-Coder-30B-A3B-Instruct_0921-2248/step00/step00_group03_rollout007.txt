def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)

    overlap = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant parts (shifts 0 to n_points-1)
    overlaps = overlap[len(h)-1:]
    # Multiply by dx to get actual overlap value
    return np.max(overlaps) * dx
