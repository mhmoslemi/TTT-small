import numpy as np
from scipy import signal
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    h1 = 1 - h
    # Compute full correlation
    corr = signal.correlate(h, h1, mode='full')
    # The indices correspond to lags from -n+1 to n-1
    lags = np.arange(-n_points + 1, n_points)
    # Find maximum over all lags
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction: a simple symmetric step function
        n_points = 100
        h = np.ones(n_points) * 0.5  # Start with uniform
        dx = 2.0 / n_points
        
    # Start time measurement
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_iter = 0
    
    # Initial lag sampling
    lag_step = 0.2
    k_vals = np.arange(-2, 2 + lag_step/2, lag_step)
    current_lags = np.unique(np.round(k_vals * n_points / 2).astype(int))
    
    # Iteratively refine the constraints
    iter_count = 0
    patience = 5
    prev_best = float('inf')
    
    while True:
        if time.time() - start_time > budget_s - 1:
            break
            
        iter_count += 1
        if iter_count > 100:
            break  # safety break
            
        # Solve optimization problem with current constraints
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1]
        
        # Mean constraint
        constraints.append(cp.sum(h_var) * dx == 1.0)
        
        # Overlap constraints
        for k in current_lags:
            # Compute the convolution manually
            # For shift k, we compute sum h[i] * (1 - h[(i + k) % n_points])
            expr = 0
            for i in range(n_points):
                shift_idx = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[shift_idx])
            expr *= dx
            constraints.append(expr <= C5)
            
        # Minimize C5
        prob = cp.Problem(cp.Minimize(C5), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
            except:
                break  # Stop if we can't solve
                
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break
            
        if C5.value is None:
            break
            
        h_new = h_var.value
        if h_new is None:
            break
            
        # Compute actual C5
        h_new = np.clip(h_new, 0, 1)
        c5_new = evaluate_erdos_solution(h_new)
        
        # Compute all overlaps to find worst violations
        corr_full = signal.correlate(h_new, 1 - h_new, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        overlaps = corr_full * dx
        
        # Get the 10% worst overlaps
        sorted_indices = np.argsort(-overlaps)
        top_indices = sorted_indices[:max(1, len(sorted_indices) // 10)]
        
        # Add these lags to current constraints
        new_lags = set(k_vals)
        for i in top_indices:
            k_idx = lags[i]
            k_frac = k_idx * 2.0 / n_points  # To match the original scale
            new_lags.add(np.round(k_frac / lag_step) * lag_step)
        current_lags = np.unique(np.round(list(new_lags) * n_points / 2).astype(int))
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_iter = iter_count
            
        # Early stopping conditions
        if c5_new < best_c5:
            prev_best = best_c5
        elif abs(c5_new - prev_best) < 1e-5:
            patience -= 1
            if patience <= 0:
                break
                
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
