from scipy import optimize
import time

    def objective(h):
        dx = 2.0 / n_points
        # Compute overlaps for all shifts
        overlaps = []
        for k in range(n_points):
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps.append(overlap * dx)
        return max(overlaps)
    
    # Constraints
    def constraint_sum(h):
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0

    def constraint_lower(h):
        return h - 0.0

    def constraint_upper(h):
        return 1.0 - h
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_lower},
        {'type': 'ineq', 'fun': constraint_upper}
    ]
    
    best_h = h0.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Simple gradient-free method using Nelder-Mead
    # Starting from the initial solution
    result = optimize.minimize(
        objective, 
        best_h, 
        method='Nelder-Mead',
        options={'maxiter': 1000, 'disp': False}
    )
    
    if result.success:
        h_opt = result.x
        c5_opt = objective(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        h_pert = np.clip(h_pert, 0, 1)
        # Re-normalize
        h_pert = h_pert / np.sum(h_pert) * 2.0
        
        c5_test = objective(h_pert)
        if c5_test < best_c5:
            best_c5 = c5_test
