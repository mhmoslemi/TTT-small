import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Extract the valid correlations (not including the zero-padding)
    start = n_points - 1
    end = 2 * n_points - 1
    correlations = corr[start:end]
    # Find maximum correlation
    max_corr = np.max(correlations)
    return max_corr * dx

def generate_initial_h_values():
    """Generate a feasible starting solution"""
    n = 45
    h = np.ones(n) * 0.5
    # Make it a valid step function with integral 1
    h[0] = 0.0
    h[1] = 1.0
    h[2] = 0.0
    h[3] = 1.0
    h[4] = 0.0
    h[5] = 1.0
    h[6] = 0.0
    h[7] = 1.0
    h[8] = 0.0
    h[9] = 1.0
    # Ensure the total is 1
    total = np.sum(h)
    target = 45 / 2
    h = h * (target / total)
    return h

def make_feasible(h):
    """Ensure h stays in [0,1] and sum(h) = n/2"""
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust to meet the integral constraint
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-8:
        # Add the difference equally to avoid large changes
        h += diff / n_points
        h = np.clip(h, 0, 1)
    
    return h

def propose_move(h):
    """Propose a local move to improve h"""
    n_points = len(h)
    # Choose a random interval of 1 or 3 points to flip
    length = np.random.choice([1, 3])  # Length of interval to flip
    start = np.random.randint(0, n_points - length + 1)
    h_new = h.copy()
    # Flip interval [start, start+length) - using a symmetric operation
    h_new[start:start+length] = 1 - h_new[start:start+length]
    return h_new

def run(seed=42, budget_s=600, **kwargs):

    # Ensure feasibility
    h = make_feasible(h)
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = 0.1  # Initial temperature
    alpha = 0.999  # Cooling factor
    max_iter = 1000000
    iter_count = 0
    
    while iter_count < max_iter:
        # Check time budget
        if time.time() - start_time > budget_s:
            break
            
        # Propose a move
        h_proposed = propose_move(h)
        
        # Ensure proposal is feasible
        h_proposed = make_feasible(h_proposed)
        
        # Evaluate proposed solution
        proposed_c5 = evaluate_erdos_solution(h_proposed)
        
        # Accept or reject
        if proposed_c5 < current_c5:
            h = h_proposed
            current_c5 = proposed_c5
            if proposed_c5 < best_c5:
                best_h = h_proposed.copy()
                best_c5 = proposed_c5

            # Accept with probability based on temperature
            delta = proposed_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_proposed
                current_c5 = proposed_c5
                
        # Cool down
        T *= alpha
        
        iter_count += 1
    
    # Final validation
    best_h = make_feasible(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
