import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Problem parameters
    dx = 2.0 / n_points
    n_shifts = 2 * n_points - 1
    
    # Create problem
    h = cp.Variable(n_points, name="h")
    C = cp.Variable(name="C")
    
    # Constraints
    constraints = []
    
    # Bound constraints
    constraints += [h >= 0]
    constraints += [h <= 1]
    
    # Sum constraint
    constraints += [cp.sum(h) == n_points / 2]
    
    # Overlap constraints for all shifts
    for k in range(-n_points + 1, n_points):
        # Indices for current shift
        i_range = np.arange(n_points)
        j_range = (i_range + k) % n_points
        
        # Compute overlap
        overlap = cp.sum(cp.multiply(h[i_range], 1 - h[j_range]))
        constraints += [overlap <= C / dx]
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    solved = False
    best_h = None
    best_C = float('inf')
    
    try:
        # Set solver options
        prob.solve(solver=cp.ECOS, max_iters=200, verbose=False, feastol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            h_val = h.value
            C_val = C.value
            if not np.isnan(C_val) and not np.isinf(C_val):
                if C_val < best_C:
                    best_C = C_val
                    best_h = h_val.copy()
                    solved = True
    except Exception as e:
        print(f"CVXPY solver error: {e}")
    
    # Fallback: try warm start from a known construction if needed
    if not solved:
        h_val = np.array(h0)
        # Clip to feasible range
        h_val = np.clip(h_val, 0, 1)
        # Normalize
        h_val = h_val * (n_points / 2) / np.sum(h_val)
        # Final check and validation
        if np.all(h_val >= 0) and np.all(h_val <= 1) and abs(np.sum(h_val) - n_points / 2) < 1e-6:
            best_h = h_val
            # Evaluate this solution manually
            cross_corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
            best_C = np.max(cross_corr)
    
    if best_h is None:
        # Default approach if all fails: initialize and evaluate
        h_val = np.array(h0)
        h_val = np.clip(h_val, 0, 1) 
        h_val = h_val * (n_points / 2) / np.sum(h_val)
        best_h = h_val
        cross_corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
        best_C = np.max(cross_corr)
    
    # Final constraints validation
    if not (np.all(best_h >= 0) and np.all(best_h <= 1) and 
            abs(np.sum(best_h) - n_points / 2) < 1e-6):
        # Rerun or adjust
        print("Warning: incorrect feasibility")
        h_val = np.clip(best_h, 0, 1)
        h_val = h_val * (n_points / 2) / np.sum(h_val)
        best_h = h_val
        cross_corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
        best_C = np.max(cross_corr)
    
    # Return the best solution found
    return (best_h.tolist(), best_C, n_points)
