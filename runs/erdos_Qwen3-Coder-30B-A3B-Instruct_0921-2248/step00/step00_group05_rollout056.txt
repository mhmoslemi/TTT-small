import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Tent function
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x <= 1:
                h[i] = x
            else:
                h[i] = 2 - x
    
    # Normalize
    h = h / np.sum(h) * target_sum
    h = np.clip(h, 0, 1)
    
    # Precompute FFT of h and (1-h)
    H = fft(h)
    H_conj = np.conj(H)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Optimization loop
    iteration = 0
    patience = 0
    max_patience = 50
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        iteration += 1
        
        # Compute overlaps in frequency domain
        GH = fft(1 - h)
        overlap_fft = H * GH
        
        # Convert back to time domain for overlap computation
        overlap_time = ifft(overlap_fft).real
        max_shift = np.argmax(overlap_time)
        
        # Gradient-based update - focus on worst overlap shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            shift = (i - max_shift) % n_points
            if shift < n_points // 2:
                # Positive shift
                grad[i] = (1 - h[(i + shift) % n_points]) - h[i]
            else:
                # Negative shift
                grad[i] = (1 - h[(i - shift + n_points) % n_points]) - h[i]
        
        # Apply gradient with adaptive step size
        step_size = 0.01 / (1 + iteration / 100)
        h_new = h - step_size * grad
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_sum
        
        # Evaluate new solution
        new_C5 = compute_C5(h_new, dx)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            
        # Use new h for next iteration
        h = h_new
        
        # Periodic cleanup: keep bounds tight
        if iteration % 20 == 0:
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * target_sum
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Use numpy convolution
    correlations = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(correlations) * dx
    return C5
