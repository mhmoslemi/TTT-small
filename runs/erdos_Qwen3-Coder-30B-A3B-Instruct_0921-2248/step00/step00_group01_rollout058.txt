import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, n_points):
    # This function is pre-imported, but we implement a basic version to confirm correctness
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 200
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure the sum constraint
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a binary vector with half ones and half zeros then shuffle
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode="full")
        return -np.max(correlation) * dx  # Negative because we minimize
    
    # Optimization with time limit
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Run SLSQP optimization
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
        
        if not res.success:
            break
            
        h = res.x
        # Project to bounds
        h = np.clip(h, 0, 1)
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
