import numpy as np
import random
import time
from scipy import correlate

def evaluate_c5(h):
    """Compute the C5 bound for a given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap for all shifts
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def initialize_h(n_points):
    """Initialize a binary h vector with sum = n_points/2."""
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    return h

def get_neighbor(h):
    """Generate a neighbor by swapping two random 0 and 1."""
    h_copy = h.copy()
    zeros = np.where(h_copy == 0)[0]
    ones = np.where(h_copy == 1)[0]
    if len(zeros) == 0 or len(ones) == 0:
        return h_copy
    i = random.choice(zeros)
    j = random.choice(ones)
    h_copy[i] = 1
    h_copy[j] = 0
    return h_copy

    assert np.sum(h_current) == n_points // 2
    
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current)
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    min_T = 1e-6
    
    # Start optimization loop
    while time.time() - start_time < budget_s:
        # Generate neighbor
        h_neighbor = get_neighbor(h_current)
        
        # Evaluate
        c5_neighbor = evaluate_c5(h_neighbor)

        delta = c5_neighbor - best_c5
        if delta < 0 or random.random() < np.exp(-delta / T):
            h_current = h_neighbor
            if c5_neighbor < best_c5:
                best_h = h_neighbor.copy()
                best_c5 = c5_neighbor
        
        # Cooling
        T *= alpha
        if T < min_T:
            T = min_T
            
    return (list(best_h), best_c5, n_points)
