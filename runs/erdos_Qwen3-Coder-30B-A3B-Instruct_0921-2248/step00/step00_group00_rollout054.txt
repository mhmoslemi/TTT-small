import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    # Evaluate the C5 value for the given h
    n = len(h)
    dx = 2.0 / n
    # Compute the maximum overlap integral across all shifts
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Start with initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure bounds and integral constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    start_time = time.time()
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Parameters
    T0 = 0.1
    alpha = 0.999
    T = T0
    max_iter = 1_000_000
    iter_count = 0
    
    # Simulated annealing main loop
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Generate a neighbor by flipping interval of length 1 or 3
        new_h = h.copy()
        flip_length = random.choice([1, 3])
        start_idx = random.randint(0, n_points - flip_length)
        
        # Flip the values in the interval
        for i in range(start_idx, start_idx + flip_length):
            new_h[i] = 1 - new_h[i]
        
        # Ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h)
        
        # Accept or reject based on SA criteria
        if new_c5 < current_c5:
            h = new_h
            current_c5 = new_c5
        else:
            # Accept worse solution with probability
            delta = new_c5 - current_c5
            if random.random() < np.exp(-delta / T):
                h = new_h
                current_c5 = new_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cool down
        T *= alpha
        
        # Early stopping if improvement is too slow
        if iter_count % 1000 == 0:
            elapsed = time.time() - start_time
            if elapsed > 0.9 * budget_s:
                break
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
