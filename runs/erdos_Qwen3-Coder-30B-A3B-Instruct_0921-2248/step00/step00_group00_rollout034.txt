import numpy as np

import time

def evaluate_erdos_solution(h):
    """Evaluates the Erdős overlap constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # The maximum correlation is at the center of the full correlation
    max_k = int((len(corr) - 1) / 2)
    return max(corr) * dx
