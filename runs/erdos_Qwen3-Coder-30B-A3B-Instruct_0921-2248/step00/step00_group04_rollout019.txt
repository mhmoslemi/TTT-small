import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h, dx):
    """Evaluate C5 for given h using the full convolution."""
    h_padded = np.concatenate([h, np.zeros(len(h) - 1)])
    h_flip = np.flip(h)
    corr = correlate(h_padded, h_flip, mode="full")
    # corr is now of size 2*len(h) - 1
    # The first len(h) values are from k=0 to k=len(h)-1
    # We want from k=0 to k=2 (the full range)
    full_corr = np.zeros(2*len(h) - 1)
    full_corr[:len(h)] = corr[:len(h)]
    full_corr[len(h):] = corr[len(h)-1:0:-1]
    # k = 0,1,...,2*len(h)-2
    # We want k from -len(h)+1 to len(h)-1?
    # No! The original problem is with k = 0, ..., 2 (i.e., negative shifts
    # are also considered! The overlap integral:
    # C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    # So we look at all possible k in [0,2] in steps of dx
    # Our k values are [0., 1., 2.]
    # k_index = 1., 2., 3. in full_corr (which has indices 0, ..., 2*len(h)-2)
    # Older indices (from the left-pointer) correspond to negative k
    # So for k=0, we take i=0 to len(h)-1 (indices 0 to len(h)-1)
    # k=1, indices 1 to len(h) -> corr[1] to corr[len(h)]
    # k=2, indices 2 to len(h)+1 -> corr[2] to corr[len(h)+1]
    # So we have to process all lags in [0,2] and take max
    # Full convolution is computed over 2*len(h) - 1 elements
    # With indexing from 0 to 2*len(h)-2
    # Compute maximum of layer 0 to 2 
    # For floating point k, we interpolate.
    # Usually, grid spacing = dx = 2/n
    n = len(h)
    max_val = 0
    # k = 0 to 2 with step = dx
    for k in np.arange(0, 2+1e-10, dx):
        # correlation of h with (1-h) with shift k (how?
        # The kernel 1-h is also indexed from 0 to n-1
        # So we do convolution of h with 1-h, with positive shift
        # That corresponds to indices i and i+k
        # idx = k/dx (steps) 
        # val = convolution(k)
        # So we calculate it more directly
        # Let us actually compute overlap for discrete k
        # Just use correlation or compute sum directly
        # If k is between two points then interpolate
        k_idx = k / dx
        k_int = int(k_idx)
        k_frac = k_idx - k_int
        if k_int < len(h) - 1 and k_int + 1 < 2*len(h) - 1:
            val0 = corr[k_int]
            val1 = corr[k_int + 1]
            val = val0 * (1 - k_frac) + val1 * k_frac
        elif k_int < len(h) - 1:
            val = corr[k_int]
        elif k_int >= len(h) - 1:
            val = corr[k_int]
        if val > max_val:
            max_val = val
    return max_val * dx


def evaluate_c5_exact(h, dx):
    max_val = 0.
    for i in range(len(h)):
        for k in range(len(h)):
            j = (i + k) % len(h)
            # h[i] * ( 1 - h[j] )
            overlap = h[i] * (1.0 - h[j])
            # now compute the corresponding integral term
            # which is d_x for sampling
            if overlap > max_val:
                max_val = overlap
    return max_val * dx


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Use initial solution if provided
    # We need h as a Vector in [0,1] with sum = 1, and length n
    n_points = 42  # Start with the known one
    dx = 2.0 / n_points
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n_points) / 2
    
    # Normalize sum to be 1.0 exactly
    h_sum = np.sum(h) * dx
    h = h * (1.0 / h_sum)

    # Signifcantly start from a coarse grid
    max_k = 2.0
    step_k = 0.1
    k_vals = np.arange(0, max_k + step_k/2, step_k)  # e.g., [0, 0.1, ..., 2]
    
    start_time = time.time()
    
    # Hyperparameters
    max_iter = 50
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0.0
    
    iterations = 0
    while time.time() - start_time < budget_s - 5 and iterations < max_iter:
        # Compute overlaps for all k (regularized/grid scan)
        k_lags = k_vals[1:]  # leave out 0 for now
        if len(k_lags) == 0:
            break
        iteration_start = time.time()
        
        # Old evaluation method - slow but precise for full overlap
        # Instead we approximate the continuous overlap for this specific purpose
        # Since the resource bounds are tight, compute a reasonably accurate upper bound on C5
        # For a fixed h, find max_k ∫ h(x)(1 - h(x+k)) dx
        # We note that the overlap only depends on the convolution at lag k, so:
        # This can be expressed as a continuous integral of product over x.
        # But we are using a discretized version, so just compute max over grid
        
        c5_vals = []
        for k in k_lags:
            # k is in [0,2] in steps of 0.1
            k_index = int(k / dx)  # nearest index
            left_index = k_index
            right_index = k_index + 1
            
            if right_index < 2 * n_points - 1: # ensure we don't go out-bounds
                # interpolate the overlap
                k_frac = k / dx - k_index
                # This is a messy operation; we simulate a correlation-based
                # but properly for discrete h.
                # Let us compute overlap in a slightly better way
                overlap = 0.0
                for i in range(n_points):
                    j = (i + k_index) % n_points
                    overlap += h[i] * (1 - h[j])
                    # and for the right index
                    if right_index < n_points :
                        j_right = (i + right_index) % n_points
                        # interpolate
                        inter = overlaps = h[i] * (1 - h[j]) * (1-k_frac) + h[i] * (1 - h[j_right]) * k_frac
                        overlap = inter
                overlap *= dx
                c5_vals.append(overlap)
            else:
                # handle boundary
                overlap = 0.0
                for i in range(n_points):
                    j = (i + k_index) % n_points
                    overlap += h[i] * (1 - h[j])
                overlap *= dx
                c5_vals.append(overlap)

        # Use a coarse version, snapshot all true overlaps rather than grid
        # Let's compute true correlation at all lags up to 2 * n_points - 1
        h_flip = np.flip(h)
        h_padded = np.concatenate([h, np.zeros(len(h) - 1)])
        corr = correlate(h_padded, h_flip, mode="full")
        overlap_vals = [0.0] * len(k_lags)
        for i, k in enumerate(k_lags):
            idx = int(k / dx)  # this will curve possibly from DX scaling
            if idx < len(corr):
                overlap_vals[i] = corr[idx] * dx
            else:
                overlap_vals[i] = 0.0
        max_val = max(overlap_vals) if overlap_vals else 0.0
        max_val = max_val * 2.0  # estimate is too small for high k, so scale up
        max_val = max(max_val, max(overlap_vals))
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()

        # Now see which lags give the worst results, and add those as constraints
        if max_val > 0.3808:  # early break if already worse than target
            break

        # Simple fixed-step improvement now
        h_orig = h.copy()
        h_r = 0.1 * (1.0 - h)
        h_new = h - 0.01 * h_r  # decrease h where 1-h is large
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (1.0 / (np.sum(h_new) * dx))  # normalize
        h = h_new
        iterations += 1
    
    # Final refinement
    h_refined = h.copy()
    for _ in range(5):
        # Perform some refinement steps
        pass

    # Compute final value
    final_c5 = best_c5
    return (best_h, final_c5, n_points)
