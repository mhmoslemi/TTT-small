import numpy as np
import random
import time
from scipy import signal

def evaluate_solution(h):
    # Compute the maximum overlap integral
    h = np.array(h)
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h) 
    corr = signal.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points // 2  # sum(h) should be exactly n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h = (initial_h > np.percentile(initial_h, 50)).astype(int)
        # Make sure sum equals target_sum
        current_sum = np.sum(h)
        if current_sum > target_sum:
            # Zero out some 1s
            ones_indices = np.where(h == 1)[0]
            to_zero = current_sum - target_sum
            zero_indices = np.random.choice(ones_indices, to_zero, replace=False)
            h[zero_indices] = 0
        elif current_sum < target_sum:
            # Set some 0s to 1s
            zeros_indices = np.where(h == 0)[0]
            to_one = target_sum - current_sum
            one_indices = np.random.choice(zeros_indices, to_one, replace=False)
            h[one_indices] = 1
    else:
        # Initialize randomly
        h = np.zeros(n_points, dtype=int)
        ones_indices = np.random.choice(n_points, target_sum, replace=False)
        h[ones_indices] = 1
    
    # Initial evaluation  
    current_c5 = evaluate_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    max_iter = 100000
    iter_per_temp = max_iter // 1000  # adjust cooling schedule
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Generate neighbor by swapping one 0 and one 1
        zeros_indices = np.where(h == 0)[0]
        ones_indices = np.where(h == 1)[0]
        
        if len(zeros_indices) == 0 or len(ones_indices) == 0:
            # This should not happen if we maintain valid constraint
            continue
            
        i0 = np.random.choice(zeros_indices)
        i1 = np.random.choice(ones_indices)
        
        # Create new solution
        new_h = h.copy()
        new_h[i0] = 1
        new_h[i1] = 0
        
        # Evaluate new solution
        new_c5 = evaluate_solution(new_h)
        
        # Accept with probability
        delta = new_c5 - current_c5
        if delta < 0 or random.random() < np.exp(-delta / T):
            h = new_h
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Cooling schedule
        if iteration % iter_per_temp == 0:
            T *= alpha
    
    return (list(best_h), best_c5, n_points)
