    # Variable for h
    h_vars = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) * dx == 1,
        C5_var >= 0
    ]
    
    # Lag constraints
    for k in lags:
        # Compute shifted indices
        k_int = int(np.round(k / dx))
        if k_int > 0:
            shifted = np.concatenate([h_vars[k_int:], h_vars[:k_int]])

            k_int = -k_int

        else:
            shifted = h_vars
        # Overlap integral approximation

        constraints.append(integral <= C5_var)
    
    # Objective
    objective = cp.Minimize(C5_var)
    prob = cp.Problem(objective, constraints)
    
    # Solve

        prob.solve(solver=cp.GLPK, verbose=False)
