import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lag = 2.0
    lag_step = 0.1
    lag_samples = np.arange(-max_lag, max_lag + lag_step/2, lag_step)

    # Formulate optimization problem
    h = cp.Variable(n_points)
    C5 = cp.Variable()

    # Objective
    objective = cp.Minimize(C5)

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) * dx == 1,
        C5 >= 0
    ]

    # Add overlap constraints for each lag
    for k in lag_samples:
        if time.time() - start_time > budget_s * 0.95:
            break
        k_idx = int(round(k / lag_step))
        if k_idx < 0:
            # Handle negative shift
            shifted_h = np.concatenate([h[k_idx:], h[:k_idx]])
        elif k_idx > 0:
            # Handle positive shift
            shifted_h = np.concatenate([h[-k_idx:], h[:-k_idx]])
        else:
            shifted_h = h
            
        # Compute overlap integral approximation
        overlap = cp.sum(cp.multiply(h, 1 - shifted_h)) * dx
        constraints.append(overlap <= C5)

    # Problem setup
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
