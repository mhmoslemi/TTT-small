import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.001
    alpha_decay = 0.99
    tolerance = 1e-6
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    # Normalize h to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        full_corr = correlate(h, 1 - h, mode="full") * dx
        C5 = np.max(full_corr)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Find the worst shift
        k_max = np.argmax(full_corr)
        k_max -= (n_points - 1)  # Convert to actual shift
        
        # Compute gradient signs
        gradient_signs = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k_max) % n_points
            gradient_signs[j] = np.sign(1 - h[idx] - h[j])
        
        # Update h
        h = h - alpha * gradient_signs
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptively decrease alpha
        if i % 50 == 0:
            alpha *= alpha_decay
    
    # Final validation
    h = best_h.copy()
    full_corr = correlate(h, 1 - h, mode="full") * dx
    final_C5 = np.max(full_corr)
    
    return (h.tolist(), final_C5, n_points)
