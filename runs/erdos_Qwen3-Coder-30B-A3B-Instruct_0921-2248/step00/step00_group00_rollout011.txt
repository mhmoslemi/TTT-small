import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations for different shifts
    corr = correlate(h, 1 - h, mode="full")
    # Max of all correlations scaled by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure bounds and constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    # Initial evaluation
    C5 = evaluate_erdos_solution(h)
    best_C5 = C5
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.999
    T = T0
    max_iter = 1_000_000
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try a move: flip intervals
        # Choose interval length (1 or 3)
        length = np.random.choice([1, 3])
        # Choose start position
        start_idx = np.random.randint(0, n_points - length + 1)
        
        # Create candidate h'
        h_new = h.copy()
        
        # Flip the interval (which maintains sum constraint)
        for i in range(start_idx, start_idx + length):
            h_new[i] = 1 - h_new[i]
        
        # Ensure feasible (clip and re-scale)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate
        C5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject
        if C5_new < C5:
            h = h_new
            C5 = C5_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        else:
            # Metropolis acceptance
            delta = C5_new - C5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                C5 = C5_new
        
        # Cool temperature
        T = T * alpha
    
    # Final evaluation and return
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
