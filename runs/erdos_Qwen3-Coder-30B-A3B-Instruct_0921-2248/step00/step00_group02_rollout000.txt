import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute overlap integrals for all shifts
    conv = signal.correlate(h, 1 - h, mode='full')
    # Normalize by dx
    conv *= dx
    # Return maximum overlap
    return np.max(conv)

def check_feasibility(h, n_points):
    """Check constraints: sum = n_points/2, bounds in [0,1]"""
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    return np.isclose(sum_h, 0.5, atol=1e-6) and np.all(h >= 0) and np.all(h <= 1)

def perturb_h(h, n_points, swap_count=1):
    """Generate a perturbed version"""
    h_new = h.copy()
    dx = 2.0 / n_points
    
    # Randomly select indices to swap
    indices = list(range(n_points))
    random.shuffle(indices)
    swap_indices = indices[:swap_count]
    
    for i in swap_indices:
        h_new[i] = 1 - h_new[i]
        
    # Ensure sum constraint (adjust one element)
    old_sum = np.sum(h) * dx
    new_sum = np.sum(h_new) * dx
    diff = old_sum - new_sum
    # Adjust last element to maintain sum
    if abs(diff) > 1e-10:
        h_new[-1] = min(1, max(0, h_new[-1] - diff/dx))
    
    return h_new

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    best_feasible = check_feasibility(h, n_points)
    
    # Perturbations loop
    iteration = 0
    max_iter = 100000  # Just in case, but will exit early due to timeout
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        # Generate perturbed h
        h_new = perturb_h(h, n_points, swap_count=1)
        c5_new = evaluate_h(h_new, dx)
        
        # Check feasibility
        is_feasible = check_feasibility(h_new, n_points)

            best_h = h_new.copy()
            best_c5 = c5_new
            best_feasible = True
            
        # Continue with new h (random walk step)
        h = h_new.copy()
    
    if not best_feasible:
        # Fallback to the original if no feasible solution found
        best_h = h.copy()
        best_c5 = evaluate_h(h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
