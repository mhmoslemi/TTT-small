import numpy as np
import cvxpy as cp
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5  # uniform initial guess
    
    # Define the optimization variables
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Set constraints
    constraints = []
    
    # Bound constraints
    constraints += [h >= 0]
    constraints += [h <= 1]
    
    # Sum constraint
    constraints += [cp.sum(h) == n_points / 2]
    
    # Overlap constraints for all shifts
    # We compute all cross-correlations using the full mode
    # and manually extract them for each shift
    full_overlap = cp.convolution(h, 1 - h, mode='full')
    
    # The length of the full convolution is 2*n_points - 1
    # The indices from -n_points+1 to n_points-1 correspond
    for k in range(-n_points+1, n_points):
        # Extract corresponding element from full overlap
        # For shift k, the valid indices are:
        # when k >= 0, it means shift h forward by k
        # when k < 0, it means shift h backward by -k
        # The overlap sum is at index (k + n_points - 1) in full_overlap
        idx = k + n_points - 1
        if 0 <= idx < len(full_overlap):
            # Normalize by dx to match the evaluator's scaling
            constraints += [full_overlap[idx] <= C / dx]
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try solving with ECOS
    try:
        prob.solve(solver=cp.ECOS, max_iters=200, verbose=False, time_limit=budget_s)
    except Exception:
        try:
            prob.solve(solver=cp.CVXOPT, max_iters=200, verbose=False, time_limit=budget_s)
        except Exception:
            # Fallback to coordinate descent for potential convergence
            pass
    
    # Extract solution
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_sol = h.value
        c5_sol = C.value
    else:
        # Fallback to initial solution if optimization doesn't succeed
        h_sol = h0
        # Evaluate the overlap integral for the initial solution
        overlaps = []
        for k in range(-n_points+1, n_points):
            cor = correlate(h0, 1-h0, mode='full')
            idx = k + n_points - 1
            if 0 <= idx < len(cor):
                overlaps.append(cor[idx] * dx)
        c5_sol = max(overlaps)
    
    # Make sure we return a valid solution
    if h_sol is None:
        h_sol = h0
        
    # Ensure h_sol satisfies constraints
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
    
    # Final check with evaluation
    overlaps = []
    for k in range(-n_points+1, n_points):
        cor = correlate(h_sol, 1-h_sol, mode='full')
        idx = k + n_points - 1
        if 0 <= idx < len(cor):
            overlaps.append(cor[idx] * dx)
    final_c5 = max(overlaps)
    
    # Return the best result available
    return (h_sol.tolist(), final_c5, n_points)
