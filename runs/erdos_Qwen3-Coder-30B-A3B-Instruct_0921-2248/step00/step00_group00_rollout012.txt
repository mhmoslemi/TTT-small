import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45  # From initial_h_values
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population_size = 100
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        population.append(h.copy())
    else:
        # Create random feasible solution
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        population.append(h.copy())
    
    # Fill rest of population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        population.append(h.copy())
    
    # Optimization variables
    best_h = None
    best_c5 = float('inf')
    stagnant_generations = 0
    max_stagnant = 500
    mutation_rate = 0.01

    def evaluate_c5(h):
        # Compute cross-correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        # Take maximum correlation value and scale by dx
        max_corr = np.max(corr)
        return max_corr * dx

    def is_feasible(h):
        return (np.all(h >= 0) and np.all(h <= 1) and 
                abs(np.sum(h) - target_sum) < 1e-8)

    def mutate(h):
        h = h.copy()
        num_mutations = max(1, int(mutation_rate * len(h)))
        indices = np.random.choice(len(h), num_mutations, replace=False)
        for idx in indices:
            # Simple Gaussian mutation
            delta = np.random.normal(0, 0.05)
            h[idx] += delta
            h[idx] = np.clip(h[idx], 0, 1)
        
        # Renormalize
        h = h * (target_sum / np.sum(h))
        return h

    def crossover(h1, h2):
        # Uniform crossover
        mask = np.random.rand(len(h1)) > 0.5
        h_new = np.where(mask, h1, h2)
        # Renormalize
        h_new = h_new * (target_sum / np.sum(h_new))
        return h_new

    def tournament_selection(pop, k=5):
        k = min(k, len(pop))
        indices = np.random.choice(len(pop), k, replace=False)
        fitnesses = [evaluate_c5(pop[i]) for i in indices]
        best_idx = indices[np.argmin(fitnesses)]
        return pop[best_idx]

    generation = 0
    while time.time() - start_time < budget_s - 1 and stagnant_generations < max_stagnant:
        # Evaluate population
        fitness = [evaluate_c5(h) for h in population]
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitness)].copy()
            stagnant_generations = 0
        else:
            stagnant_generations += 1
            
        # Create new population
        new_population = [best_h.copy()]  # Elitism
        
        # Selection, crossover, mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            
            if np.random.rand() < 0.7:  # Crossover probability
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
                
            child = mutate(child)
            new_population.append(child)
            
        population = new_population[:population_size]
        generation += 1

    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h)
    else:
        # If no improvement, just evaluate first member
        final_c5 = evaluate_c5(population[0])
        best_h = population[0].copy()
        
    return (best_h.tolist(), final_c5, n_points)
