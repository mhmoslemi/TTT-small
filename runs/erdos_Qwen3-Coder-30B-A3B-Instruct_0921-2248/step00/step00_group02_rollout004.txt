import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Create a simple initial construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Evaluate initial C5
    def evaluate_c5(h_vec):
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Normalize by dx and take max
        max_val = np.max(conv) * dx
        return max_val
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Create new candidate via perturbation
        candidate_h = best_h.copy()
        
        # Try a combination of perturbations:
        # 1. Flip two points between 0 and 1
        if np.random.rand() < 0.5 and np.sum(candidate_h > 0) > 1 and np.sum(candidate_h < 1) > 1:
            # Find positions where we can flip
            zeros = np.where(candidate_h == 0)[0]
            ones = np.where(candidate_h == 1)[0]
            if len(zeros) > 0 and len(ones) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                candidate_h[i] = 0
                candidate_h[j] = 1
                
        # 2. Adjust a few points with small noise
        elif np.random.rand() < 0.5:
            n_adjust = np.random.randint(1, min(5, n_points//4))
            indices = np.random.choice(n_points, size=n_adjust, replace=False)
            # Adjust selected points with small random changes
            for idx in indices:
                delta = np.random.normal(0, 0.05)
                candidate_h[idx] = np.clip(candidate_h[idx] + delta, 0, 1)
                
        # 3. Check if sum constraint is kept and correct if necessary
        current_sum = np.sum(candidate_h)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Adjust the last element to maintain sum
            candidate_h[-1] += (n_points/2 - current_sum)
            candidate_h[-1] = np.clip(candidate_h[-1], 0, 1)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate_h)
        
        # Update best if improved
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            
        # Early stop if we are close to target
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
