```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h, dx):
    # compute the maximum overlap for all lags k
    h_rev = 1 - h
    corr = correlate(h, h_rev, mode='full')
    # only the valid lags in [0, 2]
    n = len(h)
    k_indices = range(n - 1, 2 * n - 1)
    max_val = 0
    for i, k in enumerate(k_indices):
        if k >= n - 1 and k <= n + 1:  # within [0,2]
            val = corr[i] * dx
            if val > max_val:
                max_val = val
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    dx = 2.0 / n_points
    
    # Initial construction from parent
    try:
        h0 = np.array(initial_h_values)
        h0 /= np.sum(h0) * dx
    except:
        # fallback: a simple center spike
        h0 = np.zeros(n_points)
        center = n_points // 2
        h0[center] = 1.0 / dx  # scaled to sum to 1
    
    h_opt = h0.copy()
    current_c5 = evaluate_c5(h0, dx)
    
    # Generate initial k values
    k_step = 0.05
    k_vals = np.arange(0, 2 + k_step/2, k_step)
    k_vals = k_vals[k_vals <= 2]
    
    # Adaptive exclusion of lags k
    max_iter = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # solve optimization with current k_vals
        h = Variable(n_points)
        C5 = Variable()
        
        # constraints
        constraints = []
        constraints.append(sum(h) == 1.0 / dx)  # integral constraint
        for i in range(n_points):
            constraints.append(0 <= h[i])
            constraints.append(h[i] <= 1)
            
        for k in k_vals:
            # compute correlations for lag k
            corr = 0
            for i in range(n_points):
                j = (i - int(k/dx)) % n_points
                corr += h[i] * (1 - h[j])
            constraints.append(corr * dx <= C5)
            
        # objective
        objective = Minimize(C5)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = np.array([float(h[i].value) for i in range(n_points)])
                new_c5 = float(C5.value)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    h_opt = new_h.copy()
            else:
                break
        except:
            break
            
        # Now recompute C5 values for all lags to see worst ones
        # Create a broader range of k values to check, mostly those in [0,2]
        refined_k = np.linspace(0, 2, 100)
        max_overlaps = []
        for k in refined_k:
            val = 0
            for i in range(n_points):
                j = (i - int(k/dx)) % n_points
                val += h_opt[i] * (1 - h_opt[j])
            max_overlaps.append(val * dx)
        
        # add the worst 10 k values to constraints
        worst_k_indices = np.argsort(max_overlaps)[-10:]
        k_vals = refined_k[worst_k_indices]
        
        if current_c5 < 0.3807:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(h_opt, dx)
    return (h_opt.tolist(), final_c5, n_points)
```
