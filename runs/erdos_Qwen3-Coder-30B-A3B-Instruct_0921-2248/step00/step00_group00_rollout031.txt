import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute C5 bound from h values"""
    # Compute full cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum of these correlations
    max_corr = np.max(corr)
    return max_corr * dx

def objective(h):
    """Objective function to minimize - negative of the C5 bound"""
    dx = 2.0 / len(h)
    return -evaluate_c5(h, dx)

def constraint_eq(h):
    """Equality constraint: sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    return np.sum(h) * dx - 1.0

    dx = 2.0 / n_points
    h0 = h_init.copy()
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = [
        {'type': 'eq', 'fun': constraint_eq}
    ]
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    best_n_points = n_points

    # Optimization parameters
    max_iter = 500
    tolerance = 1e-7
    
    # Optimize with SLSQP
    result = minimize(

        bounds=bounds,

        options={'maxiter': max_iter, 'ftol': tolerance, 'gtol': tolerance},

    )
    
    # Check if optimization succeeded
    if result.success:
        h_opt = result.x
        c5_opt = -objective(h_opt)
        if c5_opt < best_c5:
