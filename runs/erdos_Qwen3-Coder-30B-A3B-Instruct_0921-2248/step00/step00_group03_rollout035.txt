    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_h(h):
        h = np.array(h)
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        # Compute all overlaps
        overlaps = []
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps.append(overlap * dx)
        return max(overlaps), h
    
    # Objective function for optimization
    def obj(h):
        _, h_adj = evaluate_h(h)
        return np.max(np.correlate(h_adj, 1 - h_adj, mode="full")) * dx

    # Constraint: h[i] in [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum of h equals target_sum
    def constraint_sum(h):
        return np.sum(h) - target_sum

    # Use scipy.optimize.minimize with SLSQP
    cons = {'type': 'eq', 'fun': constraint_sum}
    res = optimize.minimize(
        obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}
    )
    
    if res.success or not np.isnan(res.fun):
        best_h = res.x
        c5_bound, _ = evaluate_h(best_h)

        best_h = best_h * target_sum / current_sum
        
    return (best_h.tolist(), c5_bound, n_points)
