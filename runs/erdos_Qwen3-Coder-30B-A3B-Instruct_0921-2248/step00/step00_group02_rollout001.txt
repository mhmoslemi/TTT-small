import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for a step function h"""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral for shift k corresponds to corr[n-1+k]
    # We want the maximum over all valid shifts
    n = len(h)
    max_overlap = 0
    for k in range(-n + 1, n):
        idx = n - 1 + k
        overlap = corr[idx] * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def is_feasible(h, n_points):
    """Check if h satisfies all constraints"""
    dx = 2.0 / n_points
    total = np.sum(h) * dx
    return abs(total - 1.0) < 1e-10 and np.all(h >= 0) and np.all(h <= 1)

def perturb_h(h, n_points, n_swaps=3):
    """Generate a perturbed version of h by swapping 0s and 1s"""
    h_new = h.copy()
    # Identify 0s and 1s
    zeros = np.where(h_new == 0)[0]
    ones = np.where(h_new == 1)[0]
    
    # For a small number of swaps
    for _ in range(n_swaps):
        if len(zeros) > 0 and len(ones) > 0:
            # Pick a random zero and one to swap
            i = random.choice(zeros)
            j = random.choice(ones)
            h_new[i] = 1
            h_new[j] = 0
            zeros = np.where(h_new == 0)[0]
            ones = np.where(h_new == 1)[0]
    
    return h_new
