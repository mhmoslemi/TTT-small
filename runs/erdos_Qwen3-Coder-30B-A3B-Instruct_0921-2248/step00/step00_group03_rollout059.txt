import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available, otherwise initialize randomly
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
    else:
        h_init = np.ones(n_points) * 0.5

    # Create variables
    h = cp.Variable(n_points, bounds=(0, 1))
    C = cp.Variable()

    # Constraints: For each shift k, sum h[i] * (1 - h[(i+k) % n_points]) <= C/dx
    constraints = []
    
    # Add sum constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Generate all shift constraints
    for k in range(-n_points + 1, n_points):
        # Compute valid indices for the current shift
        expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h[i] * (1 - h[j])
        constraints.append(expr <= C / dx)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Set solver options
    prob.solve(solver=cp.ECOS, max_iter=200, verbose=False, time_limit=budget_s)
    
    # Extract the best solution found
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        C_opt = C.value
    else:
        # Fallback to initial solution if optimization fails
        h_opt = h_init.copy()
        C_opt = np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx
        
    # Final validation
    h_opt = np.clip(h_opt, 0, 1)
    if abs(np.sum(h_opt) - target_sum) > 1e-6:
        h_opt = h_opt * (target_sum / np.sum(h_opt))
    
    # Ensure we don't exceed our time limit
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.95:
        C_opt = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx

    return (h_opt.tolist(), float(C_opt), n_points)
