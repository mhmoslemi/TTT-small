import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except NameError:
        # Fallback: create a uniform distribution
        n_points = 70
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy constraint ∫ h(x) dx = 1
    dx = 2.0 / n_points
    h_init = h_init * (1.0 / np.sum(h_init) * n_points * dx)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    solver_time = 0
    
    # Try different numbers of points if time allows
    if budget_s - solver_time > 100:
        for n_try in [70, 100]:
            if n_try > n_points:
                # Extend the array
                h_new = np.zeros(n_try)
                h_new[:n_points] = h_init
                h_new[n_points:] = h_init[:n_try - n_points]
                h_try = h_new
            else:
                h_try = h_init[:n_try]
            
            # Optimization using convex programming
            h_vars = cp.Variable(n_try)
            C = cp.Variable()
            
            # Objective: minimize C
            objective = cp.Minimize(C)
            
            # Constraints
            constraints = [0 <= h_vars, h_vars <= 1]
            
            # Sum constraint
            constraints.append(cp.sum(h_vars) == n_try / 2)
            
            # Overlap constraints
            for k in range(n_try):
                # Compute convolution manually (only for one side)
                # overlap_k = sum(h[i] * (1 - h[(i+k)%n_try]) for i in range(n_try))
                # Equivalent to: h * (1 - h_shifted)
                h_shifted = cp.hstack([h_vars[k:], h_vars[:k]])
                overlap_k = cp.sum(cp.multiply(h_vars, 1 - h_shifted))
                constraints.append(overlap_k <= C * dx)
            
            # Solve
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_opt = np.array(h_vars.value).flatten()
                    c5_opt = C.value * dx
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
            except Exception:
                pass  # Skip if optimization fails
            
            # Update timer
            solver_time = time.time() - start_time
            if solver_time > budget_s - 20:
                break
    
    # Final refinement using local search if needed
    solver_time = time.time() - start_time
    if solver_time < budget_s - 10:
        # Apply a simple gradient-based update with bounded shifts
        h_refined = best_h.copy()
        for _ in range(5000):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Small random perturbation
            delta = np.random.normal(0, 0.005, len(h_refined))
            h_new = h_refined + delta
            
            # Clip to bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to sum to n_points / 2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new C5
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                h_refined = h_new.copy()
        
    # Final check
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), c5_final, len(best_h))

def evaluate_erdos_solution(h, dx):
    """Return the maximum overlap of h with 1-h across all shifts."""
    # Compute correlation of h with 1-h (taking maximum over all shifts)
    h = np.array(h)
    k = 1 - h
    correlations = correlate(h, k, mode='full')
    max_overlap = np.max(correlations) * dx
    return max_overlap
