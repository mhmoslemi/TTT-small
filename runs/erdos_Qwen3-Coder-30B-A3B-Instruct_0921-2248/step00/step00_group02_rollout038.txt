import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h based on available initial_h_values or create from scratch
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        # Create a piecewise constant construction with some variation
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid] = 0.5
        # Normalize to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Precompute lags
    lags = np.arange(-n_points + 1, n_points)
    
    step = 0
    while time.time() - start_time < budget_s - 1.0:
        step += 1
        # Evaluate current overlaps
        overlaps = evaluate_overlaps(h, dx)
        max_overlap = np.max(overlaps)
        max_idx = np.argmax(overlaps)
        
        if max_overlap <= best_c5:
            break
            
        # Adjust around the maximum overlap lag
        k = lags[max_idx]
        
        # Find good indices to adjust
        indices = find_adjustment_indices(h, k, dx)
        if not indices:
            # If no indices found, try small random adjustments
            i = np.random.randint(0, n_points)
            j = (i + k) % n_points if k >= 0 else (i + k + n_points) % n_points
            if 0 <= j < n_points:
                indices.append((i, j))
        
        if not indices:
            # If still no proper indices, jump out of current loop
            continue

        adjustments = generate_adjustments(h, indices, k, dx)
        if not adjustments:
            continue
            
        # Find best adjustment
        best_adj = None
        min_c5 = max_overlap
        for adj_h in adjustments:
            new_c5 = evaluate(adj_h, dx)
            if new_c5 < min_c5:
                min_c5 = new_c5
                best_adj = adj_h
        
        if best_adj is not None:
            h = best_adj
            if min_c5 < best_c5:
                best_c5 = min_c5
                best_h = h.copy()
        else:
            # No improvement found, break to prevent infinite loop
            break
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5"""
    # Using full convolution
    conv = signal.correlate(h, 1 - h, mode='full')
    # The indexing assumes input is already in correct order
    # But we should slice the relevant part
    # It starts at -n+1 to n-1, with length 2n-1
    # For overlap, we want relative positions from -n+1 to n-1 (total 2n-1)
    # But the actual maximum integral value is the max over the overlap integrals
    # For each k, we compute sum_{i, k < 0: i+k >= 0 otherwise i < n-k} h[i] * (1 - h[i+k])
    # Then use the correlogram
    # Shape is (2n-1,)
    # The full convolution is valid for h with elements in [0,1]
    # max_k integral = max_k sum h[i] (1 - h[i+k])
    # This is equivalent to maximum value in the correlation result
    # However, we need to interpret the indexing correctly:
    # corr_result corresponds to lag_k = -n+1, ..., n-1
    # So we take maximum value in corr_result
    
    if len(conv) != 2 * len(h) - 1:
        raise ValueError("Convolution size mismatch")
    
    max_overlap = np.max(conv) * dx
    return max_overlap

def evaluate_overlaps(h, dx):
    """Compute all overlap integrals for different lags"""
    # This is more efficient than computing max(corr) for all lags
    n = len(h)
    # Full convolution h * (1-h) 
    # Result length: 2n-1
    corr = signal.correlate(h, 1 - h, mode='full')
    # Each element corr[x] corresponds to overlap at lag x - n+1
    overlaps = corr * dx
    return overlaps

def find_adjustment_indices(h, k, dx):
    """Find good pairs of indices to try to adjust"""
    n = len(h)
    # Ensure that the k shift doesn't go out of bounds
    # We look for h[i] = 1 and h[i+k] = 0 or vice versa
    # Or where there's an opportunity to reduce overlap
    indices = []
    
    for i in range(n):
        if not (0 <= i+k < n):
            continue
        # Analyze change (when we do h[i] -= delta, h[i+k] += delta)
        # Overlap contribution (if k > 0): 
        # Lags that involve i, i+k can change
        if abs(k) >= 0:
            indices.append((i, i+k))
    return indices

def generate_adjustments(h, indices, k, dx):
    """Generate adjustment vectors to try"""
    adjustments = []
    max_delta = 0.1
    delta = max_delta
    
    for i, j in indices:
        # Try lowering h[i] and raising h[j]
        h_new = h.copy()
        h_new[i] = max(0.0, h[i] - delta)
        h_new[j] = min(1.0, h[j] + delta)
        
        # Check if constraint still hold
        if abs(np.sum(h_new) - np.sum(h)) < 1e-8:
            adjustments.append(h_new)
            
        # Try the opposite
        h_new = h.copy()
        h_new[i] = min(1.0, h[i] + delta)
        h_new[j] = max(0.0, h[j] - delta)
        if abs(np.sum(h_new) - np.sum(h)) < 1e-8:
            adjustments.append(h_new)
            
    return adjustments
