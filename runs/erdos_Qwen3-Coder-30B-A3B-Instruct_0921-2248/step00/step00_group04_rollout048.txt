import numpy as np
from scipy import optimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 42
        # Initial construction using a simple step function
        h0 = np.ones(n_points)
        h0[:n_points//2] = 0
        h0 = h0 * 2.0 / n_points  # Normalize sum to 1
    
    dx = 2.0 / n_points
    h0 = h0 / np.sum(h0) * (1.0/dx)  # enforce integral = 1
    
    # Start timing
    start_time = time.time()
    
    # Sample k values at coarse resolution (max k = 2, step = 0.1)
    k_vals = np.arange(0, 2 + 1e-8, 0.1)
    C5 = 0.5  # Initial guess or upper bound
    h_best = h0.copy()
    C5_best = float('inf')
    
    # Prepare solver and variables
    h_vars = Variable(n_points)
    C5_var = Variable()
    
    # Constraints for initial set of k values
    constraints = [
        sum(h_vars) == 1.0/dx,
        h_vars >= 0,
        h_vars <= 1
    ]
    
    # Lag iteration settings
    lag_step = 0.1
    max_lags = 20
    max_iter = 50
    iteration = 0
    
    # Iterate to improve bounds
    while time.time() - start_time < budget_s * 0.95:
        if iteration >= max_iter:
            break
            
        # Compute overlaps for current h and k values
        overlaps = []
        lag_indices = np.arange(0, len(k_vals))
        
        for k_index in lag_indices:
            k = k_vals[k_index]
            if k == 0:
                # Special case: no shift needed
                overlap = np.sum(h0 * (1 - h0)) * dx
            else:
                shift_idx = int(k / dx)
                # Account for periodicity and integer shift
                h_shifted = np.roll(h0, shift_idx)
                overlap = np.sum(h0 * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        
        max_overlaps = sorted(overlaps, reverse=True)[:max_lags]
        top_k_indices = np.argsort(overlaps)[::-1][:max_lags]
        
        # Update set of k values to include worst-case lags
        for i, k_i in enumerate(top_k_indices):
            k = k_vals[k_i]
            if k == 0:
                continue
            shift_idx = int(k / dx)
            h_shifted = np.roll(h_vars, shift_idx)
            constraints.append(sum(h_vars * (1 - h_shifted)) <= C5_var)
        
        # Re-solve optimization
        objective = Minimize(C5_var)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
        except:
            try:
                prob.solve(solver=ECOS, verbose=False)
            except:
                break
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            C5_new = C5_var.value
            
            # Evaluate the resulting h
            h_new = h_vars.value.copy()
            if np.any(h_new < 0):
                h_new = np.maximum(h_new, 0)
            if np.any(h_new > 1):
                h_new = np.minimum(h_new, 1)
            
            # Check if this is better
            if C5_new < C5_best:
                C5_best = C5_new
                h_best = h_new.copy()
            
            C5 = C5_new
        else:
            break
            
        iteration += 1

    # Final verification of the result
    h_final = h_best
    C5_final = C5_best

    # Clip values to ensure feasibility
    h_final = np.clip(h_final, 0, 1)
    
    # Correct normalization in case we have slight deviations
    h_final = h_final / np.sum(h_final) * (1.0/dx)

    return (h_final.tolist(), float(C5_final), n_points)
