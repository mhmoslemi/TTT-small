import numpy as np
from scipy import correlate
import cvxpy as cp
import time


def evaluate_C5(h, dx):
    conv = correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback to a symmetric tent function
        n_points = 100
        j = np.arange(-n_points // 2, n_points // 2)
        h = np.maximum(0, 1 - 2 * np.abs(j) / n_points)
        h = h / np.sum(h) * n_points / 2  # Correct the integral to 1

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # QP solver settings
    solver = cp.ECOS
    
    # Optimization loop
    max_iter = 1000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlaps
        conv = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv)
        max_indices = np.where(conv == max_overlap)[0]
        
        # Address all worst shifts
        shifts = max_indices - (n_points - 1)  # Convert to shift values
        shifts = shifts[shifts >= -n_points + 1]  # Only valid shifts
        
        # Define QP problem for worst overlaps
        h_var = cp.Variable(n_points)
        constraints = [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(h_var) == n_points / 2]
        
        # Objective: Minimize sum of overlaps for worst shifts
        objectives = []
        for k in shifts:
            if k >= 0:
                obj = cp.sum([
                    h_var[i] * (1 - h_var[i + k])
                    for i in range(n_points - k)
                ])
            else:
                obj = cp.sum([
                    h_var[i - k] * (1 - h_var[i])
                    for i in range(-k, n_points)
                ])
            objectives.append(obj)
        
        objective = cp.Minimize(cp.sum(objectives))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, eps=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Ensure it is valid: correct infeasibility
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    # Normalize
                    new_h = new_h / np.sum(new_h) * n_points / 2
                    h = new_h
                    C5 = evaluate_C5(h, dx)
                    if C5 < best_C5:
                        best_C5 = C5
                        best_h = h.copy()
        except:
            pass
    
    return (best_h.tolist(), best_C5, n_points)
