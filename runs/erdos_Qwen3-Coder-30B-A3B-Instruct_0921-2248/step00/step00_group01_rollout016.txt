import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_vals, n_points):
    dx = 2.0 / n_points
    # Compute all correlations
    corr = np.correlate(h_vals, 1 - h_vals, mode="full")
    # Max correlation (scaled)
    max_corr = np.max(corr) * dx
    return max_corr
