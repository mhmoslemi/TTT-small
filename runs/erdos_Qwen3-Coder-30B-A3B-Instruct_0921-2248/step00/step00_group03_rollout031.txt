import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple starting point with 1s and 0s
        n_points = 70
        h = np.zeros(n_points)
        # Place 1s in alternating fashion
        h[::2] = 1.0
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    def evaluate(h_vals):
        # Compute correlation values using numpy
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_func(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    def project_to_feasible(h_vals):
        # Clamp values to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Renormalize to sum to n_points/2
        sum_h = np.sum(h_vals)
        if sum_h > 0:
            h_vals = h_vals * (n_points / 2) / sum_h
        return h_vals
    
    # Optimize using projected gradient descent
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-6
    
    start_time = time.time()
    current_h = h.copy()
    current_c5 = evaluate(current_h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute maximum overlap index
        corr = np.correlate(current_h, 1 - current_h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        # Derivative of overlap with respect to h[i]
        # overlap_k = sum h[i] * (1 - h[(i+k)%n])
        # gradient w.r.t h[i] = 1 - h[(i+k)%n] for validity
        gradient = np.zeros_like(current_h)
        for i in range(n_points):
            j = (i + max_k) % n_points
            gradient[i] = 1 - current_h[j]
        
        # Update h
        updated_h = current_h - step_size * gradient
        
        # Project back to feasible region
        updated_h = project_to_feasible(updated_h)
        updated_c5 = evaluate(updated_h)
        
        # Accept update if it improves the C5
        if updated_c5 < current_c5:
            current_h = updated_h
            current_c5 = updated_c5
        else:
            # Reduce step size if no improvement
            step_size *= 0.99
            
        if iteration % 100 == 0:
            if current_c5 < 0.38085:
                break
    
    # Final check
    final_h = project_to_feasible(current_h)
    final_c5 = evaluate(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
