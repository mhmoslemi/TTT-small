import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Compute C5 bound from the step function h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with the parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create initial guess as a simple step function
        n_points = 40
        h = np.ones(n_points) * 0.5
        # Make it have exactly the right integral
        h = h * 2.0 / n_points
    
    # Set up the optimization problem
    dx = 2.0 / n_points
    
    def objective(h_vec):
        # Convert to proper size if needed
        h_reshaped = np.array(h_vec)
        return evaluate_erdos_solution(h_reshaped)
    
    def constraint_eq(h_vec):
        # Integral constraint: sum(h) * dx = 1
        return np.sum(h_vec) * dx - 1.0
    
    def constraint_ineq(h_vec):
        # Bounds: 0 <= h_i <= 1
        return np.hstack([h_vec, 1 - h_vec])
    
    # Constraints for scipy
    constraints = [
        {'type': 'eq', 'fun': constraint_eq},
        {'type': 'ineq', 'fun': lambda x: constraint_ineq(x)[len(x):]},  # upper bounds
        {'type': 'ineq', 'fun': lambda x: constraint_ineq(x)[:len(x)]}   # lower bounds
    ]
    
    # Bounds (0, 1) for all variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    start_time = time.time()
    max_iter = 500
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Define optimizer function that respects timeout
    def optimization_step():
        # Try SLSQP
        try:
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-7, 'gtol': 1e-7}
            )
            if result.success:
                return result.x
        except Exception:
            pass
        return h
        
    # Continue optimization while time allows
    while time.time() - start_time < budget_s * 0.95:
        new_h = optimization_step()
        new_c5 = evaluate_erdos_solution(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        h = new_h
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
