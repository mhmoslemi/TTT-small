def evaluate_erdos_solution(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx """
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Max over all lags (which are from -n+1 to n-1)
    max_corr = np.max(corr)
    return max_corr * dx
