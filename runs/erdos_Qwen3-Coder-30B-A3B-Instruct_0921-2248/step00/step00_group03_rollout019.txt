from scipy.optimize import minimize

    def eval_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Function to evaluate constraints
    def eval_constraints(h):
        h = np.array(h)
        # Check bounds
        bounds_violation = np.sum(np.maximum(0, h) + np.maximum(0, 1 - h))
        # Check sum constraint
        sum_violation = abs(np.sum(h) * dx - 1.0)
        return bounds_violation + sum_violation
    
    # Optimization function
    def objective(h):
        return eval_C5(h)
    
    # Constraints for bounds and sum
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}
    ]
    
    # Optimize with different starting points
    best_h = h0.copy()
    best_C5 = eval_C5(best_h)
    
    # Use scipy minimize
    max_iter = 1000
    try:
        for i in range(min(10, max_iter)):

                break
