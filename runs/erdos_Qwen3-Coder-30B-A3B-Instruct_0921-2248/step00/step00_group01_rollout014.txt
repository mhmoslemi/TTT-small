import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function is assumed to be provided
    # Computes the maximum overlap integral for step function h
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up discretization
    n_points = 200  # Reasonable number for balance of accuracy and speed
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create binary-like starting point and shuffle
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)
    
    # Use SLSQP to optimize
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Re-project to feasible set every few iterations
        if iter_count % 10 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Optimize using SLSQP
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, 
                       options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8}, 
                       callback=lambda x: None)
        
        if res.success:
            h = res.x
        else:
            # If optimization fails, do a gradient projection or fallback
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        iter_count += 1
        
        # Early stopping if no improvement for a while
        if iter_count % 50 == 0 and iter_count > 50:
            if abs(best_c5 - evaluate_erdos_solution(best_h)) < 1e-6:
                break
                
    return (best_h.tolist(), best_c5, n_points)
