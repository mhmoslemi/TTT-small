import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = int(1e5)
    initial_T = 1.0
    cooling_rate = 0.9995
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Evaluate initial solution
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        full_corr = corr[len(corr)//2:]
        return np.max(full_corr) * dx
    
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    T = initial_T
    
    # Simulated Annealing
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create neighbor solution
        h_new = h.copy()
        # Perturb few elements
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Renormalize to sum = target_sum
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute new C5
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        delta_C5 = new_C5 - current_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_new
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_h = h_new.copy()
                best_C5 = new_C5
                
        # Cool down temperature
        T *= cooling_rate
    
    # Final check
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
