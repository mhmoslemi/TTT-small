import numpy as np
from scipy import signal
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given h"""
    dx = 2.0 / len(h)
    # Compute convolution of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take max over all possible shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_feasible(h):
    """Project h to feasible region: [0,1] with sum = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h / h_sum * (2.0 / len(h))  # scale to correct integral
    return h

def perturb_h(h, num_swaps=3, swap_radius=2):
    """Create a perturbed version of h"""
    h_new = h.copy()
    n = len(h)
    # Random swaps
    for _ in range(num_swaps):
        i = np.random.randint(0, n)
        j = np.random.randint(max(0, i - swap_radius), min(n, i + swap_radius + 1))
        if i != j:
            h_new[i], h_new[j] = h_new[j], h_new[i]
    
    # Adjust one point randomly
    i = np.random.randint(0, n)
    h_new[i] = np.random.random()
    
    # Project back to feasibility
    h_new = project_to_feasible(h_new)
    return h_new
