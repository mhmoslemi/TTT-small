import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_shift = 2 * n_points  # 0 to 2*points - 1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    start_time = time.time()
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Prevent early exit for suboptimal solutions
    i = 0
    while time.time() - start_time < budget_s - 5:
        i += 1
        try:
            # Compute C5 using correlation
            corr = correlate(h, 1 - h, mode='full')
            c5_val = np.max(corr) * dx
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
            
            # Use simple gradient descent approach for modification
            h_new = h.copy()
            # Perturb the solution a bit
            noise = np.random.normal(0, 0.01, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            # Normalize
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            h = h_new

        except Exception:
            break

    # Final check with actual overlap computation
    try:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    except:
        # Fallback to cautious evaluation
        final_c5 = np.inf

    # Ensure final result
    final_c5 = min(final_c5, best_c5)
    return (best_h.tolist(), final_c5, n_points)
