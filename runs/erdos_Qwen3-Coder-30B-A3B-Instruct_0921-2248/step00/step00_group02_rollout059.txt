def evaluate_overlaps(h, dx):
    n_points = len(h)
    max_overlap = 0.0
    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                overlap += h[i] * (1.0 - h[j])
            else:
                break
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def evaluate_erdos_solution(h_values):
    """Use the pre-imported evaluator function"""
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Use full convolution to compute overlaps for all shifts
    correlation = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(correlation)
    return max_correlation * dx
