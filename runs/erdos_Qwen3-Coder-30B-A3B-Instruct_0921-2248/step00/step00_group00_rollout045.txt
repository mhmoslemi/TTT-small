from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h):
    """Compute C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # max over all lags
    max_corr = np.max(corr)
    return max_corr * dx
