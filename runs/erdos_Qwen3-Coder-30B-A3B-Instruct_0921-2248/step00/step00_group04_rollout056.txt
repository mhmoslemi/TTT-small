import numpy as np
import time
from scipy import signal

def evaluate(h, dx):
    # Compute C5 using correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum of the correlation
    max_corr = np.max(corr)
    return max_corr * dx

def normalize(h, target_sum):
    # Ensure the sum of h is exactly target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Normalize to target sum
    h = h * target_sum / current_sum
    return h

def perturb(h, epsilon, target_sum):
    # Perturb some elements while maintaining constraints
    h_new = h.copy()
    n = len(h)
    # Select a few random indices to modify
    indices = np.random.choice(n, size=min(5, n), replace=False)
    for i in indices:
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[i] += delta
    # Enforce bounds
    h_new = np.clip(h_new, 0, 1)
    # Renormalize
    h_new = normalize(h_new, target_sum)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set problem parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize(h, target_sum)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = normalize(h, target_sum)
    
    # Start timing
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Simulation Annealing parameters
    T = 1.0
    cooling_rate = 0.9995
    epsilon = 0.1
    
    # Number of iterations to run (adjustable for timing)
    max_iter = int(1e6)
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_new = perturb(h, epsilon, target_sum)
        c5_new = evaluate(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature and epsilon
        T *= cooling_rate
        epsilon *= cooling_rate
        
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
