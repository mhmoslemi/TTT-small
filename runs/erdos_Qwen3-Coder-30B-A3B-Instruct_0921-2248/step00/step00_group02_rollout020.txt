import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    n_points = 45
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a binary-like initial guess with sum = n_points / 2
        h = np.zeros(n_points)
        half_idx = n_points // 2
        h[:half_idx] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:  # leave 1s for cleanup
        iter_count += 1
        
        # Generate candidate h by perturbing best_h
        candidate_h = best_h.copy()
        
        # Try some random changes that keep sum = n_points / 2 and bounds [0,1]
        # 1. Flip a 1 to 0 and a 0 to 1
        if np.random.rand() < 0.5:
            ones = np.where(candidate_h > 0.9)[0]
            zeros = np.where(candidate_h < 0.1)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                candidate_h[i] = 0.0
                candidate_h[j] = 1.0
        else:
            # 2. Adjust a single point with small change
            i = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            candidate_h[i] += delta
            # Ensure bounds
            candidate_h[i] = max(0.0, min(1.0, candidate_h[i]))
        
        # Ensure sum constraint
        current_sum = np.sum(candidate_h)
        correction = (n_points / 2 - current_sum) / n_points
        candidate_h += correction
        
        # Clamp to bounds
        candidate_h = np.clip(candidate_h, 0.0, 1.0)
        
        # Evaluate new candidate
        C5 = evaluate_C5(candidate_h, dx)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = candidate_h.copy()
        
        # Early exit if good enough
        if best_C5 < 0.3807:
            break
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx):
    # Use the exact evaluation method from the task
    conv = np.correlate(h, 1 - h, mode='full')
    max_conv = np.max(conv)
    return max_conv * dx
