import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    step_size = 0.01
    
    # Initialize h: use initial_h_values if available, else create symmetric tent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (n_points / 2)
        h = h * (n_points / 2) / np.sum(h)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get all correlations
        corr = correlate(h, 1 - h, mode="full")
        # Scale by dx
        corr = corr * dx
        
        # Find maximum value
        max_val = np.max(corr)
        max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        # Gradient step for each maximum index
        grad = np.zeros(n_points)
        for idx in max_indices:
            shift = idx - (n_points - 1)
            for i in range(n_points):
                if 0 <= i + shift < n_points:
                    grad[i] += (1 - h[i + shift])
                if 0 <= i - shift < n_points:
                    grad[i] -= h[i - shift]
        
        # Gradient descent step
        h_new = h - step_size * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    corr = corr * dx
    return np.max(corr)
