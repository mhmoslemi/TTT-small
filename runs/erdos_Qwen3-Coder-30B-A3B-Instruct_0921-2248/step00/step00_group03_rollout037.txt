import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.001
    alpha_decay = 0.99
    tolerance = 1e-6
    
    # Get initial point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h[j] - h[i]
        
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        
        # Project to correct integral
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        if iteration % 50 == 0:
            alpha *= alpha_decay
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    n = len(h)
    # Compute cross-correlation
    corr = correlate(h, 1 - h, mode='full')
    # Maximum overlap value
    return np.max(corr) * dx
