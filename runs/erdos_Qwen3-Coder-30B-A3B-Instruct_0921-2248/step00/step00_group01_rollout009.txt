import numpy as np
import random
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h"""
    # Compute cross-correlation of h and (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value of the correlation
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 200
        # Initialize binary h with approximately half 1s
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Ensure sum equals exactly n_points/2
    target_sum = n_points // 2
    current_sum = np.sum(h)
    if current_sum != target_sum:
        # Flip some bits to get exact sum
        indices_0 = np.where(h == 0)[0]
        indices_1 = np.where(h == 1)[0]
        to_flip = abs(current_sum - target_sum)
        if to_flip > 0:
            flip_1_to_0 = np.random.choice(indices_1, to_flip, replace=False)
            flip_0_to_1 = np.random.choice(indices_0, to_flip, replace=False)
            h[flip_1_to_0] = 0
            h[flip_0_to_1] = 1
    
    dx = 2.0 / n_points
    current_C5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_C5 = current_C5
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.999
    T = T0
    start_time = time.time()
    max_iter = int(1e5)
    
    # Perform simulated annealing
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor by swapping a 0 and 1
        indices_0 = np.where(h == 0)[0]
        indices_1 = np.where(h == 1)[0]
        if len(indices_0) == 0 or len(indices_1) == 0:
            continue
        i0 = np.random.choice(indices_0)
        i1 = np.random.choice(indices_1)
        # Create neighbor
        h_new = h.copy()
        h_new[i0] = 1
        h_new[i1] = 0
        
        new_C5 = evaluate_h(h_new, dx)
        
        # Accept with probability based on SA criterion
        if new_C5 < current_C5 or random.random() < np.exp(-(new_C5 - current_C5) / T):
            h = h_new
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_h = h.copy()
                best_C5 = current_C5
                
        # Cool down temperature
        T *= alpha
    
    return (best_h.tolist(), best_C5, n_points)
