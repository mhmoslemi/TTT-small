def evaluate_erdos_solution(h):
    """Evaluates the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts 0 to 2n-2
    h_extended = np.concatenate([h, h])
    overlaps = []
    for k in range(2 * n - 1):
        overlap = np.sum(h * (1 - h_extended[k:k+n])) * dx
        overlaps.append(overlap)
    return max(overlaps)
