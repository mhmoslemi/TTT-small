import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    lag_step = 0.2
    max_lags = 200
    top_lags_ratio = 0.1
    
    # Initial lag set
    lags = np.arange(-2.0, 2.0 + lag_step/2, lag_step)
    lags = lags[abs(lags) <= 2.0]
    
    best_C5 = np.inf
    best_h = None
    
    # Start from initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
        C5 = evaluate_erdos_solution(h, n_points)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Solving LP with current lags
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) * dx == 1, C5_var >= 0]
        
        for k in lags:
            lag_idx = int(k / lag_step) + len(lags) // 2
            kernel = np.roll(np.ones(n_points), -lag_idx)
            if k < 0:
                kernel = np.roll(kernel, -int(-k / lag_step))
            else:
                kernel = np.roll(kernel, int(k / lag_step))
            
            # Compute convolution explicitly
            conv = np.convolve(h_var, 1 - kernel, mode='full')[:n_points]
            conv = conv * dx
            constraints.append(conv <= C5_var)
        
        # Solve
        objective = cp.Minimize(C5_var)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            break
            
        if prob.status != cp.OPTIMAL:
            break
            
        h = h_var.value
        current_C5 = C5_var.value
        
        # Update best if improved
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Identify bad lags
        cross_corr = correlate(h, 1-h, mode='full')
        # Normalized cross-correlation
        lags_full = np.arange(-n_points+1, n_points)
        max_overlaps = np.abs(cross_corr) * dx
        
        # Percentile to get worst lags
        threshold = np.percentile(max_overlaps, 100*(1-top_lags_ratio))
        problematic_lags = lags_full[np.abs(max_overlaps) >= threshold]
        
        # Update lag set
        lags = np.unique(np.concatenate([lags, problematic_lags * dx]))
        # Keep limited number of lags
        lags = lags[np.abs(lags) <= 2.0]
        if len(lags) > max_lags:
            lags = lags[::2]  # reduce to a subset
            
    if best_h is None:
        if 'initial_h_values' in globals():
            return (initial_h_values, evaluate_erdos_solution(initial_h_values, len(initial_h_values)), len(initial_h_values))
        else:
            # Fallback
            h = np.ones(n_points) * 0.5
            h[0] = 1.0
            h[1] = 0.0
            h[2] = 1.0
            h[3] = 0.0
            h[4] = 0.0
            h[5] = 1.0
            h[6] = 0.0
            return (h, evaluate_erdos_solution(h, n_points), n_points)
    
    # Final check and return
    final_C5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h, final_C5, n_points)
