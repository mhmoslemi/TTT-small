import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * target_sum
    else:
        # Tent function
        h = 0.5 * (n_points - np.abs(np.arange(n_points) - n_points/2)) / (n_points/2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
    
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    # Start timing
    start_time = time.time()
    
    def compute_C5(h_vals):
        h = np.array(h_vals)
        # Compute all correlations and find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        # Trim to valid range
        corr = corr[len(h)-1:]
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    def obj_func(h_vals):
        return compute_C5(h_vals)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization parameters
    max_iter = 5000
    tol = 1e-6
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    # Adaptive trust region
    trust_radius = 0.1
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use scipy.optimize for the constrained optimization
        try:
            result = minimize(
                obj_func,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'ftol': tol, 'gtol': tol},
                callback=lambda x: None
            )
            
            if result.success:
                h = result.x
                new_C5 = compute_C5(h)
                
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h.copy()
        except:
            pass
            
        # Small random perturbation to escape local minima
        if it % 10 == 0:
            perturb = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + perturb, 0, 1)
            # Renormalize
            h = h * target_sum / np.sum(h)
            
        # Early stopping if no improvement in a while
        if it > 200 and it % 20 == 0 and compute_C5(h) > best_C5 * 0.9999:
            break
    
    return best_h.tolist(), best_C5, n_points
