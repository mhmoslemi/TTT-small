def evaluate_erdos_solution(h):
    """Evaluate C5 function using correlation"""
    # assuming h is already normalized 
    # and will be internally scaled properly in correlation calc
    h = np.asarray(h)
    n = len(h)
    dx = 2.0 / n
    # compute full convolution h(x)(1-h(x+k))
    corr = np.correlate(h, 1 - h, mode="full")
    # Due to FFT operations, we have to split
    # Use 'full' mode and then look at the maximum over correct indices
    # We are only interested in some shifted versions of 1-h (positive shifts)
    # The resulting array has length 2*n - 1
    # For reference: for h in [0,2] and corresponding shift k=0,2
    # index k corresponds to corrs[n-k-1] to corrs[n+k-1]  (when including shift 0)
    res = np.array([0.0] * (n + n - 1))
    for i in range(len(corr)):
        j = i - (n - 1)
        if 0 <= j < n:
            res[j] += corr[i]
    # Find maximum over valid indices (shifting with overlap)
    # The overlapping region from [0,2] can be approximated 
    # by sliding h over all positions for shift(dx) calculation
    c5 = np.max(corr) * dx
    return (c5, corr)
