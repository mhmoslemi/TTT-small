import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Sample lags from [-2, 2] with step 0.1
    lags = np.arange(-2.0, 2.01, 0.1)
    n_lags = len(lags)
    
    # Define the optimization variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) * dx == 1,
        C5 >= 0
    ]
    
    # Add overlap constraints
    for k in lags:
        # Compute shifted indices for lag k
        shift = int(k / dx)
        if shift == 0:
            # no shift
            corr = cp.sum(cp.multiply(h, 1 - h))
        elif shift > 0:
            # forward shift
            h_shifted = cp.hstack([h[shift:], h[:shift]])
            corr = cp.sum(cp.multiply(h, 1 - h_shifted))
        else:
            # backward shift
            shift = abs(shift)
            h_shifted = cp.hstack([h[-shift:], h[:-shift]])
            corr = cp.sum(cp.multiply(h, 1 - h_shifted))
        constraints.append(corr * dx <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False)

        subset_size = min(20, len(lags))
