import numpy as np
from scipy import signal
import time
import random

def evaluate(h, dx):
    # Compute the overlap integral
    corr = signal.correlate(h, 1 - h, mode='full')
    # Only consider the valid overlaps (k from -1 to +1)
    # But actually compute all, then take max
    max_overlap = np.max(corr) * dx
    return max_overlap

def is_valid(h, n_points):
    # Check bounds and sum constraint
    if np.any(h < 0) or np.any(h > 1):
        return False
    if abs(np.sum(h) - n_points / 2.0) > 1e-10:
        return False
    return True

def perturb(h, n_points):
    # Randomly perturb h by swapping a 0 and 1, or adjusting a point
    h_new = h.copy()
    # Try to swap 1 and 0 positions
    ones = np.where(h_new == 1)[0]
    zeros = np.where(h_new == 0)[0]
    
    if len(ones) > 0 and len(zeros) > 0:
        # Swap one 1 with one 0
        i1 = random.choice(ones)
        i0 = random.choice(zeros)
        h_new[i1], h_new[i0] = h_new[i0], h_new[i1]
    
    # Adjust one point slightly (range [0,1])
    i = random.randint(0, n_points - 1)
    h_new[i] = max(0, min(1, h_new[i] + random.uniform(-0.1, 0.1)))
    
    # Ensure the sum constraint
    current_sum = np.sum(h_new)
    diff = n_points / 2.0 - current_sum
    
    # Distribute diff to ensure sum = n_points/2
    if abs(diff) > 1e-10:
        # Add diff to one element
        idx = random.randint(0, n_points - 1)
        h_new[idx] = max(0, min(1, h_new[idx] + diff))
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    n_points = 45  # as per initial_h_values size

    else:

        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # Adjust sum to be exactly n_points / 2
        h[0] = 1.0  # Ensure exact sum
        h[1] = 0.0
        # Rescale properly
        h = h * (n_points / 2.0) / np.sum(h)
        # just make it a full binary as starting config
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    best_h = h.copy()
    best_C5 = evaluate(best_h, dx)
    
    iters = 0
    while time.time() - start_time < budget_s - 1:  # leave some margin
        h_new = perturb(h, n_points)
        if is_valid(h_new, n_points):
            C5_new = evaluate(h_new, dx)
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
            # Update current h
            h = h_new
        iters += 1
    
    return (list(best_h), best_C5, n_points)
