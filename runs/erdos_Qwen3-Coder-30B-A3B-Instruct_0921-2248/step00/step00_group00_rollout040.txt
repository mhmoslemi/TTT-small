import numpy as np
from scipy import signal
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute cross-correlation of h with (1-h) with full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value, scaled by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Setup parameters
    n_points = 45  # Use the provided initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Required integral constraint
    
    # Initialize h from initial_h_values if available, otherwise random
    if 'initial_h_values' in globals():

    else:
        # Create a random feasible h with integral=1
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    
    # Ensure h is in [0,1] and sum is correct
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Evaluate initial C_5
    current_C5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_C5 = current_C5
    
    # Simulated Annealing parameters
    T0 = 0.1  # Initial temperature
    alpha = 0.999  # Cooling rate
    T = T0
    max_iter = 1000000
    start_time = time.time()
    
    def is_valid_h(h_test):
        """Check if h_test satisfies all constraints."""
        return (np.all(h_test >= 0) and np.all(h_test <= 1) and
                abs(np.sum(h_test) - target_sum) < 1e-8)
    
    # Flip interval: 1 or 3 points
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Make a move: flip interval
        # Randomly select interval length (1 or 3)
        length = random.choice([1, 3])
        # Select random start index
        start = random.randint(0, n_points - length)
        
        # Create candidate h' by flipping values in interval
        h_candidate = h.copy()
        for i in range(start, start + length):
            h_candidate[i] = 1 - h_candidate[i]
        
        # Ensure feasibility
        # Adjust the values to maintain constraint sum(target_sum)
        current_sum = np.sum(h_candidate)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            to_adjust = min(abs(diff), 1.0)
            # Adjust randomly selected points (keep in [0,1])
            if diff > 0:  # Need to increase
                indices = np.random.choice(n_points, int(to_adjust / dx), replace=False)
                for i in indices:
                    if h_candidate[i] < 1:
                        h_candidate[i] = min(1, h_candidate[i] + dx)
            else:  # Need to decrease
                indices = np.random.choice(n_points, int(-to_adjust / dx), replace=False)
                for i in indices:
                    if h_candidate[i] > 0:
                        h_candidate[i] = max(0, h_candidate[i] - dx)
        
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        # Re-adjust sum to target
        current_sum = np.sum(h_candidate)
        if abs(current_sum - target_sum) > 1e-8:
            # Add small adjustment to make sum exactly target
            factor = target_sum / current_sum
            h_candidate = h_candidate * factor
            # Ensure final bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            # Final correction to ensure sum is correct
            diff = target_sum - np.sum(h_candidate)
            if abs(diff) > 1e-8:
                # Pick a few random indices and adjust
                if diff > 0:
                    idxs = np.random.choice(n_points, min(int(diff/dx), n_points), replace=False)
                    for i in idxs:
                        h_candidate[i] = min(1, h_candidate[i] + dx)
                else:
                    idxs = np.random.choice(n_points, min(int(-diff/dx), n_points), replace=False)
                    for i in idxs:
                        h_candidate[i] = max(0, h_candidate[i] - dx)
        
        # Re-evaluate
        if is_valid_h(h_candidate):
            candidate_C5 = evaluate_erdos_solution(h_candidate)
            
            # Accept or reject based on SA
            delta = candidate_C5 - current_C5
            if delta < 0 or random.random() < np.exp(-delta / T):
                h = h_candidate
                current_C5 = candidate_C5
                if current_C5 < best_C5:
                    best_h = h.copy()
                    best_C5 = current_C5
        
        # Cool down
        T = T * alpha
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    actual_sum = np.sum(best_h)
    if abs(actual_sum - target_sum) > 1e-8:
        # Adjust for final sum
        best_h = best_h * target_sum / actual_sum
        # Clip again
        best_h = np.clip(best_h, 0, 1)
    
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
