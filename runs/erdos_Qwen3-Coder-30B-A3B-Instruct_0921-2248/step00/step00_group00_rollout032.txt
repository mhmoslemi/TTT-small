import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h):
    """Compute C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The full convolution goes from -n+1 to n-1, so need to handle appropriately
    # But we're really interested in the maximum value which is at k = 0 to 2*n-2
    # Let's extract the "valid" part for comparison.
    # Actually, for C5 = max_k int h(x)(1-h(x+k)) dx, we take a full correlation
    # and this should account for k from -n+1 to n-1.
    # To get max of these integrals, we need the max of dx * corr values
    # since dx scales the integral. 
    return np.max(corr) * dx

    np.random.seed(seed)
    start_time = time()

        h0[min(n_points//2, 10):] = 0.0
        h0 = h0 / np.sum(h0) * 2.0  # scale to sum = 1
    
    # Check that h0 satisfies constraint
    if abs(np.sum(h0) * 2.0 / n_points - 1.0) > 1e-6:
        # Scale to meet constraint
        h0 = h0 / np.sum(h0) * (n_points / 2.0)
    
    def objective(h):
        if time() - start_time > budget_s * 0.95:
            return float('inf')  # signal don't continue
        return evaluate_erdos_solution(h)

    def constraint_eq(h):
        return np.sum(h) * 2.0 / n_points - 1.0
    
    def constraint_ineq(h):
        return np.concatenate([h, 1.0 - h])
    
    # bounds for each h_i: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Setup constraints for scipy.optimize
    constraints = [
        {'type': 'eq', 'fun': constraint_eq},
        {'type': 'ineq', 'fun': lambda x: constraint_ineq(x)[n_points:]}  # h_i <= 1
    ]
    
    # Initial guess
    h_current = h0.copy()
    
    best_h = h_current.copy()
    best_c5 = objective(h_current)
    
    # Iteratively optimize
    max_iter = 500
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Run optimization with bounds and constraints
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                          bounds=bounds, constraints=constraints,
                          options={'ftol': 1e-7, 'maxiter': 100})
            
            if not res.success or np.isnan(res.fun):
                break
                
            h_candidate = res.x
            c5_candidate = objective(h_candidate)
            
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                
            # update h_current for next iteration
            h_current = h_candidate.copy()
            
        except Exception:
            break
            
        # Regularly check time limit
        if time() - start_time > budget_s:
            break
    
    # Final refinement on best result
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h)
    
    # Ensure final solution is feasible
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2.0)
    
    return (final_h.tolist(), final_c5, n_points)
