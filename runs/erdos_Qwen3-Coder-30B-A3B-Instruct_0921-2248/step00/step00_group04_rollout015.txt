    h = Variable(n_points)
    C5 = Variable()

    constraints = []
    
    # Mass constraint
    constraints.append(sum_entries(h) == n_points / 2)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints
    for k in lags:
        # Compute shifted indices
        k_int = int(np.round(k / dx))
        if abs(k_int * dx - k) > 1e-10:
            # Interpolate for non-integer shifts
            k_frac = k / dx - k_int
            l_idx = np.arange(n_points)
            r_idx = np.roll(l_idx, -k_int)
            r_idx = np.where(r_idx < 0, r_idx + n_points, r_idx)
            r_idx = np.where(r_idx >= n_points, r_idx - n_points, r_idx)

                w2 = k_frac
                integral += h[i] * (1 - w1 * h[j1] - w2 * h[j2]) * dx
