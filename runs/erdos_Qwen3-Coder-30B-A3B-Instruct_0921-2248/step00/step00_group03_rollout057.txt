import numpy as np
import cvxpy as cp
from time import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # fallback to constructing a reasonable starting point
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[10:20] = 1.0
        h0[50:60] = 0.0
    
    # Scale to correct integral
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    dx = 2.0 / n_points
    max_k = n_points - 1
    
    # Define problem
    h = cp.Variable(n_points, name="h")
    C = cp.Variable(1, name="C")
    
    # Constraints: C >= integral for all shifts k
    constraints = []
    
    # sum(h) = n_points / 2
    constraints.append(cp.sum(h) == n_points / 2)
    
    # 0 <= h[i] <= 1
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints
    for k in range(-max_k, max_k + 1):
        if time() - start_time > budget_s - 5:
            break
        # Compute the correlation-like term using manual indexing
        if k >= 0:
            weights = h[k:] * (1 - h[:n_points-k])
            corr_val = cp.sum(weights)  # C_i = h_i * (1 - h_{i+k})
        else:
            weights = h[:k] * (1 - h[-k:])
            corr_val = cp.sum(weights)
        # Normalize by dx for consistent units
        constraints.append(corr_val <= C * dx)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Formulate and solve
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.ECOS, max_iters=200, verbose=False, timeout=budget_s - (time() - start_time) - 5)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_vals = h.value
            c5_bound = C.value
        else:
            # Fallback to heuristic
            h_vals = h0.copy()
            c5_bound = 0.5  # placeholder
    except Exception:
        # Fallback to initial construction
        h_vals = h0.copy()
        if 'initial_h_values' in locals():
            c5_bound = np.max(correlate(h_vals, 1 - h_vals, mode="full")) * dx
        else:
            c5_bound = 0.5
        
    # Final check
    h_vals = np.clip(h_vals, 0, 1)
    
    # Ensure correct integral
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    final_c5 = np.max(correlate(h_vals, 1 - h_vals, mode="full")) * dx if not np.isnan(c5_bound) else 0.5
    c5_bound = final_c5
    
    elapsed = time() - start_time
    if elapsed > budget_s - 5:
        # recompute final bounds if time is almost up
        final_c5 = np.max(correlate(h_vals, 1 - h_vals, mode="full")) * dx
        c5_bound = final_c5
    
    return (h_vals.tolist(), float(c5_bound), n_points)
