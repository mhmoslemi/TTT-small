import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h):
    """Compute C5 bound from step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum value scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx
