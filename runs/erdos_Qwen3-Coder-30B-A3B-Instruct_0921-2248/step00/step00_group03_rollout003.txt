import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        # Start with a piecewise constant function, more evenly spaced ones
        h = np.zeros(n_points)
        # Place 1s in approximately even intervals to satisfy sum condition
        # sum(h) = n_points / 2
        num_ones = n_points // 2
        indices = np.linspace(0, n_points - 1, num_ones, dtype=int)
        h[indices] = 1.0
        # Normalize to required integral
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 120  # At most 120 iterations with budget
    start_time = time.time()
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Main optimization loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.98:
            break
        
        # Compute the full correlation for all k
        # This computes=np.correlate(h, 1-h, mode="full")
        # The full correlation length is 2*n_points - 1
        corr = np.correlate(h, 1 - h, mode="full")
        # Need to mult by dx to get the actual integral values
        corr = corr * dx
        # maximum overlap for any shift
        max_overlap = np.max(corr)
        max_idx = np.argmax(corr)
        # The actual shift k is between -n_points+1 and n_points-1
        # So shift k = max_idx - (n_points - 1)
        k_max = max_idx - (n_points - 1)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # We now want to reduce the overlap corresponding to shift k_max
        # We compute the gradient of the overlap with respect to each h[i]
        # overlap_k = sum(h[i] * (1 - h[(i + k) % n_points]))
        # d/dh[i] overlap_k = (1 - h[(i + k) % n_points])
        
        # The gradient for each h[i] when overlap_k is the max
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # When overlap_k is computed, we're summing h[i] * (1 - h[(i+k)%n_points])
            # The derivative w.r.t. h[i] is (1 - h[(i+k)%n_points])
            j = (i + k_max) % n_points
            gradient[i] = 1.0 - h[j]
            
        # Update h in direction of negative gradient, but with some adaptive step size
        # Let's use a small step size, and clip the update
        step_size = 0.005
        h_new = h - step_size * gradient
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        
        # now we need to ensure sum(h) = n_points / 2
        # adjust the correction so that it respects the constraint
        old_sum = np.sum(h)
        new_sum = np.sum(h_new)
        adjustment = (n_points / 2 - new_sum) / n_points
        h_new = h_new + adjustment
        h_new = np.clip(h_new, 0.0, 1.0)
        
        h = h_new
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_erdos_solution(h_values):
    # Compute overlap integral for all shifts k and return max
    # Use np.correlate and mode="full" to compute all overlaps
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # np.correlate(h, 1-h, mode="full") computes the full convolution/autocorrelation
    # Then multiply by dx to convert to continuous integral
    corr = np.correlate(h, 1 - h, mode="full")
    corr = corr * dx
    return np.max(corr)
