import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode="full")
    # Only consider valid lags (0 to n-1)
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

def initial_h():
    """Create a feasible starting point"""
    global initial_h_values
    try:
        return np.array(initial_h_values)
    except:
        # Fallback: create a simple feasible solution
        n = 45
        h = np.ones(n) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h[2] = 0.3
        h[3] = 0.7
        # Normalize to make sum = n/2
        h = h * (n / 2) / np.sum(h)
        return h

def generate_neighbor(h, n_points):
    """Generate a neighbor by flipping an interval"""
    h_new = h.copy()
    # Choose interval length (1 or 3)
    length = np.random.choice([1, 3])
    # Choose start position
    start = np.random.randint(0, n_points - length + 1)
    # Flip the interval: h_new[start:start+length] = 1 - h_new[start:start+length]
    for i in range(start, start + length):
        h_new[i] = 1 - h_new[i]
    # Ensure bounds
    h_new = np.clip(h_new, 0, 1)
    # Project back to correct integral
    current_sum = np.sum(h_new)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Adjust one element to fix sum
        diff = target_sum - current_sum
        h_new[-1] = max(0, min(1, h_new[-1] + diff))
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved C5 bound"""
    np.random.seed(seed)
    
    # Initialize
    h = initial_h()
    n_points = len(h)
    dx = 2.0 / n_points
    C5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_C5 = C5
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.999
    T = T0
    max_iter = int(1e6)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate neighbor
        h_new = generate_neighbor(h, n_points)
        C5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject
        if C5_new < C5:
            h = h_new
            C5 = C5_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
        else:
            # Accept with probability
            if np.random.rand() < np.exp((C5 - C5_new) / T):
                h = h_new
                C5 = C5_new
        
        # Cool down
        T *= alpha
        
        # Occasionally print progress
        if i % 10000 == 0:
            print(f"Iteration {i}, C5 = {C5:.6f}, Best = {best_C5:.6f}")
    
    # Final validation
    final_C5 = evaluate_erdos_solution(best_h)
    
    # Ensure final constraints
    assert np.all(0 <= best_h) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points / 2) < 1e-6
    
    h_result = best_h.tolist()
    return (h_result, final_C5, n_points)
