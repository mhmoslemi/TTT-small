import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback initialization
        n_points = 70
        h = np.zeros(n_points)
        # Place 1s at regular intervals
        for i in range(n_points // 2):
            h[i * 2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate_c5(h_vec):
        # Compute all cross-correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Multiply by dx for integral scaling
        corr = corr * dx
        # Return max overlap value
        return np.max(corr)
    
    def project_and_normalize(h_vec):
        # Project to [0, 1]
        h_vec = np.clip(h_vec, 0, 1)
        # Normalize to sum = target_sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def optimize_step(h_vec):
        c5_val = evaluate_c5(h_vec)
        # Find position of maximum overlap
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        k_max = np.argmax(corr)
        # Get the contribution of each point
        grad = np.zeros_like(h_vec)
        # Loop through all positions
        for i in range(n_points):
            # Check the shift k_max
            j = (i + k_max) % n_points  # Index in shifted array
            grad[i] = (1 - h_vec[j])
        # Update h
        step_size = 0.01
        h_new = h_vec - step_size * grad
        h_new = project_and_normalize(h_new)
        return h_new, c5_val
    
    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h)
    
    iteration = 0
    max_iter = 1200
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        current_h, prev_c5 = optimize_step(current_h)
        new_c5 = evaluate_c5(current_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
            
    return (best_h.tolist(), best_c5, len(best_h))
