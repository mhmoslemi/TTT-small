import numpy as np
import scipy.fft as fft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initial guess: symmetric tent function
        n_points = 100
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (mid + 0.5)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    max_iter = 10000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute FFT plan for optimization
    H = fft.fft(h)
    GH = fft.fft(1 - h)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap in frequency domain
        GH_freq = H * np.conj(H)
        overlap = fft.ifft(GH_freq).real
        
        # Normalize and find maximum overlap
        max_overlap_idx = np.argmax(overlap)
        max_overlap_value = overlap[max_overlap_idx] * dx
        
        if max_overlap_value < best_c5:
            best_c5 = max_overlap_value
            best_h = h.copy()
        
        # Early stopping if tolerance is met
        if best_c5 <= 0.3807:
            break
            
        # Gradient-based update (simplified approach)
        # Trying to reduce the largest overlap
        # This is basic gradient descent approach - could be improved
        grad = np.zeros(n_points)
        shift = max_overlap_idx
        for i in range(n_points):
            j = (i + shift) % n_points
            # Gradient: partial derivative of overlap w.r.t. h[i]
            grad[i] = (1 - h[j]) * dx
            
        # Update h with gradient step
        step_size = 0.001
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to satisfy sum constraint
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Use new h for next iteration
        h = h_new
        H = fft.fft(h)
        GH = fft.fft(1 - h)
    
    # Final evaluation
    H = fft.fft(h)
    GH = fft.fft(1 - h)
    overlap = fft.ifft(GH * np.conj(H)).real
    final_c5 = np.max(overlap) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
