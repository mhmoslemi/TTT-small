import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points // 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = (h > 0.5).astype(int)
        current_sum = np.sum(h)
        # Adjust to target sum
        if current_sum < target_sum:
            # Set zeros to ones
            zero_indices = np.where(h == 0)[0]
            for i in range(target_sum - current_sum):
                if i < len(zero_indices):
                    h[zero_indices[i]] = 1
        elif current_sum > target_sum:
            # Set ones to zeros
            one_indices = np.where(h == 1)[0]
            for i in range(current_sum - target_sum):
                if i < len(one_indices):
                    h[one_indices[i]] = 0
    else:
        h = np.zeros(n_points, dtype=int)
        h[:target_sum] = 1
        np.random.shuffle(h)
    
    # Evaluate initial solution
    def evaluate(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    current_c5 = evaluate(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing
    T0 = 1.0
    T = T0
    alpha = 0.995
    max_iter = 100000
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping two random 0 and 1
        zero_indices = np.where(h == 0)[0]
        one_indices = np.where(h == 1)[0]
        
        if len(zero_indices) > 0 and len(one_indices) > 0:
            i = np.random.choice(zero_indices)
            j = np.random.choice(one_indices)
            
            # Swap
            h[i], h[j] = h[j], h[i]
            
            new_c5 = evaluate(h)
            
            # Accept or reject
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            else:
                # Revert swap
                h[i], h[j] = h[j], h[i]
                
        T *= alpha
    
    # Ensure final solution meets constraints
    actual_sum = np.sum(best_h)
    if actual_sum != target_sum:
        # Adjust to meet constraint
        if actual_sum < target_sum:
            # Turn some 0s to 1s
            zero_indices = np.where(best_h == 0)[0]
            for i in range(target_sum - actual_sum):
                if i < len(zero_indices):
                    best_h[zero_indices[i]] = 1
        elif actual_sum > target_sum:
            # Turn some 1s to 0s
            one_indices = np.where(best_h == 1)[0]
            for i in range(actual_sum - target_sum):
                if i < len(one_indices):
                    best_h[one_indices[i]] = 0
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
