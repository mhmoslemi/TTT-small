import numpy as np
import random
import time
from scipy import signal

def evaluate_h(h, dx):
    """Compute C5 bound for given h"""
    # Cross-correlation of h and (1-h) with full mode
    correlation = signal.correlate(h, 1 - h, mode='full')
    # Maximum value gives us the overlap
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with initial construction if available, or create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = (h > np.median(h)).astype(int)
    else:
        n_points = 200
        h = np.zeros(n_points, dtype=int)
        # Set roughly half to 1s
        h[:n_points//2] = 1
        np.random.shuffle(h)
        
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    current_C5 = evaluate_h(h, dx)
    best_C5 = current_C5
    T = 1.0
    alpha = 0.9995
    start_time = time.time()
    
    # Simulated Annealing loop
    while time.time() - start_time < budget_s:
        # Create a neighbor by swapping two values
        i, j = random.sample(range(n_points), 2)
        if h[i] == h[j]:
            continue
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        
        # Check the constraint (sum should be n_points/2)
        if np.sum(new_h) != n_points / 2:
            continue
            
        new_C5 = evaluate_h(new_h, dx)
        
        # Accept or reject the new solution based on SA criterion
        if new_C5 < current_C5 or random.random() < np.exp(-(new_C5 - current_C5) / T):
            h = new_h
            current_C5 = new_C5
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        T *= alpha
    
    return (best_h.tolist(), best_C5, n_points)
