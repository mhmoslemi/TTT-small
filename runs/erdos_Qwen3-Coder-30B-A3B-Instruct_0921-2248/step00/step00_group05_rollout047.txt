import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 200
        h = np.ones(n_points) * 0.5  # symmetric tent function  
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    max_iter = 10000
    alpha = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integrals (correlations)
        conv = signal.correlate(h, 1-h, mode='full')
        C5_val = np.max(conv) * dx
        
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h.copy()
            
        # Find the shift(s) that achieve maximum overlap
        max_indices = np.where(conv == np.max(conv))[0]
        
        # For each such shift, compute gradient
        grad = np.zeros(n_points)
        for idx in max_indices:
            if idx < n_points:
                # h[0] to h[idx-1] contribute -h[i] 
                grad[:idx] -= h[:idx]
                # h[idx] to h[n_points-1] contribute 1-h[i+idx] 
                if idx < n_points:
                    grad[idx:] += 1 - h[:n_points-idx]
            else:
                # full shift
                shift = idx - n_points + 1 
                if shift > 0:
                    grad[:n_points-shift] += 1 - h[shift:]
                    grad[n_points-shift:] -= h[:shift]
                else:
                    grad[-shift:] += 1 - h[:n_points+shift]
                    grad[:n_points+shift] -= h[-shift:]
        
        # Normalize gradient
        grad = grad / (np.linalg.norm(grad) + 1e-10)
        
        # Update h using gradient descent with projection
        h_new = h - alpha * grad
        
        # Project onto feasible set
        # 1. Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # 2. Adjust to meet sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_C5, n_points)
