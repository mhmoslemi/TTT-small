    """Evaluate the C5 bound using the provided h values."""
    dx = 2.0 / n_points
    # Compute cross-correlation of h with (1-h)
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode='full')
    # Extract the valid range (shifts from -(n_points-1) to n_points-1)
    # The middle section corresponds to valid overlaps
    start_idx = n_points - 1
    end_idx = 2 * n_points - 2
    overlaps = correlation[start_idx:end_idx+1]
    # Each overlap is scaled by dx for proper integration
    C5 = np.max(overlaps) * dx
    return C5
