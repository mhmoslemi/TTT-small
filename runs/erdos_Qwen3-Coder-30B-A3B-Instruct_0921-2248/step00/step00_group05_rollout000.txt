import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 250
    dx = 2.0 / n_points
    max_iterations = 10000
    initial_alpha = 0.01
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)  # Normalize to sum = n_points/2
    else:
        # Construct a symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)  # Normalize to sum = n_points/2

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute the full correlation for all shifts
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        
        # Find the shift that gives the maximum correlation
        shift = max_corr_idx - (len(h) - 1)
        
        # Compute gradient for this shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + shift >= 0 and i + shift < len(h):
                grad[i] = 1 - h[i + shift]
            else:
                grad[i] = -h[i]
        
        # Perform gradient descent step
        alpha = initial_alpha / (1 + iteration / 1000)
        h_new = h - alpha * grad
        
        # Project onto feasible set: clip to [0,1] and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        h = h_new
        
        # Update if we found a better solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Compute the overlap integral C5
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
