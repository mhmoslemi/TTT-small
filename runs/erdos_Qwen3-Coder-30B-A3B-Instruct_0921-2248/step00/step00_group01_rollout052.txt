import numpy as np
import random
import time
from scipy import correlate

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    # Compute cross-correlation of h and (1 - h)
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600):
    # Set random seed for reproducibility
    random.seed(seed)
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        # Fill top half of indices with 1s
        h[:n_points//2] = 1
        # Shuffle to get random initial configuration
        indices = list(range(n_points))
        random.shuffle(indices)
        h = h[indices]
    
    # Ensure the constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-8:
        # Reconstruct such that it sums to n_points/2
        sorted_h = np.sort(h)[::-1]
        target_count = n_points // 2
        for i in range(n_points):
            if i < target_count:
                h[i] = 1
            else:
                h[i] = 0
    
    # SA parameters
    T0 = 0.5
    alpha = 0.999
    T = T0
    max_iter = 1000000
    start_time = time.time()
    
    # Current best
    best_h = np.copy(h)
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    
    # SA loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate neighbor
        h_new = np.copy(h)
        # Swap two random elements (one 0 and one 1)
        zeros = np.where(h_new == 0)[0]
        ones = np.where(h_new == 1)[0]
        
        if len(zeros) > 0 and len(ones) > 0:
            idx0 = random.choice(zeros)
            idx1 = random.choice(ones)
            h_new[idx0] = 1
            h_new[idx1] = 0
            
            # Evaluate
            new_c5 = evaluate_c5(h_new, dx)
            
            # Accept or reject based on Metropolis criterion
            delta = new_c5 - current_c5
            if delta < 0 or random.random() < np.exp(-delta / T):
                h = h_new
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = np.copy(h)
            
            # Cool down temperature
            T *= alpha
    
    # Final check of the constraint, ensure h values are binary
    h_final = np.round(best_h).astype(int)
    h_final = h_final.astype(float)
    
    final_c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
