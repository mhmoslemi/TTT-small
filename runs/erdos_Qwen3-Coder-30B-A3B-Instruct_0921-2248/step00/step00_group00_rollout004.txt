import numpy as np
from scipy import correlate
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    # Compute the maximum overlap integral
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all cross-correlations
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral for each lag k
    max_k = n_points - 1
    max_overlap = 0
    for k in range(max_k + 1):
        # k corresponds to lag in the cross-correlation
        # lag k = k gives the correct alignment
        # Cross correlation at k=0 is k=0 in full mode
        # For lag k, we want corr[n_points-1+k] for positive k
        # for negative k, we want corr[n_points-1+k] where k is negative
        # The actual lag index in corr is from 0 to 2*n_points - 2
        idx = n_points - 1 + k if k >= 0 else n_points - 1 + k
        # Only consider valid overlaps
        if idx >= 0 and idx < len(corr):
            # The integral is corr[idx] * dx / (2*dx)
            # But we need to think of corr as computing the full convolution
            # so corr[k] (for k in range(2*n_points - 1)) gives the integral
            # of h(x) * (1 - h(x+k)) at lag k, which is the correct scaling
            val = corr[idx] * dx
            max_overlap = max(max_overlap, val)
    return max_overlap
