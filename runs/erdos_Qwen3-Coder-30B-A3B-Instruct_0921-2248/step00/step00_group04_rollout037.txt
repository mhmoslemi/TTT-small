import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Get the maximum overlap for all lags
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with a reasonable size
    dx = 2.0 / n_points
    max_iter = 1000
    tol = 1e-4
    lag_step = 0.2
    top_k = 10
    
    # Initial lag set
    lag_range = np.arange(-2, 2 + lag_step, lag_step)
    lag_set = np.round(lag_range, decimals=2)
    lag_set = np.unique(lag_set)
    
    # Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_initial = np.array(initial_h_values)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    else:
        h_initial = np.ones(n_points) * 0.5
        h_initial[0] = 0.4
        h_initial[1] = 0.6
    
    # Current best solution
    h_current = h_initial.copy()
    C5_current = evaluate_erdos_solution(h_current, n_points)
    
    # Main loop
    prev_C5 = float('inf')
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Construct constraints
        # LP variables
        h = Variable(n_points)
        C5 = Variable()
        
        # Constraints
        constraints = [0 <= h[i] for i in range(n_points)]
        constraints += [h[i] <= 1 for i in range(n_points)]
        constraints += [sum(h) * dx == 1]
        constraints += [C5 >= 0]
        
        # Add overlap constraints
        for k in lag_set:
            k_int = int(np.round(k / dx))
            corr_sum = 0
            for i in range(n_points):
                j = (i - k_int) % n_points
                corr_sum += h[i] * (1 - h[j])
            constraints.append(corr_sum * dx <= C5)
        
        # Solve optimization problem
        objective = Minimize(C5)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=GLPK, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_sol = h.value
                C5_sol = prob.value
            else:
                break  # break if no solution
        except:
            break
        
        # Validate solution
        if not np.all(0 <= h_sol) or not np.all(h_sol <= 1):
            break
        h_sol = np.maximum(np.minimum(h_sol, 1), 0)
        h_sol = h_sol / np.sum(h_sol) * (n_points / 2)
        
        # Check if we improved
        if C5_sol < C5_current - tol:
            h_current = h_sol.copy()
            C5_current = C5_sol
        else:
            # Re-evaluate with full correlation to update lag set
            pass
        
        # Compute all overlaps for all lags
        corr_values = []
        for k in np.arange(-2, 2, 0.05):
            k_int = int(np.round(k / dx))
            corr_sum = 0
            for i in range(n_points):
                j = (i - k_int) % n_points
                corr_sum += h_sol[i] * (1 - h_sol[j])
            corr_values.append(corr_sum * dx)
        
        # Find worst k values
        sorted_indices = np.argsort(corr_values)[::-1]
        top_indices = sorted_indices[:max(top_k, 1)]
        new_lags = [k for i, k in enumerate(np.arange(-2, 2, 0.05)) if i in top_indices]
        
        # Update lag set
        lag_set = np.unique(np.concatenate((lag_set, new_lags)))
        
        if abs(prev_C5 - C5_current) < tol:
            break
        prev_C5 = C5_current
    
    # Final evaluation
    final_C5 = evaluate_erdos_solution(h_current, n_points)
    
    return (h_current, final_C5, n_points)
