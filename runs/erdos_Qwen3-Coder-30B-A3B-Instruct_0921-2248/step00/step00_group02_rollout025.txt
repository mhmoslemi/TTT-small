import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Compute overlaps for all lags
    overlaps = correlate(h, 1 - h, mode="full")
    # Scale by dx
    overlaps *= dx
    # Return max overlap
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):

        # Add a small taper in the middle to help optimize
        h[n_points//2] = 0.5
        h[n_points//2 + 1] = 0.5
        h = h / np.sum(h) * (n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    
    # Parameter initialization
    max_iter = 120
    delta = 0.1
    step_size = 0.05
    max_stall = 20
    stall_count = 0

        if time.time() - start_time > budget_s - 1:  # Leave 1s margin
            break
            
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        overlaps *= (2.0 / n_points)
        max_overlap = np.max(overlaps)
        
        # Find indices where overlaps are not too far from max_overlap
        eps = 0.01 * max_overlap
        high_overlap_indices = np.where(overlaps >= max_overlap - eps)[0]

        if len(high_overlap_indices) == 0:
            high_overlap_indices = np.argsort(overlaps)[-10:]  # Top 10 overlap indices

        adjustments_made = 0
        for k_idx in high_overlap_indices:
            if adjustments_made >= 3:  # Limit per iteration
                break
            k = k_idx - (n_points - 1)  # Convert index to actual lag

                    new_h[x] -= delta

                    # Clamp bounds and adjust if necessary to preserve sum
                    if new_h[x] < 0:

                        new_h[x] = 0

                        new_h[x] -= delta2
                        new_h[x + k] = 1

                    total_diff = np.sum(new_h) - np.sum(h)

                        new_h[0] -= total_diff

        # Update current solution
        h = best_h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
