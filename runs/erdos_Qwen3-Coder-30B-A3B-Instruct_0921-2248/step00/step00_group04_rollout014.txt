import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    # Compute C5 using full cross-correlation
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value and scale by dx
    return np.max(corr) * dx

def normalize_h(h, target_sum):
    # Ensure h sums to target_sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    return h

def project_h_to_bounds(h):
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def perturb_h(h, epsilon, target_sum):
    # Make small random changes to h and maintain sum
    n = len(h)
    indices = np.random.choice(n, size=max(1, n//10), replace=False)
    delta = np.random.normal(0, epsilon, len(indices))
    
    # Adjust so we don't go out of bounds by redistributing
    new_h = h.copy()
    for i, d in zip(indices, delta):
        new_h[i] += d
    
    # Project to [0, 1]
    new_h = project_h_to_bounds(new_h)
    
    # Renormalize
    new_h = normalize_h(new_h, target_sum)
    
    return new_h

    # Ensure feasibility

    h = normalize_h(h, target_sum)
    
    # Initial evaluation
    C5 = evaluate_overlap(h, dx)
    best_C5 = C5
    best_h = h.copy()
    
    # Simulated annealing parameters
    T = 1.0
    cooling_rate = 0.9995
    min_T = 1e-6
    epsilon = 0.1
    
    # Timing
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Perturb h
        new_h = perturb_h(h, epsilon, target_sum)
        
        # Evaluate
        new_C5 = evaluate_overlap(new_h, dx)
        
        # Accept or reject
        if new_C5 < C5 or np.random.rand() < np.exp(-(new_C5 - C5) / T):

            C5 = new_C5
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
        
        # Update temperature and epsilon
        T *= cooling_rate
        if T < min_T:
