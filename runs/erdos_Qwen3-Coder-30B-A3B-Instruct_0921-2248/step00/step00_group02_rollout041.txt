from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

def evaluate_c5(h, n_points, dx):
    """Evaluate C5 from the overlap integral."""
    # Create the full convolution using numpy
    h_full = np.concatenate([h, np.zeros(n_points)])
    h_flip = np.concatenate([np.zeros(n_points), 1 - h])
    overlap = np.correlate(h_full, h_flip, mode="full")
    # Extract the valid overlaps (they are indexed from n_points-1 to 2*n_points-2)
    valid_overlaps = overlap[n_points-1 : 2*n_points-1]
    return np.max(valid_overlaps) * dx

def solve_feasible_h(h_start, c5_target, n_points, dx, time_limit):
    """Attempt to find a feasible h with overlap <= c5_target."""
    # Use scipy minimize to optimize h to minimize overlap
    # We must enforce:
    # 1. h[i] in [0, 1]
    # 2. integral condition
    # 3. overlap constraint
    
    def objective(h):
        # This aims to minimize C5 (hence we use the actual overlaps)
        return evaluate_c5(h, n_points, dx)
    
    def constraint_integral(h):
        return np.sum(h) * dx - 1.0
        
    def constraint_overlap(h):
        # This will be enforced via bounds in the constraints
        # More sophisticated approach using scipy.optimize with bounds
        return None  # We'll add non-overlap in constraints
        
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize
    res = minimize(
        objective,
        h_start,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 1000},
        constraints={'type': 'eq', 'fun': constraint_integral}
    )
    
    if res.success:
        h_opt = res.x
        actual_c5 = evaluate_c5(h_opt, n_points, dx)
        if actual_c5 <= c5_target:
            return h_opt
        # If we got here but didn't meet target, we try fixing the constraints
        else:
            # We take the last h that was feasible
            return h_start
    
    # Try a simple approach:
    return h_start


def refine_h(h0, n_points, dx, time_limit):
    """Try to further refine the solution to lower C5."""
    start_time = time.time()
    
    def obj(h):
        return evaluate_c5(h, n_points, dx)
    
    def cons_integral(h):
        return np.sum(h) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(
        obj,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 1000},
        constraints={'type': 'eq', 'fun': cons_integral}
    )
    
    if result.success and time.time() - start_time < time_limit:
        return result.x
    
    return h0
