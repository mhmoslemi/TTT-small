import numpy as np
import cvxpy as cp
from time import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50
    lag_step = 0.2
    lag_range = np.arange(-2, 2 + lag_step/2, lag_step)

    if 'initial_h_values' in globals():

        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time()
    
    # Constraint refinement loop
    K = set(lag_range)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 5:
            break
            
        # Create LP problem
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        constraints = [
            cp.sum(h_vars) * dx == 1,
            h_vars >= 0,
            h_vars <= 1,
            C5 >= 0
        ]
        
        # Add overlap constraints
        for k in K:
            # Because of periodicity, handle convolution with proper indexing
            lag_idx = int(round(k / dx))
            if lag_idx < 0:
                # h[i] * (1 - h[i + |lag_idx|])
                lagged = np.roll(h_vars, -lag_idx)
                expr = cp.sum(h_vars * (1 - lagged)) * dx
            elif lag_idx > 0:
                # h[i] * (1 - h[i - lag_idx])
                lagged = np.roll(h_vars, lag_idx)
                expr = cp.sum(h_vars * (1 - lagged)) * dx
            else:
                # The case k=0 may not be needed in full correlation, but good to include
                # We can do it manually in this loop using full correlation
                pass
            
            # Load actual h values for computing correlation
            if k != 0:
                lag_idx = int(round(k / dx))
                lag_func = np.roll(h, -lag_idx) if lag_idx > 0 else np.roll(h, lag_idx)
                expr = cp.sum(h_vars * (1 - lag_func)) * dx
                constraints.append(expr <= C5)
        
        # Objective function
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_sol = h_vars.value
                C5_val = C5.value
            else:
                break
        except:
            break
            
        # Validate
        if C5_val is not None and not np.isnan(C5_val) and np.isfinite(C5_val):
            if C5_val < best_C5:
                best_C5 = C5_val
                best_h = h_sol.copy()
        
        # Compute all correlations to find worst cases
        h_pos = h_sol.reshape(-1, 1)
        h_neg = (1 - h_sol).reshape(-1, 1)
        
        # Manual full convolution for accuracy
        corr = correlate(h_sol, h_neg, mode='full')
        lags_full = np.arange(-n_points+1, n_points)
        
        corr_vals = corr * dx
        max_corr_idx = np.argmax(corr_vals)
        max_corr_val = corr_vals[max_corr_idx]
        max_corr_lag = lags_full[max_corr_idx]
        
        # Ensure we're not just playing with the built-in relationship
        # Stick with an improved over the current set
        if len(K) <= 1 and max_corr_val <= 1e-8:
            break
        elif len(K) <= 1 or max_corr_val >= 1e-8:
            # Compute more detailed set of bad lags
            # Add some random lags that are close to the maximum
            bad_lags = [max_corr_lag]
            # Add a few close lags to refine
            for d in np.linspace(-0.2, 0.2, 5):
                test_lag = max_corr_lag + d
                if abs(test_lag) <= 2 and test_lag not in K:
                    K.add(test_lag)
            # Add 10% of worst ones in case this gets refined
            top_bad_idx = np.argsort(corr_vals)[-max(5, len(corr_vals)//20):]
            top_bad_lags = lags_full[top_bad_idx]
            for lag in top_bad_lags:
                if abs(lag) <= 2 and lag not in K:
                    K.add(lag)
        
        # Replace k with precise lag values
        K.clear()
        K.update(np.arange(-2, 2 + lag_step/2, lag_step))
        
    # Final verification with bounding check
    # Evaluate final h values directly with overlap integral
    h = best_h if best_h is not None else h
    h = np.array(h)
    h = h / np.sum(h) * (n_points / 2)
    
    # Evaluate C5 with the final result edge case exclusion for small values
    h_tiled = np.concatenate([h, h])
    overlap_integral = np.zeros(2 * n_points - 1)
    for i in range(2 * n_points - 1):
        total = 0
        for j in range(n_points):
            k = i - j
            if 0 <= k < n_points:
                total += h[j] * (1 - h[k])
        overlap_integral[i] = total * dx
    final_C5 = np.max(overlap_integral)
    
    # Safety safeguard ensure C5 is finite
    if not np.isfinite(final_C5):
        final_C5 = 1.0 if 'initial_h_values' not in globals() else None
        h = np.ones(n_points) * 0.5 if final_C5 == 1.0 else h

    # Return best found h values or fallback
    if np.isnan(final_C5) or not np.isfinite(final_C5):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = h / np.sum(h) * (n_points / 2)
        else:
            h = np.ones(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
        final_C5 = 1.0
    
    # Ensure final h is valid
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    return (h, float(final_C5), n_points)
