import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.001
    alpha_decay = 0.99
    tolerance = 1e-8
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Compute all overlaps for all shifts
        # Use correlate with reverse of (1-h) to compute overlap efficiently
        h_rev = 1 - h
        correlations = correlate(h, h_rev, mode='full')
        # correlations is of length 2*n_points - 1
        # take the middle n_points elements corresponding to k in [0, n_points-1]
        # but with proper indexing for shifts
        
        # overlap_k = sum h[i] * (1 - h[(i+k)%n_points]) * dx
        # This is just correlate(h, 1-h, mode='full') but the shift index is offset
        
        # For shift k (in range 0 to n_points-1):
        # correlations[n_points - 1 - k] gives the correlation for shift k in the forward direction
        # Actually letting h be padded in the correlogram, we index properly:
        # shift from 0 to 2*n_points-2:
        # k=0: correlation[2*n_points-2] (but now correctly treated as shift 0)
        # corr for shift k=0,1,...,n_points-1 are corr[2*n_points-2], corr[2*n_points-3], ..., corr[n_points-1]
        # slice [n_points-1 : 3*n_points-3]
        # The number of correlations is 2*n_points-1
        # So for shift k, we look at index (2*n_points - 2) - k
        
        # shift from 0 to 2*n_points-2 (full)
        # shift 0 -> corr[2*n_points-2]
        # shift k -> corr[2*n_points-2-k]
        # But our valid shifts are 0 to n_points-1
        # Overlaps for shift 0 to n_points-1: corresponding to indices [2*n_points-2 : 2*n_points-n_points-1]
        # Wait, this is not right: our range should be 2*n_points - 2 (for k=0) down to n_points-1 (for k=n_points-1)
        # idx_k = 2*n_points - 2 - k, so idx_max = 2*n_points - 2, idx_min = n_points - 1   
        # So indices are [n_points-1 : 2*n_points-1], length = n_points
        # These correspond to shifts k=0 to k=n_points-1
        
        overlaps = correlations[n_points-1 : 2*n_points-1]
        assert len(overlaps) == n_points
        
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            
        # Multiply by dx to get the actual integral value
        # Actually the correlation already accounts for dx if we're careful with normalization
        # compute gradient of overlap_k_max w.r.t. h[i]
        # overlap_k_max = sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k_max)%n_points])
        # d/dh[i] overlap_k_max = 1 - h[(i+k_max)%n_points] - h[i]
        # actually gradient sign is sign(h[(i + k_max) % n_points] + h[i] - 1)
        
        grad_signs = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad_signs[i] = np.sign(h[j] + h[i] - 1)
        
        # Update h
        h = h - alpha * grad_signs
        h = np.clip(h, 0, 1)
        
        # Normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2.0) / current_sum
        
        if iteration % 50 == 0:
            alpha *= alpha_decay
            
    # Final evaluation
    final_C5 = compute_C5(best_h, dx)
    return (best_h.tolist(), final_C5, n_points)

def compute_C5(h, dx):
    h_rev = 1 - h
    correlations = correlate(h, h_rev, mode='full')
    overlaps = correlations[len(h)-1 : 2*len(h)-1]
    return np.max(overlaps) * dx
