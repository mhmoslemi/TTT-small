def evaluate_erdos_solution(h, dx):
    """Evaluate C5 = max_k int h(x) (1 - h(x+k)) dx """
    # Compute the full cross-correlation h * (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Get lags
    n = len(h)
    lags = np.arange(-n+1, n)
    # Evaluate all overlaps
    overlaps = corr * dx
    return np.max(overlaps)
