import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant for given h values"""
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a binary initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        
    # Optimization parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint function
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Objective function for optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Ensure constraints
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Time tracking
    start_time = time.time()
    
    # Optimization loop
    max_iter = 5000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Use scipy optimization
        try:
            res = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                h = res.x
                h = project(h)
                c5 = objective(h)
                
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        except:
            pass
            
        # Random projection to escape local optima
        h = project(h)
        c5 = objective(h)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Perturb solution slightly
        perturbation = np.random.normal(0, 0.01, n_points)
        h = h + perturbation
        h = project(h)
        
        iter_count += 1
        
        # Early termination if solution is very good
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    try:
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            refined_h = project(res.x)
            refined_c5 = objective(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
