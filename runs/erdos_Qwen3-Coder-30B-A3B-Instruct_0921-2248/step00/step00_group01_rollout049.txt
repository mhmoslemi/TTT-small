def evaluate_erdos_solution(h):
    """Auxiliary function to compute C_5 for a given h"""
    # The h is already normalized
    corr = np.correlate(h, 1 - h, mode='full')  
    dx = 2.0 / len(h)
    corr *= dx
    return np.max(corr)
