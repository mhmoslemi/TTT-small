import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all correlations
    correlations = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    max_overlap = np.max(correlations) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h values
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Create the variables
    h = cp.Variable(n_points, bounds=(0, 1))
    C = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Overlap constraints
    # For each shift k in [-n_points+1, n_points-1]
    for k in range(-(n_points-1), n_points):
        # Compute indices for the shifted array
        shift_indices = np.arange(n_points)
        if k > 0:
            shift_indices = (shift_indices + k) % n_points
        elif k < 0:
            shift_indices = (shift_indices + k + n_points) % n_points
            
        # Compute overlap
        overlap = cp.sum(cp.multiply(h, 1 - h[shift_indices]))
        constraints.append(overlap <= C / dx)
        
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, max_iters=200, time_limit=budget_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_opt = C.value
        else:
            h_opt = h_init
            c5_opt = evaluate_erdos_solution(h_init)
    except Exception:
        h_opt = h_init
        c5_opt = evaluate_erdos_solution(h_init)
    
    # Final validation
    if np.any(h_opt < 0) or np.any(h_opt > 1):
        h_opt = np.clip(h_opt, 0, 1)
    if abs(np.sum(h_opt) * dx - 1.0) > 1e-6:
        # Adjust h_opt to meet sum constraint
        scale_factor = (n_points / 2) / np.sum(h_opt)
        h_opt = h_opt * scale_factor
        
    c5_bound = evaluate_erdos_solution(h_opt)
    
    return (h_opt.tolist(), float(c5_bound), n_points)
