import numpy as np
from scipy import fft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent construction if available

        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fallback to a simple construction
        n_points = 100
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//4] = 1.0
        h0[n_points//4:3*n_points//4] = 0.0
        h0[3*n_points//4:] = 1.0
        h0 = h0 / np.sum(h0) * 2.0  # Normalize to integral 1
        
    dx = 2.0 / n_points
    h = h0.copy()
    
    # Helper to project h onto feasible region
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        scalar = 2.0 / np.sum(h_vec)
        h_vec = h_vec * scalar
        return h_vec
    
    # Helper to compute C5
    def compute_C5(h_vec):
        # Compute correlation using FFT
        h_fft = fft.fft(h_vec)
        one_minus_h = 1 - h_vec
        one_minus_h_fft = fft.fft(one_minus_h)
        correlation = fft.ifft(h_fft * np.conj(one_minus_h_fft))
        # Only take the overlap part
        result = np.abs(correlation[:(2*n_points-1)])
        return np.max(result) * dx
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Gradient descent with projection
    max_iter = 5000
    lr = 0.01
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Compute gradient
        h_fft = fft.fft(h)
        one_minus_h = 1 - h
        one_minus_h_fft = fft.fft(one_minus_h)
        correlation = fft.ifft(h_fft * np.conj(one_minus_h_fft))
        # Gradient of C5 with respect to h (approximated)
        # We're going to do a simple gradient ascent, but we'll project
        # to ensure h remains feasible.
        
        # For simplicity, we do a direct gradient ascent descent 
        # on the largest correlation value, with a conservative update
        grad = np.real(fft.ifft(np.conj(one_minus_h_fft) * h_fft))
        
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = project_h(h_new)
        h = h_new
        
        C5 = compute_C5(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
    # Final validation
    final_C5 = compute_C5(best_h)
    return (best_h, final_C5, n_points)
