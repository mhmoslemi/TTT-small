import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a symmetric tent function
        h = np.array([0.5 * (1 - abs(i - (n_points-1)/2) / ((n_points-1)/2)) 
                      for i in range(n_points)])
    
    # Normalize to have integral = 1
    h = h * (1.0 / np.sum(h) * n_points * dx)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Use scipy optimization to refine
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        c5 = evaluate_c5(h, dx)
        return c5
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    def constraint_lower(h_vec):
        return h_vec - 0.0
    
    def constraint_upper(h_vec):
        return 1.0 - h_vec
    
    constraints = [
        {'type': 'eq', 'fun': constraint_integral},
        {'type': 'ineq', 'fun': constraint_lower},
        {'type': 'ineq', 'fun': constraint_upper}
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization parameters
    iter_count = 0
    max_iter = 1000
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and iter_count < max_iter:
        iter_count += 1
        # Random perturbation to avoid local minima
        h_perturbed = h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize again
        h_perturbed = h_perturbed * (1.0 / np.sum(h_perturbed) * n_points * dx)
        
        # Optimize using scipy minimize
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=constraints, 
                           options={'maxiter': 10})
            h_new = res.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (1.0 / np.sum(h_new) * n_points * dx)
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                last_improvement = iter_count
        except:
            pass
        
        # Occasionally re-start from a better point or adjust
        if iter_count % 50 == 0 or iter_count < 10:
            # Try another construction like a uniform distribution with some noise
            h_uniform = np.ones(n_points) * 0.5
            h_uniform = h_uniform * (1.0 / np.sum(h_uniform) * n_points * dx)
            h_uniform = np.clip(h_uniform + np.random.normal(0, 0.02, n_points), 0, 1)
            h_uniform = h_uniform * (1.0 / np.sum(h_uniform) * n_points * dx)
            c5_uniform = evaluate_c5(h_uniform, dx)
            if c5_uniform < best_c5:
                best_c5 = c5_uniform
                best_h = h_uniform.copy()
                
        # Early stopping if no improvement for a while
        if iter_count - last_improvement > 50:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1.0 / np.sum(best_h) * n_points * dx)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
