import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum of these correlations
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        # Rescale to meet sum constraint
        h = h * (n_points / 2) / current_sum
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200  # Start with a reasonable number
    dx = 2.0 / n_points
    # Initialize from provided construction or use a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
        if len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
    else:
        # Create a binary-like initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    def objective(h):
        return evaluate_erdos_solution(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop with time control
    while time.time() - start_time < budget_s - 1:
        # Try SLSQP with projected initial guess
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
        
        if res.success:
            h = res.x
        else:
            # Fall back to simple projection and perturbation
            h = project_h(h, n_points)
            
        h = project_h(h, n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Add small random noise for exploration
        noise_level = 0.01
        h += np.random.normal(0, noise_level, n_points)
        h = project_h(h, n_points)
        
        # Occasionally try a random perturbation
        if np.random.rand() < 0.05:
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
            
    return (best_h.tolist(), best_c5, n_points)
