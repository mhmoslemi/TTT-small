import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations for overlap integral
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    # Initialize with a smooth step function, similar to previous solution
    n = 45
    h = np.zeros(n)
    # Start with a smooth ramp from 0 to 1
    for i in range(n):
        h[i] = min(1.0, max(0.0, (i - 5) / (n - 10)))
    # Normalize the integral to 1
    integral = np.sum(h) * (2.0 / n)
    h = h * (1.0 / integral)
    return h

def generate_neighbor(h, n_points):
    # Create a new candidate by flipping intervals of 1 or 3 points
    h_new = h.copy()
    # Try flip of size 1 or 3
    flip_size = np.random.choice([1, 3])
    # Determine starting position
    start_pos = np.random.randint(0, n_points - flip_size + 1)
    
    # Pick target values: 0 -> 1 or 1 -> 0
    # Currently we first try 0 -> 1, then 1 -> 0
    values = h_new[start_pos:start_pos+flip_size]
    if np.all(values == 0):
        # Flip all to 1 (if all are 0)
        h_new[start_pos:start_pos+flip_size] = 1.0
    elif np.all(values == 1):
        # Flip all to 0 (if all are 1)
        h_new[start_pos:start_pos+flip_size] = 0.0
        # Try to make it valid again
        # This check is needed to maintain the exact sum
        if np.sum(h_new) > n_points / 2:
            # reduce some values to match exactly
            while np.sum(h_new) > n_points / 2:
                idx = np.random.randint(0, n_points)
                if h_new[idx] > 0:
                    h_new[idx] -= 0.01
        elif np.sum(h_new) < n_points / 2:
            while np.sum(h_new) < n_points / 2:
                idx = np.random.randint(0, n_points)
                if h_new[idx] < 1:
                    h_new[idx] += 0.01
    else:
        # Mix of 0 and 1 - make it all 1s (effectively)
        h_new[start_pos:start_pos+flip_size] = 1.0
        # Adjustment to preserve exact sum
        diff = np.sum(h_new) - n_points / 2
        if abs(diff) > 1e-6:
            # Try to restore sum
            while abs(np.sum(h_new) - n_points / 2) > 1e-8:
                # Revert one to 0 if too high or add one if too low
                if np.sum(h_new) > n_points / 2:
                    # Find a 1 and change it to 0
                    idx = np.random.randint(0, n_points)
                    if h_new[idx] > 0:
                        h_new[idx] = 0.0
                elif np.sum(h_new) < n_points / 2:
                    # Find a 0 and change it to 1
                    idx = np.random.randint(0, n_points)
                    if h_new[idx] < 1:
                        h_new[idx] = 1.0
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    h = initial_h_values() if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    # Adjust to meet integral constraint
    integral = np.sum(h) * dx
    h = h * (1.0 / integral)
    
    # Validate the starting point
    if not (0 <= np.min(h) and np.max(h) <= 1):
        h = np.clip(h, 0, 1)
    if abs(np.sum(h) * dx - 1) > 1e-6:
        previous_sum = np.sum(h)
        h = h * (1.0 / previous_sum) * n_points / 2
    
    C5_best = evaluate_erdos_solution(h)
    h_best = h.copy()
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.999
    T = T0
    
    iteration = 0
    while time.time() - start_time < budget_s:
        if iteration % 1000 == 0 and T < 1e-10:
            # Reset temperature if stuck in plateau
            T = T0
        iteration += 1
        h_new = generate_neighbor(h, n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to sum constraint
        integral_new = np.sum(h_new) * dx
        h_new = h_new * (1.0 / integral_new)
        if abs(np.sum(h_new) * dx - 1) > 1e-6:
            h_new = h_new * (1.0 / np.sum(h_new)) * n_points / 2
        
        C5_new = evaluate_erdos_solution(h_new)
        delta = C5_new - C5_best
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new.copy()
            C5_best = C5_new
            h_best = h.copy()
        
        T *= alpha
        
        # Every few iterations, reassess to bounce if stuck
        if iteration % 5000 == 0:
            if C5_best < 0.38:
                # Early exit if good enough
                break
            if T < 1e-9:
                T = T0  # Restart cooling

    # Final validation
    if not (0 <= np.min(h_best) and np.max(h_best) <= 1):
        h_best = np.clip(h_best, 0, 1)
    
    # Revalidate the integral
    integral_final = np.sum(h_best) * dx
    if abs(integral_final - 1) > 1e-6:
        h_best = h_best * (1.0 / integral_final) * n_points / 2
        if abs(np.sum(h_best) * dx - 1) > 1e-6:
            # Last resort: iterate to make it right
            while abs(np.sum(h_best) * dx - 1) > 1e-6:
                idx = np.random.randint(0, n_points)
                offset = 0.001 * (1 if np.sum(h_best) < n_points / 2 else -1)
                h_best[idx] = np.clip(h_best[idx] + offset, 0, 1)
    
    final_c5 = evaluate_erdos_solution(h_best)
    
    return (h_best.tolist(), final_c5, n_points)
