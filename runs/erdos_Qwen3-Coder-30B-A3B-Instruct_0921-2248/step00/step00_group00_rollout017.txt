def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given h"""
    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")
    # Get the maximum value from the correlation (excluding zero lag)
    max_corr = np.max(corr)
    return max_corr * dx
