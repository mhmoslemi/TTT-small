def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The relevant overlaps are from lag -(n-1) to n-1 (centered at 0)
    # We want the max of these, but only the ones that have meaningful
    # overlap in [0, 2] interval. These start from the n-1'th element 
    # and go to 2*n-2'th element. In Python, that is [n-1:2*n-1].
    max_overlap = np.max(corr[n-1:2*n-1]) * dx
    return max_overlap
