import numpy as np
import time
from scipy.optimize import minimize
from cvxpy import *

def evaluate_erdos_solution(h_vals, n_points):
    """Evaluate C5 using the provided h_vals"""
    dx = 2.0 / n_points
    h = np.array(h_vals)
    # Compute all overlaps with full mode
    correlation = np.correlate(h, 1 - h, mode='full')
    # Get the maximum value
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Start with a simple construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        # Set up to satisfy sum(h) * dx = 1
        dx = 2.0 / n_points
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Binary search on C5
    left = 0.0
    right = 0.38085  # current best known upper bound
    precision = 1e-5
    max_iterations = 100
    iteration = 0
    
    # Simple gradient descent approach with bounds
    # For a more robust approach without convex solver, we'll simple
    # use a local search method
    
    def objective(h_vals):
        return evaluate_erdos_solution(h_vals, n_points)
    
    def constraint_sum(h_vals):
        dx = 2.0 / n_points
        return np.sum(h_vals) * dx - 1.0
    
    def generate_candidate(h_ref, step=0.01):
        h_new = h_ref.copy()
        # Random small perturbations
        idx = np.random.randint(0, len(h_new))
        pert = np.random.uniform(-step, step)
        h_new[idx] = np.clip(h_new[idx] + pert, 0, 1)
        # Ensure sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        scale_factor = target_sum / current_sum
        h_new = h_new * scale_factor
        return h_new
    
    def solve_local_search():
        nonlocal h, best_h, best_c5
        current_h = h.copy()
        current_c5 = evaluate_erdos_solution(current_h, n_points)
        
        for i in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
            new_h = generate_candidate(current_h, 0.01)
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                
            # Keep track of best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        return best_h, best_c5

    # Start iterations
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # For small problem sizes, let's just do simple local search
        # Instead of CVX optimization which is not easily accessible
        attempt_h, attempt_c5 = solve_local_search()
        if attempt_c5 < best_c5:
            best_h = attempt_h
            best_c5 = attempt_c5
        iteration += 1
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
