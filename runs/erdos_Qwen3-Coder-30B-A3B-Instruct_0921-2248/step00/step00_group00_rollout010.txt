import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Compute the Erdős constant C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def generate_candidate(h, n_points):
    """Generate a new candidate by flipping intervals"""
    h_new = h.copy()
    # Pick random interval to flip, either length 1 or 3
    interval_len = np.random.choice([1, 3])
    start = np.random.randint(0, n_points - interval_len + 1)
    
    # Flip the values
    for i in range(start, start + interval_len):
        h_new[i] = 1 - h_new[i]
    
    return h_new

def enforce_feasibility(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to enforce sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    diff = target_sum - current_sum
    
    # Simple adjustment: add diff/n_points to each element
    h = h + diff / n_points
    
    # Clamp again
    h = np.clip(h, 0, 1)
    
    return h

    C5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_C5 = C5
    
    # Initial temperature and cooling schedule
    T = 0.1
    alpha = 0.999
    
    # Run simulated annealing
    iterations = 0
    while True:
        # Check time budget
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Generate candidate
        h_new = generate_candidate(h, n_points)
        
        # Enforce feasibility
        h_new = enforce_feasibility(h_new, n_points)
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject move
        if C5_new < C5 or np.random.rand() < np.exp((C5 - C5_new) / T):
            h = h_new
            C5 = C5_new
            
            if C5_new < best_C5:
                best_h = h.copy()
                best_C5 = C5_new
                
        # Cool down temperature
        T *= alpha
        iterations += 1
    
    # Final validation
    final_C5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_C5, n_points)
