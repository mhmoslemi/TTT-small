import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a binary construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        # Start with a binary construction for better chance of meeting constraints
        h = np.zeros(n_points)
        # Set first half to 1 (so sum is n_points/2)
        h[:n_points//2] = 1.0
        # Random permute
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h):
        # Compute convolution of h and (1-h), then find maximum
        conv = correlate(h, 1 - h, mode="full")
        # Multiply by dx for integration normalization
        max_val = np.max(conv) * dx
        return max_val
    
    def is_feasible(h):
        return (np.all(h >= 0) and np.all(h <= 1) and 
                abs(np.sum(h) * dx - 1.0) < 1e-10)
    
    def perturb_h(h, mode="flip"):
        # Create a copy to modify
        h_new = h.copy()
        
        if mode == "flip":
            # Flip two points: one from 1 to 0, one from 0 to 1
            ones = np.where(h_new == 1)[0]
            zeros = np.where(h_new == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i1 = np.random.choice(ones)
                i2 = np.random.choice(zeros)
                h_new[i1] = 0
                h_new[i2] = 1
        elif mode == "adjust":
            # Adjust one point with small random change (keeping in bounds)
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        elif mode == "noise":
            # Add small noise and normalize
            noise_level = 0.05
            h_new += np.random.normal(0, noise_level, n_points)
            h_new = np.clip(h_new, 0, 1)
            # Re-normalize
            h_new *= (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Start with initial h
    C5 = compute_C5(h)
    if C5 < best_C5 and is_feasible(h):
        best_C5 = C5
        best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1: # Leave 1s margin
        # Try multiple perturbation modes
        perturbation_modes = ["flip", "adjust", "noise"]
        new_h = perturb_h(h, mode=np.random.choice(perturbation_modes))
        
        # Ensure feasibility for the new h
        if not is_feasible(new_h) and np.sum(new_h) > 0:
            # Re-normalize to satisfy sum constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            if not is_feasible(new_h):
                # Hack to ensure constraints; slightly adjust
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new C5
        new_C5 = compute_C5(new_h)
        
        # Accept if improvement or with some probability (simulated annealing-like)
        if new_C5 < best_C5 and is_feasible(new_h):
            best_C5 = new_C5
            best_h = new_h.copy()
            h = new_h.copy()
        elif np.random.rand() < 0.1 and is_feasible(new_h):
            h = new_h.copy()
            
        iteration += 1
        # Break early if exhausted time
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, n_points)
