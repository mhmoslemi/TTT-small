import numpy as np
import time
from scipy import correlate
import math

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    # Compute cross-correlation of h and (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Find maximum value in correlation result and multiply by dx
    max_corr = np.max(corr)
    return max_corr * dx

def is_feasible(h, n_points):
    """Check if h satisfies constraints"""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2.0) < 1e-8)

def mutate_h(h, mutation_rate, n_points):
    """Mutate h with given rate while maintaining feasibility"""
    h = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Mutate with small random perturbation
            delta = np.random.normal(0, 0.05)
            h[i] = np.clip(h[i] + delta, 0, 1)
    
    # Renormalize to satisfy sum constraint
    current_sum = np.sum(h)
    desired_sum = n_points / 2.0
    if abs(current_sum - desired_sum) > 1e-8:
        scale = desired_sum / current_sum
        h = h * scale
    return h

def crossover_h(parent1, parent2, crossover_rate):
    """Perform crossover between two parents"""
    if np.random.random() > crossover_rate:
        return parent1.copy()
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)  # crossover point
    child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
    
    # Renormalize
    current_sum = np.sum(child)
    desired_sum = n / 2.0
    if abs(current_sum - desired_sum) > 1e-8:
        scale = desired_sum / current_sum
        child = child * scale
    return child

def tournament_selection(population, fitnesses, tournament_size):
    """Perform tournament selection to select one individual"""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial h values if provided

        h0 = h0 / np.sum(h0) * 45 / 2.0  # renormalize
        n_points = 45
    else:
        n_points = 64
        h0 = np.ones(n_points) * 0.5

    # GA parameters
    population_size = 100
    mutation_rate = 0.01
    crossover_rate = 0.7
    tournament_size = 5
    max_generations = int(budget_s * 10) if budget_s < 100 else 5000
    elite_size = int(population_size * 0.1)
    stagnation_patience = 500
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2.0  # normalize
        population.append(h)
    
    # Eval initial population
    dx = 2.0 / n_points
    fitnesses = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    best_generation = 0
    
    # Evolution loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            best_generation = gen
            
        # Early stopping if no improvement for a while
        if gen - best_generation > stagnation_patience:
            break
            
        # Keep elite
        new_population = population[:elite_size]
        new_fitnesses = fitnesses[:elite_size]
        
        # Generate new offspring
        while len(new_population) < population_size:
            # Select parents
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            # Crossover
            child = crossover_h(parent1, parent2, crossover_rate)
            
            # Mutation
            child = mutate_h(child, mutation_rate, n_points)
            
            new_population.append(child)
            new_fitnesses.append(evaluate_h(child, dx))
            
        population = new_population
        fitnesses = new_fitnesses
    
    return (best_h.tolist(), best_c5, n_points)
