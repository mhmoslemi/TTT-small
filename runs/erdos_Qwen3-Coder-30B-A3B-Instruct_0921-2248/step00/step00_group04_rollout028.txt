import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Start with the provided initial construction
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available; otherwise create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    h = h / np.sum(h) * target_sum
    
    # Function to compute C5
    def compute_C5(h_vec):
        # Correlation: h * (1 - h_shifted), full mode
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum, scale by dx
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Initial C5
    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    # Annealing parameters
    T = 1.0
    cooling_rate = 0.9995
    min_T = 1e-6
    max_iter = int(1e6)
    
    iter_count = 0
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Generate new candidate by perturbing some elements
        new_h = h.copy()
        num_changes = np.random.randint(1, n_points // 4)
        indices = np.random.choice(n_points, size=num_changes, replace=False)
        
        # Perturb selected elements
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            new_h[i] += delta
        
        # Project back to [0, 1]
        new_h = np.clip(new_h, 0, 1)
        
        # Adjust to maintain sum = target_sum
        current_sum = np.sum(new_h)
        if current_sum > 0:
            scaling = target_sum / current_sum
            new_h *= scaling
            # Renormalize if needed to be exact
            diff = target_sum - np.sum(new_h)
            if abs(diff) > 1e-10:
                # Distribute difference across all points
                new_h += diff / n_points
        
        # Compute new C5
        new_C5 = compute_C5(new_h)
        
        # Accept or reject
        if new_C5 < current_C5:
            h = new_h
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()
        else:
            # Accept with probability exp(-(new_C5 - current_C5) / T)
            delta_C5 = new_C5 - current_C5
            if np.random.rand() < np.exp(-delta_C5 / T):
                h = new_h
                current_C5 = new_C5
        
        # Cool down
        T = max(T * cooling_rate, min_T)
    
    return (best_h.tolist(), best_C5, n_points)
