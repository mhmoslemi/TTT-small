import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Return the maximum value scaled by dx
    return np.max(corr) * dx

def project_h(h, n_points):
    # Project h to [0,1] and maintain sum constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to meet constraint
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 150  # Starting with medium resolution for efficiency
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Binary to continuous construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial evaluation
    current_best = evaluate_overlap(h, dx)
    best_h = h.copy()
    
    def objective(h):
        return evaluate_overlap(h, dx)
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization using SLSQP
    try:
        while time.time() - start_time < budget_s * 0.95:
            # Run minimize
            res = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if not res.success:
                break
                
            h = res.x
            # Project h to maintain constraints
            h = project_h(h, n_points)
            
            current_val = evaluate_overlap(h, dx)
            if current_val < current_best:
                current_best = current_val
                best_h = h.copy()
                
            # Adaptive restart with random perturbation
            if np.random.rand() < 0.2:
                # Perturb h slightly to escape local minima
                h = project_h(h + np.random.normal(0, 0.01, n_points), n_points)
                
    except Exception as e:
        pass  # Continue with best found so far
        
    return (best_h.tolist(), current_best, n_points)
