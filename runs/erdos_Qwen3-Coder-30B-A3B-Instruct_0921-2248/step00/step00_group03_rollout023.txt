import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback: uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to target sum
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    start_time = time.time()
    alpha = 0.001
    last_C5 = best_C5
    no_improvement_count = 0
    
    # Main optimization loop
    for iter_no in range(10000):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps
        overlaps = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute gradient for worst shift
        gradient_signs = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient_signs[i] = 1 - h[j] - h[i]
        
        # Update h
        h = h - alpha * gradient_signs
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize to correct sum
        h = h * target_sum / np.sum(h)
        
        # Check objective
        current_C5 = compute_C5(h, dx)
        
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if iter_no % 50 == 0 and iter_no > 0:
            if no_improvement_count > 10:
                alpha *= 0.95
            no_improvement_count = 0
            
        # Early termination if no significant improvement
        if abs(last_C5 - current_C5) < 1e-8:
            break
        last_C5 = current_C5
    
    return (list(best_h), best_C5, n_points)

def compute_C5(h, dx):
    # Compute all overlaps and return maximum
    overlaps = np.correlate(h, 1 - h, mode="full")
    return np.max(overlaps) * dx
