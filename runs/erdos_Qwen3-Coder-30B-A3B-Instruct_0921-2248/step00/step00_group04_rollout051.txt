import numpy as np
from scipy import optimize
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, multiply,  Constraint
import cvxpy as cp

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    # Compute full correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap is over all shifts
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points

    # Start from initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a basic initial construction
        h = np.ones(n_points) * 0.5

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    # Sampling of lags to start
    k_samples = np.arange(0, 2.0, 0.1)  # Coarse sampling
    k_samples = k_samples[k_samples <= 2.0]

    # Iterative refinement to tighten constraints
    iter_count = 0
    improvement_threshold = 1e-4

    while time.time() - start_time < budget_s * 0.95 and iter_count < 200:
        iter_count += 1
        current_time = time.time()
        
        # Optimize with current set of lags
        try:
            # Construct problem using cvxpy
            h_var = Variable(n_points)
            C = Variable()
            constraints = []
            
            # Sum constraint
            constraints.append(cp_sum(h_var) * dx == 1.0)
            
            # Bound constraints for sampled k values
            for k in k_samples:
                # k needs to be converted to integer offset
                k_int = int(round(k / dx))
                if k_int < 0 or k_int >= n_points:
                    continue
                corr = cp_sum(multiply(h_var, 1 - cp.roll(h_var, k_int)))
                constraints.append(corr * dx <= C)
                
            # Objective
            objective = Minimize(C)
            prob = Problem(objective, constraints)
            
            # Try optimizing
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if C.value < best_c5:
                    best_c5 = C.value
                    best_h = h_var.value.copy()
            else:
                break
                
        except Exception:
            pass

        # Evaluate all k values to identify worst offenders
        all_k = np.arange(0, 2.0 + 1e-6, 0.02)
        overlaps = []
        for k in all_k:
            k_idx = int(round(k / dx))
            if k_idx < 0 or k_idx >= n_points:
                continue
            corr = np.correlate(best_h, 1 - np.roll(best_h, k_idx), mode='valid')
            overlap = np.max(corr) * dx
            overlaps.append((k, overlap))
            
        # Identify top 10 worst
        top_k = sorted(overlaps, key=lambda x: x[1], reverse=True)[:10]
        k_samples = [k for k, _ in top_k]
        
        if time.time() - current_time > 1.0:
            # Early exit if taking too long
            break
            
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h, final_c5, n_points)
