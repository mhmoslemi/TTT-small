import numpy as np
import time
from scipy import correlate

def evaluate_erdos_solution(h):
    """Compute the C5 bound for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value gives C5 (scaled by dx)
    c5 = np.max(corr) * dx
    return c5

def generate_move(h, n_points):
    """Generate a random feasible move by flipping interval length 1 or 3"""
    # Choose interval length (1 or 3)
    length = np.random.choice([1, 3])
    # Choose start index
    start = np.random.randint(0, n_points - length + 1)
    # Generate new h by flipping the selected interval
    h_new = h.copy()
    for i in range(start, start + length):
        h_new[i] = 1 - h_new[i]
    
    # Ensure constraint remains balanced (sum == n_points/2)
    # Adjust one element to balance the change
    delta = np.sum(h_new) - n_points / 2
    if abs(delta) > 1e-8:
        # Find a suitable index to adjust
        for i in range(n_points):
            if h_new[i] > 0 and h_new[i] < 1:
                h_new[i] = h_new[i] - delta
                if 0 <= h_new[i] <= 1:
                    break
                else:
                    h_new[i] = np.clip(h_new[i], 0, 1)
                    break
    return h_new
