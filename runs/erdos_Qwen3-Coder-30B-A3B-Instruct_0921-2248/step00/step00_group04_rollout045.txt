import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lag = 2.0
    lag_step = 0.1
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(initial_h_values)), 
                              initial_h_values)
    except:
        # Create a starting point
        h_init = np.ones(n_points) * 0.5
        
    # Generate lags
    lags = np.arange(-max_lag, max_lag + lag_step, lag_step)
    lags = lags[abs(lags) <= max_lag]  # Ensure within bounds
    
    # Optimization problem
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) * dx == 1,
        C5 >= 0
    ]
    
    # Add overlap constraints for all lags
    for k in lags:
        if abs(k) < 1e-10:
            # For k=0, just integrate h(x)(1-h(x))
            overlap = cp.sum(cp.multiply(h, 1 - h)) * dx
        else:
            # Shift h to the left by k (in discrete steps)
            # Negative k shifts right, positive k shifts left
            shift = int(round(k / dx))
            if shift > 0:
                # h left shifted by shift
                h_shifted = cp.hstack([h[shift:], h[:shift]])
            elif shift < 0:
                # h right shifted by -shift
                h_shifted = cp.hstack([h[-shift:], h[:-shift]])
            else:
                h_shifted = h
            
            overlap = cp.sum(cp.multiply(h, 1 - h_shifted)) * dx
            
        constraints.append(overlap <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers = ['GLPK', 'MOSEK', 'SCIP', 'CPLEX', 'GUROBI']
    solver = None
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solver = s
                break
        except:
            continue
    
    # If none work, fallback to default
    if solver is None:
        try:
            prob.solve(solver='ECOS', verbose=False)
            solver = 'ECOS'
        except:
            pass
    
    # Extract solution
    if prob.status in ['optimal', 'optimal_inaccurate']:
        h_opt = h.value
        c5_bound = C5.value
    else:
        # Fallback to initial solution if optimization failed
        h_opt = h_init.copy()
        # Compute C5 using initial values
        c5_bound = 0.0
        corr = np.correlate(h_opt, 1-h_opt, mode="full")
        c5_bound = np.max(corr) * dx
    
    # Validate constraints
    assert np.all(h_opt >= 0) and np.all(h_opt <= 1)
    assert abs(np.sum(h_opt) * dx - 1.0) < 1e-10
    
    # Return best available solution within time limit
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.9:
        # If we're running low on time, return partial solution
        pass
    
    # Final computation to ensure correct bound
    corr = np.correlate(h_opt, 1-h_opt, mode="full")
    true_c5 = np.max(corr) * dx
    return (h_opt, true_c5, n_points)
