import numpy as np
from scipy import signal
import time

def evaluate_overlap(h, dx):
    # Compute overlap integral using convolution
    # h is the step function
    # dx is the sampling step
    # We need to compute max_k int h(x)(1-h(x+k))dx
    # For a step function, we compute cross-correlation between h and (1-h)
    # The convolution is over the full length to get all shifts
    conv_result = signal.correlate(h, 1 - h, mode='full')
    # Since we're using full mode, the shift indices go from -len(h)+1 to len(h)-1
    # But we only care about valid overlaps, so we take the max of the central part
    # The overlap integral is summing over all time shifts, so we need to scale
    # by dx to get the continuous integral
    # The maximum C5 is the maximum of the convolution result
    return np.max(conv_result) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial solution if available
    # This is the standard import from the pre-imported section
    try:

        n_points = len(h)
    except NameError:
        # If no initial construction, create standard one
        n_points = 45
        h = np.zeros(n_points)
        # Simple binary construction with 1s in the first half
        h[:n_points//2] = 1.0
        # Adjust to meet constraint sum(h) = 2.0 (with dx=2.0/n_points)
        # So sum(h) should be n_points/2 = 22.5
        # We use binary only, so we must make a good choice
        # Also, leave a small degree of freedom by allowing one entry to float
        h = np.zeros(n_points)
        target_sum = n_points / 2.0
        # Set the first floor(target_sum) to 1
        h[:int(target_sum)] = 1.0
        if target_sum > int(target_sum):
            # Make one floating point value
            h[int(target_sum)] = target_sum - int(target_sum)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Allow for 500 iterations maximum for good coverage
    max_iter = 500
    iter = 0
    
    # PerturbATIVE search around current best
    # For a few swaps, see improvement
    num_swaps = 3
    num_trials = 20
    threshold = 0.3807  # Target bound
    
    while time.time() - start_time < budget_s - 0.5 and iter < max_iter:
        for _ in range(num_trials):
            # Make a copy of the current best to perturb
            h_trial = best_h.copy()
            # Try random swaps
            indices = np.random.choice(n_points, size=num_swaps, replace=False)
            # Use the first elements as 1s, rest as 0s to maintain sum
            # To keep sum constraint, we'll adjust pairs
            # Simply swap some values so that the sum constraint is preserved
            # and draw from range [0,1] to allow float values
            for i, idx in enumerate(indices):
                # Set to opposite value
                if h_trial[idx] == 1:
                    h_trial[idx] = 0
                else:
                    h_trial[idx] = 1
            
            h_trial = np.clip(h_trial, 0, 1)
            # Our total sum should still be 1 anyway
            
            # Check if sum constraint is respected
            actual_sum = np.sum(h_trial) * dx
            target_sum = 1.0
            diff = np.abs(actual_sum - target_sum)
            
            # Correct by adjusting a single element to restore sum
            if diff > 1e-6:
                adjust_idx = np.random.randint(0, n_points)
                # And adjust first to maintain ratio
                amount = target_sum - actual_sum
                h_trial[adjust_idx] = np.clip(h_trial[adjust_idx] + amount, 0, 1)
            
            # Evaluate
            c5 = evaluate_overlap(h_trial, dx)
            
            # Check if we improved
            if c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5
                if best_c5 < threshold:
                    break
                
        iter += 1
        if best_c5 < threshold:
            break
    
    # Final validation
    c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), float(c5), n_points)
