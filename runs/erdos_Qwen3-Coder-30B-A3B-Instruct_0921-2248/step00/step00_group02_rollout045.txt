import numpy as np
from scipy import signal
import time
import random

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum of these values scaled by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def is_feasible(h, n_points):
    """Check if h satisfies constraints."""
    dx = 2.0 / n_points
    total = np.sum(h) * dx
    # Check sum constraint (should be exactly 1)
    if abs(total - 1.0) > 1e-6:
        return False
    # Check bounds
    if np.any(h < 0) or np.any(h > 1):
        return False
    return True
