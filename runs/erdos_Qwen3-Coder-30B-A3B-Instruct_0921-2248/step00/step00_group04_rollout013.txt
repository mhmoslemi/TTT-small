import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # This is the maximum of the cross-correlation of h with (1-h)
    corr = correlate(h, 1-h, mode='full')
    # Scale by dx
    corr *= dx
    # Return the maximum value, which is the peak of the correlation
    return np.max(corr)

def normalize_h(h, dx):
    # Ensure sum(h) * dx = 1, so sum(h) = 1/dx = n_points/2
    target_sum = 1.0 / dx
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust h to have the correct sum
    scale = target_sum / current_sum
    h = h * scale
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Re-adjust to ensure sum constraint
    h = h * (target_sum / np.sum(h))
    return np.clip(h, 0, 1)

def perturb_h(h, epsilon, dx):
    # Create a copy of h
    new_h = h.copy()
    # Select random indices to modify
    n = len(h)
    # Randomly pick some indices to adjust
    num_changes = max(1, n // 4)
    indices = np.random.choice(n, size=num_changes, replace=False)
    # Modify the selected indices
    for i in indices:
        delta = np.random.uniform(-epsilon, epsilon)
        new_h[i] += delta
    # Clamp to [0,1] and re-normalize
    new_h = np.clip(new_h, 0, 1)
    new_h = normalize_h(new_h, dx)
    return new_h

def run(seed=42, budget_s=600, n_points=42):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize randomly with correct sum
        h = np.random.rand(n_points)
        h = normalize_h(h, 2.0/n_points)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    epsilon_start = 0.1
    min_T = 1e-6
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        # Adapt epsilon based on current temperature
        epsilon = epsilon_start * T
        new_h = perturb_h(h, epsilon, dx)
        new_c5 = evaluate(new_h, dx)
        
        # Accept or reject based on Metropolis criterion
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = new_h
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        
        # Cool down temperature
        T = max(min_T, T * cooling_rate)
        
        # Occasionally check for early termination
        if iter_count % 1000 == 0 and time.time() - start_time > 0.9 * budget_s:
            break

    # Final evaluation of best solution
    c5_bound = evaluate(best_h, dx)
    return (list(best_h), c5_bound, n_points)
