import numpy as np
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Create symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = h / np.sum(h) * target_sum
    
    # Precompute FFT of h and (1-h)
    H = fft(h)
    GH = 1.0 - h
    GH = fft(GH)
    GH_conj = np.conj(GH)
    
    # Frequency domain computation
    # Adjusting to include the case with 'full' mode in correlation properly
    overlap_freq = H * GH_conj
    # Transform back to get overlap in time domain
    overlap_time = ifft(overlap_freq).real
    # max is taken over all valid shifts
    max_overlap = np.max(overlap_time)
    C5 = max_overlap * dx
    
    best_h = h.copy()
    best_C5 = C5
    
    # Optimizer parameters
    max_iter = 10000
    alpha = 0.01
    patience = 50
    no_improve_count = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap samples in time domain
        overlap_time = ifft(overlap_freq).real
        # Max overlap and its index
        max_idx = np.argmax(overlap_time)
        if max_idx >= len(overlap_time) // 2:
            max_idx -= len(overlap_time)
        max_overlap = overlap_time[max_idx]
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Update h based on gradient: descent on overlap
        # Relationship of overlap derivative to h 
        # Using fft to compute shift-based gradients
        g = np.zeros_like(h)
        if max_idx >= 0:
            g[max_idx:] = h[:-max_idx]  
            g[:max_idx] = h[-max_idx:]  
            dh = g - h 
        else:
            shift_idx = -max_idx
            g[:-shift_idx] = h[shift_idx:]  
            g[-shift_idx:] = h[:shift_idx]
            dh = g - h
            
        # Modify h in direction of descent
        # Assuming gradient magnitude is approximated by values of many points
        # Delta in h for each point: simple approach
        # But here, use standard descent but also ensure np.clip and sum constraint
        h = h - alpha * dh
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Correct sum constraint: h^T * dx = 1; i.e. sum(h) = n_points/2
        current_sum = np.sum(h)
        delta_sum = target_sum - current_sum
        h += delta_sum / n_points
        
        # Ensure constraint
        h = np.clip(h, 0, 1)
        
        H = fft(h)
        GH = 1.0 - h
        GH = fft(GH)
        GH_conj = np.conj(GH)
        
        # Update overlap
        overlap_freq = H * GH_conj
        overlap_time = ifft(overlap_freq).real
        C5 = np.max(overlap_time) * dx
        
        if iter_num % 50 == 0:
            pass  # Periodic check during optimization
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        else:
            no_improve_count += 1
            if no_improve_count > patience:
                # Reduce learning rate or exit early if no helpful progress
                alpha *= 0.9
                no_improve_count = 0
                
    # Final check and redistribution if needed
    if np.sum(h) != target_sum:
        h *= target_sum / np.sum(h)
        
    # Compute final C5
    h = np.clip(h, 0, 1)
    H = fft(h)
    GH = fft(1.0 - h)
    GH_conj = np.conj(GH)
    overlap_freq = H * GH_conj
    overlap_time = ifft(overlap_freq).real
    C5 = np.max(overlap_time) * dx
    
    if C5 < best_C5:
        best_C5 = C5
        best_h = h.copy()
        
    return (best_h.tolist(), float(best_C5), int(n_points))
