import numpy as np
import scipy.signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 42
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h_vec):
        # Compute the maximum overlap integral
        # C5 = max_k ∫ h(x)(1 - h(x+k)) dx
        # Using cross-correlation over full mode
        h_vec = np.array(h_vec)
        corr = scipy.signal.correlate(h_vec, 1 - h_vec, mode='full')
        # The result is indexed such that the center is at len(h)-1
        # Total length = 2*len(h) - 1
        # The maximum overlap is at indices from len(h)-1 to len(h)-1 + (len(h)-1)
        # So we want the maximum over indices len(h)-1 to 2*len(h)-2
        max_corr = np.max(corr[len(h_vec)-1:])
        return max_corr * (2.0 / n_points)
    
    def perturb_h(h_vec, temp):
        h_new = h_vec.copy()
        # Perturb a random subset of points
        n_perturb = max(1, int(0.1 * len(h_vec)))
        indices = random.sample(range(len(h_vec)), n_perturb)
        for i in indices:
            # Random perturbation within bounds
            dp = random.uniform(-0.1, 0.1) * temp
            h_new[i] += dp
            h_new[i] = max(0, min(1, h_new[i]))
        
        # Adjust to preserve sum constraint
        old_sum = np.sum(h_vec)
        new_sum = np.sum(h_new)
        if new_sum > 0:
            scale = old_sum / new_sum
            h_new = h_new * scale
            
        # Ensure it's still within bounds
        h_new = np.clip(h_new, 0, 1)
        # Adjust again to keep sum exactly correct
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    start_time = time.time()
    current_h = h
    current_C5 = compute_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5
    temp = 1.0
    cooling_rate = 0.999
    
    i = 0
    while time.time() - start_time < budget_s:
        i += 1
        # Perturb solution
        new_h = perturb_h(current_h, temp)
        new_C5 = compute_C5(new_h)
        
        # Accept or reject
        if new_C5 < current_C5 or random.random() < np.exp(-(new_C5 - current_C5) / temp):
            current_h = new_h
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = current_C5
        
        # Cool down
        temp *= cooling_rate
        
        if i % 1000 == 0:
            # Check time budget occasionally
            if time.time() - start_time >= budget_s * 0.98:
                break
    
    return (list(best_h), best_C5, n_points)
