import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = 2 * i / mid
            else:
                h[i] = 2 * (n_points - i) / mid
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find maximum correlation shift
        max_idx = np.argmax(corr)
        shift = max_idx - (len(h) - 1)
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + shift >= 0 and i + shift < n_points:
                grad[i] = 1 - h[i + shift]
            if i - shift >= 0 and i - shift < n_points:
                grad[i] -= h[i - shift]
        
        # Gradient step
        h_new = h - alpha * grad
        
        # Project onto feasible set
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Adjust to satisfy sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        h_new = h_new * target_sum / current_sum
        
        h = h_new
    
    # Final evaluation
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
