import numpy as np
from scipy.optimize import minimize
from scipy.linalg import block_diag
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Construct basic tent-shaped initial solution
        n_points = 70
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(n_points):
            x = i * dx
            h[i] = 1.0 - abs(x - 1.0)
        h = np.clip(h, 0, 1)
        # Normalize to correct integral
        h *= (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    dx = 2.0 / n_points
    
    def compute_C5(h):
        # Correlation computation
        h_ext = np.concatenate([h, np.zeros(n_points)])
        corr = np.correlate(h_ext, 1 - h, mode='full')[:2*n_points-1]
        max_corr = np.max(corr)
        return max_corr * dx
    
    def barrier(h, sigma=1e3, eta=1e-8):
        """Barrier function for constraints"""
        h = np.clip(h, eta, 1 - eta)
        return -sigma * (np.sum(np.log(h)) + np.sum(np.log(1 - h)) + 
                         np.log(np.sum(h) - n_points/2 + eta) + 
                         np.log(n_points/2 - np.sum(h) + eta))
    
    def barrier_grad(h, sigma=1e3, eta=1e-8):
        h = np.clip(h, eta, 1 - eta)
        grad = np.zeros_like(h)
        grad = sigma * (1/h - 1/(1-h) + 
                        (1/(np.sum(h) - n_points/2 + eta) - 
                         1/(n_points/2 - np.sum(h) + eta)) * np.ones_like(h))
        return grad
    
    def optimize_step(h, step_size=0.1, max_iter=50):
        current_C5 = compute_C5(h)
        best_C5 = current_C5
        best_h = h.copy()
        
        for _ in range(max_iter):
            # Compute all correlations efficiently
            h_ext = np.concatenate([h, np.zeros(n_points)])
            corr = np.correlate(h_ext, 1 - h, mode='full')[:2*n_points-1]
            k_indices = np.argsort(-corr)[:max(1, n_points//10)]  # Top shifts
            
            # Gradient approximation using top correlations
            grad = np.zeros_like(h)
            for k in k_indices:
                # Compute gradient for shift k
                # h[i] affects term: h[i] * (1 - h[i + k]) for i < n_points - k
                # and (1 - h[i]) * h[i - k] for i >= k
                if k < n_points:
                    for i in range(n_points - k):
                        if i + k < n_points:
                            grad[i] += (1 - h[i + k]) - h[i]  # gradient w.r.t. h[i]
                else:
                    for i in range(n_points):
                        if i + k < 2*n_points:
                            grad[i] += -h[i + k - n_points]  # gradient w.r.t. h[i]
            
            # Add barrier gradient
            grad += barrier_grad(h)
            
            # Simple gradient descent step
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Project to integral constraint
            h_new_sum = np.sum(h_new)
            h_new = h_new * (n_points/2) / h_new_sum
            
            # Check improvement
            new_C5 = compute_C5(h_new)
            if new_C5 < best_C5:
                h = h_new.copy()
                best_C5 = new_C5
                best_h = h.copy()
            
            if time.time() - start_time > budget_s * 0.95:
                break
                
        return best_h, best_C5
    
    # Main optimization loop
    last_improvement = 0
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    
    while time.time() - start_time < budget_s * 0.95:
        h, C5 = optimize_step(h, step_size=0.1, max_iter=30)
        if C5 < best_C5:
            best_h = h.copy()
            best_C5 = C5
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > 50:
                break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * (n_points/2) / final_sum
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
