import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a base construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple alternating pattern
        n_points = 70
        h = np.ones(n_points)
        h[::2] = 0.0
    
    # Normalize to sum = n_points / 2
    dx = 2.0 / len(h)
    h = h * (len(h) / 2) / np.sum(h)
    
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient-based approach
    max_iter = 1000
    tolerance = 1e-6
    step_size = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        correlations = np.correlate(h, 1 - h, mode="full")
        # We're looking at the first n_points-1 elements which are the "shifts"
        overlaps = correlations[:n_points] * dx
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # If overlap is too small, stop
        if max_overlap < 1e-4:
            break
            
        # Compute the gradient of overlap w.r.t h
        # For overlap_k = sum(h[i] * (1 - h[(i+k)%n])) 
        # gradient w.r.t h[i] is (1 - h[(i+k)%n]) 
        # Only consider elements where 1 - h[(i+k)%n] > 0
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h[j]
        
        # Update h
        h = h - step_size * grad
        
        # Project h back to [0,1] box constraints
        h = np.clip(h, 0, 1)
        
        # Renormalize
        h = h * (len(h) / 2) / np.sum(h)
        
        # Check if we should stop early
        if iteration > 0 and abs(max_overlap - overlaps[0]) < tolerance:
            break
            
    return (best_h.tolist(), best_c5, len(best_h))
