import numpy as np
from scipy import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    """Compute C5 bound from h values"""
    conv = correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create symmetric tent function
        n_points = 200
        h = np.zeros(n_points)
        for i in range(n_points):
            j = i - n_points // 2
            h[i] = max(0, 1 - 2 * abs(j) / n_points)
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    # Iterative QP optimization
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find current worst overlaps
        conv = correlate(h, 1 - h, mode='full')
        max_val = np.max(conv)
        max_indices = np.where(conv == max_val)[0]
        
        # Collect all shifts contributing to max overlap
        shifts = []
        for idx in max_indices:
            if idx >= n_points - 1:  # In full convolution, these are shifted indices
                shifts.append(idx - n_points + 1)
            else:
                shifts.append(idx - n_points + 1)
        
        # Create QP problem for worst-case shifts
        # Minimize sum over all influenced shifts
        h_var = cp.Variable(n_points)
        constraints = []
        
        # Box constraints
        constraints += [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Sum constraint
        constraints += [cp.sum(h_var) == n_points / 2]
        
        # Objective: minimize the maximum overlap term
        # We'll minimize the sum of overlaps for all active shifts
        objective_val = 0
        for k in shifts:
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    objective_val += h_var[i] * (1 - h_var[i + k])
                
        objective = cp.Minimize(objective_val)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = np.array(h_var.value).flatten()
                # Ensure feasibility
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except Exception:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
