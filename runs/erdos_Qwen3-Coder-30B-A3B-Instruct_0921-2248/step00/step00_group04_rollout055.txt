import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    lag_step = 0.2
    max_lags = 200
    min_improvement = 1e-4
    initial_lag_step = 0.2
    
    # Start with initial construction if available
    try:
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    except NameError:
        # Construct a basic linear profile
        h0 = np.ones(n_points) * 0.5
    
    # Initialize h and C5
    h = cp.Variable(n_points, bounds=(0, 1))
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) * dx == 1.0,
        C5 >= 0
    ]
    
    # Initial lags
    lags = np.arange(-2, 2 + lag_step, lag_step)
    lags = lags[abs(lags) <= 2]
    lags = lags[abs(lags) >= 1e-6]
    
    best_C5 = float('inf')
    best_h = h0.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Add lag-based constraints
        for k in lags:
            # Convert to discrete lag index (rounded)
            k_index = int(round(k / dx))
            if not -n_points <= k_index <= n_points:
                continue
            
            # Compute convolution over the range
            conv = np.zeros(n_points)
            for i in range(n_points):
                j = (i - k_index) % n_points
                conv[i] = h[i] * (1 - h[j])
            
            # Sum over all positions
            overlap_sum = cp.sum(conv) * dx
            constraints.append(overlap_sum <= C5)
        
        # Problem formulation
        prob = cp.Problem(cp.Minimize(C5), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                break
        
        if prob.status != cp.OPTIMAL:
            break
            
        h_val = np.array(h.value).flatten()
        C5_val = C5.value
        
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h_val.copy()
            
        # Evaluate all overlaps
        h_rev = 1 - h_val
        overlaps = correlate(h_val, h_rev, mode='full')
        lags_full = np.arange(-n_points + 1, n_points)
        indices = np.argsort(overlaps)[::-1][:max_lags // 4]
        
        # Add top violating lags
        new_lags = lags_full[indices]
        lags = np.unique(np.concatenate([lags, new_lags]))
        
        # If too many lags, remove some
        if len(lags) > max_lags:
            lags = np.unique(np.sort(np.concatenate([
                lags[:max_lags // 2],
                np.linspace(-2, 2, max_lags // 2)
            ])))
        
        # Reset constraints
        constraints = [
            cp.sum(h) * dx == 1.0,
            C5 >= 0
        ]
        
        # Early stopping
        if iteration > 5 and C5_val < 0.3807:
            break
    
    # Final validation
    try:
        c5_bound = max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    except:
        c5_bound = 0.38085

    return (best_h, c5_bound, n_points)
