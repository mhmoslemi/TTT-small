import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate C5 for a given h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def feasible_h(h):
    """Ensure h is in [0,1] and sums to 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (len(h) / 2) / h_sum
    return h

def mutate_h(h, rate=0.01):
    """Mutate h with given rate"""
    h = h.copy()
    n_points = len(h)
    n_mutations = int(rate * n_points)
    indices = np.random.choice(n_points, n_mutations, replace=False)
    for i in indices:
        h[i] = np.random.random()
    return feasible_h(h)

def crossover_h(h1, h2):
    """Uniform crossover between two h vectors"""
    h = h1.copy()
    mask = np.random.random(len(h)) > 0.5
    h[mask] = h2[mask]
    return feasible_h(h)

def tournament_selection(population, fitnesses, k=5):
    """Tournament selection"""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    population_size = 100

    # Fill remaining population with random feasible h
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = feasible_h(h)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    last_improvement = 0
    
    # Evolution loop
    for generation in range(max_generations):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h) for h in population]
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            last_improvement = generation
            
        # Early termination if stagnant
        if generation - last_improvement > 500:
            break
            
        # Selection and reproduction
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.random() < crossover_rate:
                child = crossover_h(parent1, parent2)
            else:
                child = parent1.copy()
                
            child = mutate_h(child, mutation_rate)
            new_population.append(child)
            
        population = new_population
        
    if best_h is None:
