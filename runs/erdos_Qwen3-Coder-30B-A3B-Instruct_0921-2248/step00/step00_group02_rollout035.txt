import numpy as np
import time
from scipy import signal

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for given h."""
    dx = 2.0 / len(h)
    # Note: correlate with 1-h computes the integral
    conv = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to maintain sum(h) = n_points / 2 and bounds [0,1]."""
    h = np.array(h, dtype=float)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    # Adjust by adding a constant to make sum correct
    h += (target_sum - current_sum) / n_points
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    return h

def perturb_h(h, n_points, method="flip"):
    """Create a perturbed version of h."""
    h_new = np.array(h, dtype=float)
    if method == "flip":
        # Flip two random elements: one 0 to 1 and one 1 to 0
        zeros = np.where(h_new == 0)[0]
        ones = np.where(h_new == 1)[0]
        if len(zeros) > 0 and len(ones) > 0:
            i0 = np.random.choice(zeros)
            i1 = np.random.choice(ones)
            h_new[i0] = 1
            h_new[i1] = 0
    elif method == "adjust":
        # Adjust one random element
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_new[i] += delta
    elif method == "noise":
        # Add small noise to elements, then project
        noise = np.random.normal(0, 0.02, n_points)
        h_new += noise
    # Ensure constraints
    h_new = project_h(h_new, n_points)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():

    else:
        # Start with a simple binary construction
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1
    h_current = project_h(h_current, n_points)
    best_h = h_current.copy()
    best_C5 = evaluate_overlap(h_current)
    
    start_time = time.time()
    iterations = 0
    
    while time.time() - start_time < budget_s:
        iterations += 1
        method = np.random.choice(["flip", "adjust", "noise"], p=[0.4, 0.3, 0.3])
        h_candidate = perturb_h(h_current, n_points, method=method)
        C5_candidate = evaluate_overlap(h_candidate)
        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = h_candidate.copy()
        h_current = h_candidate.copy()
        # Occasionally restart from best found
        if np.random.rand() < 0.01:
            h_current = best_h.copy()
    
    # Final validation
    assert np.isfinite(best_C5), f"Invalid C5 value: {best_C5}"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "h not in [0,1]"
    assert np.abs(np.sum(best_h) - n_points/2) < 1e-10, "Sum constraint violated"
    return (list(best_h), best_C5, n_points)
