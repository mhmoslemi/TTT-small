import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * target_sum

    # Store best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.001
    decay_factor = 0.99
    patience = 100
    patience_counter = 0
    last_c5 = best_c5
    start_time = time.time()
    
    # Main optimization loop
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation to find worst shift
        correlation = signal.correlate(h, 1 - h, mode='full')
        # We want to maximize the negative of h[i] * (1 - h[i+k]) which is 
        # same as minimizing (1-h[i]) * h[i+k]. This will be maximized by 
        # looking at the lowest correlation values for s < n_points
        # Since we are computing overlap = h(x)(1-h(x+k)) and we're minimizing it
        # we compute the IFFT then find max of real positive values
        
        # Actually compute based on full correlation
        # The correlation gives the overlap for each shift k (starting at 0)
        # When convolution is done, the full mode gives length n_points*2 - 1,
        # and the indices 0 to 2*n_points-2 map to shifts -n_points+1, ..., n_points-1
        # We want to look at shift k = 0 to n_points-1, which is indices from n_points-1 to 2*n_points-2 in the full result.
        k_shift = np.arange(n_points)
        overlaps = np.array([correlation[n_points-1 + k] for k in k_shift])
        k_max = np.argmin(overlaps)  # Look for worst shift (minimum cross-correlation)
        
        # Calculate gradient_signs: for the specific k_max shift
        # we want to minimize overlap_k_max = sum h[i] * (1 - h[(i+k_max)%n_points])
        # Then d/dh[i] overlap_k_max = 1 - h[(i+k_max)%n_points] - h[i]
        gradient_signs = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient_signs[i] = 1 - h[j] - h[i]
        
        # Update h
        h = h - alpha * gradient_signs
        h = np.clip(h, 0, 1)
        
        # Normalize sum
        h = h / np.sum(h) * target_sum
        
        # Check progress
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptively decrease alpha
        if iteration % 50 == 0 and iteration > 0:
            alpha *= decay_factor

    # Final correction to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Calculate the overlap integral for all shifts k
    n = len(h)
    # Use the full cross-correlation, then extract the relevant part
    cross_corr = np.correlate(h, 1 - h, mode='full')
    # The overlap corresponds to the central slice
    # For shift k=0 to n-1, they correspond to indices n-1 to 2*n-2 in correlation
    overlap_values = [cross_corr[n-1 + k] for k in range(n)]
    # Return the maximum overlap, scaled by dx
    return max(overlap_values) * dx
