import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to target sum if needed
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.001
    tolerance = 1e-6
    last_c5 = best_c5
    alpha_decay = 0.99
    decay_interval = 50
    
    start_time = time.time()
    
    # Iteration loop
    iteration = 0
    while time.time() - start_time < budget_s - 0.01:
        # Compute all overlaps
        overlaps = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant overlaps
        overlap_vals = overlaps[len(h)-1:]
        k_max = np.argmax(overlap_vals)
        max_overlap = overlap_vals[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # If converged, break
        if abs(last_c5 - max_overlap) < tolerance:
            break
        last_c5 = max_overlap
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = 1 - h[j] - h[i]
            
        # Update
        h = h - alpha * gradient
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
            
        # Adjust alpha
        iteration += 1
        if iteration % decay_interval == 0:
            alpha *= alpha_decay
            
        # Occasionally check for early termination
        if iteration % 100 == 0 and best_c5 < 0.3808:
            break
            
    # Final check of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    # Compute cross-correlation to get all overlaps
    corr = np.correlate(h, 1 - h, mode='full')
    # Get the valid overlap values
    overlaps = corr[len(h)-1:]
    return np.max(overlaps) * dx
