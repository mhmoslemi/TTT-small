import numpy as np
import time
from scipy import signal

def evaluate_overlap(h, dx):
    # Compute the overlap integral as specified
    conv = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def validate_solution(h, n_points):
    # Check that h is valid: in [0,1] and sum = n_points / 2
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2) < 1e-10)

def perturb_h(h, n_points):
    # Create a new candidate by perturbing h
    h_new = h.copy()
    # Try a simple local perturbation
    # 1. Choose two random indices to swap 0 and 1 (if possible)
    # Or, adjust a few points
    num_changes = 3
    indices = np.random.choice(n_points, num_changes, replace=False)
    
    # Try to make small changes preserving sum constraint
    delta = 0.05
    for i in indices:
        if h_new[i] >= delta:
            h_new[i] -= delta
        else:
            h_new[i] = 0.0
    
    # Adjust remaining points to preserve sum
    total_diff = n_points / 2 - np.sum(h_new)
    if abs(total_diff) > 1e-10:
        # Distribute correction
        for i in range(min(num_changes, n_points)):
            if total_diff > 0:
                add = min(delta, total_diff)
                if h_new[i] + add <= 1:
                    h_new[i] += add
                    total_diff -= add
                else:
                    add = 1 - h_new[i]
                    h_new[i] = 1
                    total_diff -= add
            elif total_diff < 0:
                sub = min(delta, -total_diff)
                if h_new[i] - sub >= 0:
                    h_new[i] -= sub
                    total_diff += sub
                else:
                    sub = h_new[i]
                    h_new[i] = 0
                    total_diff += sub
            
            if abs(total_diff) < 1e-10:
                break
    return h_new

    # Ensure valid initial state
    assert validate_solution(h, n_points), "Initial solution invalid"
    
    dx = 2.0 / n_points
    best_c5 = evaluate_overlap(h, dx)
