def evaluate_erdos_solution(h, dx):
    """Returns C5 = max_k int h(x)(1 - h(x+k)) dx for step function h."""
    # API usage assumes pre-constructed h via uniform grid [0, 2], sum(h)*dx = 1
    # Internally, this computes auto-correlation-like structure

    c5 = np.max(corr) * dx
    return c5
