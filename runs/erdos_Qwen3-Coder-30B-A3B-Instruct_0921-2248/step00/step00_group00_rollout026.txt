import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Scale by dx and find maximum
    c5 = np.max(corr) * dx
    return c5

def is_feasible(h):
    """Check if h satisfies constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2."""
    n = len(h)
    dx = 2.0 / n
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1.0) < 1e-8)

def mutate_h(h, mutation_rate=0.01):
    """Mutate h by slightly changing values while maintaining feasibility."""
    h_new = h.copy()
    n = len(h)
    # Mutate a fraction of the points
    n_mutate = max(1, int(n * mutation_rate))
    indices = np.random.choice(n, size=n_mutate, replace=False)
    for i in indices:
        # Perturb the value with small random change
        delta = np.random.uniform(-0.1, 0.1)
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
    # Adjust to maintain sum constraint
    current_sum = np.sum(h_new)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) > 1e-6:
        # Simple fix: scale values
        if current_sum != 0:
            h_new = h_new * (target_sum / current_sum)
        # Ensure within bounds
        h_new = np.clip(h_new, 0, 1)
    return h_new

def crossover_h(h1, h2):
    """Perform crossover between h1 and h2."""
    n = len(h1)
    crossover_point = np.random.randint(1, n)
    h_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    # Ensure sum constraint
    current_sum = np.sum(h_new)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) > 1e-6:
        if current_sum != 0:
            h_new = h_new * (target_sum / current_sum)
        h_new = np.clip(h_new, 0, 1)
    return h_new

def select_parents(population, fitnesses, tournament_size=5):
    """Tournament selection."""
    idxs = np.random.choice(len(population), size=tournament_size, replace=False)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]
