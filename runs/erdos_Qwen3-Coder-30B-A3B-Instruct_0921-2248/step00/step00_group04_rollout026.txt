import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation: h(x) * (1 - h(x+k))
    corr = correlate(h, 1 - h, mode="full")
    # The maximum is at the center of the correlation result
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h):
    """Project h to satisfy constraints: sum(h) = n/2 and 0 <= h[i] <= 1."""
    n = len(h)
    target_sum = n / 2.0
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust values to meet sum constraint
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        # Distribute diff uniformly among non-1 entries
        non_one = np.where(h < 1)[0]
        if len(non_one) > 0:
            d = diff / len(non_one)
            h[non_one] = np.minimum(h[non_one] + d, 1.0)
    else:
        # Need to decrease values
        # Distribute diff uniformly among non-0 entries
        non_zero = np.where(h > 0)[0]
        if len(non_zero) > 0:
            d = -diff / len(non_zero)
            h[non_zero] = np.maximum(h[non_zero] - d, 0.0)
    return h

def perturb_h(h, epsilon):
    """Apply small random perturbations to h, then project to constraints."""
    n = len(h)
    h_new = h.copy()
    # Choose random indices to modify
    indices = np.random.choice(n, size=max(1, int(n * 0.2)), replace=False)
    for i in indices:
        # Modify h[i] by small amount
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[i] += delta
    # Project to satisfy constraints
    h_new = project_h(h_new)
    return h_new
