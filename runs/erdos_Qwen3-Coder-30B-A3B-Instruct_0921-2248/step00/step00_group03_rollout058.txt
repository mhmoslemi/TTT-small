import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    # Compute overlap integral using full cross-correlation
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value corresponds to C5 bound
    return np.max(corr) / len(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    initial_h = getattr(__builtins__, 'initial_h_values', None)
    if initial_h is not None:
        h_init = np.array(initial_h)
        n_points = len(h_init)
    else:
        # fallback: create uniform distribution
        n_points = 70
        h_init = np.full(n_points, 0.5)

    # Set solver params
    solver_opts = {
        'max_iters': 200,
        'time_limit': budget_s - 1,
    }
    
    # Try to solve with smaller resolution
    for n in [70]:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n
        h = cp.Variable(n, name="h")
        C = cp.Variable(name="C")
 
        # Build constraints for all valid shifts
        constraints = []
        # Variable bounds
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        # Sum constraint
        constraints.append(cp.sum(h) == n / 2.0)

        # Overlap integral constraints for all shifts
        for shift in range(-n+1, n):
            if time.time() - start_time > budget_s - 5:
                break
            
            # correct indices
            idxs = np.arange(n)
            shifted = (idxs + shift) % n
            # overlap for this shift is h[i] * (1 - h[i+shift])
            # form the overlap sum with modular indexing
            if shift >= 0:
                product = h[shifts] * (1 - h[:n-shift])
            else:
                product = h[:n+shift] * (1 - h[-shift:])
            
            constraints.append(cp.sum(product) <= C)
            
        # Objective
        objective = cp.Minimize(C)
        
        # Problem formulation
        prob = cp.Problem(objective, constraints)
        
        # Try solving with ECOS (fast) and CVXOPT as fallback
        try:
            prob.solve(solver=cp.ECOS, **solver_opts)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT, **solver_opts)
            except Exception:
                pass # Will return previous best if any

        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_sol = h.value
            c5_val = C.value
            
            if h_sol is None or np.isnan(c5_val):
                continue
                
            # validate solution
            if (np.all(h_sol >= 0) and np.all(h_sol <= 1) and 
                abs(np.sum(h_sol) - n/2.0) < 1e-6):
                # compute actual C5 to check
                actual_c5 = evaluate_erdos_solution(h_sol)
                if actual_c5 < c5_val:
                    c5_val = actual_c5
                h_final = h_sol
                return (h_final, c5_val, n)
        else:
            # print("Solver status:", prob.status)
            pass
            
        if time.time() - start_time > budget_s - 5:
            break
        
    # If we made it here, try more optimized search
    # Create a refined optimization from initial solution if it's valid
    h_final = h_init
    if initial_h is not None and n_points == len(h_init):
        # initial solution is available, try gradient descent
        h_current = h_init.copy()
        
        def compute_c5(h_v):
            return evaluate_erdos_solution(h_v)
            
        def compute_grad(h_v):
            # Simple finite difference for gradient estimate
            eps = 1e-5
            grad = np.zeros_like(h_v)
            base_c5 = compute_c5(h_v)
            for i in range(len(h_v)):
                h_plus = h_v.copy()
                h_plus[i] += eps
                grad[i] = (compute_c5(h_plus) - base_c5) / eps
            return grad

        # Simple gradient ascent descent
        step_size = 1e-2 / (n_points ** 0.5)
        for iter in range(5000):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = compute_grad(h_current)
            # Projected gradient descent with masking
            h_current += step_size * grad
            h_current = np.clip(h_current, 0, 1)
            # Maintain sum = 1
            h_current = h_current * (n_points / 2.0) / np.sum(h_current)
            
        h_final = h_current
        c5_final = compute_c5(h_final)
        return (h_final, c5_final, n_points)
    
    return (h_final, evaluate_erdos_solution(h_final), n_points)
