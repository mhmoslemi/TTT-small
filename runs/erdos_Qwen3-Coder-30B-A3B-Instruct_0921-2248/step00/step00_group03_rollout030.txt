import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        # Fallback: start with a simple binary-like pattern

        # Place 1s evenly to satisfy sum(h)=n_points/2
        ones_positions = np.arange(0, n_points, 2)
        h[ones_positions] = 1.0
        h = h * (n_points / (2 * np.sum(h)))  # Normalize
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_C5(h_vec):
        # Compute overlap integral for all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Shift to align properly
        corr = corr[len(h_vec)-1:]
        C5 = np.max(corr) * dx
        return C5
    
    def project_h(h_vec):
        # Ensure h is within [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Normalize to sum = target_sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (target_sum / current_sum)
        return h_vec
    
    best_C5 = evaluate_C5(h)
    best_h = h.copy()
    
    # Iteration loop
    max_iter = 1200
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Evaluate current overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        # Only consider valid overlaps (shifts from 0 to n_points-1)
        corr = corr[len(h)-1:]
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient of max_overlap w.r.t. h (simplified approach)
        # For shift k_max, we want to reduce h[i] * (1 - h[(i+k_max) % n]) 
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h[j]
        
        # Apply gradient descent step (projected)
        step_size = 0.01  # Dynamic step size could improve convergence
        h_new = h - step_size * grad
        h_new = project_h(h_new)
        
        new_C5 = evaluate_C5(h_new)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
        
        h = h_new
        iteration += 1
    
    # Ensure final constraints
    h = project_h(h)
    C5_bound = evaluate_C5(h)
    
    # Return best known solution
    return (list(best_h), float(C5_bound), int(n_points))
