from scipy.optimize import minimize
import time

        # Create a symmetric tent function as initial guess
        n_points = 70
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 * (1.0 - abs(i - (n_points-1)/2) / ((n_points-1)/2))
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Function to evaluate C5 bound
    def evaluate_C5(h_vals):
        # Compute all cross-correlations
        h_ext = np.concatenate([h_vals, np.zeros(n_points)])
        correlations = np.correlate(h_ext, 1 - h_vals, mode='full')
        # Only consider the relevant part
        corr_vals = correlations[n_points-1:2*n_points-1]
        return np.max(corr_vals) * dx
    
    # Constraint function: sum(h) - target_sum = 0
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    # Bounds for each variable: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization options
    options = {'maxiter': 1000, 'disp': False}
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial performance
    best_C5 = evaluate_C5(h)
    best_h = h.copy()
    
    # Optimization - trust region method
    def objective(h_vals):
        return evaluate_C5(h_vals)
    
    # Algorithm: trust-region method with bounds and constraints
    # We'll try to improve using L-BFGS-B
    try:
        # Use scipy to find a better local solution
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options=options)
        if res.success:
            h_local = res.x
            C5_local = evaluate_C5(h_local)
            if C5_local < best_C5:
                best_C5 = C5_local
                best_h = h_local
    except:
        pass
    
    # Done with optimization; if no improvement was made by local search, return initial guess
    return (best_h, best_C5, n_points)
