    h = h * (1.0 / np.sum(h) * 2)

    def c5_objective(h_values):
        return evaluate_erdos_solution(h_values, dx)

    # Use scipy.optimize.minimize with L-BFGS-B for bounds
    def optimize_step(h_current, max_iter=50):
        def objective(h_vec):
            h_new = h_vec.reshape(-1)
            return evaluate_erdos_solution(h_new, dx)
            
        def jac(h_vec):
            # Approximate gradient
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2*eps)
            return grad
            
        # Optimization bounds
        bounds = [(0, 1) for _ in range(len(h_current))]
        # Project to sum constraint
        def project(h_vec):
            h_new = h_vec.copy()
            h_new = h_new / np.sum(h_new) * 2
            return np.clip(h_new, 0, 1)
        
        x0 = h_current.flatten()
        result = optimize.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            jac=jac,
            options={'maxiter': max_iter, 'ftol': 1e-6, 'gtol': 1e-6}
        )

def evaluate_erdos_solution(h_values, dx):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    h = np.array(h_values)
    # Compute all correlations via FFT
    # Compute h and 1-h
    h = np.array(h)
    one_minus_h = 1.0 - h
    
    # Full convolution via fft
    # First we pad the arrays to avoid circular convolution
    n = len(h)
    # Set size for full convolution (n + n - 1 = 2*n - 1)
    fft_len = 2 * n - 1
    
    # Pad to this size
    h_padded = np.pad(h, (0, fft_len - n), 'constant')
    one_minus_h_padded = np.pad(one_minus_h, (0, fft_len - n), 'constant')
    
    # Forward FFTs
    H = fft(h_padded)
    OneMinusH = fft(one_minus_h_padded)
    
    # Convolution result (in frequency domain)
    # Convolution of h(x) * (1-h(x+k)) -> FFT(h)* conjugate(FFT(1-h))
    correlation = H * np.conj(OneMinusH)
    
    # Transform back to time domain
    corr_time = ifft(correlation).real
    
    # Take maximum over shifts, note we added padding so 0 to 2*fft_len-2
    # shift values are from 0 to 2*n-2
    # Correlation values start at shift 0 (which is max shift possible) and go up
    # to shift 2*n-2
    # The shift that corresponds to the correlation between overlapping h(x) and 1-h(x+k)
    c5 = np.max(corr_time[:2*n-1]) * dx
    return c5
