def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h."""

    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Get the maximum value
    max_corr = np.max(corr)
    return max_corr * dx
