import numpy as np
import time
from scipy import correlate

def evaluate_h(h):
    """Evaluate the overlap integral C_5 for given h."""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h and (1-h) with full mode
    corr = correlate(h, 1 - h, mode='full')
    # The maximum correlation value corresponds to C_5 * dx
    max_corr = np.max(corr)
    return max_corr * dx

def is_feasible(h, n_points):
    """Check if h satisfies constraints."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2.0) < 1e-8)

def mutate_h(h, mutation_rate=0.01):
    """Mutation operator: randomly perturb some values."""
    h_mut = h.copy()
    n_points = len(h)
    # Mutate around 1% of points
    n_mutate = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        # Random perturbation respecting bounds
        delta = np.random.normal(0, 0.1)
        h_mut[i] = np.clip(h_mut[i] + delta, 0, 1)
    # Re-normalize if needed
    sum_h = np.sum(h_mut)
    if abs(sum_h - n_points / 2.0) > 1e-8:
        h_mut = h_mut * (n_points / 2.0) / sum_h
    return h_mut

def crossover_h(h1, h2):
    """Single-point crossover."""
    n_points = len(h1)
    if n_points != len(h2):
        raise ValueError("H vectors must have same length")
    point = np.random.randint(1, n_points)
    h_new = np.concatenate([h1[:point], h2[point:]])
    # Renormalize
    sum_h = np.sum(h_new)
    if abs(sum_h - n_points / 2.0) > 1e-8:
        h_new = h_new * (n_points / 2.0) / sum_h
    return h_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

        if abs(np.sum(h_initial) - n_points / 2.0) > 1e-8:

    # Parameters
    pop_size = 100
    max_generations = 10000
    mutation_rate = 0.01
    crossover_rate = 0.7
    elite_size = 1
    patience = 500
    best_c5 = float('inf')

    # Initial population
    population = [h_initial.copy()]

        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2.0) / np.sum(h_rand)
        population.append(h_rand)
    
    # Track best solution
    best_fitnesses = []
    
    # Simple selection pressure
    for gen in range(max_generations):
        # Measure elapsed time
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h) for h in population]
        min_fitness = min(fitnesses)

        new_population = []

        elite_indices = np.argsort(fitnesses)[:elite_size]
