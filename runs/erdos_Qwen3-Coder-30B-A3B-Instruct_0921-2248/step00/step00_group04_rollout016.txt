import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate lag samples
    lags = np.arange(-2.0, 2.01, 0.1)
    
    # Check if initial_h_values is available
    try:
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(initial_h_values)), 
                              initial_h_values)
    except:
        # Fallback: create a simple initial function
        h_init = np.ones(n_points) * 0.5
    
    # Initialize optimization variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Integral constraint
    constraints.append(cp.sum(h) * dx == 1.0)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints for each lag
    for k in lags:
        # Handle shift with wraparound
        k_idx = int(k / dx)
        if k_idx < 0:
            shifted_h = np.concatenate([h[k_idx:], h[:k_idx]])
        elif k_idx > 0:
            shifted_h = np.concatenate([h[k_idx:], h[:k_idx]])
        else:
            shifted_h = h
            
        # Correlation for this lag
        corr = cp.sum(cp.multiply(h, 1 - shifted_h)) * dx
        constraints.append(corr <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.GLPK, verbose=False, time_limit=budget_s - (time.time() - start_time))
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
            h_sol = h.value
            c5_bound = C5.value

            # Fallback to initial solution if optimization failed
            h_sol = h_init
