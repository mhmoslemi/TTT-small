import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # init params
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # initial h: use provided or create symmetric tent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # create symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = 2.0 * i / (mid - 1.0)
        for i in range(mid, n_points):
            h[i] = 2.0 * (n_points - 1 - i) / (mid - 1.0)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
    
    # optimization loop
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # lr
    lr = 0.01
    iter_since_improvement = 0
    max_iter_since_improvement = 50
    
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # compute overlap in frequency domain
        H = fft.fft(h)
        H_conj = np.conj(H)
        GH = H * H_conj
        overlap_freq = fft.ifft(GH).real
        C5_vals = np.maximum(0, overlap_freq[:n_points]) / dx  # max needed
        max_idx = np.argmax(C5_vals)
        max_overlap = C5_vals[max_idx]
        
        # try to reduce that overlap
        # gradient: difference between actual and desired (1) value at index
        grad = np.zeros(n_points)
        grad[max_idx] = 1.0
        
        # convert gradient back to time domain, flip sign
        grad_freq = fft.fft(grad)
        grad_time = fft.ifft(grad_freq * H_conj).real
        
        # update h
        h = h - lr * grad_time
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        
        # evaluation
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # decay rate
        if iter_since_improvement > 20:
            lr *= 0.95
            
        if iter_since_improvement > max_iter_since_improvement:
            break
            
    return (best_h.tolist(), float(best_c5), n_points)

def evaluate_c5(h, dx):
    # reference computation
    h_flip = 1 - h
    conv = np.correlate(h, h_flip, mode='full')
    # take only the valid part for overlaps
    valid_conv = conv[len(h)-1:]
    max_overlap = np.max(valid_conv) * dx
    return max_overlap
