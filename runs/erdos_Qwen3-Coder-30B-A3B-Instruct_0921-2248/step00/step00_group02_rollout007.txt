import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Use a binary construction as fallback
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * (n_points / 2)  # ensure sum = n_points/2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    def evaluate_C5(h):
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        # Maximize over all lags
        max_val = np.max(corr) * dx
        return max_val
    
    def is_feasible(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) * dx - 1.0) < 1e-10
    
    def perturb_binary(h, num_flips=2):
        # Flip exactly num_flips 0s to 1s and 1s to 0s
        h_new = h.copy()
        indices = np.random.permutation(n_points)
        for i in range(num_flips):
            idx = indices[i]
            h_new[idx] = 1.0 - h_new[idx]
        return h_new
    
    def perturb_smooth(h, noise_level=0.05):
        # Add noise while keeping sum and bounds
        h_new = h + np.random.normal(0, noise_level, n_points)
        # Project into [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Adjust to maintain sum
        current_sum = np.sum(h_new) * dx
        adjustment = (n_points/2 - current_sum) / n_points
        h_new += adjustment
        return np.clip(h_new, 0, 1)
    
    # Start optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < 100000:
        iter_count += 1
        
        # Try different perturbation strategies
        strategies = [
            lambda x: perturb_binary(x, 1),
            lambda x: perturb_binary(x, 2),
            lambda x: perturb_smooth(x, 0.03),
            lambda x: perturb_smooth(x, 0.05),
        ]
        
        # Sample a small number of strategies
        num_strategies = min(3, len(strategies))
        selected_strategies = np.random.choice(strategies, num_strategies, replace=False)
        
        for strategy in selected_strategies:
            candidate = strategy(best_h)
            
            # Ensure feasibility and compute C5
            if is_feasible(candidate):
                C5 = evaluate_C5(candidate)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = candidate.copy()
                    last_improvement = iter_count
                    
        # Occasionally add larger perturbations
        if iter_count % 100 == 0:
            candidate = perturb_binary(best_h, 3)
            if is_feasible(candidate):
                C5 = evaluate_C5(candidate)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = candidate.copy()
        
        # Early stopping based on no improvement
        if iter_count - last_improvement > 5000:
            break
    
    # Final validation
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
