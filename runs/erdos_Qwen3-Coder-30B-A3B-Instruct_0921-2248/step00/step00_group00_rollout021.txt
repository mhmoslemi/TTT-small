import numpy as np
import cvxpy as cp
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all cross-correlations
    corr = correlate(h_values, 1 - h_values, mode="full")
    # Get the maximum value over all lags (excluding trivial lags at edges)
    max_corr = np.max(corr[n-1:])  # Only consider valid lags
    return max_corr * dx

    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Convex solver parameters
    solver = cp.CVXOPT
    
    # Iteration logic
    last_improvement = 0
    iteration = 0

        iteration += 1
        # Compute all cross-correlations and identify max lag
        corr = correlate(h, 1 - h, mode="full")

        max_corr_idx = np.argmax(corr[n-1:]) + (n-1)  # valid lag indices only
        max_lag = max_corr_idx - (n-1)
        
        # Linearize the bilinear term around current h for max lag
        # h[i](1 - h[i + max_lag]) ≈ h0[i](1 - h0[i + max_lag]) 
        # + (1 - h0[i + max_lag])(h[i] - h0[i]) - h0[i](h[i + max_lag] - h0[i + max_lag])
        # But we must ensure indices are valid
        linearized_terms = np.zeros(n)

        if max_lag >= 0:  # h[i] * (1 - h[i + max_lag])
            for i in range(n - max_lag):
                linearized_terms[i] += h[i] * (1 - h[i + max_lag])
                linearized_terms[i] += (1 - h[i + max_lag]) * (h[i] - h[i])
                linearized_terms[i] -= h[i] * (h[i + max_lag] - h[i + max_lag])
        else:  # h[i - max_lag] * (1 - h[i])
            for i in range(-max_lag, n):
                linearized_terms[i - (-max_lag)] += h[i - (-max_lag)] * (1 - h[i])
                linearized_terms[i - (-max_lag)] += (1 - h[i]) * (h[i - (-max_lag)] - h[i - (-max_lag)])
                linearized_terms[i - (-max_lag)] -= h[i - (-max_lag)] * (h[i] - h[i])
                
        # The approximation is not linear, so we simplify procedure:
        # We approximate h[i] * (1 - h[i + k]) ≈ constant + (1-h0[i+k])*(h[i]-h0[i]) - h0[i]*(h[i+k]-h0[i+k])
        # To enforce that the constraints at the maximum cross-correlation lag are satisfied,
        # We formulate the problem to minimize C5.
        # The constraints we enforce are:
        # ∑_{i} h[i] * (1 - h[i + k]) dx ≤ C5, for all k
        
        prob_c5 = cp.Variable()
        h_new = cp.Variable(n_points)
        # Make a linear constraint:
        # h[i](1 - h[i + k]) ≈ h0[i](1 - h0[i+k]) + (1-h0[i+k])(h[i] - h0[i]) - h0[i](h[i+k] - h0[i+k])
        # Sum them over all valid i and lags k to approximate vanishing the cross-correlation at max_lag.

                # i in [0, n_points), i+k in [0, n_points), so shift k offset
                if 0 <= i + k < n_points:
                    weights[k + n_points - 1, i] = 1.0  # First dimension represents k directly

        for l in range(2 * n_points - 1):
            for i in range(n_points):
                if l - (n_points - 1) >= 0 and i < n_points - (l - (n_points - 1)):
                    weights[l, i] = 1.0
                elif l - (n_points - 1) < 0 and i >= -(l - (n_points - 1)):
                    weights[l, i] = 1.0
                else:
                    weights[l, i] = 0.0

        # We define computation in optimization only:
        C5 = cp.Variable()

        constrs = []
        # The slow method is:
        for k in range(1 - n_points, n_points):
            corr_k = 0.0
            for i in range(max(0, -k), min(n_points, n_points - k)):
                corr_k += hvar[i] * (1 - hvar[i + k]) * dx
            constrs.append(corr_k <= C5)
            
        constrs.append(cp.sum(hvar) * dx == 1.0)
        for i in range(n_points):
            constrs.append(hvar[i] >= 0)
            constrs.append(hvar[i] <= 1)

        # Problem:
        prob = cp.Problem(cp.Minimize(C5), constrs)

        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new_val = hvar.value
                if h_new_val is not None:
                    h = np.clip(h_new_val, 0, 1)
                    new_c5 = evaluate_erdos_solution(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        last_improvement = iteration
        except Exception:
            pass  # regression might happen, try next iteration
        
        # Termination condition
        if iteration - last_improvement > 50:
            break
        if time() - start_time > budget_s - 1:
            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
