import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all lags
    conv = np.correlate(h, 1-h, mode="full")
    # The maximum overlap corresponds to the maximum of these values
    c5_bound = np.max(conv) * dx
    return c5_bound

        # Set a ramp from 0 to 1 and back to 0 to roughly achieve sum = n_points/2
        for i in range(n_points):
            if i < mid:
                h[i] = i / mid
            else:
                h[i] = 1 - (i - mid) / (n_points - mid)
        # Adjust the sum to be exactly n_points/2
        scale = (n_points / 2) / np.sum(h)
        h *= scale

    h = np.clip(h, 0, 1)

        conv = np.correlate(h, 1-h, mode="full")
        conv *= 2.0 / n_points  # dx = 2.0 / n_points

        max_k = len(conv) // 2

            # For lag k, overlapping indices are:
            # x in [0, n_points - 1 - abs(k)] where x+k is in [0, n_points - 1]
            # (we assume k is within reasonable bounds to fit in a sliding window)
            lim = n_points - 1 - abs(offset)
            
            if offset >= 0:
                # Positive lag: h[x] * (1 - h[x + offset])
                i_range = np.arange(lim)

                i_range = np.arange(-offset, lim)
