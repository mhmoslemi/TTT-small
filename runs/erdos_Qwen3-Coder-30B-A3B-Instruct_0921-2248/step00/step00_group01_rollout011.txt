import numpy as np
import random
import time
from scipy import correlate

def evaluate_h(h, dx):
    # Compute overlap integral
    # h is a vector of length n_points
    # Compute correlation of h with (1-h) in full mode
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum value scaled by dx
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Use initial construction if available, but binaryize it properly
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Binirize to be exactly 0/1, set sum to n_points/2
        n_points = len(h_init)
        # Convert to binary with same number of 1s as required
        sorted_indices = np.argsort(h_init)[::-1]  # descending order
        target_ones = n_points // 2
        h = np.zeros(n_points)
        h[sorted_indices[:target_ones]] = 1.0
    else:
        n_points = 200
        h = np.zeros(n_points)
        # Set n_points/2 elements to 1
        ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones_indices] = 1.0

    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial energy
    current_energy = evaluate_h(h, dx)
    best_h = h.copy()
    best_energy = current_energy
    
    T0 = 1.0
    alpha = 0.99
    T = T0
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # Generate neighbor state by swapping a 1 and 0
        # Find 0s and 1s
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            # Pick one 1 and one 0 to swap
            i1 = random.choice(ones_indices)
            i0 = random.choice(zeros_indices)
            
            # Create neighbor
            h_new = h.copy()
            h_new[i1] = 0
            h_new[i0] = 1
            
            new_energy = evaluate_h(h_new, dx)
            
            # Accept or reject
            if new_energy < current_energy or random.random() < np.exp(-(new_energy - current_energy) / T):
                h = h_new
                current_energy = new_energy
                
                if current_energy < best_energy:
                    best_energy = current_energy
                    best_h = h.copy()
        
        # Cooling
        T *= alpha
        iteration += 1
    
    # Convert back to float values in [0,1] (since we binarized)
    # But actually we are working with binary, for our task we want the continuous
    # but let's return what minimizes correlation most closely as an answer
    # it's okay if h is binary for this problem since everything evaluates correctly
    
    return (best_h.tolist(), best_energy, n_points)
