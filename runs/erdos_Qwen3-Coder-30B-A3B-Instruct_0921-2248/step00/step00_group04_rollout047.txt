import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points for balance of speed and accuracy
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        # Simple construction
        h0 = np.ones(n_points) * 0.5

    # Sample lags from [-2, 2] with step 0.1
    lags = np.arange(-2.0, 2.01, 0.1)
    n_lags = len(lags)

    # Optimization parameters
    max_iter = 100
    best_c5 = float('inf')
    best_h = None
    
    # Generate all pairwise correlations considering wrap-around
    # Add a small epsilon to improve numerical stability
    eps = 1e-8
    
    def compute_overlap(h_vals, k):
        """Compute overlap integral at lag k for a given h"""
        n = len(h_vals)
        k_idx = int(k / dx)
        if k_idx >= 0:
            shifted_h = np.concatenate([h_vals[k_idx:], h_vals[:k_idx]])
        else:
            k_idx = -k_idx
            shifted_h = np.concatenate([h_vals[k_idx:], h_vals[:k_idx]])
        return np.sum(h_vals * (1 - shifted_h)) * dx
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Build instance of the optimization
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constrs = []
        
        # Measure constraint
        constrs.append(cp.sum(h) == n_points / 2)
        
        # Bound constraints
        constrs.append(h >= 0)
        constrs.append(h <= 1)
        
        # Overlap constraints
        for k in lags:
            # Compute convolution-like integral for each lag
            k_idx = int(k / dx)
            if k_idx >= 0:
                shifted_h = cp.hstack([h[k_idx:], h[:k_idx]])
            else:
                k_idx = -k_idx
                shifted_h = cp.hstack([h[k_idx:], h[:k_idx]])
            integral = cp.sum(cp.multiply(h, 1 - shifted_h)) * dx
            constrs.append(integral <= C5)

        # Set objective
        objective = cp.Minimize(C5)
        
        # Create and solve problem
        prob = cp.Problem(objective, constrs)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_vals = h.value
                c5_val = C5.value
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_vals
        except:
            pass
        
        # Early stopping if very close to target
        if best_c5 <= 0.3807:
            break
            
        # Perturb h0 slightly to escape local minima
        h0 = h0 * (1 + np.random.normal(0, 0.01, n_points))
        h0 = np.clip(h0, 0, 1)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        
    # Final evaluation
    if best_h is None:
        best_h = h0
    else:
        # Validation
        h = best_h
        C5 = 0.0
        for k in lags:
            k_idx = int(k / dx)
            if k_idx >= 0:
                shifted_h = np.concatenate([h[k_idx:], h[:k_idx]])
            else:
                k_idx = -k_idx
                shifted_h = np.concatenate([h[k_idx:], h[:k_idx]])
            val = np.sum(h * (1 - shifted_h)) * dx
            C5 = max(C5, val)
        best_c5 = C5

    return (best_h, best_c5, n_points)
