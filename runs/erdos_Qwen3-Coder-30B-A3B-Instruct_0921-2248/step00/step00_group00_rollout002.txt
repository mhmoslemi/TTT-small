import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h):
    """Compute C5 from a step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # Take maximum over all lags
    max_corr = np.max(corr)
    # Multiply by dx for integral normalization
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        n_points = len(h)
    else:
        # Default: start with a simple construction
        n_points = 40
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6

    # Ensure sum is correct
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum

    # Define the objective function
    def objective(h_vec):
        if time() - start_time > budget_s * 0.95:
            return float('inf')
        h_vec = np.array(h_vec)
        # Enforce bounds
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)

    # Define bounds
    bounds = [(0,1) for _ in range(n_points)]

    # Define constraint: integral must be 1
    def constraint_func(h_vec):
        return np.sum(h_vec) * dx - 1.0

    cons = {'type': 'eq', 'fun': constraint_func}

    # Optimization settings
    options = {'maxiter': 500, 'ftol': 1e-7, 'gtol': 1e-7, 'disp': False}

    # Initial guess
    x0 = h.copy()

    try:
        res = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options=options,
            callback=lambda x: None  # Placeholder
        )
        if res.success or res.nit > 0:
            h_opt = np.clip(res.x, 0, 1)
            c5_value = evaluate_erdos_solution(h_opt)
        else:
            h_opt = h.copy()
            c5_value = evaluate_erdos_solution(h_opt)
    except Exception as e:
        h_opt = h.copy()
        c5_value = evaluate_erdos_solution(h_opt)

    return (h_opt.tolist(), c5_value, n_points)
