import numpy as np
from scipy import optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.full(n_points, 1.0 / n_points)
    
    # Create variables
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Constraints
    constraints = []
    
    # Bound constraints
    constraints += [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    
    # Sum constraint
    constraints += [cp.sum(h) == n_points / 2.0]
    
    # Overlap constraints for all possible shifts
    dx = 2.0 / n_points
    for k in range(n_points):
        # Compute overlap for shift k
        overlap_k = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap_k += h[i] * (1 - h[j])
        constraints += [overlap_k * dx <= C]
    
    # Problem setup
    prob = cp.Problem(objective, constraints)
    
    # Try solving
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.array(h.value).flatten()
        else:
            # Fallback: revert to initial if problem infeasible
            h_opt = h0.copy()
    except:
        h_opt = h0.copy()
    
    # Ensure feasibility
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)
    
    # Evaluate final solution
    c5_bound = 0.0
    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap += h_opt[i] * (1 - h_opt[j])
        c5_bound = max(c5_bound, overlap * dx)
    
    return (h_opt.tolist(), c5_bound, n_points)
