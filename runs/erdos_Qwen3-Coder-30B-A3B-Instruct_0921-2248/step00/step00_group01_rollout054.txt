import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    # Compute overlap integral
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points // 2
    
    # Initialize solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = (h > np.percentile(h, 50)).astype(int)
    else:
        h = np.zeros(n_points, dtype=int)
        h[:target_sum] = 1
        np.random.shuffle(h)
    
    # Track best solution
    best_h = h.copy()
    current_c5 = evaluate(h, dx)
    best_c5 = current_c5
    
    # Simulated Annealing
    T0 = 1.0
    alpha = 0.995
    T = T0
    max_iter = 100000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Create neighbor by swapping two elements
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Ensure constraint: exactly half are 1
        ones_count = np.sum(h_new)
        if ones_count > target_sum:
            # Need to flip some 1s to 0s
            to_flip = ones_count - target_sum
            flip_indices = np.where(h_new == 1)[0]
            flip_indices = np.random.choice(flip_indices, to_flip, replace=False)
            h_new[flip_indices] = 0
        elif ones_count < target_sum:
            # Need to flip some 0s to 1s
            to_flip = target_sum - ones_count
            flip_indices = np.where(h_new == 0)[0]
            flip_indices = np.random.choice(flip_indices, to_flip, replace=False)
            h_new[flip_indices] = 1
            
        # Evaluate new solution
        new_c5 = evaluate(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Cooling schedule
        T *= alpha
        iter_count += 1
        
        # Occasionally print progress
        if iter_count % 1000 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break
    
    return (best_h.tolist(), best_c5, n_points)
