import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    # Evaluate the overlap integral as defined in the problem
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute all overlaps using full correlation
    corr = correlate(h, 1 - h, mode='full')
    # The overlaps correspond to shifts from -n+1 to n-1
    # We're interested in the maximum overlap
    max_overlap = max(corr)
    return max_overlap * dx

        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()

            h_vars >= 0,

        ]

        for k in lag_set:
