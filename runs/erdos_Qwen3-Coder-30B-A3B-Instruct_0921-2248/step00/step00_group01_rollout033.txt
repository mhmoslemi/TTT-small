from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation: h * (1 - h_shifted)
    corr = np.correlate(h, 1 - h, mode="full")
    # Maximum over all shifts
    c5 = np.max(corr) * dx
    return c5
