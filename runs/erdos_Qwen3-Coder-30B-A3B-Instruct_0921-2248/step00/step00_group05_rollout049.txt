import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step function parameters
    n_points = 100  # Less than 1k as requested
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Tent function: h[i] = 0.5 * (n - abs(k))/n where k = i - n/2
        h = np.zeros(n_points)
        for i in range(n_points):
            k = i - n_points / 2
            h[i] = 0.5 * (1 - abs(k) / (n_points / 2))
        # Normalize
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    def objective(h_vec):
        # Compute all correlations for all shifts
        h_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # We consider only valid ones where the shift does not go outside bounds
        # But match the logic of the evaluation function
        # full correlation includes k = -n to n-1
        # We want the overlap integral for k = 0 to 2h_vec_len-1
        # Extend h to make corr length 2*n-1 directly
        corr = h_corr[len(h_vec)-1:]
        # Now maximum over all k (currently in range 0 to 2n-2)
        # This is max correlation over shifts
        return np.max(corr) * dx
    
    def constraint_eq(h_vec):
        # Integral constraint
        return np.sum(h_vec) - n_points / 2
    
    def constraint_lb(h_vec):
        # Lower bound: h[i] >= 0
        return h_vec
    
    def constraint_ub(h_vec):
        # Upper bound: h[i] <= 1
        return 1 - h_vec
    
    # Define constraints as a list
    constraints = [
        {'type': 'eq', 'fun': constraint_eq},
        {'type': 'ineq', 'fun': constraint_lb},
        {'type': 'ineq', 'fun': constraint_ub}
    ]
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop: alternating between gradient-based and random search
    last_improvement = 0
    best_h = h.copy()
    best_c5 = objective(h)
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Optimize with scipy
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, 
                          options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 20}, 
                          callback=lambda x: None)
        
        if result.success:
            h = result.x
        else:
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
        
        # Ensure bounds and constraint
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Update best solution
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = i
        
        # Early stopping based on improvement
        if i - last_improvement > 50:
            break
        
        # Occasionally perturb to escape local minima
        if i % 20 == 0 and i > 0:
            h += np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
    
    return (best_h, best_c5, n_points)
