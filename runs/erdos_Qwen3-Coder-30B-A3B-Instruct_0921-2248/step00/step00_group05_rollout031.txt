import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create symmetric tent function
        h = np.ones(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = (i + 1) / (mid + 1)
            h[n_points - 1 - i] = h[i]
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 and find maximum overlap indices
        correlations = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(correlations)
        max_val = correlations[max_k]
        c5 = max_val * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute gradient for this maximum overlap shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Look at the contribution to correlation at shift k
            for k in range(n_points):
                if i + k < n_points:
                    grad[i] += (1 - h[i + k]) * (k == max_k)
                if i - k >= 0:
                    grad[i] -= h[i - k] * (k == max_k)
        
        # Update with negative gradient (descent)
        h = h - alpha * grad
        
        # Project onto feasible set
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    # Compute overlap integral
    correlations = np.correlate(h, 1 - h, mode="full")
    return np.max(correlations) * dx
