import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h_vals):
    n = len(h_vals)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    correlations = signal.correlate(h_vals, 1 - h_vals, mode='full')
    # The length is 2*n - 1, so we should look at the middle part
    # for valid overlaps (k from 0 to n-1)
    # But we care about max over any k in [0, n-1]
    # Actually, let's think...
    # For k=0 to n-1 we want overlap = int h(x) * (1 - h(x+k)) dx
    # This corresponds to convolution, now we look at indices from n-1 to 2*n-2
    # indices 0 to n-2 represent negative k's
    # indices n-1 to 2*n-2 represent k=0,...,n-1
    # So we take indices n-1 to 2*n-2
    overlaps = correlations[n-1:]
    # scale by dx
    overlaps *= dx
    return np.max(overlaps)

def is_feasible(h_vals, n_points):
    # Check bounds and sum constraints
    return (np.all(h_vals >= 0) and 
            np.all(h_vals <= 1) and 
            abs(np.sum(h_vals) - n_points / 2.0) < 1e-8)

def flip_interval(h_vals, start_idx, length):
    # Flip the interval from start_idx to start_idx + length - 1
    # by setting h[i] = 1 - h[i] for each i
    new_h = h_vals.copy()
    for i in range(start_idx, min(start_idx + length, len(new_h))):
        new_h[i] = 1 - new_h[i]
    return new_h

def compute_feasible_move(h_vals, n_points):
    # Make a small change: flip 1 or 3 consecutive points (maintaining sum)
    # 1 point flip involves just changing one point
    # 3 point flip involves flipping 3 consecutive points 
    # Algorithmically choose which one to do, weighted toward small moves
    if np.random.rand() < 0.7:  # 70% chance to flip 1 point
        flip_length = 1
    else:
        flip_length = 3
    # Pick a random start index
    max_start = n_points - flip_length
    if max_start <= 0:
        start_idx = 0
    else:
        start_idx = np.random.randint(0, max_start + 1)
    new_h = flip_interval(h_vals, start_idx, flip_length)
    return new_h

        h = h * (n_points / 2.0) / h_sum
    assert is_feasible(h, n_points), "Initial h should be feasible!"
    
    # Online optimization
    T = 0.1      # Initial temperature
    alpha = 0.999  # Cooling factor
    iteration = 0
    n_iter = 0
    
    # to prevent getting stuck
    last_improvement = 0
    patience = 10000  # after this many steps without improvement, restart
    
    while time.time() - start_time < budget_s - 0.1:  # keep 0.1s buffer
        # Generate new state with a small perturbation
        new_h = compute_feasible_move(h, n_points)
        
        # Evaluate
        try:
            new_c5 = evaluate_erdos_solution(new_h)
        except Exception:
            # if we get a numerical problem, fall back to current
            new_c5 = float('inf')
            
        # Accept or reject the move
        # Always accept better solutions
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            # Accept with some probability based on temperature
            if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                h = new_h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                
        # Cooling schedule
        T = max(1e-8, T * alpha)
        iteration += 1
        n_iter += 1
        
        # If we're stuck, restart with a new random feasible solution
        if last_improvement > patience:
            # create a new random feasible solution
            h = np.random.rand(n_points)
            h_sum = np.sum(h)

    # Final validation of best solution

    c5 = evaluate_erdos_solution(h)
    return (h.tolist(), float(c5), int(n_points))
