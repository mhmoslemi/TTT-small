import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create uniform distribution
    if 'initial_h_values' in globals():

    else:
        h = np.ones(70) * 0.5  # Uniform initialization
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Keep track of best solution
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Parameters
    alpha = 0.001
    decay_factor = 0.99
    max_iterations = 1000
    tolerance = 1e-8
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute correlation for all shifts and find the one that maximizes it
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient of overlap w.r.t. h for the worst-case shift
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            # Gradient of h[i] * (1 - h[j]) w.r.t. h[i] is (1 - h[j])
            # Gradient of h[i] * (1 - h[j]) w.r.t. h[j] is -h[i]
            gradient[i] += 1 - h[j]
            if j != i:
                gradient[j] -= h[i]
        
        # Update h
        h = h - alpha * gradient
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum to target_sum
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        
        # Compute C5
        corr = correlate(h, 1 - h, mode='full')
        C5 = np.max(corr) * dx
        
        # Update best if better
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Adaptively decrease alpha
        if iteration % 50 == 0:
            alpha *= decay_factor
            
    return (best_h.tolist(), best_C5, n_points)
