import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available (n=70 samples)
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except:
        # Fallback: use uniform distribution
        n_points = 70
        h_init = np.ones(n_points) * 0.5
        h_init[0] = 0.6  # Adjust slightly to satisfy constraint
        h_init[-1] = 0.4
        h_init = h_init / np.sum(h_init) * n_points / 2
    
    # Set up variables
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Create constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Overlap constraints
    dx = 2.0 / n_points
    # Evaluate correlations manually for QCQP formulation
    # We'll build full overlap expressions for all shifts
    for k in range(n_points):
        # Compute overlap_k = sum(h[i] * (1 - h[(i+k)%n_points]) * dx) for each k
        # This represents a circular correlation term
        overlap_k = 0
        for i in range(n_points):
            overlap_k += h[i] * (1 - h[(i + k) % n_points])
        constraints.append(overlap_k * dx <= C)
    
    # Define objective
    objective = cp.Minimize(C)
    
    # Create problem and solve
    prob = cp.Problem(objective, constraints)
    
    # Try solving
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_result = h.value
        else:
            h_result = h_init
    except:
        h_result = h_init
    
    # Ensure it's within bounds
    h_result = np.clip(h_result, 0, 1)
    
    # Renormalize
    h_result = h_result / np.sum(h_result) * n_points / 2
    
    # Final evaluation using numpy correlate
    h_eval = h_result
    full_correlation = correlate(h_eval, 1 - h_eval, mode="full")
    max_overlap = np.max(full_correlation) * dx
    
    return h_eval, max_overlap, n_points
