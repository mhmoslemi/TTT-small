def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

    """Project h to feasible region."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to sum to n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
