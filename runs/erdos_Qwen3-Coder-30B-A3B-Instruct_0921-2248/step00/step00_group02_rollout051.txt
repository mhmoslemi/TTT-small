import numpy as np
import time
from scipy import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    conv = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def is_feasible(h, n_points):
    """Check if h satisfies constraints."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2.0) < 1e-10)

def perturb_h(h, n_points, dx):
    """Generate perturbed version of h."""
    h_new = h.copy()
    # Toss a coin to decide perturbation type
    if np.random.rand() > 0.5:
        # Binary flip move: Find 1 and 0, flip them
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            idx1 = np.random.choice(ones)
            idx2 = np.random.choice(zeros)
            h_new[idx1] = 0
            h_new[idx2] = 1
    else:
        # Adjust one point slightly
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        # Project back to [0,1]
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        # Renormalize
        h_new *= (n_points / 2.0) / np.sum(h_new)
    return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h

    else:
        # Binary solution with sum = n_points/2
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    start_time = time.time()
    
    # Perturbative search
    while time.time() - start_time < budget_s - 1:
        h_new = perturb_h(h, n_points, dx)
        if not is_feasible(h_new, n_points):
            continue
        c5 = evaluate_overlap(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new.copy()
        
    # Final evaluation of best solution
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
