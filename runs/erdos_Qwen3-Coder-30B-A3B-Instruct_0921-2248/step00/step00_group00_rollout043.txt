import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h function"""
    # Compute the full correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value among all lags
    max_corr = np.max(corr)
    return max_corr * dx

def validate_h(h, n_points):
    """Validate that h satisfies constraints"""
    if not (np.all(h >= 0) and np.all(h <= 1)):
        return False
    if not abs(np.sum(h) - n_points / 2.0) < 1e-8:
        return False
    return True

def generate_random_h(n_points, dx):
    """Generate a random valid h function"""
    h = np.random.rand(n_points)
    # Normalize to desired sum
    h = h * (n_points / 2.0) / np.sum(h)
    # Clip to [0,1] range
    h = np.clip(h, 0, 1)
    return h

def mutate_h(h, mutation_rate):
    """Mutate a given h function"""
    new_h = h.copy()
    # Randomly select indices to mutate
    num_mutations = int(len(h) * mutation_rate)
    indices = np.random.choice(len(h), num_mutations, replace=False)
    for i in indices:
        # Add a small random change
        new_h[i] += np.random.normal(0, 0.1)
        # Clip to valid range
        new_h[i] = np.clip(new_h[i], 0, 1)
    # Re-normalize to maintain sum
    dx = 2.0 / len(h)
    new_h = new_h * (len(h) / 2.0) / np.sum(new_h)
    new_h = np.clip(new_h, 0, 1)
    return new_h

def crossover_h(h1, h2):
    """Perform crossover between two h functions"""
    # Simple uniform crossover for half the vector each
    n = len(h1)
    point = n // 2
    new_h = np.zeros(n)
    new_h[:point] = h1[:point]
    new_h[point:] = h2[point:]
    # Ensure sum constraint is met
    dx = 2.0 / n
    new_h = new_h * (n / 2.0) / np.sum(new_h)
    new_h = np.clip(new_h, 0, 1)
    return new_h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual from population via tournament"""
    tournament_indices = np.random.choice(len(population), size=tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

    mutation_rate = 0.01
    crossover_rate = 0.7
    max_generations = 10000
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(generate_random_h(n_points, 2.0 / n_points))
    
    # Evaluation
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Track best solution
    best_h = initial_h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    generation = 0
    stagnation_count = 0
    last_improvement = 0

        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)

            best_h = current_best_h.copy()

        new_population = [best_h.copy()]  # Elitism
