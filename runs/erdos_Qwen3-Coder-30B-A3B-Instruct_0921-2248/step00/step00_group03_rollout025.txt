import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h[2] = 0.3
        h[3] = 0.7
        if n_points > 4:
            h[4:] = 0.5
    h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    alpha = 0.001
    tolerance = 1e-6
    last_c5 = float('inf')
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute all overlaps using correlation
        corr = correlate(h, 1 - h, mode='full')
        # Only consider valid overlaps (shifts from -n+1 to n-1)
        overlap_vals = corr[len(corr)//2:] * dx  # Positive shifts
        k_max_idx = np.argmax(overlap_vals)
        k_max = k_max_idx
        c5 = overlap_vals[k_max]
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # If convergence, break
        if abs(last_c5 - c5) < tolerance:
            break
        last_c5 = c5
        
        # Compute gradient sign vector
        gradient_signs = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient_signs[i] = 1 - h[j] - h[i]
            # Ensure gradient is sign only, not magnitude
            if gradient_signs[i] > 0:
                gradient_signs[i] = 1
            elif gradient_signs[i] < 0:
                gradient_signs[i] = -1
            else:
                gradient_signs[i] = 0
                
        # Update h
        h = h - alpha * gradient_signs
        # Project onto [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to maintain integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptively reduce alpha
        if iter_count % 50 == 0:
            alpha *= 0.99
            
        # Prevent very small alpha
        if alpha < 1e-8:
            alpha = 1e-8
    
    # Final computation of C5
    corr = correlate(best_h, 1 - best_h, mode='full')
    overlap_vals = corr[len(corr)//2:] * dx  # Positive shifts
    final_c5 = np.max(overlap_vals)
    
    return (best_h.tolist(), float(final_c5), n_points)
