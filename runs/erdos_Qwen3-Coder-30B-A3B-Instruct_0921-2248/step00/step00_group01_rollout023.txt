import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points // 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = (h > 0.5).astype(int)
        # Adjust to match target sum
        current_sum = np.sum(h)
        if current_sum < target_sum:
            # Set zeros to ones
            zero_indices = np.where(h == 0)[0]
            indices_to_flip = zero_indices[:target_sum - current_sum]
            h[indices_to_flip] = 1
        elif current_sum > target_sum:
            # Set ones to zeros
            one_indices = np.where(h == 1)[0]
            indices_to_flip = one_indices[:current_sum - target_sum]
            h[indices_to_flip] = 0
    else:
        # Random binary initialization
        h = np.zeros(n_points, dtype=int)
        h[:target_sum] = 1
        np.random.shuffle(h)
    
    # Precompute 1 - h
    h_complement = 1 - h
    
    def compute_c5(h_vec):
        # Use correlation to compute overlaps
        corr = correlate(h_vec, h_complement, mode="full")
        # We only care about the valid overlaps
        # The full correlation is of length 2*n_points - 1
        # Valid shifts go from -(n_points-1) to (n_points-1)
        # We want the maximum for k = 1 to n_points - 1 in the shifted form
        max_overlap = np.max(corr[n_points-1:])
        return max_overlap * dx
    
    # Initial computation
    c5_current = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_current
    
    # Simulated Annealing
    T0 = 1.0
    alpha = 0.99
    T = T0
    max_iter = 100000
    iter_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Generate neighbor by swapping two random 0 and 1
        zero_indices = np.where(h == 0)[0]
        one_indices = np.where(h == 1)[0]
        
        if len(zero_indices) == 0 or len(one_indices) == 0:
            # This should not happen with target_sum > 0 and < n_points
            break
            
        i0 = np.random.choice(zero_indices)
        i1 = np.random.choice(one_indices)
        
        # Create new h by swapping
        h_new = h.copy()
        h_new[i0] = 1
        h_new[i1] = 0
        
        # Compute new C5
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            h = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h.copy()
                c5_best = c5_current
                
        # Cool down
        T *= alpha
        iter_count += 1
        
        # Occasionally print status
        if iter_count % 10000 == 0 and iter_count > 0:
            elapsed = time.time() - start_time
            # print(f"Iter {iter_count}, Time {elapsed:.1f}s, C5 {c5_current:.5f}, Best {c5_best:.5f}")
    
    # Final validation
    actual_sum = np.sum(h_best)
    assert actual_sum == target_sum, f"Sum mismatch: expected {target_sum}, got {actual_sum}"
    
    return (h_best.tolist(), c5_best, n_points)
