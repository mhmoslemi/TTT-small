import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral for a given step function h"""
    dx = 2.0 / len(h)
    correlation = correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx

def is_feasible(h, n_points):
    """Check if h satisfies all constraints"""
    if not all(0 <= h[i] <= 1 for i in range(n_points)):
        return False
    if abs(np.sum(h) - n_points / 2.0) > 1e-8:
        return False
    return True

def mutate_h(h, mutation_rate=0.01):
    """Mutate h by adding noise while maintaining feasibility"""
    h_new = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            delta = np.random.normal(0, 0.1)
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
    
    # Renormalize to ensure sum constraint
    current_sum = np.sum(h_new)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-8:
        scaling = target_sum / current_sum
        h_new *= scaling
    
    return h_new

def crossover_h(h1, h2):
    """Perform uniform crossover between h1 and h2"""
    n_points = len(h1)
    mask = np.random.random(n_points) < 0.5
    h_new = np.where(mask, h1, h2)
    return h_new

def genetic_algorithm(initial_h_values, n_points, budget_s):
    """Run a genetic algorithm to optimize the step function h"""
    start_time = time.time()
    population_size = 100
    max_generations = int(budget_s * 10)  # Roughly 1000 generations
    mutation_rate = 0.01
    
    # Initialize population
    pop = []
    if initial_h_values is not None:
        pop.append(initial_h_values.copy())
        # Ensure feasibility
        current_sum = np.sum(pop[0])
        pop[0] *= (n_points / 2.0) / current_sum
        pop[0] = np.clip(pop[0], 0, 1)
        if not is_feasible(pop[0], n_points):
            print("Initial solution is not feasible!")
            return initial_h_values, evaluate_h(initial_h_values), n_points

    while len(pop) < population_size:
        h = np.random.random(n_points)
        # Normalize to correct sum
        current_sum = np.sum(h)
        h *= (n_points / 2.0) / current_sum
        # Clip to [0,1] range
        h = np.clip(h, 0, 1)
        if is_feasible(h, n_points):
            pop.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h) for h in pop]
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generations_without_improvement = 0
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Tournament selection with size 5
        new_pop = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, size=5, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_pop.append(pop[winner_idx].copy())
        
        # Reproduce
        offsprings = []
        for i in range(0, len(new_pop), 2):
            parent1 = new_pop[i]
            parent2 = new_pop[(i+1) % len(new_pop)]
            
            if np.random.random() < 0.7:  # Crossover
                offspring1 = crossover_h(parent1, parent2)
                offspring2 = crossover_h(parent2, parent1)
            else:
                offspring1 = parent1
                offspring2 = parent2
                
            # Mutate
            offspring1 = mutate_h(offspring1, mutation_rate)
            offspring2 = mutate_h(offspring2, mutation_rate)
            
            # Ensure feasibility
            current_sum = np.sum(offspring1)
            offspring1 *= (n_points / 2.0) / current_sum
            offspring1 = np.clip(offspring1, 0, 1)
            
            current_sum = np.sum(offspring2)
            offspring2 *= (n_points / 2.0) / current_sum
            offspring2 = np.clip(offspring2, 0, 1)
            
            offsprings.extend([offspring1, offspring2])
        
        # Replace worst with offspring
        population_fitness = [(i, fitness[i]) for i in range(len(pop))]
        population_fitness.sort(key=lambda x: x[1])
        worst_indices = [x[0] for x in population_fitness[-population_size//2:]]
        
        for i, worst_idx in enumerate(worst_indices):
            if i < len(offsprings):
                pop[worst_idx] = offsprings[i].copy()
        
        # Evaluate population
        fitness = [evaluate_h(h) for h in pop]
        current_best_idx = np.argmin(fitness)
        
        if fitness[current_best_idx] < best_c5:
            best_idx = current_best_idx
            best_h = pop[best_idx].copy()
            best_c5 = fitness[best_idx]
            generations_without_improvement = 0
        else:
            generations_without_improvement += 1
            
        if generations_without_improvement > 500:
            break
    
    return best_h, best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization to find a new upper bound for C5"""
    np.random.seed(seed)
    
    # Try to use initial_h_values if available
    try:
        h_init = initial_h_values.copy()
        n_points = len(h_init)
    except:
        # Create a random initial solution if no initial values are provided
        n_points = 45
        h_init = np.random.random(n_points)
        # Normalize to correct sum
        current_sum = np.sum(h_init)
        h_init *= (n_points / 2.0) / current_sum
        h_init = np.clip(h_init, 0, 1)
    
    try:
        best_h, best_c5, n_points = genetic_algorithm(h_init, n_points, budget_s)
        return best_h, best_c5, n_points
    except Exception as e:
        print(f"Error during optimization: {e}")
        return h_init, evaluate_h(h_init), n_points
