import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Get initial solution, fallback to uniform if not available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure sum is correct
    h = h * target_sum / np.sum(h)
    
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    
    # Initialize tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h.copy(), dx)
    
    # Start timer
    start_time = time.time()
    
    # SCA iterations
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Compute all correlations to find max lag
        corr = correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        max_k = lags[np.argmax(corr)]
        
        # Get correlation for max lag
        corr_k = corr[lags == max_k][0]
        
        # Linearization around current h
        h_linearized = h.copy()
        h_shifted = np.roll(h, max_k)
        
        # Constraint: h[i](1 - h[i + k]) <= C_k
        # Linearize around h^0:
        # h[i](1 - h[i + k]) ≈ h^[0][i] (1 - h^[0][i + k]) + (1 - h^[0][i + k]) (h[i] - h^[0][i]) - h^[0][i] (h[i + k] - h^[0][i + k])
        
        # Simplify constraints: 
        # h[i](1 - h[i + k]) <= C_k
        # => -h[i]*h[i+k] + h[i] <= C_k
        # Linearized:
        # h[i](1 - h^[0][i + k]) + (1 - h^[0][i + k]) (h[i] - h^[0][i]) - h^[0][i] (h[i + k] - h^[0][i + k]) <= C_k
        # => -h^[0][i] * h^[0][i + k] + h[i] (1 - h^[0][i + k]) + h^[0][i] - h^[0][i] h^[0][i + k] - h^[0][i] h[i + k] + h^[0][i]^2 h^[0][i + k] <= C_k
        # => -2 * h^[0][i] h^[0][i + k] + h[i] (1 - h^[0][i + k]) + h^[0][i] - h^[0][i] h[i + k] + h^[0][i]^2 h^[0][i + k] <= C_k
        # => h[i] (1 - h^[0][i + k]) - h^[0][i] h[i + k] - h^[0][i] h^[0][i + k] + h^[0][i] + h^[0][i]^2 h^[0][i + k] <= C_k
        
        # Simplified formulation:
        # h[i] (1 - h^[0][i + k]) - h^[0][i] h[i + k] <= C_k + h^[0][i] h^[0][i + k] - h^[0][i] - h^[0][i]^2 h^[0][i + k]
        
        # Use CVXPY to solve new linearized problem
        h_new = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum(cp.pos(h_new)))  # Minimize sum to promote sparsity
        
        # Objective to minimize max correlation
        # Constraints: pairwise linearity
        constraints = []
        for i in range(n_points):
            if i + max_k >= 0 and i + max_k < n_points:
                right = (1 - h_linearized[i + max_k]) * h_new[i] - h_linearized[i] * h_new[i + max_k]
                # Approximate constraint using previous max value
                approx = (
                    h_linearized[i] * (1 - h_linearized[i + max_k]) +
                    (1 - h_linearized[i + max_k]) * (h[i] - h_linearized[i]) - 
                    h_linearized[i] * (h[i + max_k] - h_linearized[i + max_k])
                )
                constraints.append(right <= approx)
        
        # Add norm constraint
        constraints.append(cp.sum(h_new) == target_sum)
        constraints.append(h_new >= 0)
        constraints.append(h_new <= 1)
        
        # Solve
        prob = cp.Problem(obj, constraints)
        
        try:
            # Increase solver tolerance and timeout if needed
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new_arr = h_new.value
                if h_new_arr is not None:
                    h = np.clip(h_new_arr, 0, 1)
                    # Project to maintain sum
                    h = h * target_sum / np.sum(h)
                    c5_current = evaluate_erdos_solution(h, dx)
                    if c5_current < best_c5:
                        best_c5 = c5_current
                        best_h = h.copy()
        except:
            pass  # Fail gracefully and continue
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    """Evaluate C5 for given h"""
    corr = np.correlate(h, 1 - h, mode="full")
    lags = np.arange(-len(h) + 1, len(h))
    max_lag = lags[np.argmax(corr)]
    return max(corr)
