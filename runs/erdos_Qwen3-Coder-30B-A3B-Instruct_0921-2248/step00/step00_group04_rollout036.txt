import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_h(h, dx):
    """Compute C5 bound for step function h"""
    n = len(h)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Get the maximum correlation (excluding the zero shift, which is not constrained)
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 42  # Use the initial construction if available
    dx = 2.0 / n_points
    
    # Start with initial_h_values if available, otherwise initialize
    try:
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.ones(n_points) * 0.5
    except:
        h0 = np.ones(n_points) * 0.5
    
    # Optimization using CVXPY with sequential constraint refinement
    h = Variable(n_points)
    C5 = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) * dx == 1,
        C5 >= 0
    ]
    
    # Start with coarse set of lags
    k_lags = np.arange(0, 2.0, 0.25)
    k_lags = k_lags[k_lags > 0]
    
    old_C5 = float('inf')
    best_h = h0.copy()
    best_C5 = evaluate_h(h0, dx)
    
    # Iterative refinement
    while time.time() - start_time < budget_s - 1:
        # Update constraints with current lags
        test_h = best_h.copy()
        for k in k_lags:
            # Compute the overlap integral for key lag k
            k_idx = int(k / dx)
            if k_idx >= n_points:
                continue
            # Correlation for shift k, 0 indexing
            corr = np.sum(test_h * np.concatenate([test_h[k_idx:], test_h[:k_idx]]))
            constraints.append(corr * dx <= C5)
        
        # Solve optimization
        prob = Problem(Minimize(C5), constraints)
        try:
            prob.solve(solver=MOSEK, verbose=False)
        except:
            break

        if prob.status != 'optimal':
            break
            
        # Extract solution
        h_val = np.array(h.value).flatten()
        current_C5 = C5.value
        
        # Evaluate actual C5
        actual_C5 = evaluate_h(h_val, dx)
        
        # Check if we improved
        if actual_C5 < best_C5:
            best_h = h_val.copy()
            best_C5 = actual_C5
            
        # Now check for new constraints to add
        new_constraints = []
        corr_values = []
        test_h = np.array(best_h)
        
        # Restart with temporary lags to compute worst-case
        for k in np.arange(0.01, 2.0, 0.05):
            k_idx = int(k / dx)
            if k_idx >= n_points:
                continue
            corr = np.sum(test_h * np.concatenate([test_h[k_idx:], test_h[:k_idx]]))
            corr_values.append((k, corr * dx))
            
        # Add top 10 worst constraints
        corr_values.sort(key=lambda x: x[1], reverse=True)
        k_lags = [k for k, _ in corr_values[:10]]
        # Remove the old constraints we've already added
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) * dx == 1,
            C5 >= 0
        ]
        if C5.value < 1e-5:
            break
            
        # Early termination
        if abs(old_C5 - current_C5) < 1e-4:
            break
        old_C5 = current_C5
        
    return (best_h.tolist(), best_C5, n_points)
