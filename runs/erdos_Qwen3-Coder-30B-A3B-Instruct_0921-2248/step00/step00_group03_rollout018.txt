import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        # Create a reasonably balanced step function
        h = np.zeros(n_points)
        # Split into 3 segments of 1/3 each, 1s in middle, 0s elsewhere
        seg_size = n_points // 3
        h[seg_size:2*seg_size] = 1.0
        # Normalize to integral = 1 (sum = n_points/2)
        h *= (n_points / 2) / np.sum(h)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        # Compute overlap integral for all shifts
        # This can be done via correlation
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # The overlap is for shifts 0 to 2*n_points-2 (which is effectively n_points-1 shifts up to 2)
        # We want the maximum of the last n_points-1 entries
        max_overlap = np.max(corr[n_points-1:])
        return max_overlap * dx
    
    def objective(h_vec):
        # We want to minimize max overlap
        return compute_C5(h_vec)
    
    def constraint_eq(h_vec):
        # Integral constraint: sum(h) * dx = 1
        return np.sum(h_vec) * dx - 1
    
    def constraint_lb(h_vec):
        # Lower bound constraint: h[i] >= 0
        return h_vec
    
    def constraint_ub(h_vec):
        # Upper bound constraint: h[i] <= 1
        return 1 - h_vec
    
    # Generate bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_eq},
    ]
    
    # Optimizer settings
    max_iter = 200  # To fit within budget
    iter_count = 0
    start_time = time.time()
    
    # For each iteration, find the shift that maximizes overlap and change h to reduce it
    h_best = h.copy()
    c5_best = compute_C5(h_best)
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Determine current max overlap shift
        h_vec = np.array(h_best)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max = np.argmax(corr[n_points-1:]) + (n_points-1)
        
        if k_max >= n_points:
            k_max = k_max - 2*n_points + 2
        
        # Find gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h_vec[j]
        
        # Update
        h_new = h_vec - 0.01 * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to maintains integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution and update if better
        c5_new = compute_C5(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            
        iter_count += 1
        
        # Check if out of time
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement using scipy minimize
    # Constraint: sum(h) = n_points/2
    def obj(h): return compute_C5(h)
    def cons(h): return np.sum(h) * dx - 1
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': cons}]
    
    res = minimize(obj, h_best, method='SLSQP', bounds=bounds, constraints=constraints, 
                    options={'maxiter': 50, 'ftol': 1e-6})
    
    h_final = np.clip(res.x, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = compute_C5(h_final)
    
    return (h_final.tolist(), float(c5_final), n_points)
