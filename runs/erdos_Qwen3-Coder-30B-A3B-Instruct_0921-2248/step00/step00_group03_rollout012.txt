import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from initial_h_values if available, otherwise create a simple construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values)
    else:
        # Simple construction: step function with value 1 in the first half, 0 in the second half
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0

    # Adjust initial h to satisfy sum constraint
    current_sum = np.sum(h_init)
    h_init = h_init * (target_sum / current_sum)
    
    # Solve the optimization problem
    start_time = time.time()
    
    # Create variables
    h = cp.Variable(n_points)
    C = cp.Variable()

    # Constraints
    constraints = [h >= 0, h <= 1]
    
    # Sum constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Overlap constraints for all shifts
    # For each shift k, we compute the correlation with (1-h)
    # The range of k values spans from -(n_points-1) to (n_points-1)
    max_k = n_points - 1
    
    for k in range(-max_k, max_k + 1):
        # Compute the shifted indices for the correlation
        idx1 = np.arange(n_points)
        idx2 = (idx1 + k) % n_points
        
        # Compute the overlap for this shift
        # It's the sum over i of h[i] * (1 - h[(i+k) % n_points])
        overlap = cp.sum(cp.multiply(h[idx1], 1 - h[idx2]))
        
        # The constraint is overlap <= C / dx
        constraints.append(overlap <= C / dx)

    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, max_iters=200, verbose=False, time_limit=budget_s)
    except:
        # If solver fails, return the initialization
        return (h_init.tolist(), max(np.correlate(h_init, 1-h_init, mode='full')) * dx, n_points)

    # Check for feasibility
    if prob.status != cp.OPTIMAL:
        # Return initialization if no good solution found
        return (h_init.tolist(), max(np.correlate(h_init, 1-h_init, mode='full')) * dx, n_points)

    # Get optimized h
    h_opt = h.value
    c5_bound = C.value
    
    # Validate solution
    if h_opt is None or np.any(h_opt < 0) or np.any(h_opt > 1):
        return (h_init.tolist(), max(np.correlate(h_init, 1-h_init, mode='full')) * dx, n_points)
    
    # Check sum constraint
    if abs(np.sum(h_opt) - target_sum) > 1e-6:
        return (h_init.tolist(), max(np.correlate(h_init, 1-h_init, mode='full')) * dx, n_points)
        
    # Compute actual C5 from cross-correlation to verify
    actual_C5 = max(np.correlate(h_opt, 1-h_opt, mode='full')) * dx
    
    # Ensure final value is correct
    c5_bound = actual_C5
    
    return (h_opt.tolist(), c5_bound, n_points)
