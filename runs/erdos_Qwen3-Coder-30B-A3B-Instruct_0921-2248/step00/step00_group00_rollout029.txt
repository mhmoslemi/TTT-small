```python
import numpy as np
import time
from scipy import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for a given h"""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1-h, mode='full')
    # The maximum value scaled by dx gives C5
    return np.max(corr) * dx

def check_constraints(h):
    """Check if h satisfies the constraints"""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - len(h)/2) < 1e-8)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population of feasible h vectors"""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
        pop_size -= 1
    
    for _ in range(pop_size):
        # Random feasible h
        h = np.random.rand(n_points)
        h = h * 2  # Scale to [0,2]
        # Adjust to have correct sum
        sum_h = np.sum(h)
        if abs(sum_h) > 1e-10:  # Avoid divide by zero
            h = h * (n_points/2) / sum_h
        # Clamp values to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
        
    return population

def crossover(parent1, parent2):
    """Uniform crossover between two parents"""
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, mutation_rate=0.01):
    """Mutation: slightly perturb a segment"""
    h = h.copy()
    n = len(h)
    if n == 0:
        return h
    # Mutate a segment
    seg_len = max(1, int(n * mutation_rate))
    start = np.random.randint(0, n - seg_len + 1)
    # Slightly perturb
    delta = np.random.normal(0, 0.1, seg_len)
    h[start:start+seg_len] += delta
    # Ensure constraints
    h = np.clip(h, 0, 1)
    # Adjust to correct sum
    sum_h = np.sum(h)
    if abs(sum_h) > 1e-10:
        h = h * (len(h)/2) / sum_h
    return h

def selection(tournament_size, population, fitnesses):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), 
                                          size=tournament_size, 
                                          replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    best_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    n_points = 45 if initial_h is None else len(initial_h)
    
    # Parameters
    pop_size = 100
    gen_size = 200  # generations to run in each batch
    mutation_rate = 0.05
    tournament_size = max(2, pop_size // 10)
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_h = None
    best_c5 = np.inf
    no_improvement_count = 0
    best_generation = 0
    
    try:
        for gen in range(gen_size):
            # Check time
            if time.time() - start_time > budget_s * 0.99:
                break
                
            # Evaluate fitness
            fitnesses = [evaluate_h(h) for h in population]
            for i, (h, fit) in enumerate(zip(population, fitnesses)):
                if fit < best_c5:
                    best_c5 = fit
                    best_h = h.copy()
                    best_generation = gen
                    no_improvement_count = 0
                elif i == 0:  # Usually first individual is updated due to initial_h
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
            # Early stopping
            if no_improvement_count > 500:
                break
                
            # Selection and reproduction
            new_population = []
            
            # Elitism - keep best
            elite_idx = np.argmin(fitnesses)
            new_population.append(population[elite_idx])
            
            # Generate offspring
            while len(new_population) < pop_size:
                parent1 = selection(tournament_size, population, fitnesses)
                parent2 = selection(tournament_size, population, fitnesses)
                
                if np.random.rand() < 0.7:
                    child1, child2 = crossover(parent1, parent2)
                else:
                    child1, child2 = parent1.copy(), parent2.copy()
                    
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                
                new_population.extend([child1, child2])
                
            # Trim to exact size
            population = new_population[:pop_size]
            
            # Decrease mutation rate if improvement was slow
            if gen > 100 and no_improvement_count > 100:
                mutation_rate = min(0.2, mutation_rate * 1.1)
                
    except Exception as e:
        pass
    
    # Final evaluation of best solution
    if best_h is not None:
        best_c5 = evaluate_h(best_h)
        # Ensure constraints are still satisfied
        if not check_constraints(best_h):
            # Resort to a known good from the GP
            best_h = initial_h if initial_h is not None else np.array([0.5] * n_points)
        # Reevaluate to be sure
        best_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
```
