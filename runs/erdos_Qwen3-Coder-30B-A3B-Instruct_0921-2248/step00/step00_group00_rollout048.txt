from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h using full correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap (C5) is the maximum of these correlations
    c5 = np.max(corr) * dx
    return c5
