import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available, otherwise create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create simple alternating step function
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 2 == 0 else 0
        h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    # Optimization iteration loop
    max_iter = 120
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        h_extended = np.concatenate([h, h])
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                overlap += h[i] * (1 - h_extended[i + k])
            overlaps[k] = overlap * dx
            
        # Find the maximum overlap and its shift
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient contribution from shift k_max
            j = (i + k_max) % n_points
            grad[i] = 1 - h[j]
            
        # Update h in direction of reducing overlap
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to sum(h) = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final validation
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate final C5
    h_extended = np.concatenate([h, h])
    overlaps = np.zeros(2 * n_points - 1)
    for k in range(2 * n_points - 1):
        overlap = 0
        for i in range(n_points):
            overlap += h[i] * (1 - h_extended[i + k])
        overlaps[k] = overlap * dx
    
    c5_bound = np.max(overlaps)
    
    return (h.tolist(), c5_bound, n_points)
