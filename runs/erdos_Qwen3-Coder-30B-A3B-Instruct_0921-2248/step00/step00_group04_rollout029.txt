import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using full convolution (cross-correlation) with 1-h
    h_reversed = np.flip(1 - h)
    correlation = signal.convolve(h, h_reversed, mode='full')
    # The maximum value of the correlation (divided by dx for normalization)
    return np.max(correlation) * dx

def normalize_h(h, target_sum):
    # Ensure sum(h) == target_sum and h[i] in [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def perturb_h(h, epsilon, target_sum):
    # Modify a few random elements while maintaining sum constraint
    h_new = h.copy()
    n = len(h)
    # Randomly select a few indices to perturb
    num_perturb = min(5, n // 2)
    indices = np.random.choice(n, num_perturb, replace=False)
    for idx in indices:
        # Perturb this element
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[idx] += delta
    # Ensure feasible solution
    h_new = normalize_h(h_new, target_sum)
    return h_new

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial evaluation
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters for Simulated Annealing
    T = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-8
    epsilon = 0.1
    
    iterations = 0
    max_iter = int(1e6)  # Avoid infinite loop
    
    while time.time() - start_time < budget_s and T > min_temp:
        if iterations >= max_iter:
            break
            
        # Perturb h
        h_new = perturb_h(h, epsilon, target_sum)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Acceptance probability
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            current_c5 = c5_new
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Cool down
        T *= cooling_rate
        epsilon = max(1e-6, epsilon * 0.9999)
        iterations += 1
    
    # Final validation
    best_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)
