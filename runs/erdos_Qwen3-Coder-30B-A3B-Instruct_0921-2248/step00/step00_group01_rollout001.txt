import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    # Compute the maximum overlap integral
    n = len(h_values)
    dx = 2.0 / n
    conv = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

        # Cosine shaped for smooth start
        h = 0.5 + 0.3 * np.cos(np.pi * x) * (x < 1) + 0.3 * np.cos(np.pi * (x-1)) * (x >= 1)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        if time.time() - start_time > budget_s - 0.1:
            return float('inf')
        # Ensure h is within bounds
        h_vec = np.clip(h_vec, 0, 1)
        # Scale to meet sum constraint
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        # Compute overlaps
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Create bounds (0,1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial objective
    best_c5 = objective(h)
    best_h = h.copy()
    
    # Optimization with single step
    try:

        for i in range(100):

            h = res.x
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    except:
        pass
    
    # Final scaling to ensure sum constraint
    current_sum = np.sum(best_h)
    if current_sum > 0:
        best_h = best_h * target_sum / current_sum
    # Final check and eval
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
