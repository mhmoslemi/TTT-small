import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    # compute overlap integral C5 using full correlation
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def normalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def perturb(h, eps, target_sum):
    # perturb a few random elements
    n = len(h)
    mask = np.random.rand(n) < 0.2
    if not np.any(mask):
        mask[0] = True
    deltas = np.random.uniform(-eps, eps, n) * mask
    h_new = h + deltas
    h_new = normalize(h_new, target_sum)
    return h_new
