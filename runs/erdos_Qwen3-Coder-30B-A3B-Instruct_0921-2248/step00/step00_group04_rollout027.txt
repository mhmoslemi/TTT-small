import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a reasonable starting point
        n_points = 42
        h = np.ones(n_points) * 0.5
        # Ensure correct sum
        h[:] = h * (n_points / 2.0) / np.sum(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Annealing parameters
    T = 1.0
    cooling_rate = 0.9995
    epsilon = 0.1
    
    # Main loop
    while time.time() - start_time < budget_s:
        # Create new candidate
        new_h = h.copy()
        
        # Random perturbations
        num_pert = np.random.randint(1, n_points // 4)
        indices = np.random.choice(n_points, num_pert, replace=False)
        
        for i in indices:
            # Add small random change
            delta = np.random.uniform(-epsilon, epsilon)
            new_h[i] += delta
            
            # Enforce bounds
            if new_h[i] < 0:
                new_h[i] = 0
            elif new_h[i] > 1:
                new_h[i] = 1
                
        # Maintain sum constraint
        old_sum = np.sum(h)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            # Adjust to maintain correct sum
            diff = (old_sum - new_sum)
            # Distribute the difference
            if abs(diff) > 1e-10:
                idx = np.random.randint(0, n_points)
                new_h[idx] = max(0, min(1, new_h[idx] - diff))
        
        # Re-normalize if needed
        if np.abs(np.sum(new_h) - n_points/2.0) > 1e-10:
            new_h = new_h * (n_points/2.0) / np.sum(new_h)
            
        # Compute new C5
        new_C5 = evaluate_C5(new_h, dx)
        
        # Accept or reject
        if new_C5 < best_C5:
            h = new_h
            best_h = new_h
            best_C5 = new_C5
        else:
            # Probability of acceptance (SA)
            delta = new_C5 - best_C5
            if np.random.random() < np.exp(-delta / T):
                h = new_h
                # Don't update best_h here
        
        # Cool down
        T *= cooling_rate
        epsilon *= 0.999
        
        if T < 1e-10:
            T = 1e-10
    
    return (list(best_h), best_C5, n_points)

def evaluate_C5(h, dx):
    n_points = len(h)
    # Compute correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    corr = corr * dx
    # Return maximum value
    return np.max(corr)
