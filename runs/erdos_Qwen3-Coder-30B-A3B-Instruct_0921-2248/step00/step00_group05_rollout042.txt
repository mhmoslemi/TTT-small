import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h - use initial if available, otherwise create symmetric tent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Create symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            dist = abs(i - mid)
            h[i] = max(0, 1 - dist / (n_points / 4))
        h = h / np.sum(h) * target_sum
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    max_no_improvement = 50
    no_improvement_count = 0
    step_size = 0.01
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Update best if better
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
                
        # Compute frequency domain representation
        H = fft.fft(h)
        H_conj = np.conj(H)
        G = fft.fft(1 - h)
        GH = H_conj * G
        
        # Convert back to time domain for overlap analysis
        overlap_in_time = fft.ifft(GH).real
        max_overlap_idx = np.argmax(overlap_in_time)
        
        # Create gradient update for worst case
        gradient = np.zeros(n_points)
        for i in range(n_points):
            shift = (i - max_overlap_idx) % n_points
            # Approximate gradient of overlap w.r.t. h[i]
            # Using forward difference of the approximated product
            gradient[i] = (overlap_in_time[max_overlap_idx] - 
                          overlap_in_time[(max_overlap_idx + 1) % n_points]) 
            
        # Smoothing based on local neighborhood
        smooth_gradient = np.zeros(n_points)
        for i in range(n_points):
            smooth_gradient[i] = 0.25 * (gradient[(i-1)%n_points] + 
                                       2 * gradient[i] + 
                                       gradient[(i+1)%n_points])
            
        # Apply update
        h_new = h - step_size * smooth_gradient
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_sum
        
        # Apply slight smoothing
        h = 0.9 * h + 0.1 * h_new
        
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
    
def evaluate_c5(h, dx):
    # Use correlation directly for accuracy
    h_array = np.array(h)
    conv = np.correlate(h_array, 1 - h_array, mode='full')
    # Normalize by dx for proper integration scaling
    # The mode 'full' gives 2*n-1 output values, center is at n-1
    max_val = np.max(conv)
    return max_val * dx
