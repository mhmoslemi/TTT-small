import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_integral = n_points / 2.0
    
    # Initialize h using the provided initial construction or a symmetric tent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_integral
    else:
        # Symmetric tent function
        h = np.array([0.5 * (1.0 - abs(i - (n_points-1)/2) / ((n_points-1)/2)) 
                      for i in range(n_points)])
        h = h / np.sum(h) * target_integral
    
    def evaluate_C5(h_vec):
        h_vec = np.array(h_vec)
        # Manual correlation computation
        corr = np.zeros(2*n_points - 1)
        for i in range(n_points):
            for j in range(n_points):
                if 0 <= i+j < len(corr):
                    corr[i+j] += h_vec[i] * (1 - h_vec[j])
        return np.max(corr) * dx
    
    # Constraint function and bounds
    def constraint_func(h_vec):
        return np.sum(h_vec) - target_integral
    
    cons = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop
    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Use L-BFGS-B with bounds for efficient optimization
    while time.time() - start_time < budget_s * 0.95:
        # Take a few steps with different algorithms for refinement
        for _ in range(10):
            result = minimize(evaluate_C5, h, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-6})
            if not result.success:
                break
            h = np.clip(result.x, 0, 1)
            # Enforce integral constraint
            h = h / np.sum(h) * target_integral
            
            C5 = evaluate_C5(h)
            if C5 < best_C5:
                best_h = h.copy()
                best_C5 = C5
        
        # Simple perturbation and hill climbing
        perturbed_h = h + np.random.normal(0, 0.01, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * target_integral
        
        if evaluate_C5(perturbed_h) < evaluate_C5(h):
            h = perturbed_h
        
        if time.time() - start_time > budget_s * 0.99:
            break
    
    return (best_h.tolist(), best_C5, n_points)
