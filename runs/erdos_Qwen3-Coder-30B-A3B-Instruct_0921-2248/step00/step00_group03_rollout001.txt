import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default starting construction: binary with 1s at even positions
        n_points = 70
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * n_points / (2 * np.sum(h))
    
    # Ensure that sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Constants
    dx = 2.0 / n_points
    max_iter = 10000
    tolerance = 1e-6
    
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute full correlation (overlap) for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        # The correlation is of length 2*n_points - 1
        # The maximum overlap occurs at the middle indices
        k_max = np.argmax(corr) - (n_points - 1)  # map index to shift
        
        # Find indices where h[i] * (1 - h[i + k_max]) is significant
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_shifted = (j + k_max) % n_points
            grad[j] = 1 - h[j_shifted]
        
        # Step size based on gradient magnitude
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project back to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to sum to n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, len(best_h))
