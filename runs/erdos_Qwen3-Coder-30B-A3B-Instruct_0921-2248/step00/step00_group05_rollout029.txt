import numpy as np
from scipy import optimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 200  # Use a reasonable balance of accuracy and speed
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (n_points / 2)
        h = h / np.sum(h) * (n_points / 2)

    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Optimization parameters
    max_iter = 5000
    alpha = 0.01
    patience = 50
    
    last_improvement = 0
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current overlaps in frequency domain
        H = fft(h)
        GH = fft(1 - h)
        overlap_fft = H * np.conj(GH)
        overlap_time = np.real(ifft(overlap_fft))
        
        # Find maximum overlap and worst shifts
        max_overlap = np.max(overlap_time)
        worst_shifts = np.where(overlap_time >= max_overlap - 1e-8)[0]
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for shift in worst_shifts:
            # Properly handle circular convolution
            shift_idx = shift if shift < n_points else shift - n_points
            for i in range(n_points):
                j = (i + shift_idx) % n_points
                grad[i] += (1 - h[j]) - h[i]
        grad *= dx
        
        # Update h
        h_new = h - alpha * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Check if new point is better
        new_C5 = compute_C5(h_new, dx)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            last_improvement = iter
        elif iter - last_improvement > patience:
            # If no improvement in patience iterations, reduce learning rate
            alpha *= 0.5
            if alpha < 1e-6:
                break
        
        h = h_new.copy()
        
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Direct computation using convolution
    h_reversed = h[::-1]
    overlap = np.correlate(h, 1 - h_reversed, mode='full')
    return np.max(overlap) * dx
