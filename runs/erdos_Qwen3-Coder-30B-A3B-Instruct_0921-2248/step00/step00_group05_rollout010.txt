import numpy as np
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Tent function
        n_points = 200
        h = np.zeros(n_points)
        for i in range(n_points):
            j = i - n_points // 2
            h[i] = max(0, 1 - 2 * abs(j) / n_points)
        h /= np.sum(h) * 2.0 / n_points  # Normalize to integral = 1

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    start_time = time.time()

    # QP solver for convex optimization
    def solve_qp(h_current, shifts):
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1]
        constraints.append(sum(h) == n_points / 2)

        # Objective: minimize weighted sum of overlaps for worst shifts
        obj = 0
        for k in shifts:
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    obj += h[i] * (1 - h_current[j])
        
        objective = Minimize(obj)
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                return np.array(h.value).flatten()
        except:
            pass
        return h_current.copy()

    # Iterate
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        h_correlate = np.correlate(h, 1 - h, mode="full")
        C5_current = h_correlate * dx
        max_correlation = np.max(C5_current)
        # Find indices where max occurs
        max_indices = np.where(np.abs(C5_current - max_correlation) < 1e-8)[0]
        
        # Convert full indices to relative shifts
        shift_indices = max_indices - (n_points - 1)
        unique_shifts = list(set(shift_indices))

        # Solve QP to minimize overlap for worst shifts
        new_h = solve_qp(h, unique_shifts)
        
        # Apply bounds and normalization
        new_h = np.clip(new_h, 0, 1)
        new_h *= (n_points / 2) / np.sum(new_h)

        # Update h and check solution
        h = new_h
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
