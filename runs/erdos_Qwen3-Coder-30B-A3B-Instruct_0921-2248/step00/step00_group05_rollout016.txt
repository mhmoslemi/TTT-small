import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a symmetric tent-like function
        n_points = 200
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * n_points / 2  # normalize to sum = n_points/2

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent parameters
    alpha = 0.01
    max_iter = 10000
    patience = 50
    patience_count = 0
    prev_best = best_c5 + 1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience_count = 0
        else:
            patience_count += 1
            if patience_count > patience:
                alpha *= 0.95
        
        # Find max overlap shift
        correlations = correlate(h, 1 - h, mode='full')
        max_shift = np.argmax(correlations)
        max_val = correlations[max_shift]
        
        # Compute gradient for this shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient contribution from shift max_shift
            if max_shift <= i:
                # h[i] contributes to overlap with h[i - max_shift]
                if i - max_shift >= 0:
                    grad[i] += -1
            else:
                # h[i] contributes to overlap with h[i + (n_points - max_shift)]
                if i + (n_points - max_shift) < n_points:
                    grad[i] += 1
        
        # Update h using gradient descent
        h = h - alpha * grad
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum  # rescale to maintain sum constraint
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    correlations = correlate(h, 1 - h, mode='full')
    return np.max(correlations) * dx
