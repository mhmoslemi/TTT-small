import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    # Compute the overlap integral using cross-correlation
    # We compute the maximum of the correlation with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The correlation is computed over the full range, we need to scale appropriately
    # maximum overlap is at the center of correlation for full overlap
    c5 = np.max(corr) * (2.0 / len(h))
    return c5


def initialize_h(n_points):
    """Initialize h with the given number of points, try to use initial_h_values if available."""
    try:
        # If initial_h_values is available, we use it
        h = np.array(initial_h_values)
        if len(h) == n_points:
            return h.copy()
        elif len(h) > n_points:
            # Too many points: truncate and normalize
            h = h[:n_points]
            h = h * (n_points / (2.0 * np.sum(h)))
            return h
        else:
            # Too few points: pad and normalize
            h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
            h = h * (n_points / (2.0 * np.sum(h)))
            return h
    except NameError:
        # If not available, create a random initial guess
        h = np.random.rand(n_points)
        # Normalize to ensure integral is 1
        h = h * (n_points / (2.0 * np.sum(h)))
        return h


def perturb_h(h, epsilon):
    """Perturb h in a way that maintains constraints."""
    # Make a copy to avoid modifying input
    h_new = h.copy()
    # Select a few random indices to adjust
    indices = np.random.choice(len(h), size=min(5, len(h)), replace=False)
    for i in indices:
        # Random perturbation in [-epsilon, epsilon]
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[i] += delta
    # Project back to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Rescale to ensure the integral constraint is met
    integral = np.sum(h_new)
    h_new = h_new * (len(h_new) / (2.0 * integral))
    return h_new


def run(seed=42, budget_s=600):
    """Run the optimization to find a better upper bound for C5."""
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42  # Reasonable compromise between resolution and speed
    
    # Initialize h
    h = initialize_h(n_points)
    
    # Current best
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Simulated Annealing parameters
    temp = 1.0  # Initial temperature
    cooling_factor = 0.999  # Geometric cooling
    epsilon = 0.1  # Initial perturbation size
    
    iter_count = 0
    max_iter = int(1e8)  # Cap the number of iterations to avoid infinite loops
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Make a perturbation
        h_new = perturb_h(h, epsilon)
        
        # Evaluate new C5
        c5_new = evaluate_c5(h_new)
        
        # Accept criteria
        delta_c5 = c5_new - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / max(temp, 1e-10)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

        # Update parameters
        temp *= cooling_factor
        epsilon = max(1e-5, epsilon * cooling_factor)  # Lower epsilon gradually
        iter_count += 1
        if iter_count % 1000 == 0:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return (best_h.tolist(), best_c5, n_points)
