import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate C5 bound for given h."""
    conv = correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Create symmetric tent function
        h = np.zeros(n_points)
        for i in range(n_points):
            j = i - n_points // 2
            h[i] = max(0, 1 - 2 * abs(j) / n_points)
        h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Iterative optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find maximum overlap and shifts contributing to it
        conv = correlate(h, 1 - h, mode="full")
        max_val = np.max(conv)
        max_indices = np.where(conv == max_val)[0]
        
        # Use only the first shift for stability
        k_indices = [max_indices[0]]
        
        # Create QP problem for this shift
        # Minimize: sum h[i] * (1 - h[i+k]) for relevant k's
        prob_h = cp.Variable(n_points)
        
        # Objective: weighted sum to approximate min of convolved terms
        # Instead of directly modeling the convolution sum, we use a heuristic
        # approach to update h locally around maximum contributions
        obj = 0
        for k in k_indices:
            # We will modify h to reduce these specific overlap values
            for i in range(n_points):
                if k < n_points:
                    j = i + k
                    if j < n_points:
                        obj += prob_h[i] * (1 - prob_h[j])
        
        # Simplify approach: solve a local QP one shift at a time
        # Get indices that contribute to max overlap
        idxs = []
        for k in k_indices:
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    idxs.append((i, j))
        
        # Construct QP
        obj = cp.Minimize(cp.sum(prod * (1 - prob_h[prod[1]])) for prod in idxs)
        
        # Use simpler convex approx approach
        # Just focus on minimizing directly the violation 
        # Let's instead construct the real QP properly
        # Minimize sum_{k} sum_i h[i](1-h[i+k])
        # Note: we iterate over a small set of violations. Here we pick one max contributor
        # Instead, let's optimize with gradient step direction or use a greedy model
        
        # Simpler and direct: just solve a local step
        # Our actual goal: adjust h to reduce max( conv(h, 1-h) * dx ) 
        
        # Direct QP approach
        prob_h = cp.Variable(n_points)
        
        # The target function to minimize is a scalar representing overlap 
        # And we want to minimize it over all (k, i) pairs that matter
        # Simplistic: we only affect the zone causing max violation
        
        # Let's virtually create QP:
        # Minimize sum_{k} sum_{i in range(n)} h[i] * (1 - h[i+k]) for worst k
        # For best approximation, iterate by improving.
        
        # Start numerical approach:
        # Update using gradient descent like idea, but with projection
        
        # Enhance this via constraint relaxation:
        # * 0 <= h[i] <= 1.0
        # * sum h = target_sum
        # * Want to minimize one max overlap term
        # Slow iterative refinement using a QP.
        
        # Let's construct a clean QP for one violation
        active_k = k_indices[0]
        
        # Identify terms: h[i]* (1 - h[i+active_k]) that contribute to max
        # Build a relaxation with a few chosen i's
        # To simplify, we'll just update via gradient approach that respects convexity
        
        # Instead, solve a true QP:
        # Optimize h[] so as to minimize sum of that overlap index
        # This is convexifed using *Jacobian approximation*
        
        # Simple idea: create gradient update directly (still iterated QP)
        # Target convex relaxation:  
        # Rely on cvxpy than full general subproblem
        
        # Start a local adjustment QP with the worst contribution
        # QP with variables for the h[]
        # What to minimize is sum of specific hh conv terms
        
        # Build a simpler but worked QP with approximate solver:
        # We'll extract pairs (i,j=some_k+i) that contribute to maximum
        # Where current overlap is especially problematic
        
        # Find i j pairs:
        conv = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(conv)
        max_val = conv[max_k]
        pairs = []
        if max_k < n_points:
            pairs.append((0, max_k))
        elif max_k >= n_points:
            pairs.append((max_k - n_points, n_points - 1))
            
        # Let's do smart QP for small block:
        prob_h = cp.Variable(n_points)
        obj = cp.Minimize(
            cp.sum([prob_h[i] * (1 - prob_h[i + active_k]) 
                   for i in range(n_points) if i + active_k < n_points])
        )
        
        constraints = [prob_h[i] >= 0 for i in range(n_points)]
        constraints += [prob_h[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(prob_h) == target_sum]
        
        problem = cp.Problem(obj, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = prob_h.value
            else:
                h_new = h
        except:
            h_new = h
            
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_sum
        h = h_new
        
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Final cleanup
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * target_sum
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
