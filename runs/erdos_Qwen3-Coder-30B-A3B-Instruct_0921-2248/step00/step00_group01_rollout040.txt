import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    # Compute the overlap integral for each shift and return max
    correlation = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = (h > 0.5).astype(int)
        n_points = len(h)
    else:
        n_points = 200  # Default resolution
        h = np.zeros(n_points, dtype=int)
        # Initialize with 1s in the first half and 0s in the second half
        h[:n_points//2] = 1
    
    # Ensure sum(h) = n_points/2
    current_sum = np.sum(h)
    desired_sum = n_points // 2
    if current_sum != desired_sum:
        # Adjust to have exactly desired_sum 1s
        indices_0 = np.where(h == 0)[0]
        indices_1 = np.where(h == 1)[0]
        # Convert some 1s to 0s
        if current_sum > desired_sum:
            num_to_change = current_sum - desired_sum
            for i in range(num_to_change):
                if i < len(indices_1):
                    h[indices_1[i]] = 0
        # Convert some 0s to 1s  
        if current_sum < desired_sum:
            num_to_change = desired_sum - current_sum
            for i in range(num_to_change):
                if i < len(indices_0):
                    h[indices_0[i]] = 1
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    T = 1.0
    alpha = 0.995
    max_iter = 1000000  # Large upper bound
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        # Generate neighbor by swapping one 0 and one 1
        indices_0 = np.where(h == 0)[0]
        indices_1 = np.where(h == 1)[0]
        
        if len(indices_0) == 0 or len(indices_1) == 0:
            break
            
        i0 = np.random.choice(indices_0)
        i1 = np.random.choice(indices_1)
        
        # Create new solution
        h_new = h.copy()
        h_new[i0] = 1
        h_new[i1] = 0
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T *= alpha
        iter_count += 1
        
        # Occasionally check if we can exit early
        if iter_count % 5000 == 0 and time.time() - start_time > 0.9 * budget_s:
            break
    
    return (best_h.tolist(), best_c5, n_points)
