import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2] = 0.5  # Middle point with value 0.5 to make sum = n_points/2
    
    # Ensure the sum constraint: integral of h should be 1
    # Since dx = 2/n_points, we want sum(h) * dx = 1, i.e., sum(h) = n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Scale h to have the correct sum
        h = h * (target_sum / current_sum)
    
    def evaluate_C5(h_vec):
        # Compute overlap integral using convolution
        # Use signal.correlate with mode='full' to get all lags
        # The convolution of h with (1 - h) will give us overlap integral
        # But we want the maximum, not the whole array
        h_minus = 1 - h_vec
        overlap_full = signal.correlate(h_vec, h_minus, mode='full')
        # Extract the relevant part. 
        # The lags range from -n_points+1 to n_points-1
        # But only consider the valid lags (lags that don't exceed array bounds)
        # The valid lags correspond to these indices in overlap_full:
        # indices from n_points-1 to 2*n_points-2
        # But to compute max overlap, we consider all lags
        # however, we need to interpret the results properly.
        # Actually use computational approach that replicates what the evaluators do
        # and later we confirm that overlap_k = sum(h[i] * (1 - h[i+k]))
        # We will compute directly using correlation definition. Slightly tricky to interpret
        # But for now, proceed using np.correlate, and see what max is
        
        dx = 2.0 / n_points
        # The result from np.correlate has length 2*n_points - 1
        # Take the maximum of all the convolved values (multiply by dx)
        return np.max(overlap_full) * dx
    
    def compute_all_overlaps(h_vec):
        # Returns the full set of overlaps
        h_minus = 1 - h_vec
        overlap_full = signal.correlate(h_vec, h_minus, mode='full')
        return overlap_full * (2.0 / n_points)  # dx = 2/n_points
    
    current_C5 = evaluate_C5(h)
    
    best_h = h.copy()
    best_C5 = current_C5
    max_evals = 1000  # Max evals to prevent long run
    n_evals = 0 
    delta = 0.1
    min_delta = 1e-4
    improvement_stall = 0
    max_stall = 20
    
    while time.time() - start_time < budget_s - 2.0:
        n_evals += 1
        if n_evals > max_evals:
            break
            
        # Compute all overlaps for current h
        overlaps = compute_all_overlaps(h)
        
        # Find maximum overlap and associated indices (lags)
        max_overlap = np.max(overlaps)
        max_lags_indices = np.where(overlaps == max_overlap)[0]
        max_lags = max_lags_indices - (n_points - 1)  # Convert index back to lag
        
        # Focus on the top staggered portion (largest overlaps)
        # Take top 5% or top 5 lags (whichever is higher)
        top_count = max(1, min(len(max_lags), int(0.05 * len(overlaps)))) 
        top_lags = max_lags[:top_count]
        
        # Try multiple adjustments
        best_adjustment = None
        best_delta_C5 = 0
        num_attempts = 0
        for lag_k in top_lags:
            num_attempts += 1
            if num_attempts > 50: break  # prevent taking too long
            
            # Compute the overlap with this lag
            # h[i] * (1 - h[i+k]) for valid i
            if lag_k >= 0: # positive lag
                h1 = h[:-lag_k] if lag_k > 0 else h
                h2 = h[lag_k:] if lag_k > 0 else h
                initial_overlap = np.sum(h1 * (1 - h2))
            else:
                h1 = h[-lag_k:] if lag_k < 0 else h
                h2 = h[:lag_k] if lag_k < 0 else h
                initial_overlap = np.sum(h1 * (1 - h2))
            
            if initial_overlap <= max_overlap * 0.99: continue  # not contributing to max
            
            # Try small adjustments to reduce this overlap
            # Try to reduce h[i] and increase h[i+k] 
            # but only within [0,1] and maintaining sum = n_points/2
            # Find candidates for adjustments
            
            feasible_x = []
            # Find positions where we can decrease h[x] and increase h[x+k]
            if lag_k > 0:
                # We look for i where h[i] > 0 and h[i+k] < 1 
                for i in range(n_points - lag_k):
                    if h[i] > delta and h[i + lag_k] < 1 - delta:
                        feasible_x.append(i)
            elif lag_k < 0:
                # -k > 0, so look for i where h[i-k] > 0 and h[i] < 1
                for i in range(-lag_k, n_points):
                    if h[i] < 1 - delta and h[i - lag_k] > delta:
                        feasible_x.append(i)
            else:
                # lag_k = 0, skip
                continue
            
            if not feasible_x:
                continue
                
            # Try adjustments
            for x in feasible_x[:5]: # 5 candidates at most
                new_h = h.copy()
                delta_adjust = delta
                
                # decrease h[x] and increase h[x+k] or h[x-k]
                if lag_k > 0:
                    # This means reduced h[x] and increased h[x+k]
                    new_h[x] = max(0, new_h[x] - delta_adjust)
                    new_h[x + lag_k] = min(1, new_h[x + lag_k] + delta_adjust)
                    # correct for sum
                    correction = (new_h[x] + new_h[x + lag_k]) - (h[x] + h[x + lag_k])
                    if abs(correction) > 0:
                        # find other x to adjust
                        for i in range(n_points):
                            if i != x and i != x + lag_k and h[i] > delta_adjust:
                                new_h[i] = max(0, new_h[i] - correction)
                                break
                        
                elif lag_k < 0:
                    # reduce h[x], increase h[x-k]
                    new_h[x] = max(0, new_h[x] - delta_adjust)
                    new_h[x - lag_k] = min(1, new_h[x - lag_k] + delta_adjust)
                    correction = (new_h[x] + new_h[x - lag_k]) - (h[x] + h[x - lag_k])
                    if abs(correction) > 0:
                        for i in range(n_points):
                            if i != x and i != (x - lag_k) and h[i] > delta_adjust:
                                new_h[i] = max(0, new_h[i] - correction)
                                break
                
                # Enforce sum constraint - make sure sum = target_sum
                new_sum = np.sum(new_h)
                if abs(new_sum - target_sum) > 1e-8:
                    # slightly adjust one element 
                    adjustment = (target_sum - new_sum) / n_points
                    new_h = new_h + adjustment  # Distribute evenly for small correction
                    # clamp to [0,1]
                    new_h = np.clip(new_h, 0, 1)
                
                # Verify feasibility  
                if not (np.all(new_h >= 0) and np.all(new_h <= 1)):
                    continue
                    
                new_C5 = evaluate_C5(new_h)
                delta_C5 = current_C5 - new_C5  # positive if improved
                
                if delta_C5 > best_delta_C5:
                    best_delta_C5 = delta_C5
                    best_adjustment = (x, lag_k, delta_adjust, new_h.copy())

        # If found an improvement, apply it
        if best_adjustment is not None:
            x, k, d, new_h = best_adjustment
            h = new_h
            current_C5 = evaluate_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
            improvement_stall = 0
            delta = max(min_delta, delta * 0.95)  # gradually decrease delta
        else:
            improvement_stall += 1
            if delta > min_delta:
                delta *= 0.95
            if improvement_stall > max_stall:
                break

    return (best_h.tolist(), best_C5, n_points)
