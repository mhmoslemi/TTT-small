import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Create a symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h[i] = i / mid
            else:
                h[i] = 1.0 - (i - mid) / (n_points - mid)
        h = h / np.sum(h) * target_sum
    
    # Ensure it's within bounds
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1.0 and iteration - last_improvement < 50:
        iteration += 1
        
        # Compute overlap in frequency domain
        H = fft(h)
        H_conj = np.conj(H)
        GH = H_conj * H
        overlap_fd = ifft(GH)
        
        # Extract the overlap (full convolution)
        overlap_values = np.real(overlap_fd)
        
        # Adjust to get the right offset (we're looking for max of 1 - h(x+k) * h(x))
        # The convolution is about h(x+k) limited to overlap with h(x)
        # We want max of sum(h(x) * (1 - h(x+k))) which maps to max of 
        # sum(h(x) - h(x)*h(x+k)) = sum(h) - sum(h(x)*h(x+k))
        # max of above over k => min of sum(h(x)*h(x+k)) over k.
        # So let's compute 1 - sum(h(x) * h(x+k) for each k
        c5_value = np.max(overlap_values)
        
        # Get the index of the worst overlap 
        worst_idx = np.argmax(overlap_values)
        if worst_idx >= n_points:
            worst_idx = worst_idx - n_points
            
        # We want update to reduce this overlap
        # Compute gradient in time domain for update
        # We know that overlap at shift k is:
        # sum over i of h[i] * h[(i+k)%n] 
        # gradient w.r.t h[i] at shift k is h[(i+k)%n] 
        # But we want to minimize sum(h[i] * (1- h[i+k]))
        # gradient w.r.t h[i] = 1 - h[i+k] - h[i+k]
        # = 1 - 2*h[i+k]
        # Here we look at all shifts, and consider the one that gives max (hard constraint)
        # Exact gradients for overlap don't help much.
        
        # Use simpler gradient descent approach on overlapping sums  
        # For each position, compute neighbor overlaps
        delta = 0.01
        h_new = h.copy()
        for i in range(n_points):
            # Try updating i with gradient
            grad = 0.0
            for k in range(n_points):
                grad += 1.0 - 2.0 * h[(i+k)%n_points]
            h_new[i] = h[i] - delta * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_sum
        
        # Compute new C5 and accept if better
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
        elif iteration - last_improvement > 10:
            # Decrease learning rate if no improvement in 10 iters
            delta *= 0.95
            
    # Final check
    if np.all(h < 0):
        h = best_h
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    # Compute all overlaps for all shifts k
    n_points = len(h)
    h_padded = np.concatenate([h, h])  # To handle wraparound easily
    # Also, note that we need 1-h in the convolution
    one_minus_h = 1.0 - h
    # Correlate h with (1-h) with full mode
    # It's like: sum_{x} h[x] * (1 - h[ x + k ])
    correlations = np.correlate(h, one_minus_h, mode='full')
    # Take from second half to get k from [0, n_points]
    correlations = correlations[n_points - 1:]
    # Find the maximum overlap
    c5 = np.max(correlations) * dx
    return c5
