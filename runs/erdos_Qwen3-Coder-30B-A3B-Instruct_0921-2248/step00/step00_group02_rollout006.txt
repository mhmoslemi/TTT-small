import numpy as np
import time
from scipy import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    conv = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def is_feasible(h, n_points):
    """Check if h is feasible (bounds and sum constraints)."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2) < 1e-10)

def perturb_h(h, n_points, dx):
    """Generate a new candidate h by small perturbations."""
    h_new = h.copy()
    # Try to flip two points: one 0->1 and one 1->0
    zeros = np.where(h_new == 0)[0]
    ones = np.where(h_new == 1)[0]
    if len(zeros) > 0 and len(ones) > 0:
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        h_new[i] = 1.0
        h_new[j] = 0.0
    else:
        # Adjust a point with small noise
        i = np.random.randint(0, len(h_new))
        delta = np.random.uniform(-0.1, 0.1)
        h_new[i] += delta
        # Project back to [0,1]
        h_new[i] = np.clip(h_new[i], 0, 1)
    # Project back to correct sum
    current_sum = np.sum(h_new)
    correction = (n_points / 2 - current_sum) / len(h_new)
    h_new += correction
    # Clamp to bounds again after correction
    h_new = np.clip(h_new, 0, 1)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Using the provided starting point size
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Build a reasonable starting point
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[1] = 1.0
        h[2] = 0.0
        h[3] = 1.0
        # Adjust to enforce sum constraint
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    start_time = time.time()
    
    # Start optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        h_new = perturb_h(h, n_points, dx)
        if not is_feasible(h_new, n_points):
            continue
        c5_new = evaluate_overlap(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Try a different approach if we're stuck
            if iter_count % 10 == 0:
                h = perturb_h(h, n_points, dx)
        # Occasionally reset to best if stuck
        if iter_count % 50 == 0 and abs(best_c5 - evaluate_overlap(h, dx)) < 1e-6:
            h = best_h.copy()
    
    return (list(best_h), best_c5, n_points)
