import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create symmetric tent function as starting point
        n_points = 200
        x = np.linspace(0, 2, n_points)
        h = 1.0 - np.abs(x - 1.0)
        h = h / np.sum(h) * n_points / 2  # normalize to sum = n_points/2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = 0.01
    
    # Optimization loop
    iteration = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        if iteration % 100 == 0:
            # Adaptive learning rate
            if iteration - last_improvement > 500:
                alpha *= 0.95
                last_improvement = iteration
                
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        max_shift = np.argmax(corr)
        max_corr = corr[max_shift]
        
        # Determine which shifts influence the maximum
        shifts = []
        for k in range(len(corr)):
            if abs(corr[k] - max_corr) < 1e-10:
                shifts.append(k)
                
        # Compute gradient for the maximum shift
        grad = np.zeros_like(h)
        for k in shifts:
            # Compute gradient w.r.t. shift k
            for i in range(n_points):
                if k <= i:
                    # h[i] contributes to h[i-k] * (1-h[i])
                    if i - k >= 0:
                        grad[i] += -h[i - k]
                else:
                    # h[i] contributes to h[i+k] * (1-h[i])
                    if i + k < n_points:
                        grad[i] += (1 - h[i + k])
        
        # Gradient descent step
        h_new = h - alpha * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * n_points / 2
        
        h = h_new
        
        # Check for improvement
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        if iteration % 50 == 0:
            # Check if we're plateauing
            if iteration - last_improvement > 1000:
                break
    
    return (list(best_h), best_c5, n_points)

def compute_c5(h, dx):
    # Compute correlation for all shifts
    corr = signal.correlate(h, 1 - h, mode='full')
    # Return the maximum value scaled by dx
    return np.max(corr) * dx
