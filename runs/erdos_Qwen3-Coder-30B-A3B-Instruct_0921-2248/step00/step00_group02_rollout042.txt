def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts k in [0, 2] with step dx
    overlaps = []
    for k in range(n):
        # Shift h by k positions with wrapping
        shifted = np.concatenate([h[k:], h[:k]])
        overlap = np.sum(h * (1 - shifted)) * dx
        overlaps.append(overlap)
    return max(overlaps)
