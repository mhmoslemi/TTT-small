import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Start with a uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    alpha = 0.001
    tolerance = 1e-6
    max_iter = 10000
    
    # Initial evaluation
    def compute_C5(h_vec):
        # Compute full correlation
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (shifts 0 to n_points-1)
        overlap = corr[len(corr)//2:] * dx
        return np.max(overlap)
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Optimization loop
    iter_count = 0
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            
        # Compute all overlaps to find worst case
        corr = correlate(h, 1 - h, mode='full')
        overlap = corr[len(corr)//2:] * dx
        k_max = np.argmax(overlap)
        
        # Compute gradient (analytical derivative)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h[j] - h[i]
        
        # Update step
        h = h - alpha * grad
        
        # Projection: Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize to fix sum
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        
        # Adaptive decrease of alpha
        if iter_count % 50 == 0 and iter_count > 0:
            alpha *= 0.99
            
        iter_count += 1
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        h = best_h
    
    return (best_h.tolist(), float(best_C5), n_points)
