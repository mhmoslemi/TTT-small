import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Initialize with uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize tracking variables
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Initial computation
    start_time = time.time()
    
    # Compute initial overlap to identify worst shift
    correlation = signal.correlate(h, 1 - h, mode='full')
    C5 = np.max(correlation) * dx
    if C5 < best_C5:
        best_C5 = C5
        best_h = h.copy()
    
    # Gradient-based optimization with adaptive step size
    alpha = 0.001
    tolerance = 1e-6
    last_C5 = C5
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlations for all shifts
        correlation = signal.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlation)
        
        # Get the actual correlation value
        worst_overlap = correlation[k_max] * dx
        
        # Compute gradient signs (for shift k_max)
        # We compute the gradient of overlap_k_max w.r.t. h
        grad_signs = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad_signs[i] = np.sign(1 - h[j] - h[i])
        
        # Update h
        h = h - alpha * grad_signs
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum to target
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        
        # Update best
        correlation = signal.correlate(h, 1 - h, mode='full')
        C5 = np.max(correlation) * dx
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Adaptive step size
        if iteration % 50 == 0:
            if abs(C5 - last_C5) < tolerance:
                alpha *= 0.99
            last_C5 = C5
        
        # Check for convergence
        if abs(C5 - last_C5) < 1e-8:
            break

    if best_C5 == float('inf'):
        # fallback to initial if needed
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
        else:
            h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        correlation = signal.correlate(h, 1 - h, mode='full')
        best_C5 = np.max(correlation) * dx
        best_h = h
    
    return (best_h.tolist(), best_C5, len(best_h))
