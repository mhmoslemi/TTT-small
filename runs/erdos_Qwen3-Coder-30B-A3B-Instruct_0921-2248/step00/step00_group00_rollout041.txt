import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Maximum value over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def is_feasible(h):
    """Check if h satisfies constraints"""
    return (np.all(h >= 0) and 
            np.all(h <= 1) and 
            abs(np.sum(h) - len(h) / 2) < 1e-8)

def mutate(h, mutation_rate):
    """Mutate h with given rate"""
    h_new = h.copy()
    n_points = len(h)
    # Flip some bits with probability mutation_rate
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            h_new[i] = np.random.random()
    # Project back to feasible region
    h_new = np.clip(h_new, 0, 1)
    # Renormalize to meet sum constraint
    current_sum = np.sum(h_new)
    target_sum = n_points / 2
    # Add correction proportional to deviation
    scale = target_sum / current_sum if current_sum > 0 else 1
    h_new = np.clip(h_new * scale, 0, 1)
    return h_new

def crossover(h1, h2):
    """Perform uniform crossover of two h vectors"""
    n = len(h1)
    mask = np.random.random(n) > 0.5
    h_new = np.where(mask, h1, h2)
    # Renormalize to meet sum constraint
    current_sum = np.sum(h_new)
    target_sum = n / 2
    scale = target_sum / current_sum if current_sum > 0 else 1
    h_new = np.clip(h_new * scale, 0, 1)
    return h_new

def select_tournament(population, fitnesses, tournament_size):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45  # Use the specified size
    population_size = 100
    max_generations = 5000
    mutation_rate = 0.05
    tournament_size = max(2, int(0.05 * population_size))
    
    # Initialize population
    population = []
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        population.append(h0)
    else:
        # Generate random feasible h with sum = n_points/2
        h0 = np.random.random(n_points)
        h0 = np.clip(h0, 0, 1)
        current_sum = np.sum(h0)
        if current_sum > 0:
            h0 = h0 * (n_points / 2) / current_sum
        population.append(h0)
    
    # Generate rest of population randomly
    for i in range(population_size - 1):
        h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        population.append(h)
    
    # Compute initial fitnesses
    fitnesses = [evaluate_h(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolution loop
    generations_since_improvement = 0
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best if improved
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            generations_since_improvement = 0
        else:
            generations_since_improvement += 1
        
        # Adaptive mutation rate
        if generations_since_improvement > 200:
            mutation_rate = min(0.2, mutation_rate * 1.1)
        
        # Create new population
        new_population = [population[0]]  # Keep best
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = select_tournament(population, fitnesses, tournament_size)
            parent2 = select_tournament(population, fitnesses, tournament_size)
            
            # Crossover
            if np.random.random() < 0.7:
                offspring = crossover(parent1, parent2)
            else:
                offspring = parent1.copy()
            
            # Mutation
            offspring = mutate(offspring, mutation_rate)
            
            new_population.append(offspring)
        
        # Update population and fitnesses
        population = new_population[:population_size]
        fitnesses = [evaluate_h(h) for h in population]
    
    # Final evaluation

            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        final_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
