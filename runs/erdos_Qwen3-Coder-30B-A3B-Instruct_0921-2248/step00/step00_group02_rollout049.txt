import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Create a binary initial construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * (n_points / 2) 

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Define the objective function
    def compute_c5(h_vec, dx):
        # Compute the overlap integral for all shifts k
        conv = correlate(h_vec, 1 - h_vec, mode="full")
        max_conv = np.max(conv) * dx
        return max_conv
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Generate a candidate h by perturbing current h
        candidate_h = best_h.copy()
        
        # Try a few randomized perturbations
        # 1. Flip two random positions (maintaining sum)
        flip_indices = np.random.choice(n_points, size=2, replace=False)
        if candidate_h[flip_indices[0]] == 1:
            candidate_h[flip_indices[0]] = 0
            candidate_h[flip_indices[1]] = 1
        else:
            candidate_h[flip_indices[0]] = 1
            candidate_h[flip_indices[1]] = 0
        
        # 2. Adjust one position with a small amount
        adjust_index = np.random.choice(n_points)
        adjustment = np.random.normal(0, 0.02)
        candidate_h[adjust_index] += adjustment
        
        # 3. Clamp to [0,1]
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # 4. Scale to maintain sum = n_points/2
        current_sum = np.sum(candidate_h)
        scale_factor = (n_points / 2) / current_sum
        candidate_h *= scale_factor
        
        # Evaluate candidate
        c5 = compute_c5(candidate_h, dx)
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate_h.copy()
            
        if iteration % 100 == 0 and time.time() - start_time > budget_s - 1:
            break  # Early exit if near budget limit

    # Final validation and compute C5
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
