import numpy as np
from scipy import signal
import time

def evaluate_overlap(h, dx):
    """Evaluate max overlap integral for given h."""
    # Compute full convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum of the convolution values
    max_overlap = np.max(conv) * dx
    return max_overlap

def check_feasibility(h, n_points):
    """Check if h satisfies constraints."""
    dx = 2.0 / n_points
    # Check bounds
    if np.any(h < 0) or np.any(h > 1):
        return False
    # Check sum constraint
    if abs(np.sum(h) * dx - 1.0) > 1e-10:
        return False
    return True

def perturb_h(h, n_points, n_swaps=3):
    """Generate a perturbed version of h."""
    # Create a copy to avoid modifying the original
    h_new = h.copy()
    
    # Select random positions to swap
    indices = np.random.choice(n_points, size=n_swaps, replace=False)
    indices = list(indices)
    
    # Keep track of changes
    changed_positions = set()
    
    # For each selected position, either toggle it (if binary) or randomize it
    for pos in indices:
        # If it's nearly 0 or 1, we can make a binary change
        if abs(h_new[pos] - 0) < 1e-5 or abs(h_new[pos] - 1) < 1e-5:
            # Flip the value
            h_new[pos] = 1 - h_new[pos]
            changed_positions.add(pos)
        else:
            # Set to a different value in (0,1)
            h_new[pos] = np.random.rand()
            changed_positions.add(pos)
    
    # If we're not already at a good constraint satisfaction, we try a small adjustment
    # Usually it's better to pick an appropriate assignment
    # But with strategy, we don't need to do anything extremely fancy
    
    return h_new

    iter_count = 0

        iter_count += 1
        # Generate perturbed h
        h_new = perturb_h(h, n_points, n_swaps=3)
        
        # Enforce the constraint on the sum
        current_sum = np.sum(h_new) * dx
