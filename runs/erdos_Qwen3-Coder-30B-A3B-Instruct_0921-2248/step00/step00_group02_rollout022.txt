import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # params
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # binary construction with approx target sum
        h = np.zeros(n_points)
        half_points = int(target_sum)
        h[:half_points] = 1.0
        # ensure exact sum
        h[0] = h[0] + (target_sum - np.sum(h))
    
    # ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # perturbation functions
    def perturb_flip(h, n_flips=2):
        h_new = h.copy()
        indices = np.random.choice(n_points, size=n_flips*2, replace=False)
        for i in range(0, len(indices), 2):
            idx0, idx1 = indices[i], indices[i+1]
            if h_new[idx0] == 1:
                h_new[idx0] = 0
                h_new[idx1] = 1
        return h_new
    
    def perturb_adjust(h, n_adjust=2):
        h_new = h.copy()
        indices = np.random.choice(n_points, size=n_adjust, replace=False)
        for i in indices:
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
        return np.clip(h_new, 0, 1)
    
    def perturb_noise(h, noise_level=0.03):
        h_new = h + np.random.normal(0, noise_level, n_points)
        h_new = np.clip(h_new, 0, 1)
        # rescale to exact sum
        h_new = h_new * target_sum / np.sum(h_new)
        return h_new
    
    # main loop
    num_evals = 0
    while time.time() - start_time < budget_s - 1:
        num_evals += 1
        
        # generate candidate using one of the perturbation strategies
        if np.random.random() < 0.4:  # 40% flip
            candidate = perturb_flip(best_h, n_flips=2)
        elif np.random.random() < 0.6:  # 20% adjust
            candidate = perturb_adjust(best_h, n_adjust=1)
        else:  # 40% noise
            candidate = perturb_noise(best_h, noise_level=0.02)
        
        # evaluate C5
        c5_candidate = evaluate_c5(candidate, dx)
        
        # update best
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = candidate.copy()
            
        if num_evals % 100 == 0:
            # recalculate for safety
            best_c5 = evaluate_c5(best_h, dx)
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h_vals, dx):
    h = np.array(h_vals)
    # compute convolution manually for: int h(x) (1 - h(x + k)) dx
    # using full mode equivalent to np.correlate
    conv = np.correlate(h, 1 - h, mode='full')
    # only consider the parts with k in [0, 2]
    # full convolution is of length 2*n_points-1
    # middle n_points elements are the "valid" convolution values
    # but we want the k in [0, 2], which means we look only at
    # k from 0 to n_points-1, which are elements n_points-1 to 2*n_points-2
    # that's the last n_points elements of the convolve result
    max_overlap = np.max(conv)
    return max_overlap * dx
