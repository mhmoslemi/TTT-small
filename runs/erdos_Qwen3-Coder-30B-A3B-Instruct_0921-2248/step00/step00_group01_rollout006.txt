import numpy as np
import time
from scipy.fft import fft, ifft, fftfreq
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100  # reasonable compromise
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # Try a simple construction
        h0 = np.ones(n_points) * 0.5
    
    h_current = h0.copy()
    h_current = h_current / np.sum(h_current) * target_sum
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, n_points)
    
    # Gradient descent in Fourier domain
    max_iter = 5000
    lr = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Project to valid range [0,1]
        h_projected = np.clip(h_current, 0, 1)
        h_projected = h_projected / np.sum(h_projected) * target_sum
        
        # Compute current C5
        c5 = evaluate_erdos_solution(h_projected, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_projected.copy()
        
        # Gradient in frequency domain
        # We compute for the overlap function conv(h, 1-h) and try to minimize its max
        if i % 10 == 0:
            h_fft = fft(h_projected)
            h_conj = np.conj(h_fft)
            one_minus_h = 1 - h_projected
            one_minus_h_fft = fft(one_minus_h)
            
            # Compute cross-correlation in frequency domain (convolution)
            cross_corr = h_fft * one_minus_h_fft
            cross_corr_real = ifft(cross_corr).real
            
            # Need to evaluate max over all shifts
            # full convolution length is 2*n_points - 1
            # We take the part from shift -n_points+1 to n_points-1
            full_corr = np.concatenate([cross_corr_real[-(n_points-1):], cross_corr_real[:n_points]])
            
            # Use simple gradient w.r.t. to input h
            max_pos = np.argmax(full_corr)
            h_grad = np.zeros_like(h_projected)
            
            # Basic gradient estimation by finite differences
            eps = 1e-5
            for j in range(n_points):
                h_plus = h_projected.copy()
                h_plus[j] += eps
                
                # Project
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / np.sum(h_plus) * target_sum
                
                c5_plus = evaluate_erdos_solution(h_plus, n_points)
                h_grad[j] = (c5_plus - c5) / eps
                
            # Update step
            h_current = h_projected - lr * h_grad
            # Project again
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current / np.sum(h_current) * target_sum
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h, n_points):
    """Compute C5, the maximum overlap integral"""
    dx = 2.0 / n_points
    h = np.array(h)
    one_minus_h = 1 - h
    
    # Use full convolution mode to compute all overlaps
    correlation = np.correlate(h, one_minus_h, mode="full") * dx
    # Return the maximum
    return np.max(correlation)
