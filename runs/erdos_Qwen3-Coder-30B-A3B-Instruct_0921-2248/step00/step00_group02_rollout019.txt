from scipy import signal

def evaluate_h(h):
    """Evaluate the overlap integral for a given h."""
    dx = 2.0 / len(h)
    # Compute convolution of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value, scale by dx
    max_overlap = np.max(corr) * dx
    return max_overlap
