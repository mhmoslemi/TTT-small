import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h_values"""
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps using full convolution
    overlap = correlate(h, 1 - h, mode="full")
    # The maximum overlap corresponds to the peak in the convolution
    c5 = np.max(overlap) * dx
    return c5

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure the constraint sum(h) = target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

        h_new = h.copy()
        for i in inds:
            h_new[i] = np.clip(h[i] + delta, 0, 1)
        return h_new
    
    def get_largest_overlaps(h, top_percent=0.05):
        """Identify the largest overlaps (C_5) and their contributing indices"""
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        indices = np.where(overlap == max_overlap)[0]
        # Get lag indices and their actual values
        lags = indices - (len(h) - 1)
        return lags, max_overlap
    
    # Optimization loop
    iter_count = 0
