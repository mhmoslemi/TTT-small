import numpy as np
from scipy.optimize import linprog
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 100
    min_improvement = 1e-5
    lag_step = 0.5
    add_lags_per_iter = 10
    
    # Start with initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[1] = 0.2
        h[-1] = 0.15
        h[-2] = 0.05
        h = h / np.sum(h) * (n_points / 2)
    
    # Get current C5 bound
    def compute_overlap(h, k):
        """Compute overlap for a specific shift k"""
        padded_h = np.concatenate([h, h[:k]])
        return np.sum(h * (1 - padded_h[k:])) * dx

    def evaluate_c5(h):
        """Evaluate the maximum overlap over all possible shifts"""
        max_overlap = 0
        k_vals = np.arange(0, 2, lag_step)
        for k in k_vals:
            k_idx = int(k / dx)
            overlap = compute_overlap(h, k_idx)
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    # Optimization process
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Initialize constraints matrix
    A_ub = []
    b_ub = []
    
    # Initial set of lags to consider
    k_vals = np.arange(0, 2, lag_step)
    lags_for_constraint = set()
    
    for i in range(100):  # Max iterations
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute current overlaps for all lags
        overlaps = []
        for k in k_vals:
            k_idx = int(k / dx)
            # Compute integral h[x] * (1 - h[x + k])
            overlap = 0
            for x in range(n_points):
                y = (x + k_idx) % n_points
                overlap += h[x] * (1 - h[y])
            overlap *= dx
            overlaps.append((overlap, k))
        
        # Sort by overlap descending
        overlaps.sort(reverse=True)
        
        # Add top lags to constraints
        for overlap_val, k in overlaps[:add_lags_per_iter]:
            if overlap_val > best_c5 * 0.9999:  # Skip small improvements
                k_idx = int(k / dx)
                if k_idx not in lags_for_constraint:
                    lags_for_constraint.add(k_idx)
                    # Add constraint: sum h[i] * (1 - h[i + k]) <= C5
                    constraint = np.zeros(n_points + 1)  # last element is C5
                    for x in range(n_points):
                        y = (x + k_idx) % n_points
                        constraint[x] = h[x] - h[x] * h[y]  # Simplified linearization
                    constraint[n_points] = -1  # C5 coefficient
                    A_ub.append(constraint)
                    b_ub.append(0)
                    
        # Solve LP problem
        # Objective: minimize C5 = sum over h + C5 term, 

        # Rebuild the constraint matrix for more precision
        # We use an LP with variable for C5 and try to minimize it
        # Constraint: sum of h_i * (1 - h_i+k) <= C5  for each lag k  

        # More systematic approach:
        # Variables: h[0], ..., h[n-1], C5
        # Minimize: C5
        # Subject to:
        #     sum(h[i]) * dx = 1
        #     for i = 0..(n-1), k = 0..(n-1): h[i] * (1 - h[(i+k)%n]) <= C5

        # Build a list of constraints
        A_eq = np.zeros((1, n_points + 1))
        A_eq[0, :n_points] = 1
        A_eq[0, n_points] = 0
        b_eq = [1]
        
        A_inner = []
        b_inner = []
        for k in list(lags_for_constraint):
            row = np.zeros(n_points + 1)
            for i in range(n_points):
                j = (i + k) % n_points
                row[i] = h[i] - h[i] * h[j]  # KLUDGE: approximate/linearize constraint
            row[n_points] = -1
            A_inner.append(row)
            b_inner.append(0.)

        # Initial conditions
        c = np.zeros(n_points + 1)
        c[n_points] = 1  # Minimize C5

        A = np.vstack([A_eq, A_inner])
        b = np.hstack([b_eq] + b_inner)
        bounds = [(0, 1) for _ in range(n_points)] + [(0, None)]  # h in [0,1], C5 >= 0

        try:
            # Attempt to solve LP using linprog with method = 'highs'
            result = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if result.success:
                h_new = result.x[:n_points]
                new_c5 = result.x[n_points]
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass

        # Evaluate C5 on full set of lags to update best
        h = best_h.copy()
        temp_c5 = evaluate_c5(h)
        if temp_c5 < best_c5:
            best_c5 = temp_c5

    # Final result
    return (best_h.tolist(), best_c5, n_points)
