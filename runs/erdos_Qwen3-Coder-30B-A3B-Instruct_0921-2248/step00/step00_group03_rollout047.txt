import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values or create a reasonable starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a binary-like initial guess with 1s in roughly even positions
        n_points = 70
        h = np.zeros(n_points)
        for i in range(0, n_points, 2):
            h[i] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / len(h)
    
    # Precompute constants to avoid recalculation
    n_points = len(h)
    total_integral = n_points / 2.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Iteration parameters
    max_iter = 2000
    tolerance = 1e-6
    warmup = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps using correlation
        # This computes all shifts and finds the maximum overlap
        full_correlation = correlate(h, 1 - h, mode='full')
        all_overlaps = full_correlation * dx
        max_overlap_idx = np.argmax(all_overlaps)
        max_overlap = all_overlaps[max_overlap_idx]
        
        # Only proceed if we're actually improving
        if iteration > warmup and max_overlap < best_c5 - tolerance:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # k_index maps to actual shift: shift = k_index - (n_points - 1)
        shift = max_overlap_idx - (n_points - 1)
        # Compute gradient for this shift by taking the derivative
        # i.e., d/dh[i] of overlap at shift k = - h[i+k] + h[i+k-1]...
        # So for shift k, gradient is partial overlap w.r.t h[i]: h[i+k] - h[i+k-1] for some indices,
        # but we're doing a more direct approach
        
        # We expand the definition
        gradient = np.zeros_like(h, dtype=float)
        for i in range(n_points):
            # To reduce overlap at shift shift, we reduce h[i] if h[(i+shift)%n_points] is significant
            if shift >= 0:
                j = (i + shift) % n_points
            else:
                j = (i - (-shift)) % n_points
            # Gradient contribution to overlap at shift
            gradient[i] += 1.0 - h[j]
        
        # Update h in gradient descent direction
        step_size = 0.001
        h_new = h - step_size * gradient
        
        # Project h_new back to feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain integral = total_integral
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * total_integral / current_sum
            
        h = h_new
        
        # Report progress occasionally
        if iteration % 100 == 0:
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    # Create a full cross-correlation over all shifts
    # The overlap integral for shift k is:
    #     int h(x) (1 - h(x+k)) dx
    # Using discrete convolution (correlation) and scaling:
    n = len(h)
    dx = 2.0 / n
    # Compute: h * (1 - h) for all shifts using full convolution
    conv = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv * dx)
    return max_overlap
