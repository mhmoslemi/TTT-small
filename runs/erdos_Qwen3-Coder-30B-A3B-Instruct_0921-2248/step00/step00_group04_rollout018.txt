import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lag = 2.0
    lag_step = 0.1
    lags = np.arange(-max_lag, max_lag + lag_step/2, lag_step)
    
    # Initialize with a good starting point
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * n_points / 2
    else:
        # Simple initial guess
        h0 = np.ones(n_points) * 0.5
    
    # Build optimization problem
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) * dx == 1.0,
        C5 >= 0
    ]
    
    # Add overlap constraints for sampled lags
    for k in lags:
        # Compute shifted indices
        shift = int(np.round(k / dx))
        if shift == 0:
            overlap = h @ (1 - h)
        else:
            # Handle periodic boundary conditions
            if shift > 0:
                h_shifted = np.concatenate([h[shift:], h[:shift]])
            else:
                h_shifted = np.concatenate([h[shift:], h[:shift]])
            overlap = h @ (1 - h_shifted)
        constraints.append(overlap * dx <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Optimization
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with GLPK if available, otherwise MOSEK
    try:
        prob.solve(solver=cp.GLPK, verbose=False, time_limit=budget_s)
    except:
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s)
        except:
            # Fallback: just use initial values
            h_opt = h0.copy()
            C5_val = evaluate_erdos_solution(h_opt)
            return (h_opt, C5_val, n_points)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        C5_val = C5.value
    else:
        h_opt = h0.copy()
        C5_val = evaluate_erdos_solution(h_opt)
        
    return (h_opt, C5_val, n_points)

def evaluate_erdos_solution(h):
    """Internal evaluator to test solution quality."""
    # Using numpy's correlate to compute overlaps
    h = np.array(h)
    c5_vals = np.correlate(h, 1 - h, mode="full")
    return np.max(c5_vals) * (2.0 / len(h))
