import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h[:n_points]
        if len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), mode='constant', constant_values=0)
    else:
        # Symmetric tent function
        h = np.array([min(i, 2*n_points - i) for i in range(2*n_points)]).astype(float)
        h = h[:n_points]
        h = h / np.sum(h) * (n_points / 2)
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute FFTs
    H = fft.fft(h)
    GH = fft.fft(1 - h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps in frequency domain
        overlap_fft = H * np.conj(GH)
        overlap_time = fft.ifft(overlap_fft).real
        
        # Find maximum overlap
        max_idx = np.argmax(overlap_time)
        max_val = overlap_time[max_idx]
        
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
            
        # Compute gradient w.r.t. h
        # Approximate gradient for the maximum overlap
        grad = np.zeros_like(h)
        for k in range(n_points):
            shift = (k - max_idx) % n_points
            grad[k] += (1 - h[(k + shift) % n_points]) * dx
        
        # Apply gradient step
        h += lr * grad
        
        # Project to constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update FFTs
        H = fft.fft(h)
        GH = fft.fft(1 - h)
        
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            lr = max(lr * 0.95, 1e-6)
    
    return (best_h.tolist(), best_c5, n_points)
