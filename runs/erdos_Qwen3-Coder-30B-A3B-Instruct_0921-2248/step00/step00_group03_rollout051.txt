            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1.0 - h[j])
            overlaps.append(overlap * dx)
        return max(overlaps)

    def constraint_sum(h):
        return np.sum(h) * dx - 1.0

    constraints = {'type': 'eq', 'fun': constraint_sum}

    result = minimize(objective, h0, method='SLSQP', bounds=bounds,
