import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum of conv * dx over all valid shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def construct_initial_h():
    """Construct an initial h vector from initial_h_values if available"""
    try:
        h = np.array(initial_h_values, dtype=float)
        return h
    except NameError:
        # Fallback to a simple construction
        n = 45
        h = np.zeros(n)
        # Set roughly half to 1.0
        h[:n//2] = 1.0
        return h

def perturb_h(h, n_perturb=5):
    """Generate a perturbed version of h"""
    h_new = h.copy()
    n = len(h)
    dx = 2.0 / n
    
    # Perform perturbations
    for _ in range(n_perturb):
        if np.random.random() < 0.5:
            # Swap a 0 with a 1, maintaining sum constraint
            zeros = np.where(h_new == 0)[0]
            ones = np.where(h_new == 1)[0]
            if len(zeros) > 0 and len(ones) > 0:
                i = np.random.choice(zeros)
                j = np.random.choice(ones)
                h_new[i] = 1.0
                h_new[j] = 0.0
        else:
            # Adjust a value with a small random change
            idx = np.random.randint(0, n)
            # Add small random perturbation, clamp to [0,1]
            h_new[idx] = np.clip(h_new[idx] + np.random.uniform(-0.1, 0.1), 0, 1)
    
    # Ensure sum constraint is maintained
    current_sum = np.sum(h_new) * dx
    if abs(current_sum - 1.0) > 1e-10:
        # Adjust all values to correct the sum
        scale = 1.0 / current_sum
        h_new = h_new * scale
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Initial construction
    h_best = construct_initial_h()
    h_best = h_best / np.sum(h_best) * (n_points / 2)  # Ensure correct sum
    
    c5_best = evaluate_overlap(h_best, dx)
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        # Generate multiple candidates
        n_candidates = 10
        candidates = []
        for _ in range(n_candidates):
            h_candidate = perturb_h(h_best, n_perturb=5)
            c5_candidate = evaluate_overlap(h_candidate, dx)
            candidates.append((h_candidate, c5_candidate))
        
        # Select the best candidate
        h_new, c5_new = min(candidates, key=lambda x: x[1])
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            # print(f"New best C5: {c5_best:.6f} at iteration {iter_count}")
    
    # Final check for feasibility
    h_best = np.clip(h_best, 0, 1)
    actual_sum = np.sum(h_best) * dx
    h_best = h_best * (1.0 / actual_sum)
    
    return (h_best.tolist(), float(c5_best), int(n_points))
