import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations with 1-h using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the max of these correlations
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2.0)  # scale to sum = n_points/2
    else:
        h = np.ones(n_points) * 0.5  # fallback to uniform
    
    # Use CVXPY with MOSEK solver, if installed
    try:
        solver = cp.MOSEK
    except AttributeError:
        solver = cp.CLARABEL
    finally:
        pass
    
    def solve_with_constraints(k_list, current_h):
        n = len(current_h)
        h_var = cp.Variable(n)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) * dx == 1
        ]
        
        # Add overlap constraints
        for k in k_list:
            # Weights will be 1 if i < n-k, 1 if i >= k, 0 otherwise
            # But this is better with convolution-like indexing
            # For each k, compute convolution index
            # h[i] * (1 - h[(i+k) % n])
            # This is equivalent to: sum over i of h[i] * (1 - h[(i+k)%n])
            # Build the constraint: sum h[i] * (1 - h[(i + k) % n]) <= C
            shift = k % n  # cyclic shift
            overlap = 0.0
            for i in range(n):
                overlap += h_var[i] * (1 - h_var[(i + shift) % n])
            constraints.append(overlap * dx <= 1.0)
        
        objective = cp.Minimize(1.0)  # dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value.copy()
                return new_h
        except Exception:
            pass
        return current_h.copy()
    
    # Refined optimization approach
    C_current = evaluate_erdos_solution(h)
    C_best = C_current
    h_best = h.copy()
    print(f"Initial C5: {C_current:.6f}")
    
    # Generate set of k values to initially consider, step of 0.1
    k_values = np.arange(0.0, 2.0, 0.1) 
    lags_to_consider = [int(np.round(k/dx)) for k in k_values if k < 2.0]
    lags_to_consider = list(set(lags_to_consider))  # unique
    
    # Iteratively optimize
    iterations = 0
    while time.time() - start_time < budget_s - 2.0:
        iterations += 1
        # Evaluate all lags to see what is the worst one
        max_overlap = -1
        worst_k = -1
        for k in lags_to_consider:
            # compute overlap
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1.0 - h[j])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k
                
        if max_overlap <= C_current:
            break
            
        # Add the worst k to the problem
        try:
            h = solve_with_constraints([worst_k], h)
        except Exception:
            break
            
        C_current = evaluate_erdos_solution(h)
        if C_current < C_best:
            C_best = C_current
            h_best = h.copy()
            
        if time.time() - start_time > budget_s - 5.0:
            break
            
        # Add some more lags for active search
        # Do not infinite expand lags, just add the top few
        new_lags = []
        top_k = 20  # Top evals
        for k in range(n_points//2):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1.0 - h[j])
            overlap *= dx
            if overlap > C_current * 0.6:  # Add that potential worse
                new_lags.append(k)
                
        #.Select top
        if len(new_lags) > 5:
            # Sort by overlap value descending, but return 5
            new_lags = sorted(new_lags, key=lambda k: sum(h[i]* (1-h[(i+k)%n_points]) for i in range(n_points))*dx, reverse=True)[:5]
        lags_to_consider = list(set(lags_to_consider + new_lags))
        
    # Final value
    final_c5 = evaluate_erdos_solution(h_best)
    # Ensure final h is valid (sums correctly, bounded)
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best / np.sum(h_best) * (n_points / 2.0)

    return (h_best, final_c5, n_points)
