import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Solve using CVXPY
    h = cp.Variable(n_points, name='h')
    C = cp.Variable(1, name='C')
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints for all shifts
    for k in range(-n_points + 1, n_points):
        if time.time() - start_time > budget_s - 5:  # Leave buffer for final evaluation
            break
            
        # Compute valid indices for the shift
        indices = np.arange(n_points)
        shifted_indices = (indices + k) % n_points
        
        # Compute overlap integral for shift k
        # This is equivalent to summing h[i] * (1 - h[shifted_indices[i]])
        overlap_expr = cp.sum(cp.multiply(h, 1 - h[shifted_indices]))
        constraints.append(overlap_expr <= C * dx)
        
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solver options
    prob.solve(solver=cp.ECOS, max_iters=200, verbose=False, time_limit=budget_s)
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        c5_bound = C.value
    else:
        # fallback to initial solution
        h_opt = h_init
        c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    # Ensure feasibility and return
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)  # adjust sum to exactly n_points/2
    c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    # Ensure return is valid
    h_opt = np.array(h_opt).tolist()
    return (h_opt, c5_bound, n_points)
