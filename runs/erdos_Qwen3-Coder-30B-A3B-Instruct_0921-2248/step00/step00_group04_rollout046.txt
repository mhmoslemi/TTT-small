import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lag = 2.0
    lag_step = 0.1
    
    # Generate lag samples
    lags = np.arange(-max_lag, max_lag + lag_step/2, lag_step)
    n_lags = len(lags)
    
    # Initialize h with the initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        # Simple starting point: uniform distribution
        h_init = np.ones(n_points) * 0.5
    
    # Precompute the correlation matrix for efficient constraint evaluation
    def compute_overlap(h_vec, k):
        """Compute overlap integral for lag k."""
        h_shifted = np.roll(h_vec, -int(k/dx))
        return np.sum(h_vec * (1 - h_shifted)) * dx

    # Initialize variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        h >= 0,
        h <= 1,
        C5 >= 0
    ]
    
    # Add overlap constraints for sampled lags
    for k in lags:
        if time.time() - start_time > budget_s - 5:  # Leave time for final check
            break
        # Approximate the overlap integral
        # Use a simpler formulation with explicit sum
        # h[i] * (1 - h[(i-k)%n]) 
        # Evaluate terms in the convolution-like sum
        def build_overlap_constraint(k_val):
            if k_val == 0:
                return cp.sum(h * (1 - h)) * dx <= C5
            else:
                # For simplicity, build this with simple indexing
                # This is not exactly the continuous case but works well with the 
                # discrete nature of the problem
                i_indices = np.arange(n_points)
                k_indices = ((i_indices - int(k_val / dx)) % n_points).astype(int)
                term = cp.sum(h[i_indices] * (1 - h[k_indices])) * dx
                return term <= C5

        try:
            constraints.append(build_overlap_constraint(k))
        except Exception:
            # If building constraint fails, skip it
            pass
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve using GLPK (more confident in speed)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            c5_bound = C5.value
        else:
            # Fallback to initial solution if optimization fails
            h_sol = h_init
            c5_bound = float('inf')
            evals = [compute_overlap(h_init, k) for k in lags]
            c5_bound = max(evals)
    except Exception:
        # If optimization fails completely, use the initial
        h_sol = h_init
        c5_bound = float('inf')
        evals = [compute_overlap(h_init, k) for k in lags]
        c5_bound = max(evals)

    # Final validation
    if time.time() - start_time > budget_s - 5:
        # Return best found
        pass

    # Final evaluation using the actual evaluator function (assumed to exist)
    # Since the evaluator is not provided, we compute C5 directly with
    # sampling as in the problem description (assumes np.correlate is available)
    try:
        # Recompute C5 with pooling
        h_temp = np.array(h_sol)
        actual_c5 = max(np.correlate(h_temp, 1 - h_temp, mode="full")) * dx
        c5_bound = actual_c5
    except:
        pass

    return (h_sol, c5_bound, n_points)
