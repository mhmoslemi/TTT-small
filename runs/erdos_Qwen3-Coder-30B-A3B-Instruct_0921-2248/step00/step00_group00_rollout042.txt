import numpy as np
import time
from scipy import correlate

def evaluate_h(h):
    """Evaluate the C5 constant for given h."""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value adjusted by dx
    c5 = np.max(corr) * dx
    return c5

def is_feasible(h):
    """Check if h satisfies constraints."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - len(h) / 2.0) < 1e-8)

def mutate(h, mutation_rate=0.01):
    """Apply mutation to h vector."""
    h_mut = h.copy()
    n = len(h)
    # Mutate a random subset of elements
    num_mutations = max(1, int(n * mutation_rate))
    indices = np.random.choice(n, num_mutations, replace=False)
    for i in indices:
        # Mutate by small random amount avoiding boundaries
        delta = np.random.normal(0, 0.05)
        h_mut[i] = np.clip(h_mut[i] + delta, 0, 1)
    return h_mut

def crossover(h1, h2):
    """Perform uniform crossover between two vectors."""
    n = len(h1)
    mask = np.random.rand(n) > 0.5
    offspring1 = np.where(mask, h1, h2)
    offspring2 = np.where(mask, h2, h1)
    return offspring1, offspring2

def tournament_select(population, fitnesses, tournament_size=5):
    """Perform tournament selection."""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

    # GA parameters
    population_size = 100
    max_generations = 6000
    mutation_rate = 0.01
    
    # Initialize population with h_init and random members
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)  # normalize
        population.append(h)
    
    # Evaluation
    fitnesses = [evaluate_h(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection
        selected = tournament_select(population, fitnesses, 5)
        
        # Crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < population_size:
            # Select two parents
            p1_idx, p2_idx = np.random.choice(len(selected), 2, replace=False)
            p1, p2 = selected[p1_idx], selected[p2_idx]
            
            # Crossover
            if np.random.rand() < 0.7:
                c1, c2 = crossover(p1, p2)
                # Ensure feasibility
                c1 = c1 * (n_points / 2.0) / np.sum(c1)
                c2 = c2 * (n_points / 2.0) / np.sum(c2)
                new_population.extend([c1, c2])

                # Just copy parents
                new_population.extend([p1, p2])

        new_population = new_population[:population_size]
