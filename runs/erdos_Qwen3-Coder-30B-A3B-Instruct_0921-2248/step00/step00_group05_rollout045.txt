import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    tol = 1e-6
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a symmetric tent-shaped initial function
        h = np.zeros(n_points)
        mid = n_points // 2
        width = n_points // 4
        for i in range(mid - width, mid + width):
            if 0 <= i < n_points:
                h[i] = 1.0 - abs(i - mid) / width
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation and find maximum overlap
        corr = signal.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient for the specific shift causing max overlap
        grad = np.zeros(n_points)
        if max_k < n_points:
            # Convolution h * (1-h) place 0 means h[0] * (1-h[0]) degree 0
            # Convolution place k means h[i] * (1-h[i+k]) 
            for i in range(n_points):
                if i + max_k < n_points:
                    grad[i] = -(1 - h[i + max_k])
                else:
                    # If overflow wrap or treat as 0 (this shouldn't occur in valid shifts)
                    grad[i] = 0
        else:
            # For shifts > n_points, align properly
            k_offset = max_k - n_points + 1
            for i in range(n_points):
                if i < k_offset:
                    grad[i] = 0
                else:
                    grad[i] = -(1 - h[i - k_offset])
        
        # Update using projected gradient
        h_new = h - alpha * grad
        
        # Project onto constraints
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Check convergence
        if np.linalg.norm(h_new - h, ord=2) < tol:
            break
            
        h = h_new
        
        # Keep best
        current_C5 = compute_C5(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
    
    # Final check on best solution
    best_C5 = compute_C5(best_h, dx)
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    corr = signal.correlate(h, 1 - h, mode='full')
    max_k = np.argmax(corr)
    return corr[max_k] * dx
