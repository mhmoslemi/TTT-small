import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for given h"""
    corr = signal.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def is_feasible(h, n_points):
    """Check if h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2) < 1e-8)

def mutate_h(h, mutation_rate=0.01):
    """Mutate h with given rate while maintaining feasibility"""
    h_new = h.copy()
    n_points = len(h)
    if n_points == 0:
        return h_new
    num_mutations = max(1, int(n_points * mutation_rate))
    for _ in range(num_mutations):
        i = random.randint(0, n_points - 1)
        delta = random.gauss(0, 0.05)
        h_new[i] = max(0, min(1, h_new[i] + delta))
    
    # Renormalize to satisfy sum constraint
    scale = (n_points / 2) / np.sum(h_new)
    h_new = h_new * scale
    return h_new

def crossover_h(h1, h2):
    """Uniform crossover between two h vectors"""
    n_points = len(h1)
    if n_points != len(h2):
        return h1.copy()
    mask = np.random.rand(n_points) > 0.5
    h_new = np.where(mask, h1, h2)
    
    # Renormalize
    scale = (n_points / 2) / np.sum(h_new)
    h_new = h_new * scale
    return h_new

    tournament_size = 5
    max_generations = 6000
    stagnation_limit = 500
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)

    best_c5 = fitness[best_idx]
    
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s and generation < max_generations:
        generation += 1
        # Tournament selection
        new_population = []
        for _ in range(population_size):
            # Tournament
            tournament_indices = random.sample(range(population_size), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Create offspring via crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[(i + 1) % population_size]
            if random.random() < crossover_rate:
                child1 = crossover_h(parent1, parent2)
                child2 = crossover_h(parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate_h(child1, mutation_rate)
            child2 = mutate_h(child2, mutation_rate)
            
            offspring.extend([child1, child2])
        
        # Replace worst individuals
        population = offspring[:population_size]
        
        # Evaluate new population
        fitness = [evaluate_h(h, 2.0 / n_points) for h in population]
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_h = population[new_best_idx].copy()
            best_c5 = fitness[new_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        if stagnation_count >= stagnation_limit:
            mutation_rate = min(0.1, mutation_rate * 1.1)
            stagnation_count = 0
        
        if generation % 100 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break
    
    # Final verification
    if not is_feasible(best_h, n_points):
        # Try to fix by renormalizing
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return best_h, best_c5
