import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    
    # Initialize h from initial_h_values or use a symmetric tent function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a symmetric tent function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = (i + 1) / (mid + 1)
            h[n_points - 1 - i] = h[i]
        h[mid] = 1.0
        # Scale to sum to n_points / 2
        h = h / np.sum(h) * (n_points / 2)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the correlation to find maximum overlap
        corr = signal.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        
        # Determine the shift that causes the maximum correlation
        shift = max_corr_idx - (len(h) - 1)
        
        # Compute gradient: if shift is positive, we compute d/dh_i of integral over n shifts
        # For position i, the gradient of integral with respect to h_i is (1 - h[i+shift]) when valid
        # and 0 otherwise.
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i + shift < len(h):
                grad[i] = 1 - h[i + shift]
            else:
                # treat out-of-bounds as zero contribution to overlap
                grad[i] = 0
        
        # Update h with gradient descent
        h_new = h - alpha * grad
        
        # Project h_new back onto the feasible set: [0,1] and sum=1
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        h = h_new
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
