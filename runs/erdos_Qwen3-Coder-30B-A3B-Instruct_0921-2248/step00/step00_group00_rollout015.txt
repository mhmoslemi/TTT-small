import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_C5(h):
    """Evaluate C5 bound for given h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective(h):
    """Objective function to minimize (negative for minimization)"""
    return -evaluate_C5(h)

def constraint_eq(h):
    """Equality constraint: sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):

    # Discretization
    dx = 2.0 / n_points
    
    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Optimize using SLSQP
    result = None
    best_h = h0.copy()
    best_c5 = evaluate_C5(h0)
    
    # SLSQP with tighter tolerances
    opt_options = {
        'maxiter': 500,
        'ftol': 1e-7,
        'gtol': 1e-7,
        'disp': False
    }
    
    # Initial optimization with SLSQP
    try:
        result = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=opt_options,
            callback=lambda x: None  # Empty callback for now
        )
        
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            c5_opt = evaluate_C5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception as e:
        pass
    
    # Refinement with different starting points and a local refinement step
    for i in range(10):  # Limited restarts
        if time.time() - start_time > budget_s - 1:

        h_new = best_h + np.random.normal(0, 0.05, n_points)

                constraints=constraints,

                h_test = np.clip(res.x, 0, 1)
                c5_test = evaluate_C5(h_test)
                if c5_test < best_c5:
                    best_h = h_test
                    best_c5 = c5_test
        except:
            continue
    
    # Final evaluation
    final_c5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
