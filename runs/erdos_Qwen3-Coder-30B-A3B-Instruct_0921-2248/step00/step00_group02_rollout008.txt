import numpy as np
from scipy.signal import correlate
import time


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial solution if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Create initial binary solution and perturb
        n_points = 45
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid] = 0.5
        h[mid + 1:] = 0.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to satisfy constraint
        h = h * target_sum / current_sum
    
    # Initial evaluation
    def eval_c5(h_vec):
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # full convolution gives 2*n - 1 elements, but we want max over k = 0 to 2*n-2
        # but actual lag in first element is -(n-1) to (n-1)
        # so we map full conv indices to actual lags: 
        # lag 0 corresponds to index (n-1) in full conv
        # lag k corresponds to index (n-1+k) in full conv
        # lag -k corresponds to index (n-1-k) in full conv
        max_k = (n_points - 1) * 2
        conv_vals = conv[(n_points-1):]  # get positive lags
        conv_vals_neg = conv[:(n_points-1)]  # get negative lags
        all_conv = np.concatenate([conv_vals_neg[::-1], conv_vals])  # map to lags -n+1 to n-1
        result = np.max(all_conv) * dx
        return result
    
    # Helper to adjust h
    def adjust_h(h_vec, idx1, k, delta):
        # h[idx1] -= delta, h[idx1 + k] += delta
        h_new = h_vec.copy()
        if 0 <= idx1 < n_points and 0 <= idx1 + k < n_points:
            if h_new[idx1] >= delta and 0 <= h_new[idx1 + k] + delta <= 1:
                h_new[idx1] -= delta
                h_new[idx1 + k] += delta
                return h_new
        return h_vec
    
    # Adjust multiple points
    def adjust_multiple(h_vec, deltas, selected_indices, k_vals):
        h_new = h_vec.copy()
        total_delta = 0
        for idx, k, delta in zip(selected_indices, k_vals, deltas):
            # Make sure indices are valid and adjustments are feasible
            if 0 <= idx < n_points and 0 <= idx + k < n_points:
                if h_new[idx] >= delta and 0 <= h_new[idx + k] + delta <= 1:
                    h_new[idx] -= delta
                    h_new[idx + k] += delta
                    total_delta += delta
                else:
                    # Revert if not feasible
                    h_new = h_vec.copy()
                    return h_new
        return h_new

    start_time = time.time()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # stepsize
        delta = max(0.01, 0.1 * (1.0 - (time.time() - start_time) / budget_s))
        
        # Find maximum overlaps
        conv = correlate(h, 1-h, mode='full')
        conv_vals = conv[(n_points-1):]  # positive lags
        conv_vals_neg = conv[:(n_points-1)]  # negative lags
        all_conv = np.concatenate([conv_vals_neg[::-1], conv_vals])
        max_val = np.max(all_conv)
        
        # Get top 5% lags that contribute most to max
        threshold = max_val * 0.99  # Consider lags contributing to >99% of max
        high_lag_indices = np.where(all_conv >= threshold)[0]
        
        # For every such lag found, try to find and adjust such pairs of points
        shrank = False
        for i in range(len(high_lag_indices)):
            # We'll just adjust a fixed number
            if time.time() - start_time > budget_s - 1:
                break
            
            # attempt to adjust 
            # to find tradeoffs, look at problematic pairs more carefully
            
            # Get corresponding lag index from full convolution to actual lag
            lag_idx = high_lag_indices[i % len(high_lag_indices)]
            
            # Map lag_idx to actual lag
            k = lag_idx - (n_points - 1)
            
            # Find pairs x, x+k such that h[x] = 1 and h[x+k] = 0
            # We'll just brute-force a few candidates
            possible_indices = []
            for x in range(n_points):
                if (0 <= x + k < n_points and h[x] > 0.95 and h[x + k] < 0.05):
                    possible_indices.append((x, k))
            
            if len(possible_indices) == 0:
                continue
            
            best_new_h = h
            best_new_c5 = best_c5
            
            # Try pairs of indices
            for (x, k) in possible_indices[:5]:  # don't try too many
                if (x + k) < 0 or (x + k) >= n_points:
                    continue
                
                # Try adjusting from 1 to 1-delta and 0 to 0+delta in corresponding pair
                new_h = adjust_h(h, x, k, delta)
                new_c5 = eval_c5(new_h)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_new_h = new_h
            
            if best_new_c5 < best_c5:
                h = best_new_h
                best_c5 = best_new_c5
                shrank = True
                
        # If improvement is small, re-evaluate with more aggressive adjustments
        if not shrank:
            # Try joint adjustments
            if len(high_lag_indices) >= 1:
                converged = True
                current_c5 = eval_c5(h)
                for i in range(20):  # test as long as needed
                    cands = []
                    weights = []
                    if len(high_lag_indices) == 0:
                        break
                    
                    # Focus on some key indices
                    selected_lags = high_lag_indices[:3]  # top lags
                    for k_idx in selected_lags:
                        k = k_idx - (n_points-1)
                        candidates = []
                        for x in range(n_points):
                            if 0 <= x + k < n_points:
                                if h[x] > 0.9 and h[x+k] < 0.1:
                                    candidates.append((x, k, min(0.1, h[x])))
                        if candidates:
                            cands.extend(candidates)
                    
                    if not cands:
                        continue
                        
                    # Try triplet of such adjustments (to reduce sum and spatial pairs)
                    for i1 in range(len(cands)):
                        if i1 + 2 >= len(cands):
                            break
                        (x1, k1, d1) = cands[i1]
                        (x2, k2, d2) = cands[i1+1]
                        (x3, k3, d3) = cands[i1+2] if i1+2 < len(cands) else (x1, k1, d1)
                        
                        # try adjusting three at once (check feasibility of each)
                        try:
                            new_h = h.copy()
                            if (0 <= x1 < n_points and 0 <= x1 + k1 < n_points and 
                                0 <= x2 < n_points and 0 <= x2 + k2 < n_points and
                                0 <= x3 < n_points and 0 <= x3 + k3 < n_points):
                                if (new_h[x1] >= d1 and 0 <= new_h[x1 + k1] + d1 <= 1 and
                                    new_h[x2] >= d2 and 0 <= new_h[x2 + k2] + d2 <= 1 and
                                    new_h[x3] >= d3 and 0 <= new_h[x3 + k3] + d3 <= 1):
                                    new_h[x1] -= d1
                                    new_h[x1 + k1] += d1
                                    new_h[x2] -= d2
                                    new_h[x2 + k2] += d2
                                    new_h[x3] -= d3
                                    new_h[x3 + k3] += d3
                                    new_c5 = eval_c5(new_h)
                                    if new_c5 < current_c5:
                                        h = new_h
                                        current_c5 = new_c5
                                        converged = False
                        except:
                            pass
                    
                    if converged:
                        break

        if best_c5 < eval_c5(best_h):
            best_h = h.copy()
    
    if best_c5 < eval_c5(h):
        h = best_h
    
    # Ensure constraint
    h_sum = np.sum(h)
    h = h * target_sum / h_sum
    
    final_c5 = eval_c5(h)
    return (h.tolist(), float(final_c5), n_points)
