import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_h(h, dx):
    # Compute the overlap integral C5
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def ensure_feasible(h, n_points):
    # Ensure h is in [0,1] and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-8:
        # Adjust by scaling
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    # Uniform crossover
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, mutation_rate):
    # Mutate by adding normal noise and clipping
    h = h + np.random.normal(0, 0.05, len(h)) * mutation_rate
    h = ensure_feasible(h, len(h))
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    # Select one individual using tournament selection
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 45  # Use initial construction if available
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    fitnesses = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = ensure_feasible(h, n_points)
        population.append(h)
        fitness = evaluate_h(h, dx)
        fitnesses.append(fitness)
    else:
        # Generate a random feasible h
        h = np.random.rand(n_points)
        h = ensure_feasible(h, n_points)
        population.append(h)
        fitness = evaluate_h(h, dx)
        fitnesses.append(fitness)
    
    # Generate rest of the population
    for i in range(population_size - 1):
        h = np.random.rand(n_points)
        h = ensure_feasible(h, n_points)
        population.append(h)
        fitness = evaluate_h(h, dx)
        fitnesses.append(fitness)
    
    # Track the best solution found
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    mutation_rate = 0.01
    stagnation_count = 0
    max_stagnation = 500
    
    while time.time() - start_time < budget_s - 0.1:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top individuals
        elite_size = population_size // 2
        new_population = population[:elite_size]
        new_fitnesses = fitnesses[:elite_size]
        
        # Generate new offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if random.random() < 0.7:  # Crossover probability
                child1, child2 = crossover(parent1, parent2)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutate children
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            new_fitnesses.extend([
                evaluate_h(child1, dx),
                evaluate_h(child2, dx)
            ])
            
            # Check time again
            if time.time() - start_time > budget_s - 0.1:
                break
        
        # Trim to right size
        population = new_population[:population_size]
        fitnesses = new_fitnesses[:population_size]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adapt mutation rate
        if stagnation_count > 100:
            mutation_rate = min(mutation_rate * 1.1, 0.1)
        elif stagnation_count > 250:
            mutation_rate = 0.1
            stagnation_count = 0
        
        if stagnation_count > max_stagnation:
            break
    
    # Final evaluation of best solution
    best_fitness = evaluate_h(best_h, dx)
    return (best_h.tolist(), best_fitness, n_points)
