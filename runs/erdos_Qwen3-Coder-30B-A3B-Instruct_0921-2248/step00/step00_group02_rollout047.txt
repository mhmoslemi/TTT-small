import numpy as np
from scipy import signal
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute overlap using convolution
    conv = signal.correlate(h, 1 - h, mode='full')
    # Normalize by dx to convert to integral
    conv *= dx
    # Return maximum overlap value over all shifts
    return np.max(conv)

def is_feasible(h, n_points):
    """Check if h satisfies constraints."""
    dx = 2.0 / n_points
    # Check bounds
    if np.any(h < 0) or np.any(h > 1):
        return False
    # Check sum constraint
    if abs(np.sum(h) * dx - 1.0) > 1e-10:
        return False
    return True

def perturb_h(h, n_points, swaps=2):
    """Create a perturbed version of h."""
    h_new = h.copy()
    dx = 2.0 / n_points
    
    # Random swaps between 0 and 1
    indices = np.random.choice(n_points, size=swaps*2, replace=False)
    
    for i in range(0, len(indices), 2):
        idx1, idx2 = indices[i], indices[i+1]
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
    
    # Ensure sum constraint
    total_new = np.sum(h_new) * dx
    h_new *= (1.0 / total_new)
    
    # Clamp values to [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    return h_new
