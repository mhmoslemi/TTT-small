import numpy as np
import time
from scipy import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    conv = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def validate_solution(h, n_points):
    """Ensure h is feasible."""
    dx = 2.0 / n_points
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1.0) < 1e-10)

def perturb_h(h, n_points, dx):
    """Generate a new feasible h by small perturbations."""
    h_new = h.copy()
    # Try a few random swaps to keep sum intact
    indices = np.random.choice(n_points, size=4, replace=False)
    for i in range(0, len(indices), 2):
        if i + 1 < len(indices):
            idx1, idx2 = indices[i], indices[i+1]
            # Swap 0 and 1 while maintaining sum
            if h_new[idx1] == 1 and h_new[idx2] == 0:
                h_new[idx1] = 0
                h_new[idx2] = 1
            elif h_new[idx1] == 0 and h_new[idx2] == 1:
                h_new[idx1] = 1
                h_new[idx2] = 0
            elif h_new[idx1] < 1 and h_new[idx2] > 0:
                # adjust values while maintaining sum
                diff = min(h_new[idx2], 1 - h_new[idx1])
                h_new[idx1] += diff
                h_new[idx2] -= diff
            elif h_new[idx1] > 0 and h_new[idx2] < 1:
                # adjust values while maintaining sum
                diff = min(h_new[idx1], 1 - h_new[idx2])
                h_new[idx1] -= diff
                h_new[idx2] += diff
    # Ensure bounds and sum
    h_new = np.clip(h_new, 0, 1)
    current_sum = np.sum(h_new) * dx
    scale_factor = (n_points / 2) / current_sum if current_sum != 0 else 1
    h_new = h_new * scale_factor
    return h_new

        iterations += 1
        h_new = perturb_h(h, n_points, dx)
        
        if validate_solution(h_new, n_points):
            c5_new = evaluate_overlap(h_new, dx)
            if c5_new < best_c5:
