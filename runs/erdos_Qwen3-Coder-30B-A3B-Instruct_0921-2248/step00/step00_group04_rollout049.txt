import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def evaluate_C5(h, dx):
    # Compute overlap integral for all lags
    n = len(h)
    # Create correlation of h with (1-h) in full mode
    conv = np.correlate(h, 1 - h, mode='full')
    # The lags go from -(n-1) to (n-1)
    # We only care about lags 0 to 2, which are indices (n-1) to (2*n-2)
    # For lag k, we need to be careful with boundaries
    max_val = 0
    for i in range(2 * n - 1):
        k = i - (n - 1)
        if 0 <= k <= 2:
            max_val = max(max_val, conv[i] * dx)
    return max_val

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create default starting point
        n_points = 42
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.4  # Perturb slightly
        h_start[-1] = 0.6
    
    dx = 2.0 / n_points
    h = h_start.copy()
    
    # Initialize with coarse lag sampling
    lag_step = 0.5
    lags = np.arange(0, 2 + lag_step/2, lag_step)
    max_lag = 2.0
    last_C5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Initialize optimization variables
        h_vars = cp.Variable(n_points, bounds=(0, 1))
        C5 = cp.Variable()
        
        # Constraints based on sampled lags
        constraints = []
        constraints.append(cp.sum(h_vars) * dx == 1)
        
        # Compute correlation for each lag
        for k in lags:
            # Compute correlation with 1-h shifted by k
            # For discrete case, use convolution for each valid k
            # this is an approximation since discrete convolution for continuous case is complex
            # But we will sample k from a grid
            corr = 0
            for i in range(n_points):
                j = (i + int(k/dx)) % n_points
                corr += h_vars[i] * (1 - h_vars[j])
            # Add lag constraint
            constraints.append(corr * dx <= C5)
        
        # Objective
        problem = cp.Problem(cp.Minimize(C5), constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            C5_val = C5.value
            
            if not np.isfinite(C5_val):
                break
                
            # Update current solution
            h = h_vars.value
            
            # Re-evaluate actual C5 with more dense lag evaluation
            # Find worst lag by sampling more densely
            best_lag = -1
            max_overlap = 0
            if C5_val < last_C5 * 0.99:
                # Sample more lags for detailed check
                dense_lags = np.linspace(0, 2, 100)
                for k in dense_lags:
                    corr = 0
                    for i in range(n_points):
                        j = (i + int(k/dx)) % n_points
                        corr += h[i] * (1 - h[j])
                    overlap = corr * dx
                    if overlap > max_overlap:
                        max_overlap = overlap
                        best_lag = k
                
                # Add the worst lag as constraint if beneficial
                if max_overlap > C5_val * 0.9995:
                    lags = np.append(lags, best_lag)
                    if len(lags) > 200:  # Cap constraint count
                        lags = lags[1:]  # Remove first
        
        except Exception as e:
            break
            
        if time.time() - start_time > budget_s - 1 or abs(last_C5 - C5_val) < 1e-5:
            break
            
        last_C5 = C5_val
    
    # Final check with dense grid
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    final_C5 = evaluate_C5(h, dx)
    
    return (h, final_C5, n_points)
