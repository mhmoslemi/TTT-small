import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h_values, n_points):
    h = np.array(h_values)
    dx = 2.0 / n_points
    overlaps = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlaps) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Start from initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a reasonable initial guess
        h_init = np.ones(n_points) * 0.5
        # Ensure sum matches constraint
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    start_time = time.time()
    
    # Binary search parameters
    low = 0.0
    high = 0.38085
    tolerance = 1e-5
    max_iterations = 20
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        mid = (low + high) / 2.0
        # Create optimization variables
        h = Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
        
        # Approximate overlaps using McCormick envelopes
        # For simplicity, we'll proceed with direct numerical optimization
        # Set up the objective to minimize C
        def objective(h_vals):
            h_arr = np.array(h_vals)
            return evaluate_erdos_solution(h_arr, n_points)
        
        # Use scipy minimize to improve the solution
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            objective,
            h_init,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        
        if result.success:
            h_opt = result.x
            c5_new = evaluate_erdos_solution(h_opt, n_points)
            if c5_new < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_new
            if c5_new <= mid:
                high = mid
            else:
                low = mid
        else:
            break
            
        if high - low < tolerance:
            break
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    # Return result
    return (list(best_h), final_c5, n_points)
