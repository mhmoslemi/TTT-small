import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        # Place 1s in a staggered pattern
        for i in range(n_points // 2):
            h[2*i] = 1.0
        # Normalize
        h = h * n_points / (2 * np.sum(h))
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_overlap(h_vec):
        # Compute C5 = max_k int h(x)(1-h(x+k)) dx
        # Using full cross-correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # We care about the shift k values from 0 to 2*n_points - 2
        # But let's extract the [1, 2*n_points-2] part (non-overlapping parts)
        # Actually compute just the positive shifts (0 through n_points-1) 
        # and compare to max of all lags
        max_corr = -np.inf
        for k in range(2*n_points - 1):
            # We need correct convolution for proper overlap
            # overlap_k = sum(h[i] * (1 - h[(i + k) % n_points]))
            # This is equivalent to: sum(h[i] * 1) - sum(h[i] * h[(i + k) % n_points])
            # = sum(h) - cross_correlation[k]
            # 
            # But in fact, 
            # overlap_k = sum(h[i] * (1 - h[(i + k) % n_points]))
            #         = sum(h[i]) - sum(h[i] * h[(i + k) % n_points])
            # The sum(h[i]) is always target_sum, independent of k
            # But that's wrong since the convention of convolution in np is different.
            # Correct way:
            # compute all lags: correlation from k=0 to k=2n-2, shift k means
            # h[0] * (1 - h[k]), h[1] * (1 - h[(k+1)%n]), ...
            # corr calculation:
            # Truncate to ensure we have proper alignment
            if k < n_points:
                val = np.sum(h_vec[:n_points-k] * (1 - h_vec[k:]))
            else:
                val = np.sum(h_vec[k-n_points+1:] * (1 - h_vec[:n_points-(k-n_points+1)]))
            if k >= n_points-1:
                val = np.sum(h_vec[:n_points-(k-n_points+1)] * (1 - h_vec[k-n_points+1:]))
            max_corr = max(max_corr, val)
        return max_corr * dx
    
    def objective_and_gradient(h_vec):
        # Return (C5_value, gradient_wrt_h)
        # We compute overlap over all k and find the max
        # Then compute gradients w.r.t. h for this max
            
        # Compute correlations for all shifts
        h_rev = h_vec[::-1]
        corr_full = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Correlations are over indices [0, ... 2*n_points-2] for shift k = 0..2*n_points-2
        
        # We have overlap_k = sum(h[i] * (1 - h[(i+k)%n]))  = sum(h[i]) - sum(h[i] * h[(i+k)%n])
        # Which is:
        # sum(h) - corr_full[k + n - 1] = target_sum - corr_full[k + n - 1]
        # So overlap = target_sum - corr_full[k + n - 1]
        # Then overlap_k  = sum(h[i]) - \sum_i h[i](1-h[i+k]) = sum(h[i]) - \sum_i h[i]*h[i+k] =target_sum - corr_full[k+ n-1]
        
        # But integrating over all shifts:
        # Each term is: sum(h[i] * (1-h[i+k]))  or            sum(h[i] * h[i+k])
        overlaps = []
        # compute overlap_k for shift k = 0 through 2 * n_points - 2
        for k in range(2 * n_points - 1):
            # After flipping to match convolution
            # index k goes to corr_full[2*n - 2 - k]
            # specifically for which efficient indexing:
            # actually: h[i] (1 - h[i+k]) requires h * (1 - h_shifted)
            # Makes more sense to avoid complex 
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_vec[i] * (1.0 - h_vec[j])
            overlaps.append(overlap)
        overlaps = np.array(overlaps) * dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        # Compute analytical gradient w.r.t. h_vec of this max overlap
        # The gradient involves (1-h_{i+k}) terms and (h_{i+k} terms) for contribution.
        # i.e. gradient of overlap_k = \sum_j h[j] * (1 - h[j+k]), w.r.t. h_i
        # Each term in sum is h[j] * (1 - h[j+k]), 
        # if j=i then contributes (1-h[i+k]) 
        # if j = i-k then contributes -h[i]
        # For max_overlap_idx = k_max, the gradient wrt h[i] is:
        # 0 if i does not affect overlap_k_max (i.e. i+k_max >= n or i < k_max)
        # 1-h[i+k_max] if j = i (so i in support of sum and k_max shift)
        # -h[i+k_max] if j = i-k_max (i>k_max, i is pre-images that also on the right)
        # makes sense: h[i] affects it if i + k_max < n_points or if i - k_max >=0
        # But in the full overlap matrix, what matters is overlap for shift k_max:
        # So terms come from i+j = k_max or (i + k) % n  = k_max
        # which is: i = (k_max - j) % n
        
        grad = np.zeros_like(h_vec)
        k_max = max_overlap_idx
        # For sum h[i] * (1-h[(i+k_max)%n])
        # i contributes 1-h[(i+k_max)%n] 
        # j contributes -h[i] if i+j=k_max (or i = (k_max - j) % n)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] += 1.0 - h_vec[j]
        # Also: i*(k_max) contribute term -h_i [only if (j-k_max)%n = i]
        for i in range(n_points):
            j = (i + k_max) % n_points
        # Plus derivative of h[i] effect on BLUE_TERM (sum h[i] - sum h[i]*h[(i+k)%n] taken at k_max )
        # But actually we just know:
        # gradient of out sum of h[i] * (1 - h[j]) over terms of overlap_k
        # ===  gradient of [h[i] - h[i]*h[i+k_max]] 
        # dz[i] /dh[i] = 1 - h[i+k_max] if somehow makes sense?
        
        # Simpler derivation:
        # For overlap = sum(h[i] * (1 - h[(i+k)%n]))
        # grad[i] = (1 - h[i+k%n]) 
        # over all i such that i+k%n < n and (i+k)%n = (i+k)%n 
        # For last index, j = (i+k)%n, j=n-1 means i = (n-1-k)%n 
        # In fact let's enumerate all indices i:
        # term is: h[i] * (1 - h[(i+k)%n]) = h[i] - h[i] * h[(i+k)%n]
        # So the gradient wrt h[i] = 1 - h[(i+k)%n] if we wanted the gradient wrt h[i]
        # but overlap_k = sum_{i} h[i] * (1 - h[i + k])
        # we should compute per i: i is present only once, but so are h[i+k] 
        # Let's shift index g: in for p_i, look at i+k = g, i = g-k, if g now is i+k, 
        # then gradient wrt h[p] = 1 - h[p] if p = g = i+k (i contributes for this overlap) 
        # grad_loops out [1-h[p]] for all appearances of this k-lag = i
        # Oh, i go as 0 to n-1 in sum, for this fixed k=shift
        # So partial of overlap_k wrt h[i] = 1 - h[i+k]
        
        # Actually i is ignored in care of j, since we try to get over all j in h[j], 
        # but need to compute for which i, term j = i+k mod n:
        # h_i * (1 - h_{i+k})
        
        grad = np.zeros_like(h_vec)
        if k_max < n_points:
            # The right side (iursion in forward direction) 
            # i: current index i, and h[i+k] = h[(i+k)%n] 
            # => gradient of (h[i](1-h[i+k])) wrt h[i] = (1 - h[i+k])
            grad = 1.0 - h_vec[k_max:]   # Could be numpy behavior...
            # However, when k_max >= i, we're indexing h[i+k] and invalid. 
            idx = [i for i in range(n_points)]
            # Label shifts in different part of indexing region
            for i in range(n_points):  
                effect_idx = (i + k_max) % n_points
                grad[i] = 1.0 - h_vec[effect_idx]
        else:
            # This case we had computed over multiple parts, hard to compute by index
            grad = np.zeros_like(h_vec)
            for i in range(n_points):
                idx = (i + k_max) % n_points
                grad[i] = 1.0 - h_vec[idx]
                
        return max_overlap_val, grad
    
    # Iterative improvement using global optimization
    def obj_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Renormalize to sum = target_sum exactly
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return evaluate_overlap(h_vec)
        
    # Use gradient methods to update step
    def update_h_grad_descent(h_vec, max_iter=100):

        for it in range(max_iter):
            # Evaluate current objective and gradient
            val, grad = objective_and_gradient(h_vec)
            if val < 1e-10:

            # Take step
            step_size = 0.01  # Maybe adapt
            update = step_size * grad
            
            # Clip and project
            h_new = h_vec - update
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            if np.allclose(h_new, h_vec, rtol=1e-6):
                break
            h_vec = h_new
            
        return h_vec
    
    # Maximize iterations based on time
    max_iter = 120  # Prevent large outage
    iter_count = 0
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:

        h_new = update_h_grad_descent(h, max_iter=3)
        c5_new = evaluate_overlap(h_new)
        if c5_new < best_c5:
            # Go to better
            best_h = h_new.copy()
            best_c5 = c5_new
        # Random shift + small perturb as extra
        h_pert = h_new.copy()
        idx = np.random.randint(0, n_points)
        # Add small noise to a random point
        h_pert[idx] = np.clip(h_pert[idx] + np.random.normal(0, 0.01), 0, 1)
        # Renormalize
        h_pert = h_pert * target_sum / np.sum(h_pert)
        c5_pert = evaluate_overlap(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
        h = h_new.copy()
        iter_count += 1
        
        if (time.time() - start_time) > budget_s - 1:
            break
    
    return (list(best_h), best_c5, n_points)
