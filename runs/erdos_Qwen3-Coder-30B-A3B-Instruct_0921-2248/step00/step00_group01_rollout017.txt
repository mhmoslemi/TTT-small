import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations: h * (1 - h_shifted)
    correlations = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of all correlations, scaled by dx
    max_overlap = np.max(correlations) * dx
    return max_overlap
