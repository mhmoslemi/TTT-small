import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Fallback: uniform distribution
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Ensure sum(h) == n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    remaining_time = budget_s - (time.time() - start_time)
    
    # Try different configurations
    configs = [
        (n_points, "ECOS"),
        (min(100, n_points + 30), "ECOS"),
        (n_points, "SCIP"),
    ]
    
    for n, solver in configs:
        if time.time() - start_time > budget_s - 1:
            break
            
        if n != n_points:
            # Resize h0 to n_points
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
            h0 = h0 * (n / 2) / np.sum(h0)
            n_points = n
            
        try:
            prob = solve_continuous_erdos(n_points, h0, solver)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = prob.variables()[0].value
                h_sol = np.clip(h_sol, 0, 1)
                # Normalize sum to n_points/2
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                c5_sol = evaluate_erdos_solution(h_sol)
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
            elif prob.status == "infeasible":
                # Adjust a bit and retry
                h0 = np.clip(h0 + np.random.normal(0, 0.01, n_points), 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
        except Exception as e:
            continue
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, len(best_h))

def solve_continuous_erdos(n_points, h0, solver_name="ECOS"):
    # Create variables
    h = cp.Variable(n_points)
    C = cp.Variable()

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C >= 0
    ]

    dx = 2.0 / n_points

    # For each shift, add overlap constraint
    for k in range(n_points):
        # Compute overlap for shift k
        overlap = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap += h[i] * (1 - h[j])
        constraints.append(overlap * dx <= C)

    # Objective
    objective = cp.Minimize(C)

    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SOLVER_MAP[solver_name] if solver_name in cp.SOLVER_MAP else cp.ECOS, 
                   verbose=False, max_iters=10000)
    except:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=10000)
        
    return prob
