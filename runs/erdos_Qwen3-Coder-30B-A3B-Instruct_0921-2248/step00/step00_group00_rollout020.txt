def evaluate_erdos_solution(h):
    # h is a numpy array representing the step function
    # compute the maximum overlap integral with the given h
    n = len(h)
    dx = 2.0 / n
    # correlate h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
