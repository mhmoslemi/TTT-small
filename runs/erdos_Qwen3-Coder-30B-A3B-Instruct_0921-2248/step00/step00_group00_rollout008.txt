import numpy as np
from scipy import correlate
import time

def evaluate_h(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all cross-correlations for k from 0 to 2*dx
    corr = correlate(h, 1 - h, mode="full")
    # Only consider the valid shifts
    # k = 0 to 2*delta, step = delta
    max_k = n_points
    max_corr = -1.0
    for i in range(max_k):
        val = corr[i] * dx
        if val > max_corr:
            max_corr = val
    return max_corr

def is_valid_h(h, n_points):
    dx = 2.0 / n_points
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1) < 1e-8)

def generate_move(h, n_points):
    # Generate a random interval flip: 1 or 3 points
    length = np.random.choice([1, 3])
    start = np.random.randint(0, n_points - length + 1)
    new_h = h.copy()
    for i in range(start, start + length):
        new_h[i] = 1 - new_h[i]
    return new_h
