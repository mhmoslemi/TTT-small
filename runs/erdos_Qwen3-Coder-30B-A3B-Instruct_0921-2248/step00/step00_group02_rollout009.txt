import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the maximum overlap integral for the given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap is the maximum of the correlation (times dx)
    # dx = 2.0 / len(h)
    dx = 2.0 / len(h)
    return np.max(corr) * dx
