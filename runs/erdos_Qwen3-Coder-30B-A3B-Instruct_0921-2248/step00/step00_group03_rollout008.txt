import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.001
    alpha_decay = 0.99
    tolerance = 1e-6
    best_c5 = float('inf')
    best_h = None
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5  # uniform initialization
    
    h = h / np.sum(h) * (n_points / 2)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        # Extract the relevant part (shifts from -n+1 to n-1)
        shifts = np.arange(-n_points + 1, n_points)
        # Only consider positive shifts which are valid
        valid_shifts = shifts[n_points-1:]
        overlaps = corr[n_points-1:]
        
        # Find the maximum overlap
        k_max = valid_shifts[np.argmax(overlaps)]
        max_overlap = np.max(overlaps)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient w.r.t h for the worst case shift
        # Overlap_k = sum h[i] * (1 - h[(i+k) % n])
        # gradient[i] = d/dh[i] overlap_k = 1 - h[(i+k) % n] - h[i]
        gradient_signs = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient_signs[i] = 1 - h[j] - h[i]
        
        # Update h
        h -= alpha * gradient_signs
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum to n_points/2
        h = h / np.sum(h) * (n_points / 2)
        
        # Adaptive step size
        if iteration % 50 == 0:
            alpha *= alpha_decay
            
    # Final evaluation
    if best_h is not None:
        h = best_h
    corr = correlate(h, 1 - h, mode='full')
    shifts = np.arange(-n_points + 1, n_points)
    valid_shifts = shifts[n_points-1:]
    overlaps = corr[n_points-1:]
    final_c5 = np.max(overlaps) * dx
    
    # Ensure no NaN or invalid values
    if np.isnan(final_c5) or not np.isfinite(final_c5):
        final_c5 = 1.0
        
    return (h.tolist(), final_c5, n_points)
