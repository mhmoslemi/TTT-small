import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound using the given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all overlaps using cross-correlation
    overlaps = correlate(h_values, 1 - np.array(h_values), mode='full')
    # Only consider the valid overlaps (not the full correlation)
    # The valid ones are from index n_points-1 to 2*n_points-2
    valid_overlaps = overlaps[n_points-1:]
    max_overlap = np.max(valid_overlaps) * dx
    return max_overlap

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create uniform

    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize the best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Time limit
    time_limit = start_time + budget_s
    
    # Try different configurations
    for iteration in range(1000):
        if time.time() > time_limit:
            break
            
        # We'll try a local optimization approach by perturbing the h values
        # to see if we can improve
        h_pert = h.copy()
        # Make small changes
        for i in range(n_points):
            if np.random.rand() < 0.1:  # 10% chance to perturb
                delta = np.random.normal(0, 0.02) if np.random.rand() < 0.5 else -np.random.normal(0, 0.02)
                h_pert[i] += delta
                h_pert[i] = np.clip(h_pert[i], 0, 1)
        
        # Re-normalize
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        # Check
        c5_new = evaluate_erdos_solution(h_pert)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_pert.copy()
            if best_c5 < 0.3807:
                break  # Early stopping if target achieved

    # Try optimization with CVXPY for better solution
    try:
        if time.time() > time_limit:
            return (best_h.tolist(), best_c5, len(best_h))
        n = n_points
        dx = 2.0 / n
        
        # CVXPY variables
        h_vars = cp.Variable(n)
        C = cp.Variable()
        
        # Constraints
        constraints = []
        # h in [0, 1]
        constraints += [h_vars >= 0, h_vars <= 1]
        # sum(h) = n_points / 2
        constraints += [cp.sum(h_vars) == n / 2]
        # C >= overlap_k for all k
        # We compute overlaps manually
        for k in range(n):
            # overlap_k = sum h[i] * (1 - h[(i + k) % n])
            overlap = 0
            for i in range(n):
                overlap += h_vars[i] * (1 - h_vars[(i + k) % n])
            constraints += [overlap * dx <= C]
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Try to solve
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_vars.value
            if h_opt is not None:
                # Normalize manually
                h_opt = h_opt * (n / 2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                
    except Exception as e:
        pass  # Skip if CVXPY fails

    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
