import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a symmetric tent function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 200
        h = np.array([min(i, 2*n_points - i - 2) for i in range(2*n_points)]).astype(float)
        h = h[:n_points]
        h = h / np.sum(h) * (n_points / 2)
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Adaptive gradient descent with frequency domain enhancements
    alpha = 0.01
    max_iter = 10000
    no_improve_count = 0
    max_no_improve = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute convolution in frequency domain
        H = fft.fft(h)
        G = 1 - h
        GH = fft.fft(G)
        conv_result = fft.ifft(H * np.conj(GH))
        
        # Find maximum overlap and its location
        max_val = np.real(conv_result).max()
        max_idx = np.real(conv_result).argmax()
        
        # Convert back to proper time domain indices
        k = max_idx - n_points + 1 if max_idx >= n_points else max_idx
        
        # Project gradient to decrease overlap
        # Compute gradient for each point
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient w.r.t. h[i]: derivative of integral
            # The overlap at shift k is approximated by terms around i
            # Here we approximate using delta functions around the peak
            j = (i - k) % n_points
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
            
            # Other terms in the overlap (simplified gradient)
            if 0 <= i < n_points:
                j = (i + k) % n_points
                if 0 <= j < n_points:
                    grad[i] -= h[j] * dx
        
        # Update step with projection
        h_new = h - alpha * grad
        
        # Project to constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint with exact projection
        current_sum = np.sum(h_new)
        h_new = h_new * target_sum / current_sum
        
        # Update with new solution
        h = h_new
        
        # Compute C5
        C5 = compute_C5(h, dx)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Adaptive learning rate
        if no_improve_count > 10:
            alpha *= 0.95
        if no_improve_count > max_no_improve:
            break
            
    return (best_h, best_C5, n_points)

def compute_C5(h, dx):
    """Compute maximum overlap as defined in problem."""
    h_rev = 1 - h
    # Use full convolution to handle circular shifts
    conv = np.correlate(h, h_rev, mode="full")
    # Convert conv values to C5 = max_k integral for shifted versions
    return np.max(conv) * dx
