import numpy as np
import time
from scipy import fftpack
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial if available, otherwise create symmetric tent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 200
        h = np.linspace(0, 1, n_points//2)
        h = np.concatenate([h, h[::-1]])
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure proper sum
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based update with frequency domain enhanced optimization
    max_iter = 10000
    alpha = 0.01
    patience = 50
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps in frequency domain
        H = fftpack.fft(h)
        G = 1 - h
        GH = fftpack.fft(G)
        
        # Overlap in frequency domain (convolution)
        overlap_freq = np.real(fftpack.ifft(H * np.conj(GH)))
        # Only consider full convolution (valid mode)
        overlap_full = overlap_freq[:n_points]  # Valid values
        max_overlap_idx = np.argmax(overlap_full)
        max_overlap_val = overlap_full[max_overlap_idx]
        
        # Gradient-based update targeting worst overlap
        # Sample the gradient of overlap w.r.t. h
        grad = np.zeros(n_points)
        # Since overlap = h * (1-h) shifted, gradient is 1 - 2*h at shift locations
        # Approximate error direction by solving for dC5/dh = 0
        # Using direct adaptation, moving in direction of gradient of overlap at max point
        
        # Create shifted versions
        shift = max_overlap_idx
        h_shifted = np.roll(h, shift)
        h_shifted_back = np.roll(h, -shift)  # useful for backshift
        
        # Approximate the gradient in time domain
        for i in range(n_points):
            # We are computing d/dh_i int h(x)(1-h(x+k))dx
            # This is -(1-h_i) if shifting by k from i to i+k
            # Backwards difference approximation
            # For k = shift, reward/penalize h_i based on h_i+shift
            grad_i = 0
            for j in range(n_points):
                adj_idx = (j + shift) % n_points
                grad_i += (1 - h[adj_idx]) * (j == i)
            grad[i] = grad_i

        # Normalize gradient to standard scale        
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-8:
            grad = grad / grad_norm * alpha

        # Update h
        h_new = h - grad
        
        # Apply constraints: clip h to [0,1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if new h is better
        C5_new = compute_c5(h_new, dx)
        if C5_new < best_c5:
            best_c5 = C5_new
            best_h = h_new.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive step adjustment
        if patience_counter > 20:
            patience_counter = 0
            alpha *= 0.8
            if alpha < 1e-6:
                alpha = 1e-6
                
        h = h_new.copy()
        
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Full convolution approach
    # h to compute overlaps
    result = np.correlate(h, 1 - h, mode='full')
    # Only take from valid (non-zero padding) section
    full_len = len(result)
    valid_start = len(h) - 1
    valid_end = valid_start + len(h)
    result_valid = result[valid_start:valid_end]
    # Take max and scale by dx
    return np.max(result_valid) * dx
