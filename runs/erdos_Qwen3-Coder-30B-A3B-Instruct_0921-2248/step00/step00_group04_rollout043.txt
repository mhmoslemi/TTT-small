import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 42  # Use the provided initial construction size if available
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Normalize to satisfy sum(h) == target_sum
        h = h * target_sum / np.sum(h)
    else:
        # Start with a random feasible solution
        h = np.random.rand(n_points)
        # Scale to meet sum constraint
        h = h * target_sum / np.sum(h)
    
    # Function to compute C5
    def compute_C5(h_vec):
        # Compute the cross-correlation
        # np.correlate with mode='full' computes 
        # sum h[i] * (1 - h[i+k]) over all valid i
        # dilated by k from -n+1 to n-1
        correlation = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum correlation scaled by dx
        return np.max(correlation) * dx
    
    # Function to perturb h
    def perturb_h(h_vec, temp, epsilon=0.05):
        # Make a copy to modify
        new_h = h_vec.copy()
        # Choose random indices to perturb
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        # Perturb values
        for i in indices:
            delta = np.random.uniform(-epsilon, epsilon)
            new_h[i] += delta
        # Ensure bounds [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Ensure sum matches target using a simple redistribution
        current_sum = np.sum(new_h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            # Evenly redistribute the difference
            evenly_distributed = diff / len(new_h)
            new_h = np.clip(new_h + evenly_distributed, 0, 1)
            # Adjust again to enforce exact sum
            adjustment = target_sum - np.sum(new_h)
            if abs(adjustment) > 1e-10:
                for i in range(n_points):
                    if adjustment < 0:
                        reduction = min(-adjustment, new_h[i])
                        new_h[i] -= reduction
                        adjustment += reduction
                    else:
                        increase = min(adjustment, 1.0 - new_h[i])
                        new_h[i] += increase
                        adjustment -= increase
                    if abs(adjustment) < 1e-10:
                        break
        return new_h

    # Simulated Annealing
    start_time = time.time()
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        # Generate new candidate solution
        new_h = perturb_h(current_h, temp)
        new_C5 = compute_C5(new_h)
        
        # Accept with probability
        delta = new_C5 - current_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = new_h
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = current_C5
        
        # Cool temperature
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
    
    return (best_h.tolist(), best_C5, n_points)
