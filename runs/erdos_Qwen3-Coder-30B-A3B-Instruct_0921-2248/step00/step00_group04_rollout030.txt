import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lag = 2.0
    lag_step = 0.1
    lag_indices = np.arange(-max_lag, max_lag + lag_step/2, lag_step)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    else:
        # Create a simple starting point
        h_start = np.ones(n_points) * 0.5
    
    # Variables for optimization
    h = cp.Variable(n_points)
    C5 = cp.Variable()

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) * dx == 1,
        C5 >= 0
    ]
    
    # Add overlap constraints
    for k in lag_indices:
        if time.time() - start_time > budget_s * 0.95:
            break
        lag_int = int(round(k / lag_step))
        if abs(lag_int * lag_step - k) > 1e-10:
            continue
        # For discrete lag handling
        lag_shift = lag_int
        shifted_h = cp.hstack([h[-lag_shift:], h[:-lag_shift]]) if lag_shift > 0 else cp.hstack([h[-lag_shift:], h[:-lag_shift]])
        overlap = cp.sum(cp.multiply(h, 1 - shifted_h)) * dx
        constraints.append(overlap <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = C5.value
        else:
            h_opt = h_start
            c5_bound = evaluate_erdos_solution(h_opt)
    except:
        h_opt = h_start
        c5_bound = evaluate_erdos_solution(h_opt)
        
    # Final evaluation
    final_bound = evaluate_erdos_solution(h_opt)
    
    return (h_opt, final_bound, n_points)
