import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Compute C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode="full")
    # Extract the relevant portion (k=0 to n-1)
    c5 = np.max(corr[len(h)-1:])
    return c5 * dx

def initial_h_values():
    """Generate initial h values from the parent construction if available"""
    try:
        return initial_h_values
    except NameError:
        # If no parent, return a reasonable starting point
        n = 45
        h = np.ones(n) * 0.5
        # Make it a step function
        h[:n//2] = 1.0
        h[n//2:] = 0.0
        # Normalize integral to 1
        h = h * 2.0 / np.sum(h)
        return h

def is_feasible(h):
    """Check if h satisfies constraints"""
    n = len(h)
    dx = 2.0 / n
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1.0) < 1e-8)

def propose_move(h, n_points):
    """Propose a new h by flipping a small interval"""
    # Select interval of length 1 or 3 to flip
    length = np.random.choice([1, 3])
    start = np.random.randint(0, n_points - length + 1)
    new_h = h.copy()
    # Flip values in interval
    for i in range(start, start + length):
        new_h[i] = 1.0 - new_h[i]
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Starting with parent size if exists
    h = initial_h_values()
    dx = 2.0 / n_points
    assert len(h) == n_points, "Size mismatch"
    
    # Ensure feasibility
    if not is_feasible(h):
        # Reconstruct if needed
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * 2.0 / np.sum(h)
    
    c5_current = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5_current
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.999
    max_iter = int(budget_s * 1000)  # Roughly estimated
    iteration = 0
    
    while iteration < max_iter:
        # Check time budget
        if time.time() - start_time > budget_s - 0.1:  # Give a bit of buffer
            break
            
        # Propose move
        h_proposed = propose_move(h, n_points)
        
        # Ensure feasibility by projection if needed
        if not is_feasible(h_proposed):
            # Project to feasible region
            h_proposed = np.clip(h_proposed, 0, 1)
            # Adjust to satisfy integral constraint
            current_sum = np.sum(h_proposed) * dx
            target_sum = 1.0
            scale = target_sum / current_sum
            h_proposed *= scale
            
        if not is_feasible(h_proposed):
            # Try just reprojecting
            h_proposed = np.clip(h_proposed, 0, 1)
            # Adjust sum to match constraint
            if abs(np.sum(h_proposed) * dx - 1.0) > 1e-6:
                h_proposed = h_proposed * (1.0 / (np.sum(h_proposed) * dx)) * 1.0
                
        c5_proposed = evaluate_erdos_solution(h_proposed)
        
        # Metropolis acceptance criterion
        if c5_proposed < c5_current or np.random.rand() < np.exp((c5_current - c5_proposed) / T):
            h = h_proposed
            c5_current = c5_proposed
        
        # Update best if needed
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        # Cool down temperature
        T *= alpha
        iteration += 1
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    # Ensure final solution is valid
    assert is_feasible(best_h), "Final solution infeasible"
    return (best_h.tolist(), float(final_c5), int(n_points))
