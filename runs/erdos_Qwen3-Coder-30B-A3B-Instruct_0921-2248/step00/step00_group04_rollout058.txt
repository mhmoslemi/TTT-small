import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with a reasonable size
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.random.random(n_points)
        h = h / np.sum(h) * target_sum
    
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    h = h / np.sum(h) * target_sum
    
    # Helper to compute C5
    def compute_C5(h_vec):
        # Compute convolution
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum value among all lags
        max_val = np.max(conv)
        return max_val * dx
    
    # SA parameters
    T = 1.0
    cooling_rate = 0.9995
    min_T = 1e-6
    max_iter = 1000000
    iter_count = 0
    start_time = time.time()
    
    # Best solution tracking
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5
    
    # Main optimization loop
    while iter_count < max_iter and (time.time() - start_time) < budget_s:
        iter_count += 1
        
        # Perturb h
        h_new = current_h.copy()
        
        # Choose random indices to perturb
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Perturb values
        eps = T * 0.1
        for i in indices:
            delta = np.random.uniform(-eps, eps)
            h_new[i] += delta
        
        # Ensure bounds and sum constraint
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to keep sum correct
        current_sum = np.sum(h_new)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # Distribute difference among all elements
            h_new += diff / n_points
        
        # Clamp again after adjustment
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint again
        h_new = h_new / np.sum(h_new) * target_sum
        
        # Compute new C5
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        delta_C5 = new_C5 - current_C5
        if delta_C5 < 0 or np.random.random() < np.exp(-delta_C5 / T):
            current_h = h_new
            current_C5 = new_C5
            
            # Update best if improved
            if current_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = current_C5
        
        # Cool temperature
        T = max(min_T, T * cooling_rate)
    
    return (best_h.tolist(), best_C5, n_points)
