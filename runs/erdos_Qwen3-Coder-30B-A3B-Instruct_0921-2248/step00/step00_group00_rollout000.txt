def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    # Compute full cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation scaled by dx
    dx = 2.0 / len(h)
    return np.max(corr) * dx
