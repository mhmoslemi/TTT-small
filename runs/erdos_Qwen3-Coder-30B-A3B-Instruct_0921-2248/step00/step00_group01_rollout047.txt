import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap integral
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a reasonable starting point
        n_points = 100
        h_init = np.ones(n_points) * 0.5
        h_init[10:20] = 1.0  # Some spikes
        h_init[40:50] = 0.0  # Some zeros
        h_init[70:80] = 1.0  # Some more spikes
    
    dx = 2.0 / n_points
    
    # Normalize initial function to have integral = 1
    h = h_init.copy()
    h_sum = np.sum(h) * dx
    h = h * (1.0 / h_sum)
    
    # Bounds: h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function to minimize
    def objective(h_vec):
        # Correlation with shifted version
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Optimization parameters
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # SGD-style optimization
    # SLSQP works better on this problem
    approximation_order = 1000  # Try to estimate convergence
    for it in range(approximation_order):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize
        try:
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_new = res.x.copy()
                
                # Project to feasible space
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new) * dx
                h_new = h_new * (1.0 / h_new_sum)
                
                c5_new = evaluate_erdos_solution(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
                h = h_new.copy()
            else:
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h) * dx
                h = h * (1.0 / h_sum)
                
        except Exception as e:
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h) * dx
            h = h * (1.0 / h_sum)
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
