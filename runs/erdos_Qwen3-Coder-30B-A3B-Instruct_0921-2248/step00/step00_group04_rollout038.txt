import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the C5 bound for given h_values"""
    h = np.array(h_values)
    dx = 2.0 / n_points
    # Compute all correlations with 1-h
    corr = correlate(h, 1-h, mode="full")
    # Extract the valid range
    mid = len(corr) // 2
    valid_corr = corr[mid - n_points + 1:mid + n_points]
    return np.max(valid_corr) * dx

    constraints = [

        h <= 1,
        cp.sum(h) * dx == 1,
        C5 >= 0
    ]
    
    # Lag set: coarse initial sampling
    lags = np.arange(-2, 2 + 0.1, 0.2)
    lags = lags[1:-1]  # remove the 0 lag since it doesn't contribute to the problem
    
    # Add constraints for all lags
    for k in lags:
        # Compute shift index
        shift_idx = int(k / dx)
        if shift_idx >= 0:
            h_shifted = h[shift_idx:] + np.zeros(n_points)  # Padding with zeros

            h_shifted = np.zeros(n_points) + h[:shift_idx]  # Padding with zeros

        valid_h = valid_h[:n_points]
        
        # Constraint for this lag
        shift_constraint = cp.sum(h * (1 - valid_h)) * dx <= C5
        constraints.append(shift_constraint)

    objective = cp.Minimize(C5)

    prob = cp.Problem(objective, constraints)
