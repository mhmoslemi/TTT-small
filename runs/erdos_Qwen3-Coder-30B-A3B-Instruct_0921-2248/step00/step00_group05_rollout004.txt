import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Suitable size for fast optimization
    
    # Initialize h: use initial_h_values if available, else create symmetric tent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create symmetric tent function
        h = np.zeros(n_points)
        for i in range(n_points):
            k = i - n_points // 2
            h[i] = max(0, 1.0 - abs(k) / (n_points / 2))
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    max_iter = 10000
    tol = 1e-6
    
    def compute_C5(h_vec):
        # Make h_vec a numpy array if it isn't
        h_vec = np.array(h_vec)
        # Compute correlation for all shifts using numpy's correlate function
        # Return maximum of all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def objective(h_vec):
        return compute_C5(h_vec)
    
    # Constraint: sum is exactly n_points / 2
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for each variable
    bounds = [(0.0, 1.0)] * n_points
    
    # Optimization using SLSQP
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol}, 
                      callback=lambda x: None)
 
    if result.success:
        h_optimized = result.x
        c5_result = compute_C5(h_optimized)
    else:
        # Fallback to original h if optimization failed
        h_optimized = h
        c5_result = compute_C5(h_optimized)
    
    # Ensure constraints are met
    h_optimized = np.clip(h_optimized, 0, 1)
    h_optimized = h_optimized / np.sum(h_optimized) * (n_points / 2)
    
    return (h_optimized, c5_result, n_points)
