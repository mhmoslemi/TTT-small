import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a reasonable initial guess
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[0] = 0.45
        h[1] = 0.55
        h[-1] = 0.45
        h[-2] = 0.55
    
    # Ensure sum constraint
    dx = 2.0 / n_points
    h *= (n_points / 2.0) / np.sum(h)
    
    # Evaluate initial C5
    def compute_C5(h_values):
        # Cross-correlate h with (1-h) and find maximum
        h = np.array(h_values)
        h_rev = h[::-1]
        corr = correlate(h, 1 - h_rev, mode='full')
        # Take only the relevant part (max shift is 2)
        # The full correlation has length 2*n-1
        center = len(corr) // 2
        # Get all correlations for shifts in [0, 2] which means we look at indices near center
        # We're mostly interested in the overlaps at different k values in [0, 2]  
        # But since convolution gives overlaps with different lags, we map those lags to k
        # The lag in correlation output is offset by (n-1)
        # So for lag = 0: h[0] matches 1-h[0], lag = 1: h[1] matches 1-h[0], etc.
        # And for lags k in [0, n-1], the actual shift = k - (n-1), but we need rational shifts over [0,2]
        # Proper evaluation is to take full correlation and extract the required values
        
        # Let's compute for a range of k=0 to 2 in steps of 0.1 (say, resolution)
        # This is in fact equivalent to max of all the correlations 
        # since those come from different shifts, and any shift in [0,2] fits in the array
        # But computing manually via overlap integrals per k is exact.
        # For the overlaps, max over all lags
        corr_val = np.max(corr) * dx
        return corr_val

    # Initial evaluation
    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()

    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.9995
    min_T = 1e-6
    max_iter = int(1e6)
    
    # Timing control
    start_time = time.time()
    
    # Number of perturbations per step
    num_perturbations = max(1, min(5, n_points // 10))

    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Create new solution by perturbing h
        new_h = h.copy()
        alpha = 0.1 * T
        
        # Choose a few random indices to perturb
        indices = np.random.choice(n_points, size=min(num_perturbations, n_points), replace=False)
        
        for i in indices:
            # Add some small random change
            delta = np.random.uniform(-alpha, alpha)
            new_h[i] += delta
            
        # Ensure bounds and constraint
        new_h = np.clip(new_h, 0, 1)
        
        # Re-normalize to satisfy sum(h) = n/2
        scale = (n_points / 2.0) / np.sum(new_h)
        new_h *= scale
        
        # Evaluate the new solution
        new_C5 = compute_C5(new_h)
        
        # Accept or reject the new solution
        delta_C5 = new_C5 - current_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = new_h
            current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        # Update temperature
        T = max(min_T, T * cooling_rate)
        
    return (best_h.tolist(), best_C5, n_points)
