import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize

    else:

        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (mid + 1)
        h = h / np.sum(h) * target_sum
        
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Gradient-based updates with frequency domain analysis
    max_iter = 1000
    alpha = 0.01
    patience = 50
    no_improve_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        C5 = compute_C5(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Compute C5 over all shifts through convolution
        conv_h_1mh = np.correlate(h, 1 - h, mode='full')
        conv_h_1mh *= dx
        max_shift = np.argmax(conv_h_1mh)
        # For max_shift, we compute partial derivatives to adjust h
        # For simplicity, use basic gradient update on representative shifts
        
        # Gradient approximation through small perturbations
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            if h_plus[j] > 1: h_plus[j] = 1.0
            if h_plus[j] < 0: h_plus[j] = 0.0
            C5_plus = compute_C5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            if h_minus[j] > 1: h_minus[j] = 1.0
            if h_minus[j] < 0: h_minus[j] = 0.0
            C5_minus = compute_C5(h_minus, dx)
            
            grad[j] = (C5_plus - C5_minus) / (2 * eps)

        h -= alpha * grad
        # Clip values
        h = np.clip(h, 0, 1)
        # Adjust sum
        h = h / np.sum(h) * target_sum

            alpha = min(0.05, alpha * 1.05)

            if no_improve_count >= patience:
                break
    
    # Final validation
    final_C5 = compute_C5(best_h, dx)
    return (best_h.tolist(), final_C5, n_points)

def compute_C5(h, dx):
    # Compute the maximum overlap integral
    conv = np.correlate(h, 1 - h, mode='full')
    conv *= dx
    return np.max(conv)
