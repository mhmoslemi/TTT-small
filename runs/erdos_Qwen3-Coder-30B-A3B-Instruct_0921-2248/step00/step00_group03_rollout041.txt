    h = cp.Variable(n_points, name="h")
    C = cp.Variable(name="C")

    constraints = []

    constraints.extend([h >= 0, h <= 1])

    constraints.append(cp.sum(h) == n_points / 2)
    
    # Overlap constraints
    for k in range(-max_shift, max_shift + 1):

        i_indices = np.arange(n_points)

        if k >= 0:

            expr = cp.sum(cp.multiply(h[i_indices[:-k]], 1 - h[j_indices[k:]]))

            expr = cp.sum(cp.multiply(h[i_indices[-k:]], 1 - h[j_indices[:-k]]))
        
        constraints.append(expr <= C / dx)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
