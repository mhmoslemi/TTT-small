def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # The full correlation array has length 2*n - 1
    # We only care about the overlaps, so we take the max
    c5 = np.max(corr) * dx
    return c5

def generate_candidate(h, n_points):
    """
    Generate a new candidate solution by flipping an interval.
    """
    h_new = h.copy()
    # Select a random interval
    length = np.random.choice([1, 3])
    start = np.random.randint(0, n_points - length + 1)
    for i in range(start, start + length):
        h_new[i] = 1.0 - h_new[i]
    return h_new

def ensure_feasible(h, n_points):
    """Adjust h to ensure it's feasible (bounds and integral constraint)."""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust to maintain integral constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-8:
        # Simple linear adjustment
        factor = target_sum / current_sum
        h = h * factor
        h = np.clip(h, 0, 1)
    return h

    T0 = 0.1

        # Metropolis acceptance criterion
        if c5_new < c5_old:

        # Update best solution

            best_c5 = c5_old
            
        # Temperature cooling
        T = T * alpha
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
