import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the maximum overlap integral for given h."""
    dx = 2.0 / len(h)
    # Compute full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # The overlap integral for lag k is conv[k + len(h) - 1]
    # We're interested in the maximum among all valid lags
    # For a signal of length n, valid lags are from -(n-1) to (n-1)
    max_overlap = max(conv)
    return max_overlap * dx

def initialize_h(n_points):
    """Initialize h based on initial_h_values or create a simple construction."""
    try:
        h = np.array(initial_h_values)
        if len(h) == n_points:
            return h.copy()
    except NameError:
        pass
    # Create a simple binary-like construction
    h = np.zeros(n_points)
    mid = n_points // 2
    # Assign values to make sum = n_points / 2, with some variation
    h[:mid] = 1
    excess = sum(h) - n_points / 2
    if excess > 0:
        h[mid] = max(0, 1 - excess)
    return h

def perturb_h(h, delta=0.1):
    """Generate a perturbed version of h by making small adjustments."""
    h_new = h.copy()
    n = len(h)
    
    # Identify overlaps that are near the current maximum
    dx = 2.0 / n
    conv = correlate(h, 1 - h, mode='full')
    max_overlap = max(conv)
    
    # Look for the top 10% of contributing lags
    thresholds = np.sort(conv)[-max(1, n // 10):]
    high_lags = [i for i, val in enumerate(conv) if val >= thresholds[0]]
    
    # Now for each high lag, find x where h[x]=1 and h[x+k]=0
    # Try small adjustments
    
    adjust_candidates = []
    for k in high_lags[:min(20, len(high_lags))]:
        # k is an index in full convolution, adjust for actual lag
        lag = k - (n - 1)
        if lag > 0:
            # We want to reduce overlap when we compute h[i] * (1 - h[i+lag])
            # Check for 1 and 0 values
            for i in range(n - lag):
                if h[i] >= 0.99 and h[i + lag] <= 0.01:
                    adjust_candidates.append((i, i + lag, -delta, delta))
        elif lag < 0:
            # We want to reduce overlap when we compute h[i] * (1 - h[i-lag])
            for i in range(-lag, n):
                if h[i] >= 0.99 and h[i + lag] <= 0.01:
                    adjust_candidates.append((i, i + lag, -delta, delta))
        else:
            # Zero lag
            for i in range(n):
                if h[i] >= 0.99:
                    adjust_candidates.append((i, i, -delta, delta))
    
    # Evaluate adjustments
    best_h = h_new.copy()
    best_c5 = evaluate_c5(h_new)
    actual_updates = []
    
    # Consider at most 5 adjustments
    for i, (x1, x2, d1, d2) in enumerate(adjust_candidates[:5]):
        h_cand = h_new.copy()
        h_cand[x1] = max(0, min(1, h_cand[x1] + d1))
        h_cand[x2] = max(0, min(1, h_cand[x2] + d2))
        # Maintain constraint on sum
        diff = (d1 + d2)
        h_cand[-1] -= diff  # Adjust last element to enforce sum
        
        # Clamp the result
        h_cand = np.clip(h_cand, 0, 1)
        # Re-scale if needed to keep sum correct
        if abs(np.sum(h_cand) - n/2) > 1e-10:
            factor = (n / 2) / np.sum(h_cand)
            h_cand = h_cand * factor
            
        c5 = evaluate_c5(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()
            actual_updates.append((x1, x2, d1, d2))
    
    if actual_updates:
        h_new = best_h
    return h_new, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 80  # Experiment with a reasonable size for multiple iterations
    
    # Initialize h, might be using initial_h_values if available
    h = initialize_h(n_points)
    
    # Evaluate initial solution
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters for optimization
    max_iter = 1000
    max_no_improve = 20
    
    no_improve_count = 0
    last_best = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Perturb h
        next_h, new_c5 = perturb_h(h)
        
        # If improvement, move to new point; otherwise, continue
        if new_c5 < current_c5:
            h = next_h
            current_c5 = new_c5
            no_improve_count = 0
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                last_best = best_c5
        else:
            no_improve_count += 1
        if no_improve_count >= max_no_improve:
            break
    
    return (best_h.tolist(), best_c5, n_points)
