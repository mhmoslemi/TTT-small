from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    target_time = time.time() + budget_s
    
    # Objective function
    def obj(h_vec):
        h_vec = np.array(h_vec)
        # Ensure bounds
        h_vec = np.clip(h_vec, 0, 1)
        # Ensure integral constraint
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        
        # Compute overlap
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraint function
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Bounds
    bnds = [(0, 1) for _ in range(n_points)]
    
    # Optimize
    best_h = h.copy()
    best_c5 = obj(best_h)
    
    # Get initial C5
    current_c5 = best_c5
    
    # Optimizer iteration
    iteration = 0
    last_improvement = 0
    
    while time.time() < target_time and iteration - last_improvement < 50:
        iteration += 1
        # Slightly perturb h
        new_h = best_h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.3:
                delta = np.random.normal(0, 0.03)
                new_h[i] = np.clip(new_h[i] + delta, 0, 1)
        
        # Normalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate
        c5 = obj(new_h)
        if c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = c5
            last_improvement = iteration
        
        # Gradient-based step with trust region approach
        if iteration % 5 == 0:
            # Simple gradient descent with different step size
            step_size = 0.01
            grad = np.zeros(n_points)
            h_clip = np.clip(best_h, 1e-6, 1 - 1e-6)
            for shift in range(n_points):
                corr = np.correlate(h_clip, 1 - h_clip, mode='full')
                k = shift if shift < n_points else shift - n_points
                if k >= 0:
                    grad += (1 - h_clip[k:]) * np.concatenate([np.ones(len(h_clip[k:])), np.zeros(k)])
                else:
                    grad += -h_clip[:k] * np.concatenate([np.zeros(-k), np.ones(len(h_clip[-k:]))])
            # Normalize gradient
            grad = grad / (np.sum(np.abs(grad)) + 1e-10)
            # Update
            temp_h = best_h - step_size * grad
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
            temp_c5 = obj(temp_h)
            if temp_c5 < best_c5:
                best_h = temp_h.copy()
                best_c5 = temp_c5
                last_improvement = iteration
    
    # Final evaluation
    final_c5 = obj(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
