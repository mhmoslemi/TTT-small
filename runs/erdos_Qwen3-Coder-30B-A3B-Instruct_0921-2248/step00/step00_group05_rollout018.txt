import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Create a symmetric tent function
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x <= 1.0:
                h[i] = x
            else:
                h[i] = 2.0 - x
        # Normalize
        h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent parameters
    max_iter = 10000
    alpha = 0.01
    tol = 1e-6
    last_c5 = best_c5 + 10
    patience = 0
    max_patience = 200
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Find the worst shift
        corr = signal.correlate(h, 1 - h, mode='full')
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from shift max_shift
            j = i + max_shift
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
            if 0 <= i + max_shift < n_points:
                grad[i] -= h[i + max_shift] * dx
        
        # Update step
        h = h - alpha * grad
        
        # Project
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            h = h * target_sum / current_sum
            
        # Adaptive step size
        if abs(c5 - last_c5) < tol:
            patience += 1
            if patience > max_patience:
                alpha *= 0.9
                patience = 0
        else:
            patience = 0
        last_c5 = c5
        
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
