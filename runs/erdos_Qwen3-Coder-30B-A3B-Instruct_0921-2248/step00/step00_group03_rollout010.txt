import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.001
    alpha_decay = 0.99
    tolerance = 1e-6
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5  # Default initialization
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute overlaps for all shifts
        correlation = signal.correlate(h, 1 - h, mode='full')
        # The full correlation includes indices from -(n-1) to (n-1), but we're interested only in positive shifts
        # Extract from index n-1 to 2*(n-1) for shifts from 0 to n-1
        overlap_values = correlation[len(h)-1:] * dx
        
        # Find the maximum overlap index
        k_max = np.argmax(overlap_values)
        
        # Compute gradient approximation: 
        # d/dh_i (h_i * (1 - h_{i+k_max})) = 1 - h_{i+k_max}
        # and we also consider the symmetric term
        # For all terms (h_i * (1 - h_{i+k_max})), the gradient w.r.t. h_i is (1 - h_{i+k_max})
        # But we need to be careful about the periodicity. 
        # For each h[i], gradient contribution from (h[i] * (1 - h[i+k_max])) is (1 - h[(i+k_max)%n_points])
        # Actually, we take gradient of h[i] to modify it. For t=h[i] * (1 - h[(i+k_max)%n_points]),  
        # we get d/dh[i] t = (1 - h[(i+k_max)%n_points])
        # But for maximizing h[i], we may need to think in reverse direction for the sign.
        # However, to decrease C5 which is max_k overlap_k, we want to decrease the worst-case overlap.
        # So, for shift k_max, if overlap_k = \sum h[i] * (1 - h[i+k_max]), then gradient w.r.t. h_i is:
        # (1 - h[(i + k_max) % n_points]).  
        # We need to move in direction that decreases this overlap which means quite a bit of math.
        # A simpler approach is to consider gradient of the overlap function w.r.t. h:
        # \partial_{h_i} \sum_j h_j (1 - h_{j+k_max}) = (1 - h_{i+k_max})
        # To reduce overlap, we move h in negative gradient direction, that is h[i] -= alpha * (1 - h[(i+k_max)%n_points])
        
        # Compute the sign of gradient for the worse shift, to move h in direction to decrease overlap
        gradient_sign = 1 - h[(np.arange(n_points) + k_max) % n_points]
        
        # Update h 
        h = h - alpha * gradient_sign
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum to be n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute new C5
        C5 = compute_C5(h, dx)
        
        if C5 < best_C5:
            best_h = h.copy()
            best_C5 = C5
            
        # Adaptive adjustment of step size
        if iter_count % 50 == 0:
            alpha *= alpha_decay
            
    # Return best candidate
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Use scipy.signal.correlate with 'full' to compute all overlaps
    correlation = signal.correlate(h, 1 - h, mode='full')
    overlap_values = correlation[len(h)-1:] * dx  # Only positive shifts
    return np.max(overlap_values)
