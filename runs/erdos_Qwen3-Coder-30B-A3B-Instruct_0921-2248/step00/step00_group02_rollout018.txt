import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    # Compute the overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    # Using convolution to compute all shifts efficiently
    conv = signal.correlate(h, 1 - h, mode='full')
    # The overlap is computed over [0,2] so we need to adjust indexing
    # For full convolution, we get a result of length 2*n - 1
    # We actually want the overlap for shifts k in [0,2]
    # The maximum is over the middle part of the convolution
    # The shift range corresponds to k from 0 to 2 = 2*n - 1 elements
    # matches shifted indices from 0 to 2*n - 2
    # indices in [n-1, n-1 + 2*n - 1] = [n-1, 3*n - 2] correspond to k = 0 to 2
    # Actually, we need full overlap over all k, but computational effect is same.
    # for each shift k from 0 to 2 via convolution
    # Correct method: use full to get the overlap values for all shifts
    # Overlap values are at indices n-1 to 2*n-2
    overlap_values = conv[n-1:]
    max_overlap = np.max(overlap_values) * dx
    return max_overlap

    np.random.seed(seed)
    n = 45
    dx = 2.0 / n
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Get initial h
    h = get_initial_h()
    h = h.astype(float)
    
    # Initialize with the best known construction if available
    current_h = h.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 20000:
        iter_count += 1
        # Make small perturbations
        new_h = current_h.copy()

        # Flip a few elements
        num_flips = np.random.randint(1, 6)
        flip_indices = np.random.choice(n, size=num_flips, replace=False)
        for idx in flip_indices:
            new_h[idx] = 1 - new_h[idx]
        
        # Randomly adjust one value slightly (as continuous)
        adjust_idx = np.random.randint(0, n)
        delta = np.random.uniform(-0.1, 0.1)
        new_h[adjust_idx] = np.clip(new_h[adjust_idx] + delta, 0, 1)
        
        # Project to satisfy the constraint sum(h) = n/2
        current_sum = np.sum(new_h)
        target_sum = n / 2
        diff = target_sum - current_sum
        
        if abs(diff) > 1e-6:
            # Re-weight a few points
            idxs = np.random.choice(n, size=min(5, int(abs(diff) / dx + 1)), replace=False)
            for idx in idxs:
                if diff < 0 and new_h[idx] > 0:
                    decrease = min(-diff, new_h[idx])
                    new_h[idx] -= decrease
                    diff += decrease
                elif diff > 0 and new_h[idx] < 1:
                    increase = min(diff, 1 - new_h[idx])
                    new_h[idx] += increase
                    diff -= increase
                if abs(diff) < 1e-6:
                    break
        
        # Ensure bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Evaluate
        c5 = evaluate_h(new_h, dx)
        
        if c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = c5
            
        current_h = new_h.copy()
    
    return (best_h.tolist(), best_c5, n)
