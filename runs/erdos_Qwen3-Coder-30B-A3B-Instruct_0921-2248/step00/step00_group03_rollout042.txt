import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Problem setup
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    C = cp.Variable()

    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints for all shifts
    # We compute overlap for each shift k in [-n_points+1, n_points-1]
    for k in range(-n_points + 1, n_points):
        # Shifted indices
        shifted_indices = np.arange(n_points)
        if k > 0:
            shifted_indices = np.roll(shifted_indices, -k)
        elif k < 0:
            shifted_indices = np.roll(shifted_indices, -k)
        # Compute overlap for this shift
        overlap = cp.sum(cp.multiply(h, 1 - h[shifted_indices]))
        constraints.append(overlap <= C * dx)

    # Problem
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.ECOS, max_iters=200, verbose=False, 
                   time_limit=budget_s - (time.time() - start_time))
    except:
        pass

    # Postprocess the result
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_val = h.value
        c5_val = C.value
    else:
        # Fallback to initial solution if optimization failed
        h_val = h0
        c5_val = np.max(np.correlate(h_val, 1-h_val, mode="full")) * dx
        
    # Ensure feasibility and clean up
    h_val = np.clip(h_val, 0, 1)
    h_val = h_val * (n_points / 2) / np.sum(h_val)
    c5_val = np.max(np.correlate(h_val, 1-h_val, mode="full")) * dx
    
    # Final check and tiebreaks
    if np.isnan(c5_val) or np.isinf(c5_val):
        h_val = h0
        c5_val = np.max(np.correlate(h_val, 1-h_val, mode="full")) * dx
    
    return (h_val, c5_val, n_points)
