import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 20
    max_iter = 2000
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def get_neighbors(h, num_neighbors=10):
        """Generate neighbor solutions by flipping pairs of 0s and 1s"""
        neighbors = []
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            for _ in range(num_neighbors):
                # Sample a 1 and a 0 and flip them
                i1 = np.random.choice(ones)
                i0 = np.random.choice(zeros)
                h_new = h.copy()
                h_new[i1] = 0
                h_new[i0] = 1
                neighbors.append(h_new)
        
        return neighbors
    
    def local_search(h_start, max_iter=50):
        """Perform local continuous optimization around a discrete solution"""
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        con = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        best_h_local = h_start.copy()
        best_c5_local = evaluate_c5(h_start, dx)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Random perturbation
            h_pert = best_h_local + np.random.normal(0, 0.02, n_points)
            h_pert = project(h_pert)
            
            try:
                result = minimize(
                    objective,
                    h_pert,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=con,
                    tol=1e-6,
                    options={'maxiter': 30, 'disp': False}
                )
                if result.success:
                    h_opt = project(result.x)
                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5_local:
                        best_h_local = h_opt
                        best_c5_local = c5_opt
            except:
                pass
                
        return best_h_local, best_c5_local
    
    # Main Tabu search loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbors
        neighbors = get_neighbors(best_h)
        
        if not neighbors:
            continue
            
        # Evaluate all neighbors
        neighbor_c5s = [evaluate_c5(h, dx) for h in neighbors]
        
        # Find best non-tabu neighbor
        best_neighbor_idx = None
        best_neighbor_c5 = float('inf')
        
        for j, (h_n, c5_n) in enumerate(zip(neighbors, neighbor_c5s)):
            # Check if neighbor is in tabu list
            is_tabu = False
            for tabu_h, tabu_c5 in tabu_list:
                if np.allclose(h_n, tabu_h, atol=1e-6) and abs(c5_n - tabu_c5) < 1e-6:
                    is_tabu = True
                    break
            
            if not is_tabu and c5_n < best_neighbor_c5:
                best_neighbor_c5 = c5_n
                best_neighbor_idx = j
                
        if best_neighbor_idx is not None:
            # Accept the best neighbor
            new_h = neighbors[best_neighbor_idx].copy()
            
            # Update best solution
            if best_neighbor_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = best_neighbor_c5
            
            # Update tabu list
            tabu_list.append((new_h.copy(), best_neighbor_c5))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            # Occasionally perform local search around new solution
            if i % 10 == 0:
                refined_h, refined_c5 = local_search(new_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        
        # Adapt tabu tenure based on progress
        if i % 50 == 0 and i > 0:
            if len(tabu_list) > 0:
                recent_improvements = sum(1 for _, c5 in tabu_list if c5 < best_c5 + 1e-4)
                if recent_improvements > 5:
                    tabu_tenure = max(10, tabu_tenure - 2)
                else:
                    tabu_tenure = min(50, tabu_tenure + 1)
        
        # Occasionally try a random restart
        if i % 500 == 0:
            h_random = np.zeros(n_points)
            h_random[:n_points//2] = 1.0
            np.random.shuffle(h_random)
            h_random = project(h_random)
            c5_random = evaluate_c5(h_random, dx)
            if c5_random < best_c5:
                best_h = h_random
                best_c5 = c5_random
    
    # Final local optimization around best solution
    final_h, final_c5 = local_search(best_h)
    
    return (final_h, final_c5, n_points)
