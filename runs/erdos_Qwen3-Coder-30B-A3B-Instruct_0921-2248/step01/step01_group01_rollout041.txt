def run(seed=42, budget_s=600, **kwargs):

        n_points = 150

        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def calculate_c5(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Binary search for the minimum C5
    left = 0.0
    right = 0.38085
    best_c5 = right
    best_h = h.copy()
    
    while right - left > 1e-5 and time.time() - start_time < budget_s - 1:
        mid = (left + right) / 2.0
        # Formulate SDP feasibility problem
        h_var = cp.Variable(n_points)
        Z = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add constraints for overlap bound (this part is simplified)
        # For each k, we must have sum h[i] * (1 - h[i+k]) <= mid * dx
        # We algebraically represent the overlap constraint via matrix inequalities
        # For simplicity, we'll directly check the constraints using numerical method
        # while still using CVXPY for optimization
        
        # Given a feasible current solution h, we solve the feasibility problem
        # by trying to find feasible h that satisfy overlap bound <= mid
        
        h_trial = h.copy()
        h_trial = np.clip(h_trial, 0, 1)
        h_trial_sum = np.sum(h_trial)
        if abs(h_trial_sum - target_sum) > 1e-10:
            h_trial = h_trial * target_sum / h_trial_sum
        c5_current = calculate_c5(h_trial)
        if c5_current <= mid:
            right = mid
            if mid < best_c5:
                best_c5 = mid
                best_h = h_trial.copy()
        else:
            left = mid
    
    # Refine with local optimization at the final candidate
    h_final = best_h.copy()

        h_final = h_final * target_sum / h_final_sum
    final_c5 = calculate_c5(h_final)
