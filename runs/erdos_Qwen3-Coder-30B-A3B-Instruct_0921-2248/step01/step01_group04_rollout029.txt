import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def mutate_h(h, mutation_rate=0.05):
    """Mutate h by perturbing some elements"""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, 0.05, len(h))
    h_new = h + noise * mask
    return project_feasible(h_new, len(h))

def crossover_h(h1, h2):
    """Crossover two h functions by segment swapping"""
    n = len(h1)
    if n < 2:
        return h1.copy()
    cut_point = np.random.randint(1, n)
    h_new = np.concatenate([h1[:cut_point], h2[cut_point:]])
    return h_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual via tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def ga_optimize(initial_h, n_points, budget_s):
    """Genetic algorithm to optimize h"""
    np.random.seed(42)
    start_time = time.time()
    
    n_pop = 50
    max_gens = 1000
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(n_pop - 1):
        h = mutate_h(initial_h, 0.1)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    for gen in range(max_gens):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitnesses[min_idx]
        
        # Selection + crossover + mutation
        new_population = []
        
        # Keep best
        best_idx = np.argmin(fitnesses)
        new_population.append(population[best_idx].copy())
        
        while len(new_population) < n_pop:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child = crossover_h(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate_h(child, mutation_rate)
            
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5

        h = h + noise
        h = project_feasible(h, n_points)
    
    # Run GA
    best_h, best_c5 = ga_optimize(h, n_points, budget_s)
    
    # Final polish with SLSQP if time allows
    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]

            return np.sum(x) - n_points / 2.0
