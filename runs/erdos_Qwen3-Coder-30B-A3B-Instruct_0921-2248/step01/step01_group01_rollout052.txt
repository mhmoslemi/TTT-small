def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[::2] = 1.0  # Place 1s at even indices
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx

    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Track number of evaluations
    num_evals = 0

    # Optimization at different scales
    scales = [50, 100, 150]
    for scale in scales:
        if time.time() - start_time > budget_s - 1:
            break
            
        if scale > n_points:
            # Interpolate to higher resolution
            old_h = h.copy()
            new_points = scale
            h = np.interp(np.linspace(0, 2, new_points), np.linspace(0, 2, n_points), old_h)
            n_points = new_points
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
        for _ in range(5):  # Multiple restarts
            if time.time() - start_time > budget_s - 1:
                break
                
            # Ensure feasible h
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
                
            # Optimize
            try:
                res = minimize(
                    objective,
                    h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6, 'gtol': 1e-6}
                )
                
                if res.success:
                    h = res.x.copy()
                    c5 = objective(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        
                    # Perturbation based on best overlap
                    noise_scale = min(0.075, 0.025 + 0.05 * c5) 
                    noise = np.random.normal(0, noise_scale, n_points)
                    h = h + noise
                else:
                    # If optimization fails, try another approach
                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)
                    
            except Exception as e:
                # If optimization fails, try random search
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)

            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
            num_evals += 1

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
