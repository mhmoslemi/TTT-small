import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a structured construction
        n_points = 150
        h0 = np.zeros(n_points)
        # Create a step-like function with spread
        split = n_points // 2
        h0[:split] = 1.0
        # Perturb a bit for better optimization
        for i in range(0, n_points, 5):
            if i < n_points:
                h0[i] = np.random.uniform(0, 1)
        # Normalize to integral = 1
        h0 = h0 * (n_points / 2) / np.sum(h0)
        
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Create a more constrained optimization approach
    def finite_difference(h, k):
        # Compute the overlap at shift k
        h_shifted = np.roll(h, k)
        return np.sum(h * (1 - h_shifted))
    
    # Helper function to evaluate maximum overlap
    def max_overlap(h):
        max_val = 0
        for k in range(2 * n_points):
            val = finite_difference(h, k)
            if val > max_val:
                max_val = val
        return max_val * dx
    
    # Bounds: h[i] in [0, 1] and sum = n_points / 2
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Fine-tune with local search
    for i in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.005, n_points)
        h_pert = project(h_pert)
        
        # Optimize with L-BFGS-B
        result = minimize(
            max_overlap,
            h_pert,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = max_overlap(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                
        # Occasionally do a larger random perturbation
        if i % 10 == 0:
            h_pert = best_h + np.random.normal(0, 0.02, n_points)
            h_pert = project(h_pert)
            
            result = minimize(
                max_overlap,
                h_pert,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 50, 'disp': False}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = max_overlap(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
