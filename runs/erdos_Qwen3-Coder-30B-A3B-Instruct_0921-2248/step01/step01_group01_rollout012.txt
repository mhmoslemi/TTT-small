def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 150
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Binary search bounds
    left, right = 0.0, 0.38085
    best_c5 = 0.38085
    best_h = h.copy()
    
    # Objective function to minimize C5 
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Feasibility check using SDP relaxation
    def is_feasible(c5_bound):
        # Formulate SDP 
        h_var = Variable(n_points)
        Z = Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
        ]
        
        # Incorporate overlap constraints
        for k in range(1, 2*n_points-1):
            # Compute convolution index
            if k < n_points:
                corr_val = np.sum(h_var[:n_points-k] * (1 - h_var[k:]))
            else:
                corr_val = np.sum(h_var[k-n_points+1:] * (1 - h_var[:n_points-(k-n_points+1)]))
            
            # Approximate with quadratic constraint 
            constraints.append(corr_val >= (1 - c5_bound) / dx)
        
        # Z constraints (relaxation)
        for i in range(n_points):
            for j in range(n_points):
                constraints.append(Z[i, j] <= h_var[i])
                constraints.append(Z[i, j] <= h_var[j])
                constraints.append(Z[i, j] >= 0)
        
        # PSD constraint
        constraints.append(Z >> 0)
        
        # Minimize to find if feasible
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=CPLEX, verbose=False)
            return prob.status == 'optimal'
        except:
            return False

    # Refine with local optimization
    def refine_solution(h_vec):
        # Project to feasible region
        h_vec = np.clip(h_vec, 0, 1)
        h_vec_sum = np.sum(h_vec)
        if abs(h_vec_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / h_vec_sum
        return h_vec

    # Binary search over C5
    while time.time() - start_time < budget_s - 2 and right - left > 1e-6:
        mid = (left + right) / 2.0
        
        # Check if we can achieve this C5
        feasible = is_feasible(mid) if mid < 0.38085 else False
        if not feasible:
            left = mid
        else:
            right = mid
            # Optimize around this value
            # Refine solution using local optimization
            h_refined = refine_solution(h_vec=h.copy())
            c5 = compute_c5(h_refined)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_refined.copy()

            # Local search
            def obj(h_vec):
                return compute_c5(h_vec) 
            
            h_init = h_refined.copy()
            try:
                res = minimize(obj, h_init, method='L-BFGS-B', bounds=[(0, 1)] * n_points, 
                               options={'ftol': 1e-8, 'gtol': 1e-8}, 
                               callback=lambda x: None)
                if res.success:
                    h_new = refine_solution(res.x)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
            except:
                pass

    # Final check of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
