import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2):
    """Perform segment-based crossover"""
    n = len(parent1)
    if n < 2:
        return parent1.copy(), parent2.copy()
    
    # Choose crossover point
    cx_point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Apply gaussian mutation to h"""
    n_points = len(h)
    mask = np.random.random(n_points) < mutation_rate
    noise = np.random.normal(0, 0.05, n_points)
    h_mut = h + noise * mask
    return project_to_feasible(h_mut, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h0)
    else:
        n_points = 100
        # Create a reasonable initial guess using binary pattern
        h0 = np.zeros(n_points)
        half = n_points // 2
        h0[:half] = 1
    
    # GA parameters
    n_pop = 50
    generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.8
    elite_size = 2
    
    # Initialize population
    population = [project_to_feasible(h0.copy(), n_points)]
    for i in range(1, n_pop):
        # Perturb initial solution a bit
        h_perturbed = h0 + np.random.normal(0, 0.05, n_points)
        population.append(project_to_feasible(h_perturbed, n_points))
    
    # Store best solution
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for entire population
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Track best in current generation
        current_best_idx = np.argmin(fitnesses)
        current_best_h = population[current_best_idx]
        current_best_c5 = fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5
        
        # Selection, crossover, and mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < n_pop:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim population to exact size
        population = new_population[:n_pop]
    
    # Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use best individual as starting point for polishing
    polished_h = best_h.copy()
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        polished_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    if result.success:
        polished_h = result.x
    polished_h = project_to_feasible(polished_h, n_points)
    final_c5 = evaluate_erdos_solution(polished_h)
    
    return (list(polished_h), final_c5, n_points)
