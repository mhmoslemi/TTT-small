import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value of the correlation (scaled by dx)
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create one
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Create a binary vector with approximately half ones
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1
        np.random.shuffle(h0)
    dx = 2.0 / n_points
    
    # Semi-continuous optimization with constraints
    def obj_func(h):
        # Ensure h is within bounds and integral constraint is met
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)

    def constraint_eq(x):
        return np.sum(x) - n_points / 2

    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Optimization parameters
    opt_options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    # Initialize with h0 and keep track of best
    best_h = h0.copy()
    best_c5 = obj_func(best_h)
    
    # Run gradient-based optimization with multiple restarts
    restarts = 5
    for rep in range(restarts):
        if time.time() - start_time > budget_s - 1:
            break
        # Randomly perturb the current best solution
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points) * (1 - best_h)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(obj_func, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options=opt_options)
            if res.success and res.fun < best_c5:
                # Ensure result satisfies the constraint
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_val = obj_func(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
        except Exception:
            pass

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
