import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default: create a vector with 1s and 0s to satisfy constraint
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Scale to match the constraint sum(h) = n_points / 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    # Apply convex optimization approach to minimize C5
    # Variables
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum,
        C5_var >= 0
    ]
    
    # Compute overlap for all shifts and add constraints
    for k in range(n_points):
        # For shift k, compute sum h_i * (1 - h_{i+k}) for valid indices
        # This corresponds to correlation with shifted version of (1-h)
        if k == 0:
            # Skip the k=0 case, it's just the integral of h*(1-h)
            continue
        else:
            # For shift k, we compute the overlap with the shift
            overlap = 0
            first_overlap = 0
            # Compute the valid indices for the correlator
            # In the overlap h*(1-h), for shift k,
            # we consider h[i] * (1 - h[i-k]) for valid i
            # This means: i-k >= 0, so i >= k, and i < n_points
            # Also i < n_points, so we loop from max(k,0) to min(n_points, n_points-k)
            # But the correlation uses i-k, so i from 0 to n_points-1, i-k can be negative, etc.
            # Actually, full mode means we take all correlations including beyond boundary.
            # For the overlap part: sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k)%n_points])
            # But this is a circular convolution. Let's be more precise.
            # We actually want: sum_i h[i] * (1 - h[i-k])
            # This is a convolution which is constrained to be <= C5*dx
            
            # For a shift k, we compute the overlap with shifted version of (1-h)  
            # clip: i ranges from 0 to n_points-1, and i-k must also be in [0, n_points-1]
            # h[i] * (1 - h[i-k]) for all valid i
            
            shift_indices = np.arange(k, n_points)
            valid_indices = np.arange(0, n_points - k)
            overlap_expr = cp.sum(cp.multiply(h_var[valid_indices], 1 - h_var[shift_indices]))
            constraints.append(overlap_expr <= C5_var * (n_points / 2))
    
    # For k=0 case (this is the diagonal term)
    # It's sum of h[i]*(1-h[i]) for all i
    overlap_diag = cp.sum(cp.multiply(h_var, 1 - h_var))
    constraints.append(overlap_diag <= C5_var * (n_points / 2))
    
    # Objective: minimize C5
    objective = cp.Minimize(C5_var)
    prob = cp.Problem(objective, constraints)
    
    # Solve with MOSEK (best solver for this quadratic problem)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-7, reltol=1e-7)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = h_var.value
            c5_value = C5_var.value
        else:
            # fallback to initial solution if optimization fails
            h_solution = h.copy()
            c5_value = np.max(np.correlate(h_solution, 1 - h_solution, mode="full")) * dx
    except Exception:
        h_solution = h.copy()
        c5_value = np.max(np.correlate(h_solution, 1 - h_solution, mode="full")) * dx
    
    # Final validation
    h_solution = np.clip(h_solution, 0, 1)
    final_sum = np.sum(h_solution)
    h_solution = h_solution * target_sum / final_sum
    actual_c5 = np.max(np.correlate(h_solution, 1 - h_solution, mode="full")) * dx
    
    return (h_solution.tolist(), float(actual_c5), n_points)
