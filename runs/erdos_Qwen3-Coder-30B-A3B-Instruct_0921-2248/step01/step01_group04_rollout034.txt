import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

        h = 0.5 + 0.3 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    bounds = [(0, 1) for _ in range(n_points)]
