import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback construction with small number of points
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Track best solution
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution grid refinement
    upscale_sequence = [2, 2, 12]  # This sequence might need tuning
    
    current_h = h.copy()
    current_n = n_points
    
    for upscale_factor in upscale_sequence:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Upsample
        new_n = current_n * upscale_factor
        h_upsampled = np.repeat(current_h, upscale_factor)
        h_upsampled = h_upsampled[:new_n]
        
        # Add controlled noise
        noise_strength = 0.05
        noise = np.random.uniform(-noise_strength, noise_strength, new_n)
        h_upsampled = np.clip(h_upsampled + noise, 0, 1)
        
        # Normalize sum
        h_upsampled = h_upsampled * target_sum / np.sum(h_upsampled)
        
        # Optimize with LBFGS-B
        bounds_new = [(0.0, 1.0)] * new_n
        cons_new = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        try:
            res = minimize(obj_func, h_upsampled.copy(), method='L-BFGS-B', 
                          bounds=bounds_new, constraints=cons_new,
                          options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * target_sum / np.sum(h_opt)
                c5_opt = obj_func(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                current_h = h_opt
                current_n = new_n
            else:
                current_h = h_upsampled
                current_n = new_n
        except Exception:
            current_h = h_upsampled
            current_n = new_n
            
        # Random restart with small probability
        if np.random.rand() < 0.1:
            current_h = np.random.rand(current_n)
            current_h = current_h * target_sum / np.sum(current_h)
    
    # Final validation with original evaluation function
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure final solution is valid
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    
    return (best_h.tolist(), float(final_c5), len(best_h))
