import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Upsample to higher resolution
    n_upsample = 500
    x_orig = np.linspace(0, 2, n_points)
    x_upsample = np.linspace(0, 2, n_upsample)
    h_upsample = np.interp(x_upsample, x_orig, h)
    
    # Normalize to target sum
    target_sum = n_upsample / 2.0
    h_upsample = h_upsample * target_sum / np.sum(h_upsample)
    
    best_h = h_upsample.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use Nelder-Mead for optimization with restarts
    for iter_no in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * target_sum / np.sum(h_vec)
            overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
            max_overlap = np.max(overlap)
            return max_overlap * (2.0 / len(h_vec))
        
        # Random perturbation
        perturbed = h_upsample + np.random.normal(0, 0.02, len(h_upsample))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        try:
            res = minimize(obj_func, perturbed, method='Nelder-Mead', 
                          options={'maxiter': 200, 'fatol': 1e-8, 'xatol': 1e-8})
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
        
        # Add randomness to avoid local minima
        h_upsample = best_h.copy()
        noise = np.random.normal(0, 0.01, len(h_upsample))
        h_upsample = np.clip(h_upsample + noise, 0, 1)
        h_upsample = h_upsample * target_sum / np.sum(h_upsample)
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), n_upsample)
