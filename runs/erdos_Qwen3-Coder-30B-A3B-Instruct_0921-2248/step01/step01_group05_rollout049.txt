import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary step function
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute objective value
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(overlap) * dx
    
    # Helper to enforce sum constraint
    def project_to_feasible(h_vec):
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return np.clip(h_vec, 0, 1)
    
    # Helper to upsample
    def upsample(h_old, factor):
        n_old = len(h_old)
        h_new = np.repeat(h_old, factor)
        if len(h_new) > n_points:
            h_new = h_new[:n_points]
        elif len(h_new) < n_points:
            # Pad with last value
            h_new = np.pad(h_new, (0, n_points - len(h_new)), constant_values=h_new[-1])
        return h_new
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Upsample factors: 1, 2, 4, 12 (to reach 47*12 = 564 eventual size)
    # But we cap it at 200 to avoid excessive memory usage
    upsample_factors = [1, 2, 4]
    upscale_targets = [47, 94, 188]
    if upscale_targets[-1] > 200:
        upscale_targets = [47, 94, 188]
    new_n_points = 200
    
    # Resample to target n points
    def resample_to_target(h_vec):
        if len(h_vec) < new_n_points:
            h_resampled = np.resize(h_vec, new_n_points)
        elif len(h_vec) > new_n_points:
            h_resampled = h_vec[:new_n_points]
        else:
            h_resampled = h_vec.copy()
        return h_resampled
    
    # Optimization loop
    time_remaining = budget_s - (time.time() - start_time)
    
    # Stage 1: Start from current solution
    current_h = h.copy()
    current_h = project_to_feasible(current_h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    # Stage 2: Optimize on finer grids
    for i, target_n in enumerate(upscale_targets):
        if i >= len(upsample_factors):
            break
        if target_n > 200:
            break
        factor = upsample_factors[i]
        if i == 0:
            # Use current solution
            h_upsampled = current_h
        else:
            # Upsample previous solution
            h_upsampled = upsample(h_upsampled, factor)
        
        # Add some noise
        noise_magnitude = 0.1
        h_upsampled = h_upsampled + np.random.uniform(-noise_magnitude, noise_magnitude, len(h_upsampled))
        h_upsampled = project_to_feasible(h_upsampled)
        
        # Optimize with LBFGS-B
        bounds = [(0.0, 1.0)] * len(h_upsampled)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        # Re-target for optimization
        h_vec = h_upsampled.copy()
        try:
            # Optimize for 150s per stage or until timeout
            stage_start = time.time()
            while time.time() - stage_start < 150 and time.time() - start_time < budget_s - 1:
                res = minimize(objective, h_vec, method='L-BFGS-B', bounds=bounds, 
                               options={'maxiter': 50, 'gtol': 1e-6, 'ftol': 1e-6})
                if res.success:
                    h_vec = res.x
                else:
                    h_vec = project_to_feasible(h_vec)
                
                h_vec = project_to_feasible(h_vec)
                c5_new = objective(h_vec)
                if c5_new < best_c5:
                    best_h = h_vec.copy()
                    best_c5 = c5_new
                
                # Random restart with small probability
                if np.random.rand() < 0.15:
                    h_vec = np.random.rand(len(h_vec))
                    h_vec = project_to_feasible(h_vec)
                    
                # Re-evaluate after optimization steps
                time_remaining = budget_s - (time.time() - start_time)
                if time_remaining < 5:
                    break
                    
        except Exception:
            # Prevent failure from terminating entire run
            pass
            
        # Resample to target length
        current_h = resample_to_target(h_vec)
        current_h = project_to_feasible(current_h)
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
                
    # Final validation
    best_h = project_to_feasible(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
