import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Simple binary vector
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Upsample to higher resolution
    n_upsample = 500
    h_upsampled = np.interp(np.linspace(0, 2, n_upsample), 
                           np.linspace(0, 2, n_points), h)
    h = h_upsampled
    n_points = n_upsample
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        # Make sure it's in the valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Nelder-Mead optimization
    for iteration in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Use Nelder-Mead for optimization
            res = minimize(obj_func, h.copy(), method='Nelder-Mead', 
                           options={'maxiter': 200, 'initial_simplex_size': 0.1})
            
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint with projection
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            
            # Random restart with small perturbations
            h = best_h + np.random.normal(0, 0.02, n_points)
            # Project back to feasible region
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * target_sum / current_sum
            
        except Exception as e:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
