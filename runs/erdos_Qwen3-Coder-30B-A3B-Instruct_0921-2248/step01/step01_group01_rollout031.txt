def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with adaptive noise
    max_starts = 10
    for start_iter in range(max_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize from either parent solution or random
        if 'initial_h_values' in globals() and start_iter == 0:
            h = np.array(initial_h_values, dtype=float)
        else:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        
        # Perturb to avoid local optima
        noise_scale = 0.01 + 0.02 * start_iter / max_starts
        noise = np.random.normal(0, noise_scale, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * target_sum / h_sum
            
        # Optimize
        try:
            res = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
            )

            if res.success:
                h = res.x.copy()
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except Exception:
            pass
            
        # Ensure feasibility
        best_h = np.clip(best_h, 0, 1)
        best_h_sum = np.sum(best_h)
        if abs(best_h_sum - target_sum) > 1e-10:
            best_h = best_h * target_sum / best_h_sum
            
        if time.time() - start_time > budget_s - 1:
            break
    
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
