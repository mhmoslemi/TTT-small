import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    try:

        n_points = len(h0)
    except NameError:
        # Fall back to a structured construction
        n_points = 100

        # Create a piecewise constant function with some variation
        segments = 4
        segment_size = n_points // segments
        for i in range(segments):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < segments - 1 else n_points
            h0[start_idx:end_idx] = 1.0 if i % 2 == 0 else 0.0
        # Normalize to sum to n_points/2
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Optimization using different strategies
    strategies = [
        # Strategy 1: Direct optim with SLSQP
        lambda h, dx: minimize(objective, h, method='SLSQP', bounds=bounds, constraints=con, 
                               options={'maxiter': 100, 'disp': False}),
        # Strategy 2: Random restarts with bounds
        lambda h, dx: minimize(objective, np.random.uniform(0, 1, n_points), 
                               method='SLSQP', bounds=bounds, constraints=con, 
                               options={'maxiter': 50, 'disp': False})
    ]
    
    max_iter = 2000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different strategies
        for strat in strategies:
            try:
                # Random perturbation
                h_pert = best_h + np.random.normal(0, 0.01, n_points) * (np.random.rand(n_points) > 0.7)
                h_pert = project(h_pert)
                
                result = strat(h_pert, dx)
                
                if result.success:
                    h_opt = project(result.x)
                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
                        break  # Found better solution, try next iteration
            except:
                continue  # Skip failed attempts
    
    # Final refinement with LBFGS
    try:
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=con,
                          options={'maxiter': 100, 'disp': False})
        if result.success:
            h_refined = project(result.x)
            c5_refined = evaluate_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass  # Skip if failed
    
    return (best_h, best_c5, n_points)
