import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        f = interp1d(x_orig, h_init, kind='linear', fill_value='extrapolate')
        h_init = f(x_target)
        n_points = len(h_init)
    else:
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
    
    # Genetic Algorithm Parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 3
    elite_size = 5
    max_mutations = 5
    
    # Initialize population
    def initialize_population():
        population = []
        # Start with h_init
        h_base = h_init.copy()
        population.append(h_base.copy())
        
        # Add variations
        for _ in range(population_size - 1):
            h = h_base.copy()
            # Add small random perturbations
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            # Normalize to sum
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
        return population
    
    def fitness(h):
        return evaluate_erdos_solution(h)
    
    def tournament_selection(population, fitnesses):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        # Single point crossover
        point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Blend crossover with alpha=0.5
        alpha = 0.5
        diff = np.abs(parent1 - parent2)
        child1 = np.clip(parent1 - alpha * diff, 0, 1)
        child2 = np.clip(parent2 + alpha * diff, 0, 1)
        
        # Ensure sum constraint
        child1_sum = np.sum(child1)
        child1 = child1 * (n_points / 2) / child1_sum
        child2_sum = np.sum(child2)
        child2 = child2 * (n_points / 2) / child2_sum
        
        return child1, child2
    
    def mutate(h):
        h = h.copy()
        # Random reset
        reset_indices = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
        h[reset_indices] = np.random.uniform(0, 1, len(reset_indices))
        
        # Gaussian perturbation
        perturb_indices = np.random.choice(n_points, size=min(max_mutations, n_points), replace=False)
        noise = np.random.normal(0, 0.05, len(perturb_indices))
        h[perturb_indices] += noise
        h = np.clip(h, 0, 1)
        
        # Normalize sum
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        return h
    
    # Initialize
    population = initialize_population()
    fitnesses = [fitness(h) for h in population]
    best_h = population[np.argmin(fitnesses)]
    best_c5 = min(fitnesses)
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism
        new_population = sorted_population[:elite_size]
        new_fitnesses = sorted_fitnesses[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.append(child1)
            new_fitnesses.append(fitness(child1))
            if len(new_population) < population_size:
                new_population.append(child2)
                new_fitnesses.append(fitness(child2))
        
        population = new_population
        fitnesses = new_fitnesses
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
