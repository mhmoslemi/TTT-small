def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = h.copy()

    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx

    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    # Multi-start optimization with different initializations
    n_starts = 10
    best_h_local = None
    best_c5_local = float('inf')

    # Start with parent or default construction
    initial_h = h.copy()
    h0 = project_to_feasible(initial_h)
    
    for start in range(n_starts):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Random initialization with perturbation
        h_start = np.random.rand(n_points)
        h_start = h_start * target_sum / np.sum(h_start)
        
        try:
            res = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            if res.success:
                h_opt = res.x.copy()
                h_opt = project_to_feasible(h_opt)
                c5 = objective(h_opt)
                if c5 < best_c5_local:
                    best_c5_local = c5
                    best_h_local = h_opt.copy()
        except:
            pass
            
        # Try with parent solution
        try:
            res = minimize(
                objective,
                h0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            if res.success:
                h_opt = res.x.copy()
                h_opt = project_to_feasible(h_opt)
                c5 = objective(h_opt)
                if c5 < best_c5_local:
                    best_c5_local = c5
                    best_h_local = h_opt.copy()
        except:
            pass

    if best_h_local is not None:
        best_h = best_h_local
        best_c5 = best_c5_local

    # Final refinement using large number of iterations
    max_iter = 300
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # local exploration: small perturbations
        perturbation = np.random.normal(0, 0.02, n_points)
        h_new = best_h + perturbation
        h_new = project_to_feasible(h_new)
        
        c5_new = objective(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
