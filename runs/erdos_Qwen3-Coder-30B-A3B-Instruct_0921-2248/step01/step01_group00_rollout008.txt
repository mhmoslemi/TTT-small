import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    
    def obj_func(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)

    def constraint_eq(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    opt_options = {'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    
    best_h = h0.copy()
    best_c5 = obj_func(best_h)
    
    # Try multiple random restarts
    n_restarts = 5
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 5:
            break
        # Create diverse starting points
        if restart == 0:
            h_init = h0.copy()
        else:
            # Generate a random binary vector with exact sum n/2
            h_init = np.zeros(n_points)
            h_init[:n_points//2] = 1
            np.random.shuffle(h_init)
            # Add some variation
            for i in range(n_points):
                if np.random.rand() < 0.1 and h_init[i] == 1:
                    h_init[i] = 0
                
        # Optimize
        try:
            res = minimize(obj_func, h_init, method='SLSQP', bounds=bounds, constraints=cons, options=opt_options)
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_val = obj_func(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
        except Exception:
            pass
            
        # Try a gradient-free local search with local perturbations
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(20):
            if time.time() - start_time > budget_s - 5:
                break
            # Make small random perturbation
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points) * 0.1
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Evaluate and accept improve
            c5_val = obj_func(h_perturbed)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_perturbed.copy()
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
