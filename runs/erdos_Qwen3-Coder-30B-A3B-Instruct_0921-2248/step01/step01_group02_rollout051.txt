import numpy as np
from scipy.interpolate import interp1d
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

        h_init[:n_points//4] = 1
        h_init[n_points//4:3*n_points//4] = 0

    num_generations = 1000
    mutation_rate = 0.1
    elite_size = 5
    tournament_size = 3

    def normalize_h(h):
        """Ensure sum(h) = n_points / 2 and h[i] in [0,1]"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        return h

    def mutate(h):
        """Apply mutation while preserving constraints"""
        h_mut = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h_mut[i] += np.random.normal(0, 0.05)
        return normalize_h(h_mut)

    def crossover(h1, h2):
        """Single-point crossover with normalization"""
        point = np.random.randint(1, len(h1))
        child1 = np.concatenate([h1[:point], h2[point:]])
        child2 = np.concatenate([h2[:point], h1[point:]])
        return normalize_h(child1), normalize_h(child2)

    def tournament_select(population, fitnesses):
        """Select an individual via tournament"""

        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return population[best_idx].copy()

        population = []
