import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Function to evaluate overlap
    def evaluate_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # CVXPY solver function
    def cvx_optimize(h_vec):
        h_var = cp.Variable(n_points)
        # Objective: maximize sum of correlations (which is equivalent to minimizing negative)
        # We approximate by directly optimizing the maximum
        # However, CVXPY doesn't support max() directly in a linear way in this context,
        # So we focus on local constraint satisfaction and use standard optimization
    
        # Define constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Use scipy for constrained optimization
        def objective_scipy(h_flat):
            corr = np.correlate(h_flat, 1 - h_flat, mode="full")
            return -np.max(corr) * dx  # negative because we minimize

        result = minimize(objective_scipy, h_vec, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, options={'ftol': 1e-8, 'gtol': 1e-8})
        
        if result.success:
            return result.x
        else:
            # fallback to projection
            h_proj = np.clip(h_vec, 0, 1)
            h_sum = np.sum(h_proj)
            h_proj = h_proj * target_sum / h_sum
            return h_proj
    
    def project_to_feasible(h_vec):
        h_proj = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_proj)
        if abs(h_sum - target_sum) > 1e-10:
            h_proj = h_proj * target_sum / h_sum
        return h_proj
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    no_improvement_count = 0
    noise_scale = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Optimize current solution
        h_opt = cvx_optimize(h.copy())
        
        # Project back to feasible region
        h_opt = project_to_feasible(h_opt)
        
        # Evaluate new candidate
        c5 = evaluate_overlap(h_opt)
        
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            no_improvement_count = 0
            noise_scale = max(0.001, noise_scale * 0.99)  # Reduce noise scale slowly
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                # introduce more randomization
                noise_scale = min(0.1, noise_scale * 1.1)
        
        # perturbation
        noise = np.random.normal(0, noise_scale, n_points)
        h = h_opt + noise
        h = project_to_feasible(h)
        
        # occasionally increase resolution and interpolate
        if iteration % 20 == 0 and n_points < 150:
            # Interpolate to higher resolution
            n_high = min(n_points * 3, 150)
            h_new = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, n_points), h)
            h = h_new
            n_points = n_high
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            
        if no_improvement_count > 100:
            break
    
    # Final validation
    final_c5 = evaluate_overlap(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
