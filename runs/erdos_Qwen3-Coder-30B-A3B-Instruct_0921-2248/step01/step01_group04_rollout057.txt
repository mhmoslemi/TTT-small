import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (n_points / 2.0) / np.sum(h)
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform segment-based crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    cut_point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
    child2 = np.concatenate([parent2[:cut_point], parent1[cut_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate h by perturbing some elements"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    return h

def genetic_algorithm(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    n_points = 47  # Default to the size of initial_h_values if available
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        h_init = np.zeros(n_points)
        half = n_points // 2
        h_init[:half] = 1
    
    # Create initial population
    n_pop = 50
    population = []
    for _ in range(n_pop):
        h = h_init.copy()
        # Add some noise
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = project_to_feasible(h, n_points)
        population.append(h)

    best_h = None
    best_c5 = float('inf')
    
    # Evolution loop
    generations = 1000
    stagnation_count = 0
    stagnation_threshold = 50
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for stagnation
        if gen > 0 and abs(fitness[0] - fitness[-1]) < 1e-6:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count > stagnation_threshold:
            # Reset population with diversified solutions
            for i in range(n_pop):
                h = h_init.copy()
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = project_to_feasible(h, n_points)
                population[i] = h
            stagnation_count = 0

        # Selection
        new_population = []
        for _ in range(n_pop // 2):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Project to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
            
        population = new_population.copy()
        
    # Final polishing with local search
    if best_h is not None:
        # Try to improve with local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            final_h = result.x
        else:
            final_h = best_h
            
        final_h = project_to_feasible(final_h, n_points)
        final_c5 = evaluate_erdos_solution(final_h)
        
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    return best_h, best_c5, n_points

            h[:half] = 1
        h = project_to_feasible(h, n_points)
        c5_bound = evaluate_erdos_solution(h)
        
    return (list(h), c5_bound, n_points)
