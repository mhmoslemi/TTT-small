def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Population size
    pop_size = 10
    population = []
    
    # Initialize population
    for _ in range(pop_size):
        h_temp = h.copy()
        h_temp += np.random.normal(0, 0.01, n_points)
        h_temp = np.clip(h_temp, 0, 1)
        h_sum = np.sum(h_temp)
        if abs(h_sum - target_sum) > 1e-10:
            h_temp = h_temp * target_sum / h_sum
        population.append(h_temp)
    
    iteration = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1:  # leave 1 seconds for final eval
        # Local optimization for each individual in population
        for i in range(pop_size):
            h = population[i].copy()
            try:
                res = minimize(
                    objective,
                    h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                if res.success:
                    h = res.x.copy()
                else:
                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)
            except Exception:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
            
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
                
            population[i] = h.copy()
            
        # Evaluate population
        population_c5 = [objective(h) for h in population]
        
        # Update best
        min_idx = np.argmin(population_c5)
        if population_c5[min_idx] < best_c5:
            best_c5 = population_c5[min_idx]
            best_h = population[min_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Mutation for diversity
        if no_improvement > 20 or iteration % 30 == 0:
            # Add random mutations
            for i in range(pop_size):
                if np.random.rand() < 0.5:
                    noise = np.random.normal(0, 0.02, n_points)
                    population[i] = population[i] + noise
                    population[i] = np.clip(population[i], 0, 1)
                    h_sum = np.sum(population[i])
                    if abs(h_sum - target_sum) > 1e-10:
                        population[i] = population[i] * target_sum / h_sum
                    
        # Crossover: create new solutions from best ones
        if iteration % 5 == 0 and pop_size > 1:
            elite = sorted(range(len(population)), key=lambda k: population_c5[k])[:pop_size//2]
            for i in range(pop_size//2, pop_size):
                parent1 = population[elite[np.random.randint(len(elite))]]
                parent2 = population[elite[np.random.randint(len(elite))]]
                # Simple crossover
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Mutation
                noise = np.random.normal(0, 0.01, n_points)
                child = child + noise
                child = np.clip(child, 0, 1)
                h_sum = np.sum(child)
                if abs(h_sum - target_sum) > 1e-10:
                    child = child * target_sum / h_sum
                population[i] = child
                
        iteration += 1
        if no_improvement > 100:
            break
    
    # Final optimization of best solution
    h = best_h.copy()
    try:
        res = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        if res.success:
            h = res.x.copy()
    except Exception:
        pass
        
    # Final check and normalization
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-10:
        h = h * target_sum / h_sum
    
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, len(h))
