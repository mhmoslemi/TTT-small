import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))
