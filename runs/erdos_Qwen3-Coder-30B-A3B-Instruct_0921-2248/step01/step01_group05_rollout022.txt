import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Evaluates the C5 constant for given h
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for interval [0, 2]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Upsample to 500 points to improve resolution
    n_upsampled = 500
    x_orig = np.linspace(0, 2, n_points)
    x_new = np.linspace(0, 2, n_upsampled)
    h_upsampled = np.interp(x_new, x_orig, h)
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * (2.0 / len(h_vec))
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Initial best
    best_h = h_upsampled.copy()
    best_c5 = obj_func(best_h)
    
    # Nelder-Mead optimization with restarts
    max_iter = 200
    n_restarts = 5
    n_evals = 0
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the solution slightly for restart
        h_start = best_h.copy()
        perturbation = np.random.normal(0, 0.01, len(h_start))
        h_start += perturbation
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * target_sum / np.sum(h_start)
        
        try:
            res = minimize(obj_func, h_start, method='Nelder-Mead', 
                          options={'maxiter': max_iter, 'adaptive': True, 'initial_simplex': None})
            
            if res.success and res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except Exception:
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), len(final_h))
