import numpy as np
from scipy.interpolate import interp1d
import time
import random

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

    pop_size = 50

    fitness = [evaluate_erdos_solution(h) for h in population]

    best_h = population[best_idx].copy()

    # Genetic algorithm main loop
    gen = 0

def tournament_select(population, fitness, k=3):
    """Tournament selection"""
    selected_idx = random.randint(0, len(population) - 1)
    for _ in range(k - 1):
        idx = random.randint(0, len(population) - 1)
        if fitness[idx] < fitness[selected_idx]:
            selected_idx = idx
    return selected_idx

def crossover(parent1, parent2):
    """Single-point crossover with repair"""
    n = len(parent1)
    crossover_point = random.randint(1, n - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Repair to enforce sum constraint
    child1_sum = np.sum(child1)
    child2_sum = np.sum(child2)
    
    child1 = child1 * (n / 2) / child1_sum
    child2 = child2 * (n / 2) / child2_sum
    
    return child1, child2

def mutate(individual, mutation_rate):
    """Gaussian mutation with normalization"""
    n = len(individual)
    for i in range(n):
        if random.random() < mutation_rate:
            individual[i] += np.random.normal(0, 0.1)
    
    # Clip to [0,1]
    individual = np.clip(individual, 0, 1)
    
    # Normalize to sum = n/2
    individual_sum = np.sum(individual)
    individual = individual * (n / 2) / individual_sum
