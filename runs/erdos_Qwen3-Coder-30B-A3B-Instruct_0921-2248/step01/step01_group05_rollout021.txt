import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Target sum constraint
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Upscaling sequence: 47 -> 94 -> 188 -> 200
    upsample_factors = [2, 2, 12]
    upsampled_sizes = [47, 94, 188, 200]
    
    # Optimize at each resolution level
    for i, size in enumerate(upsampled_sizes):
        if time_left() < 5:
            break
            
        # Upsample
        if size > n_points:
            h_up = np.repeat(h, upsample_factors[i-1] if i > 0 else 1)
            h = h_up[:size]
            n_points = size
            dx = 2.0 / n_points
        else:
            n_points = size
            dx = 2.0 / n_points
        
        # Add controlled noise
        noise_level = max(0.05, 0.1 * (time_left() / budget_s))
        noise = np.random.uniform(-noise_level, noise_level, n_points)
        h = np.clip(h + noise, 0, 1)
        
        # Project onto feasible region
        current_sum = np.sum(h)
        h = h * target_sum / current_sum
        
        # Optimize using LBFGS-B
        maxiter = min(50, max(10, int(100 * (time_left() / 200))))
        
        try:
            res = minimize(obj_func, h.copy(), method='L-BFGS-B', bounds=bounds, constraints=cons, 
                           options={'maxiter': maxiter, 'ftol': 1e-8, 'gtol': 1e-8})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                raise ValueError("Optimization failed")
        except Exception as _:
            pass  # Continue with current best
        
        # Random restart with probability 0.1
        if np.random.rand() < 0.1 and time_left() > 5:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
