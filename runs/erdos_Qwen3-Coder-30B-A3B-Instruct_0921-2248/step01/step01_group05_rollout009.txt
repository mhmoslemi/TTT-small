import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Upsample to increase resolution
    n_upsampled = 500
    h_upsampled = np.interp(np.linspace(0, 2, n_upsampled), 
                           np.linspace(0, 2, n_points), h)
    
    # Target sum constraint
    target_sum = n_upsampled / 2.0
    dx = 2.0 / n_upsampled
    
    def obj_func(h_vec):
        # Ensure bounds
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap with proper scaling
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Use bounds and constraints
    bounds = [(0.0, 1.0)] * n_upsampled
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Best solution tracking
    best_h = h_upsampled.copy()
    best_c5 = obj_func(best_h)
    
    # Optimization process
    for i in range(50):  # Limited iterations
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Start from current best with noise
            h_start = best_h.copy()
            noise_level = 0.05
            noise = np.random.normal(0, noise_level, n_upsampled)
            h_start = np.clip(h_start + noise, 0, 1)
            # Project to sum constraint
            current_sum = np.sum(h_start)
            h_start = h_start * target_sum / current_sum
            
            # Optimize with Nelder-Mead
            res = minimize(obj_func, h_start, method='Nelder-Mead',
                          options={'maxiter': 200, 'initial_simplex': None,
                                  'adaptive': True})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
            
        # Random perturbation for escape
        if i % 5 == 0:
            h_pert = best_h.copy()
            pert = np.random.normal(0, 0.02, n_upsampled)
            h_pert = np.clip(h_pert + pert, 0, 1)
            current_sum = np.sum(h_pert)
            h_pert = h_pert * target_sum / current_sum
            c5_pert = obj_func(h_pert)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
    
    # Final validation 
    final_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(final_h)
    final_h = final_h * target_sum / current_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), n_upsampled)
