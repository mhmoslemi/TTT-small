import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create an initial construction with specific pattern
        n_points = 150
        h_init = np.zeros(n_points)
        # Create a simple step function with two levels
        h_init[:n_points//3] = 0.5
        h_init[n_points//3:2*n_points//3] = 1.0
        h_init[2*n_points//3:] = 0.0
        
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    # To track best solution
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # A more robust optimization approach using sequential least squares
    def smooth_optimization(h0, bounds, constraints, objective, max_iter=100):
        """
        Try to improve solution with a hybrid approach
        """
        h_current = h0.copy()
        c5_current = objective(h_current)
        
        # Iteration loop
        for i in range(max_iter):
            # SLSQP for local optimization
            try:
                res = minimize(objective, h_current, method='SLSQP', 
                              bounds=bounds, constraints=constraints, 
                              options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_candidate = res.x.copy()
                    h_candidate = np.clip(h_candidate, 0, 1)
                    # Project to satisfy sum
                    h_sum = np.sum(h_candidate)
                    h_candidate = h_candidate * (n_points / 2) / h_sum
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < c5_current:
                        h_current = h_candidate.copy()
                        c5_current = c5_candidate
                else:
                    # If local optimization fails, project current
                    h_projected = np.clip(h_current, 0, 1)
                    h_sum = np.sum(h_projected)
                    h_projected = h_projected * (n_points / 2) / h_sum
                    c5_candidate = objective(h_projected)
                    if c5_candidate < c5_current:
                        h_current = h_projected.copy()
                        c5_current = c5_candidate
            except Exception:
                # Fallback
                h_projected = np.clip(h_current, 0, 1)
                h_sum = np.sum(h_projected)
                h_projected = h_projected * (n_points / 2) / h_sum
                c5_candidate = objective(h_projected)
                if c5_candidate < c5_current:
                    h_current = h_projected.copy()
                    c5_current = c5_candidate
            
            if time.time() - start_time > budget_s - 1:
                break
                
        return h_current, c5_current
    
    # Main loop with multiple restarts
    while time.time() - start_time < budget_s - 1:
        # Random perturbation
        h_start = h_init.copy()
        if np.random.random() < 0.7:
            # Small random noise
            noise = np.random.normal(0, 0.05, n_points)
            h_start += noise
            h_start = np.clip(h_start, 0, 1)
            # Normalize
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Optimize
        h_opt, c5_opt = smooth_optimization(h_start, bounds, constraints, objective, max_iter=50)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
        
        # Occasionally improve current solution with a better restart
        if np.random.random() < 0.1 and time.time() - start_time < budget_s * 0.8:
            # Try a different initialization
            h_rand = np.zeros(n_points)
            h_rand[:n_points//3] = 1.0
            h_rand[n_points//3:2*n_points//3] = 0.0
            np.random.shuffle(h_rand)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            h_opt2, c5_opt2 = smooth_optimization(h_rand, bounds, constraints, objective, max_iter=50)
            if c5_opt2 < best_c5:
                best_h = h_opt2.copy()
                best_c5 = c5_opt2
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
