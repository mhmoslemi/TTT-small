import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Compute correlations manually for efficiency
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Extract the relevant part (maximum k values)
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if abs(current_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize best solution
    h = project_to_feasible(h)
    best_c5 = compute_overlap(h)
    best_h = h.copy()
    
    # Multi-scale optimization
    resolutions = [50, 150]
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            old_h = h.copy()
            n_points = res
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(old_h)), old_h)
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            h = project_to_feasible(h)
        
        # Local optimization with SLSQP
        max_iter = 500 if res == 50 else 300
        
        for it in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update and project
            h = project_to_feasible(h)
            
            # Local optimization step
            try:
                # Direct function evaluation for the gradient-free approach
                def objective(h_vec):
                    return compute_overlap(h_vec)
                    
                # Use SLSQP with bounds
                bounds = [(0, 1) for _ in range(n_points)]
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - target_sum

                constraints = {'type': 'eq', 'fun': constraint_sum}
                
                res_opt = minimize(
                    objective,
                    h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
                )

                if res_opt.success:
                    h = res_opt.x.copy()
                    c5 = compute_overlap(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                else:
                    # Random perturbation if optimization fails
                    h = h + np.random.normal(0, 0.01, n_points) * best_c5
                    h = project_to_feasible(h)
                    
            except Exception:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
            
            h = project_to_feasible(h)
            
            # Periodic global mutation
            if it % 20 == 0:
                # Slight randomization to escape local minima
                noise_scale = min(0.1, best_c5 * 0.05)
                noise = np.random.normal(0, noise_scale, n_points)
                h = h + noise
                h = project_to_feasible(h)
    
    # Final project
    best_h = project_to_feasible(best_h)
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
