import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        # Make it a bit more structured
        for i in range(0, n_points, 4):
            if i < n_points:
                h0[i] = 1.0
        for i in range(1, n_points, 4):
            if i < n_points:
                h0[i] = 0.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Use a smarter local search
    max_iter = 2000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Construct neighbor with selected mutation
        h_pert = best_h.copy()
        # Random mutation
        idx = np.random.randint(0, n_points)
        h_pert[idx] += np.random.normal(0, 0.03)
        
        # Make sure it's feasible
        h_pert = project(h_pert)
        
        # Optimize with L-BFGS-B
        result = minimize(
            objective,
            h_pert,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 50, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

        # Occasionally do a global perturbation for escape
        if i % 50 == 0:
            h_pert = best_h + np.random.normal(0, 0.02, n_points)
            h_pert = project(h_pert)
            
            result = minimize(
                objective,
                h_pert,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 30, 'disp': False}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
