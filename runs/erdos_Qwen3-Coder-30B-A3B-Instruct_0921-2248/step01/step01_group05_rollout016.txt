import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function computes the C5 value for the given h
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback construction with 47 points
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-6
    
    # Constraints function
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Objective function
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    # Optimization bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Constraint structure
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Keep track of best result
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization
    upscale_factors = [2, 2, 12]  # From 47 -> 94 -> 188 -> 200
    resolutions = [47, 94, 188, 200]
    stage_times = [100, 100, 100, 100]
    
    current_h = h.copy()
    current_n = n_points
    
    for stage, (upscale_factor, resolution) in enumerate(zip(upscale_factors, resolutions)):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Up-sample the function
        current_h = np.repeat(current_h, upscale_factor)
        current_h = current_h[:resolution]
        current_n = resolution
        
        # Optimization
        time_left = budget_s - (time.time() - start_time)
        iter_budget = min(max_iter, int(time_left / 10))
        
        # Attempt optimization with L-BFGS-B
        try:
            res = minimize(obj_func, current_h.copy(), method='L-BFGSB', bounds=bounds[:current_n], constraints=cons, 
                           options={'maxiter': iter_budget, 'ftol': tol, 'gtol': tol})
            
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
        
        # Random restarts with projection
        for k in range(5):
            if time.time() - start_time > budget_s - 5:
                break
            h_restart = np.random.rand(current_n) * 0.5
            current_sum = np.sum(h_restart)
            h_restart = h_restart * target_sum / current_sum
            c5_restart = obj_func(h_restart)
            if c5_restart < best_c5:
                best_h = h_restart.copy()
                best_c5 = c5_restart
                
    # Final check
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
