import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h0 = initial_h_values
        n_points = len(h0)

        n_points = 100

        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_tenure = 20
    max_iter = 5000
    tabu_list = []
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Local optimization refinement
    def refine_solution(h):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        con = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        try:
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 200, 'disp': False}
            )
            if result.success:
                h = project(result.x)
                return h
        except:
            pass
        return h
    
    # Neighborhood search: flip two elements
    def get_neighbors(h, num=5):
        neighbors = []
        h = h.copy()
        for _ in range(num):
            i, j = np.random.choice(n_points, size=2, replace=False)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            # Maintain sum constraint with project
            h_new = project(h_new)
            neighbors.append(h_new)
        return neighbors
    
    # Check if solution is in tabu list
    def is_tabu(h, tabu_list, dx):
        for h_tabu, c5_tabu in tabu_list:
            if np.allclose(h, h_tabu, atol=1e-6):
                return True
        return False
    
    # Tabu search loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate neighbors
        neighbors = get_neighbors(best_h, 10)
        
        # Find best non-tabu neighbor
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for neighbor in neighbors:
            if not is_tabu(neighbor, tabu_list, dx):
                c5 = evaluate_c5(neighbor, dx)
                if c5 < best_neighbor_c5:
                    best_neighbor = neighbor
                    best_neighbor_c5 = c5
        
        # Accept best neighbor
        if best_neighbor is not None:
            best_h = best_neighbor
            best_c5 = best_neighbor_c5
            
            # Refine with local search
            best_h = refine_solution(best_h)
            best_c5 = evaluate_c5(best_h, dx)
            
            # Add to tabu list
            tabu_list.append((best_h.copy(), best_c5))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        
        # Occasionally accept a worse solution to escape local optimum
        if np.random.random() < 0.05:
            # Random neighbor
            h_rand = np.random.rand(n_points)
            h_rand = project(h_rand)
            c5 = evaluate_c5(h_rand, dx)
            if c5 < best_c5:
                best_h = h_rand
                best_c5 = c5
    
    # Final refinement
    best_h = refine_solution(best_h)
    best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h, best_c5, n_points)
