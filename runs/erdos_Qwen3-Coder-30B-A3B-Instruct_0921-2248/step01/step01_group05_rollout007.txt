import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        np.random.shuffle(h)

    best_c5 = evaluate_erdos_solution(h)
    
    # Upscaling stages
    upsample_factors = [2, 2, 12]
    stages = []
    total_time = 0
    
    # Time allocations for different stages
    stage_times = [100, 100, 100]
    
    for i, upscale in enumerate(upsample_factors):
        if total_time >= budget_s - 1:
            break
        time_left = budget_s - total_time - 1
        
        # Create upsampled solution
        h_upsampled = np.tile(h, upscale)
        n_upsampled = len(h_upsampled)
        dx = 2.0 / n_upsampled
        
        # Add small noise for better exploration
        perturb = np.random.uniform(-0.05, 0.05, n_upsampled)
        h_upsampled = np.clip(h_upsampled + perturb, 0, 1)
        
        # Project to correct sum
        target_sum = n_upsampled / 2.0
        current_sum = np.sum(h_upsampled)
        h_upsampled = h_upsampled * target_sum / current_sum
        
        # Define objective function
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
            max_overlap = np.max(overlap)
            return max_overlap * dx
        
        # Define constraint
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        bounds = [(0.0, 1.0)] * n_upsampled
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize using L-BFGS-B for efficiency with bounds
        try:
            res = minimize(obj_func, h_upsampled, method='L-BFGS-B', bounds=bounds,

            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                # Project to sum constraint again
                current_sum = np.sum(h_opt)
                h_opt = h_opt * target_sum / current_sum
                c5_opt = obj_func(h_opt)
                
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt

                h_test = np.random.rand(n_upsampled)
                h_test = h_test * target_sum / np.sum(h_test)
                c5_test = obj_func(h_test)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
        except Exception as e:
            pass
        
        h = h_upsampled

    final_c5 = evaluate_erdos_solution(best_h)
