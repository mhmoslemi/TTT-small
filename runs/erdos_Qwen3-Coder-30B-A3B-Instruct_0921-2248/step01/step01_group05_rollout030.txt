import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default: Simple step function
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Optimization parameters
    max_iter_per_stage = 50
    stage_times = [100, 100, 100, 100]  # Stage times in seconds
    upscale_factors = [2, 2, 12]  # Upscale at each stage
    random_restarts = 5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Helper for constraint
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Helper for bounds
    bounds = [(0.0, 1.0)] * len(h)

    # Constraints dict
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    current_h = h.copy()
    current_n = n_points
    
    # Stages of optimization
    stage_start = 0
    for upsample_factor in upscale_factors:
        if time.time() - start_time > budget_s - 5:
            break
            
        stage_time_limit = stage_times[min(stage_start, len(stage_times)-1)]
        stage_start += 1
        
        # Upsample
        new_n = current_n * upsample_factor
        new_h = np.repeat(current_h, upsample_factor)
        new_h = new_h[:new_n]  # trim if needed
        new_h = new_h.astype(float)
        
        # Adjust to target sum
        current_sum = np.sum(new_h)
        new_h = new_h * target_sum / current_sum
        
        current_h = new_h
        current_n = new_n
        dx = 2.0 / current_n
        
        # Optimize at this resolution
        stage_end_time = time.time() + stage_time_limit
        iter_count = 0
        
        while time.time() < stage_end_time and iter_count < max_iter_per_stage:
            # Optimization
            try:
                # Use L-BFGS-B for better convergence
                res = minimize(
                    lambda x: np.max(np.correlate(x, 1 - x, mode="full")) * dx,
                    current_h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    h_new = np.clip(res.x, 0, 1)
                    current_sum = np.sum(h_new)
                    h_new = h_new * target_sum / current_sum
                    c5_new = evaluate_erdos_solution(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        # Decrease this to refocus search if needed
                        if c5_new < 0.3808:
                            max_iter_per_stage = min(30, max_iter_per_stage)
                            
                    current_h = h_new.copy()
                else:
                    # Random restart
                    current_h = np.random.rand(current_n)
                    current_h = current_h * target_sum / np.sum(current_h)
            except:
                # Failure: random restart
                current_h = np.random.rand(current_n)
                current_h = current_h * target_sum / np.sum(current_h)
            
            iter_count += 1
            
            # Small random perturbations
            if iter_count % 5 == 0 and time.time() < stage_end_time - 5:
                for _ in range(random_restarts):
                    if time.time() - start_time > budget_s - 5:
                        break
                    h_pert = current_h + np.random.normal(0, 0.05, current_n)
                    h_pert = np.clip(h_pert, 0, 1)
                    h_pert = h_pert * target_sum / np.sum(h_pert)
                    c5_pert = evaluate_erdos_solution(h_pert)
                    if c5_pert < best_c5:
                        best_h = h_pert.copy()
                        best_c5 = c5_pert

    # Final refinement
    if time.time() - start_time < budget_s - 10:
        # Final optimiztion of best solution
        try:
            res = minimize(
                lambda x: np.max(np.correlate(x, 1 - x, mode="full")) * dx,
                best_h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 20, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_final)
                h_final = h_final * target_sum / current_sum
                c5_final = evaluate_erdos_solution(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (best_h.tolist(), float(best_c5), len(best_h))
