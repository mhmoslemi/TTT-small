def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 50
    
    # Initialize from parent solution or create default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if len(h) != n_points:
            # Interpolate to target resolution
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-10:
        h = h * (n_points/2) / h_sum
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Population of 15 candidates
    population = [h.copy()]
    for _ in range(14):
        new_h = h + np.random.normal(0, 0.02, n_points)
        population.append(project(new_h))
    
    # Adjustment for convergence tracking
    no_improvement = 0
    max_no_improvement = 100
    noise_scale = 0.01

    while time.time() - start_time < budget_s - 1:
        if no_improvement > max_no_improvement:
            break
            
        np.random.shuffle(population)
        current_best_h = population[0]
        current_best_c5 = objective(current_best_h)
        
        # Try optimizing each candidate using local optimization
        for i in range(len(population)):
            h_candidate = population[i]
            h_candidate = project(h_candidate)
            
            def local_obj(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode="full")
                return np.max(corr) * dx
            
            # Use SLSQP local optimization
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            
            try:
                res = minimize(
                    local_obj,
                    h_candidate,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    updated_h = res.x
                    updated_h = project(updated_h)
                    new_c5 = local_obj(updated_h)
                    
                    if new_c5 < current_best_c5:
                        current_best_c5 = new_c5
                        current_best_h = updated_h
                        
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = updated_h.copy()
                        no_improvement = 0
                        
            except Exception:
                pass
                
            # Add some diversity with perturbation
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, noise_scale, n_points)
                mutated = h_candidate + noise
                population[i] = project(mutated)
            else:
                population[i] = h_candidate
            
        # Global random perturbation
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, noise_scale, n_points)
            population[0] = project(population[0] + noise)
            noise_scale = min(0.1, noise_scale + 0.002)
            
        # Progress tracking
        if current_best_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Update population with new best
        if any(np.allclose(h, current_best_h, atol=1e-10) for h in population):
            pass
        else:
            population.append(current_best_h)
            population.sort(key=lambda x: objective(x))
            if len(population) > 15:
                population.pop()  # Remove worst solution
    
    # Final validation and refinement
    final_h = project(best_h)
    final_c5 = objective(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
