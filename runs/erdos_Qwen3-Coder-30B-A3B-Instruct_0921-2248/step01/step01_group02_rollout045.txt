import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia_w = 0.7
    cognitive_w = 1.5
    social_w = 1.5
    max_vel = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        if np.random.random() < 0.3:
            # Random perturbation of h_init
            h = h_init.copy()
            indices = np.random.choice(n_points, size=n_points//4, replace=False)
            h[indices] = 1 - h[indices]
            np.random.shuffle(h)
        else:
            h = h_init.copy()
            if np.random.random() < 0.5:
                np.random.shuffle(h)
        # Ensure sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h))
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           cognitive_w * r1 * (pbest[i] - swarm[i]) + 
                           social_w * r2 * (gbest - swarm[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            swarm[i] += velocities[i]
            # Enforce bounds
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Ensure sum constraint
            h_sum = np.sum(swarm[i])
            swarm[i] = swarm[i] * (n_points / 2) / h_sum
            
            # Evaluate
            curr_val = evaluate_erdos_solution(swarm[i])
            
            # Update personal best
            if curr_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = curr_val
                
                # Update global best
                if curr_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = curr_val
        
        # Periodic SLSQP refinement
        if iteration % 20 == 0 and iteration > 0:
            top_indices = np.argsort(pbest_vals)[:swarm_size // 10]
            for idx in top_indices:
                h0 = pbest[idx].copy()
                bounds = [(0, 1) for _ in range(n_points)]
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2
                
                constraints = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h):
                    corr = np.correlate(h, 1 - h, mode="full")
                    dx = 2.0 / n_points
                    return np.max(corr) * dx
                
                try:
                    res = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                                  constraints=constraints, options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x.copy()
                        h_refined = np.clip(h_refined, 0, 1)
                        h_sum = np.sum(h_refined)
                        h_refined = h_refined * (n_points / 2) / h_sum
                        refined_val = evaluate_erdos_solution(h_refined)
                        if refined_val < pbest_vals[idx]:

                            pbest_vals[idx] = refined_val
                            # Update gbest if necessary
                            if refined_val < gbest_val:
                                gbest = h_refined.copy()
                                gbest_val = refined_val
                except:
                    pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(gbest)
    
    return (gbest.tolist(), final_c5, n_points)
