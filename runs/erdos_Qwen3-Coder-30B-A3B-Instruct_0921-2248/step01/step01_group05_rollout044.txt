import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
    
    # Scale to match the constraint sum(h) = n_points / 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Construct correlation constraints for all shifts
    max_shift = 2 * n_points  # Full overlap range
    
    # Create CVXPY variables
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # Add overlap constraints
    for k in range(n_points):
        # Compute correlation for this shift
        corr = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                corr += h[i] * (1 - h[j])
        constraints.append(corr <= C5 * (n_points / 2.0))
    
    # Objective: minimize C5
    objective = cp.Minimize(C5)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Solve using MOSEK (more reliable/deterministic than others)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-6, reltol=1e-6)
    except Exception as e:
        # Fallback to ECOS if MOSEK fails
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # Return initial solution if everything fails
            return (h_init.tolist(), float(np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx), n_points)
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h.value
        c5_sol = C5.value
    else:
        h_sol = h_init
        c5_sol = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    
    # Ensure feasible solution
    h_sol = np.clip(h_sol, 0, 1)
    
    # Fixed sum constraint
    h_sol = h_sol * target_sum / np.sum(h_sol)
    
    # Final C5 evaluation
    final_c5 = np.max(np.correlate(h_sol, 1 - h_sol, mode="full")) * dx
    
    return (h_sol.tolist(), float(final_c5), n_points)
