def run(seed=42, budget_s=600, **kwargs):

        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def correlation(h_vec):
        """Compute maximum overlap using correlation."""
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Local optimization using CVXPY
    def optimize_local(h_vec):
        h_cvx = cp.Variable(n_points)
        h_vec = np.array(h_vec)
        
        # Create a simple quadratic approximation
        # We don't directly minimize the exact correlation, but we can try 
        # to find a quadratic approximation that is easier to solve
        
        # We'll use a more direct approach: solve the unconstrained problem, then 
        # project to constraints
        
        # Define constraints
        constraints = [
            h_cvx >= 0,
            h_cvx <= 1,
            cp.sum(h_cvx) == target_sum
        ]
        
        # Simple objective: minimize sum of h * (1-h) across shifts
        # We'll use a quadratic approximation to the overlap integral
        try:
            # Use quadratic approximation to the correlation (simplified)
            # This is a rough approximation, but CVXPY can at least solve this
            obj = cp.Minimize(cp.sum_squares(h_cvx))
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=CPLEX, verbose=False, eps=1e-8)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_cvx.value
        except:
            pass

        try:
            def objective(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode="full")
                return -np.max(corr) * dx
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(
                objective,
                h_vec.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if result.success:
                return result.x
        except:
            pass

        h_proj = np.clip(h_vec, 0, 1)

        return h_proj

        # Slowly decrease noise over time
        if no_improvement_count > 50:
            noise_scale = min(0.1, noise_scale + 0.005)
        
        # Check time again
        if time.time() - start_time > budget_s - 1:
            break
            
        # Occasionally restart with diverse samples
        if np.random.rand() < 0.01:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    final_c5 = correlation(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
