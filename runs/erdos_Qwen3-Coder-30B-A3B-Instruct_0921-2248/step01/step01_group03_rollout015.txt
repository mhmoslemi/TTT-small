import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to structured initial guess
        n_points = 100
        h0 = np.zeros(n_points)
        # Place a step at the center to avoid trivial solutions
        mid = n_points // 2
        h0[:mid] = 1.0
        # Slightly perturb for variety
        h0 += np.random.normal(0, 0.02, n_points)
        h0 = np.clip(h0, 0, 1)
        h_sum = np.sum(h0)
        if h_sum > 0:
            h0 = h0 * (n_points / 2) / h_sum
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Line search with momentum
    def line_search(h, grad, alpha=0.01):
        h_new = h - alpha * grad
        return project(h_new)
    
    # Gradient-based optimization patch for refinement
    def refine_solution(h, max_iter=200):
        h = h.copy()
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            old_c5 = evaluate_c5(h, dx)
            # Compute approximate gradient via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_j = h.copy()
                h_j[j] += eps
                h_j = project(h_j)
                c5_plus = evaluate_c5(h_j, dx)
                grad[j] = (c5_plus - old_c5) / eps
            
            # Try a line search
            alpha = 0.01
            h_trial = line_search(h, grad, alpha)
            new_c5 = evaluate_c5(h_trial, dx)
            
            if new_c5 < old_c5:
                h = h_trial
                if new_c5 < best_c5:
                    best_h[:], best_c5 = h.copy(), new_c5
            else:
                alpha *= 0.5
                h_trial = line_search(h, grad, alpha)
                new_c5 = evaluate_c5(h_trial, dx)
                if new_c5 < old_c5:
                    h = h_trial
                    if new_c5 < best_c5:
                        best_h[:], best_c5 = h.copy(), new_c5
        
        return h
    
    # Main loop - alternating between local refinement and random perturbations
    max_iter = 2000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb solution
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = project(h_pert)
        
        # Local refinement
        h_refined = refine_solution(h_pert, 50)
        c5_refined = evaluate_c5(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
        
        # Random walk enhancement
        h_walk = best_h + np.random.normal(0, 0.01, n_points)
        h_walk = project(h_walk)
        c5_walk = evaluate_c5(h_walk, dx)
        if c5_walk < best_c5:
            best_h = h_walk
            best_c5 = c5_walk
    
    return (best_h, best_c5, n_points)
