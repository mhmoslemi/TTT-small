import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = len(h0)
    else:
        # Default construction
        n_points = 50
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    h0 = h0 * (n_points / 2.0) / np.sum(h0)
    
    def compute_c5(h_vec):
        dx = 2.0 / len(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Population-based optimization with local refinement
    population_size = 15
    population = [h0.copy()]
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.05, n_points)
        h_pert = h0 + noise
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2.0) / np.sum(h_pert)
        population.append(h_pert)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    max_no_improvement = 100
    no_improvement_count = 0
    noise_scale = 0.01
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iter_count += 1
        # Local search using scipy
        new_population = []
        
        for h in population:
            # Constraint sum
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            # Objective function to minimize
            def objective(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode="full")
                return -np.max(corr)  # negative because minimize
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(
                    objective,
                    h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_new = np.clip(res.x, 0, 1)
                    h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                    new_population.append(h_new)
                else:
                    # fallback to current
                    new_population.append(h.copy())
            except Exception:
                new_population.append(h.copy())
        
        # Evaluate all in population
        pop_c5 = [compute_c5(h) for h in new_population]
        current_best_idx = np.argmin(pop_c5)
        current_best_c5 = pop_c5[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = new_population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Ensure no_improvement_count is not too high
        if no_improvement_count >= max_no_improvement:
            break
        
        # Update population
        population = [new_population[current_best_idx].copy()]
        for _ in range(population_size - 1):
            h_crossover = population[0].copy()
            # Introduce some diversity
            idx = np.random.randint(0, n_points, size=50)  # random indices to modify
            noise_mask = np.zeros(n_points)
            noise_mask[idx] = 1
            noise = np.random.normal(0, noise_scale, n_points)
            h_crossover += noise * noise_mask
            h_crossover = np.clip(h_crossover, 0, 1)
            h_crossover = h_crossover * (n_points / 2.0) / np.sum(h_crossover)
            population.append(h_crossover)
            
        # Increase noise scale periodically
        if iter_count % 50 == 0 and noise_scale < 0.05:
            noise_scale += 0.005
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
