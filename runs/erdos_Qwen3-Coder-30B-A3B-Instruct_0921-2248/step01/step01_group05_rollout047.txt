import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Resolution stages: 47 -> 94 -> 188 -> 200
    stages = [94, 188, 200]
    upscale_factors = [2, 2, 12//2]  # Last is to make 200 from 188
    
    current_h = h.copy()
    current_n = n_points
    
    # Process stages
    for stage_idx, (next_n, upscale_factor) in enumerate(zip(stages, upscale_factors)):
        if time_left() < 10:
            break
            
        # Upsample
        current_h = np.repeat(current_h, upscale_factor)
        current_n = len(current_h)
        
        # Ensure final size
        if current_n < next_n:
            pad_len = next_n - current_n
            current_h = np.pad(current_h, (0, pad_len), constant_values=0)
        elif current_n > next_n:
            current_h = current_h[:next_n]
        current_n = next_n
            
        # Precompute dx and target sum
        dx = 2.0 / current_n
        
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
            return np.max(overlap) * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        bounds = [(0.0, 1.0)] * current_n
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize
        try:
            # Initialize with current_h
            h_start = np.copy(current_h)
            res = minimize(obj_func, h_start, method='L-BFGSB', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8})
            
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                # Project to satisfy sum constraint (more accurate projection)
                current_sum = np.sum(h_opt)
                h_opt = h_opt * target_sum / current_sum
                c5_val = obj_func(h_opt)
                if c5_val < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_val
            else:
                # Fallback: use current_h but with normalization
                h_opt = np.clip(current_h, 0, 1)
                current_sum = np.sum(h_opt)
                h_opt = h_opt * target_sum / current_sum
                c5_val = obj_func(h_opt)
                if c5_val < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_val
                
            current_h = h_opt
        except Exception:
            pass
            
    # Local search: random restarts and crossover
    if time_left() > 10:
        for restart in range(5):
            if time_left() < 5:
                break
                
            # Re-initialize from the best
            h_restart = best_h.copy()
            h_restart += np.random.normal(0, 0.05, len(h_restart))
            h_restart = np.clip(h_restart, 0, 1)
            current_sum = np.sum(h_restart)
            h_restart = h_restart * target_sum / current_sum
            
            try:
                h_start = np.copy(h_restart)
                res = minimize(evaluate_erdos_solution, h_start, method='L-BFGSB',
                               bounds=[(0.0, 1.0)] * current_n, 
                               options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
                
                if res.success or not np.isnan(res.fun):
                    h_opt = np.clip(res.x, 0, 1)
                    current_sum = np.sum(h_opt)
                    h_opt = h_opt * target_sum / current_sum
                    c5_val = evaluate_erdos_solution(h_opt)
                    if c5_val < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_val
                        
            except Exception:
                pass
                
            # Crossover
            try:
                crossover_point = np.random.randint(1, len(best_h))
                h_cross = np.concatenate([best_h[:crossover_point], h_restart[crossover_point:]])
                h_cross = np.clip(h_cross, 0, 1)
                current_sum = np.sum(h_cross)
                h_cross = h_cross * target_sum / current_sum
                c5_val = evaluate_erdos_solution(h_cross)
                if c5_val < best_c5:
                    best_h = h_cross.copy()
                    best_c5 = c5_val
            except Exception:
                pass
                
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    best_h = best_h * target_sum / current_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure final constraints are met exactly
    final_sum = np.sum(best_h)
    assert abs(final_sum - target_sum) < 1e-6, "Sum constraint violated"
    
    return (best_h.tolist(), float(final_c5), len(best_h))
