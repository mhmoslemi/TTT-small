import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 100
        # Create a structured initial guess with better distribution
        h0 = np.ones(n_points)
        h0[:n_points//2] = 0.0
        # Add some randomness
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Gradient-based optimization with line search
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation with adaptive step size
        step_size = max(0.001, 0.01 * (1 - i / max_iter))
        h_pert = best_h + np.random.normal(0, step_size, n_points)
        h_pert = project(h_pert)
        
        # Optimize with SLSQP
        try:
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 100, 'disp': False}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
        
        # Occasionally try a more aggressive search
        if i % 20 == 0:
            h_pert = best_h + np.random.normal(0, 0.05, n_points)
            h_pert = project(h_pert)
            
            try:
                result = minimize(
                    objective,
                    h_pert,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=con,
                    tol=1e-6,
                    options={'maxiter': 50, 'disp': False}
                )
                
                if result.success:
                    h_opt = project(result.x)
                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
            except:
                pass
    
    # Final refinement with gradient-based approach
    try:
        h_pert = best_h + np.random.normal(0, 0.001, n_points)
        h_pert = project(h_pert)
        
        result = minimize(
            objective,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except:
        pass
    
    return (best_h, best_c5, n_points)
