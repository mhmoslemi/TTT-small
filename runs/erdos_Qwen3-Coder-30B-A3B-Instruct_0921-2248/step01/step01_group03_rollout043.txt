import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Optimization approach: use convex relaxation with SDP
    # Method: start with gradient-based optimization on good initial solution
    
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Optimizer: try multiple strategies
    strategies = [
        # Strategy 1: SLSQP with random perturbations
        lambda h: minimize(objective, h, method='SLSQP', bounds=bounds, constraints=con, 
                           options={'maxiter': 50, 'disp': False}),
        # Strategy 2: L-BFGS-B
        lambda h: minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 50, 'disp': False})
    ]
    
    for i in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use the best solution found so far as current start
        h_start = best_h.copy()
        
        # Introduce a small perturbation
        pert = np.random.normal(0, 0.01, n_points)
        h_pert = h_start + pert
        h_pert = project(h_pert)
        
        # Try multiple optimization strategies
        for strategy in strategies:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                result = strategy(h_pert)
                if result.success:
                    h_opt = project(result.x)
                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
                        break  # Found new best
            except:
                continue
    
    return (best_h, best_c5, n_points)
