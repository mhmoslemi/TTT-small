import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 100
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Parameters for genetic algorithm
    n_pop = 50
    n_gen = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = []
    for _ in range(n_pop):
        temp_h = h.copy()
        # Add some noise to get diversity
        noise = np.random.normal(0, 0.05, n_points)
        temp_h += noise
        temp_h = np.clip(temp_h, 0, 1)
        # Rescale to sum to n_points/2
        temp_h = temp_h * (n_points / 2.0) / np.sum(temp_h)
        population.append(temp_h)
    
    # Track best in generation
    generation_best = float('inf')
    stagnation_counter = 0
    stagnation_limit = 100
    
    for gen in range(n_gen):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        
        # Track best in generation
        gen_best_idx = np.argmin(fitness)
        if fitness[gen_best_idx] < generation_best:
            generation_best = fitness[gen_best_idx]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Selection (tournament)
        new_population = []
        for _ in range(n_pop):
            # Tournament selection
            tournament_indices = np.random.choice(n_pop, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone winner
            new_individual = population[winner_idx].copy()
            
            # Crossover with random other individual
            if np.random.random() < crossover_rate:
                crossover_parent = population[np.random.choice(n_pop)]
                # Segment crossover
                crossover_point = np.random.randint(1, n_points)
                new_individual[:crossover_point] = crossover_parent[:crossover_point]
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Randomly mutate some elements
                num_muts = max(1, int(n_points * mutation_rate))
                mutation_indices = np.random.choice(n_points, num_muts, replace=False)
                for idx in mutation_indices:
                    new_individual[idx] += np.random.normal(0, 0.1)
            
            # Ensure feasibility
            new_individual = np.clip(new_individual, 0, 1)
            new_individual = new_individual * (n_points / 2.0) / np.sum(new_individual)
            
            new_population.append(new_individual)
        
        # Replace population
        population = new_population
        
        # Update global best
        current_best_idx = np.argmin([evaluate_erdos_solution(ind) for ind in population])
        current_best = population[current_best_idx]
        current_c5 = evaluate_erdos_solution(current_best)
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Reset if stagnation
        if stagnation_counter > stagnation_limit:
            # Restart with some best individuals and perturbations
            fitness = [evaluate_erdos_solution(ind) for ind in population]
            top_indices = np.argsort(fitness)[:n_pop//2]
            new_pop = [population[i] for i in top_indices]
            
            # Add fresh individuals
            for _ in range(n_pop - len(new_pop)):
                temp_h = h.copy()
                noise = np.random.normal(0, 0.05, n_points)
                temp_h += noise
                temp_h = np.clip(temp_h, 0, 1)
                temp_h = temp_h * (n_points / 2.0) / np.sum(temp_h)
                new_pop.append(temp_h)
                
            population = new_pop
            stagnation_counter = 0
    
    # Final polish with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Evaluate best solution and polish
    final_h = best_h.copy()
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        final_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    
    if result.success:
        final_h = result.x
    
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2.0) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
