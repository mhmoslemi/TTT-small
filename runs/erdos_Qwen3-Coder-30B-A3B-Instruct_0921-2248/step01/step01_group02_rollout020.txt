from scipy.interpolate import interp1d
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def tournament_selection(fitness, k=3):
    """Perform tournament selection"""
    indices = np.random.choice(len(fitness), k)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return best_idx

def crossover(parent1, parent2):
    """Single-point crossover with repair"""
    n = len(parent1)
    point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Repair to maintain sum constraint
    sum1 = np.sum(child1)
    sum2 = np.sum(child2)
    child1 = child1 * (n / 2) / sum1
    child2 = child2 * (n / 2) / sum2
    
    return child1, child2

def mutate(h, rate):
    """Mutation with repair"""
    for i in range(len(h)):
        if np.random.random() < rate:
            if np.random.random() < 0.5:
                # Random reset
                h[i] = np.random.random()
            else:
                # Gaussian perturbation
                h[i] += np.random.normal(0, 0.05)
    
    # Repair
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - (len(h) / 2)) > 1e-10:
        h = h * (len(h) / 2) / h_sum
    
    return h
