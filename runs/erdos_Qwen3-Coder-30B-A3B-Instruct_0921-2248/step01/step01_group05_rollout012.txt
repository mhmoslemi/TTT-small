import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default construction: simple binary vector
        n_points = 100
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
    
    # Scale initial point to match target integral constraint
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Define variables
    h = cp.Variable(n_points, bounds=(0, 1))
    C5 = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == target_sum]
    
    # Add overlap constraints
    max_shift = n_points - 1
    for k in range(max_shift):
        # Compute correlation for shift k
        corr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
        constraints.append(corr <= C5 * target_sum)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Set solver and solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-6)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = C5.value
        else:
            # Fallback to initial solution if optimization fails
            h_opt = h_init.copy()
            c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    except Exception:
        # Fallback in case of solver issues
        h_opt = h_init.copy()
        c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx

    # Ensure valid range and sum
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * target_sum / np.sum(h_opt)
    
    # Final evaluation
    final_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    return (h_opt.tolist(), float(final_c5), n_points)
