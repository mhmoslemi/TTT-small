import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

    if n_points >= 50:

        # Progressive decrease of perturbation scale

    if time.time() - start_time < budget_s - 5:
        # Define bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
