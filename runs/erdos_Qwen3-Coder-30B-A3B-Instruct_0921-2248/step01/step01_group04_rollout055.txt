import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 200
        # Start with binary mixture
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach
    # Start with a small subset of shifts
    possible_shifts = list(range(2 * n_points - 1))
    current_shifts = np.random.choice(possible_shifts, size=min(50, len(possible_shifts)), replace=False)
    current_shifts = sorted(list(current_shifts))
    
    # Iteratively improve
    max_iter = 1000
    iteration = 0
    
    while iteration < max_iter:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Build LP problem
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(cp.sum(h_vars) == n_points / 2)
        constraints.append(c5_var >= 0)
        
        # Add constraints for current shifts
        for k in current_shifts:
            # Compute correlation for shift k
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(c5_var >= corr_sum * (2.0 / n_points))
        
        # Objective
        objective = cp.Minimize(c5_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_vars.value).flatten()
                c5_new = c5_var.value
                
                # Project back to feasible region
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
                # Find the most violated constraint
                max_violation = -1
                best_k = -1
                
                # Check all potentially missing shifts
                for k in possible_shifts:
                    if k not in current_shifts:
                        corr_sum = 0
                        for i in range(n_points):
                            j = (i + k) % n_points
                            corr_sum += h_new[i] * (1 - h_new[j])
                        violation = corr_sum * (2.0 / n_points) - c5_new
                        if violation > max_violation:
                            max_violation = violation
                            best_k = k
                
                # Add most violated shift if significant improvement
                if best_k != -1 and max_violation > 1e-5:
                    current_shifts.append(best_k)
                    current_shifts = sorted(list(set(current_shifts)))
                else:
                    # If no improvement, perturb the solution
                    h_new += np.random.normal(0, 0.01, n_points)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            else:
                # If not optimal, proceed with current solution
                pass
                
        except Exception as e:
            # Fallback to simple gradient method
            pass
            
        iteration += 1
    
    # Final polish with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Try to refine with SLSQP
    try:
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = best_h.copy()
    except:
        h_final = best_h.copy()
    
    # Final projection
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    final_c5 = evaluate_erdos_solution(h_final)
    
    if final_c5 < best_c5:
        best_h = h_final.copy()
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
