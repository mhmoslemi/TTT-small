import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h0 = initial_h_values
        n_points = len(h0)
    except NameError:

        n_points = 200

        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_tenure = 20
    tabu_list = []
    max_iter = 10000
    iter_without_improvement = 0
    max_no_improve = 500
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def evaluate_h(h):
        return evaluate_c5(h, dx)
    
    # Generate neighbor by flipping two elements
    def get_neighbor(h):
        h_new = h.copy()
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        # Flip the values at positions i1 and i2
        h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        # Project to satisfy constraint
        return project(h_new)
    
    # Start with initial solution
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidates
        candidates = []
        for _ in range(10):  # Generate multiple candidates
            neighbor = get_neighbor(current_h)
            c5_val = evaluate_h(neighbor)
            candidates.append((neighbor, c5_val))
            
        # Find best candidate not in tabu list
        best_candidate = None
        best_c5_candidate = float('inf')
        
        for h_cand, c5_cand in candidates:
            cand_tuple = tuple(h_cand.round(6))  # Round for hashing
            if cand_tuple not in tabu_list and c5_cand < best_c5_candidate:
                best_candidate = h_cand
                best_c5_candidate = c5_cand
        
        # If no valid candidate found, take any best candidate
        if best_candidate is None:
            for h_cand, c5_cand in candidates:
                if c5_cand < best_c5_candidate:
                    best_candidate = h_cand
                    best_c5_candidate = c5_cand
        
        # Update current solution
        current_h = best_candidate
        current_c5 = best_c5_candidate
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            iter_without_improvement = 0
        else:
            iter_without_improvement += 1
            
        # Update tabu list
        tabu_list.append(tuple(current_h.round(6)))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
            
        # Adaptive tabu tenure
        if iter_without_improvement > 100 and tabu_tenure > 5:
            tabu_tenure -= 1
        
        # Early stop if no improvement for a while
        if iter_without_improvement > max_no_improve:
            break
    
    # Final continuous optimization around the best solution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Local optimization
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=con,
        tol=1e-6,
        options={'maxiter': 100, 'disp': False}
    )
    
    if result.success:
        h_final = project(result.x)
        c5_final = evaluate_c5(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (best_h, best_c5, n_points)
