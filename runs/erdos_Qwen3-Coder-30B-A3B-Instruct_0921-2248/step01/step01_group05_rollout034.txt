import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    # Helper to project solution to feasible region
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0)] * n_points
    
    # Optimization parameters
    max_iter = 100
    ftol = 1e-8
    gtol = 1e-8
    perturb_strength = 0.1
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Try upscaling scheme: 47 -> 94 -> 188 -> 200
    upscale_factors = [2, 2, 12]
    upscale_points = [47, 94, 188, 200]
    
    current_h = h.copy()
    current_n = n_points
    
    for i, upscale_factor in enumerate(upscale_factors):
        if time.time() - start_time > budget_s - 5:
            break
            
        new_n = upscale_points[i+1] if i < len(upscale_points) - 1 else current_n
        if current_n < new_n:
            # Upscale
            new_h = np.zeros(new_n)
            for j in range(current_n):
                new_h[j * upscale_factor] = current_h[j]
            # Interpolate for remaining positions
            for j in range(1, upscale_factor):
                for k in range(current_n):
                    if (k * upscale_factor + j) < new_n:
                        new_h[k * upscale_factor + j] = current_h[k]
            current_h = new_h
            current_n = new_n
        else:
            # Keep same size, just add noise and optimize
            pass
        
        # Add noise for perturbation
        noise = np.random.uniform(-perturb_strength, perturb_strength, current_n)
        current_h = current_h + noise
        current_h = project(current_h)
        
        # Optimization using L-BFGS-B
        try:
            result = minimize(
                obj_func,
                current_h.copy(),
                method='L-BFGSB',
                bounds=bounds,
                options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol}
            )
            if result.success:
                optimized_h = project(result.x)
                c5 = obj_func(optimized_h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = optimized_h.copy()
            else:
                # If optimization failed, use the current one
                current_h = project(current_h)
                c5 = obj_func(current_h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current_h.copy()
        except Exception:
            # If optimization causes error, just proceed after projection
            pass
            
        # Add random restart
        if i < len(upscale_factors) - 1 and np.random.random() < 0.3:
            current_h = np.random.rand(current_n)
            current_h = project(current_h)
        
        # Cool down perturbation
        perturb_strength *= 0.8
    
    # Final projection to ensure feasibility
    final_h = project(best_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), len(final_h))
