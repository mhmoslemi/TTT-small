def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)

        # Default construction: binary vector with half 1s and half 0s
        n_points = 47

        initial_h[:n_points//2] = 1.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        # Compute all correlations using full mode
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum value
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Keep track of best solution
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Multi-scale optimization approach
    resolutions = [50, 100, 150]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new h with current resolution
        h = np.zeros(res)
        # Interpolate from previous h
        if len(initial_h) <= res:
            indices = np.linspace(0, len(initial_h)-1, res, dtype=int)
            h = initial_h[indices]
        else:
            # Downsample if necessary
            indices = np.linspace(0, len(initial_h)-1, res, dtype=int)
            h = initial_h[indices]
        
        # Ensure sum constraint and feasibility
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Add noise to prevent from premature convergence
        h += np.random.normal(0, 0.01, res)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Optimize using L-BFGS-B with a lower tolerance
        bounds = [(0, 1) for _ in range(res)]
        
        def objective(x):
            return compute_c5(x)
        
        # Constraint for sum
        def constraint_sum(x):
            return np.sum(x) - target_sum
            
        constraints = {'type': 'eq', 'fun': constraint_sum}

        attempts = 3

            if time.time() - start_time > budget_s - 1:
                break

                    objective,
                    h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6, 'gtol': 1e-6}

                    h_opt = res_min.x.copy()

                        best_c5 = c5
                        best_h = h_opt.copy()

                    noise_level = min(0.05, 0.1 * c5)  # scale noise based on current best

                else:

                    h = np.random.rand(res)
                    h = h * target_sum / np.sum(h)

            best_h = best_h * target_sum / best_h_sum
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
