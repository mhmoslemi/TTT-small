import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

        # Start with a smooth bell-shaped function
        h = np.exp(-((x - 1) ** 2) / 0.1)
        # Normalize to area = 1
        h = h / np.sum(h) * 2.0  # scale to sum = 2
        h = np.clip(h, 0, 1)
        # Slightly adjust to sum = 1
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Fourier-based optimization approach
    M = 10  # number of cosine basis functions
    # Define basis functions
    def basis_functions(x):
        basis = np.ones((len(x), M))
        for m in range(1, M):
            basis[:, m] = np.cos(m * np.pi * x / 2.0)
        return basis
    
    def make_h_from_coeffs(a_coeffs, x_points):
        basis_vals = basis_functions(x_points)
        h_vals = np.dot(basis_vals, a_coeffs)
        return np.clip(h_vals, 0, 1)
    
    # Derivative of h with respect to coefficients
    def basis_derivatives(x):
        basis_d = np.zeros((len(x), M))
        basis_d[:, 0] = 1.0
        for m in range(1, M):
            basis_d[:, m] = -m * np.pi / 2.0 * np.sin(m * np.pi * x / 2.0)
        return basis_d
    
    # Initial coefficients
    x_points = np.linspace(0, 2, n_points)
    basis_vals = basis_functions(x_points)
    a_initial = np.linalg.lstsq(basis_vals, h, rcond=None)[0]

    max_iter = 3000
    learning_rate = 0.01
    best_a = a_initial.copy()
    best_c5_a = best_c5

            break
            
        # Compute h from current coefficients
        h_current = make_h_from_coeffs(best_a, x_points)
        # Ensure area constraint
        h_current = h_current * (n_points / 2.0) / np.sum(h_current)
        
        # Compute C5 for the current h
        c5_current = evaluate_erdos_solution(h_current)
        
        if c5_current < best_c5_a:
            best_c5_a = c5_current
            best_a = best_a.copy()
        
        # Compute gradient via finite differences for efficiency
        epsilon = 1e-5
        gradient = np.zeros(M)
        for i in range(M):
            a_plus = best_a.copy()

            h_plus = make_h_from_coeffs(a_plus, x_points)

    bounds = [(-10, 10) for _ in range(M)]
    cons = {'type': 'eq', 'fun': constraint_area}

    # Return the best solution found
    return (list(final_h), final_c5, n_points)
