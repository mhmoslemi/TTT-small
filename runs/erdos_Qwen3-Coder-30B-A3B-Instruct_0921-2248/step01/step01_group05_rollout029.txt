import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Fallback: create a reasonable initial vector
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        np.random.shuffle(h_init)
    
    # Scale to match the required integral constraint
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Define variables
    h = cp.Variable(n_points, bounds=(0, 1))
    C5 = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == target_sum]
    
    # Add overlap constraints for each shift
    # We limit shifts to avoid too many constraints
    max_shift = min(n_points, 100)  # Reduce for time efficiency
    
    for k in range(max_shift):
        # Create correlation constraint
        corr = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                corr += h[i] * (1 - h[j])
        # Add constraint: corr <= C5 * (target_sum)
        constraints.append(corr <= C5 * target_sum)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s * 0.9)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            if h_opt is not None:
                h_final = np.clip(h_opt, 0, 1)
                # Ensure sum constraint
                h_final = h_final * target_sum / np.sum(h_final)
                c5_bound = prob.value
            else:
                h_final = h_init
                c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
        else:
            h_final = h_init
            c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    except Exception:
        h_final = h_init
        c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    
    # Final check
    if not (np.all(h_final >= 0) and np.all(h_final <= 1)):
        h_final = np.clip(h_final, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    eval_c5 = np.max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
    
    return (h_final.tolist(), float(eval_c5), n_points)
