import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Target sum constraint
    target_sum = n_points / 2.0
    
    # Optimization parameters
    max_iter = 100
    max_time_per_stage = 100  # seconds per upsample stage
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Scales to upscale
    scales = [2, 2, 12/2]  # 47 -> 94 -> 188 -> 200
    current_n = n_points
    
    # Iterative upsampling and optimization
    scale_idx = 0
    while scale_idx < len(scales) and (time.time() - start_time) < budget_s - 5:
        upscale_factor = scales[scale_idx]
        new_n = int(current_n * upscale_factor)
        
        # Upsample
        h_upsampled = np.repeat(h, upscale_factor)
        h = h_upsampled[:new_n]
        current_n = new_n
        dx = 2.0 / current_n
        target_sum = current_n / 2.0
        
        # Optimization function with constraints
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
            max_overlap = np.max(overlap)
            return max_overlap * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        # Bounds and constraints
        bounds = [(0.0, 1.0)] * current_n
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Local optimization with L-BFGS-B
        stage_start = time.time()
        while (time.time() - stage_start) < max_time_per_stage and (time.time() - start_time) < budget_s - 5:
            # Optimize
            try:
                res = minimize(obj_func, h.copy(), method='L-BFGSB', bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8})
                
                if res.success or res.fun < best_c5:
                    h_new = np.clip(res.x, 0, 1)
                    current_sum = np.sum(h_new)
                    if abs(current_sum - target_sum) > 1e-6:
                        h_new = h_new * target_sum / current_sum
                    c5_new = obj_func(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except Exception:
                pass
            
            # Random restart with perturbed solution
            try:
                h_perturb = h.copy() + np.random.normal(0, 0.05, current_n)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * target_sum / np.sum(h_perturb)
                c5_perturb = obj_func(h_perturb)
                if c5_perturb < best_c5:
                    best_h = h_perturb.copy()
                    best_c5 = c5_perturb
            except Exception:
                pass
            
            # Restart from current best
            h = best_h.copy() + np.random.normal(0, 0.01, current_n)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            
        # Update current solution
        h = best_h.copy()
        scale_idx += 1
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    # Return final best
    return (final_h.tolist(), float(final_c5), len(final_h))
