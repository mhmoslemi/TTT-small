import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        np.random.shuffle(h_init)
    
    # Start with initial solution
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try multiple restarts with variations
    for restart in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random restart with perturbation
        h0 = h_init.copy()
        if np.random.random() < 0.5:
            # Add small random noise
            noise = np.random.normal(0, 0.02, n_points)
            h0 = h0 + noise
            h0 = np.clip(h0, 0, 1)
            
        # Project to correct sum
        h0 = h0 * (n_points / 2) / np.sum(h0)
        
        # Local optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        options = {'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode="full")
            dx = 2.0 / n_points
            return np.max(corr) * dx
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options=options)
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                # Re-project to sum constraint
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
        except Exception:
            pass
            
        # Periodic refinement
        if restart % 10 == 0:
            # Try a volatility-based gradient descent improvement
            current_h = best_h.copy()
            current_c5 = best_c5
            for step in range(5):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute gradients via finite differences
                eps = 1e-4
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = current_h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = evaluate_erdos_solution(h_plus)
                    
                    h_minus = current_h.copy()
                    h_minus[i] -= eps
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    c5_minus = evaluate_erdos_solution(h_minus)
                    
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                # Descent step
                learning_rate = 0.1 / (1.0 + restart/10)
                new_h = current_h - learning_rate * grad
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = evaluate_erdos_solution(new_h)
                
                if new_c5 < current_c5:
                    current_h = new_h
                    current_c5 = new_c5
                else:
                    break
                    
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
