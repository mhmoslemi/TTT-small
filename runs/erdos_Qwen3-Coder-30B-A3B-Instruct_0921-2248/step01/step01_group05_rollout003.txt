import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization
    upscale_factors = [2, 2, 12]  # 47 -> 94 -> 188 -> 200
    current_h = h.copy()
    current_n = n_points
    
    for upscale in upscale_factors:
        if time.time() - start_time > budget_s - 5:
            break
        new_n = current_n * upscale
        new_h = np.zeros(new_n)
        for i in range(new_n):
            new_h[i] = current_h[i // upscale]
        
        # Optimization with L-BFGS-B
        try:
            res = minimize(obj_func, new_h, method='L-BFGS-B', bounds=bounds[:new_n], 
                           constraints=cons, options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception:
            pass
        
        current_h = new_h
        current_n = new_n
        
        # Randomized local searches
        for _ in range(5):
            if time.time() - start_time > budget_s - 5:
                break
            h_pert = current_h + np.random.normal(0, 0.05, current_n)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * target_sum / np.sum(h_pert)
            c5_new = obj_func(h_pert)
            if c5_new < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_new
    
    # Final optimization with L-BFGS-B
    try:
        res = minimize(obj_func, best_h, method='L-BFGS-B', bounds=bounds[:len(best_h)], 
                       constraints=cons, options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * target_sum / np.sum(h_final)
            c5_final = obj_func(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    except Exception:
        pass
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
