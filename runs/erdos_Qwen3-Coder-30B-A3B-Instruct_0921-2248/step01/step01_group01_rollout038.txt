import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize population with diverse candidates
    population_size = 15
    population = []
    for _ in range(population_size):
        h_temp = h.copy() + np.random.normal(0, 0.05, n_points)
        h_temp = np.clip(h_temp, 0, 1)
        h_sum = np.sum(h_temp)
        if abs(h_sum - target_sum) > 1e-10:
            h_temp = h_temp * target_sum / h_sum
        population.append(h_temp)
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:  # leave 1 seconds for final eval
        # Evaluate all candidates
        for i, candidate in enumerate(population):
            c5 = objective(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Update population with new candidates
        new_population = []
        for i, candidate in enumerate(population):
            # Local optimization with SLSQP for each candidate
            try:
                res = minimize(
                    objective,
                    candidate.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                if res.success:
                    new_candidate = res.x.copy()
                else:
                    new_candidate = candidate.copy()
            except:
                new_candidate = candidate.copy()
                
            # Ensure feasibility
            new_candidate = np.clip(new_candidate, 0, 1)
            h_sum = np.sum(new_candidate)
            if abs(h_sum - target_sum) > 1e-10:
                new_candidate = new_candidate * target_sum / h_sum
            
            # Add noise for diversity if similar to best
            if np.random.random() < 0.3 or iteration % 10 == 0:
                noise_level = 0.02 + 0.02 * (iteration / 100)
                new_candidate = new_candidate + np.random.normal(0, noise_level, n_points)
                new_candidate = np.clip(new_candidate, 0, 1)
                h_sum = np.sum(new_candidate)
                if abs(h_sum - target_sum) > 1e-10:
                    new_candidate = new_candidate * target_sum / h_sum
                    
            new_population.append(new_candidate)
            
        population = new_population
        
        # Global perturbation every 10 iterations
        if iteration % 10 == 0: 
            # Perturb best solution
            best_h = best_h + np.random.normal(0, 0.015, n_points)
            best_h = np.clip(best_h, 0, 1)
            h_sum = np.sum(best_h)
            if abs(h_sum - target_sum) > 1e-10:
                best_h = best_h * target_sum / h_sum
                
            # Add it back to population
            if len(population) > 0:
                population[np.random.randint(0, len(population))] = best_h.copy()
        
        # Early stopping if no improvement
        if no_improvement_count > 100:
            break
            
        iteration += 1
    
    # Ensure final solution is valid
    if np.any(best_h < 0) or np.any(best_h > 1):
        best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
