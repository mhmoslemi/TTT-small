import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_tenure = 20
    tabu_list = []
    max_iter = 2000
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def get_neighbors(h):
        # Generate neighbors by flipping pairs
        neighbors = []
        # Try flipping each 0 to 1 and 1 to 0
        for i in range(n_points):
            for j in range(i+1, n_points):
                if h[i] == 0 and h[j] == 1:
                    # Flip to create neighbor
                    h_new = h.copy()
                    h_new[i] = 1
                    h_new[j] = 0
                    neighbors.append(h_new)
                elif h[i] == 1 and h[j] == 0:
                    h_new = h.copy()
                    h_new[i] = 0
                    h_new[j] = 1
                    neighbors.append(h_new)
        return neighbors
    
    # Make sure initial solution is correct
    best_h = project(best_h)
    best_c5 = evaluate_c5(best_h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        neighbors = get_neighbors(best_h)
        if not neighbors:
            continue
            
        # Evaluate all neighbors
        neighbor_evals = []
        for neighbor in neighbors:
            neighbor = project(neighbor)
            c5 = evaluate_c5(neighbor, dx)
            neighbor_evals.append((neighbor, c5))
            
        # Find best non-tabu neighbor
        best_neighbor = None
        best_c5_neighbor = float('inf')
        
        for neighbor, c5 in neighbor_evals:
            # Check if neighbor is in tabu list
            is_tabu = any(np.array_equal(neighbor, h_tabu) for h_tabu, _ in tabu_list)
            if not is_tabu and c5 < best_c5_neighbor:
                best_neighbor = neighbor
                best_c5_neighbor = c5
        
        # If we found a better neighbor, update
        if best_neighbor is not None:
            best_h = best_neighbor
            best_c5 = best_c5_neighbor
            # Add to tabu list
            tabu_list.append((best_h.copy(), best_c5))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        else:
            # No improvement found, still update tabu list with current solution
            tabu_list.append((best_h.copy(), best_c5))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
    
    # Final local optimization around best solution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
        
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=con,
        tol=1e-6,
        options={'maxiter': 100, 'disp': False}
    )
    
    if result.success:
        h_opt = project(result.x)
        c5_opt = evaluate_c5(h_opt, dx)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
