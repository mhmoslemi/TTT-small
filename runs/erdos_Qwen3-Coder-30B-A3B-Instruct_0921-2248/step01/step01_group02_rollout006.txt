import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        n_points = len(h_parent)
        # Upsample to target resolution
        x_orig = np.linspace(0, 2, n_points)
        x_target = np.linspace(0, 2, 100)
        f = interp1d(x_orig, h_parent, kind='linear', fill_value='extrapolate')
        h_parent = f(x_target)
        n_points = len(h_parent)
    else:
        n_points = 100
        h_parent = np.zeros(n_points)
        h_parent[:n_points//2] = 1
    
    # Population and GA parameters
    pop_size = 50
    max_gen = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    
    # Initialize population
    def create_individual():
        # Start with parent
        h = h_parent.copy()
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        return h
    
    def population_create():
        pop = [create_individual() for _ in range(pop_size)]
        return pop
    
    def fitness(h):
        return evaluate_erdos_solution(h)
    
    def selection_tournament(pop, fitnesses, k=3):
        selected = []
        for _ in range(len(pop)):
            idxs = np.random.choice(len(pop), k)
            best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
            selected.append(pop[best_idx].copy())
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Single point crossover
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Blend crossover with alpha=0.5
        alpha = 0.5
        for i in range(n_points):
            min_val = min(parent1[i], parent2[i])
            max_val = max(parent1[i], parent2[i])
            range_val = max_val - min_val
            if range_val > 0:
                child1[i] = min_val - alpha * range_val + np.random.rand() * (2 * alpha * range_val)
                child2[i] = min_val - alpha * range_val + np.random.rand() * (2 * alpha * range_val)
        
        # Repair and re-normalize
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)
        sum1 = np.sum(child1)
        sum2 = np.sum(child2)
        child1 = child1 * (n_points / 2) / sum1 if sum1 > 0 else child1
        child2 = child2 * (n_points / 2) / sum2 if sum2 > 0 else child2
        
        return child1, child2
    
    def mutate(h):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    # Gaussian perturbation
                    h_mut[i] += np.random.normal(0, 0.1)
                else:
                    # Random reset
                    h_mut[i] = np.random.rand()
        # Ensure feasibility
        h_mut = np.clip(h_mut, 0, 1)
        sum_h = np.sum(h_mut)
        if sum_h > 0:
            h_mut = h_mut * (n_points / 2) / sum_h
        return h_mut
    
    # Main GA loop
    population = population_create()
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(max_gen):
        # Evaluate fitness
        fitnesses = [fitness(individual) for individual in population]
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitnesses[min_idx]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism
        new_pop = population[:elite_size]
        
        # Selection and reproduction
        selected = selection_tournament(population, fitnesses)
        for i in range(elite_size, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.extend([child1, child2])
        
        # Trim to population size if needed
        population = new_pop[:pop_size]
        
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Final evaluation
    if best_h is None:
        # Fallback to last population
        fitnesses = [fitness(individual) for individual in population]
        min_idx = np.argmin(fitnesses)
        best_h = population[min_idx]
        best_c5 = fitnesses[min_idx]
    
    # Ensure final constraints
    sum_h = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h if sum_h > 0 else best_h
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
