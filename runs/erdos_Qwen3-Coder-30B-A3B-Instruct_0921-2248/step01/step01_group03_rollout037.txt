import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:

        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_tenure = 20
    tabu_list = []
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    # Helper for project to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Flip a small number of entries
    def get_neighbor(h, n_flips=2):
        h_new = h.copy()
        indices = np.random.choice(n_points, size=n_flips, replace=False)
        for i in indices:
            h_new[i] = 1 - h_new[i]
        return project(h_new)
    
    # Evaluate objective
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate neighbor
        neighbor = get_neighbor(best_h)
        
        # Check tabu list
        neighbor_tuple = tuple(neighbor)
        if neighbor_tuple in tabu_list:
            continue
        
        # Evaluate neighbor
        c5_val = objective(neighbor)
        
        # Accept if better or improve the list
        if c5_val < best_c5:
            best_h = neighbor
            best_c5 = c5_val
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
        
        # Update tabu list
        tabu_list.append(neighbor_tuple)
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # If no improvement for a while, restart with better neighbor
        if iter_since_improvement > max_no_improvement:
            # Try a random improvement of best_h
            h_pert = best_h + np.random.normal(0, 0.02, n_points)
            h_pert = project(h_pert)
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
                tol=1e-6,
                options={'maxiter': 50, 'disp': False}
            )
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
            iter_since_improvement = 0
    
    return (best_h, best_c5, n_points)
