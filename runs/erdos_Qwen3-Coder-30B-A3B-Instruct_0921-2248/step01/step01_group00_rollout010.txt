import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value of the correlation (scaled by dx)
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create one
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Create a binary vector
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1
    
    # Create binary decision variables
    h = cp.Variable(n_points, boolean=True)
    C = cp.Variable()
    
    # Constraints: sum of h equals n_points/2, and C >= 0
    constraints = [cp.sum(h) == n_points/2, C >= 0]
    
    # Compute the overlap for each shift
    for k in range(1, n_points):
        # Compute the sum of h_i * h_{i+k} for valid indices
        z = cp.Variable(n_points - k, boolean=True)
        # Add constraints for z_i = h_i * h_{i+k}
        for i in range(n_points - k):
            constraints.append(z[i] <= h[i])
            constraints.append(z[i] <= h[i + k])
            constraints.append(z[i] >= h[i] + h[i + k] - 1)
        
        # The overlap for this shift
        S_k = cp.sum(h[:n_points - k]) - cp.sum(z)
        constraints.append(S_k <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)
    
    # Try solving with different solvers
    solver_list = [cp.SCIP, cp.GLPK_MI, cp.CBC]
    best_c5 = float('inf')
    best_h = None
    
    # Try to solve with the available solvers
    for solver in solver_list:
        try:
            # Allow only 80% of the budget for solving
            problem.solve(solver=solver, time_limit=budget_s * 0.8)
            if problem.status in ["optimal", "optimal_inaccurate"]:
                sol_h = [int(v.value) for v in h]
                c5_val = evaluate_erdos_solution(sol_h)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = sol_h
            break  # If first solver succeeds, return it
        except Exception as e:
            continue

    # Fallback to heuristic search if MILP fails
    if best_h is None:
        def obj_func(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h)

        def constraint_eq(x):
            return np.sum(x) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        opt_options = {'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
        
        best_h = h0.copy()
        best_c5 = obj_func(best_h)
        
        # Run multiple local optimizations with different starting points
        for _ in range(10):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb the current best
            h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            try:
                res = minimize(obj_func, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options=opt_options)
                if res.success and res.fun < best_c5:
                    h_opt = np.clip(res.x, 0, 1)
                    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                    c5_val = obj_func(h_opt)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_opt.copy()
            except Exception:
                pass
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
