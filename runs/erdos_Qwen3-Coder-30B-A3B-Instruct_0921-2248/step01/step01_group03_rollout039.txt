import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

        # Place 1s in a spread pattern to satisfy integral constraint
        ones_count = n_points // 2
        positions = np.random.choice(n_points, size=ones_count, replace=False)
        h0[positions] = 1.0
    
    # Ensure the sum constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 20
    max_iterations = 10000
    iteration = 0
    
    # Helper to project onto feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def get_neighbor(h, iterations):
        # Make a small change
        h_new = h.copy()
        # Flip a small chunk of 2-5 elements
        n_flips = random.randint(2, 5)
        # Ensure two different values
        indices = random.sample(range(n_points), n_flips)
        for i in indices:
            h_new[i] = 1 - h_new[i]
        
        # Project back to feasible region
        h_new = project(h_new)
        return h_new
    
    # Main Tabu search loop
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 1:
        # Get neighbors
        neighbors = []
        for _ in range(10):
            neighbor = get_neighbor(best_h, iteration)
            neighbors.append(neighbor)
        
        # Evaluate neighbors
        c5_values = [evaluate_c5(h, dx) for h in neighbors]
        
        # Find best non-tabu neighbor
        best_neighbor_idx = -1
        best_c5_neighbor = float('inf')
        
        for idx, (h, c5) in enumerate(zip(neighbors, c5_values)):
            # Check if this solution or its reverse is in tabu list
            tabu_found = False
            for tabu_h, tabu_c5 in tabu_list:
                if np.allclose(h, tabu_h, atol=1e-6) or np.allclose(h, 1 - tabu_h, atol=1e-6):
                    tabu_found = True
                    break
            
            if not tabu_found and c5 < best_c5_neighbor:
                best_c5_neighbor = c5
                best_neighbor_idx = idx
        
        # Update best if improved
        if best_neighbor_idx != -1:
            best_h = neighbors[best_neighbor_idx]
            best_c5 = best_c5_neighbor
            
            # Add current solution to tabu list
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append((best_h.copy(), best_c5))
        
        # Occasionally perform a local search
        if iteration % 100 == 0:
            # Try continuous optimization around current best
            bounds = [(0, 1) for _ in range(n_points)]

                return evaluate_c5(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            con = {'type': 'eq', 'fun': constraint_sum}
            
            # Perturb a bit and optimize
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = project(h_pert)
            
            try:
                result = minimize(
                    objective,
                    h_pert,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=con,
                    tol=1e-6,
                    options={'maxiter': 50, 'disp': False}

                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
            except:
                pass  # Ignore optimization failures
        
        iteration += 1
    
    return (best_h, best_c5, n_points)
