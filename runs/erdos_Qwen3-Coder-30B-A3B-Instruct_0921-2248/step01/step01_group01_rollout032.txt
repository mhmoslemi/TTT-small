import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Coarse-to-fine optimization approach
    resolutions = [50, 75, 100, 125, 150]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize from parent if available, otherwise random
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values, dtype=float)
            if len(h) != n_points:
                # Interpolate or pad
                h_interp = np.interp(np.linspace(0, 2, n_points), 
                                     np.linspace(0, 2, len(h)), h)
                h = h_interp
        else:
            h = np.random.rand(n_points)
            
        # Project to satisfy sum constraint
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
        
        # Objective function
        def objective(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode="full")
            max_overlap = np.max(corr)
            return max_overlap * dx
            
        # Constraint function
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Multi-start optimization with adaptive perturbations
        for start in range(10):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Use previous best as initialization if available
            if start > 0 and best_h is not None:
                h_trial = best_h.copy() + np.random.normal(0, 0.02, n_points)
            else:
                h_trial = h.copy()
                
            # Ensure feasibility after perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * target_sum / np.sum(h_trial)
            
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    h_opt = res.x.copy()
                    c5 = objective(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        best_n = n_points
                        
            except Exception:
                pass
                
            # Kick-start with new random solution
            h_trial = np.random.rand(n_points)
            h_trial = h_trial * target_sum / np.sum(h_trial)
            
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    h_opt = res.x.copy()
                    c5 = objective(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        best_n = n_points
                        
            except Exception:
                pass
    
    # Final evaluation with best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, best_n)
