import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Create initial feasible solution if needed
    h0 = np.clip(h0, 0, 1)
    h_sum = np.sum(h0)
    if h_sum > 0:
        h0 = h0 * (n_points / 2) / h_sum
    
    # Tabu search parameters
    tabu_tenure = 20
    tabu_list = []
    max_iter = 5000
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def get_neighbors(h, n_flip=2):
        neighbors = []
        indices = np.where(h > 0)[0]
        if len(indices) < n_flip:
            return neighbors
            
        # Generate neighbor by flipping pairs of 0s and 1s
        for i in range(min(50, len(indices))):
            idx_1 = indices[np.random.randint(len(indices))]
            idx_2 = np.random.randint(n_points)
            if h[idx_2] > 0:  # Swap 1 with 0
                h_new = h.copy()
                h_new[idx_1] = 0
                h_new[idx_2] = 1
                neighbors.append(h_new)
        return neighbors
    
    def local_optimize(h, max_iter=20):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        con = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        for _ in range(max_iter):
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 10, 'disp': False}
            )
            if result.success:
                h = project(result.x)
            else:
                break
        return h
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get neighbors
        neighbors = get_neighbors(current_h, n_flip=2)
        if not neighbors:
            continue
            
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for neighbor in neighbors:
            if neighbor is None:
                continue
            c5 = evaluate_c5(neighbor, dx)
            tabu_key = tuple(np.round(neighbor, decimals=6))
            if tabu_key not in [tuple(np.round(h, decimals=6)) for h, _ in tabu_list]:
                if c5 < best_neighbor_c5:
                    best_neighbor = neighbor
                    best_neighbor_c5 = c5
        
        # Accept best neighbor if better
        if best_neighbor is not None:
            current_h = best_neighbor
            current_c5 = best_neighbor_c5
            
            # Local optimization to refine
            refined_h = local_optimize(current_h, max_iter=5)
            refined_c5 = evaluate_c5(refined_h, dx)
            
            if refined_c5 < current_c5:
                current_h = refined_h
                current_c5 = refined_c5
            
            # If better than best, update
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            
            # Add to tabu list
            tabu_list.append((current_h.copy(), current_c5))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        else:
            # Move to random neighbor
            current_h = np.random.rand(n_points)
            current_h = project(current_h)
            current_c5 = evaluate_c5(current_h, dx)
    
    # Final optimization of best solution
    final_h = project(best_h)
    final_h = local_optimize(final_h, max_iter=20)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h, final_c5, n_points)
