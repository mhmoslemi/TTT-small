import numpy as np
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 50
        h = np.zeros(n_points)
        # Create two clusters of 1s for better diversity
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Project h to ensure it satisfies the sum constraint
    def project_h(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2.0) / np.sum(x)
        return x
    
    # Simulated Annealing
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    
    # Upsample if needed
    if n_points < 100:
        target_n = 100
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, target_n)
        h_interp = np.interp(x_new, x_orig, h)
        h = h_interp
        n_points = target_n
    
    if n_points < 200:
        target_n = 200
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, target_n)
        h_interp = np.interp(x_new, x_orig, h)
        h = h_interp
        n_points = target_n
    
    T = T0
    temp_steps = 0
    
    while T > T_min and time.time() - start_time < budget_s - 5:
        temp_steps += 1
        # Perform multiple perturbations at current temperature
        for step in range(100):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Create candidate solution
            h_new = h.copy()
            # Pick a random index
            idx = np.random.randint(0, n_points)
            # Perturb by a random amount
            delta = np.random.normal(0, T * 0.1)
            h_new[idx] += delta
            # Project back to feasible set
            h_new = project_h(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept or reject based on simulated annealing probability
            if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / T):
                h = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
        
        # Cool down
        T *= alpha
        
    # Final refinement
    # Use SLSQP for final polishing
    from scipy.optimize import minimize
    
    def objective(x):
        return evaluate_erdos_solution(x)
    
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    if result.success:
        final_h = project_h(result.x)
    else:
        final_h = best_h.copy()
    
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
