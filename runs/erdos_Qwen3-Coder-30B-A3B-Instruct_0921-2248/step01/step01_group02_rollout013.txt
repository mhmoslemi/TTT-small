import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a step function with more optimal placement
        n_points = 200
        h_init = np.zeros(n_points)
        # Place 1s in first half and 0s in second half
        h_init[:n_points//2] = 1
        # Randomize placement to encourage better optimization
        np.random.shuffle(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try different optimization approaches
    # Basic gradient-based optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    # Additional global optimizations
    # Try simulated annealing approach or random restarts
    for i in range(100):  # Limited but robust iteration
        if time.time() - start_time > budget_s - 2:
            break
            
        # Random perturbation
        h_test = best_h.copy()
        # Perturb some elements
        idxs = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
        for idx in idxs:
            h_test[idx] = np.clip(h_test[idx] + np.random.normal(0, 0.05), 0, 1)
        
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        c5_test = evaluate_erdos_solution(h_test)
        if c5_test < best_c5:
            best_h = h_test.copy()
            best_c5 = c5_test
    
    # Refine with optimization
    try:
        res = minimize(objective, best_h.copy(), method='SLSQP', 
                        bounds=bounds, constraints=constraints, 
                        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6})
        if res.success:
            h_opt = res.x.copy()
            h_opt = np.clip(h_opt, 0, 1)
            # Project to satisfy constraint sum(h) == n_points/2
            h_sum = np.sum(h_opt)
            h_opt = h_opt * (n_points / 2) / h_sum
            c5 = evaluate_erdos_solution(h_opt)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5
    except Exception:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
