import numpy as np
from scipy.interpolate import interp1d
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Population and GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 3
    elite_size = 10
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        n_points = len(h_parent)
        # Upsample to desired resolution
        x_parent = np.linspace(0, 2, n_points)
        x_target = np.linspace(0, 2, 200)
        f = interp1d(x_parent, h_parent, kind='linear', fill_value='extrapolate')
        h_target = f(x_target)
        h_init = h_target
        n_points = len(h_target)
    else:
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
    
    # Initialize population
    def create_individual():
        h = h_init.copy()
        # Small random mutations for diversity
        for i in range(n_points):
            if np.random.random() < 0.1:
                h[i] += np.random.normal(0, 0.05)
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        return h
    
    population = [create_individual() for _ in range(pop_size)]
    fitness = [evaluate_erdos_solution(ind) for ind in population]
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    while time.time() - start_time < budget_s - 1:
        # Selection
        def tournament_selection():
            idxs = np.random.choice(pop_size, size=tournament_size, replace=False)
            best = idxs[np.argmin([fitness[i] for i in idxs])]
            return population[best]
        
        # Crossover
        def crossover(parent1, parent2):
            if np.random.random() > crossover_rate:
                return parent1.copy(), parent2.copy()
            i = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:i], parent2[i:]])
            child2 = np.concatenate([parent2[:i], parent1[i:]])
            return child1, child2
        
        # Mutation
        def mutate(ind):
            new_ind = ind.copy()
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    # Random reset
                    if np.random.random() < 0.5:
                        new_ind[i] = np.random.random()
                    else:
                        # Gaussian perturbation
                        new_ind[i] += np.random.normal(0, 0.05)
            # Ensure feasibility constraints
            new_ind = np.clip(new_ind, 0, 1)
            h_sum = np.sum(new_ind)
            new_ind = new_ind * (n_points / 2) / h_sum
            return new_ind

        # Generate new population
        new_population = []
        # Keep elite
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            p1 = tournament_selection()
            p2 = tournament_selection()
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1)
            c2 = mutate(c2)
            new_population.extend([c1, c2])
        
        # Trim to exact size
        new_population = new_population[:pop_size]
        
        # Evaluate new population
        new_fitness = [evaluate_erdos_solution(ind) for ind in new_population]
        
        # Update best
        best_new_idx = np.argmin(new_fitness)
        if new_fitness[best_new_idx] < best_c5:
            best_h = new_population[best_new_idx].copy()
            best_c5 = new_fitness[best_new_idx]
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Adaptive mutation rate based on stagnation
        if len(new_population) > 10:
            worst = np.argmin(fitness)
            best = np.argmin(fitness)
            if abs(fitness[worst] - fitness[best]) < 1e-6:
                mutation_rate = min(0.3, mutation_rate * 1.1)
            else:
                mutation_rate = max(0.05, mutation_rate * 0.99)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
