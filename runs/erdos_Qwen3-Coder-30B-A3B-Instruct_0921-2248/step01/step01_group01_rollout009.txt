import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        """Compute the maximum overlap integral."""
        # Compute correlation for all lags
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx

    def optimize_single(h_vec):
        """Optimize a single point using CVXPY."""
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-8, max_iters=1000)
            if prob.status == 'optimal':
                return h_var.value
        except:
            pass
        
        # Fallback to scipy if CVXPY doesn't work
        def obj_func(x):
            return np.sum(x * (1 - x))
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj_func, h_vec, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            return res.x
        return h_vec

    def evolve_population(population):
        """Evolve a population of solutions."""
        new_pop = []
        for i, h in enumerate(population):
            # Mutation with some probability
            if np.random.rand() < 0.3:
                h_mut = h.copy() + np.random.normal(0, 0.02, n_points)
                h_mut = np.clip(h_mut, 0, 1)
                h_mut_sum = np.sum(h_mut)
                if abs(h_mut_sum - target_sum) > 1e-10:
                    h_mut = h_mut * target_sum / h_mut_sum
                h = h_mut
                
            # Local optimization
            h_refined = optimize_single(h)
            new_pop.append(h_refined)
        return new_pop

    def get_best_solution(population):
        """Get the best solution based on overlap."""
        best_h = None
        best_c5 = float('inf')
        for h in population:
            c5 = compute_overlap(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        return best_h, best_c5

    # Initialize population
    population = []
    for i in range(10):
        if i == 0:
            h_init = h.copy()
        else:
            h_init = h + np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_sum = np.sum(h_init)
        if abs(h_sum - target_sum) > 1e-10:
            h_init = h_init * target_sum / h_sum
        population.append(h_init)

    best_h = None
    best_c5 = float('inf')
    no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evolve population
        population = evolve_population(population)
        
        # Check for best solution
        curr_best_h, curr_c5 = get_best_solution(population)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_best_h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Check if we should increase resolution
        if iteration % 50 == 0 and iteration > 0 and n_points < 150:
            n_points = 150
            old_h = best_h.copy()
            best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(old_h)), old_h)
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
        
        # Check time
        if time.time() - start_time > budget_s - 1:
            break
            
        iteration += 1
        if no_improvement > 100:
            break

    # Final optimization of best solution
    if best_h is not None:
        final_h = optimize_single(best_h)
        final_c5 = compute_overlap(final_h)
    else:
        final_h = h.copy()
        final_c5 = compute_overlap(final_h)

    return (final_h.tolist(), final_c5, len(final_h))
