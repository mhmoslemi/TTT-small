import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback initial construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Multi-resolution optimization with careful restarts
    upscale_factors = [2, 2, 12]
    max_iter_per_stage = 150  # seconds per stage
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # track current h and size for upscaling
    current_h = h.copy()
    current_n = n_points
    
    # helper to add noise
    def add_noise(h_vec, noise_level):
        noise = np.random.uniform(-noise_level, noise_level, len(h_vec))
        return np.clip(h_vec + noise, 0, 1)
    
    # helper to upscale and maintain sum constraint
    def upscale(h_vec, factor):
        new_len = len(h_vec) * factor
        new_h = np.zeros(new_len)
        for i in range(len(h_vec)):
            new_h[i * factor:(i+1) * factor] = h_vec[i]
        # Scale to maintain sum constraint
        new_h = new_h * target_sum / np.sum(new_h)
        return new_h, new_len
    
    # helper: constraint function for scipy optimization
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # loop stages
    for stage, upscale_factor in enumerate(upscale_factors):
        if time.time() - start_time > budget_s - 5:
            break
            
        # upscale (repeating values is simple and maintains feasibility)
        new_h, new_n = upscale(current_h, upscale_factor)
        current_h = new_h
        current_n = new_n
        dx = 2.0 / current_n
        target_sum = current_n / 2.0
        
        # add initial noise
        noise_level = 0.1 * (0.8 ** stage)  # decrease over stages
        current_h = add_noise(current_h, noise_level)
        
        # random restart at beginning of each stage
        bounds = [(0.0, 1.0)] * current_n
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # optimize with LBFGS-B
        stage_end = time.time() + max_iter_per_stage
        while time.time() < stage_end:
            # try to optimize
            try:
                res = minimize(
                    lambda x: np.max(np.correlate(x, 1 - x, mode="full")) * dx,
                    current_h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                if res.success:
                    h_new = np.clip(res.x, 0, 1)
                    # enforce sum exactly
                    h_new *= target_sum / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                    current_h = h_new.copy()
                else:
                    # fallback if optimization fails
                    current_h = add_noise(current_h, 0.05)
                    current_h *= target_sum / np.sum(current_h)
            except Exception as e:
                # fallback to noise
                current_h = add_noise(current_h, 0.05)
                current_h *= target_sum / np.sum(current_h)

            # occasional random restart
            if np.random.random() < 0.1:
                current_h = add_noise(current_h, 0.1)
                current_h *= target_sum / np.sum(current_h)
    
    # final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), len(final_h))
