def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: uniform step function
        n_points = 50
        h = np.zeros(n_points)
        h[::2] = 1.0  # Place 1s at even indices
        h = h * (n_points / 2.0) / np.sum(h)  # Scale to sum = n_points/2
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization rounds with increasing resolution
    resolutions = [50, 100, 150]
    noise_scales = [0.05, 0.025, 0.01]
    
    for res, noise_scale in zip(resolutions, noise_scales):
        if time.time() - start_time > budget_s - 1:
            break
            
        n_points = res
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Interpolate from previous solution if available
        if res > len(h):
            h_new = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h)), h)
            h = h_new
        
        # Perturb and optimize
        for _ in range(3):  # multiple restarts
            if time.time() - start_time > budget_s - 1:
                break
                
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
                
            try:
                res_opt = minimize(
                    objective,
                    h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6}
                )
                
                if res_opt.success:
                    h = res_opt.x.copy()
                    c5 = objective(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        
                    # Small perturbation
                    noise = np.random.normal(0, noise_scale, n_points)
                    h = h + noise
            except:
                pass
                
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum

    if time.time() - start_time < budget_s - 1:
        n_points = 150

        target_sum = n_points / 2.0
        bounds = [(0, 1) for _ in range(n_points)]
        
        if len(best_h) < n_points:
            h_new = np.interp(np.linspace(0, 2, n_points),

            h = h_new
        else:
            h = best_h[:n_points]

            res_opt = minimize(
                objective,
                h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8}

                c5 = objective(h)

                    best_h = h.copy()
        except:
            pass
    
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if abs(final_sum - target_sum) > 1e-10:
        final_h = final_h * target_sum / final_sum
        
    final_c5 = objective(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
