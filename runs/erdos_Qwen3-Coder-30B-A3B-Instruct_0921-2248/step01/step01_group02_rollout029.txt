import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a piecewise constant construction for better starting point
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        
    # Initialize variables
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Gradient-based optimization using scipy
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Special case: try known good constructions
    # Searches in frequency domain to find optimal solution
    x = Variable(n_points)
    # Set up constraints
    constraints = [x >= 0, x <= 1, sum(x) == n_points / 2]
    
    # Try to optimize with periodic boundary conditions to prevent local trap
    n_iter = 0
    max_iter = 1000  # limit iterations
    
    # Simple alternating optimization between local search and global projection
    while time.time() - start_time < budget_s - 1 and n_iter < max_iter:
        # Try local optimization
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        
        try:
            res = minimize(objective, h_rand, method='SLSQP', 
                          bounds=bounds, constraints=constraints, 
                          options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-5})
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
            else:
                h_project = np.clip(h_rand, 0, 1)
                h_project = h_project * (n_points / 2) / np.sum(h_project)
                c5 = evaluate_erdos_solution(h_project)
                if c5 < best_c5:
                    best_h = h_project.copy()
                    best_c5 = c5
                    
        except Exception:
            pass
        
        # Try another exploration approach with different initializations
        h_f = np.ones(n_points) * 0.5
        # Randomly flip some values
        idx = np.random.choice(n_points, size=n_points//4, replace=False)
        h_f[idx] = 1 - h_f[idx]
        
        # Project and evaluate
        h_f = np.clip(h_f, 0, 1)
        h_f = h_f * (n_points / 2) / np.sum(h_f)
        c5 = evaluate_erdos_solution(h_f)
        if c5 < best_c5:
            best_h = h_f.copy()
            best_c5 = c5
            
        n_iter += 1
        
        # Early termination if sufficient improvement
        if time.time() - start_time > budget_s * 0.9:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
