import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Shuffled binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Increase resolution and interpolate
    n_upsample = 500
    x_orig = np.linspace(0, 2, n_points)
    x_new = np.linspace(0, 2, n_upsample)
    h_upsampled = np.interp(x_new, x_orig, h)
    h = h_upsampled
    n_points = n_upsample
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0

    def obj_func(h_vec):
        # Make sure it's in the valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx

    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Use Nelder-Mead optimization with restarts
    for iteration in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random restart
        h_restart = np.random.rand(n_points)
        h_restart = h_restart * target_sum / np.sum(h_restart)
        
        # Nelder-Mead optimization
        try:
            res = minimize(obj_func, h_restart, method='Nelder-Mead', 
                           options={'maxiter': 200, 'initial_simplex': None, 'xatol': 1e-6, 'fatol': 1e-6})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint with projection
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception as e:
            pass
        
        # Perturb best solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        h_perturbed = h_perturbed * target_sum / current_sum
        c5_perturbed = obj_func(h_perturbed)
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
