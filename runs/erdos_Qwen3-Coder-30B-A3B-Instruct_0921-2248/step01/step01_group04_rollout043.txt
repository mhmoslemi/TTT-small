import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def crossover(parent1, parent2, rate=0.8):
    """Perform segment-based crossover"""
    n = len(parent1)
    if np.random.rand() > rate:
        return parent1.copy(), parent2.copy()
    
    # Choose crossover point
    point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, rate=0.05):
    """Mutate h by perturbing some elements"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < rate:
            h[i] += np.random.normal(0, 0.05)
    return h

def tournament_selection(population, fitness, k=5):
    """Select one individual using tournament selection"""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Default to small, as per requirements
    n_pop = 50
    n_generations = 500
    
    # Initialize population
    population = []
    fitness = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        population.append(h_start.copy())
        fitness.append(evaluate_erdos_solution(h_start))
    else:
        h_start = np.ones(n_points) * 0.5
        population.append(h_start)
        fitness.append(evaluate_erdos_solution(h_start))
        
    # Generate diverse initial population
    for _ in range(n_pop - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    stagnation_count = 0
    stagnation_limit = 50
    
    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection and reproduction
        new_population = []
        
        # Elitism: keep best
        new_population.append(best_h.copy())
        
        for _ in range(n_pop - 1):
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, rate=0.05)
            child2 = mutate(child2, rate=0.05)
            
            # Projection
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.append(child1)
            new_population.append(child2)
        
        # Evaluation of new population
        new_fitness = []
        for h in new_population:
            new_fitness.append(evaluate_erdos_solution(h))
        
        # Keep best individuals
        combined_pop = new_population + population
        combined_fit = new_fitness + fitness
        
        # Sort by fitness
        idxs = np.argsort(combined_fit)
        population = [combined_pop[i] for i in idxs[:n_pop]]
        fitness = [combined_fit[i] for i in idxs[:n_pop]]
        
        # Check for improvement
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Reset if stagnation
        if stagnation_count > stagnation_limit:
            for i in range(n_pop):
                h = np.random.rand(n_points)
                h = project_to_feasible(h, n_points)
                population[i] = h
                fitness[i] = evaluate_erdos_solution(h)
            stagnation_count = 0
    
    # Final polishing using SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    
    if result.success:
        best_h = result.x
        best_h = project_to_feasible(best_h, n_points)
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), best_c5, n_points)
