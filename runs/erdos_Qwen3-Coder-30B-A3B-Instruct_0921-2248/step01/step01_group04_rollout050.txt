import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use cosine basis expansion approach
    # Define basis functions
    def basis_func(x, m):
        return np.cos(m * np.pi * x / 2)

    M = 6
    coeffs = np.zeros(M + 1)  # including constant term

        # Project initial values to cosine basis
        # Simple approach: use FFT to get coefficients
        initial_h = np.array(initial_h_values)
        # Normalize to have integral 1
        initial_h = initial_h * (n_points / 2.0) / np.sum(initial_h)
        # Interpolate to our grid
        x_vals = np.linspace(0, 2, n_points)
        # Use least squares to fit cosine basis
        A = np.column_stack([np.ones(n_points)] + [basis_func(x_vals, m) for m in range(1, M+1)])
        coeffs = np.linalg.lstsq(A, initial_h, rcond=None)[0]

    max_iter = 1000
    learning_rate = 0.01
    
    def compute_h_from_coeffs(coeffs):
        x = np.linspace(0, 2, n_points)
        h_vals = coeffs[0] + np.sum([coeffs[m+1] * basis_func(x, m+1) for m in range(M)], axis=0)
        return np.clip(h_vals, 0, 1)

        h_vals = compute_h_from_coeffs(coeffs)

        # compute cross correlation manually in frequency domain for efficiency
        H = np.fft.fft(h_vals)
        H_conj = np.conj(H)
        one_minus_h = 1 - h_vals
        One_minus_h = np.fft.fft(one_minus_h)
        
        cross_corr = np.fft.ifft(H * One_minus_h)
        max_corr_idx = np.argmax(np.abs(cross_corr))
        max_corr = np.real(cross_corr[max_corr_idx])

        grad = np.zeros_like(coeffs)

            coeffs_pert[i] += eps
            h_pert = compute_h_from_coeffs(coeffs_pert)

            h_ref = h_vals_refine
