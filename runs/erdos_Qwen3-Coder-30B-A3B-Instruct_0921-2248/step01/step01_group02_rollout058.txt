import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a more carefully crafted initial construction
        n_points = 47
        h_init = np.zeros(n_points)
        # Put most mass in first half, but with some variation
        h_init[:n_points//2] = 1
        # Slightly randomize the boundary
        h_init[np.random.randint(0, n_points//2, 3)] = 0
        h_init[np.random.randint(n_points//2, n_points, 2)] = 1
        np.random.shuffle(h_init)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find maximum (scaled by dx = 2/n_points)
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    # Initialize best solution
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimizer parameters
    options = {'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}
    method = 'SLSQP'
    
    # Main optimization loop
    last_time = start_time
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Random restarts
        if iter_count % 10 == 0:
            h0 = np.random.rand(n_points)
            h0 = h0 * (n_points / 2) / np.sum(h0)
        elif np.random.random() < 0.3:
            h0 = np.copy(h_init)
            # Perturb slightly
            h0 += np.random.normal(0, 0.05, n_points)
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (n_points / 2) / np.sum(h0)
        else:
            h0 = best_h.copy()
            h0 += np.random.normal(0, 0.01, n_points)
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (n_points / 2) / np.sum(h0)
                        
        try:
            # Try multiple optimization runs
            for _ in range(3):
                res = minimize(objective, h0, method=method, bounds=bounds, 
                              constraints=constraints, options=options)

                if res.success:
                    h_opt = res.x.copy()
                    h_opt = np.clip(h_opt, 0, 1)
                    # Project to satisfy constraint
                    h_sum = np.sum(h_opt)
                    if h_sum > 0:
                        h_opt = h_opt * (n_points / 2) / h_sum
                    # Evaluate
                    c5 = evaluate_erdos_solution(h_opt)
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
                        break
                else:
                    h_projected = np.clip(h0, 0, 1)
                    h_sum = np.sum(h_projected)
                    if h_sum > 0:
                        h_projected = h_projected * (n_points / 2) / h_sum
                    c5 = evaluate_erdos_solution(h_projected)
                    if c5 < best_c5:
                        best_h = h_projected.copy()
                        best_c5 = c5
                        
        except Exception:
            # Fallback to projection
            h_projected = np.clip(h0, 0, 1)
            h_sum = np.sum(h_projected)
            if h_sum > 0:
                h_projected = h_projected * (n_points / 2) / h_sum
            c5 = evaluate_erdos_solution(h_projected)
            if c5 < best_c5:
                best_h = h_projected.copy()
                best_c5 = c5
        
        if time.time() - last_time > 2.0:
            last_time = time.time()
            if iter_count > 50 and time.time() - start_time > budget_s * 0.7:
                break
    
    # Return best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
