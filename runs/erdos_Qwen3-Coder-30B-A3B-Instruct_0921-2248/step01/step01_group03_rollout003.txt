import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

        # Adjust to meet constraint sum(h) = n_points/2
        h0 = np.clip(h0, 0, 1)
        h_sum = np.sum(h0)
        if h_sum > 0:
            h0 = h0 * (n_points / 2) / h_sum
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Optimization loop
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = project(h_pert)
        
        # Optimize with SLSQP
        try:
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 100, 'disp': False}
            )
            
            if result.success:

                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
        
        # Gradient-based refinement
        try:
            h_refined = best_h.copy()
            lr = 0.005
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:

                grad = np.zeros_like(h_refined)

                eps = 1e-4

                    c_plus = evaluate_c5(h_plus, dx)

        except:
            pass

    return (best_h, best_c5, n_points)
