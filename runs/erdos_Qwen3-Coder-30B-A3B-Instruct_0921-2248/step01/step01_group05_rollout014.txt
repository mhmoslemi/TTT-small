import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default: create a simple smooth initial guess
        n_points = 100
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
    
    # Scale initial vector to meet integral constraint
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    dx = 2.0 / n_points
    
    # Build the CVXPY optimization problem
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        C5 >= 0,
    ]
    
    # Add constraints for each shift k
    # We compute from k = 0 to n_points-1
    for k in range(n_points):
        # This is the sum over i of h[i] * (1 - h[i+k]) - we need to be careful
        # about indexing (including wraparound)
        expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h[i] * (1 - h[j])
        constraints.append(expr <= C5 * (target_sum))
    
    # Objective
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Solve with MOSEK (if available) or fallback to ECOS
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s*0.9)
    except Exception:
        try:
            # Fallback to ECOS if MOSEK unavailable
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            # Fallback to default solver
            try:
                prob.solve(verbose=False)
            except Exception:
                # If all solvers fail, return initial construction
                h_final = h_init
                c5_final = np.max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
                return (h_final.tolist(), float(c5_final), n_points)
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_final = h.value
        if h_final is None:
            h_final = h_init
        c5_final = C5.value
        if c5_final is None:
            c5_final = np.max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
    else:
        h_final = h_init
        c5_final = np.max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
    
    # Ensure feasibility
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    
    # Final evaluation with standard function
    c5_final = np.max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
    
    return (h_final.tolist(), float(c5_final), n_points)
