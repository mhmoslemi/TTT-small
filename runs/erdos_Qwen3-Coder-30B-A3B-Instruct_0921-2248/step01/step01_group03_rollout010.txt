from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

        if h_sum > 0:
            h = h * (n_points / 2) / h_sum

        h_new = h.copy()
        # try to flip two bits
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        # Flip the values
        h_new[i1] = 1 - h_new[i1]
        h_new[i2] = 1 - h_new[i2]
        # Project to ensure sum constraint
        h_new = project(h_new)
        return h_new

        for h_tabu, _ in tabu_list:
            if np.allclose(h_candidate, h_tabu, atol=1e-6):

        return False

    def update_tabu(h_candidate, c5_val, tabu_list):
        tabu_list.append((h_candidate.copy(), c5_val))
        if len(tabu_list) > max_tabu_size:
            tabu_list.pop(0)
        return tabu_list

        bounds = [(0, 1) for _ in range(n_points)]

            return np.sum(h) - n_points / 2

            return evaluate_c5(h, dx)

            h_start,
            method='SLSQP',

            constraints=con,
