```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 200
        # Start with binary mixture and shuffle
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach
    max_iterations = 500
    # Starting set of shifts
    K = set()
    # Add some random shifts to start
    num_initial_shifts = min(50, 2 * n_points - 1)
    for _ in range(num_initial_shifts):
        K.add(np.random.randint(0, 2 * n_points - 1))
    K = list(K)
    
    # Track feasibility
    K_set = set(K)
    
    iterations = 0
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 5:
        iterations += 1
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Create linear program
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        for k in K:
            # Compute the overlap integral for shift k
            # h[i] * (1 - h[(i+k)%n_points])
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(C5 >= overlap_expr * (2.0 / n_points))
            
        # Bounds on variables
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            # If solver fails, try with a different one or fallback to bounds
            try:
                prob.solve(solver=cp.GLOP, verbose=False)
            except:
                # Fallback to bounds and gradient based method
                break
        
        # Extract solution
        if prob.status == cp.OPTIMAL:
            h_new = h_vars.value
            c5_new = C5.value
            
            # Project h_new back to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            # Evaluate
            c5_eval = evaluate_erdos_solution(h_new)
            
            # Accept better solutions
            if c5_eval < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_eval
                
            # Find the most constraining shift not in K
            max_relax = 0
            best_k = None
            for k in range(2 * n_points - 1):
                if k not in K_set:
                    overlap_expr = 0
                    for i in range(n_points):
                        j = (i + k) % n_points
                        overlap_expr += h_new[i] * (1 - h_new[j])
                    rel = overlap_expr * (2.0 / n_points) - c5_eval
                    if rel > max_relax:
                        max_relax = rel
                        best_k = k
                        
            if best_k is not None and max_relax > 1e-5:
                K.append(best_k)
                K_set.add(best_k)
            else:
                # Try polishing with SLSQP on the best solution if not improving significantly
                break
        else:
            # If solver fails, break and finish with polish
            break
    
    # Final polish with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimize with SLSQP based on current best h
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    if result.success:
        h_final = result.x
    else:
        h_final = best_h
    
    # Final projection to ensure constraints
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    # Evaluate final
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (list(h_final), final_c5, n_points)
```
