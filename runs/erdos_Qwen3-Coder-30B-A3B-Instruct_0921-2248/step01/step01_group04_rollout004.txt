import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)

    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define sine/cosine basis functions
    def basis_func(x, m):
        return np.cos(m * np.pi * x / 2)
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.01
    decay_every = 200
    patience = 50
    
    # Initialize coefficients
    M = 6  # Number of cosine basis functions (excluding constant)
    a = np.random.normal(0, 0.01, M+1) # +1 for constant term
    a[0] = 0.5  # Start with mean value
    
    # Normalize to have integral 1
    def coeffs_to_h(coeffs, n_points):
        x = np.linspace(0, 2, n_points)
        h_vals = coeffs[0] + np.sum([coeffs[m+1] * basis_func(x, m+1) for m in range(M)], axis=0)
        # Clip and normalize
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2.0) / np.sum(h_vals)
        return h_vals
    
    # Gradient computation via finite difference
    def compute_grad(coeffs, n_points, epsilon=1e-4):
        grad = np.zeros_like(coeffs)
        base_c5 = evaluate_erdos_solution(coeffs_to_h(coeffs, n_points))
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += epsilon
            c5_plus = evaluate_erdos_solution(coeffs_to_h(coeffs_plus, n_points))
            grad[i] = (c5_plus - base_c5) / epsilon
        return grad
    
    # Optimizer loop
    no_improvement_count = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Convert coefficients to h
        h_current = coeffs_to_h(a, n_points)
        
        # Evaluate current solution
        c5_current = evaluate_erdos_solution(h_current)
        
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adapt learning rate based on no improvement
        if no_improvement_count % decay_every == 0 and no_improvement_count > 0:
            learning_rate *= 0.5
            
        # Compute gradient using finite difference
        grad = compute_grad(a, n_points)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Update coefficients
        a = a - learning_rate * grad
        
        # Re-project h to ensure area constraint
        h_current = coeffs_to_h(a, n_points)

    return (list(best_h), best_c5, n_points)
