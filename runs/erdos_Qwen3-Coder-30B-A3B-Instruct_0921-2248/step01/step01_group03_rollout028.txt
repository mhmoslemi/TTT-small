import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Adaptive optimization with structured starts
    max_iter = 200
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate a new starting point
        h_start = best_h.copy()
        # Add perturbation with structured changes
        idx = np.random.randint(0, n_points)
        h_start[idx] = np.clip(h_start[idx] + np.random.normal(0, 0.05), 0, 1)
        h_start = project(h_start)
        
        # Optimize with SLSQP
        try:
            result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 100, 'disp': False}
            )
        except:
            continue
            
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
