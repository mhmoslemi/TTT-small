import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 200
        # Start with binary mixture and shuffle
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach
    # Start with some shifts
    K = set()
    # Add a few random shifts to start
    for _ in range(min(50, 2*n_points - 1)):
        k = np.random.randint(0, 2*n_points - 1)
        K.add(k)
    
    # Initial projection to feasible set
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Try a hybrid approach: optimizers + column generation
    max_iter = 2000
    iter_count = 0
    improvement_threshold = 1e-5
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        if iter_count % 10 == 0:
            # Periodic relaxation step
            # Use SLSQP with bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(x):
                return np.sum(x) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(
                lambda x: evaluate_erdos_solution(x),
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': 50}
            )
            
            if result.success:
                h_new = result.x
            else:
                h_new = h.copy()
            
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new.copy()
        
        # Compute all C5 values for shifts not in K to find the most constraining one
        max_c5 = -1
        k_star = -1
        
        # We compute for k in [0, 2*n_points - 2] but avoid duplicates
        for k in range(0, 2*n_points - 1):
            if k in K:
                continue
            # Compute overlap for shift k
            # This is essentially the correlation at k position
            corr_val = 0
            dx = 2.0 / n_points
            for i in range(n_points):
                j = (i + k) % n_points
                corr_val += h[i] * (1 - h[j])
            c5_k = corr_val * dx
            if c5_k > max_c5:
                max_c5 = c5_k
                k_star = k
        
        # If we found an improvement, add to K
        if max_c5 > best_c5 + improvement_threshold:
            K.add(k_star)
            # Refine with full column generation logic
            h_refined = h.copy()
            c5_refined = best_c5
            # Re-solve using CVXPY on the extended K
            try:
                # Problem setup
                h_vars = cp.Variable(n_points)
                C5 = cp.Variable()
                
                # Constraints
                constraints = [0 <= h_vars[i] for i in range(n_points)]
                constraints += [h_vars[i] <= 1 for i in range(n_points)]
                constraints += [cp.sum(h_vars) == n_points / 2.0]
                
                # For all k in K, add constraint that C5 >= overlap(k)
                for k in K:
                    overlap_expr = 0
                    for i in range(n_points):
                        j = (i + k) % n_points
                        overlap_expr += h_vars[i] * (1 - h_vars[j])
                    overlap_expr *= (2.0 / n_points)
                    constraints.append(C5 >= overlap_expr)
                
                # Objective
                prob = cp.Minimize(C5)
                prob = cp.Problem(prob, constraints)
                
                # Solve
                prob.solve(solver='GLOP')
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_opt = h_vars.value
                    c5_opt = C5.value
                    
                    # Project back to [0,1]
                    h_opt = np.clip(h_opt, 0, 1)
                    h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)
                    
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        h = h_opt.copy()
            except:
                pass  # Fallback to previous solution
        else:
            # No improvement found - just continue with some randomness
            # Occasionally add a random shift to escape local modes
            if np.random.rand() < 0.05:
                k = np.random.randint(0, 2*n_points - 1)
                K.add(k)
    
    # Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 50}
    )
    
    if result.success:
        h_final = result.x
    else:
        h_final = best_h
    
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (list(h_final), final_c5, n_points)
