def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Use initial construction if available
    if 'initial_h_values' in globals():

    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 150
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Our approach:
    # 1. Binary search on C5
    # 2. For each candidate C5, we solve a convex relaxation
    # 3. If feasible, we try to improve the bound
    
    # Try to solve the problem directly using a more sophisticated method
    # Based on initial idea, but starting with more careful binary search

    # First we'll try the best solution with fast local optimization
    max_iter = 500
    iter_count = 0

    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Ensure feasible h
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
        
        # Optimize
        try:
            res = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
            )

            if res.success:
                h = res.x.copy()
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
                # Small perturbation to avoid local minima
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
            else:
                # If optimization fails, try another approach
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
                
        except Exception as e:
            # If optimization fails, try random search
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            
        # Projection to keep feasible
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
            
        # Check time
        iter_count += 1
        if time.time() - start_time > budget_s - 1:
            break
    
    # Try some additional methods
    
    # Try a greedy method
    h_greedy = h.copy()

        h_greedy = h_greedy * target_sum / h_sum

    # Apply hill climbing to improve solution
    c5_greedy = objective(h_greedy)
    if c5_greedy < best_c5:
        best_c5 = c5_greedy
        best_h = h_greedy.copy()

    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    # Return solution
    return (best_h.tolist(), final_c5, len(best_h))
