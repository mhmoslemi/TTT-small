import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create reasonable initial guess - two clusters
        n_points = 100
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        
    # Coarse-to-fine approach: first optimize with fewer points
    if n_points > 100:
        # Reduce resolution for coarse optimization
        coarse_n = min(50, n_points // 2)
        coarse_h = np.zeros(coarse_n)
        coarse_h[:coarse_n//2] = 1
        coarse_h = np.interp(np.linspace(0, coarse_n-1, n_points), 
                             np.linspace(0, coarse_n-1, coarse_n), coarse_h)
        h = coarse_h.copy()
        n_points = len(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.999
    min_T = 1e-6
    max_iter = int(budget_s * 1000)  # Iterations allowed with some safety margin
    current_T = T0
    
    # Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current solution
        h_current = h.copy()
        
        # Make a random perturbation
        i = np.random.randint(0, n_points)
        delta = (np.random.random() - 0.5) * current_T * 2
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Rescale to satisfy constraint sum(h) = n_points / 2
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject based on simulated annealing
        delta_cost = c5_new - best_c5
        if delta_cost < 0 or np.random.random() < np.exp(-delta_cost / current_T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Update temperature
        current_T = max(min_T, current_T * alpha)
    
    # Final refinement with SLSQP if there's time
    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Run final optimization with SLSQP
        try:
            result = minimize(
                lambda x: evaluate_erdos_solution(x),
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 50}
            )
            if result.success:
                refined_h = result.x
                refined_h = np.clip(refined_h, 0, 1)
                refined_h = refined_h * (n_points / 2.0) / np.sum(refined_h)
                final_c5 = evaluate_erdos_solution(refined_h)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
