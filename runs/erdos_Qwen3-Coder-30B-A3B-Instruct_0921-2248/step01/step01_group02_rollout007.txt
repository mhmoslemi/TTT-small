import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 47  # Start with the provided parent
    pop_size = 50
    max_generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 10
    
    # Get initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Upsample if needed
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            f = interp1d(x_orig, h_init, kind='linear', fill_value='extrapolate')
            h_init = f(x_new)
    else:
        # Create a basic construction
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
    
    # Normalize to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population
    for i in range(pop_size):
        if i == 0:
            h = h_init.copy()
        else:
            h = h_init.copy()
            # Add perturbation
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA loop
    for generation in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Selection: Tournament selection
        selected = []
        for _ in range(pop_size):
            idx1, idx2, idx3 = np.random.choice(pop_size, 3, replace=False)
            if fitness[idx1] < fitness[idx2] and fitness[idx1] < fitness[idx3]:
                selected.append(population[idx1])
            elif fitness[idx2] < fitness[idx1] and fitness[idx2] < fitness[idx3]:
                selected.append(population[idx2])
            else:
                selected.append(population[idx3])
        
        # Create new population
        new_pop = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_pop.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Single point crossover
                point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.1)
                
                # Ensure feasibility
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                new_pop.append(child)
                if len(new_pop) >= pop_size:
                    break
        
        # Evaluate new population
        population = new_pop[:pop_size]
        fitness = [evaluate_erdos_solution(h) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
