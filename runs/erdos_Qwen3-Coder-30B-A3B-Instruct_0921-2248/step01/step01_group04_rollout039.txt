import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 200
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Helper function to compute overlaps efficiently
    def compute_overlaps(h, n_points):
        overlaps = []
        for k in range(2 * n_points - 1):
            # Circular convolution
            shift = np.roll(1 - h, k)
            overlap = np.sum(h * shift) * (2.0 / n_points)
            overlaps.append(overlap)
        return np.array(overlaps)
    
    # Column generation approach
    K = set()  # Set of active shifts
    max_shifts = 2 * n_points - 1
    
    # Initial set of shifts
    initial_shifts = np.random.choice(max_shifts, size=min(50, max_shifts), replace=False)
    for k in initial_shifts:
        K.add(k)
    
    # Main column generation loop
    while time.time() - start_time < budget_s - 5:
        # Store current best
        current_best_c5 = evaluate_erdos_solution(best_h)
        
        # Create LP formulation
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Add overlap constraints for active shifts
        for k in K:
            # Compute overlap for shift k
            shift_h = np.roll(h_vars, k)
            overlap_expr = cp.sum(cp.multiply(h_vars, 1 - shift_h)) * (2.0 / n_points)
            constraints.append(C5 >= overlap_expr)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Solve LP
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                c5_sol = C5.value
                if c5_sol < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_sol
            else:
                # Make a random perturbation
                h_sol = h + np.random.normal(0, 0.01, n_points)
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2.0) / np.sum(h_sol)
                c5_sol = evaluate_erdos_solution(h_sol)
                if c5_sol < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_sol
        except:
            # Fallback to optimization
            h_sol = h + np.random.normal(0, 0.01, n_points)
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = h_sol * (n_points / 2.0) / np.sum(h_sol)

            if c5_sol < best_c5:

        if max_k not in K:

        if len(K) > 200:

        if len(K) > 50 and len(K) % 20 == 0:

            h_init = best_h.copy()

            def obj_func(x):
                return evaluate_erdos_solution(x)
            
            result = minimize(obj_func, h_init, method='SLSQP', bounds=bounds, 
                            options={'ftol': 1e-8, 'maxiter': 50})
            if result.success:
                h_polished = result.x
            else:
                h_polished = h_init
            
            # Project to feasible set
            h_polished = np.clip(h_polished, 0, 1)
            h_polished = h_polished * (n_points / 2.0) / np.sum(h_polished)
            c5_polished = evaluate_erdos_solution(h_polished)
            
            if c5_polished < best_c5:
                best_h = h_polished.copy()
                best_c5 = c5_polished
                
        h = best_h.copy()
    
    # Final refinement with local optimizers
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj_func(x):
        return evaluate_erdos_solution(x)
    
    result = minimize(obj_func, best_h.copy(), method='SLSQP', bounds=bounds, 
                     options={'ftol': 1e-8, 'maxiter': 200})
    if result.success:
        final_h = result.x
    else:
        final_h = best_h.copy()
        
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2.0) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    # Ensure final solution is better or equal to best
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
