import numpy as np
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value of the correlation (scaled by dx)
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

        ones_indices = np.where(h_current == 1)[0]

        i0 = np.random.choice(zeros_indices)
        
        # Create new solution by swapping
        h_new = h_current.copy()
        h_new[i1] = 0
        h_new[i0] = 1
        
        # Evaluate the new solution
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject
        delta = c5_new - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current

        iter_count += 1
