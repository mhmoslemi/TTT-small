import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization
    max_starts = 20
    iteration = 0
    
    # Coarse-to-fine grid search
    resolutions = [50, 100, 150]
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize at current resolution
        if res <= len(h):
            h_new = h[:res].copy()
        else:
            h_new = np.zeros(res)
            h_new[:len(h)] = h
            # Fill remaining with random values
            h_new[len(h):] = np.random.rand(res - len(h))
        
        # Normalize
        h_new = h_new * target_sum / np.sum(h_new)
        n_points = res
        
        for start in range(max_starts):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Try different initialization schemes
            if start % 3 == 0:
                # From parent solution
                init_h = h_new.copy()
            elif start % 3 == 1:
                # Random initialization
                init_h = np.random.rand(n_points)
                init_h = init_h * target_sum / np.sum(init_h)
            else:
                # Alternating pattern
                init_h = np.zeros(n_points)
                init_h[::2] = 1.0
                init_h = init_h * target_sum / np.sum(init_h)
            
            # Optimize
            try:
                res_opt = minimize(
                    objective,
                    init_h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res_opt.success:
                    h_opt = res_opt.x.copy()
                    c5 = objective(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                else:
                    # If optimization fails, use the initial solution
                    h_opt = init_h.copy()
                    c5 = objective(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        
            except Exception:
                # If optimization fails, try another approach
                h_opt = init_h.copy()
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
            
            # Perturb to escape local minima
            if time.time() - start_time > budget_s - 1:
                break
                
            noise = np.random.normal(0, 0.01, n_points)
            h_opt = h_opt + noise
            
            # Project to feasible region
            h_opt = np.clip(h_opt, 0, 1)
            h_opt_sum = np.sum(h_opt)
            if abs(h_opt_sum - target_sum) > 1e-10:
                h_opt = h_opt * target_sum / h_opt_sum
            
            c5_perturbed = objective(h_opt)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_opt.copy()
            
            iteration += 1
            
    # Final clean up and ensuring constraints
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
