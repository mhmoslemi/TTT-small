def run(seed=42, budget_s=600, **kwargs):

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Binary search for C5
    left, right = 0.0, 0.38085
    tolerance = 1e-5
    
    while right - left > tolerance and time.time() - start_time < budget_s - 1:
        mid = (left + right) / 2.0
        c5_candidate = mid
        
        # Formulate and solve SDP
        try:
            # Variables
            h_vars = cp.Variable(n_points)
            # Constraints
            constraints = [
                cp.sum(h_vars) == target_sum,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Check overlap constraints
            # Use a conservative bound for optimization
            max_overlap = 0
            for k in range(n_points):
                # Compute correlation term
                corr_term = 0
                for i in range(n_points):
                    j = i + k
                    if j < n_points:
                        corr_term += h_vars[i] * (1 - h_vars[j])
                # Add constraint that corr_term <= c5_candidate * dx
                constraints.append(corr_term <= c5_candidate * dx)
            
            # Objective: minimize maximum overlap (this is not strictly needed in SDP framework)
            # We'll validate feasibility for the given c5_candidate
            
            # Solve SDP
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000, eps=1e-6)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVER_ERROR:
                # Check if we found a valid solution
                h_sol = h_vars.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    # Renormalize
                    h_sol = h_sol * target_sum / np.sum(h_sol)
                    # The overlap constraint should already be satisfied in SDP
                    # Compute actual overlap for validation
                    corr = np.correlate(h_sol, 1 - h_sol, mode="full")
                    actual_c5 = np.max(corr) * dx
                    
                    if actual_c5 < best_c5:
                        best_c5 = actual_c5
                        best_h = h_sol.copy()
                    
                    # Update bounds for binary search
                    if actual_c5 <= c5_candidate:
                        right = c5_candidate
                    else:
                        left = c5_candidate
                else:
                    left = c5_candidate
            else:
                left = c5_candidate
                
        except Exception as e:
            # Handle numerical issues or solver failures
            left = c5_candidate
            
        # Random perturbations to avoid local minima
        if time.time() - start_time > budget_s - 2:
            break
        else:
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * target_sum / np.sum(h)
    
    # Perform final optimization on best solution
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    h = best_h.copy()
    # Re-optimize the best solution with full power of optimization
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Use scipy optimization on current solution
            res = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                h = res.x.copy()
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                # Perturb and try again
                noise = np.random.normal(0, 0.01, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h * target_sum / np.sum(h)
        except:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * target_sum / np.sum(h)
    
    # Final correction
    final_h = np.clip(best_h, 0, 1)
    final_h_sum = np.sum(final_h)
    if abs(final_h_sum - target_sum) > 1e-10:
        final_h = final_h * target_sum / final_h_sum
    final_c5 = objective(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
