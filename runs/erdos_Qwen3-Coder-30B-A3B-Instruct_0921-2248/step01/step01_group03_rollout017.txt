import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def gradient(h):
        # Compute gradient using finite differences for simplicity
        eps = 1e-6
        grad = np.zeros_like(h)
        base_val = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    # Use gradient-based optimization with line search
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Take gradient step
        grad = gradient(best_h)
        step_size = 0.01
        h_new = best_h - step_size * grad
        h_new = project(h_new)
        
        # Line search along the gradient direction
        for ls_iter in range(10):
            if time.time() - start_time > budget_s - 1:
                break
            h_test = project(best_h - (step_size / (2**ls_iter)) * grad)
            c5_test = evaluate_c5(h_test, dx)
            if c5_test < best_c5:
                best_h = h_test
                best_c5 = c5_test
                break
                
    # Final refinement with scipy minimize
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=con,
        tol=1e-6,
        options={'maxiter': 100, 'disp': False}
    )
    
    if result.success:
        h_opt = project(result.x)
        c5_opt = evaluate_c5(h_opt, dx)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    return (best_h, best_c5, n_points)
