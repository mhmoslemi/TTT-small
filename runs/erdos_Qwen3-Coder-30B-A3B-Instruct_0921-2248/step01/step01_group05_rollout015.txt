import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution approach
    resolutions = [47, 94, 188, 200]
    current_h = h.copy()
    current_n = n_points
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        if res > current_n:
            # Upsample
            upscale_factor = res // current_n
            new_h = np.repeat(current_h, upscale_factor)
            if len(new_h) > res:
                new_h = new_h[:res]
            elif len(new_h) < res:
                new_h = np.pad(new_h, (0, res - len(new_h)), 'constant', constant_values=0)
            current_h = new_h
            current_n = res
        elif res < current_n:
            # Downsample
            current_h = current_h[:res]
            current_n = res
        
        # Optimize with L-BFGS-B
        try:
            res_opt = minimize(
                obj_func,
                current_h.copy(),
                method='L-BFGS-B',
                bounds=bounds[:current_n],
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res_opt.success or res_opt.fun < best_c5:
                h_temp = np.clip(res_opt.x, 0, 1)
                # Enforce sum constraint
                h_temp = h_temp * target_sum / np.sum(h_temp)
                c5_temp = obj_func(h_temp)
                if c5_temp < best_c5:
                    best_h = h_temp.copy()
                    best_c5 = c5_temp
                    current_h = h_temp.copy()
                    current_n = len(h_temp)
            
        except Exception:
            pass
        
        # Local random search
        for _ in range(5):
            if time.time() - start_time > budget_s - 5:
                break
            h_pert = current_h + np.random.normal(0, 0.05, len(current_h))
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * target_sum / np.sum(h_pert)
            c5_temp = obj_func(h_pert)
            if c5_temp < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_temp
    
    # Final evaluation with evaluate_erdos_solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
