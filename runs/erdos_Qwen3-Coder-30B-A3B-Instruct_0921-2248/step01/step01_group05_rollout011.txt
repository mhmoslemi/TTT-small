import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h)

        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Upsample to higher resolution (500 points) to improve search ability
    if n_points < 500:
        h_upsampled = np.interp(np.linspace(0, 2, 500), np.linspace(0, 2, n_points), h)
        h = h_upsampled
        n_points = 500
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        # Make sure it's in the valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-level optimization: Nelder-Mead with restarts
    max_iter = 500
    restart_iters = 5
    perturb_strength = 0.05
    
    for iteration in range(restart_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random restart
        h_restart = np.random.rand(n_points)
        h_restart = h_restart * target_sum / np.sum(h_restart)
        
        # Optimize using Nelder-Mead
        res = minimize(obj_func, h_restart, method='Nelder-Mead', 
                       options={'maxiter': max_iter, 'xatol': 1e-6, 'fatol': 1e-6})
        
        if res.success or res.fun < best_c5:
            h_opt = np.clip(res.x, 0, 1)
            # Enforce sum constraint with projection
            current_sum = np.sum(h_opt)
            h_opt = h_opt * target_sum / current_sum
            c5_opt = obj_func(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                
        # Add perturbations to escape local minima
        h_perturbed = best_h + np.random.normal(0, perturb_strength, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        c5_perturbed = obj_func(h_perturbed)
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
            
        # Adaptive perturbation strength
        if iteration % 2 == 0:
            perturb_strength *= 0.95
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure that we have a valid final solution
    if not np.isfinite(final_c5) or np.isnan(final_c5):
        final_c5 = obj_func(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
