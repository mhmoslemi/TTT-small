import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Uniform crossover between two parents"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Segment-based crossover
    n = len(parent1)
    cut_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
    child2 = np.concatenate([parent2[:cut_point], parent1[cut_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.05):
    """Gaussian mutation"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return mutated

def genetic_algorithm(n_points, n_pop=50, generations=1000, budget_s=600):
    """Genetic algorithm to find optimal h"""
    start_time = time.time()
    
    # Initialize population
    population = []
    fitness = []
    
    # Use initial construction if available

        population.append(h_base.copy())
        fitness.append(evaluate_erdos_solution(h_base))
    else:
        # Create random binary-like starting population
        h_base = np.zeros(n_points)
        half = n_points // 2
        h_base[:half] = 1
        np.random.shuffle(h_base)
        population.append(h_base.copy())
        fitness.append(evaluate_erdos_solution(h_base))
    
    # Fill the rest of population
    for _ in range(n_pop - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
        fitness.append(evaluate_erdos_solution(h))
    
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = min(fitness)
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate current population
        new_fitness = [evaluate_erdos_solution(ind) for ind in population]
        
        # Track best solution
        min_fit_idx = np.argmin(new_fitness)
        if new_fitness[min_fit_idx] < best_c5:
            best_h = population[min_fit_idx].copy()
            best_c5 = new_fitness[min_fit_idx]
        
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best
        elite_idx = np.argmin(new_fitness)
        new_population.append(population[elite_idx].copy())
        
        # Generate new individuals
        while len(new_population) < n_pop:
            # Selection
            parent1 = tournament_selection(population, new_fitness)
            parent2 = tournament_selection(population, new_fitness)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Projection to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Keep only n_pop individuals
        population = new_population[:n_pop]
        fitness = [evaluate_erdos_solution(ind) for ind in population]
    
    return best_h, best_c5

        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimize with SLSQP
    result = minimize(

    )
    
    if result.success:
        best_h = result.x
    
    # Final project to feasible region
    best_h = project_feasible(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h)
