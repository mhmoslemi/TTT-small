import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a reasonable starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum  
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize swarm with 10 particles
    n_particles = 10
    particles = [h.copy() for _ in range(n_particles)]
    particle_scores = [float('inf')] * n_particles
    
    # Global best
    global_best_h = h.copy()
    global_best_c5 = float('inf')
    
    # Iteration counter for adaptive noise
    iteration = 0
    max_no_improvement = 100
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1:  # leave 1 seconds for final eval
        # Update particles
        for i in range(n_particles):
            # Optimize each particle
            try:
                h_current = particles[i].copy()
                h_current = np.clip(h_current, 0, 1)
                h_sum = np.sum(h_current)
                if abs(h_sum - target_sum) > 1e-10:
                    h_current = h_current * target_sum / h_sum
                
                # Optimize with SLSQP
                res = minimize(
                    objective,
                    h_current,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
                )

                if res.success:
                    h_new = res.x.copy()
                    c5 = objective(h_new)
                    particles[i] = h_new
                    particle_scores[i] = c5
                    if c5 < global_best_c5:
                        global_best_c5 = c5
                        global_best_h = h_new.copy()
                        no_improvement = 0
                    else:
                        no_improvement += 1
                        
                else:
                    # If optimization fails, perturb
                    noise = np.random.normal(0, 0.05, n_points) 
                    h_new = particles[i] + noise
                    h_new = np.clip(h_new, 0, 1)
                    h_sum = np.sum(h_new)
                    h_new = h_new * target_sum / h_sum
                    particles[i] = h_new
                    particle_scores[i] = objective(h_new)
            except:
                # Fallback
                noise = np.random.normal(0, 0.05, n_points) 
                h_new = particles[i] + noise
                h_new = np.clip(h_new, 0, 1)
                h_sum = np.sum(h_new)
                h_new = h_new * target_sum / h_sum
                particles[i] = h_new
                particle_scores[i] = objective(h_new)
                
        # Mutation and new solutions
        if iteration % 20 == 0 and no_improvement > 10:
            # Inject global best with noise
            noise_scale = min(0.1, 0.01 + (iteration // 100) * 0.005)
            noise = np.random.normal(0, noise_scale, n_points)
            new_h = global_best_h + noise
            new_h = np.clip(new_h, 0, 1)
            h_sum = np.sum(new_h)
            new_h = new_h * target_sum / h_sum
            particles.append(new_h)
            particle_scores.append(objective(new_h))
            
            # Remove worst particle
            worst_idx = np.argmax(particle_scores)
            particles.pop(worst_idx)
            particle_scores.pop(worst_idx)
            
            # Add crossover
            if len(particles) >= 2:
                idx1, idx2 = np.random.choice(len(particles), 2, replace=False)
                h1 = particles[idx1]
                h2 = particles[idx2]
                # Simple crossover
                crossover_point = np.random.randint(1, n_points - 1)
                new_h = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                new_h = np.clip(new_h, 0, 1)
                h_sum = np.sum(new_h)
                new_h = new_h * target_sum / h_sum
                particles.append(new_h)
                particle_scores.append(objective(new_h))
                
        # Keep best
        if global_best_c5 < best_c5:
            best_c5 = global_best_c5
            best_h = global_best_h.copy()
        
        iteration += 1
        if no_improvement >= max_no_improvement:
            break
            
    # Final refinement on best solution
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_sum = np.sum(h_final)
    if abs(h_sum - target_sum) > 1e-10:
        h_final = h_final * target_sum / h_sum
    
    final_c5 = objective(h_final)
    
    return (h_final.tolist(), final_c5, len(h_final))
