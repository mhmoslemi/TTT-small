import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum= n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def crossover(parent1, parent2):
    """Perform segment-based crossover"""
    n = len(parent1)
    if n < 2:
        return parent1.copy(), parent2.copy()
    
    # Random crossover point
    cx_point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.1):
    """Mutate h with Gaussian noise"""
    n = len(h)
    mask = np.random.random(n) < mutation_rate
    noise = np.random.normal(0, 0.1, n)
    h_new = h + noise * mask
    return project_feasible(h_new, len(h))

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual via tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Default to initial construction size
    n_pop = 50
    n_gen = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        population = [initial_h.copy()]
        # Add some variations
        for _ in range(n_pop - 1):
            h = initial_h + np.random.normal(0, 0.05, n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    else:
        # Create random initial population
        population = []
        for _ in range(n_pop):
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    
    # Track best
    best_h = None
    best_c5 = float('inf')
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h)
        fitness.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Evolution loop
    for gen in range(n_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep best
        elite_idx = np.argmin(fitness)
        new_population.append(population[elite_idx].copy())
        
        # Generate offspring
        while len(new_population) < n_pop:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        new_population = new_population[:n_pop]
        
        # Evaluate
        fitness = []
        for h in new_population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Update population
        population = new_population
    
    # Final refinement with local optimization
    if best_h is not None:
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        refined_result = minimize(
            lambda x: evaluate_erdos_solution(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if refined_result.success:
            final_h = refined_result.x
        else:
            final_h = best_h
    else:
        final_h = population[0]
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
