import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: key points at 0.5, 1.5 for uniformity
        n_points = 50
        h = np.zeros(n_points)
        for i in range(0, n_points, 2):
            h[i] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Using correlation for overlap calculation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Multi-scale optimization
    scales = [50, 100, 150]
    scale_idx = 0
    
    while time.time() - start_time < budget_s - 1:
        if scale_idx >= len(scales):
            break
            
        current_n = scales[scale_idx]
        if current_n > n_points:
            # Increase resolution by interpolation
            old_h = h.copy()
            old_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, current_n)
            h = np.interp(new_x, old_x, old_h)
            h = project_to_feasible(h)
            n_points = current_n
            dx = 2.0 / n_points
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Random restarts
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Optimize with L-BFGS-B
            try:
                res = minimize(
                    objective,
                    h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6, 'gtol': 1e-6}
                )
                if res.success:
                    h = res.x.copy()
                else:
                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)
            except Exception:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
            
            h = project_to_feasible(h)
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Random perturbation scaled with current overlap
            noise_scale = min(0.075, 0.025 + best_c5 * 0.05)
            noise = np.random.normal(0, noise_scale, n_points)
            h = h + noise
            h = project_to_feasible(h)
            
            if time.time() - start_time > budget_s - 1:
                break
        
        scale_idx += 1
    
    # Final check on best solution
    final_h = project_to_feasible(best_h)
    final_c5 = objective(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
