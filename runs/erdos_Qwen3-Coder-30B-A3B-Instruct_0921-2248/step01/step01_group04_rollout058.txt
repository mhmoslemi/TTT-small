import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)

        h[:half] = 1
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # GA parameters
    n_pop = 50
    n_gen = 1000
    mut_rate = 0.05
    cross_rate = 0.8
    tour_size = 5
    
    # Create initial population
    population = []
    for i in range(n_pop):
        if i == 0:
            individual = h.copy()
        else:
            # Create variation of initial
            individual = h.copy()
            # Add some noise
            noise = np.random.normal(0, 0.1, n_points)
            individual += noise
            individual = np.clip(individual, 0, 1)
            # Rescale
            individual = individual * (n_points / 2.0) / np.sum(individual)
        population.append(individual)
    
    # Track best overall
    best_overall = best_h.copy()
    best_overall_c5 = best_c5
    
    # Evolution loop
    for gen in range(n_gen):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = evaluate_erdos_solution(individual)
            fitness.append(c5)
        
        # Selection (tournament)
        new_pop = []
        for _ in range(n_pop):
            # Tournament
            tournament_indices = np.random.choice(n_pop, tour_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_pop.append(population[winner_idx].copy())
        
        # Crossover and Mutation
        next_pop = []
        for i in range(0, n_pop, 2):
            parent1 = new_pop[i]
            parent2 = new_pop[i+1] if i+1 < n_pop else new_pop[0]
            
            if np.random.random() < cross_rate:
                # Segment crossover
                cut_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
                child2 = np.concatenate([parent2[:cut_point], parent1[cut_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.random() < mut_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.random() < mut_rate:
                    child2[j] += np.random.normal(0, 0.05)
            
            # Project back to feasible region
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2.0) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2.0) / np.sum(child2)
            
            next_pop.extend([child1, child2])
        
        # Trim to exact population size
        population = next_pop[:n_pop]
        
        # Update best
        for individual in population:
            c5 = evaluate_erdos_solution(individual)
            if c5 < best_overall_c5:
                best_overall = individual.copy()
                best_overall_c5 = c5
    
    # Final optimization using SLSQP on best solution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Try to refine the best solution found
    h_final = best_overall.copy()
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        h_final,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    
    if result.success:
        h_final = result.x
    else:
        # If optimization fails, keep the best solution found
        pass
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(h_final)
    
    # Ensure final solution meets constraints
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    return (list(h_final), final_c5, n_points)
