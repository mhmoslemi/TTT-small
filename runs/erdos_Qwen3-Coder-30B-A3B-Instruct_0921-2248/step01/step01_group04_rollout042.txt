import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform segment-based crossover"""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate h by randomly perturbing elements"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h[i] = np.random.rand()
    return h

    if 'initial_h_values' in globals():

    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution loop
    max_generations = 1000

    stagnation_limit = 100

        if time.time() - start_time > budget_s - 5:
            break
        
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Track best solution in this generation
        gen_best_idx = np.argmin(fitnesses)
        gen_best_h = population[gen_best_idx]
        gen_best_c5 = fitnesses[gen_best_idx]
        
        if gen_best_c5 < best_c5:
            best_h = gen_best_h.copy()
            best_c5 = gen_best_c5

        if stagnation_count > stagnation_limit:

        while len(new_population) < n_pop:

            child1, child2 = crossover(parent1, parent2)
