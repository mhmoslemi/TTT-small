import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback: simple binary construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution sequence
    resolutions = [47, 94, 188, 200]
    current_h = h.copy()
    current_n = n_points
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Upsample
        upscale_factor = res // current_n
        if upscale_factor > 1:
            new_h = np.repeat(current_h, upscale_factor)
            if len(new_h) > res:
                new_h = new_h[:res]
            elif len(new_h) < res:
                new_h = np.pad(new_h, (0, res - len(new_h)), 'constant', constant_values=0)
            current_h = new_h
            current_n = res
            dx = 2.0 / current_n
        
        # Optimization within budget
        budget_per_res = min(100, budget_s - (time.time() - start_time))
        if budget_per_res < 10:
            break
            
        # Multiple random restarts
        best_restart_h = current_h.copy()
        best_restart_c5 = obj_func(current_h)
        
        restarts = min(5, max(1, int(budget_per_res / 20)))
        for _ in range(restarts):
            if time.time() - start_time > budget_s - 5:
                break
                
            try:
                # Random perturbation
                h_pert = current_h + np.random.normal(0, 0.05, current_n)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * target_sum / np.sum(h_pert)
                
                # Optimize
                res_opt = minimize(obj_func, h_pert, method='L-BFGSB', bounds=bounds, constraints=cons,
                                   options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8})
                
                if res_opt.success:
                    h_new = np.clip(res_opt.x, 0, 1)
                    h_new = h_new * target_sum / np.sum(h_new)
                    c5_new = obj_func(h_new)
                    if c5_new < best_restart_c5:
                        best_restart_h = h_new.copy()
                        best_restart_c5 = c5_new
                else:
                    h_new = np.clip(h_pert, 0, 1)
                    h_new = h_new * target_sum / np.sum(h_new)
                    c5_new = obj_func(h_new)
                    if c5_new < best_restart_c5:
                        best_restart_h = h_new.copy()
                        best_restart_c5 = c5_new
                        
            except Exception:
                continue
                
        current_h = best_restart_h
        new_c5 = best_restart_c5
        if new_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = new_c5
        
        if current_n >= 200:
            # Final intensification
            for _ in range(5):
                if time.time() - start_time > budget_s - 5:
                    break
                h_pert = best_h + np.random.normal(0, 0.02, len(best_h))
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * target_sum / np.sum(h_pert)
                c5_new = obj_func(h_pert)
                if c5_new < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_new
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-6:
        best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
