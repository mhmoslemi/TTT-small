import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default construction
        n_points = 100
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        np.random.shuffle(h_init)
    
    # Scale h_init to match target sum
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Try to improve using convex optimization
    dx = 2.0 / n_points
    max_shift = n_points - 1
    
    # CVXPY optimization
    try:
        h = cp.Variable(n_points, name='h')
        C5 = cp.Variable(name='C5')
        
        # Constraints
        constraints = [
            cp.sum(h) == target_sum,
            h >= 0,
            h <= 1
        ]
        
        # Add shift constraints
        for k in range(max_shift):
            # Compute shift k correlation contribution
            shift_sum = 0
            for i in range(n_points):
                if i + k < n_points:
                    shift_sum += h[i] * (1 - h[i + k])
            constraints.append(shift_sum <= C5 * target_sum)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Try MOSEK solver, fallback to ECOS if necessary
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=5000, feastol=1e-6, abstol=1e-6)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                # Fallback to SLSQP if CVXPY fails
                return fallback_optimization(h_init, n_points, target_sum, start_time, budget_s)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h.value
            c5_bound = float(C5.value)
            if np.all(h_result >= 0) and np.all(h_result <= 1):
                h_result = h_result / target_sum * (n_points / 2.0)
                return (h_result.tolist(), c5_bound, n_points)
            else:
                # Fallback to SLSQP
                return fallback_optimization(h_init, n_points, target_sum, start_time, budget_s)
        else:
            # Fallback to SLSQP
            return fallback_optimization(h_init, n_points, target_sum, start_time, budget_s)
            
    except Exception as e:
        # Fallback to SLSQP
        return fallback_optimization(h_init, n_points, target_sum, start_time, budget_s)

def fallback_optimization(h_init, n_points, target_sum, start_time, budget_s):
    # Use the previous good solver approach for fallback
    from scipy.optimize import minimize
    
    dx = 2.0 / n_points
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Local optimization steps
    while time.time() - start_time < budget_s - 1:
        try:
            res = minimize(obj_func, h.copy(), method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-6})
            
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_new)
                if current_sum > 0:
                    h_new = h_new * target_sum / current_sum
                    c5_new = obj_func(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            
            # Random restart
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            
        except Exception as e:
            break
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * target_sum / final_sum
        final_c5 = obj_func(final_h)
        return (final_h.tolist(), float(final_c5), n_points)
    else:
        # Return the initial solution
        return (h_init.tolist(), float(obj_func(h_init)), n_points)
