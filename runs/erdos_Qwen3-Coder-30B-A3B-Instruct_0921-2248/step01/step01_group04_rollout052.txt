import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define bounds for each point
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization loop with restarts
    max_iter = 1000
    restarts = 0
    max_restarts = 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Periodically restart with new random guess to escape local minima
        if iteration % 100 == 0 and iteration > 0:

                h = h * (n_points / 2.0) / np.sum(h)
                restarts += 1
        
        # Make a copy for this iteration
        h_current = h.copy()
        
        # Optimization step with SLSQP
        try:
            result = minimize(
                lambda x: evaluate_erdos_solution(x),
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': 30}
            )
            
            if result.success:
                h_new = result.x
            else:
                h_new = h_current
        except:
            # Fallback in case of optimization error
            h_new = h_current

        h_new = np.clip(h_new, 0, 1)

        c5_new = evaluate_erdos_solution(h_new)

        # Use new h for next iteration
        h = h_new.copy()
        
        # Occasionally add small random perturbation
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
