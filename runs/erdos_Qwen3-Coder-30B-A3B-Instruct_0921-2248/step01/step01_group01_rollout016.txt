import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with different perturbations
    num_starts = 20
    for start in range(num_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize from current or previous best
        if start == 0:
            h_init = h.copy()
        else:
            # Initialize with small random perturbation or from best solution
            h_init = best_h.copy()
            # Add noise based on current best
            noise_level = min(0.1, max(0.01, 1.0 / (1.0 + best_c5 * 10)))
            noise = np.random.normal(0, noise_level, n_points)
            h_init = h_init + noise
            
        # Ensure feasible h_init
        h_init = np.clip(h_init, 0, 1)
        h_sum = np.sum(h_init)
        if abs(h_sum - target_sum) > 1e-10:
            h_init = h_init * target_sum / h_sum
        
        try:
            res = minimize(
                objective,
                h_init.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                h_opt = res.x.copy()
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    
        except Exception as e:
            continue
            
    # Final projection to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    # Final evaluation
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
