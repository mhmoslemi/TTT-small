import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_feasible(h, n_points):
    """Project h to feasible set: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def mutate_individual(h, mutation_rate=0.1):
    """Mutate individual by perturbing some elements"""
    h = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * mutation_rate), replace=False)
    for i in indices:
        # Add small random change
        h[i] += np.random.normal(0, 0.05)
    return project_feasible(h, len(h))

def crossover_individuals(h1, h2):
    """Perform segment-based crossover"""
    n = len(h1)
    crossover_point = np.random.randint(1, n-1)
    child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitness, tournament_size=5):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), size=tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 100
        initial_h = np.zeros(n_points)
        # Create binary mixture
        half = n_points // 2
        initial_h[:half] = 1
        np.random.shuffle(initial_h)
    
    # Initialize population
    n_pop = 50
    population = []
    fitness = []
    
    # Seed with initial solution and variations
    population.append(initial_h.copy())
    fitness.append(evaluate_erdos_solution(initial_h))
    
    # Add random variations
    for _ in range(n_pop - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    max_gen = 1000
    
    # Main evolution loop
    for generation in range(max_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        new_fitness = []
        for h in population:
            f = evaluate_erdos_solution(h)
            new_fitness.append(f)
            
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best
        elite_idx = np.argmin(new_fitness)
        new_population.append(population[elite_idx].copy())
        
        while len(new_population) < n_pop:
            # Selection
            parent1 = tournament_selection(population, new_fitness)
            parent2 = tournament_selection(population, new_fitness)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover_individuals(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            child1 = mutate_individual(child1, mutation_rate)
            child2 = mutate_individual(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:n_pop]
        fitness = new_fitness[:n_pop]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            
        # Occasionally add diversity
        if generation % 100 == 0 and generation > 0:
            for i in range(0, n_pop, 2):
                if len(population) > i:
                    h = np.random.rand(n_points)
                    h = project_feasible(h, n_points)
                    population[i] = h
            
    # Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Try polishing the best solution
    h_current = best_h.copy()
    try:
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            h_current,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        if result.success:
            h_final = result.x
            c5_final = evaluate_erdos_solution(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass  # Skip polishing on error
        
    # Final check of feasibility
    best_h = project_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
