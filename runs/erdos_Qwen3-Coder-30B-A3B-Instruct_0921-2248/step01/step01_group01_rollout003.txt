import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default: construct a good initial solution
        n_points = 100
        h = np.zeros(n_points)
        # Create a piecewise constant function
        split = n_points // 2
        h[:split] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization
    iterations = 0
    max_iter = 50
    num_starts = 10
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        for _ in range(num_starts):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Try both random and structured initialization
            if np.random.rand() > 0.5:
                h_start = np.random.rand(n_points)
            else:
                # Continue with current best if available, otherwise use structured
                if iterations > 0:
                    h_start = best_h.copy()
                else:
                    h_start = np.random.rand(n_points)
            
            # Normalize
            h_start = h_start * target_sum / np.sum(h_start)
            
            try:
                res = minimize(
                    objective,
                    h_start,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    h_opt = res.x.copy()
                    c5 = objective(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        
            except Exception:
                pass
            
            # Add small random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h_start = h_start + noise
            
            iterations += 1
            if time.time() - start_time > budget_s - 1:
                break

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
