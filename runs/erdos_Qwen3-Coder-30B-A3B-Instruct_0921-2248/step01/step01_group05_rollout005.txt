import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 47
    
    # Use the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def get_upsampled_h(h_old, factor):
        h_new = np.repeat(h_old, factor)
        if len(h_new) > len(h_old) * factor:
            h_new = h_new[:len(h_old) * factor]
        return h_new
    
    times = [100, 200, 300, 400, 500]
    upscale_factors = [2, 2, 12]
    
    for i, t in enumerate(times):
        if time.time() - start_time > budget_s - 1:
            break
            
        if i < len(upscale_factors):
            h = get_upsampled_h(h, upscale_factors[i])
            n_points = len(h)
            dx = 2.0 / n_points
            bounds = [(0.0, 1.0)] * n_points
            cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Add small random noise
        noise = np.random.uniform(-0.05, 0.05, n_points)
        h_noise = h + noise
        h = project_to_feasible(h_noise)
        
        # Optimization using LBFGSB
        try:
            res = minimize(objective, h.copy(), method='L-BFGS-B', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
            
            if res.success:
                h_opt = project_to_feasible(res.x)
            else:
                h_opt = project_to_feasible(h)
        except:
            h_opt = project_to_feasible(h)
        
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            
        # Random restart
        if i % 2 == 0:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
    
    # Final validation
    if best_h is not None:
        final_c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        return (h.tolist(), float(objective(h)), n_points)
