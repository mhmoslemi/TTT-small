import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default: binary construction
        n_points = 100
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
    
    # Scale to target size
    if n_points < 200:
        h_init = np.tile(h_init, 200 // n_points + 1)[:200]
        n_points = 200
    elif n_points > 200:
        h_init = h_init[:200]
        n_points = 200
    
    # dx for interval [0, 2]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Solver settings
    solver = cp.MOSEK
    max_time = budget_s * 0.9
    
    # Define variables
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable()

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        C5 >= 0
    ]
    
    # Add overlap constraints for all shifts
    # For each shift k, we compute the overlap integral
    # and add constraint that it's <= C5 * (n/2)
    max_shift = n_points
    
    # We use a subset of shifts for performance
    shift_range = range(1, min(max_shift, 50))
    
    for k in shift_range:
        # Compute correlation for shift k (right shift)
        # We do correlation manually with appropriate slicing
        corr_sum = 0
        for i in range(n_points - k):
            corr_sum += h[i] * (1 - h[i + k])
        # Add constraint for each shift
        constraints.append(corr_sum <= C5 * target_sum)
    
    # Define the objective
    objective = cp.Minimize(C5)
    
    # Create the problem and solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False, time_limit=max_time)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_val = C5.value
        else:
            # Fallback to initial solution if solver fails
            h_opt = h_init
            c5_val = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    except Exception:
        # If anything fails, fallback
        h_opt = h_init
        c5_val = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    
    # Ensure bounds and normalize sum
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = np.sum(h_opt)
    h_opt = h_opt * target_sum / current_sum
    
    # Evaluate final C5
    final_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    return (h_opt.tolist(), float(final_c5), n_points)
