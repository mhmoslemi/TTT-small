import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        h0 = np.random.permutation(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Try to improve with convex optimization approach
    solver = 'MOSEK'  # or 'GUROBI' if available
    
    # Convert to CVXPY variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        C >= 0
    ]
    
    # Compute all possible correlations
    for k in range(2 * n_points - 1):
        # Compute correlation term for shift k
        correlation = 0
        if k < n_points:
            # Forward shift
            correlation = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            # Backward shift
            correlation = cp.sum(cp.multiply(h_var[n_points-(2*n_points-1-k):], 1 - h_var[:2*n_points-1-k]))
        constraints.append(C >= correlation * dx)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Solve once with better starting point
    h_start = best_h.copy()
    h_var.value = h_start
    
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_var.value
            c5_new = C.value
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass  # Continue with other approach if SDP fails
    
    # Refine using local optimization
    def objective_local(h):
        return evaluate_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Optimizer loop
    max_iter = 200
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try random search
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = project(h_pert)
        
        result = minimize(
            objective_local,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
        # Occasionally restart with better h
        if i % 10 == 0:
            h_pert = best_h + np.random.normal(0, 0.005, n_points)
            h_pert = project(h_pert)
            result = minimize(
                objective_local,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 50, 'disp': False}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h, final_c5, n_points)
