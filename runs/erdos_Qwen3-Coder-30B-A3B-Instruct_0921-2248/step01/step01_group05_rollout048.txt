import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Multi-resolution optimization
    resolutions = [47, 94, 188, 200]
    current_h = h.copy()
    current_c5 = obj_func(current_h)
    
    if current_c5 < 0.3807:
        return (current_h.tolist(), float(current_c5), n_points)
    
    for i, res in enumerate(resolutions):
        if time.time() - start_time > budget_s - 5:
            break
            
        if res > n_points:
            # Upsample
            upsample_factor = res // n_points
            h_upsampled = np.repeat(current_h, upsample_factor)
            if len(h_upsampled) > res:
                h_upsampled = h_upsampled[:res]
            elif len(h_upsampled) < res:
                h_upsampled = np.pad(h_upsampled, (0, res - len(h_upsampled)), 'constant')
            
            current_h = h_upsampled.copy()
            n_points = res
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            
            # Re-define bounds and constraints
            bounds = [(0.0, 1.0)] * n_points
            cons = {'type': 'eq', 'fun': constraint_sum}

        # Optimize
        max_iter = 100
        opt_time = min(30, budget_s - (time.time() - start_time) - 5)
        
        try:
            res = minimize(obj_func, current_h.copy(), method='L-BFGSB', 
                           bounds=bounds, constraints=cons, options={'maxiter': max_iter})
            
            if res.success or res.fun < current_c5:
                h_new = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                
                if c5_new < current_c5:
                    current_h = h_new.copy()
                    current_c5 = c5_new
                    
        except Exception:
            pass

    # Final local search
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Random restarts with slight perturbations
        h_pert = best_h + np.random.normal(0, 0.05, len(best_h))
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert = h_pert * target_sum / current_sum
        c5_new = obj_func(h_pert)
        
        if c5_new < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_new

    # Validate final solution
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
