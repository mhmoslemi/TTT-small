import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # ADMM-inspired approach with random restarts
    max_iter = 5000
    rho = 1.0
    max_rho = 100.0
    
    # ADMM step function
    def admm_step(h, rho, lambda_val):
        # Simple gradient-like update with projection
        # Compute gradient approximation
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                j = (i - k) % n_points
                grad[i] += (1 - h[j]) * (1 if k < n_points else 0)
        
        h_new = h - 0.01 * grad + lambda_val / rho
        h_new = project(h_new)
        return h_new
    
    lambda_val = 0.0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = project(h_pert)
        
        # Optimize with SLSQP
        result = minimize(
            objective,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
        # ADMM step
        h_admm = admm_step(best_h, rho, lambda_val)

            best_c5 = c5_admm
        
        # Update Lagrange multiplier
        lambda_val += rho * (np.sum(best_h) - n_points / 2)
        
        # Increase rho if needed
        if np.abs(np.sum(best_h) - n_points / 2) > 1e-3:
            rho = min(rho * 1.1, max_rho)
    
    return (best_h, best_c5, n_points)
