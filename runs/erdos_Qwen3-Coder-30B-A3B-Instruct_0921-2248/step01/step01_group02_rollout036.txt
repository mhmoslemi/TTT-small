import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

    start_time = time.time()

    # Upsample if needed
    if len(h_init) < n_points:
        x_orig = np.linspace(0, 2, len(h_init))
        x_new = np.linspace(0, 2, n_points)
        f = interp1d(x_orig, h_init, kind='linear', fill_value='extrapolate')
        h_init = f(x_new)
    
    # Parameters
    pop_size = 50
    max_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_init.copy()
        # Perturb
        mask = np.random.rand(n_points) < 0.3
        h[mask] += np.random.normal(0, 0.1, np.sum(mask))
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    last_improvement = 0
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            last_improvement = generation
        
        # Create next generation
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection (tournament)
            parent1 = tournament_selection(population, fitness, 3)
            parent2 = tournament_selection(population, fitness, 3)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:pop_size]
        
        # Evaluate new population
        fitness = [evaluate_erdos_solution(h) for h in new_population]
        
        population = new_population
        generation += 1

            break
    
    # Final evaluation with better precision
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)

def tournament_selection(population, fitness, k):
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    n = len(parent1)
    # Single point crossover
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Blend crossover alternative (better for continuous space)
    alpha = 0.5
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    
    # Repair
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    child1 = child1 * (len(child1) / 2) / np.sum(child1)
    child2 = child2 * (len(child2) / 2) / np.sum(child2)
    
    return child1, child2

def mutate(h, rate):
    n = len(h)
    mask = np.random.rand(n) < rate
    h[mask] += np.random.normal(0, 0.1, np.sum(mask))
    h = np.clip(h, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h
