import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup initial construction
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach
    # Initialize the set of shifts K
    K = set()
    num_initial_shifts = min(50, 2 * n_points - 1)  # Start with some random shifts
    
    # Add the first few shifts
    for i in range(num_initial_shifts):
        K.add(i)
    
    # Column generation iterations
    max_iter = 1000  # reasonable limit
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Solve LP to find current h and best C5
        # Variables
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2)
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Add overlap constraints for all k in K
        for k in K:
            # Convolution term: sum_{i} h[i] * (1 - h[(i+k) % n_points])
            conv_term = 0
            for i in range(n_points):
                conv_term += h_vars[i] * (1 - h_vars[(i + k) % n_points])
            if k < 2 * n_points - 1:
                constraints.append(C5_var >= conv_term * 2.0 / n_points)
        
        # Objective
        objective = cp.Minimize(C5_var)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            h_solution = h_vars.value
            c5_solution = C5_var.value
        except:
            # If solver fails, fallback to simpler approach
            h_solution = h.copy()
            c5_solution = best_c5 + 1  # Forced descent
        
        # Check if we've found a better solution
        if c5_solution < best_c5:
            best_h = h_solution.copy()
            best_c5 = c5_solution
            
        # Compute for all shifts not in K
        max_increase = 0
        new_k = -1
        
        for k in range(2 * n_points - 1):
            if k not in K:
                # Compute overlap term
                conv_term = 0
                for i in range(n_points):
                    conv_term += h_solution[i] * (1 - h_solution[(i + k) % n_points])
                overlap_val = conv_term * 2.0 / n_points
                if overlap_val > max_increase:
                    max_increase = overlap_val
                    new_k = k
                    
        # If we found a new constraint to add
        if new_k != -1:
            K.add(new_k)

            if len(K) > 350:  # Arbitrary but reasonable limit
                break
        else:

            break
    
    # Final polishing with gradient-based optimization
    if best_c5 < 0.5:  # Only if we have a promising solution

            return evaluate_erdos_solution(h_vec)
            
        # Define bounds and constraint again for local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Local optimization on best solution
        # Use the best solution as starting point
        result = minimize(
            obj_func,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = best_h
            
        # Enforce feasibility after optimization
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2.0) / np.sum(h_final)
        
        final_c5 = evaluate_erdos_solution(h_final)
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
