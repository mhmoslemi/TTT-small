def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def tournament_selection(population, fitnesses, k):
    """Select an individual using tournament selection"""
    selected_indices = np.random.choice(len(population), k)
    selected_fitnesses = [fitnesses[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitnesses)]
    return population[winner_idx]

def crossover(parent1, parent2):
    """Single-point crossover with blend interpolation"""
    n = len(parent1)
    point = np.random.randint(1, n-1)
    
    child1 = np.zeros(n)
    child2 = np.zeros(n)
    
    child1[:point] = parent1[:point]
    child1[point:] = parent2[point:]
    
    child2[:point] = parent2[:point]
    child2[point:] = parent1[point:]
    
    # Apply blend crossover for better structure preservation
    alpha = 0.5
    for i in range(n):
        if np.random.random() < 0.5:
            # Blend with parent1
            vmin = min(parent1[i], parent2[i])
            vmax = max(parent1[i], parent2[i])
            range_val = vmax - vmin
            child1[i] = vmin - alpha * range_val + np.random.random() * (2 * alpha * range_val)
            child2[i] = vmin - alpha * range_val + np.random.random() * (2 * alpha * range_val)
    
    return child1, child2

def mutate(h):
    """Apply mutation to an individual"""
    # Random reset
    for i in range(len(h)):
        if np.random.random() < 0.1:
            h[i] = np.random.random()
        # Add gaussian noise
        h[i] += np.random.normal(0, 0.03)
        h[i] = max(0, min(1, h[i]))
    
    return h

def project_to_feasible(h, n_points):
    """Project to satisfy sum(h) == n_points / 2 and 0 <= h <= 1"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    
    # Normalize
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    
    return h
