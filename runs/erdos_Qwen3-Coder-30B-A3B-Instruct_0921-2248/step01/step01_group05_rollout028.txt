import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default: use a smoothed version of a binary construction
        n_points = 100
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        # Add some randomness to avoid local minima
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Try to improve using a convex optimization approach
    # We'll optimize a relaxed version using CVXPY
    
    # Create variables
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")  # To minimize
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # Approximate the max over shifts as constraints
    # For each shift k from 0 to n_points-1, we want:
    # sum over i of h[i] * (1 - h[i+k]) <= C5 * (n_points/2)
    
    # We add constraints for a subset of shifts to reduce complexity
    max_shifts = min(20, n_points)  # limit shift range for speed
    for k in range(max_shifts):
        # Compute the correlation for this shift
        # Only sum over valid indices where both h[i] and h[i+k] exist
        if k == 0:
            corr = cp.sum(cp.multiply(h, 1 - h))  # self-correlation
        else:
            corr = cp.sum(cp.multiply(h[:-k], 1 - h[k:]))  # cross-correlation
        constraints.append(corr <= C5 * target_sum)
    
    # Objective: minimize C5
    objective = cp.Minimize(C5)
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers_to_try = ['MOSEK', 'SCIP', 'GLPK', 'ECOS']
    best_result = None
    best_c5 = float('inf')
    best_h = h_init
    
    # Try a few different solvers in case one is better for this problem
    for solver in solvers_to_try:
        if time.time() - start_time > budget_s - 5:
            break
        try:
            prob.solve(solver=solver, verbose=False, max_iters=1000)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                sol_h = h.value
                if sol_h is not None:
                    sol_h = np.clip(sol_h, 0, 1)
                    # Enforce sum constraint properly
                    sol_h = sol_h * target_sum / np.sum(sol_h)
                    # Compute actual C5
                    c5_val = np.max(np.correlate(sol_h, 1 - sol_h, mode="full")) * dx
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = sol_h.copy()
                        best_result = sol_h.copy()
        except Exception:
            continue
    
    # If we didn't solve with CVXPY, fallback to previous method
    if best_result is None:
        best_h = h_init
        best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    
    # Apply some final local optimization if time allows
    if time.time() - start_time < budget_s - 1:
        # Run a quick gradient descent on the result
        try:
            # Simple gradient-based update with projection
            h_current = best_h.copy()
            step_size = 0.01
            for _ in range(100):  # Limited iterations
                if time.time() - start_time > budget_s - 1:
                    break
                # Compute gradient approximation using finite differences
                grad = np.zeros_like(h_current)
                for i in range(n_points):
                    h_plus = h_current.copy()
                    h_plus[i] += 1e-5
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * target_sum / np.sum(h_plus)
                    val_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode="full")) * dx
                    
                    h_minus = h_current.copy()
                    h_minus[i] -= 1e-5
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * target_sum / np.sum(h_minus)
                    val_minus = np.max(np.correlate(h_minus, 1 - h_minus, mode="full")) * dx
                    
                    grad[i] = (val_plus - val_minus) / (2e-5)
                
                # Update step
                h_current = h_current - step_size * grad
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * target_sum / np.sum(h_current)
                
                # Compute new value
                new_c5 = np.max(np.correlate(h_current, 1 - h_current, mode="full")) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_current.copy()
        except Exception:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
