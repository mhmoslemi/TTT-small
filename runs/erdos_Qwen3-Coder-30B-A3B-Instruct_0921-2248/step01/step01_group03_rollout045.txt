def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

        mid = n_points // 2
        h0[:mid] = 1.0

        # Randomize a bit
        indices = np.random.permutation(n_points)[:n_points//4]
        h0[indices] = np.random.uniform(0, 1, len(indices))
        # Normalize to sum to n_points/2
        h0 = h0 * (n_points / 2) / np.sum(h0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)

    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    def objective(h):
        return evaluate_c5(h, dx)
    
    # Use L-BFGS-B with feasible projections
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        h_pert = best_h + np.random.normal(0, 0.005, n_points)
        h_pert = project(h_pert)
        
        # Optimize with L-BFGS-B
        result = minimize(
            objective,
            h_pert,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=con,
