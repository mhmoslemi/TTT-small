import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the overlap integral using np.correlate
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a structured construction
        n_points = 150
        h0 = np.zeros(n_points)
        # Create a piecewise constant function with 3 segments
        split1 = n_points // 3
        split2 = 2 * n_points // 3
        h0[:split1] = 1.0
        h0[split1:split2] = 0.5
        h0[split2:] = 0.0
        
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Optimization loop with random search and local refinement
    max_iter = 500
    patience = 0
    max_patience = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = project(h_pert)
        
        # Local optimizer
        result = minimize(
            objective,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Occasionally try larger perturbations if stuck
        if patience > 20 and i % 50 == 0:
            h_pert = best_h + np.random.uniform(-0.1, 0.1, n_points)
            h_pert = project(h_pert)
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 50, 'disp': False}
            )
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    patience = 0
        
        # Occasionally try random restarts
        if i % 100 == 0 and patience > 50:
            h_pert = np.random.rand(n_points)
            h_pert = project(h_pert)
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 50, 'disp': False}
            )
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    patience = 0
            
    return (best_h, best_c5, n_points)
