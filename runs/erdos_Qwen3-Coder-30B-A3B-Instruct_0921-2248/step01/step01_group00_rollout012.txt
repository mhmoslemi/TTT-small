import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value of the correlation (scaled by dx)
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

    # Global best
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()

    # PSO main loop
    iter_count = 0
    for _ in range(max_iter):

            break

            r1, r2 = np.random.random(2)

            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            particles[i] += velocities[i]

            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])

            fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if fitness < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = fitness

            if fitness < g_best_val:
