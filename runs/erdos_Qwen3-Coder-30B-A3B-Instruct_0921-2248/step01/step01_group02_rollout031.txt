import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

    # Global best
    gbest_idx = np.argmin(pbest_scores)

            h_new = h + v

            h_new = np.clip(h_new, 0, 1)

    return (gbest.tolist(), final_c5, n_points)
