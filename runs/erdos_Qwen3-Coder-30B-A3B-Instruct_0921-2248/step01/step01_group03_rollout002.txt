import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Refine with gradient ascent
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = project_to_feasible(h_pert, n_points)
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        con = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            objective,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_opt = project_to_feasible(result.x, n_points)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                
        # Try to sample a better starting point for next iteration
        h_sample = np.random.uniform(0, 1, n_points)
        h_sample = project_to_feasible(h_sample, n_points)
        result = minimize(
            objective,
            h_sample,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 50, 'disp': False}
        )
        
        if result.success:
            h_opt = project_to_feasible(result.x, n_points)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
