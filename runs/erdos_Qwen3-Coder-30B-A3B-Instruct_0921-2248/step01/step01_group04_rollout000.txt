import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 50
        h = np.zeros(n_points)
        # Place two clusters of 1s
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Upsample to target resolution
    if n_points < 200:
        upscale_factor = 200 // n_points
        h_upsampled = np.repeat(h, upscale_factor)
        # Add remaining points by interpolation
        remaining = 200 - len(h_upsampled)
        if remaining > 0:
            h_upsampled = np.concatenate([h_upsampled, 
                                         np.linspace(h_upsampled[-1], 0, remaining)])
        h = h_upsampled
        n_points = 200
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perform simulated annealing
    T = 0.1  # Initial temperature
    alpha = 0.999  # Cooling rate
    min_T = 1e-6  # Minimum temperature
    max_iter_per_T = 100  # Iterations per temperature level
    progress_counter = 0
    stable_count = 0
    last_best = best_c5
    
    while T > min_T and time.time() - start_time < budget_s - 5:
        for _ in range(max_iter_per_T):
            if time.time() - start_time > budget_s - 5:
                break
            # Make a copy of current solution
            h_current = h.copy()
            
            # Random perturbation
            i = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_current[i] += delta
            
            # Projection to feasible region
            h_current = np.clip(h_current, 0, 1)
            rescale_factor = (n_points / 2.0) / np.sum(h_current)
            h_current *= rescale_factor
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_current)
            
            # Accept or reject based on SA rule
            delta_E = c5_new - best_c5
            if delta_E < 0 or np.exp(-delta_E / T) > np.random.rand():
                h = h_current
                if c5_new < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_new
                    
        # Cool down
        T *= alpha
        
        # Check for stagnation
        if abs(best_c5 - last_best) < 1e-6:
            stable_count += 1
            if stable_count > 5:
                # Lower temperature to restart exploration
                T *= 0.9
                stable_count = 0
        else:
            last_best = best_c5
            stable_count = 0
    
    # Final refinement with SLSQP
    if time.time() - start_time < budget_s - 10:
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Use better solution as starting point for refinement
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            rescale_factor = (n_points / 2.0) / np.sum(refined_h)
            refined_h *= rescale_factor
            refined_c5 = evaluate_erdos_solution(refined_h)
            
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
