import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 2000
    dual_step_size = 0.1
    
    # Initialize Lagrange multiplier
    lambda_ = 0.0
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Augmented Lagrangian
    def augmented_lagrangian(h, lambda_, rho):
        # Compute the correlation
        correlation = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(correlation)
        
        # Lagrangian term
        penalty = lambda_ * constraint_sum(h) + (rho / 2) * (constraint_sum(h))**2
        
        return max_corr * dx + penalty
    
    # Gradient of the objective w.r.t. h
    def gradient_h(h):
        # This is approximate: we numerically differentiate
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (evaluate_c5(h_plus, dx) - evaluate_c5(h_minus, dx)) / (2 * eps)
        return grad
    
    # ADMM iteration
    h_current = h0.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Update h by minimizing augmented Lagrangian
        # Using gradient descent
        grad = gradient_h(h_current)
        h_new = h_current - 0.01 * grad
        
        # Project to feasible set
        h_new = project(h_new)
        
        # Update dual variable lambda
        lambda_ = lambda_ + dual_step_size * constraint_sum(h_new)
        
        # Check if better solution found
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update current iterate
        h_current = h_new.copy()

        if iteration % 100 == 0 and iteration > 0:

            h_current = best_h.copy()
            
    return (best_h, best_c5, n_points)
