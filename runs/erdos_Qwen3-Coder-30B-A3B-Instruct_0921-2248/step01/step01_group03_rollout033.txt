```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Create structured initial guess
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Gradient-based optimization with projected updates
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = project(h_pert)
        
        # Local optimization using L-BFGS
        result = minimize(
            objective,
            h_pert,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-6,
            options={'maxiter': 100}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
        # Occasionally refine with a better local search
        if i % 20 == 0 and i > 0:
            h_pert = best_h + np.random.normal(0, 0.005, n_points)
            h_pert = project(h_pert)
            
            result = minimize(
                objective,
                h_pert,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                tol=1e-6,
                options={'maxiter': 50}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
```
