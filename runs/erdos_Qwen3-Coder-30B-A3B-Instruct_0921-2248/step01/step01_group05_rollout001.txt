import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization
    upscale_factors = [2, 2, 12]
    resolutions = [47, 94, 188, 200]
    
    for res_idx, n_points_new in enumerate(resolutions[:-1]):  # Last resolution is treated separately
        if time.time() - start_time > budget_s - 30:
            break
            
        # Upsample if needed
        if n_points_new > n_points:
            upsample_factor = n_points_new // n_points
            h_new = np.repeat(h, upsample_factor)
            if len(h_new) > n_points_new:
                h_new = h_new[:n_points_new]
            elif len(h_new) < n_points_new:
                h_new = np.pad(h_new, (0, n_points_new - len(h_new)), constant_values=0)
            h = h_new.copy()
            n_points = n_points_new
            dx = 2.0 / n_points
            
        # Optimize at current resolution
        try:
            # Use L-BFGS-B method for faster convergence
            res = minimize(obj_func, h.copy(), method='L-BFGSB', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint with projection
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            
            # Random restarts
            for _ in range(5):
                if time.time() - start_time > budget_s - 10:
                    break
                h_rnd = np.random.rand(n_points)
                h_rnd = h_rnd * target_sum / np.sum(h_rnd)
                c5_rnd = obj_func(h_rnd)
                if c5_rnd < best_c5:
                    best_h = h_rnd.copy()
                    best_c5 = c5_rnd
                    
        except Exception:
            pass
            
        # Perturb the solution
        perturbation = 0.05 * np.random.rand(n_points)
        h_pert = h + perturbation
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        c5_pert = obj_func(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
    
    # Final higher resolution with 200 points if not already done
    if n_points < 200:
        upsample_factor = 200 // n_points
        h_final = np.repeat(h, upsample_factor)
        if len(h_final) > 200:
            h_final = h_final[:200]
        elif len(h_final) < 200:
            h_final = np.pad(h_final, (0, 200 - len(h_final)), constant_values=0)
        h = h_final.copy()
        n_points = 200
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        bounds = [(0.0, 1.0)] * n_points
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Final optimization
        try:
            res = minimize(obj_func, h.copy(), method='L-BFGSB', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * target_sum / np.sum(h_final)
                c5_final = obj_func(h_final)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-6:
        best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    if np.isnan(final_c5) or np.isinf(final_c5):
        # Fall back to previous best if final evaluation failed
        try:
            final_c5 = evaluate_erdos_solution(best_h)
        except:
            pass

    return (best_h.tolist(), float(final_c5), n_points)
