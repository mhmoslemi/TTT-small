import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Simple construction: 1s followed by 0s
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = np.max(np.correlate(h0, 1 - h0, mode="full")) * dx
    
    def evaluate_c5(h):
        correlation = np.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * dx
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Optimization loop with multiple strategies
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Strategy 1: Random perturbation with SLSQP
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = project(h_pert)
        
        result = minimize(
            evaluate_c5,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
        # Strategy 2: Local gradient-based search with line search
        if time.time() - start_time > budget_s - 1:
            break
        
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = project(h_pert)
        
        # Use L-BFGS for refinement
        result = minimize(
            evaluate_c5,
            h_pert,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-6,
            options={'maxiter': 50, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
