import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization with upscaling
    resolutions = [47, 94, 188, 200]
    upscale_factors = [2, 2, 12]
    
    current_h = h.copy()
    current_n = n_points
    
    for i, new_n in enumerate(resolutions):
        if time.time() - start_time > budget_s - 10:
            break
            
        if new_n > current_n:
            upscale_factor = upscale_factors[i] if i < len(upscale_factors) else 2
            new_h = np.repeat(current_h, upscale_factor)
            new_h = new_h[:new_n]
            current_h = new_h
            current_n = new_n
        elif new_n < current_n:
            current_h = current_h[:new_n]
            current_n = new_n
            
        dx = 2.0 / current_n
        target_sum = current_n / 2.0
        
        # Optimization with L-BFGS-B
        try:
            res = minimize(obj_func, current_h.copy(), method='L-BFGS-B', 
                          bounds=[(0.0, 1.0)] * current_n, 
                          options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success:
                current_h = np.clip(res.x, 0, 1)
                # Project to satisfy sum constraint
                current_sum = np.sum(current_h)
                current_h = current_h * target_sum / current_sum
                c5_new = obj_func(current_h)
                if c5_new < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_new
        except:
            pass
        
        # Randomized local search
        for _ in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            perturbed_h = current_h.copy()
            indices = np.random.choice(current_n, size=int(current_n * 0.1), replace=False)
            perturbed_h[indices] = np.random.rand(len(indices))
            # Project to satisfy sum constraint
            current_sum = np.sum(perturbed_h)
            perturbed_h = perturbed_h * target_sum / current_sum
            c5_new = obj_func(perturbed_h)
            if c5_new < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = c5_new
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), len(final_h))
