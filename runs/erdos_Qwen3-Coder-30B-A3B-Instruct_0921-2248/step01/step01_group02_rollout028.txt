import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a more sophisticated initial guess by alternating low/high values
        n_points = 100
        h_init = np.zeros(n_points)
        # Create piecewise constant function with alternating 0s and 1s, 
        # but adjusted to meet the integral constraint
        h_init[:n_points//2] = 1
        np.random.shuffle(h_init)
    
    # Create bounds for CVXPY optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Objective function that computes overlap integral
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    # Helper functions for convex relaxation approach
    def project_to_feasible(h):
        h_projected = np.clip(h, 0, 1)
        h_sum = np.sum(h_projected)
        if h_sum > 0:
            h_projected = h_projected * (n_points / 2) / h_sum
        return h_projected
    
    # Strategy: Local optimization with random restarts
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization function
    def optimize_once(h_start):
        # Use a different optimization method that's more robust
        bounds = [(0, 1) for _ in range(n_points)]
        
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6}
        
        try:
            res = minimize(objective, h_start, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options=options)
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_sum = np.sum(h_opt)
                h_opt = h_opt * (n_points / 2) / h_sum
                return h_opt, objective(h_opt)
            else:
                h_projected = project_to_feasible(h_start)
                return h_projected, objective(h_projected)
        except:
            h_projected = project_to_feasible(h_start)
            return h_projected, objective(h_projected)
    
    # Iterative refinement with random restarts
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Random restart with probability 0.3
        if np.random.random() < 0.3:
            h_rand = np.copy(h_init)
            np.random.shuffle(h_rand)
            h0 = h_rand
        else:
            h0 = best_h.copy()
        
        # Optimize
        h_new, c5_new = optimize_once(h0)
        
        # Update best solution if improved
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            last_improvement = time.time()
        
        # Early stopping: if no significant improvement in last 10 seconds, break
        if time.time() - last_improvement > 10 and time.time() - start_time > budget_s * 0.8:
            break
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
