import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 200
        # Create initial h with alternating pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach with CVXPY
    # Initial set of shifts
    K = list(range(0, min(50, 2*n_points - 1)))  # Start with first 50 shifts
    
    while time.time() - start_time < budget_s - 5:
        # Create LP model
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints for current K
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2)
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Add overlap constraints
        for k in K:
            # Compute overlap for shift k
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(C5 >= overlap_sum * (2.0 / n_points))
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h_new = np.array(h_new)
                    # Project back to feasible space
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        # Add some randomness for exploration
                        if len(K) < 2 * n_points - 1:
                            # Add a new shift from unseen range
                            unseen = set(range(2*n_points - 1)) - set(K)
                            if unseen:
                                k_add = np.random.choice(list(unseen))
                                K.append(k_add)
            else:
                # Fallback to gradient method if LP fails
                break
        except:
            break
            
        # Add more shifts gradually
        if len(K) < 2 * n_points - 1 and len(K) < 150:
            # Find most violated constraint
            h_test = best_h if np.random.random() > 0.5 else h_new
            max_viol = -1
            k_viol = -1
            for k_idx in range(2*n_points - 1):
                if k_idx not in K:
                    corr_sum = 0
                    for i in range(n_points):
                        j = (i + k_idx) % n_points
                        corr_sum += h_test[i] * (1 - h_test[j])
                    if corr_sum * (2.0 / n_points) > max_viol:
                        max_viol = corr_sum * (2.0 / n_points)
                        k_viol = k_idx
            if k_viol != -1:
                K.append(k_viol)
                
    # Final polishing with SLSQP
    if time.time() - start_time < budget_s - 10:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Use best solution as starting point
        h_current = best_h.copy()
        
        # Optimization step with SLSQP
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            h_current,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = h_current
        
        # Final projection
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2.0) / np.sum(h_final)
        
        final_c5 = evaluate_erdos_solution(h_final)
        
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
