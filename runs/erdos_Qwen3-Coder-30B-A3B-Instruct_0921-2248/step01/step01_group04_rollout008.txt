```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 200
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach
    current_h = h.copy()
    
    # Initial set of shifts
    max_shifts = 2 * n_points - 1
    available_shifts = list(range(max_shifts))
    active_shifts = set()
    
    # Add some initial shifts
    initial_count = min(50, len(available_shifts))
    initial_shifts = np.random.choice(available_shifts, initial_count, replace=False)
    for s in initial_shifts:
        active_shifts.add(s)
    
    # Iteratively improve
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 20:
        if time.time() - start_time > budget_s - 5:
            break
            
        iteration += 1
        
        # Solve LP problem with current active shifts
        try:
            # Define variables
            h_vars = cp.Variable(n_points)
            c5_var = cp.Variable()
            
            # Constraints 
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1,
            ]
            
            # Add overlap constraints for active shifts
            for k in active_shifts:
                # Compute overlap for shift k
                overlap = 0
                dx = 2.0 / n_points
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap += h_vars[i] * (1 - h_vars[j]) * dx
                constraints.append(c5_var >= overlap)
            
            # Objective
            objective = cp.Minimize(c5_var)
            prob = cp.Problem(objective, constraints)
            
            # Solve
            prob.solve(solver=cp.GLPK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                c5_new = c5_var.value
                
                # Project back to feasible set
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                
                # Evaluate
                actual_c5 = evaluate_erdos_solution(h_new)
                
                # Update if better
                if actual_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = actual_c5
                    last_improvement = iteration
                    current_h = h_new.copy()
        except:
            pass
        
        # Find most constraining shift not currently in active set
        max_overlap = -1
        best_shift = -1
        
        for k in available_shifts:
            if k not in active_shifts:
                # Compute overlap for shift k against current h
                dx = 2.0 / n_points
                overlap = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap += current_h[i] * (1 - current_h[j]) * dx
                if overlap > max_overlap:
                    max_overlap = overlap
                    best_shift = k
        
        # Add new shift if significant
        if best_shift != -1:
            active_shifts.add(best_shift)
        
        # Occasionally attempt local optimization
        if iteration % 50 == 0:
            try:
                # Use local refinement
                bounds = [(0, 1) for _ in range(n_points)]
                
                def objective(x):
                    return evaluate_erdos_solution(x)
                
                def constraint_sum(x):
                    return np.sum(x) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(
                    objective,
                    current_h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'ftol': 1e-6, 'maxiter': 50}
                )
                
                if result.success:
                    h_new = result.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        current_h = h_new.copy()
                        last_improvement = iteration
            except:
                pass
    
    # Final polishing with SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(x):
            return evaluate_erdos_solution(x)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            objective,
            best_h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2.0) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final)
            
            if final_c5 < best_c5:
                best_h = h_final.copy()
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
```
