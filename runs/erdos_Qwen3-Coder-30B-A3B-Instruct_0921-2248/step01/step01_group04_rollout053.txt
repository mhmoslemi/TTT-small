import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 200
        # Start with binary mixture and shuffle
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use column generation approach
    # Start with a subset of shifts
    max_shifts = 2 * n_points - 1
    selected_shifts = set()
    
    # Start with some random initial shifts
    initial_k = np.random.choice(max_shifts, size=min(50, max_shifts), replace=False)
    selected_shifts.update(initial_k)
    
    # Keep track of the best solution found
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Create all overlaps for current selected shifts
        # We'll use a different approach: randomized sampling and refinement
        
        # Make a perturbed version of current best
        h_perturbed = h.copy()
        # Add small noise to explore
        noise_scale = 0.02
        noise = np.random.normal(0, noise_scale, n_points)
        h_perturbed += noise
        
        # Project back to feasible bounds
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Evaluate
        c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
            
        # Run a small local optimization step as well
        def objective(x):
            return evaluate_erdos_solution(x)
            
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Local optimization with SLSQP
        h_local = h.copy()
        try:
            result = minimize(
                objective,
                h_local,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': 20}
            )
            if result.success:
                h_local = result.x
                # Project back to valid region
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points / 2.0) / np.sum(h_local)
                
                c5_local = evaluate_erdos_solution(h_local)
                if c5_local < best_c5:
                    best_h = h_local.copy()
                    best_c5 = c5_local
        except:
            pass
            
        h = best_h.copy()
    
    # Final refinement with SLSQP
    def objective(x):
        return evaluate_erdos_solution(x)
        
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2.0) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
