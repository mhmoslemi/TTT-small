import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        h = np.array(initial_h_values)
        n_points = len(h)
    else:

        h_coarse = np.zeros(n_coarse)
        half_coarse = n_coarse // 2
        h_coarse[:half_coarse] = 1
        np.random.shuffle(h_coarse)
        
        # Upsample using linear interpolation
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, 200)
        f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
        h = f(x_fine)
        n_points = len(h)
    
    # Ensure initial sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    steps_per_temp = 100
    
    # Cooling schedule
    T = T0
    temperature_schedule = []
    
    best_h_global = h.copy()
    best_c5_global = evaluate_erdos_solution(best_h_global)
    
    # Annealing loop
    while T > T_min and time.time() - start_time < budget_s - 1:
        for _ in range(steps_per_temp):
            # Make a copy for this iteration
            h_current = h.copy()
            
            # Perturb a random index
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.1 * T)
            h_new = h_current.copy()
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept with SA probability
            delta_C = c5_new - best_c5
            if delta_C < 0 or np.random.rand() < np.exp(-delta_C / T):
                h = h_new.copy()
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    
            # Update global best
            if c5_new < best_c5_global:
                best_h_global = h_new.copy()
                best_c5_global = c5_new
                
        T *= alpha
        if T < T_min:
            T = T_min
            
    # Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Simple optimization
    result = minimize(
        lambda x: evaluate_erdos_solution(x),

        method='SLSQP',
        bounds=bounds,
        constraints=cons,

    final_h = np.clip(final_h, 0, 1)

    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
