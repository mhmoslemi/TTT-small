def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 500
    noise_magnitude = 0.01
    
    # Multi-start optimization with different initializations
    for i in range(20):  # 20 starts

        if i % 2 == 0:

            h_start = h_start * target_sum / h_start_sum

            res = minimize(
