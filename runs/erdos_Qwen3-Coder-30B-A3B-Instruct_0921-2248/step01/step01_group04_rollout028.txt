import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

        np.random.shuffle(h)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Evolutionary algorithm parameters
    n_pop = 50
    generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    for _ in range(n_pop):
        candidate = h.copy() if 'initial_h_values' in globals() else np.random.rand(n_points)
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points / 2.0) / np.sum(candidate)
        population.append(candidate)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        
        # Track global best
        for i, f in enumerate(fitness):
            if f < best_c5:
                best_h = population[i].copy()
                best_c5 = f
        
        # Selection (tournament)
        new_population = []
        for _ in range(n_pop):
            # Tournament selection
            tournament_indices = np.random.choice(n_pop, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_index].copy())
        
        # Crossover and mutation
        for i in range(0, n_pop, 2):
            if np.random.rand() < crossover_rate and i + 1 < n_pop:
                parent1 = new_population[i]
                parent2 = new_population[i+1]

                crossover_point = np.random.randint(1, n_points-1)

                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = new_population[i].copy()

    # Double-check improvement
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
