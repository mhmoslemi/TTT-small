import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        np.random.shuffle(h_init)
    
    # Initialize best solution
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint function for sum
    def sum_constraint(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': sum_constraint}
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    # Try multiple random restarts
    max_restarts = 1000
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_restarts:
        # Random restart
        h0 = np.random.rand(n_points)
        # Scale to meet constraint
        h_sum = np.sum(h0)
        h0 = h0 * (n_points / 2) / h_sum
        h0 = np.clip(h0, 0, 1)
        h_sum = np.sum(h0)
        h0 = h0 * (n_points / 2) / h_sum
        
        # Local optimization with SLSQP
        try:
            res = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
            if res.success:
                h_opt = res.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                h_sum = np.sum(h_opt)
                h_opt = h_opt * (n_points / 2) / h_sum
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
            else:
                h_projected = np.clip(h0, 0, 1)
                h_sum = np.sum(h_projected)
                h_projected = h_projected * (n_points / 2) / h_sum
                c5 = evaluate_erdos_solution(h_projected)
                if c5 < best_c5:
                    best_h = h_projected.copy()
                    best_c5 = c5
        except Exception:
            h_projected = np.clip(h0, 0, 1)
            h_sum = np.sum(h_projected)
            h_projected = h_projected * (n_points / 2) / h_sum
            c5 = evaluate_erdos_solution(h_projected)
            if c5 < best_c5:
                best_h = h_projected.copy()
                best_c5 = c5
                
        iterations += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
