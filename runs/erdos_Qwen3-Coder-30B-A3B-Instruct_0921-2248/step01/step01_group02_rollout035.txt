import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def upsample_h(initial_h, target_n):
    """Upsample initial_h to target_n points using linear interpolation"""
    if len(initial_h) == target_n:
        return initial_h.copy()
    
    x_old = np.linspace(0, 2, len(initial_h))
    x_new = np.linspace(0, 2, target_n)
    f = interp1d(x_old, initial_h, kind='linear', fill_value='extrapolate')
    return f(x_new)

def initialize_population(n_points, initial_h=None, population_size=50):
    """Initialize a population of feasible h vectors"""
    population = []
    
    # Start with the initial construction if available
    if initial_h is not None:
        h_upsampled = upsample_h(initial_h, n_points)
        # Add some variation to this initial solution
        for _ in range(population_size // 2):
            h_perturbed = h_upsampled + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_sum = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (n_points / 2) / h_sum
            population.append(h_perturbed)
    else:
        # Create random feasible solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            h = np.clip(h, 0, 1)
            population.append(h)
    
    # Add some random solutions to diversify
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def mutate_h(h, mutation_rate=0.1, sigma=0.1):
    """Mutate a h vector respecting constraints"""
    h_mut = h.copy()
    for i in range(len(h_mut)):
        if np.random.random() < mutation_rate:
            # Apply Gaussian perturbation
            h_mut[i] += np.random.normal(0, sigma)
            # Random reset
            if np.random.random() < 0.3:
                h_mut[i] = np.random.random()
    
    # Clip values to [0,1]
    h_mut = np.clip(h_mut, 0, 1)
    # Re-normalize
    h_sum = np.sum(h_mut)
    h_mut = h_mut * (len(h_mut) / 2) / h_sum
    
    return h_mut

def crossover_h(h1, h2, alpha=0.5):
    """Blend crossover with repair"""
    child = alpha * h1 + (1 - alpha) * h2
    # Ensure feasibility
    child = np.clip(child, 0, 1)
    # Re-normalize
    h_sum = np.sum(child)
    child = child * (len(child) / 2) / h_sum
    return child

def selection_tournament(pop, fitnesses, k=3):
    """Tournament selection"""
    selected_indices = []
    for _ in range(len(pop)):
        # Select k individuals randomly
        tournament_indices = np.random.choice(len(pop), k)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        # Select the best
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_idx)
    return [pop[i] for i in selected_indices]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 100  # fallback resolution
        h_init = None
    
    # Adjust resolution if too small or large
    n_points = min(max(n_points, 50), 500)
    
    # Initialize population
    population = initialize_population(n_points, h_init, population_size=50)
    
    # Evaluation
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # GA parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    max_generations = 5000
    stagnation_limit = 100
    stagnation_count = 0
    last_best = best_c5
    
    # Main GA loop
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Selection
        selected = selection_tournament(population, fitnesses, k=3)
        
        # Create new population
        new_population = []
        
        # Keep best individuals
        top_indices = np.argsort(fitnesses)[:population_size // 4]
        for idx in top_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Crossover
            if np.random.random() < crossover_rate:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                child = crossover_h(parent1, parent2)
            else:
                # Just pick one parent and mutate it
                parent = np.random.choice(selected)
                child = parent.copy()
            
            # Mutate
            child = mutate_h(child, mutation_rate=mutation_rate, sigma=0.05)
            
            new_population.append(child)
        
        # Truncate population
        population = new_population[:population_size]
        
        # Evaluate
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        current_best = fitnesses[current_best_idx]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early stopping if no improvement for a while
        if stagnation_count > stagnation_limit:
            break
            
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
