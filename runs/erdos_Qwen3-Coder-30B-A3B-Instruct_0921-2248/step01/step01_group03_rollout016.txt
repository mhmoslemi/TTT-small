import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Define the problem using a more structured approach
    # We'll make use of both gradient optimization and a heuristic refinement
    
    # Gradient-based optimization with constraints using SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Try several optimization strategies
    strategies = []
    
    # Strategy 1: Random perturbations with optimization
    h_candidates = []
    max_candidates = 50
    for i in range(max_candidates):
        if time.time() - start_time > budget_s - 1:
            break
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = project(h_pert)
        strategies.append(h_pert)
        
    # Strategy 2: Improve using a known good heuristic pattern
    if n_points >= 100:
        h_heuristic = np.zeros(n_points)
        # Make a somewhat optimized pattern
        half = n_points // 2
        h_heuristic[:half] = 1.0
        h_heuristic = np.random.permutation(h_heuristic)
        # Make it a bit more smoothed
        for i in range(1, n_points-1):
            h_heuristic[i] = 0.5 * h_heuristic[i] + 0.25 * (h_heuristic[i-1] + h_heuristic[i+1])
        h_heuristic = project(h_heuristic)
        strategies.append(h_heuristic)
    
    # Strategy 3: Use direct optimization for a subset of the space for a few steps
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = project(h_pert) 
        
        try:
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 50, 'disp': False}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass  # Do nothing on failure

    # Evaluate all candidate strategies
    for h in strategies:
        if time.time() - start_time > budget_s - 1:
            break
        h_projected = project(h)
        c5 = evaluate_c5(h_projected, dx)
        if c5 < best_c5:
            best_h = h_projected
            best_c5 = c5

    # Final optimization refinement
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        h_pert = best_h + np.random.normal(0, 0.005, n_points)
        h_pert = project(h_pert)
        
        try:
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 20, 'disp': False}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass

    return (best_h, best_c5, n_points)
