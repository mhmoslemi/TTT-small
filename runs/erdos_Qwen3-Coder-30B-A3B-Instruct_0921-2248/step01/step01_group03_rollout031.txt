import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Multi-start optimization
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random initialization with Sobol-like structure
        h_pert = np.random.rand(n_points) * 0.5
        h_pert = project(h_pert)
        
        # Local optimization with different methods
        methods = ['L-BFGS-B', 'TNC', 'SLSQP']
        for method in methods:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                result = minimize(
                    objective,
                    h_pert,
                    method=method,
                    bounds=bounds,
                    constraints=con,
                    tol=1e-6,
                    options={'maxiter': 100, 'disp': False}
                )
                if result.success:
                    h_opt = project(result.x)
                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
            except:
                continue
                
        # Add small perturbation to explore
        h_pert = best_h + np.random.normal(0, 0.005, n_points) * (1 - np.random.rand(n_points) * 0.5)
        h_pert = project(h_pert)
        
        try:
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 50, 'disp': False}
            )
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            continue
    
    # Final refinement with gradient-based method
    try:
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=con,
            tol=1e-8,
            options={'maxiter': 200}
        )
        if result.success:
            h_refined = project(result.x)
            c5_refined = evaluate_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    return (best_h, best_c5, n_points)
