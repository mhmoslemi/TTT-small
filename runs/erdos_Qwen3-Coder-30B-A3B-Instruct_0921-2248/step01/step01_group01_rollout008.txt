def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx

    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiplicative factor for noise perturbations
    noise_factor = 0.05
    
    # Main optimization loop
    iteration = 0
    no_improvement_count = 0

    while time.time() - start_time < budget_s - 1:  # leave 1 seconds for final eval
        # Ensure feasible h
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
            
        # Local optimization using scipy
        try:
            res = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
            )

            if res.success:
                h = res.x.copy()
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
                
        except Exception as e:
            # If optimization fails, try random search
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        
        # Regularization using noise for diversity
        if iteration % 20 == 0:
            noise = np.random.normal(0, noise_factor, n_points)
            h = h + noise

        # Small perturbation with decreasing noise scale
        if no_improvement_count > 10:
            noise = np.random.normal(0, noise_factor / 5, n_points)
            h = h + noise
            noise_factor *= 1.1

        # Projection to keep feasible
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
            
        # Occasionally increase resolution if we're close to good solution
        if iteration % 100 == 0 and n_points == 50:
            # Interpolate to higher resolution
            n_points_high = 100
            h_high = np.interp(np.linspace(0, 2, n_points_high), 
                              np.linspace(0, 2, len(h)), h)
            h = h_high
            dx = 2.0 / n_points_high
            target_sum = n_points_high / 2.0
            bounds = [(0, 1) for _ in range(n_points_high)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
        iteration += 1

        # Check time
        if time.time() - start_time > budget_s - 1:
            break
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
