import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Helper function to project h to feasible region
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        target_sum = n_points / 2.0
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def obj_func(h_vec):
        h_vec = project_h(h_vec)
        return evaluate_erdos_solution(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - (n_points / 2.0)
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Multi-resolution approach with upsampling and optimization
    upscale_sequence = [2, 2, 12]
    remaining_budget = budget_s
    
    for upscale_factor in upscale_sequence:
        if remaining_budget <= 5:
            break
            
        # Create new grid with upsampling
        new_n_points = n_points * upscale_factor
        h_upsampled = np.repeat(h, upscale_factor)
        h = h_upsampled.copy()
        n_points = new_n_points
        
        # Add small noise to encourage exploration
        noise_level = 0.05
        noise = np.random.uniform(-noise_level, noise_level, n_points)
        h = h + noise
        h = project_h(h)
        
        # Optimization with bfgs
        op_start = time.time()
        try:
            res = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success:
                h_new = project_h(res.x)
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                h_new = project_h(h)
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception:
            h_new = project_h(h)
            c5_new = obj_func(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        op_time = time.time() - op_start
        remaining_budget -= op_time
        h = project_h(h)
        # Random restart
        if np.random.random() < 0.3 and remaining_budget > 10:
            h = np.random.rand(n_points)
            h = project_h(h)
            c5_new = obj_func(h)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        h = project_h(h)
    
    # Final randomization and optimization if time left
    if remaining_budget > 20:
        for _ in range(5):
            if time.time() - start_time > budget_s - 5:
                break
            h = np.random.rand(n_points)
            h = project_h(h)
            try:
                res = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
                if res.success:
                    h_new = project_h(res.x)
                    c5_new = obj_func(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except:
                pass
    
    # Final evaluation
    if best_h is None:
        best_h = h.copy()
    final_h = project_h(best_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    # Validate and return
    assert 0 <= np.min(final_h) <= np.max(final_h) <= 1
    assert abs(np.sum(final_h) - (n_points / 2.0)) < 1e-6
    
    return (final_h.tolist(), float(final_c5), len(final_h))
