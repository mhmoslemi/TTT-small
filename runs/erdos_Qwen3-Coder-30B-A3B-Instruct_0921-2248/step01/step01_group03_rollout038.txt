import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    correlation = correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 20
    max_iterations = 5000
    neighborhood_size = 10
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def generate_candidate(h, num_changes=2):
        candidate = h.copy()
        indices = np.random.choice(n_points, num_changes, replace=False)
        for i in indices:
            candidate[i] = 1 - candidate[i]
        return project(candidate)
    
    def is_tabu(h_candidate, tabu_list, dx):
        for h_prev, c5_prev in tabu_list:
            if np.allclose(h_candidate, h_prev, atol=1e-6):
                return True
        return False
    
    iteration = 0
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        # Tabu search neighborhood
        candidates = []
        for _ in range(neighborhood_size):
            candidate = generate_candidate(best_h)
            if not is_tabu(candidate, tabu_list, dx):
                c5_candidate = evaluate_c5(candidate, dx)
                candidates.append((candidate, c5_candidate))
        
        if candidates:
            # Find best non-tabu candidate
            best_candidate, best_c5_candidate = min(candidates, key=lambda x: x[1])
            if best_c5_candidate < best_c5:
                best_h = best_candidate
                best_c5 = best_c5_candidate
        
        # Update Tabu list
        tabu_list.append((best_h.copy(), best_c5))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Continuous local search
        if iteration % 100 == 0:
            def objective(h):
                return evaluate_c5(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            con = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = project(h_pert)
            
            try:
                from scipy.optimize import minimize
                result = minimize(
                    objective,
                    h_pert,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=con,
                    tol=1e-6,
                    options={'maxiter': 100, 'disp': False}
                )
                if result.success:
                    h_opt = project(result.x)
                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
            except:
                pass

        iteration += 1
    
    return (best_h, best_c5, n_points)
