import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Shuffled binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define upsampling sequence
    upscale_factors = [2, 2, 12]
    upsampled_sizes = [n_points, n_points*2, n_points*4, n_points*12]
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Optimization stages
    stage_budget = budget_s / len(upscale_factors) * 0.9
    
    for stage_idx, (upscale, size) in enumerate(zip(upscale_factors, upsampled_sizes)):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Upsample
        h_upsampled = np.repeat(h, upscale)
        
        # Pad to desired size
        if len(h_upsampled) < size:
            h_upsampled = np.pad(h_upsampled, (0, size - len(h_upsampled)), 'constant')
        elif len(h_upsampled) > size:
            h_upsampled = h_upsampled[:size]
        
        h = h_upsampled
        n_points = size
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        
        bounds = [(0.0, 1.0)] * n_points
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Add controlled noise
        noise_amp = 0.1 * (1 - (stage_idx + 1) / len(upscale_factors))
        noise = np.random.uniform(-noise_amp, noise_amp, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Optimize
        opt_start = time.time()
        max_iter = max(100, int(stage_budget / 10))
        
        try:
            res = minimize(obj_func, h.copy(), method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                h = h * target_sum / np.sum(h)
                c5_new = obj_func(h)
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    
        except Exception:
            # Fallback
            c5_new = obj_func(h)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
        # Random restart occasionally
        if np.random.random() < 0.1:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            c5_new = obj_func(h)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), len(final_h))
