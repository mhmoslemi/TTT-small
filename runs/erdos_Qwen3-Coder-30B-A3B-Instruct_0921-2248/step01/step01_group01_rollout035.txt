import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[::2] = 1.0  # Place 1s at even indices
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-scale optimization
    scales = [50, 100, 150]
    for scale in scales:
        if time.time() - start_time > budget_s - 1:
            break
            
        if scale > n_points:
            # Interpolate to higher resolution
            old_h = h.copy()
            h = np.interp(np.linspace(0, 2, scale), np.linspace(0, 2, n_points), old_h)
            n_points = scale
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Multiple restarts for better exploration
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb and prepare h
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
                
            # Optimize
            try:
                res = minimize(
                    objective,
                    h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6 if scale == 150 else 1e-4}
                )

                if res.success:
                    h = res.x.copy()
                    c5 = objective(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        
                    # Add noise for escape from local minima
                    noise_scale = 0.05 * best_c5 if best_c5 > 1e-10 else 0.05
                    noise = np.random.normal(0, noise_scale, n_points)
                    h = h + noise
                else:
                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)
                    
            except Exception as e:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
                
        # Project back onto feasible set
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
        
    # Final check with highest resolution if needed
    if len(best_h) < 150 and time.time() - start_time < budget_s - 1:
        n_points = 150
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        old_h = best_h.copy()
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(old_h)), old_h)
        # Optimize once more at high resolution
        try:
            res = minimize(
                objective,
                h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 250, 'ftol': 1e-8}
            )
            if res.success:
                h = res.x.copy()
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass
            
    # Final projection
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
