```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 200
        # Start with binary mixture and shuffle
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach with CVXPY
    # We will iteratively add constraints for the most violated shifts
    
    # Initialize with subset of shifts
    max_shifts = 2 * n_points - 1
    available_shifts = list(range(max_shifts))
    initial_k = min(50, len(available_shifts))
    selected_k = np.random.choice(available_shifts, initial_k, replace=False).tolist()
    
    # Track best solution
    current_h = h.copy()
    current_c5 = best_c5
    
    # Main column generation loop
    while time.time() - start_time < budget_s - 5:
        # Create optimization model
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2)
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Add overlap constraints for selected shifts
        for k in selected_k:
            # For shift k, compute overlap constraint
            # We compute: sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k) % n_points])
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(c5_var >= overlap_sum * (2.0 / n_points))
        
        # Objective
        objective = cp.Minimize(c5_var)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_sol = h_vars.value
                c5_sol = c5_var.value
                if c5_sol < current_c5:
                    current_h = np.array(h_sol)
                    current_c5 = c5_sol
                    # Check if improvement
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
        except:
            pass  # Skip iteration if infeasible or error
        
        # Find most violated shift
        max_violation = -1
        best_k = -1
        for k in available_shifts:
            if k not in selected_k:
                # Compute overlap for this shift
                overlap_sum = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap_sum += current_h[i] * (1 - current_h[j])
                violation = overlap_sum * (2.0 / n_points) - current_c5
                if violation > max_violation:
                    max_violation = violation
                    best_k = k
        
        # If a good violation found, add it
        if best_k >= 0 and max_violation > 1e-5:
            selected_k.append(best_k)
        else:
            # No significant improvement found - break
            break
    
    # Final refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization with SLSQP
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    if result.success:
        h_final = result.x
    else:
        h_final = best_h
    
    # Project back to feasible set
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_final)
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
```
