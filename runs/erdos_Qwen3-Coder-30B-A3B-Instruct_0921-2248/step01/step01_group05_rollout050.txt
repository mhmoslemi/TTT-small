import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        # Ensure h_vec values are in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Try starting with upsampling
    upsample_factors = [2, 2, 12]  # Total upsample to 47*2*2*12 = 1128, but clip to 200
    current_h = h.copy()
    current_n = n_points
    
    # Multi-resolution optimization
    for factor in upsample_factors:
        if time.time() - start_time > budget_s - 10:
            break
        next_n = current_n * factor
        if next_n > 200:
            next_n = 200
        # Upsample by repeating the values
        upsampled_h = np.repeat(current_h, factor)[:next_n]
        # Adjust to exact size if necessary
        if len(upsampled_h) < next_n:
            upsampled_h = np.pad(upsampled_h, (0, next_n - len(upsampled_h)), mode='constant')
        elif len(upsampled_h) > next_n:
            upsampled_h = upsampled_h[:next_n]
        
        # Add noise to help escape local optima
        perturbation_strength = 0.1
        noise = np.random.uniform(-perturbation_strength, perturbation_strength, next_n)
        upsampled_h = upsampled_h + noise
        upsampled_h = np.clip(upsampled_h, 0, 1)
        
        # Project back to constraint
        current_sum = np.sum(upsampled_h)
        upsampled_h = upsampled_h * (next_n / 2.0) / current_sum
        
        # Optimize the upsampled version with LBFGS-B
        try:
            bounds_upsample = [(0.0, 1.0)] * next_n
            res = minimize(obj_func, upsampled_h.copy(), method='L-BFGS-B', bounds=bounds_upsample, 
                           options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
            
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint with projection
                current_sum = np.sum(h_new)
                h_new = h_new * (next_n / 2.0) / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            current_h = h_new.copy()
            current_n = next_n
        except Exception:
            break
            
        # Add a small random mutation after each upsample to introduce variability
        if time.time() - start_time < budget_s - 10:
            mutated_h = current_h + np.random.normal(0, 0.01, current_n)
            mutated_h = np.clip(mutated_h, 0, 1)
            mutated_sum = np.sum(mutated_h)
            mutated_h = mutated_h * (current_n / 2.0) / mutated_sum
            mutated_c5 = obj_func(mutated_h)
            if mutated_c5 < best_c5:
                best_h = mutated_h.copy()
                best_c5 = mutated_c5
        else:
            break
    
    # Final refinement with high-resolution optimization if time permits
    if time.time() - start_time < budget_s - 10:
        bounds_fine = [(0.0, 1.0)] * (200 if current_n < 200 else current_n)

        try:
            res = minimize(obj_func, best_h.copy(), method='L-BFGS-B', bounds=bounds_fine,
                           options={'maxiter': 200, 'ftol': 1e-7, 'gtol': 1e-7})
            if res.success or res.fun < best_c5:
                h_final = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_final)
                h_final = h_final * (len(h_final) / 2.0) / current_sum
                c5_final = obj_func(h_final)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if abs(final_sum - (len(best_h) / 2.0)) > 1e-8:
        best_h = best_h * (len(best_h) / 2.0) / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
