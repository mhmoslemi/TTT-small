import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default: Create a smooth, balanced construction
        n_points = 100
        h_init = np.ones(n_points)
        h_init[:n_points//2] = 0.0
    
    # Scale to target sum
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Prepare for optimization
    dx = 2.0 / n_points
    
    # Build kernel matrix for correlation
    h_best = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_best)
    
    # Use CVXPY for improved optimization
    try:
        # Define optimization variables
        h = cp.Variable(n_points, name='h')
        C5 = cp.Variable(name='C5')
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum,
            C5 >= 0
        ]
        
        # Add correlation constraints
        # Compute correlation values with C5 upper bound
        # Reshape to handle convolution
        for k in range(n_points):
            # Build sum for shift k
            left = h[:n_points-k] if k < n_points else []
            right = h[k:] if k < n_points else []
            if len(left) > 0 and len(right) > 0:
                # sys using cross-correlation in a slack form
                shift_sum = cp.sum(cp.multiply(left, 1 - right))
                constraints.append(shift_sum <= C5 * (target_sum / 2.0))
        
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Solve with MOSEK
        prob.solve(solver=cp.MOSEK, verbose=False, max_iters=1000)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = h.value
            c5_val = C5.value
            if not np.isnan(c5_val):
                h_best = np.maximum(0, np.minimum(1, h_val))
                best_c5 = c5_val * dx
                # Recheck constraints
                actual_sum = np.sum(h_best)
                h_best = h_best * target_sum / actual_sum
                best_c5 = evaluate_erdos_solution(h_best)
                
    except Exception as e:
        pass
    
    # Final attempt on any further improvement by local search
    if time.time() - start_time < budget_s - 1:
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
            max_overlap = np.max(overlap)
            return max_overlap * dx
        
        # Shoot for better local optima with standard optimization
        from scipy.optimize import differential_evolution
        
        bounds = [(0.0, 1.0)] * n_points
        res = differential_evolution(obj_func, bounds, seed=seed,
                                   maxiter=200, popsize=10,
                                   mutation=(0.5, 1.0), recombination=0.7,
                                   strategy='best1bin')
        
        if res.success and res.fun < best_c5:
            h_new = np.clip(res.x, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_c5:
                h_best = h_new.copy()
                best_c5 = c5_new
    
    return (h_best.tolist(), float(best_c5), n_points)
