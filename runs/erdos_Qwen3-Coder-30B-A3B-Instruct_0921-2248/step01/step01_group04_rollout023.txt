import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Start with a reasonable construction
        n_points = 200
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach
    max_iterations = 1000
    k_set = set()
    
    # Initialize with random shifts
    for _ in range(min(50, n_points)):
        k = np.random.randint(0, 2 * n_points - 1)
        k_set.add(k)
    
    # Helper to compute overlap for a shift
    def compute_overlap(h, k):
        h_shifted = np.roll(h, k)
        return np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
    
    # Column generation loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve LP using CVXPY for current k_set
        try:
            # Variables
            h_vars = cp.Variable(n_points)
            C5 = cp.Variable()
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2, C5 >= 0]
            
            # Add overlap constraints
            for k in k_set:
                h_shifted = np.roll(h_vars.value, k) if h_vars.value is not None else np.zeros(n_points)
                shift_overlap = cp.sum(cp.multiply(h_vars, 1 - h_shifted)) * (2.0 / n_points)
                constraints.append(C5 >= shift_overlap)
            
            # Objective
            objective = cp.Minimize(C5)
            prob = cp.Problem(objective, constraints)
            
            # Solve
            prob.solve(solver=cp.GLPK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                c5_sol = C5.value
                
                # Project solution to feasible region
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2.0) / np.sum(h_sol)
                
                if c5_sol < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_sol
                    
                # Find the most constraining shift not in k_set
                max_constraint = -np.inf
                best_k = None
                
                # Check a limited number of possible shifts
                for k_candidate in range(2 * n_points - 1):
                    if k_candidate in k_set:
                        continue
                    overlap = compute_overlap(h_sol, k_candidate)
                    if overlap > max_constraint:
                        max_constraint = overlap
                        best_k = k_candidate
                
                if best_k is not None and max_constraint > c5_sol + 1e-5:
                    k_set.add(best_k)
                else:
                    # Check if we're plateauing
                    if iteration > 20 and len(k_set) > 10:
                        break
            else:
                # Fall back to local search
                pass
                
        except:
            # Fall back to SLSQP
            pass
    
    # Final polishing with SLSQP if needed
    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(x):
            return evaluate_erdos_solution(x)
        
        # Create a good starting point
        h_start = best_h.copy()
        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = best_h
        final_c5 = evaluate_erdos_solution(h_final)
        
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
