import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 300
        h = np.ones(n_points) * 0.5
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop with projected gradient
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current solution
        h_current = h.copy()
        
        # Optimization step with L-BFGS
        try:
            result = minimize(
                lambda x: evaluate_erdos_solution(x),
                h_current,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'gtol': 1e-6, 'maxiter': 100}
            )
            
            if result.success:
                h_new = result.x
            else:
                h_new = h_current
                
        except:
            # fallback to a simple gradient step if L-BFGS fails
            h_new = h_current + 0.01 * (np.random.random(n_points) - 0.5)
            h_new = np.clip(h_new, 0, 1)
            
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept better solutions
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Use new h for next iteration
        h = h_new.copy()
        
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
