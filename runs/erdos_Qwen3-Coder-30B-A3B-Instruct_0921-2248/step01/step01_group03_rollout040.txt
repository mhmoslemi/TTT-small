import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 20
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor by swapping two elements
        h_neighbor = best_h.copy()
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_neighbor[idx1], h_neighbor[idx2] = h_neighbor[idx2], h_neighbor[idx1]
        
        # Project to feasible region
        h_neighbor = project(h_neighbor)
        
        # Check Tabu list
        tabu_found = False
        for tabu_h, tabu_c5 in tabu_list:
            if np.allclose(h_neighbor, tabu_h, atol=1e-6):
                tabu_found = True
                break
                
        # Evaluate neighbor
        c5_neighbor = evaluate_c5(h_neighbor, dx)
        
        # Accept if better or not in Tabu list
        if not tabu_found or c5_neighbor < best_c5:
            if c5_neighbor < best_c5:
                best_h = h_neighbor
                best_c5 = c5_neighbor
                
            # Add to Tabu list
            tabu_list.append((h_neighbor.copy(), c5_neighbor))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        
        # Local refinement with gradient-based optimization
        if i % 50 == 0:
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = project(h_pert)
            
            try:
                result = minimize(
                    objective,
                    h_pert,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=con,
                    tol=1e-6,
                    options={'maxiter': 50}
                )
                if result.success:
                    h_opt = project(result.x)
                    c5_opt = evaluate_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
            except:
                pass
    
    return (best_h, best_c5, n_points)
