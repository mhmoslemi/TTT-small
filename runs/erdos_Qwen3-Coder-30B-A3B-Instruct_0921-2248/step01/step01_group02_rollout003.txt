import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def project_to_constraint(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2"""
    h_sum = np.sum(h)
    if h_sum > 0:
        return h * (n_points / 2) / h_sum
    else:
        return np.ones_like(h) * 0.5

    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_vel = 0.2
    
    # Initialize swarm
    swarm = np.random.random((swarm_size, n_points)) * 0.5
    velocities = np.random.random((swarm_size, n_points)) * 0.1

    personal_best = swarm.copy()

    # PSO iteration
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Evaluate current swarm
        for i in range(swarm_size):
            h = np.clip(swarm[i], 0, 1)
            h = project_to_constraint(h, n_points)
            score = evaluate_erdos_solution(h)

            if iter_count % 20 == 0 and i < swarm_size // 5:

                bounds = [(0, 1) for _ in range(n_points)]

                    return np.sum(h) - n_points / 2
