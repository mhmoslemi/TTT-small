def run(seed=42, budget_s=600, **kwargs):

        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Run optimization
    while time.time() - start_time < budget_s - 1:  # leave 1 seconds for final eval
        # Ensure feasible h
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
        
        # Optimize
        try:
            # Use L-BFGS-B with trust-constr for better convergence
            res = minimize(
                objective,
                h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-9}
            )

            if res.success:
                h = res.x.copy()
                c5 = objective(h)
                if c5 < best_c5:
