def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Compute overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start strategy with adaptive perturbations
    max_starts = 20
    for start in range(max_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize with either parent or random
        if start == 0 and 'initial_h_values' in globals():
            h_init = np.array(initial_h_values, dtype=float)
        else:
            h_init = np.random.rand(n_points)
            h_init = h_init * target_sum / np.sum(h_init)
            
        # Optimize
        try:
            res = minimize(
                compute_overlap,
                h_init.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                h_opt = res.x.copy()
                c5 = compute_overlap(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    
            # Perturbation for next start
            noise_mag = min(0.1, max(0.01, 0.01 * best_c5))
            h_pert = h_init + np.random.normal(0, noise_mag, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert_sum = np.sum(h_pert)
            h_pert = h_pert * target_sum / h_pert_sum
            
        except Exception as e:
            continue
            
        # Quick projection step
        h_proj = np.clip(h_opt, 0, 1)
        h_proj_sum = np.sum(h_proj)
        if abs(h_proj_sum - target_sum) > 1e-10:
            h_proj = h_proj * target_sum / h_proj_sum
        c5_proj = compute_overlap(h_proj)
        if c5_proj < best_c5:
            best_c5 = c5_proj
            best_h = h_proj.copy()
            
    # Final project and validation
    final_h = np.clip(best_h, 0, 1)
    final_h_sum = np.sum(final_h)
    if abs(final_h_sum - target_sum) > 1e-10:
        final_h = final_h * target_sum / final_h_sum
    final_c5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
