import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization
    upscale_factors = [2, 2, 12]  # 47 -> 94 -> 188 -> 200
    current_h = h.copy()
    current_n = n_points
    
    stage_times = [30, 30, 100]  # seconds per stage
    
    for stage_idx, (upscale, stage_time) in enumerate(zip(upscale_factors, stage_times)):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Upsample
        new_n = current_n * upscale
        new_h = np.repeat(current_h, upscale)
        # Trim or pad to exact size
        if len(new_h) > new_n:
            new_h = new_h[:new_n]
        elif len(new_h) < new_n:
            new_h = np.pad(new_h, (0, new_n - len(new_h)), 'constant', constant_values=0)
        
        current_h = new_h
        current_n = new_n
        dx = 2.0 / current_n
        target_sum = current_n / 2.0
        
        # Update bounds and constraints
        bounds = [(0.0, 1.0)] * current_n
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize with L-BFGS-B
        for _ in range(3):  # Three restarts per stage
            if time.time() - start_time > budget_s - 5:
                break
                
            try:
                # Try random start
                h_start = current_h + np.random.normal(0, 0.01, current_n)
                h_start = np.clip(h_start, 0, 1)
                h_start = h_start * target_sum / np.sum(h_start)
                
                res = minimize(obj_func, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons,
                               options={'maxiter': 50, 'ftol': 1e-7, 'gtol': 1e-7})
                
                if res.success:
                    h_new = np.clip(res.x, 0, 1)
                    h_new = h_new * target_sum / np.sum(h_new)
                    c5_new = obj_func(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        # Local random search around the better solution
                        for _ in range(5):
                            if time.time() - start_time > budget_s - 5:
                                break
                            perturbed = h_new + np.random.normal(0, 0.005, current_n)
                            perturbed = np.clip(perturbed, 0, 1)
                            perturbed = perturbed * target_sum / np.sum(perturbed)
                            c5_test = obj_func(perturbed)
                            if c5_test < best_c5:
                                best_h = perturbed.copy()
                                best_c5 = c5_test
                else:
                    # If optimization failed, try random restart
                    h_rand = np.random.rand(current_n)
                    h_rand = h_rand * target_sum / np.sum(h_rand)
                    c5_rand = obj_func(h_rand)
                    if c5_rand < best_c5:
                        best_h = h_rand.copy()
                        best_c5 = c5_rand
                        
            except Exception as e:
                pass
        
        # Optimization may not always succeed, so we will also do a final local search
        # At this point, let's do a few random local search steps to further refine
        for _ in range(10):
            if time.time() - start_time > budget_s - 5:
                break
            perturbed = best_h + np.random.normal(0, 0.005, current_n)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            c5_test = obj_func(perturbed)
            if c5_test < best_c5:
                best_h = perturbed.copy()
                best_c5 = c5_test
                
    # final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), len(final_h))
