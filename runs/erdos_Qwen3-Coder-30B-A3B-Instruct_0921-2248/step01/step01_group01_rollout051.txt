def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with 1s placed every 2 indices
        n_points = 50
        h = np.zeros(n_points)
        h[::2] = 1.0
        # Ensure sum is correct
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-scale optimization with different resolutions
    resolutions = [50, 100, 150]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 2:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            old_h = h.copy()
            h = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), old_h)
            n_points = res
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            
        # Random restarts
        for restart in range(5):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Perturb current best to avoid local minima
            if restart == 0:
                current_h = h.copy()
            else:
                # Small random perturbations with scaling
                noise_factor = 0.05 * (1.0 + 0.1 * np.random.rand())
                perturbation = np.random.normal(0, noise_factor, n_points)
                current_h = h + perturbation
                
            # Ensure feasibility
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            
            try:
                # Use L-BFGS-B for faster convergence with tighter tolerance
                res_opt = minimize(
                    objective,
                    current_h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6, 'gtol': 1e-6}
                )
                
                if res_opt.success:
                    h_opt = res_opt.x.copy()
                    c5 = objective(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        
                    # Add noise for next iteration if not last
                    if restart < 4:
                        noise_factor = 0.1 * (1.0 + 0.05 * np.random.rand())
                        noise = np.random.normal(0, noise_factor, n_points)
                        h = h_opt + noise
                    else:
                        h = h_opt.copy()
                else:
                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)
            except Exception:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
                
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
            
        # Final update of h at current resolution
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
    
    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
