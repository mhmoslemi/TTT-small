import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

    swarm_size = 50
    max_iter = 1000
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    vmax = 0.2  # maximum velocity
    
    # Initialize swarm
    swarm = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) * vmax
    
    # Initialize personal best positions and values
    personal_best_positions = swarm.copy()
    personal_best_values = np.array([evaluate_erdos_solution(p) for p in swarm])
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best_positions[i] - swarm[i]) + 
                            c2 * r2 * (global_best_position - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -vmax, vmax)

            swarm[i] += velocities[i]
            
            # Apply bounds (clip to [0,1])
            swarm[i] = np.clip(swarm[i], 0, 1)

            h_sum = np.sum(swarm[i])
            if h_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / h_sum
            
            # Evaluate
            current_value = evaluate_erdos_solution(swarm[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best_values[i] = current_value
                personal_best_positions[i] = swarm[i].copy()
                
                # Update global best if needed
                if current_value < global_best_value:
                    global_best_value = current_value
                    global_best_position = swarm[i].copy()

        if iteration % 20 == 0:
            # Select top 10% of particles for refinement

                return np.sum(h) - n_points / 2
