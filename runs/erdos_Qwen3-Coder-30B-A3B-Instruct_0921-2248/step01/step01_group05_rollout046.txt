import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback: simple construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization: upscale from 47 to 200 points
    upscale_factors = [2, 2, 12]  # 47 -> 94 -> 188 -> 200
    resolutions = [47, 94, 188, 200]
    
    for i, (upscale, res) in enumerate(zip(upscale_factors, resolutions)):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Upsample
        h_upsampled = np.repeat(h, upscale)
        h_upsampled = h_upsampled[:res]
        h = h_upsampled
        n_points = res
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        bounds = [(0.0, 1.0)] * n_points
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize at this resolution
        for _ in range(3):
            if time.time() - start_time > budget_s - 5:
                break
                
            try:
                res_opt = minimize(obj_func, h.copy(), method='L-BFGSB', bounds=bounds, constraints=cons, 
                                   options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
                
                if res_opt.success:
                    h_new = np.clip(res_opt.x, 0, 1)
                    current_sum = np.sum(h_new)
                    h_new = h_new * target_sum / current_sum
                    c5_new = obj_func(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except:
                pass
            
            # Random restart
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
    
    # Final local search
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Small perturbation
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert = h_pert * target_sum / current_sum
        c5_pert = obj_func(h_pert)
        
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
