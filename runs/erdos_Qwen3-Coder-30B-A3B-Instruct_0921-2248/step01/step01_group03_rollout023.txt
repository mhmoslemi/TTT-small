import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 200
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0, dx)
    
    # Tabu search parameters
    tabu_tenure = 20
    tabu_list = []
    max_iter = 10000
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Evaluate C5
    def evaluate_c5(h):
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(correlation)
        return max_corr * dx
    
    # Neighborhood generator
    def generate_neighbor(h):
        h_new = h.copy()
        # Randomly select two indices to flip: one 0 to 1, one 1 to 0
        zeros = np.where(h_new == 0)[0]
        ones = np.where(h_new == 1)[0]
        
        if len(zeros) > 0 and len(ones) > 0:
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
            h_new[i] = 1
            h_new[j] = 0
            
            # Ensure the sum constraint
            h_new = project(h_new)
        return h_new
    
    # Adaptive Tabu search loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate
        h_candidate = generate_neighbor(best_h)
        c5_candidate = evaluate_c5(h_candidate)
        
        # Check tabu list
        is_tabu = False
        for h_tabu, c5_tabu in tabu_list:
            if np.allclose(h_candidate, h_tabu, atol=1e-6):
                is_tabu = True
                break
        
        # Accept if better or not tabu
        if c5_candidate < best_c5 or not is_tabu:
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
            
            # Add to tabu list
            tabu_list.append((h_candidate.copy(), c5_candidate))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        
        # Sync with optimizer if time allows
        if i % 100 == 0 and time.time() - start_time < budget_s - 1:
            # Local continuous optimization on current best
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            con = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return evaluate_c5(h)
                
            result = minimize(
                objective,
                project(best_h),
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 10, 'disp': False}
            )
            if result.success:
                h_refined = project(result.x)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined

    return (best_h, best_c5, n_points)
