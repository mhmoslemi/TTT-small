import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default: binary construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Upsample to 500 points with interpolation
    original_x = np.linspace(0, 2, n_points)
    new_x = np.linspace(0, 2, 500)
    h_upsampled = np.interp(new_x, original_x, h)
    h = h_upsampled
    n_points = 500
    
    # dx for interval [0, 2]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec

    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Iteration loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 50:
        # Nelder-Mead optimization
        try:
            res = minimize(obj_func, h.copy(), method='Nelder-Mead', 
                           options={'maxiter': 200, 'initial_simplex': None, 'xatol': 1e-6, 'fatol': 1e-6})
            if res.success and not np.isnan(res.fun) and not np.isinf(res.fun):
                h_new = project_feasible(res.x)
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
        
        # Random perturbations
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_feasible(h_perturbed)
        c5_pert = obj_func(h_perturbed)
        if c5_pert < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_pert
            
        # Restart with random sample
        h = np.random.rand(n_points)
        h = project_feasible(h)
        
        iter_count += 1
    
    # Final evaluation
    final_h = project_feasible(best_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
