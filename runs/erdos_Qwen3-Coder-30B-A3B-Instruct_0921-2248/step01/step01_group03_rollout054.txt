from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

    tabu_list = []
    tabu_tenure = 20
    max_iter = 10000
    max_no_improve = 1000

    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    def get_neighbor(h, n_flips=2):
        h_new = h.copy()
        indices = np.random.choice(n_points, n_flips, replace=False)
        for i in indices:
            h_new[i] = 1 - h_new[i]
        return project(h_new)
    
    # Binary representation
    def to_binary(h):
        return (h > 0.5).astype(int)
    
    # Main Tabu search loop
    no_improve_count = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor
        h_candidate = get_neighbor(best_h)
        
        # Skip if in Tabu list
        h_binary = to_binary(h_candidate)
        if tuple(h_binary) in tabu_list:
            continue

        c5_candidate = evaluate_c5(h_candidate, dx)

        if c5_candidate < best_c5:

        if i % 1000 == 0:
