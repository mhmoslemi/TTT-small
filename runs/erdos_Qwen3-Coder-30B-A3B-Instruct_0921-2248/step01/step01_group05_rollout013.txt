import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default construction
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        np.random.shuffle(h_init)
    
    # Scale to target sum
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Initialize best solution
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * (2.0 / n_points)
    
    # Use CVXPY to solve the optimization problem
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Define constraints
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1,
        C5 >= 0
    ]
    
    # Generate all shift indices for correlation
    # This is equivalent to computing the full correlation with mode="full"
    # We'll compute max over all valid shifts, ignoring the zero padding
    # The number of full overlaps is 2*n_points - 1
    
    dx = 2.0 / n_points
    # Use a subset of shifts to reduce complexity within time budget
    max_shift = min(100, n_points)
    
    # Build constraints for the correlation terms
    # For each shift k: sum_{i} h[i] * (1 - h[i+k]) <= C5 * (n_points/2)
    for k in range(-max_shift, max_shift + 1):
        if k < 0:
            # h[i] * (1 - h[i+k]) for i in [0, n_points+k-1]
            corr_expr = cp.sum(cp.multiply(h[:n_points+k], 1 - h[-k:]))
            constraints.append(corr_expr <= C5 * target_sum)
        elif k == 0:
            # h[i] * (1 - h[i]) = h[i] - h[i]^2
            corr_expr = cp.sum(h - cp.square(h))
            constraints.append(corr_expr <= C5 * target_sum)
        else:  # k > 0
            # h[i] * (1 - h[i+k]) for i in [k, n_points-1]
            corr_expr = cp.sum(cp.multiply(h[k:], 1 - h[:n_points-k]))
            constraints.append(corr_expr <= C5 * target_sum)

    # Objective: minimize C5
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Solve with MOSEK (if available) or default solver
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 1)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_opt = C5.value
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except:
        # Fallback to a local optimization approach if cvxpy fails
        pass
    
    # Try local optimization with the current best solution
    while time.time() - start_time < budget_s - 2:
        try:
            # Simple gradient descent on a subset of the problem for fast local refinement
            h_copy = best_h.copy()
            for _ in range(50):
                # Small perturbations with gradient descent
                grad = np.zeros_like(h_copy)
                for i in range(n_points):
                    for k in range(-10, 11):
                        if 0 <= i + k < n_points and i + k >= 0:
                            grad[i] += h_copy[i] - h_copy[i] * h_copy[i + k]
                
                # Update
                h_copy = h_copy - 0.0001 * grad
                h_copy = np.clip(h_copy, 0, 1)
                h_copy = h_copy * target_sum / np.sum(h_copy)
                
            c5_new = np.max(np.correlate(h_copy, 1 - h_copy, mode="full")) * dx
            if c5_new < best_c5:
                best_h = h_copy.copy()
                best_c5 = c5_new
        except:
            break
    
    # Make sure solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), float(final_c5), n_points)
