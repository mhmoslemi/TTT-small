import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 100
        # Start with binary mixture
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Genetic Algorithm parameters
    n_pop = 50
    n_gen = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    for i in range(n_pop):
        if i == 0:  # Use best solution as first
            ind = h.copy()
        else:
            ind = np.random.rand(n_points)
            # Make it feasible
            ind = np.clip(ind, 0, 1)
            ind = ind * (n_points / 2.0) / np.sum(ind)
        population.append(ind)
    
    # Track best solution overall
    best_overall = best_h.copy()
    best_overall_c5 = best_c5
    
    # Evolution loop
    for gen in range(n_gen):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all individuals
        fitness = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind)
            fitness.append(c5)
        
        # Selection via tournament
        new_pop = []
        for _ in range(n_pop):
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(n_pop, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_pop.append(population[winner_idx].copy())
        
        # Crossover and mutation
        next_pop = []
        for i in range(0, n_pop, 2):
            parent1 = new_pop[i]
            parent2 = new_pop[(i+1) % n_pop]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Segment-based crossover
                cut_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
                child2 = np.concatenate([parent2[:cut_point], parent1[cut_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
            
            # Feasible projection
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2.0) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2.0) / np.sum(child2)
            
            next_pop.append(child1)
            if len(next_pop) < n_pop:
                next_pop.append(child2)
        
        population = next_pop
        
        # Update best
        for i, ind in enumerate(population):
            c5 = evaluate_erdos_solution(ind)
            if c5 < best_overall_c5:
                best_overall_c5 = c5
                best_overall = ind.copy()
    
    # Final polishing with local optimization
    best_h = best_overall.copy()
    best_c5 = best_overall_c5
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use SLSQP to polish
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    
    if result.success:
        final_h = result.x
    else:
        final_h = best_h
    
    # Ensure feasibility
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2.0) / np.sum(final_h)
    
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
