import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_c5(h0, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 2000
    dual_step_size = 0.1
    
    # Initialize Lagrange multiplier
    lambda_ = 0.0
    
    # Initialize h
    h = h0.copy()
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # ADMM iterations
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Save previous h
        h_prev = h.copy()
        
        # Update h by minimizing the augmented Lagrangian
        # Hessian of the smooth part is 2*I, with regularization
        # We will do gradient descent on the correlation term
        # Approximate gradient of overlap by finite differences
        h_grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus)
            c5_minus = evaluate_c5(h_minus, dx)
            
            h_grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update h with gradient descent
        h = h - dual_step_size * h_grad
        h = project_h(h)
        
        # Update Lagrange multiplier
        h_sum = np.sum(h)
        lambda_ += rho * (h_sum - n_points / 2)
        
        # Check for convergence
        if np.linalg.norm(h - h_prev) < 1e-6:
            break
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Final refinement using local search
    def objective(h):
        return evaluate_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Random perturbations to improve solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_pert = best_h + np.random.normal(0, 0.005, n_points)
        h_pert = project_h(h_pert)
        
        result = minimize(
            objective,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            tol=1e-6,

        )

            h_opt = project_h(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
