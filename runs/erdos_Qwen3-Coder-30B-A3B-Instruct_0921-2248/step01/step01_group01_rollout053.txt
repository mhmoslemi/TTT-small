import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Need to be careful with the indexing here.
        # The full correlation gives a result of length 2*n_points - 1.
        # The maximum value is at index n_points-1 (center), but we need the max over all lags.
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Running optimizations
    iteration = 0
    perturbation_scale = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < 500:
        # Ensure feasible h
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
        
        # Local optimization with SLSQP
        try:
            res = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
            )

            if res.success:
                h = res.x.copy()
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
                # Add noise for diversity
                noise = np.random.normal(0, perturbation_scale, n_points)
                h = h + noise
            else:
                # If optimization fails, try random search
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
                
        except Exception as e:
            # If optimization fails, try random search
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            
        # Projection to keep feasible
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            h = h * target_sum / h_sum
            
        # Adjustment to perturbation scale
        if iteration % 50 == 0 and iteration > 0:
            perturbation_scale = min(perturbation_scale + 0.005, 0.1)
        
        iteration += 1
        
        # Check time
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation of best solution
    final_c5 = objective(best_h)
    
    # Further optimization at higher resolution if time allows
    if time.time() - start_time < budget_s - 2 and n_points < 150:
        h_high_res = np.interp(np.linspace(0, 2, 150), np.linspace(0, 2, n_points), best_h)
        h_high_res = h_high_res * (75.0 / np.sum(h_high_res))  # Ensure the right sum
        
        best_c5 = float('inf')
        best_h = h_high_res.copy()
        
        # Continuously optimize at high resolution
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            
            h = np.clip(h_high_res, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - 75.0) > 1e-10:
                h = h * 75.0 / h_sum
            
            try:
                res = minimize(
                    objective,
                    h.copy(),
                    method='SLSQP',
                    bounds=[(0, 1) for _ in range(150)],
                    constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - 75.0},
                    options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    h = res.x.copy()
                    c5 = objective(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                else:
                    h = np.random.rand(150)
                    h = h * 75.0 / np.sum(h)
                    
            except Exception:
                h = np.random.rand(150)
                h = h * 75.0 / np.sum(h)
            
            # Projection
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - 75.0) > 1e-10:
                h = h * 75.0 / h_sum
                
            if time.time() - start_time > budget_s - 1:
                break
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
