import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a smooth approximation to initial construction
        n_points = 200
        h_init = np.ones(n_points)
        # Smooth transition for better optimization
        h_init[:n_points//2] = 0
        # Add some variation to avoid local minima
        h_init += 0.1 * np.random.random(n_points)
        h_init = np.clip(h_init, 0, 1)
        # Normalize
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Genetic algorithm approach with local optimization
    population_size = 30
    generations = 50
    mutation_rate = 0.1
    elite_size = 5
    
    def create_individual():
        # Create a random feasible hvector
        h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        # Normalize sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.random(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        # Keep feasible
        child = np.clip(child, 0, 1)
        # Normalize
        child = child * (n_points / 2) / np.sum(child)
        return child
    
    def mutate(individual):
        # Add small Gaussian noise
        mask = np.random.random(n_points) < mutation_rate
        noise = np.random.normal(0, 0.05, n_points)
        individual = individual + noise * mask
        # Clip and normalize
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:elite_size].copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            idx1 = np.random.choice(elite_size)
            idx2 = np.random.choice(elite_size)
            parent1 = population[idx1]
            parent2 = population[idx2]
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Occasionally perform local optimization on best individual
        if gen % 5 == 0:
            try:
                # Local optimization using L-BFGS-B
                h0 = best_h.copy()
                bounds = [(0, 1) for _ in range(n_points)]
                
                def objective(h):
                    corr = np.correlate(h, 1 - h, mode="full")
                    dx = 2.0 / n_points
                    return np.max(corr) * dx
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2
                
                constraints = {'type': 'eq', 'fun': constraint_sum}
                res = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, 
                              constraints=constraints, options={'maxiter': 200})
                
                if res.success:
                    h_opt = res.x.copy()
                    h_opt = np.clip(h_opt, 0, 1)
                    h_sum = np.sum(h_opt)
                    h_opt = h_opt * (n_points / 2) / h_sum
                    c5 = evaluate_erdos_solution(h_opt)
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
            except:
                pass

    return (best_h.tolist(), best_c5, n_points)
