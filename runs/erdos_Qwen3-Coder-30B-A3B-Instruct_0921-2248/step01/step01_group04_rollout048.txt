import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create initial low-resolution guess
        n_points = 100
        h = np.zeros(n_points)
        # Two clusters: first half is 1, second half is 0
        h[:n_points//2] = 1.0
        # Add some random noise to avoid degenerate cases
        noise = np.random.normal(0, 0.02, n_points)
        h += noise
        h = np.clip(h, 0, 1)
    
    # Rescale to ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    steps_per_temp = 100
    
    def project_to_feasible(x):
        """Ensure x is in [0,1] and sum equals n_points / 2"""
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2.0) / np.sum(x)
        return x
    
    # Initial cost
    current_h = h.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    
    # Cooling schedule
    T = T0
    temp_step = 0
    
    while T > T_min and time.time() - start_time < budget_s - 1:
        # Perform steps at current temperature
        for _ in range(steps_per_temp):
            # Make a copy of current solution
            h_trial = current_h.copy()
            
            # Random mutation
            i = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_trial[i] += delta
            
            # Project to feasible domain
            h_trial = project_to_feasible(h_trial)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(h_trial)
            
            # Accept or reject based on Metropolis criterion
            delta_c = new_c5 - current_c5
            if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
                current_h = h_trial
                current_c5 = new_c5
                
                # Update best solution if improved
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        # Cool temperature
        T *= alpha
        temp_step += 1
        
        # Occasionally reduce step size if plateau detected
        if temp_step % 50 == 0 and T > T_min * 10:
            steps_per_temp = max(10, steps_per_temp - 1)
    
    # Final polishing with SLSQP if time allows
    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            post_h = result.x
            post_h = project_to_feasible(post_h)
            post_c5 = evaluate_erdos_solution(post_h)
            if post_c5 < best_c5:
                best_h = post_h
                best_c5 = post_c5
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
