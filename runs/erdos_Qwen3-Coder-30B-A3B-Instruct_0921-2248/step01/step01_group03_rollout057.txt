import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            # Project to feasible region
            h_proj = h_pert.copy()
            h_sum = np.sum(h_proj)
            if h_sum > 0:
                h_proj = h_proj * (n_points / 2) / h_sum
            
            # Evaluate
            c5_eval = evaluate_c5(h_proj, dx)
            if c5_eval < best_c5:
                best_h = h_proj.copy()

        h = cp.Variable(n_points)
        C = cp.Variable()

        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2,
            C >= 0
        ]
        
        # Append correlation constraints
        def add_correlation_constraint(shift):
            if shift >= n_points:
                return
            corr_val = 0
            for i in range(n_points):
                j = (i - shift) % n_points
                corr_val += h[i] * (1 - h[j])
            constraints.append(C >= corr_val * dx)
            
        # Add constraints for shifts 0 to n_points-1
        for k in range(n_points):
            add_correlation_constraint(k)
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Solve with MOSEK or CVXOPT

                h_result = h_result * (n_points / 2) / h_sum
                c5_result = evaluate_c5(h_result, dx)
                if c5_result < best_c5:
                    best_h = h_result
                    best_c5 = c5_result
        except Exception:
            pass
            
    except Exception:
        pass
        
    return (best_h, best_c5, n_points)
