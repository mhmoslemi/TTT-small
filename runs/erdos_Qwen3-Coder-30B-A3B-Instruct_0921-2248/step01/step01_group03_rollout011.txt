import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral using cross-correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h0 = initial_h_values
        n_points = len(h0)
    except NameError:

        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 20
    max_iter = 10000
    neighborhood_size = 10
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    def generate_neighbor(h, tabu_list):
        # Generate a neighbor by flipping two elements (one 0 -> 1, one 1 -> 0)
        h_new = h.copy()
        zeros = np.where(h_new == 0)[0]
        ones = np.where(h_new == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            # Fallback to random perturbation if no flip possible
            idx1 = np.random.randint(0, n_points)
            idx2 = np.random.randint(0, n_points)
            h_new[idx1] = 1 - h_new[idx1]
            h_new[idx2] = 1 - h_new[idx2]
        else:
            # Pick one 0 and one 1 to flip
            idx0 = np.random.choice(zeros)
            idx1 = np.random.choice(ones)
            h_new[idx0] = 1
            h_new[idx1] = 0
            
        return project(h_new)
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbors and evaluate
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        for _ in range(neighborhood_size):
            neighbor = generate_neighbor(best_h, tabu_list)
            
            # Check if this candidate has been in tabu recently
            is_tabu = any(np.array_equal(neighbor, tabu_h) for tabu_h in tabu_list)
            if not is_tabu:
                neighbor_c5 = objective(neighbor)
                if neighbor_c5 < best_neighbor_c5:
                    best_neighbor = neighbor
                    best_neighbor_c5 = neighbor_c5
        
        if best_neighbor is not None:
            # Update best solution
            if best_neighbor_c5 < best_c5:
                best_h = best_neighbor
                best_c5 = best_neighbor_c5
            
            # Add current solution to tabu list
            tabu_list.append(best_h.copy())
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            # Local refinement using SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            con = {'type': 'eq', 'fun': constraint_sum}
            
            # Find good starting point for local optimization
            h_start = project(best_h + np.random.normal(0, 0.01, n_points))
            
            try:
                result = minimize(
                    objective,
                    h_start,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=con,
                    tol=1e-6,
                    options={'maxiter': 50, 'disp': False}
                )
                if result.success:
                    h_refined = project(result.x)
                    c5_refined = evaluate_c5(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (best_h, best_c5, n_points)
