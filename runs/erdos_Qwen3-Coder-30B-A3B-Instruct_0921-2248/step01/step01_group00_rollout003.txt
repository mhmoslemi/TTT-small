import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

    # We'll still use SLSQP but with better initialization and more iterations
    def constraint_eq(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    opt_options = {'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
    
    # Enhanced random search with guiding exploration
    max_iter = int(1e5)

        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(evaluate_erdos_solution, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options=opt_options)
            if res.success and res.fun < best_c5:
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_val = evaluate_erdos_solution(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()

        # Occasionally try the original initialization to escape local minima
        if i % 500 == 0:

                    if c5_val < best_c5:
                        best_c5 = c5_val

            except Exception:
                pass
