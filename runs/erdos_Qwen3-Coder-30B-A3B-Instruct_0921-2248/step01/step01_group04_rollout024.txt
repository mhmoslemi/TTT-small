import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 200
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Column generation approach
    max_iterations = 500
    k_selected = set()
    
    # Start with small subset of shifts
    n_shifts = min(50, 2 * n_points - 1)
    for _ in range(n_shifts):
        k = np.random.randint(0, 2 * n_points - 1)
        k_selected.add(k)
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Convert k_selected to sorted list for consistency
        k_list = sorted(list(k_selected))
        
        # Build CVXPY variables
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(cp.sum(h_vars) == n_points / 2)
        constraints.append(C5 >= 0)
        
        # Add the constraints for each selected shift
        for k in k_list:
            # Compute the convolution
            conv_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                conv_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(C5 >= conv_sum * (2.0 / n_points))
        
        # Solve the LP
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h_new = np.array(h_new)
                    # Project back to feasible set
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                    
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                    
                    # Find the maximizing shift
                    h_cycle = np.concatenate([h_new, h_new])
                    max_k = 0
                    max_val = -np.inf
                    for k in range(2 * n_points - 1):
                        k_val = 0.0
                        for i in range(n_points):
                            k_val += h_cycle[i] * (1 - h_cycle[i + k])
                        if k_val > max_val:
                            max_val = k_val
                            max_k = k
                    
                    # Add this to the basis if it's not there
                    if max_k not in k_selected and k_selected:
                        k_selected.add(max_k)
                        # Keep only a limited number of shifts
                        if len(k_selected) > 150:
                            # Remove the least important shift
                            k_list_check = [k for k in k_selected if k != max_k]
                            max_vals = []
                            for k in k_list_check:
                                k_val = 0.0
                                for i in range(n_points):
                                    j = (i + k) % n_points
                                    k_val += h_new[i] * (1 - h_new[j])
                                max_vals.append(k_val)
                            # Remove shift with minimum value
                            k_remove = k_list_check[np.argmin(max_vals)]
                            k_selected.remove(k_remove)
                            
                    h = h_new.copy()
        except:
            pass
    
    # Final optimization with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    if result.success:
        final_h = result.x
    else:
        final_h = best_h
    
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2.0) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
