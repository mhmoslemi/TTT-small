def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Use correlation to compute overlap
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint function for sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Objective function
    def objective(h_vec):
        return compute_overlap(h_vec)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Population-based optimization
    num_candidates = 10
    candidates = []
    for _ in range(num_candidates):
        h_candidate = h.copy()
        # Add small noise
        noise = np.random.normal(0, 0.01, n_points)
        h_candidate += noise
        h_candidate = np.clip(h_candidate, 0, 1)
        h_sum = np.sum(h_candidate)
        h_candidate = h_candidate * target_sum / h_sum
        candidates.append(h_candidate)
    
    # Evaluate candidates
    for i in range(len(candidates)):
        c5 = compute_overlap(candidates[i])
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidates[i].copy()
    
    # Main optimization loop with adaptive strategies
    max_iter = 100
    iter_no_improve = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Update population with best solution and some new random samples
        new_population = [best_h.copy()]
        for _ in range(num_candidates - 1):
            # Random solution with correction to sum constraint
            h_rand = np.random.rand(n_points)
            h_sum = np.sum(h_rand)
            h_rand = h_rand * target_sum / h_sum
            new_population.append(h_rand)
        
        # Refine solutions in population
        for i in range(len(new_population)):
            h_candidate = new_population[i].copy()
            
            # Local optimization with constraints
            try:
                # Create a new optimization problem for just this candidate
                h_vars = cp.Variable(n_points)
                constraints = [
                    h_vars >= 0,
                    h_vars <= 1,
                    cp.sum(h_vars) == target_sum
                ]
                
                # Formating overlap as quadratic form (approximation)
                # We can't directly optimize this in CVXPY with overlap terms,
                # so we will still use the correlation approach but in a constrained
                # fashion based on previous gradient descent technique
                # Proceed with SLSQP for simplicity
                def local_objective(h_vec):
                    return compute_overlap(h_vec)
                
                # SLSQP optimization for local refinement
                res = minimize(
                    local_objective,
                    h_candidate,
                    method='SLSQP',
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    h_candidate = res.x.copy()
                else:
                    # If optimization fails, attempt a simple feasibility update
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_sum = np.sum(h_candidate)
                    h_candidate = h_candidate * target_sum / h_sum
            except:
                # Should not happen, but just in case of errors
                h_candidate = np.clip(h_candidate, 0, 1)
                h_sum = np.sum(h_candidate)
                h_candidate = h_candidate * target_sum / h_sum
                
            new_population[i] = h_candidate
            
            # Evaluation
            c5 = compute_overlap(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
                iter_no_improve = 0
            else:
                iter_no_improve += 1
                
        # Perturb best solution occasionally or if no improvement
        if iter_no_improve > 20 or iter_count % 20 == 0:
            perturbation_factor = 0.01 + 0.005 * (iter_count // 50)
            noise = np.random.normal(0, perturbation_factor, n_points)
            best_h = best_h + noise
            best_h = np.clip(best_h, 0, 1)
            h_sum = np.sum(best_h)
            best_h = best_h * target_sum / h_sum
            new_population[0] = best_h.copy()
            
        # Retain new population
        candidates = new_population
        
        # Check time
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h = best_h * target_sum / best_h_sum
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
