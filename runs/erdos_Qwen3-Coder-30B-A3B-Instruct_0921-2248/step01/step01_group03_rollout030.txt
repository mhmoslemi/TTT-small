import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:

        n_points = len(h0)
    except NameError:
        # Fall back to a structured construction

        # Start with a smooth bump and add some randomness
        h0 = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Use a Gaussian-like bump
        h0 = np.exp(-((x - 1.0)**2) / (2 * 0.2**2))
        # Normalize to sum to n_points / 2
        h0 = h0 * (n_points / 2) / np.sum(h0)
        # Add a bit of randomness
        h0 += np.random.normal(0, 0.02, n_points)
        h0 = np.clip(h0, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Gradient-based local search
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different perturbation strategies
        pert_type = i % 3
        
        if pert_type == 0:  # Local perturbation
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
        elif pert_type == 1:  # Global perturbation
            h_pert = np.random.rand(n_points) * 0.5
            h_pert = project(h_pert)
        else:  # Gradient ascent search
            # Perform a simple gradient approximation
            eps = 1e-4
            grad = np.zeros(n_points)
            base_c5 = objective(best_h)
            for j in range(n_points):
                h_test = best_h.copy()
                h_test[j] += eps
                h_test = project(h_test)
                c5_plus = objective(h_test)
                grad[j] = (c5_plus - base_c5) / eps
                
            h_pert = best_h - 0.01 * grad
            h_pert = project(h_pert)
        
        # Optimize with SLSQP
        try:
            result = minimize(
                objective,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 100, 'disp': False}
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
            
        # Always keep current best
        if i % 5 == 0:
            # Occasionally try a random restart
            h_rand = np.random.rand(n_points) * 0.5
            h_rand = project(h_rand)
            c5_rand = evaluate_c5(h_rand, dx)
            if c5_rand < best_c5:
                best_h = h_rand
                best_c5 = c5_rand
    
    return (best_h, best_c5, n_points)
