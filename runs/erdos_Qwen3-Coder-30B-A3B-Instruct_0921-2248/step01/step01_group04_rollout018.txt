import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization using scipy
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Set up optimization parameters
    max_iter = 1000
    
    # First optimization: smooth approximation
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        h_current = h.copy()

        if iteration > 0 and iteration % 50 == 0:

            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = h_current + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)

                best_c5 = c5_perturbed
                
        # Optimize with L-BFGS-B

        if c5_new < best_c5:
