import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def mutate_h(h, mutation_rate=0.1):
    """Mutate h by random changes"""
    h_new = h.copy()
    n_points = len(h)
    n_mutations = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, n_mutations, replace=False)
    for i in indices:
        h_new[i] = np.random.uniform(0, 1)
    return project_to_feasible(h_new, n_points)

def crossover_h(h1, h2):
    """Perform crossover between two h functions"""
    n_points = len(h1)
    crossover_point = np.random.randint(1, n_points)
    h_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return h_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = [project_to_feasible(np.random.rand(n_points), n_points)

    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution loop
    for generation in range(n_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(individual) for individual in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitnesses[min_fitness_idx]

        new_population = []

        elite_idx = np.argmin(fitnesses)

    h_current = best_h.copy()
