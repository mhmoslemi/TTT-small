import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        n_points = len(h_parent)
        # Upsample to target resolution
        x_parent = np.linspace(0, 2, n_points)
        x_target = np.linspace(0, 2, 200)
        f = interp1d(x_parent, h_parent, kind='linear', fill_value='extrapolate')
        h_init = f(x_target)
        n_points = len(h_init)
    else:
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
    
    # Constants for GA
    population_size = 50
    max_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    tournament_size = 3
    
    # Initialize population
    def create_individual():
        ind = h_init.copy()
        # Add noise to broad pattern
        noise = np.random.normal(0, 0.1, n_points)
        ind += noise
        # Clip and normalize
        ind = np.clip(ind, 0, 1)
        h_sum = np.sum(ind)
        ind = ind * (n_points / 2) / h_sum
        return ind
    
    def create_population():
        pop = [create_individual() for _ in range(population_size)]
        return pop
    
    def fitness(individual):
        return evaluate_erdos_solution(individual)
    
    def tournament_selection(population, fitnesses, k=tournament_size):
        selected_indices = np.random.choice(len(population), k)
        selected_fitnesses = [fitnesses[i] for i in selected_indices]
        best_idx = selected_indices[np.argmin(selected_fitnesses)]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Blend crossover (BLX-0)
        alpha = 0.5
        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        
        for i in range(n_points):
            x1, x2 = parent1[i], parent2[i]
            d = abs(x1 - x2)
            min_val = min(x1, x2) - alpha * d
            max_val = max(x1, x2) + alpha * d
            child1[i] = np.clip(np.random.uniform(min_val, max_val), 0, 1)
            child2[i] = np.clip(np.random.uniform(min_val, max_val), 0, 1)
            
        # Normalize 
        child1_sum = np.sum(child1)
        child1 = child1 * (n_points / 2) / child1_sum
        child2_sum = np.sum(child2)
        child2 = child2 * (n_points / 2) / child2_sum
        
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                # Random reset with probability 0.1
                if np.random.random() < 0.1:
                    mutated[i] = np.random.random()
                else:
                    # Gaussian perturbation
                    mutated[i] += np.random.normal(0, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        
        # Normalize
        h_sum = np.sum(mutated)
        mutated = mutated * (n_points / 2) / h_sum
        return mutated
    
    # Main GA loop 
    population = create_population()
    best_individual = None
    best_fitness = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Create next generation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            
            # Mutate
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
            
        # Trim to exact size
        population = new_population[:population_size]
    
    # Final evaluation
    final_fitness = evaluate_erdos_solution(best_individual)
    
    return (best_individual.tolist(), final_fitness, n_points)
