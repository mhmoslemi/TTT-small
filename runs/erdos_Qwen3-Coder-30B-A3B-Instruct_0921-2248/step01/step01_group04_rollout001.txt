import numpy as np
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a coarse initial solution with clusters
        n_points = 100
        h = np.zeros(n_points)
        # Split into 3 clusters
        cluster_size = n_points // 3
        h[:cluster_size] = 1.0
        h[cluster_size:2*cluster_size] = 0.0
        h[2*cluster_size:] = 1.0
        # Add some randomness
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        # Rescale to sum = n_points / 2
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    def project_feasible(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2.0) / np.sum(x)
        return x
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    steps_per_temp = 100
    max_temp_steps = int(budget_s / 0.1)  # Rough estimate
    
    T = T0
    iter_count = 0
    
    # Coarse to fine: higher resolution
    target_n_points = min(200, n_points * 2)
    if target_n_points > n_points:
        # Interpolate to higher resolution
        x_orig = np.linspace(0, 1, n_points)
        x_new = np.linspace(0, 1, target_n_points)
        h = np.interp(x_new, x_orig, h)
        n_points = target_n_points
        # Project to feasible
        h = project_feasible(h)
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(best_h)
    
    while T > T_min and iter_count < max_temp_steps:
        # Within current temperature
        for _ in range(steps_per_temp):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_trial = h.copy()
            # Perturb one element at random
            i = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.1 * T)  # Adaptive to temperature
            h_trial[i] += delta
            h_trial = project_feasible(h_trial)
            
            c5_trial = evaluate_erdos_solution(h_trial)
            c5_current = evaluate_erdos_solution(h)
            
            delta_c = c5_trial - c5_current
            
            # Accept with probability
            if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
                h = h_trial
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
        
        T *= alpha
        iter_count += 1
        
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final polishing with gradient-based method
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # SLSQP refinement
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 50}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = best_h

        h_final = project_feasible(h_final)
        final_c5 = evaluate_erdos_solution(h_final)
        
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
