import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create initial construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    # Helper to evaluate objective
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    # Constraint function
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Constraints dict
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution approach with upsampling
    resolutions = [47, 94, 188, 200]
    current_h = h.copy()
    current_n = n_points
    
    for i, target_n in enumerate(resolutions):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
            
        if target_n > current_n:
            # Upsample: repeat elements
            upscale_factor = target_n // current_n
            current_h = np.repeat(current_h, upscale_factor)
            current_h = np.concatenate([current_h, np.zeros(target_n - len(current_h))])
            current_h = current_h[:target_n]  # Ensure correct length
            current_n = target_n
        elif target_n < current_n:
            # Downsample: average groups of elements
            current_h = current_h[:target_n]  # Trim
            current_n = target_n
        
        # Make sure the lengths match
        assert len(current_h) == current_n
        
        # Optimization with L-BFGS-B
        max_iter = min(100, max(50, int((budget_s - elapsed - 20) / 2)))
        try:
            res = minimize(
                obj_func,
                current_h.copy(),
                method='L-BFGS-B',
                bounds=bounds[:current_n],
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-7, 'gtol': 1e-7}
            )
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception:
            pass  # Continue to next resolution
            
        # Random restarts around current best
        for _ in range(5):
            if time.time() - start_time > budget_s - 5:
                break
            h_restart = current_h.copy()
            # Small perturbations
            idxs = np.random.choice(current_n, size=int(current_n*0.1))
            for i in idxs:
                h_restart[i] += np.random.normal(0, 0.05)
            h_restart = np.clip(h_restart, 0, 1)
            current_sum = np.sum(h_restart)
            h_restart = h_restart * target_sum / current_sum
            c5_new = obj_func(h_restart)
            if c5_new < best_c5:
                best_h = h_restart.copy()
                best_c5 = c5_new
        
        # Update current_h
        current_h = best_h.copy()
    
    # Final validation and return
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
