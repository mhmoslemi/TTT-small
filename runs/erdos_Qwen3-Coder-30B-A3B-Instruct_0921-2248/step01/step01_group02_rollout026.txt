import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a simple initialization with some structure
        n_points = 150
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        # Slightly perturb it to avoid local minima
        noise = np.random.normal(0, 0.1, n_points)
        h_init = np.clip(h_init + noise, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Use a combination of optimization approaches
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    last_improvement = 0
    
    # Optimization parameters
    max_iter = 1000
    iter_count = 0
    method = 'SLSQP'
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find maximum (scaled by dx = 2/n_points)
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    # Track current best
    current_h = h_init.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Attempt several random restarts with different strategies
        
        # Strategy 1: Random restart with local search
        h_rand = np.copy(current_h)
        # Add some noise to break symmetry
        noise_level = 0.05 if iter_count < max_iter // 2 else 0.1
        noise = np.random.normal(0, noise_level, n_points)
        h_rand = np.clip(h_rand + noise, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        
        # First try a local optimization
        try:
            res = minimize(objective, h_rand, method=method, bounds=bounds, 
                          constraints=constraints, options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_opt = res.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                # Project to satisfy constraint sum(h) == n_points/2
                h_sum = np.sum(h_opt)
                h_opt = h_opt * (n_points / 2) / h_sum
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < current_c5:
                    current_h = h_opt.copy()
                    current_c5 = c5
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
                        last_improvement = iter_count
        except:
            pass
        
        # Strategy 2: Gradient-free optimization using Nelder-Mead
        try:
            # Perturb current solution slightly
            h_pert = current_h.copy() + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            res = minimize(objective, h_pert, method='Nelder-Mead', 
                          options={'maxiter': 30, 'adaptive': True})
            if res.success:
                h_opt = res.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                h_sum = np.sum(h_opt)
                h_opt = h_opt * (n_points / 2) / h_sum
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < current_c5:
                    current_h = h_opt.copy()
                    current_c5 = c5
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
                        last_improvement = iter_count
        except:
            pass
        
        # Strategy 3: Probability-based perturbation
        try:
            # Create a better guess derived from current
            h_new = current_h / np.max(current_h) * 0.8  #Normalize to 0.8 max
            # Add directional noise to improve search
            noise = np.random.normal(0, 0.05, n_points)
            h_new = np.clip(h_new + noise, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    last_improvement = iter_count
        except:
            pass
            
        iter_count += 1
        # Occasionally restart if too long since improvement
        if iter_count - last_improvement > 50 and iter_count > 100:
            # Repopulate with some structure
            h_new = np.zeros(n_points)
            h_new[:n_points//2] = 1
            h_new = np.clip(h_new + np.random.normal(0, 0.05, n_points), 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            current_h = h_new.copy()
            current_c5 = evaluate_erdos_solution(current_h)
            last_improvement = iter_count
            
        if iter_count % 10 == 0:
            if time.time() - start_time > budget_s * 0.8:
                break
    
    # Final evaluation with the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
