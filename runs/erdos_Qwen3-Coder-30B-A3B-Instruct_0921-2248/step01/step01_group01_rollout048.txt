def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Using full convolution for overlap
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def objective(h_vec):
        return compute_overlap(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with different strategies
    strategies = [
        lambda: np.random.rand(n_points),
        lambda: h.copy(),
        lambda: np.random.permutation(np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]))
    ]
    
    loop_count = 0
    while time.time() - start_time < budget_s - 1:
        loop_count += 1
        
        # Try different strategies
        for strategy in strategies:
            h_trial = strategy()
            h_trial = h_trial * target_sum / np.sum(h_trial)
            
            # Ensure feasibility
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * target_sum / np.sum(h_trial)
            
            try:
                res = minimize(
                    objective,
                    h_trial.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-8}
                )
                
                if res.success:
                    h_opt = res.x.copy()
                    h_opt = np.clip(h_opt, 0, 1)
                    h_opt = h_opt * target_sum / np.sum(h_opt)
                    c5_trial = compute_overlap(h_opt)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_opt.copy()
                        
            except Exception:
                pass
                
            # Small perturbations to prevent local minima entrapment
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed = h_trial + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            c5_perturbed = compute_overlap(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
            
            if time.time() - start_time > budget_s - 1:
                break
                
        # Adaptive refinement - if we're doing well, try increasing resolution
        if loop_count % 10 == 0 and n_points < 150:
            n_points = min(n_points + 10, 150)
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            target_sum = n_points / 2.0
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
