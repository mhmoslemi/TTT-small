import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform segment-based crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    cut_point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
    child2 = np.concatenate([parent2[:cut_point], parent1[cut_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate h by adding Gaussian noise to some elements"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    return h
