import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Helper function to calculate the overlap integral
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Local optimization function using CVXPY
    def local_optimize(h_vec):
        h_var = cp.Variable(n_points)
        # Objective: maximize correlation
        # We minimize -correlation to maximize it
        objective = cp.Minimize(-cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, 0))))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-8, max_iters=1000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h_var.value
        except:
            pass
        return h_vec
    
    # Mutation function to improve diversity
    def mutate(h_vec, mutation_scale):
        noise = np.random.normal(0, mutation_scale, n_points)
        h_mut = h_vec + noise
        h_mut = np.clip(h_mut, 0, 1)
        h_sum = np.sum(h_mut)
        if abs(h_sum - target_sum) > 1e-10:
            h_mut = h_mut * target_sum / h_sum
        return h_mut
    
    # Evolutionary loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Local optimization on current best
        optimized = local_optimize(best_h)
        c5_opt = compute_overlap(optimized)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = optimized.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Mutation to escape local optima
        if no_improvement_count > 20 or iteration % 50 == 0:
            mutation_scale = 0.05 if no_improvement_count > 50 else 0.01
            h_perturbed = mutate(best_h, mutation_scale)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_sum = np.sum(h_perturbed)
            if abs(h_sum - target_sum) > 1e-10:
                h_perturbed = h_perturbed * target_sum / h_sum
            c5_pert = compute_overlap(h_perturbed)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_perturbed.copy()
                no_improvement_count = 0
        
        # Random restart with some probability
        if np.random.rand() < 0.05:
            h_random = np.random.rand(n_points)
            h_random = h_random * target_sum / np.sum(h_random)
            c5_random = compute_overlap(h_random)
            if c5_random < best_c5:
                best_c5 = c5_random
                best_h = h_random.copy()
                no_improvement_count = 0
                
        # Stop if no improvement over many iterations
        if no_improvement_count >= max_no_improvement:
            break
            
        # Ensure we don't use too much time
        if time.time() - start_time > budget_s - 2:
            break
    
    # Final validation
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
