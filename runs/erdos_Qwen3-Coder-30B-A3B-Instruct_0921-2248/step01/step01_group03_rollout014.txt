import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 150
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Define a more global optimization strategy:
    # Alternate between gradient-based and evolutionary local search 
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try random mutation
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = project(h_pert)
        
        # Use scipy's L-BFGS-B which is more stable for this problem
        result = minimize(
            lambda x: evaluate_c5(project(x), dx),
            h_pert,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 50, 'disp': False}
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
        # Randomly sample a new candidate occasionally
        if i % 10 == 0:
            h_sample = np.random.rand(n_points)
            h_sample = project(h_sample)
            c5_sample = evaluate_c5(h_sample, dx)
            if c5_sample < best_c5:
                best_h = h_sample
                best_c5 = c5_sample
    
    return (best_h, best_c5, n_points)
