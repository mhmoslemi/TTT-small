import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise create a reasonable initial guess
        n_points = 200
        # Start with binary mixture and shuffle
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Set up column generation parameters
    max_shifts = 2 * n_points - 1
    initial_k_set = set(np.random.choice(max_shifts, size=min(20, max_shifts), replace=False))
    
    # Keep track of shifts considered
    k_set = initial_k_set.copy()
    
    # Solve the LP with column generation
    iter_count = 0
    max_iter = 500
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Create optimization variables
        h_vars = Variable(n_points)
        c5_var = Variable(1)
        
        # Constraints for current k_set
        constraints = []
        constraints.append(sum(h_vars) == n_points / 2)
        constraints.append(0 <= h_vars)
        constraints.append(h_vars <= 1)
        
        # Add constraints for each k in k_set
        for k in k_set:
            # Create the constraint for shift k
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(c5_var >= overlap_sum * (2.0 / n_points))
        
        # Objective: minimize c5_var
        objective = Minimize(c5_var)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=GLPK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = np.array(h_vars.value).flatten()
                c5_sol = float(c5_var.value)
                
                # Update best solution if better
                if c5_sol < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_sol
                    
                # Find the k with the largest violation
                max_violation = -1
                best_k = -1
                
                for k in range(max_shifts):
                    if k in k_set:
                        continue
                    # Compute overlap for shift k
                    overlap_sum = 0
                    for i in range(n_points):
                        j = (i + k) % n_points
                        overlap_sum += h_sol[i] * (1 - h_sol[j])
                    overlap_scaled = overlap_sum * (2.0 / n_points)
                    violation = overlap_scaled - c5_sol
                    
                    if violation > max_violation:
                        max_violation = violation
                        best_k = k
                        
                # Add the most violative k if improvement is significant
                if best_k >= 0 and max_violation > 1e-5:
                    k_set.add(best_k)
                else:
                    break  # Convergence achieved
            else:
                break  # Infeasible or unbounded
        except:
            break  # Catch any CVXPY errors
            
        iter_count += 1
    
    # Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective for polishing
    def objective_polish(x):
        return evaluate_erdos_solution(x)
    
    # Final optimization
    try:
        h_final = best_h.copy()
        result = minimize(
            objective_polish,
            h_final,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = best_h.copy()
    except:
        h_final = best_h.copy()
    
    # Final projection
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (list(h_final), final_c5, n_points)
