import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Create initial binary construction
        n_points = 100
        h0 = np.zeros(n_points)
        ones_count = n_points // 2
        positions = np.random.permutation(n_points)[:ones_count]
        h0[positions] = 1.0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Tabu search parameters
    tabu_list = []
    max_tabu_size = 20
    max_iter = 10000
    tenure = 20
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    def neighbor_flip(h, i, j):
        h = h.copy()
        h[i], h[j] = h[j], h[i]
        return h
    
    def neighbor_adjust(h, i, delta):
        h = h.copy()
        h[i] = np.clip(h[i] + delta, 0, 1)
        # Adjust another element to maintain sum
        other_idx = np.random.randint(0, n_points)
        while other_idx == i:
            other_idx = np.random.randint(0, n_points)
        h[other_idx] = np.clip(h[other_idx] - delta, 0, 1)
        # Scale to maintain sum
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate by flipping two random positions
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_candidate = neighbor_flip(best_h, idx1, idx2)
        
        # Make sure candidate is feasible
        h_candidate = project(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        
        # Check if candidate is tabu or better than best
        tabu = any(np.array_equal(h_candidate, h_tabu) for h_tabu, _ in tabu_list)
        if not tabu and c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
        
        # Add current solution to tabu list
        tabu_list.append((best_h.copy(), best_c5))
        if len(tabu_list) > max_tabu_size:
            tabu_list.pop(0)
        
        # Occasionally do continuous refinement around best
        if i % 50 == 0:
            bounds = [(0, 1) for _ in range(n_points)]
            con = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
            
            # Local optimization
            result = minimize(
                compute_c5,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=con,
                tol=1e-6,
                options={'maxiter': 10, 'disp': False}
            )
            if result.success:
                h_opt = project(result.x)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
