import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 300
        h0 = np.full(n_points, 0.5)
    
    # Basis functions: cos(m*pi*x/2) for m=0..M
    M = 6
    x = np.linspace(0, 2, n_points, endpoint=False)
    basis = np.array([np.cos(m * np.pi * x / 2) for m in range(M+1)])
    
    # Initial coefficients (project h0 onto basis)
    coeffs = np.linalg.lstsq(basis.T, h0, rcond=None)[0]
    
    # Precompute for efficiency
    def h_from_coeffs(c):
        h = basis.T @ c
        # Ensure h is in [0,1] and integral is 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        return h
    
    def obj_and_grad(c):
        if time.time() - start_time > budget_s - 1:
            raise KeyboardInterrupt
        h = h_from_coeffs(c)
        corr = np.correlate(h, 1-h, mode="full")
        max_corr = np.max(corr)
        # Gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(c)
        for i in range(len(c)):
            c_plus = c.copy()
            c_plus[i] += eps
            h_plus = h_from_coeffs(c_plus)
            corr_plus = np.correlate(h_plus, 1-h_plus, mode="full")
            max_plus = np.max(corr_plus)
            grad[i] = (max_plus - max_corr) / eps
        return max_corr, grad
    
    # Optimize
    best_c = coeffs.copy()
    best_c5 = evaluate_erdos_solution(h_from_coeffs(best_c))
    
    try:
        for iter in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
            # L-BFGS-B optimization
            try:
                c_new, c5_new, info = fmin_l_bfgs_b(
                    func=obj_and_grad,
                    x0=best_c,
                    maxiter=20,
                    disp=0,
                    bounds=[(None,None) for _ in range(len(best_c))]
                )
                if not np.isnan(c5_new) and c5_new < best_c5:
                    best_c = c_new.copy()
                    best_c5 = c5_new
            except:
                break
    except KeyboardInterrupt:
        pass
    
    # Final refinement
    best_c5 = evaluate_erdos_solution(h_from_coeffs(best_c))
    
    final_coeffs = best_c
    final_h = h_from_coeffs(final_coeffs)
    
    return (list(final_h), best_c5, n_points)
