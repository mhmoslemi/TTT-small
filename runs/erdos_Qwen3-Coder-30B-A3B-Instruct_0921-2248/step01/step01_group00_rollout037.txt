import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value of the correlation (scaled by dx)
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create one
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Create a binary vector and shuffle it
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1
        np.random.shuffle(h0)
        
    dx = 2.0 / n_points

    # Initialize with h0 and keep track of best
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define the optimization parameters
    max_iter = 500
    lr = 0.02
    decay = 0.99
    patience = 10
    wait = 0
    
    def obj_func(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Adjust to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    # Gradient-based optimization with adaptive step size
    best_h = h0.copy()
    current_c5 = obj_func(best_h)
    best_c5 = current_c5
    prev_c5 = current_c5
    step_size = lr
    min_grad = 1e-6
    
    # SLSQP optimization as fallback
    from scipy.optimize import minimize
    
    def constraint_eq(x):
        return np.sum(x) - n_points / 2

    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    opt_options = {'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive step size based on progress
        if current_c5 < prev_c5:
            step_size *= decay
        else:
            step_size *= 0.999  # Slow decay to avoid oscillation
            
        # Gradient estimation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            f_plus = obj_func(h_plus)
            
            h_minus = best_h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            f_minus = obj_func(h_minus)
            
            grad[j] = (f_plus - f_minus) / (2 * eps)
            
        # Update with gradient descent
        new_h = best_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = obj_func(new_h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            best_h = new_h.copy()
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                wait = 0
            else:
                wait += 1
        else:
            wait += 1
            
        # Reset step size if stuck
        if wait > patience:
            step_size = lr
            wait = 0
            
        # Occasionally, try SLSQP for refinement 
        if i % 30 == 0 and i > 0:
            try:
                h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                res = minimize(obj_func, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options=opt_options)
                if res.success and res.fun < best_c5:
                    h_opt = np.clip(res.x, 0, 1)
                    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                    c5_val = obj_func(h_opt)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_opt.copy()
            except Exception:
                pass

    # Final evaluation 
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
