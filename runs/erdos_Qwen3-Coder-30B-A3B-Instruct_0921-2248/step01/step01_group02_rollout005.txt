import numpy as np
from scipy.interpolate import interp1d
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        n_points = len(h_parent)
        # Upsample to target resolution
        x_parent = np.linspace(0, 2, n_points)
        x_target = np.linspace(0, 2, 100)
        f = interp1d(x_parent, h_parent, kind='linear', fill_value='extrapolate')
        h_init = f(x_target)
        n_points = len(h_init)
    else:
        n_points = 100
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
    
    # Initialize population
    def create_individual():
        # Start with parent
        h = h_init.copy()
        # Add some random variations while preserving sum
        pert = np.random.normal(0, 0.05, n_points)
        h += pert
        h = np.clip(h, 0, 1)
        # Normalize sum
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def create_population(size):
        return [create_individual() for _ in range(size)]
    
    # Genetic operators
    def crossover(parent1, parent2):
        # Blend crossover
        alpha = 0.5
        child1 = alpha * parent1 + (1 - alpha) * parent2
        child2 = (1 - alpha) * parent1 + alpha * parent2
        # Repair children
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)
        # Normalize sums
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    def mutate(individual, rate=0.1):
        # Random reset mutation
        mask = np.random.random(n_points) < rate
        individual[mask] = np.random.random(np.sum(mask))
        # Gaussian perturbation
        pert = np.random.normal(0, 0.05, n_points)
        individual += pert
        # Clip
        individual = np.clip(individual, 0, 1)
        # Normalize
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    def fitness(h):
        return evaluate_erdos_solution(h)
    
    # Parameters
    pop_size = 60
    max_generations = 1000
    num_offspring = 40
    keep_parents = 20
    
    # Create population
    population = create_population(pop_size)
    population_fitness = [fitness(ind) for ind in population]
    
    # Track best solution
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]
    
    # Main loop
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Sort population
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best
        if population_fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = population_fitness[0]
            last_improvement = generation
        
        # Select parents for breeding
        parents = population[:pop_size // 3]
        
        # Generate offspring
        offspring = []
        for _ in range(num_offspring):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child1, child2 = crossover(p1, p2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            offspring.extend([child1, child2])
        
        # Keep some parents
        survivors = population[:keep_parents]
        
        # Evaluate offspring
        offspring_fitness = []
        for child in offspring:
            try:
                f = fitness(child)
                offspring_fitness.append(f)
            except:
                # If evaluation fails, use a simple proxy
                f = np.inf
                offspring_fitness.append(f)
        
        # Merge and select new population
        all_individuals = survivors + offspring
        all_fitness = [fitness(ind) if i >= len(survivors) else population_fitness[i] 
                       for i, ind in enumerate(all_individuals)]
        
        # Sort and keep best half
        sorted_indices = np.argsort(all_fitness)
        new_pop = [all_individuals[i] for i in sorted_indices[:pop_size]]
        new_fitness = [all_fitness[i] for i in sorted_indices[:pop_size]]
        
        population = new_pop
        population_fitness = new_fitness
        
        generation += 1
        
        # Early stopping if no improvement in 100 generations
        if generation - last_improvement > 100:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
