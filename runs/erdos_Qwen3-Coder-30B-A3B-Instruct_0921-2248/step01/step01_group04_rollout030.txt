import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a reasonable initial guess with 50 points, two clusters
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simulated Annealing approach
    # Initialize parameters
    temp = 0.1
    cooling_rate = 0.999
    min_temp = 1e-6
    steps_per_temp = 100
    
    # Coarse-to-fine approach
    if n_points < 100:
        n_points_target = 100
        # Interpolate to target resolution
        h_new = np.interp(np.linspace(0, 1, n_points_target), 
                          np.linspace(0, 1, n_points), h)
        h = h_new
        n_points = n_points_target
    
    if n_points < 200:
        n_points_target = 200
        h_new = np.interp(np.linspace(0, 1, n_points_target), 
                          np.linspace(0, 1, n_points), h)
        h = h_new
        n_points = n_points_target
    
    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Annealing loop
    current_h = h.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    
    # Keep track of temperature schedule and improvement
    recent_improvements = []
    last_improvement_time = time.time()
    
    while time.time() - start_time < budget_s - 5:
        # Start with current solution
        h_current = current_h.copy()
        c5_current = current_c5
        
        # Perform steps at current temperature
        for _ in range(steps_per_temp):
            # Make a copy to modify
            h_new = h_current.copy()
            
            # Choose a random index and perturb
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.05)
            h_new[idx] += delta
            
            # Project back into feasible region [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to satisfy sum constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            # Compute new cost
            c5_new = evaluate_erdos_solution(h_new)
            
            # Determine whether to accept change
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                recent_improvements.append(c5_current)
            else:
                # Accept with probability based on temperature
                delta = c5_new - c5_current
                if np.random.rand() < np.exp(-delta / temp):
                    h_current = h_new.copy()
                    c5_current = c5_new
                
            # Update best solution found so far
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        
        # Update temperature
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        # Check for improvement stagnation
        recent_improvements = recent_improvements[-20:]
        if len(recent_improvements) >= 20 and np.std(recent_improvements) < 1e-6:
            # Stagnated, cool faster
            cooling_rate = 0.9999
            
        # If current is better than best, update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Exit early if we've plateaued significantly
        if time.time() - last_improvement_time > 30 and best_c5 < 0.381:
            break
    
    # Final polishing with SLSQP
    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Try optimizing to improve further
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if result.success:
            final_h = result.x
        else:
            final_h = best_h
            
        # Ensure feasibility of final result
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2.0) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h)
        
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
