from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

    lambda_ = 0.0

    target_sum = n_points / 2

        h = np.clip(h, 0, 1)

        # Compute the maximum correlation
        correlation = np.correlate(h, 1 - h, mode="full")

        return max_corr * dx

    h = best_h.copy()

        def objective(h_vec):
            h_test = h_vec.copy()
            h_test = project_h(h_test)
