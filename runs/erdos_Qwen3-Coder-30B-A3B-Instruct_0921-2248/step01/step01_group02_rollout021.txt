import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    
    # Set up population parameters
    pop_size = 50
    n_points = 100
    max_generations = 500
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 3
    
    # Initialize population
    population = []
    if initial_h is not None:
        # Interpolate the initial h to our desired size
        x_orig = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        f = interp1d(x_orig, initial_h, kind='linear', fill_value='extrapolate')
        h_interp = np.clip(f(x_new), 0, 1)
        # Normalize to sum = n_points/2
        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        population.append(h_interp.copy())
        
        # Add some variations
        for i in range(pop_size - 1):
            h = h_interp.copy()
            # Add small random changes
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            # Clip values
            h = np.clip(h, 0, 1)
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Create random population
        for i in range(pop_size):
            h = np.random.rand(n_points)
            # Normalize to sum = n_points/2
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluation of initial population
    fitness = [evaluate_erdos_solution(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    generation = 0
    stagnation_count = 0
    last_best_c5 = best_c5
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep the best
        new_population.append(best_h.copy())
        
        # Generate offspring
        for i in range(0, pop_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                # Random reset mutation
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] = np.random.rand()
                
                # Gaussian perturbation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                
                # Clip to [0,1]
                child = np.clip(child, 0, 1)
                
                # Normalize to sum = n_points/2
                child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
            
            if len(new_population) >= pop_size:
                break
        
        # Ensure we have pop_size individuals
        while len(new_population) < pop_size:
            # Fill with random individuals
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            new_population.append(h)
        
        # Evaluate new population
        new_fitness = [evaluate_erdos_solution(h) for h in new_population]
        
        # Replace worst with new population
        worst_indices = np.argsort(fitness)[-len(new_population):]
        for i, idx in enumerate(worst_indices[:len(new_population)]):
            population[idx] = new_population[i].copy()
            fitness[idx] = new_fitness[i]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive mutation rate
        if stagnation_count > 10 and mutation_rate < 0.3:
            mutation_rate = min(0.3, mutation_rate * 1.1)
            
        # Early stopping if improvement is too slow
        if time.time() - start_time > budget_s * 0.8 and stagnation_count > 20:
            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
