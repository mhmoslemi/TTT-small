import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Shuffled binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Upsample to higher resolution
    n_upsampled = 500
    h_upsampled = np.interp(np.linspace(0, 2, n_upsampled), 
                           np.linspace(0, 2, n_points), h)
    
    dx = 2.0 / n_upsampled
    target_sum = n_upsampled / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_upsampled
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h_upsampled.copy()
    best_c5 = obj_func(h_upsampled)
    
    # Try different restart strategies
    for iteration in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Use Nelder-Mead optimization
            res = minimize(obj_func, h_upsampled.copy(), method='Nelder-Mead',
                          options={'maxiter': 200, 'initial_simplex': 0.1,
                                  'disp': False, 'adaptive': True})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                current_sum = np.sum(h_new)
                if current_sum != 0:
                    h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            
            # Random restart with some perturbation
            h_upsampled = best_h.copy()
            perturbation = np.random.normal(0, 0.02, n_upsampled)
            h_upsampled += perturbation
            h_upsampled = np.clip(h_upsampled, 0, 1)
            current_sum = np.sum(h_upsampled)
            h_upsampled = h_upsampled * target_sum / current_sum
            
        except Exception as e:
            continue
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), n_upsampled)
