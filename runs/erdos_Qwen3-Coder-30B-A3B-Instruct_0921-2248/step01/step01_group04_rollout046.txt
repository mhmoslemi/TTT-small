import numpy as np
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Create a reasonable initial guess
        n_points = 200
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Coarse-to-fine approach
    # Start at lower resolution, gradually increase
    resolutions = [50, 100, 200]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create initial solution at this resolution
        if res == 50:
            h_coarse = np.zeros(res)
            half_coarse = res // 2
            h_coarse[:half_coarse] = 1
            # Interpolate to target resolution
            h = np.interp(np.linspace(0, 1, n_points), 
                         np.linspace(0, 1, res), h_coarse)
        elif res == 100:
            # Initialize based on current best to preserve progress
            h = best_h.copy()
            
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Simulated Annealing at this resolution
        T0 = 0.1
        alpha = 0.999
        T = T0
        min_T = 1e-6
        max_iter = 1000  # changes per temperature level
        delta = 0.1  # initial perturbation
        
        current_h = h.copy()
        current_c5 = evaluate_erdos_solution(current_h)
        
        # Track best
        temp_best_h = current_h.copy()
        temp_best_c5 = current_c5
        
        iter_count = 0
        for t in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb one element
            i = np.random.randint(0, n_points)
            # Perturb amount
            pert = np.random.uniform(-delta, delta)
            h_new = current_h.copy()
            h_new[i] += pert
            
            # Project back to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to satisfy sum constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept or reject based on acceptance probability
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                current_h = h_new.copy()
                current_c5 = c5_new
                
            # Update best
            if current_c5 < temp_best_c5:
                temp_best_h = current_h.copy()
                temp_best_c5 = current_c5
                
            # Cool temperature
            T = max(min_T, T * alpha)
            
            iter_count += 1
            
            # Occasionally re-scale delta to avoid stagnation
            if t % 100 == 0:
                delta *= 0.99
            
        # Update global best with this resolution's result
        if temp_best_c5 < best_c5:
            best_h = temp_best_h.copy()
            best_c5 = temp_best_c5
        
        # Final polishing step: SLSQP to refine
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Simple version of search
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            current_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = current_h
            
        # Final projection
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2.0) / np.sum(h_final)
        
        final_c5 = evaluate_erdos_solution(h_final)
        
        if final_c5 < best_c5:
            best_h = h_final.copy()
            best_c5 = final_c5
            
        h = best_h.copy()  # prepare for next resolution
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
