import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a new initial guess with 50 points
        n_points = 50
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        np.random.shuffle(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Coarse-to-fine strategy
    resolutions = [50, 100, 200]
    for res in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        if res > n_points:
            # Interpolate to target resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            f = interp1d(x_orig, h, kind='linear', fill_value='extrapolate')
            h = f(x_new)
            n_points = res
            
        # Simulated Annealing parameters
        temp = 0.1
        cooling_rate = 0.999
        min_temp = 1e-8
        iter_per_temp = 100
        perturb_factor = 0.1
        
        for _ in range(1000):  # Limit iterations to avoid hanging
            if time.time() - start_time > budget_s - 1:
                break
                
            # Make a copy
            h_current = h.copy()
            
            # Try multiple perturbations at current temperature
            for _ in range(iter_per_temp):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Select random index
                i = np.random.randint(0, n_points)
                
                # Make perturbation
                delta = np.random.uniform(-perturb_factor, perturb_factor)
                h_new = h_current.copy()
                h_new[i] += delta
                
                # Clamp to [0,1]
                h_new[i] = np.clip(h_new[i], 0, 1)
                
                # Rescale to maintain sum constraint
                scale_factor = (n_points / 2.0) / np.sum(h_new)
                h_new = h_new * scale_factor
                
                # Evaluate
                c5_new = evaluate_erdos_solution(h_new)
                
                # Compute cost difference
                c5_old = evaluate_erdos_solution(h_current)
                delta_c = c5_new - c5_old
                
                # Accept based on simulated annealing probability
                if delta_c < 0 or np.exp(-delta_c / temp) > np.random.rand():
                    h = h_new.copy()
                    if c5_new < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_new
                
            # Cool down
            temp *= cooling_rate
            if temp < min_temp:
                temp = min_temp
        
        # Early polishing step using SLSQP
        if res == resolutions[-1]:  # Only polish at final resolution
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(x):
                return np.sum(x) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            from scipy.optimize import minimize
            
            result = minimize(
                lambda x: evaluate_erdos_solution(x),
                best_h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 100}
            )
            
            if result.success:
                best_h = result.x.copy()
                best_c5 = evaluate_erdos_solution(best_h)
        
        # Check if time's almost up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
