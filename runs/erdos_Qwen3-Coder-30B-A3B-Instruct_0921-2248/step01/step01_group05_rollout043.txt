import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default construction
        n_points = 100
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        np.random.shuffle(h_init)
    
    # Ensure the initial vector satisfies the sum constraint
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Scale and clip to [0,1] if needed
    h_init = np.clip(h_init, 0, 1)
    
    # CVX optimization approach
    # Variable: h vector of size n_points
    h = cp.Variable(n_points, name='h')
    C5 = cp.Variable(name='C5')
    
    # Objective: minimize C5
    objective = cp.Minimize(C5)
    
    # Constraints
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1,
    ]
    
    # Add overlap constraints for k = 0 to n_points-1
    # For each k, we compute the overlap for shifts
    dx = 2.0 / n_points
    
    # Create correlation constraint for each k
    for k in range(n_points):
        # Compute shift correlation
        corr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            corr += h[i] * (1 - h[j])
        constraints.append(corr <= C5 * (n_points / 2.0))
    
    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with MOSEK backend
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s * 0.8)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = C5.value * dx
        else:
            # Fallback to initial h if optimization fails
            h_opt = h_init
            c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    except Exception as e:
        # Fallback with original heuristic if CVXPY fails
        h_opt = h_init
        c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    
    # Ensure final solution is within bounds
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * target_sum / np.sum(h_opt)
    
    # Final verification
    final_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    return (h_opt.tolist(), float(final_c5), n_points)
