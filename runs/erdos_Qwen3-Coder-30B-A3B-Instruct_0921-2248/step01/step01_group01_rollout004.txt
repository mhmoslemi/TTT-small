import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default: create a structured initialization
        n_points = 50
        h = np.zeros(n_points)
        # Place 1s at even indices to create a step function
        h[::2] = 1.0
        # Normalize sum to target
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization sequence
    res_levels = [50, 100, 150]
    for res in res_levels:
        if time.time() - start_time > budget_s - 1:
            break
            
        if res > n_points:
            # Interpolate from previous resolution
            old_h = h.copy()
            h = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), old_h)
            n_points = res
            dx = 2.0 / n_points
        
        # Multiple restarts
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb current best
            h_pert = h + np.random.normal(0, 0.05, n_points) * (best_c5 if best_c5 < 1 else 0.1)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert_sum = np.sum(h_pert)
            h_pert = h_pert * target_sum / h_pert_sum
            
            try:
                res_opt = minimize(
                    objective,
                    h_pert.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6}
                )
                
                if res_opt.success:
                    h_final = res_opt.x.copy()
                    c5 = objective(h_final)
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_final.copy()
                        
                    # Perturb to escape local minima
                    h = h_final + np.random.normal(0, 0.025, n_points) * (c5 if c5 < 1 else 0.1)
                else:
                    # Random restart
                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)
            except:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
                
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
