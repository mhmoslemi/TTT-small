import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value and scale
    max_corr = np.max(corr)
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Start with a reasonable construction
        n_points = 200
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Generate all possible shift indices
    shift_indices = list(range(2 * n_points - 1))
    
    # Use column generation approach
    # Start with a small set of shifts
    K = np.random.choice(shift_indices, size=min(50, len(shift_indices)), replace=False)
    K = sorted(K)
    
    # Track time
    time_limit = budget_s - 5  # Leave buffer
    
    # Column generation loop
    iteration = 0
    while time.time() - start_time < time_limit and iteration < 1000:
        iteration += 1
        
        if len(K) >= len(shift_indices):
            break
            
        # Create CVXPY variables
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Constraints
        constraints = [0 <= h_vars, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Add overlap constraints for current shifts
        for k in K:
            # Compute the overlap for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(c5_var >= overlap_expr * (2.0 / n_points))
        
        # Objective
        objective = cp.Minimize(c5_var)
        
        # Solve LP
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            h_sol = h_vars.value
            c5_sol = c5_var.value
        except:
            # If LP fails, fallback to a smaller K
            K = K[:len(K)//2] if len(K) > 10 else [0]
            continue
            
        # Check which shift k not in K has the highest overlap
        max_overlap = -1
        best_k = -1
        
        # Check a sample of all shifts (randomized to avoid bias)
        check_shifts = np.random.choice(shift_indices, size=min(100, len(shift_indices)), replace=False)
        
        for k in check_shifts:
            if k in K:
                continue
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_sol[i] * (1 - h_sol[j])
            overlap_scaled = overlap * (2.0 / n_points)
            if overlap_scaled > max_overlap:
                max_overlap = overlap_scaled
                best_k = k
        
        # If we find a better k, add it to K
        if max_overlap > c5_sol * 1.001:  # Small margin to avoid numerical issues
            K.append(best_k)
            K = sorted(K)
        else:
            # Check if we can do better with a few more samples
            break
    
    # Final refinement with SLSQP
    # Reset h to best known with polishing if possible
    h_final = best_h.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization using SLSQP with multiple restarts
    def obj_func(x):
        return evaluate_erdos_solution(x)
    
    # Try multiple random starts to escape local minima
    for _ in range(10):
        if time.time() - start_time > time_limit:
            break
            
        # Random unless we're near a good solution
        if np.random.rand() < 0.3 or len(K) < 50:
            h_trial = np.random.uniform(0, 1, n_points)
        else:
            h_trial = h_final.copy()
            
        # Project to feasible set
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)
        
        result = minimize(obj_func, h_trial, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'ftol': 1e-8, 'maxiter': 100})
        
        if result.success:
            h_new = result.x
        else:
            h_new = h_trial
            
        # Make sure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Update the starting point for next iteration
            h_final = h_new.copy()
    
    # Final projection to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
