import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        
    # Upsample to 500 points to improve resolution
    h_upsampled = np.interp(np.linspace(0, 2, 500), np.linspace(0, 2, n_points), h)
    h = h_upsampled
    n_points = 500
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        # Make sure it's in the valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Try multiple restarts with different methods
    for iter_count in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Start with a random perturbation of the current best
            h_start = best_h + np.random.normal(0, 0.01, n_points)
            h_start = np.clip(h_start, 0, 1)
            # Project back to sum constraint
            h_start = h_start * target_sum / np.sum(h_start)
            
            # First optimize with Nelder-Mead to find local minima
            res = minimize(obj_func, h_start, method='Nelder-Mead', 
                           options={'maxiter': 200, 'adaptive': True})
            
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint with projection
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except Exception as e:
            pass
            
        # Occasionally perturb current best to escape local minima
        if iter_count % 20 == 0:
            h_perturbed = best_h + np.random.normal(0, 0.005, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            c5_perturbed = obj_func(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the result
    return (best_h.tolist(), float(final_c5), n_points)
