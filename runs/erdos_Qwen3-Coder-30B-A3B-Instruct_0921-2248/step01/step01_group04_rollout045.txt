import numpy as np
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initial guess with 50 points
        n_points = 50
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    def project_feasible(x):
        """Ensures x is in [0,1] and sum(x) = n_points/2"""
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2.0) / np.sum(x)
        return x
    
    # Coarse to fine search with simulated annealing
    resolutions = [50, 100, 200]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h = np.interp(np.linspace(0, 1, res), np.linspace(0, 1, n_points), h)
            n_points = res
            
        # Parameters for SA
        T0 = 0.1
        alpha = 0.999
        T = T0
        max_iter_per_temp = 100
        min_T = 1e-6
        flat_line_count = 0
        flat_line_threshold = 10
        
        # Temperature schedule search
        for t in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            
            if T < min_T:
                break
                
            # Sample and perturb
            for _ in range(max_iter_per_temp):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                h_trial = h.copy()
                i = np.random.randint(0, len(h_trial))
                delta = np.random.normal(0, 0.1 * T)
                h_trial[i] += delta
                h_trial = project_feasible(h_trial)
                
                # Evaluate
                c5_trial = evaluate_erdos_solution(h_trial)
                
                # Accept or reject
                delta_c5 = c5_trial - best_c5
                if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                    h = h_trial
                    if c5_trial < best_c5:
                        best_h = h_trial.copy()
                        best_c5 = c5_trial
                        
            # Cool
            T *= alpha
            
            # Early stopping if no improvement for a while
            if T < 0.01 and flat_line_count > flat_line_threshold:
                break
            flat_line_count += 1

    # Final polishing with gradient-based method
    def cost_func(x):
        return evaluate_erdos_solution(x)
        
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use SLSQP for polishing
    result = minimize(
        lambda x: cost_func(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    if result.success:
        final_h = result.x
        final_h = project_feasible(final_h)
        final_c5 = cost_func(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    return (list(best_h), best_c5, n_points)
