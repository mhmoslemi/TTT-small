import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Shuffled binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        # Make sure it's in the valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization: upsample progressively
    upscale_factors = [2, 2, 12]
    current_h = h.copy()
    current_n = n_points
    
    for i, upscale in enumerate(upscale_factors):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Upsample
        new_n = current_n * upscale
        new_h = np.repeat(current_h, upscale)
        
        # Add noise
        noise_magnitude = 0.1 * (1.0 - (time.time() - start_time) / budget_s)
        noise = np.random.uniform(-noise_magnitude, noise_magnitude, new_n)
        new_h = new_h + noise
        
        # Project to feasible region
        current_sum = np.sum(new_h)
        new_h = new_h * target_sum / current_sum
        new_h = np.clip(new_h, 0, 1)
        
        # Optimization using LBFGS-B
        try:
            res = minimize(obj_func, new_h.copy(), method='L-BFGS-B', bounds=[(0.0, 1.0)]*new_n, constraints=cons, 
                           options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-6, 'maxls': 20})
            
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint with projection
                current_sum = np.sum(h_new)
                h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception as e:
            pass
        
        # Random restart
        if i < len(upscale_factors) - 1:
            current_h = np.random.rand(new_n)
            current_h = current_h * target_sum / np.sum(current_h)
        else:
            current_h = new_h
        
        current_n = new_n
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
