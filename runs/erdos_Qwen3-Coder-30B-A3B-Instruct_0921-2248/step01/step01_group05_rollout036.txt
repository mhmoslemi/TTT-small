import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function should compute the C5 value for the given h
    # Using the provided evaluation mechanism
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: Simple binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        # Make sure it's in the valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-resolution optimization with upsampling
    upscale_factors = [2, 2, 12]  # 47 -> 94 -> 188 -> 200
    upsampled_points = [47, 94, 188, 200]
    
    # Consider the upsampling and optimization strategy
    for idx, n_up in enumerate(upsampled_points):
        # Prepare upsampling if needed
        if n_up > n_points:
            # Up-sample: repeat every element
            h_up = np.repeat(h, n_up // n_points)
            # Add slight uniform noise to break symmetry
            noise_magnitude = 0.05
            noise = np.random.uniform(-noise_magnitude, noise_magnitude, size=n_up)
            h_up = h_up + noise
            h = h_up
            n_points = n_up
            dx = 2.0 / n_points
        else:
            h = h[:n_up]
        
        # Random restart with probability 0.1 for first two stages
        if idx < 2:
            if np.random.rand() < 0.1:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
        
        # Optimization with LBFGS-B for the current resolution
        bounds = [(0.0, 1.0)] * n_points
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj_func, h.copy(), method='L-BFGS-B', bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9})
            
            if res.success or res.fun < best_c5:
                h_new = np.clip(res.x, 0, 1)
                # Enforce sum constraint with projection
                current_sum = np.sum(h_new)
                if current_sum > 0:
                    h_new = h_new * target_sum / current_sum
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception as e:
            pass
        
        h = np.random.rand(n_points)  # Unconditional random restart
        h = h * target_sum / np.sum(h)
        
        # Check time left
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    best_h = best_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
