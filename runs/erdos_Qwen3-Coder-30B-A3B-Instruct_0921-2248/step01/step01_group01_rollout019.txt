def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try different resolutions and restarts
    resolutions = [47, 94, 141]  # Very rough scaling to try multiple points
    restarts = 5
    
    # Different initialization patterns
    init_patterns = [
        lambda n: np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)]),  # alternating
        lambda n: np.array([1.0 if i < n//2 else 0.0 for i in range(n)]),    # half 1s
        lambda n: np.random.rand(n),                                         # random
    ]
    
    for n in resolutions:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Ensure we have at least one restart
        attempts = max(1, restarts // len(resolutions))
        for _ in range(attempts):
            # Try different initializations
            for init_func in init_patterns:
                h = init_func(n)
                h_sum = np.sum(h)
                if abs(h_sum - target_sum) > 1e-10:
                    h = h * target_sum / h_sum
                
                # Optimize
                try:
                    if n == 47:  # Coarse resolution
                        res = minimize(
                            objective,
                            h.copy(),
                            method='L-BFGS-B',
                            bounds=bounds,
                            constraints=constraints,
                            options={'maxiter': 200, 'ftol': 1e-4}
                        )
                    elif n == 94:  # Medium resolution
                        res = minimize(
                            objective,
                            h.copy(),
                            method='L-BFGS-B',
                            bounds=bounds,
                            constraints=constraints,
                            options={'maxiter': 300, 'ftol': 1e-5}
                        )
                    else:  # Fine resolution
                        res = minimize(
                            objective,
                            h.copy(),
                            method='L-BFGS-B',
                            bounds=bounds,
                            constraints=constraints,
                            options={'maxiter': 500, 'ftol': 1e-6}
                        )

                    if res.success:
                        h_opt = res.x.copy()
                        h_opt = np.clip(h_opt, 0, 1)
                        h_sum = np.sum(h_opt)
                        if abs(h_sum - target_sum) > 1e-10:
                            h_opt = h_opt * target_sum / h_sum
                        c5 = objective(h_opt)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_opt.copy()
                            
                        # Add variability to avoid being stuck
                        noise_level = min(0.05, 0.5 * best_c5)
                        noise = np.random.normal(0, noise_level, n)
                        h = h_opt + noise
                    else:
                        # If optimization fails, try random search
                        h = np.random.rand(n)
                        h = h * target_sum / np.sum(h)
                        
                except Exception as e:
                    # If optimization fails, try random search
                    h = np.random.rand(n)
                    h = h * target_sum / np.sum(h)
                
                # Project back to feasible region
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - target_sum) > 1e-10:
                    h = h * target_sum / h_sum
                    
                # Check time
                if time.time() - start_time > budget_s - 2:
                    break
                    
            # Rescale solution to maintain feasibility
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
                
            # Perturb slightly to escape local minima
            noise_level = min(0.02, 0.1 * best_c5) if best_c5 < float('inf') else 0.02
            noise = np.random.normal(0, noise_level, n)
            h = h + noise
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    # Compute final C5
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
