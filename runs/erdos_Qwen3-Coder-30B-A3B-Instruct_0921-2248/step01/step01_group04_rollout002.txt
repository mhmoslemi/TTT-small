import numpy as np
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initialize h from existing construction if available

        n_points = len(h)
    else:

        n_points = 50
        h = np.zeros(n_points)
        h[:25] = 1.0
        # Interpolate to higher resolution
        n_high = 200
        x_low = np.linspace(0, 1, n_points)
        x_high = np.linspace(0, 1, n_high)
        h = np.interp(x_high, x_low, h)
        h = np.clip(h, 0, 1)
        n_points = n_high

    # Ensure the integral constraint
    h = h * (n_points / 2.0) / np.sum(h)

    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Annealing parameters
    T0 = 0.1
    alpha = 0.999
    min_T = 1e-6
    temp_steps = 1000
    perturb_steps = 100

    # Simulated Annealing
    T = T0
    last_improvement = 0
    patience = 50

    for step in range(temp_steps):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb h multiple times at current temperature
        for _ in range(perturb_steps):
            # Create new solution by perturbing one element
            h_new = h.copy()
            i = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.1 * T)
            h_new[i] += delta

            # Clip to bounds
            h_new = np.clip(h_new, 0, 1)

            # Rescale to satisfy constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)

            # Compute change in cost
            c5_new = evaluate_erdos_solution(h_new)
            c5_old = evaluate_erdos_solution(h)

            delta_c = c5_new - c5_old

            # Accept with probability based on temperature
            if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
                h = h_new.copy()
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    last_improvement = step

        # Cool down
        T = max(T * alpha, min_T)

        # Early stopping if no improvement
        if step - last_improvement > patience:
            break

    # Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    cons = {'type': 'eq', 'fun': constraint_sum}

    h_final = best_h.copy()
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:

            lambda x: evaluate_erdos_solution(x),
            h_final,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 20}
        )

        if result.success:
            h_final = result.x
        else:
            break

        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2.0) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final)

        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
