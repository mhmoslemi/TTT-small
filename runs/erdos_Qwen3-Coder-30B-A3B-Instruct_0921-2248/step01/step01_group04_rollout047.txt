import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Get maximum value
    max_corr = np.max(corr)
    # Return scaled value
    return max_corr * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a coarse initial guess and upsample
        n_coarse = 50
        h_coarse = np.zeros(n_coarse)
        h_coarse[:n_coarse//2] = 1
        np.random.shuffle(h_coarse)
        
        # Upsample to target resolution
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_coarse), h_coarse)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        n_points = len(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Coarse-to-fine annealing with simulated annealing
    # Initialize temperature and parameters
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    n_steps_per_T = 100
    
    for t in range(1000):  # limit iterations to prevent long runs
        if time.time() - start_time > budget_s - 1:
            break
            
        # For each temperature
        for _ in range(n_steps_per_T):
            # Make a copy of current solution
            h_new = h.copy()
            
            # Perform random perturbation
            i = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.1 * T)  # adaptive perturbation size
            h_new[i] += delta
            h_new[i] = np.clip(h_new[i], 0, 1)
            
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept or reject
            delta_c = c5_new - best_c5
            if delta_c < 0 or np.exp(-delta_c / T) > np.random.rand():
                h = h_new.copy()
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            
        # Cool temperature
        T = max(min_T, T * alpha)
    
    # Final refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        lambda x: evaluate_erdos_solution(x),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2.0) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h)
    else:
        final_h = best_h
        final_c5 = best_c5
    
    return (list(final_h), final_c5, n_points)
