import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Upsample to 500 points for better resolution
    n_upsampled = 500
    h_upsampled = np.interp(np.linspace(0, 2, n_upsampled), 
                           np.linspace(0, 2, n_points), h)
    
    # dx for upsampling
    dx = 2.0 / n_upsampled
    target_sum = n_upsampled / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_upsampled
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h_upsampled.copy()
    best_c5 = obj_func(h_upsampled)
    
    # Iterate with randomized restarts
    for iteration in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Start from current best solution with small random perturbation
            h_start = best_h + np.random.normal(0, 0.01, n_upsampled)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * target_sum / np.sum(h_start)
            
            # Use Nelder-Mead for optimization
            res = minimize(obj_func, h_start, method='Nelder-Mead', 
                          options={'maxiter': 200, 'adaptive': True})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                c5_new = obj_func(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
            # Random restart with some perturbation
            h_restart = np.random.rand(n_upsampled)
            h_restart = h_restart * target_sum / np.sum(h_restart)
            c5_restart = obj_func(h_restart)
            if c5_restart < best_c5:
                best_h = h_restart.copy()
                best_c5 = c5_restart
                
        except Exception as e:
            continue
            
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), n_upsampled)
