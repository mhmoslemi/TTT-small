def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with 1s at even indices
        n_points = 50
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-resolution optimization
    resolutions = [50, 100, 150]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Interpolate or resize h to new resolution if needed
        if res != n_points:
            if res > n_points:
                # Upsample
                new_h = np.interp(np.linspace(0, 2, res), 
                                  np.linspace(0, 2, n_points), h)
            else:
                # Downsample
                new_h = np.array([np.mean(h[i::(n_points//res)]) 
                                 for i in range(n_points//res)])
            h = new_h
            n_points = res
        
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        
        # Optimize at this resolution
        max_iter = 250
        ftol = 1e-6 if res == 150 else 1e-4  # tighter for final resolution
        
        # Multiple restarts for robustness
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Ensure feasible h
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
            
            try:
                res_opt = minimize(
                    objective,
                    h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-6}
                )
                
                if res_opt.success:
                    h_opt = res_opt.x.copy()
                    c5 = objective(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        
                    # Small perturbation to avoid local minima
                    noise_scale = min(0.075, 0.025 + best_c5 * 0.05)
                    noise = np.random.normal(0, noise_scale, n_points)
                    h = h_opt + noise
                else:
                    # If optimization fails, try random restart with bad initial
                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)
                    
            except Exception as e:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
            
            # Projection to keep feasible
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-10:
                h = h * target_sum / h_sum
    
    # Final evaluation of the best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
