import numpy as np
from scipy.interpolate import interp1d
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        n_points = len(h_parent)
        # Upsample to target resolution if needed
        if n_points < 200:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 200)
            f = interp1d(x_orig, h_parent, kind='linear', fill_value='extrapolate')
            h_parent = f(x_new)
            n_points = 200
    else:
        n_points = 200
        h_parent = np.zeros(n_points)
        h_parent[:n_points//2] = 1

    # GA parameters
    pop_size = 75
    generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 3

    def ensure_feasible(h):
        """Ensure h is within bounds and sum equals n_points/2"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    def selection_tournament(pop, fitnesses):
        """Tournament selection"""
        idx = np.random.choice(len(pop), tournament_size)
        best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
        return pop[best_idx].copy()

    def crossover(parent1, parent2):
        """Blend crossover with repair"""
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        alpha = 0.5
        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        for i in range(len(parent1)):
            min_val = min(parent1[i], parent2[i])
            max_val = max(parent1[i], parent2[i])
            range_val = max_val - min_val
            child1[i] = np.clip(parent1[i] + alpha * range_val, 0, 1)
            child2[i] = np.clip(parent2[i] + alpha * range_val, 0, 1)
        
        # Repair
        child1 = ensure_feasible(child1)
        child2 = ensure_feasible(child2)
        return child1, child2

    def mutate(h):
        """Mutation with random reset and Gaussian perturbation"""
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                if np.random.random() < 0.5:
                    # Random reset
                    h[i] = np.random.random()
                else:
                    # Gaussian perturbation
                    h[i] += np.random.normal(0, 0.1)
        return ensure_feasible(h)

    # Initialize population
    population = []
    for _ in range(pop_size):
        if np.random.random() < 0.7:
            # Copy and mutate parent
            h = h_parent.copy()
            h = mutate(h)
        else:
            # Random initialization with constraint
            h = np.random.random(n_points)
            h = ensure_feasible(h)
        population.append(h)

    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]

    # Main GA loop
    last_improvement = 0
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitnesses[i] for i in sorted_indices]
        
        # Keep top 20% and generate new offspring
        elite_size = pop_size // 5
        new_pop = sorted_pop[:elite_size]
        
        # Create new offspring
        for _ in range(pop_size - elite_size):
            parent1 = selection_tournament(sorted_pop, sorted_fit)
            parent2 = selection_tournament(sorted_pop, sorted_fit)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.extend([child1, child2])
            if len(new_pop) >= pop_size:
                break

        # Trim to exact population size
        new_pop = new_pop[:pop_size]
        population = new_pop
        
        # Recalculate fitness
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        new_best_idx = np.argmin(fitnesses)
        if fitnesses[new_best_idx] < best_c5:
            best_c5 = fitnesses[new_best_idx]
            best_h = population[new_best_idx].copy()
            last_improvement = gen
            
        # Early stopping
        if gen - last_improvement > 25 and gen > 10:
            break

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
