import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

        # Compute gradient of sum over all correlations
        h_shifted = np.concatenate([h, h[:n_points-1]])
        grad = np.zeros_like(h)

                grad[i] += (1 - h_shifted[i-k]) * dx
        return grad
