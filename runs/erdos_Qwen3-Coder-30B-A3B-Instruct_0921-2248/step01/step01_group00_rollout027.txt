import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value of the correlation (scaled by dx)
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

    max_iter = 2000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles and velocities
    particles = np.random.rand(n_particles, n_points) * 0.5 + 0.25  # Keep in [0.25, 0.75] initially
    velocities = np.random.randn(n_particles, n_points) * 0.01
    p_best = particles.copy()
    p_best_fitness = np.array([evaluate_erdos_solution(p) for p in particles])
    
    # Initialize global best
    best_idx = np.argmin(p_best_fitness)
    g_best = p_best[best_idx].copy()
    g_best_fitness = p_best_fitness[best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Apply velocity limit
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update particle position
            particles[i] += velocities[i]
            
            # Apply boundary conditions
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Adjust sum to satisfy constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new fitness
            fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best_fitness[i] = fitness
                p_best[i] = particles[i].copy()
                
                # Update global best
                if fitness < g_best_fitness:
                    g_best_fitness = fitness
                    g_best = particles[i].copy()
    
    # Return best solution
    final_c5 = evaluate_erdos_solution(g_best)
    return (g_best.tolist(), final_c5, n_points)
