import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default: simple step function
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
    
    # Scale initial vector to satisfy sum constraint
    target_sum = n_points / 2.0
    h_init = h_init * target_sum / np.sum(h_init)
    
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Use cvxpy with MOSEK or similar QP solver
    # Define variable
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        C5 >= 0
    ]
    
    # We will need to represent the convolution constraints
    # but for now just do a small number of computed shifts
    # to stay within time budget

    dx = 2.0 / n_points
    max_shifts_to_consider = min(50, n_points//2)  # Prevent memory issues
    for shift in range(max_shifts_to_consider):
        # Compute shift for correlation
        shift_sum = 0
        for i in range(n_points - shift):
            shift_sum += h[i] * (1 - h[i + shift])
        # This requires full convolution for now
        # Approximating with a few key shifts to stay fast
        if shift < n_points//4:  # Restrict to smaller shifts for speed
            constraints.append(shift_sum <= C5 * (n_points / 2.0))
    
    # Try full convolution for a shorter sequence
    max_eval_points = 100
    if n_points > max_eval_points:
        # Downsample while preserving the sum
        h_eval = h_init[::(n_points // max_eval_points)]
        h_eval = h_eval * target_sum / np.sum(h_eval)
        n_eval = len(h_eval)
    else:
        h_eval = h_init
        n_eval = n_points
    
    # Use our best solution from previous attempts with ofer random restarts
    num_restarts = 10
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try local search around initial
        h_local = h_eval.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.05, n_eval)
        h_local += noise
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * target_sum / np.sum(h_local)
        
        # Now evaluate C5 for this case
        C5_val = np.max(np.correlate(h_local, 1 - h_local, mode="full")) * dx
        if C5_val < best_c5:
            best_c5 = C5_val
            best_h = h_local.copy()
    
    # Final refinement by solving with CVXPY for small number of shifts to be sure
    if n_eval <= 150:  # Limit to better performance
        h = cp.Variable(n_eval, name="h")
        C5 = cp.Variable(name="C5")
        
        # Original constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum,
            C5 >= 0
        ]
        
        # Use fewer shifts for computation speed
        max_shifts_for_cvx = min(30, n_eval//2)
        for shift in range(max_shifts_for_cvx):
            shift_sum = 0
            for i in range(n_eval - shift):
                shift_sum += h[i] * (1 - h[i + shift])
            constraints.append(shift_sum <= C5 * (n_eval / 2.0))
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-6, reltol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                solution = h.value
                solution = np.clip(solution, 0, 1)
                # Re-scale to correct sum
                solution = solution * target_sum / np.sum(solution)
                C5_val_new = np.max(np.correlate(solution, 1 - solution, mode="full")) * dx
                if C5_val_new < best_c5:
                    best_c5 = C5_val_new
                    best_h = solution.copy()
        except:
            pass

    # Final evaluation using the existing function
    final_func = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), float(final_func), len(best_h))
