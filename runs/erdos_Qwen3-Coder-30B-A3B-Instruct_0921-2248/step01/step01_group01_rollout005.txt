def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Initialize with uniform pattern to reduce variance
        n_points = 50
        h = np.zeros(n_points)
        # Place 1s at even indices
        h[::2] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_vec):
        # Calculate overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiple resolution optimizations
    resolutions = [50, 100, 150]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Resample to new resolution
        if res > n_points:
            # Interpolate
            old_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, res)
            h = np.interp(new_x, old_x, h)
            h = h * (res / 2.0) / np.sum(h)
            n_points = res
        elif res < n_points:
            # Downsample and interpolate
            old_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, res)
            h = np.interp(new_x, old_x, h)
            h = h * (res / 2.0) / np.sum(h)
            n_points = res
        
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        
        # Perturb and optimize
        num_restarts = 5
        for restart in range(num_restarts):
            if time.time() - start_time > budget_s - 2:
                break
                
            h_restarted = h.copy()
            
            # Add small noise to break symmetry
            noise_scale = 0.05 if res == 50 else (0.02 if res == 100 else 0.01)

            h_restarted = h_restarted + noise
            
            # Ensure feasibility
            h_restarted = np.clip(h_restarted, 0, 1)
            h_restarted_sum = np.sum(h_restarted)

                h_restarted = h_restarted * target_sum / h_restarted_sum

                res_opt = minimize(
                    objective,
                    h_restarted.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-6 if res == 150 else 1e-4}

                    h_final = res_opt.x.copy()
                    c5 = objective(h_final)

                        best_h = h_final.copy()
                        
                    # Add more noise to escape local minima
                    noise = np.random.normal(0, noise_scale*0.5, n_points)

                    h = np.random.rand(n_points)
                    h = h * target_sum / np.sum(h)

                h = np.random.rand(n_points)

    final_h_sum = np.sum(final_h)

        final_h = final_h * target_sum / final_h_sum
        
    final_c5 = objective(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
