import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This function evaluates the C5 value for the given h
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback: random binary construction
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for interval [0, 2]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Define upsampling targets
    upsampling_targets = [94, 188, 200]
    current_h = h.copy()
    current_n = n_points

    # Iterative optimization with upsampling
    for target_n in upsampling_targets:
        # Up-sample current solution
        upscale = target_n // current_n
        if upscale > 1:
            new_h = np.repeat(current_h, upscale)
            new_h = new_h[:target_n]  # truncate if necessary
            current_h = new_h
            current_n = target_n
        else:
            continue
            
        # Optimization with time limit
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 10:
            break
            
        max_iter = min(200, int(remaining_time * 0.5))
        
        try:
            # Random restart
            restart_h = current_h + np.random.normal(0, 0.01, current_n)
            restart_h = np.clip(restart_h, 0, 1)
            restart_h = restart_h * target_sum / np.sum(restart_h)
            
            # Optimize with L-BFGS-B
            res = minimize(obj_func, restart_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-8})
            
            if res.success:
                new_h = np.clip(res.x, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                new_c5 = obj_func(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
            else:
                new_h = restart_h
                new_c5 = obj_func(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
            
        except Exception:
            pass
            
        # Add more randomness for refinement
        if time.time() - start_time > budget_s - 20:
            break
            
        # Local search around current best
        for _ in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            perturbation = np.random.normal(0, 0.02, current_n)
            trial_h = best_h + perturbation
            trial_h = np.clip(trial_h, 0, 1)
            trial_h = trial_h * target_sum / np.sum(trial_h)
            trial_c5 = obj_func(trial_h)
            if trial_c5 < best_c5:
                best_h = trial_h.copy()
                best_c5 = trial_c5

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), len(final_h))
