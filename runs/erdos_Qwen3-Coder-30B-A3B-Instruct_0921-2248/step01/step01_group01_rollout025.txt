import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: binary vector with half 1s and half 0s
        n_points = 50
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def create_cvxpy_problem(h_vec):
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # For a fixed k, we compute the correlation term manually
        # Create a helper to compute the overlap for a specific k
        # This simplified version uses a basic approach
        # The actual overlap computation is the same as before
        expr = 0
        for i in range(n_points):
            expr += h_var[i] * (1-h_var[(i+1) % n_points])
        objective = cp.Minimize(expr)
        return cp.Problem(objective, constraints), h_var
    
    best_c5 = float('inf')
    best_h = h.copy()
    iterations = 0
    patience = 0
    max_patience = 100

    while time.time() - start_time < budget_s - 2:  # leave 2 seconds for final eval
        iterations += 1
        h = project_feasible(h)
        
        # Try local optimization with a simpler approach
        def objective(h_vec):
            return compute_overlap(h_vec)
            
        try:
            # Direct optimization with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                objective,
                h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
            )

            if res.success:
                h_new = res.x.copy()
                c5 = objective(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    patience = 0  # reset patience
                else:
                    patience += 1
            else:
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = project_feasible(h_new)
                c5 = objective(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    patience = 0
                else:
                    patience += 1
                    
        except Exception as e:
            # fallback: try again with random noise
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = project_feasible(h_new)
            c5 = objective(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
        
        # Random walk with low probability
        if np.random.rand() < 0.1:
            h_new = h + np.random.normal(0, 0.005, n_points)
            h_new = project_feasible(h_new)
            c5 = objective(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                patience = 0

        # No improvement for too long, perturb more aggressively
        if patience > max_patience:
            noise_scale = 0.05
            h = h + np.random.normal(0, noise_scale, n_points)
            patience = 0

        # Final projection
        best_h = project_feasible(best_h)
        
        # Check if we're running out of time
        if time.time() - start_time > budget_s - 2:
            break

    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
