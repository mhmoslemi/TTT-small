def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure feasibility of initial h
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * target_sum / h_sum
    
    # Objective function to minimize
    def objective(h_vec):
        # Calculate the overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer configuration
    maxiter = 250
    ftol = 1e-6
    gtol = 1e-6
    
    # Multiple restarts with different initialization strategies
    restarts = 5
    
    for restart in range(restarts):
        if time.time() - start_time > budget_s - 2:
            break

        if restart == 0:
            h0 = h.copy()

            # Random perturbation of the solution
            noise_scale = 0.1 * (best_c5 if best_c5 < 1 else 0.1)
            noise = np.random.normal(0, noise_scale, n_points)
            h0 = best_h + noise
            h0 = np.clip(h0, 0, 1)
        
        # Ensure h0 is feasible
        h0_sum = np.sum(h0)
        h0 = h0 * target_sum / h0_sum
        
        # Optimization with L-BFGS-B
        try:
            res = minimize(
                objective,
                h0,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol}
            )

                h_opt = res.x.copy()
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

            h = h_opt + np.random.normal(0, noise_scale, n_points)
        except Exception:
            # Fallback to simple randomization
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)

    n_fine = 100

        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_points), best_h)

        target_sum_fine = n_fine / 2.0

        # Optimization at higher resolution

            max_overlap = np.max(corr)
            return max_overlap * dx_fine

                method='L-BFGS-B',

            )
            
            if res.success:

                c5_finest = objective_fine(h_finest)
                if c5_finest < best_c5:
                    best_c5 = c5_finest
                    best_h = h_finest.copy()
        except Exception:
            pass
    
    # Final projection
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-10:
        best_h = best_h * target_sum / best_h_sum
    
    # Final evaluation
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
