import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1
        np.random.shuffle(h0)
        
    dx = 2.0 / n_points
    
    # Define the optimization problem
    def solve_with_cvxpy(h_initial):
        h = Variable(n_points, name='h')
        C = Variable(name='C')
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2,
            C >= 0
        ]
        
        # Compute overlap integral for each shift
        for k in range(1, n_points):
            # Expression for the integral over shift k
            # Need to compute sum of h[i] * (1 - h[i+k]) for valid i
            expr = 0
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i + k])
            expr *= dx
            constraints.append(expr <= C)
        
        # Objective is to minimize C
        objective = Minimize(C)
        
        prob = Problem(objective, constraints)
        
        # Try to solve it
        try:
            prob.solve(solver=SCS, max_iters=1000, eps=1e-6)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                sol_h = h.value
                sol_c = C.value
                return sol_h, sol_c
        except:
            pass
        return None, None

    # Simple gradient-based optimization with restarts
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use CVXPy to improve initial result
    cvx_h, cvx_c = solve_with_cvxpy(h0)
    if cvx_h is not None and cvx_c < best_c5:
        best_h = cvx_h
        best_c5 = cvx_c
        
    # Try different random restarts
    num_restarts = 5
    for i in range(num_restarts):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Perturb the best solution
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Normalize to maintain constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_perturbed)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_perturbed.copy()
    
    # Run a final optimization
    def obj_func(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    # Try to refine with SLSQP if we have time
    while time.time() - start_time < budget_s - 1:
        try:
            res = minimize(obj_func, best_h.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons, options={'maxiter': 100})
            if res.success:
                aux_h = np.clip(res.x, 0, 1)
                aux_h = aux_h * (n_points / 2) / np.sum(aux_h)
                c5_val = obj_func(aux_h)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = aux_h.copy()
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
