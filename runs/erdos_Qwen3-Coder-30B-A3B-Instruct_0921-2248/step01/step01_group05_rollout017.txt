import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # dx for the interval [0, 2]
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    
    def obj_func(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds
    bounds = [(0.0, 1.0)] * n_points
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    # Multi-res optimization with upsampling
    upscale_factors = [2, 2, 12]  # Total upscale to 47 * 2 * 2 * 12 = 2256
    current_n = n_points
    current_h = h.copy()
    
    for factor in upscale_factors:
        if time.time() - start_time > budget_s - 10:
            break
            
        current_n *= factor
        current_h = np.repeat(current_h, factor)
        dx = 2.0 / current_n
        target_sum = current_n / 2.0
        
        # Define new bounds and constraints for the higher resolution
        bounds = [(0.0, 1.0)] * current_n
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize
        try:
            # Random restarts to escape local minima
            best_local = current_h.copy()
            best_local_c5 = obj_func(current_h)
            
            for _ in range(5):
                if time.time() - start_time > budget_s - 10:
                    break
                    
                # Random perturbation of current solution
                perturb = np.random.rand(current_n) * 0.1
                h_start = np.clip(current_h + perturb - np.mean(perturb), 0, 1)
                h_start = h_start * target_sum / np.sum(h_start)
                
                # Optimize at this resolution
                res = minimize(obj_func, h_start, method='L-BFGSB', bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-7, 'gtol': 1e-7})
                
                if res.success:
                    h_opt = np.clip(res.x, 0, 1)
                    h_opt = h_opt * target_sum / np.sum(h_opt)
                    c5_opt = obj_func(h_opt)
                    if c5_opt < best_local_c5:
                        best_local = h_opt.copy()
                        best_local_c5 = c5_opt
                        
                        if best_local_c5 < best_c5:
                            best_h = best_local.copy()
                            best_c5 = best_local_c5
                else:
                    # If optimization fails, try to improve with simple projection
                    h_proj = np.clip(current_h, 0, 1)
                    h_proj = h_proj * target_sum / np.sum(h_proj)
                    c5_proj = obj_func(h_proj)
                    if c5_proj < best_local_c5:
                        best_local = h_proj.copy()
                        best_local_c5 = c5_proj
                        
                        if best_local_c5 < best_c5:
                            best_h = best_local.copy()
                            best_c5 = best_local_c5
                
                # Crossover with the current best
                for i in range(0, current_n, 5):
                    if np.random.rand() < 0.2:
                        best_h[i] = (best_h[i] + current_h[i]) / 2
                        
                # Reoptimize for crossover
                h_cross = np.clip(best_h, 0, 1)
                h_cross = h_cross * target_sum / np.sum(h_cross)
                c5_cross = obj_func(h_cross)
                if c5_cross < best_c5:
                    best_h = h_cross.copy()
                    best_c5 = c5_cross
            
            current_h = best_local.copy()
            
        except Exception as e:
            # Fallback to simpler approach
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            
    # Final intensive local search
    if time.time() - start_time < budget_s - 5:
        # Generate multiple random restarts close to the current solution
        best_final_h = best_h.copy()
        best_final_c5 = best_c5
        
        for _ in range(20):  # More random restarts
            if time.time() - start_time > budget_s - 2:
                break
                
            # Add Gaussian noise to current solution
            noise = np.random.normal(0, 0.05, current_n)
            h_trial = np.clip(best_final_h + noise, 0, 1)
            h_trial = h_trial * target_sum / np.sum(h_trial)
            c5_trial = obj_func(h_trial)
            
            if c5_trial < best_final_c5:
                best_final_h = h_trial.copy()
                best_final_c5 = c5_trial
                
        best_h = best_final_h
        best_c5 = best_final_c5
    
    return (best_h.tolist(), float(best_c5), current_n)
