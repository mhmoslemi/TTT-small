import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value among all shifts
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a smooth approximation to initial construction
        n_points = 100
        h_init = np.zeros(n_points)
        # Simple piecewise constant construction
        h_init[:n_points//2] = 1.0
        # Add some smoothness
        h_init = np.clip(h_init, 0, 1)
    
    # Project to correct sum constraint
    h_sum = np.sum(h_init)
    h_init = h_init * (n_points / 2) / h_sum
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Gradient-based optimization using L-BFGS-B
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    options = {'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-8, 'maxcor': 10}
    
    # Local search with multiple restarts
    restarts = min(20, int(budget_s / 10))
    for i in range(restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random restart
        h0 = h_init.copy()
        # Perturb with small noise
        h0 += np.random.normal(0, 0.01, n_points)
        h0 = np.clip(h0, 0, 1)
        
        # Normalize
        h_sum = np.sum(h0)
        if h_sum > 0:
            h0 = h0 * (n_points / 2) / h_sum
        
        try:
            res = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, 
                          constraints=constraints, options=options)
            if res.success:
                h_opt = res.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                h_sum = np.sum(h_opt)
                if h_sum > 0:
                    h_opt = h_opt * (n_points / 2) / h_sum
                
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
        except Exception:
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
