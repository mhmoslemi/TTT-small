```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fall back to a simple construction
        n_points = 100
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        h0 = np.random.permutation(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    con = {'type': 'eq', 'fun': constraint_sum}
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Objective function to minimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Helper function for gradient-based optimization
    def obj_grad(h):
        # Simple finite difference gradient
        eps = 1e-5
        grad = np.zeros_like(h)
        base_val = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    # Optimization loop with adaptive step sizes
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = project(h_pert)
        
        # Try both SLSQP and L-BFGS
        try:
            result = minimize(
                objective,
                h_pert,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=con,
                options={'maxiter': 100},
                tol=1e-6
            )
            
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
        
        # Strengthen with gradient-based line search
        try:
            grad = obj_grad(best_h)
            step_size = 0.01
            h_candidate = best_h - step_size * grad
            h_candidate = project(h_candidate)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        except:
            pass
    
    # Final refinement with more aggressive optimization
    try:
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=con,
            options={'maxiter': 50},
            tol=1e-7
        )
        
        if result.success:
            h_opt = project(result.x)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except:
        pass
    
    return (best_h, best_c5, n_points)
```
