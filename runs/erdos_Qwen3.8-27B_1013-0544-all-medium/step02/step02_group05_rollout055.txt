from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
        # Iterative projection
        for _ in range(10):
            s = h.sum()
            if abs(s - target_sum) < 1e-10:
                break
            # Adjust only non-boundary elements
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() == 0:
                break
            h[free] += (target_sum - s) / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    if nl > 1:
        diff_a = a_arr[:, None] - a_arr[None, :]
        diff_c = c_arr[:, None] - c_arr[None, :]
        valid = np.abs(diff_a) > 1e-12
        with np.errstate(divide='ignore', invalid='ignore'):
            x_pts = np.where(valid, diff_c / diff_a, np.inf)
        iu = np.triu_indices(nl, k=1)
        xv = x_pts[iu[0], iu[1]]
        xv = xv[np.isfinite(xv) & (xv >= -0.5) & (xv <= 1.5)]
        if len(xv) > 0:
            candidates = np.concatenate([candidates, xv])
    for i in range(nl):
        a_i = a_arr[i]
        c_i = c_arr[i]
        B = a_i - 1.0
        C = c_i - S_0
        disc = B * B - 4.0 * C
        if disc >= 0:
            sq = np.sqrt(disc)
            candidates = np.append(candidates, [(-B + sq) * 0.5, (-B - sq) * 0.5])
    candidates = np.clip(candidates, 0.0, 1.0)
    candidates = np.unique(np.round(candidates, 12))
    if len(candidates) == 0:
        return hm
    lin_vals = a_arr[:, None] * candidates[None, :] + c_arr[:, None]
    max_lin = np.max(lin_vals, axis=0)
    quad_vals = S_0 + candidates - candidates * candidates
    total = np.maximum(quad_vals, max_lin)
    best_idx = np.argmin(total)
    return float(candidates[best_idx])

def coordinate_descent(h, target_sum, max_sweeps=200, tol=1e-9, time_limit=None, start_time=None, rng=None):
    n = len(h)
    h = h.copy()
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if rng is not None:
            rng.shuffle(order)
        else:
            np.random.shuffle(order)
        max_change = 0.0
        for m in order:
            corr = np.correlate(h, 1.0 - h, mode="full")
            new_val = coord_solve_1d(m, h, corr)
            max_change = max(max_change, abs(new_val - h[m]))
            h[m] = new_val
        h = project_h(h, target_sum)
        if max_change < tol:
            break
    return h

def slsqp_polish(h0, target_sum, max_iter=300, beta=100):
    n = len(h0)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        h_new = project_h(res.x, target_sum)
        return h_new
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 1)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = None

    # Primary optimization at n=70 (matching parent)
    n_primary = 70
    target_sum = n_primary / 2.0

    # Build diverse starting points
    starts = []
    if h_parent is not None and len(h_parent) == n_primary:
        starts.append(h_parent.copy())
    if h_parent is not None and len(h_parent) != n_primary:
        xo = np.linspace(0, 2, len(h_parent), endpoint=False)
        xn = np.linspace(0, 2, n_primary, endpoint=False)
        h_interp = np.interp(xn, xo, h_parent)
        starts.append(project_h(h_interp, target_sum))

    # Bimodal structured start
    n_ones = int(target_sum)
    h_bm = np.zeros(n_primary)
    h_bm[:n_ones//2] = 1.0
    h_bm[n_primary//2:n_primary//2 + n_ones - n_ones//2] = 1.0
    starts.append(project_h(h_bm, target_sum))

    # Random starts
    for _ in range(2):
        starts.append(project_h(rng.random(n_primary), target_sum))

    # Phase 1: Coordinate descent from each start
    for si, h0 in enumerate(starts):
        if time_left() < 120:
            break
        h0 = project_h(h0, target_sum)
        alloc = min(time_left() * 0.5, 250)
        h_cd = coordinate_descent(h0, target_sum, max_sweeps=300, tol=1e-9,
                                  time_limit=alloc * 0.6, start_time=time.time(), rng=rng)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5
            best_n = n_primary

    # Phase 2: SLSQP polish from best CD result
    if best_h is not None and time_left() > 80:
        for beta in [60, 120, 200]:
            if time_left() < 40:
                break
            h_pol = slsqp_polish(best_h, target_sum, max_iter=500, beta=beta)
            c5 = compute_c5(h_pol)
            if c5 < best_c5:
                best_h = h_pol.copy()
                best_c5 = c5
                best_n = n_primary

    # Phase 3: More CD from polished result
    if best_h is not None and time_left() > 60:
        h_cd2 = coordinate_descent(best_h, target_sum, max_sweeps=500, tol=1e-10,
                                   time_limit=min(time_left() * 0.3, 150), start_time=time.time(), rng=rng)
        c5 = compute_c5(h_cd2)
        if c5 < best_c5:
            best_h = h_cd2.copy()
            best_c5 = c5

    # Phase 4: Try higher resolution
    for n_alt in [90, 110]:
        if time_left() < 100:
            break
        ts_alt = n_alt / 2.0
        if best_h is not None:
            xo = np.linspace(0, 2, len(best_h), endpoint=False)
            xn = np.linspace(0, 2, n_alt, endpoint=False)
            h_alt = np.interp(xn, xo, best_h)
        else:
            h_alt = rng.random(n_alt)
        h_alt = project_h(h_alt, ts_alt)
        h_alt_cd = coordinate_descent(h_alt, ts_alt, max_sweeps=200, tol=1e-9,
                                      time_limit=min(time_left() * 0.3, 120), start_time=time.time(), rng=rng)
        c5 = compute_c5(h_alt_cd)
        if c5 < best_c5:
            best_h = h_alt_cd.copy()
            best_c5 = c5
            best_n = n_alt
        if time_left() > 40 and c5 < best_c5 + 0.001:
            h_pol = slsqp_polish(h_alt_cd, ts_alt, max_iter=200, beta=100)
            c5p = compute_c5(h_pol)
            if c5p < best_c5:
                best_h = h_pol.copy()
                best_c5 = c5p
                best_n = n_alt

    # Phase 5: Final intensive CD on best solution
    if best_h is not None and time_left() > 30:
        ts = best_n / 2.0
        h_final = coordinate_descent(best_h, ts, max_sweeps=1000, tol=1e-11,
                                     time_limit=min(time_left() - 5, 120), start_time=time.time(), rng=rng)
        c5 = compute_c5(h_final)
        if c5 < best_c5:
            best_h = h_final.copy()
            best_c5 = c5

    # Phase 6: Final SLSQP
    if best_h is not None and time_left() > 15:
        ts = best_n / 2.0
        h_pol = slsqp_polish(best_h, ts, max_iter=1000, beta=300)
        c5 = compute_c5(h_pol)
        if c5 < best_c5:
            best_h = h_pol.copy()
            best_c5 = c5
        # One more CD
        if time_left() > 10:
            h_final2 = coordinate_descent(best_h, ts, max_sweeps=200, tol=1e-11,
                                          time_limit=min(time_left() - 3, 60), start_time=time.time(), rng=rng)
            c5 = compute_c5(h_final2)
            if c5 < best_c5:
                best_h = h_final2.copy()
                best_c5 = c5

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
