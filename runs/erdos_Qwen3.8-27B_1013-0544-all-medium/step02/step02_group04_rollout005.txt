def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)


def obj_full(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, _g_beta) * _g_dx


def obj_focused(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    sel = corr[_g_top_lags]
    return smooth_max(sel, _g_beta) * _g_dx


def cfun(h):
    return h.sum() - _g_target


def cjac(h):
    return np.ones(h.shape[0])


def make_grad_active(h, lag_idx):
    n = _g_n
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        for j in range(n - lag):
            grad[j] += (1.0 - h[j + lag])
    else:
        for j in range(n + lag):
            grad[j + (-lag)] += (1.0 - h[j])
            grad[j] -= h[j + (-lag)]
    grad = grad - np.mean(grad)
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    global _g_beta, _g_dx, _g_target, _g_n, _g_top_lags
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        _g_n = len(h_init)
    else:
        _g_n = 70
        h_init = np.full(_g_n, 0.5)

    n = _g_n
    _g_dx = 2.0 / n
    _g_target = n / 2.0
    best_h = project_h(h_init, _g_target)
    best_c5 = compute_c5(best_h)

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with moderate beta from initial point
    for beta in [20, 50, 100, 200]:
        if tl() < 200:
            break
        _g_beta = beta
        try:
            res = minimize(obj_full, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-16})
            h_new = project_h(res.x, _g_target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Active-set focused optimization
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    _g_top_lags = np.argsort(corr)[-6:]
    _g_beta = 500

    for _ in range(40):
        if tl() < 150:
            break
        try:
            res = minimize(obj_focused, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_h(res.x, _g_target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
                corr = np.correlate(best_h, 1.0 - best_h, mode="full")
                _g_top_lags = np.argsort(corr)[-6:]
        except Exception:
            pass

    # Phase 3: Multi-restart with diverse perturbations
    _g_beta = 400
    for attempt in range(500):
        if tl() < 30:
            break
        h0 = best_h.copy()
        strat = attempt % 8

        if strat == 0:
            h0 += rng.randn(n) * 0.006
        elif strat == 1:
            gi = _g_top_lags[rng.randint(0, len(_g_top_lags))]
            grad = make_grad_active(best_h, gi)
            h0 -= 0.004 * grad
        elif strat == 2:
            i, j = rng.choice(n, 2, replace=False)
            d = 0.025 * rng.randn()
            h0[i] += d
            h0[j] -= d
        elif strat == 3:
            blen = rng.randint(3, 10)
            bs = rng.randint(0, n - blen)
            h0[bs:bs + blen] += rng.randn(blen) * 0.012
        elif strat == 4:
            h0 = 0.9 * h0 + 0.1 * h0[::-1]
            h0 += rng.randn(n) * 0.003
        elif strat == 5:
            # Negative gradient from random active lag
            li = _g_top_lags[rng.randint(0, len(_g_top_lags))]
            lag = li - (n - 1)
            for j in range(n):
                jp = j + lag
                jn = j - lag
                if 0 <= jp < n:
                    h0[j] -= 0.002 * (1.0 - best_h[jp])
                if 0 <= jn < n:
                    h0[j] += 0.002 * best_h[jn]
        elif strat == 6:
            # Gentle annealing toward uniform
            h0 = 0.97 * h0 + 0.03 * 0.5
        else:
            # Random Fourier perturbation
            k = rng.randint(1, 4)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.arange(n)
            pert = 0.005 * np.sin(2 * np.pi * k * t / n + phase)
            h0 = h0 + pert - np.mean(pert)

        h0 = project_h(h0, _g_target)
        try:
            res = minimize(obj_full, h0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_h(res.x, _g_target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
                corr = np.correlate(best_h, 1.0 - best_h, mode="full")
                _g_top_lags = np.argsort(corr)[-6:]
        except Exception:
            pass

    # Phase 4: Final high-precision polish
    if tl() > 20:
        for beta in [1000, 5000, 10000]:
            if tl() < 10:
                break
            _g_beta = beta
            try:
                res = minimize(obj_full, best_h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 30000, 'ftol': 1e-17})
                h_new = project_h(res.x, _g_target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    best_h = project_h(best_h, _g_target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
