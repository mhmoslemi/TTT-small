import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    bc = beta * corr
    bc_max = bc.max()
    e = np.exp(bc - bc_max)
    sum_e = e.sum()
    w = e / sum_e
    
    f_val = (bc_max + np.log(sum_e)) / beta * dx + lam * (h.sum() - n/2.0)**2
    
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k] * beta * dx
        if wk < 1e-18:
            continue
        s = k - (n-1)
        if s >= 0:
            # (1-h[j-s]) for j in [s, n)
            g[s:] += wk * (1.0 - h[:n-s])
            # -h[j+s] for j in [0, n-s)
            if n-s > 0:
                g[:n-s] -= wk * h[s:]
        else:
            t = -s
            # (1-h[j+t]) for j in [0, n-t)
            if n-t > 0:
                g[:n-t] += wk * (1.0 - h[t:])
            # -h[j-t] for j in [t, n)
            g[t:] -= wk * h[:n-t]
    
    g += 2.0 * lam * (h.sum() - n/2.0)
    return f_val, g

    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False) + 1.0/n_dst
    h_interp = np.interp(x_dst, x_src, np.concatenate([h_src, [h_src[0]]]))
    # Actually, for a periodic function, we should wrap around
    # Let's use a simpler approach
    x_src2 = np.linspace(0, 2.0, n_src)  # include endpoint
    h_src2 = np.concatenate([h_src, [h_src[0]]])  # wrap
    h_interp = np.interp(x_dst, x_src2, h_src2)
    return h_interp
