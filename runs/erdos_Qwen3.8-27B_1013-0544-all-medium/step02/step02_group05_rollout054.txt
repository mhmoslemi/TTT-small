from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smax_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * dx

def smax_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for i in range(n):
        gi = 0.0
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-15:
                continue
            l = idx - (n - 1)

                gi += wi * (1.0 - h[i + l])
            if 0 <= i - l < n:
                gi -= wi * h[i - l]
        grad[i] = gi * dx
    return grad

    h = np.clip(h, 0.0, 1.0)
    d = tsum - h.sum()
    if abs(d) > 1e-10:

        h = np.clip(h, 0.0, 1.0)

                if time.time() - t0 > t_alloc:
                    break
                scale = 0.02 + 0.02 * it / 20
                h0 = base + np.random.randn(n_pts) * scale
                h0 = proj(h0, tsum)
                def obj2(x):
                    return smax_obj(x, 300.0, dx)
                def obj2g(x):
                    return smax_grad(x, 300.0, dx)
                cons = [{'type': 'eq', 'fun': lambda x, ts=tsum: x.sum() - ts,
                         'jac': lambda x, ts=tsum: np.ones(len(x))}]
                try:
                    res = minimize(obj2, h0, method='SLSQP', bounds=bounds,
                                   constraints=cons, jac=obj2g,
                                   options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
                    h_new = proj(res.x, tsum)
                    c5_new = compute_c5(h_new)
                    if c5_new < curr_c5 - 1e-12:
                        base, curr_c5 = h_new.copy(), c5_new
                        if curr_c5 < best_c5:
                            best_h, best_c5, best_n = h_new.copy(), c5_new, n_pts
                except Exception:
                    pass

    # Final polish
    if best_h is not None and time.time() - start_time < budget_s - 10:
        nf = best_n
        df = 2.0 / nf
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        consf = [{'type': 'eq', 'fun': lambda x, ts=tf: x.sum() - ts,
                  'jac': lambda x, ts=tf: np.ones(len(x))}]
        for beta in [800, 1200, 2000]:
            if time.time() - start_time > budget_s - 5:
                break
            def objf(x, b=beta, d=df):
                return smax_obj(x, b, d)
            def objfg(x, b=beta, d=df):
                return smax_grad(x, b, d)
            try:
                res = minimize(objf, best_h, method='SLSQP', bounds=bf,
                               constraints=consf, jac=objfg,
                               options={'maxiter': 2000, 'ftol': 1e-16, 'iprint': 0})
                h_new = proj(res.x, tf)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    return best_h, best_c5, best_n
