```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            # Need to move some value from a boundary
            if diff > 0:
                # Need to increase sum, so decrease some 1s
                at_one = np.where(h >= 1.0 - 1e-12)[0]
                if len(at_one) > 0:
                    idx = at_one[0]
                    amt = min(diff, 0.5)
                    h[idx] -= amt
                    diff -= amt
                continue
            else:
                at_zero = np.where(h <= 1e-12)[0]
                if len(at_zero) > 0:
                    idx = at_zero[0]
                    amt = min(-diff, 0.5)
                    h[idx] += amt
                    diff += amt
                continue
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_smooth_max(h, beta):
    """Compute gradient of smooth_max(corr(h), beta)"""
    n = len(h)
    corr = corr_of_h(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    # For numerical stability
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    weights = exps / exps.sum()
    
    # Gradient of smooth_max = sum_k weights[k] * grad(corr[k])
    # grad(corr[k]) w.r.t. h[j]
    grad = np.zeros(n)
    # corr has 2n-1 entries, index m corresponds to lag k = m - (n-1)
    for m_idx in range(2 * n - 1):
        if weights[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        w = weights[m_idx]
        if k >= 0:
            # 0 ≤ j ≤ n-1-k: contribution (1-h[j+k])
            if n - 1 - k >= 0:
                idx1 = np.arange(0, n - k)
                grad[idx1] += w * (1.0 - h[idx1 + k])
            # k ≤ j ≤ n-1: contribution -h[j-k]
            if k <= n - 1:
                idx2 = np.arange(k, n)
                grad[idx2] += w * (-h[idx2 - k])
        else:
            kp = -k
            # kp ≤ j ≤ n-1: contribution (1-h[j-kp])
            if kp <= n - 1:
                idx1 = np.arange(kp, n)
                grad[idx1] += w * (1.0 - h[idx1 - kp])
            # 0 ≤ j ≤ n-1-kp: contribution -h[j+kp]
            if n - 1 - kp >= 0:
                idx2 = np.arange(0, n - kp)
                grad[idx2] += w * (-h[idx2 + kp])
    return grad * beta  # chain rule: d/dh [smooth_max] = beta * sum(w_k * d(corr_k)/dh)
```

```python
def gradient_optimize(h0, n, target_sum, rng, time_left, min_time=10, start_beta=10, end_beta=1000):
    h = h0.copy()
    best_local = compute_c5(h)
    best_local_h = h.copy()
    
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    lr = 0.01  # initial learning rate
    
    t_start = time.time()
    total_steps = 0
    target_steps = 50000  # aim for this many steps
    
    while time_left() > min_time and total_steps < target_steps:
        batch_size = max(1, target_steps // 10)
        progress = total_steps / target_steps
        beta = start_beta * (end_beta / start_beta) ** progress
        
        # Nesterov look-ahead
        h_look = h + mu * momentum
        grad = grad_smooth_max(project_h(h_look, target_sum), beta)
```

For each lag k (0 to n-1, positive lags), the gradient contribution is:
- For j = 0 to n-1-k: grad[j] += w * (1 - h[j+k])
- For j = k to n-1: grad[j] += w * (-h[j-k])

For negative lags (k = -(n-1) to -1), similar but with k' = -k.

```python
def grad_smooth_max_vec(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for lag_idx in range(2 * n - 1):
        w = weights[lag_idx]
        if w < 1e-15:
            continue
        k = lag_idx - (n - 1)
        if k >= 0:
            length = n - k
            grad[:length] += w * one_minus_h[k:]  # (1-h[j+k]) for j=0..n-1-k
            grad[k:] += w * (-h[:length])           # -h[j-k] for j=k..n-1
        else:
            kp = -k
            length = n - kp
            grad[kp:] += w * one_minus_h[:length]   # (1-h[j-kp]) for j=kp..n-1
            grad[:length] += w * (-h[kp:])           # -h[j+kp] for j=0..n-1-kp
    
    return grad
```

Actually, let me think about whether I can do even better. The key observation is:

grad[j] = sum_k w_k * [contribution of h[j] to corr[k]]

For a given j, which lags k have h[j] in their sum?
- corr[k] = sum_i h[i]*(1-h[i+k])
- h[j] appears as h[i] when i=j (if j+k is valid, i.e., 0 ≤ j+k < n)
- h[j] appears as h[i+k] when i = j-k (if 0 ≤ j-k < n)

So for a fixed j:
- Contribution from "h[i]" role: (1-h[j+k]) for k such that 0 ≤ j+k < n, i.e., -j ≤ k < n-j
- Contribution from "h[i+k]" role: -h[j-k] for k such that 0 ≤ j-k < n, i.e., j-n < k ≤ j

grad[j] = sum_k w_k * (1-h[j+k]) * [0≤j+k<n] + sum_k w_k * (-h[j-k]) * [0≤j-k<n]

First sum = sum_{u: max(0,j-n+1)≤u≤min(n-1,j+n-1)} W[u-j] * (1-h[u])

Second sum: let v = j-k, so k = j-v, v ranges from max(0, j-(n-1)) to min(n-1, j+(n-1)):
Second sum = sum_{v: max(0,j-n+1)≤v≤min(n-1,j+n-1)} W[j-v] * h[v]

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = free.sum()
        if nfree == 0:
            if diff > 0:
                candidates = np.where(h >= 1.0 - 1e-10)[0]
            else:
                candidates = np.where(h <= 1e-10)[0]
            if len(candidates) == 0:
                break
            # Move from the most extreme
            idx = candidates[np.argmax(h)] if diff > 0 else candidates[np.argmin(h)]
            step = min(abs(diff), 0.1) * (1 if diff > 0 else -1)
            h[idx] += step
            continue
        h[free] += diff / nfree
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, beta):
    n = len(h)
    corr = corr_of_h(h)
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    total = exps.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights = exps / total
    
    grad = np.zeros(n)
    omh = 1.0 - h
    
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            L = n - k
            if L > 0:
                grad[:L] += w * omh[k:]
            if k < n:
                grad[k:] += w * (-h[:L])
        else:
            kp = -k
            L = n - kp
            if L > 0:
                grad[kp:] += w * omh[:L]
            if kp < n:
                grad[:L] += w * (-h[kp:])
    return grad

    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    momentum = np.zeros(n)
    mu = 0.85
    lr = 0.05
    
    t0 = time.time()
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() - t0 > max_time:
            break
        
        progress = min(1.0, it / max_iter)
        beta = start_beta * (end_beta / start_beta) ** (progress * 2)
        
        # Nesterov
        h_ne = project_h(h + mu * momentum, target_sum)
        g = grad_smooth_max(h_ne, beta)
        
        # Update
        momentum = mu * momentum - lr * g
        h_new = h + momentum
        h_new = project_h(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnant = 0
            # Slightly increase learning rate on improvement
            lr = min(lr * 1.01, 0.1)
        else:
            stagnant += 1
            if stagnant > 500:
                # Reduce learning rate
                lr *= 0.9
                momentum *= 0.5
                stagnant = 0
        
        h = h_new
        # Slowly decay learning rate
        lr = max(0.001, 0.05 * (0.9995 ** it))
    
    return best_h, best_c5
```
