import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -700, 700)))) / beta


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            idx = np.argsort(h)
            for i in idx:
                adj = min(1.0 - h[i], d)
                h[i] += adj
                d -= adj
                if d <= 1e-14:
                    break
        else:
            idx = np.argsort(-h)
            for i in idx:
                adj = min(h[i], -d)
                h[i] -= adj
                d += adj
                if d >= -1e-14:
                    break
    return h


def _obj_and_grad(h, n, ts, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(np.clip(beta * (corr - m), -700, 700))
    w = exps / np.sum(exps)
    val = smooth_max_val(corr, beta) * dx + lam * (h.sum() - ts) ** 2
    g = np.correlate(w, 1.0 - 2.0 * h, mode="valid") * dx
    g = g + 2.0 * lam * (h.sum() - ts)
    return val, g


def _lbgfs(h, n, ts, beta, lam, bnds, maxiter=500):
    def fun(x):
        return _obj_and_grad(x, n, ts, beta, lam)[0]

    def jac(x):
        return _obj_and_grad(x, n, ts, beta, lam)[1]

    r = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bnds,
                 options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return r.x


def _coord_descent(h, n, ts, beta, lam, max_sweeps=5):
    h = project_box_sum(h.copy(), ts)
    dx = 2.0 / n

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            h_old = h[i]

            def f_1d(t):
                h[i] = np.clip(t, 0.0, 1.0)
                s = h.sum()
                corr = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max_val(corr, beta) * dx + lam * (s - ts) ** 2

            h[i] = h_old
            res = minimize_scalar(f_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            h[i] = np.clip(res.x, 0.0, 1.0)
            if abs(h[i] - h_old) > 1e-7:
                improved = True
        h = project_box_sum(h, ts)
        if not improved:
            break
    return h


def _perturb(h, n, ts, strategy, rng):

    if strategy == 0:
        scale = 0.03
        h += rng.randn(n) * scale
    elif strategy == 1:

    elif strategy == 2:

    elif strategy == 3:

        k_star = int(np.argmax(corr)) - (n - 1)

            j2 = j + k_star

    return project_box_sum(h, ts)


def _parametric_init(n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    a1 = rng.uniform(0.1, 0.45)
    w1 = rng.uniform(1.5, 5.0) * np.pi
    p1 = rng.uniform(0, 2 * np.pi)
    a2 = rng.uniform(-0.25, 0.25)
    w2 = rng.uniform(3.0, 10.0) * np.pi
    p2 = rng.uniform(0, 2 * np.pi)
    h = 0.5 + a1 * np.sin(w1 * x + p1) + a2 * np.sin(w2 * x + p2)
    return project_box_sum(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - 5.0 - (time.time() - t0)

    def try_improve(h_candidate):
        nonlocal best_h, best_c5
        c = compute_c5(h_candidate)
        if c < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c
            return True
        return False

    # Phase 1: Progressive L-BFGS-B from initial
    beta_schedule = [(50, 10, 400), (100, 30, 400), (200, 80, 300),
                     (400, 150, 300), (800, 300, 200), (1500, 500, 150)]
    for beta, lam, iters in beta_schedule:
        if tl() < 15:
            break
        try:
            h_new = _lbgfs(best_h, n, ts, beta, lam, bnds, iters)
            h_new = project_box_sum(h_new, ts)
            try_improve(h_new)
        except Exception:
            pass

    # Phase 2: Coordinate descent polish
    if tl() > 30:
        for beta, lam in [(200, 100), (400, 200), (800, 400)]:
            if tl() < 15:
                break
            for ms in [5, 10, 15]:
                if tl() < 10:
                    break
                h_cd = _coord_descent(best_h, n, ts, beta, lam, ms)
                try_improve(h_cd)

    # Phase 3: Diversified restarts
    starts = []
    for _ in range(12):
        if tl() < 40:
            break
        starts.append(_parametric_init(n, rng))
    starts.append(best_h.copy())

    for h_start in starts:
        if tl() < 60:
            break
        for beta, lam, iters in [(100, 50, 300), (300, 100, 200), (600, 200, 150)]:
            if tl() < 15:
                break
            try:
                h_new = _lbgfs(h_start, n, ts, beta, lam, bnds, iters)
                h_new = project_box_sum(h_new, ts)
                if try_improve(h_new):
                    h_start = best_h.copy()
            except Exception:
                pass
        # Coordinate descent after L-BFGS-B
        if tl() > 20:
            h_cd = _coord_descent(h_start, n, ts, 400, 200, 8)
            try_improve(h_cd)

    # Phase 4: Intensive local search with perturbations
    it = 0
    while tl() > 60:
        it += 1
        strategy = it % 6
        scale_factor = max(0.5, 1.0 - 0.02 * it)
        h_pert = _perturb(best_h, n, ts, strategy, rng)
        h_pert = h_pert * scale_factor + best_h * (1.0 - scale_factor)
        h_pert = project_box_sum(h_pert, ts)

        for beta, lam in [(300, 150), (600, 300)]:
            if tl() < 12:
                break
            try:
                h_new = _lbgfs(h_pert, n, ts, beta, lam, bnds, 200)
                h_new = project_box_sum(h_new, ts)
                if try_improve(h_new):
                    h_pert = best_h.copy()
            except Exception:
                pass

        # Quick coordinate descent
        if tl() > 15 and it % 3 == 0:
            h_cd = _coord_descent(best_h, n, ts, 500, 300, 4)
            try_improve(h_cd)

    # Phase 5: Final high-precision polish
    if tl() > 15:
        for beta, lam, iters in [(1000, 500, 500), (2000, 1000, 300)]:
            if tl() < 10:
                break
            try:
                h_new = _lbgfs(best_h, n, ts, beta, lam, bnds, iters)
                h_new = project_box_sum(h_new, ts)
                try_improve(h_new)
            except Exception:
                pass

    # Final coordinate descent polish
    if tl() > 5:
        h_cd = _coord_descent(best_h, n, ts, 1000, 500, 10)
        try_improve(h_cd)

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
