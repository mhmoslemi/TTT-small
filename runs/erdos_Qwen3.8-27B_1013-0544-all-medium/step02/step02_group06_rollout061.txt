from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def binary_c5(h_bin, n):
    # h_bin is a binary array of length n
    # Returns C5 and the correlation array
    dx = 2.0 / n
    corr = np.correlate(h_bin.astype(float), 1.0 - h_bin.astype(float), mode="full")
    return float(np.max(corr)) * dx, corr

def swap_delta(h_bin, i, j, n):
    # Compute change in corr_m for all m when swapping h[i] (0->1) and h[j] (1->0)
    # Δ_m = (h[j+m] + h[j-m]) - (h[i+m] + h[i-m])
    # where out-of-bounds are 0
    # m ranges from -(n-1) to (n-1)
    corr = np.zeros(2*n-1)
    for m in range(-(n-1), n):
        dm = 0.0
        # h[j+m]
        jm = j + m
        if 0 <= jm < n:
            dm += h_bin[jm]
        # h[j-m]
        jmm = j - m
        if 0 <= jmm < n:
            dm += h_bin[jmm]
        # h[i+m]
        im = i + m
        if 0 <= im < n:
            dm -= h_bin[im]
        # h[i-m]
        imm = i - m
        if 0 <= imm < n:
            dm -= h_bin[imm]
        corr[m + n - 1] = dm
    return corr

Similarly, the term h[i-m]*(1-h[i]) is in the sum for lag m if and only if i-m is in [max(0,-m), min(n-1,n-1-m)]. In that case, i-m+m = i is automatically in [0,n-1].

So:

  [i in valid_range(m) ? (1 - h[i+m]) : 0]   -- change from h[i]: 0→1, term j=i

+ [j-m in valid_range(m) ? h[j-m] : 0]       -- change from h[j]: 1→0, term j=j-m

where valid_range(m) = [max(0,-m), min(n-1, n-1-m)]

And h[i+m], h[i-m], h[j+m], h[j-m] are the OLD values (before the swap). Since we're only changing h[i] and h[j], and i+m, i-m, j+m, j-m could equal i or j, we need to be careful.
