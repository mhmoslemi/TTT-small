            best_c5 = c5_new
    return h, best_c5


def local_perturb_search(h0, n, target_sum, dx, time_limit, rng):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    attempt = 0
    while time.time() < time_limit:
        attempt += 1
        h_try = best_h.copy()
        scale = 0.008 * np.exp(-attempt / 40.0) + 0.002
        strategy = attempt % 5
        if strategy == 0:
            h_try += rng.randn(n) * scale
        elif strategy == 1:
            i, j = rng.choice(n, 2, replace=False)
            d = 0.03 * rng.randn()
            h_try[i] += d
            h_try[j] -= d
        elif strategy == 2:
            bl = rng.randint(3, 8)
            bs = rng.randint(0, n - bl)
            h_try[bs:bs+bl] += rng.randn(bl) * scale * 2
        elif strategy == 3:
            h_try = 0.85 * h_try + 0.15 * h_try[::-1]
            h_try += rng.randn(n) * 0.005
        else:
            corr = np.correlate(h_try, 1.0 - h_try, mode="full")
            active = np.argmax(corr)
            lag = active - (n - 1)
            for j in range(n):
                jp = j + lag
                if 0 <= jp < n:
                    h_try[j] -= 0.003 * (1.0 - h_try[jp])
                jn = j - lag
                if 0 <= jn < n:
                    h_try[j] += 0.003 * h_try[jn]
        h_try = project_h(h_try, target_sum)
        # Quick LP step from perturbed point
        h_lp, c5_lp = lp_step(h_try, n, target_sum, dx)
        if c5_lp is not None and c5_lp < best_c5:
            best_h = h_lp
            best_c5 = c5_lp
        else:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        h_parent = np.full(70, 0.5)
        n_parent = 70

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Try multiple n_points values
    n_candidates = sorted(set([n_parent,
                                max(30, n_parent - 10),
                                n_parent + 10,
                                max(30, n_parent - 20),
                                n_parent + 20]))
    n_candidates = [n for n in n_candidates if 30 <= n <= 150]

    # Allocate time: 60% to LP+SQP across n candidates, 40% to local search
    phase1_limit = start_time + 0.6 * budget_s
    phase2_limit = start_time + 0.85 * budget_s

    for n_try in n_candidates:
        if time.time() > phase1_limit:
            break
        target_sum = n_try / 2.0
        dx = 2.0 / n_try
        if n_try == n_parent:
            h_init = h_parent.copy()
        else:
            h_init = resample(h_parent, n_try)
        h_init = project_h(h_init, target_sum)
        c5_init = compute_c5(h_init)
        if c5_init < best_c5:
            best_h = h_init
            best_c5 = c5_init

        # Time per n candidate: proportional to remaining
        remaining_to_phase1 = max(0, phase1_limit - time.time())
        num_remaining = len(n_candidates) - n_candidates.index(n_try)
        t_limit = time.time() + remaining_to_phase1 / num_remaining + 30

        h_opt, c5_opt = optimize_for_n(h_init, n_try, target_sum, dx, t_limit)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 2: Local perturbation search on the best solution
    n_best = len(best_h)
    target_sum_best = n_best / 2.0
    dx_best = 2.0 / n_best
    h_final, c5_final = local_perturb_search(best_h, n_best, target_sum_best,
                                              dx_best, phase2_limit, rng)
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final

    # Final high-beta polish
    if time.time() < start_time + budget_s - 15:
        for beta in [2000, 5000]:
            if time.time() > start_time + budget_s - 5:
                break
            h_new = sqp_smooth(best_h, n_best, target_sum_best, dx_best, beta, maxiter=6000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = project_h(best_h, target_sum_best)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
