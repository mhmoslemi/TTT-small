from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-14:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff2 / free.sum()
    return h

def solve_1d_minimax(a, b):
    """Minimize max_s(a[s]*t + b[s]) over t in [0,1]."""
    # Quick check: evaluate at 0 and 1
    g0 = np.max(b)
    g1 = np.max(a + b)
    if g0 <= g1:
        # Check if g is increasing at 0
        active0 = (b >= g0 - 1e-15)
        if not (a[active0] < -1e-10).any():
            return 0.0
    if g1 <= g0:
        active1 = (a + b >= g1 - 1e-15)
        if not (a[active1] > 1e-10).any():
            return 1.0
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) * 0.333
        m2 = hi - (hi - lo) * 0.333
        if np.max(a * m1 + b) < np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coord_descent(h, n_passes=500, time_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    t0 = time.time()
    max_lag = 2 * n - 1
    s_idx = np.arange(max_lag) - (n - 1)
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    
    for p in range(n_passes):
        if time.time() - t0 > time_limit:
            break
        for j in range(n):
            js_plus = j + s_idx
            js_minus = j - s_idx
            valid_plus = (js_plus >= 0) & (js_plus < n)
            valid_minus = (js_minus >= 0) & (js_minus < n)
            
            a = np.zeros(max_lag)
            if valid_plus.any():
                a[valid_plus] += 1.0 - h[js_plus[valid_plus]]
            if valid_minus.any():
                a[valid_minus] += -h[js_minus[valid_minus]]
            
            b = corr - a * h[j]
            t_star = solve_1d_minimax(a, b)

                    corr[valid_minus] += -delta * h[js_minus[valid_minus]]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-14:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff2 / free.sum()
    return h

def solve_1d_minimax(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) < np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5
