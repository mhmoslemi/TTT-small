            # dc/dh[j]: for j in [d, n-1]: +(1-h[j-d]); for j in [0, n-1-d]: -h[j+d]
            js = np.arange(d, n)
            grad[js] += (1.0 - h[js - d])
            js2 = np.arange(0, n - d)
            grad[js2] -= h[js2 + d]
        else:

            # c_{-m} = sum_{i=m}^{n-1} h[i-m]*(1-h[i])
            # dc/dh[j]: for j in [0, n-1-m]: +(1-h[j+m]); for j in [m, n-1]: -h[j-m]
            js = np.arange(0, n - m)
            grad[js] += (1.0 - h[js + m])
            js2 = np.arange(m, n)
            grad[js2] -= h[js2 - m]
    return grad

    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(corr, beta)

For k > 0 (where k is the lag, not the correlation index):
c_k = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])

The gradient w.r.t. h[j]:
- If j appears as h[i+k] (i.e., j = i+k, so i = j-k): contributes (1-h[j-k]) when 0 <= j-k <= n-1-k, i.e., k <= j <= n-1
- If j appears as h[i] (i.e., j = i): contributes -h[j+k] when 0 <= j <= n-1-k, i.e., 0 <= j <= n-1-k

So:
dc_k/dh[j] = (1-h[j-k]) * I[k <= j <= n-1] + (-h[j+k]) * I[0 <= j <= n-1-k]

For k < 0, let k = -m where m > 0:
c_{-m} = sum_{i=m}^{n-1} h[i-m] * (1-h[i])

The gradient w.r.t. h[j]:
- If j appears as h[i-m] (i.e., j = i-m, so i = j+m): contributes (1-h[j+m]) when m <= j+m <= n-1, i.e., 0 <= j <= n-1-m
- If j appears as h[i] (i.e., j = i): contributes -h[j-m] when m <= j <= n-1, i.e., m <= j <= n-1

So:
dc_{-m}/dh[j] = (1-h[j+m]) * I[0 <= j <= n-1-m] + (-h[j-m]) * I[m <= j <= n-1]

    # Phase 4: High-precision final subgradient polish
    if tleft() > 15:
        h_cur = best_h.copy()
        step = 0.5 / n
        iters = int(tleft() * 250)
        for it in range(iters):
            if tleft() < 8:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            mx = corr.max()
            thr = mx - 0.015 * mx
            active = np.where(corr >= thr)[0]
            grad = subgrad_vec(h_cur, active, n)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            h_try = project_h(h_cur - step * grad / gn, ts)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
                h_cur = h_try
                step *= 0.9995
            else:
                step *= 0.82
                if step < 1e-10:
                    break

    best_h = project_h(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
