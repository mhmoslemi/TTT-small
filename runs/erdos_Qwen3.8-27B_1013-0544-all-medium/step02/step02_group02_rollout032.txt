```python
def targeted_move(h, n, rng):
    """Make a move that targets the most active lag."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    k_star = np.argmax(c)
    k = k_star - (n-1)  # actual shift
    
    # For this lag, we want to reduce sum h[i]*(1-h[i+k])
    # Increasing h[i] where h[i+k] is high helps
    # Decreasing h[i] where h[i+k] is low hurts
    
    # Find i where h[i] is low but h[i+k] is high (good to increase h[i])
    # and i where h[i] is high but h[i+k] is low (good to decrease h[i])
    
    valid_i = []
    for i in range(n):
        if 0 <= i+k < n:
            valid_i.append(i)
    
    if len(valid_i) < 2:
        return None, None
    
    # Score each position: positive means "increase h here helps"
    scores = np.zeros(n)
    for i in valid_i:
        scores[i] = (1-h[i+k]) - h[i]  # if positive, increasing h[i] reduces the product
    
    # Pick the best i to increase and a random j to decrease
    i_best = max(valid_i, key=lambda i: scores[i])
    j_candidates = [j for j in valid_i if j != i_best and scores[j] < 0]
    
    if not j_candidates:
        j_candidates = valid_i
    
    j_best = rng.choice(j_candidates)
    
    return i_best, j_best
```

```python
                hp = ht
        c = compute_c5(hp, n)
        if c < cur_best_c - 1e-14:
            cur_best, cur_best_c = hp.copy(), c
    return cur_best, cur_best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 80
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c5 = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: SLSQP progressive on initial n
    for alpha in [500, 5000, 50000, 500000]:
        if rem() < 10: break
        ht = slsqp(best_h, best_n, alpha, 5000, 1e-16)
        c = compute_c5(ht, best_n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 2: Try different n values
    n_candidates = []
    for nv in [60, 64, 72, 80, 96, 100, 112, 128]:
        if nv != best_n:
            n_candidates.append(nv)
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_try in n_candidates[:4]:
        if rem() < 60: break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, n_try)
        for alpha in [2000, 20000, 200000]:
            if rem() < 15: break
            ht = slsqp(h_try, n_try, alpha, 3000, 1e-15)
            c = compute_c5(ht, n_try)
            if c < compute_c5(h_try, n_try) - 1e-14:
                h_try = ht
        c = compute_c5(h_try, n_try)
        if c < best_c5 - 1e-14:
            best_h, best_c5, best_n = h_try.copy(), c, n_try

    # Phase 3: Main SA
    if rem() > 150:
        best_h, best_c5 = sa_mixed(best_h, best_n, t0, budget_s, rng,
                                    400000, 0.08, 1e-6)

    # Phase 4: SLSQP polish with very high alpha
    for alpha in [200000, 2000000, 20000000]:
        if rem() < 10: break
        ht = slsqp(best_h, best_n, alpha, 8000, 1e-18)
        c = compute_c5(ht, best_n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 5: Dithering restarts
    if rem() > 80:
        best_h, best_c5 = dither_restarts(best_h, best_n, t0, budget_s, rng,
                                           best_c5, best_h, n_restarts=8)

    # Phase 6: Second SA round (cooler)
    if rem() > 80:
        best_h, best_c5 = sa_mixed(best_h, best_n, t0, budget_s, rng,
                                    200000, 0.02, 1e-7)

    # Phase 7: Final polish
    for alpha in [10000000, 100000000]:
        if rem() < 8: break
        ht = slsqp(best_h, best_n, alpha, 10000, 1e-18)
        c = compute_c5(ht, best_n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
```
