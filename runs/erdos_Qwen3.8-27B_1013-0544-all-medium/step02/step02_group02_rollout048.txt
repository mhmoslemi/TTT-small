    # Phase 1: Multi-start L-BFGS-B
    x = np.linspace(0, 2, n)
    starts = [best_h.copy()]
    alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    starts.append(project_feasible(alt, n))
    sine1 = 0.5 + 0.35 * np.sin(np.pi * x)
    starts.append(project_feasible(sine1, n))
    sine2 = 0.5 + 0.35 * np.sin(2 * np.pi * x)
    starts.append(project_feasible(sine2, n))
    blk = np.array([1.0 if (i % 16) < 8 else 0.0 for i in range(n)])
    starts.append(project_feasible(blk, n))
    rnd = np.zeros(n)
    idx = rng.permutation(n)
    rnd[idx[:n // 2]] = 1.0
    starts.append(project_feasible(rnd, n))

    for s in starts:
        if rem() < 20:
            break
        c = compute_c5(s, n)
        if c < best_c5:
            best_h, best_c5 = s.copy(), c

    # Phase 2: L-BFGS-B with increasing alpha
    penalty = 1e6
    for alpha in [100, 500, 2000, 10000]:
        if rem() < 30:
            break
        ht = lbfgs_step(best_h, n, alpha, penalty, 500)
        c = compute_c5(ht, n)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 3: Basin hunting with L-BFGS-B
    for sigma, alpha, nit in [(0.15, 2000, 60), (0.08, 5000, 80), (0.03, 10000, 100)]:
        if rem() < 60:
            break
        best_h, best_c5 = basin_hopping(best_h, n, t0, budget_s, rng, nit, sigma, alpha, penalty, 300)

    # Phase 4: SLSQP polish
    for alpha in [5000, 50000]:
        if rem() < 20:
            break
        ht = slsqp_smooth(best_h, n, alpha, 2000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 5: Fine basin hopping
    for sigma, nit in [(0.05, 80), (0.02, 100), (0.01, 80)]:
        if rem() < 50:
            break
        best_h, best_c5 = basin_hopping(best_h, n, t0, budget_s, rng, nit, sigma, 20000, penalty, 200)

    # Phase 6: SLSQP final polish with very high alpha
    for alpha in [100000, 500000, 2000000]:
        if rem() < 15:
            break
        ht = slsqp_smooth(best_h, n, alpha, 3000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 7: Coordinate refinement
    if rem() > 20:
        dxr = 2.0 / n
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        improved = True
        while improved and rem() > 15:
            improved = False
            for base_d in steps:
                if rem() < 10:
                    break
                for i in range(n):
                    if rem() < 5:
                        break
                    for j in range(i + 1, n):
                        for sgn in [1.0, -1.0]:
                            hi = best_h[i] + sgn * base_d
                            hj = best_h[j] - sgn * base_d
                            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                                best_h[i], best_h[j] = hi, hj
                                c = compute_c5(best_h, n)
                                if c < best_c5:
                                    best_c5 = c
                                    improved = True
                                else:
                                    best_h[i], best_h[j] = hi - sgn * base_d, hj + sgn * base_d
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5(best_h, n)

    # Phase 8: Final SLSQP
    if rem() > 10:
        ht = slsqp_smooth(best_h, n, 1000000, 5000, 1e-20)
        c = compute_c5(ht, n)
        if c < best_c5:
            best_h, best_c5 = ht, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
