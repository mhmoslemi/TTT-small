        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, beta):
    """Vectorized gradient of smooth_max(corr, beta) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -500, 500))
    w /= w.sum()
    
    grad = np.zeros(n)
    for lag_idx in range(len(corr)):
        wi = w[lag_idx] * dx
        if wi < 1e-12:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            vl = n - 1 - lag
            if vl >= 0:
                grad[:vl+1] += wi * (1.0 - h[lag:])
                grad[lag:] -= wi * h[:vl+1]
        else:
            ma = -lag
            vl = n - 1 - ma
            if vl >= 0:
                grad[ma:] += wi * (1.0 - h[:vl+1])
                grad[:vl+1] -= wi * h[ma:]
    return grad

def projected_gradient_step(h, grad, target_sum, step):
    h_new = h - step * grad
    return project_h(h_new, target_sum)

def line_search(h, grad, target_sum, initial_step, dx, max_iter=30):
    """Backtracking line search."""
    f0 = compute_c5(h)
    step = initial_step
    for _ in range(max_iter):
        h_try = project_h(h - step * grad, target_sum)
        f_try = compute_c5(h_try)
        if f_try < f0 - 1e-14:
            return h_try, f_try
        step *= 0.5
    return h, f0

    num_lags = 2 * n - 1
    weights = np.ones(num_lags) / num_lags
    eta = 0.1  # learning rate for weight updates
    lr = 0.1   # learning rate for h updates
    
    cur_c5 = compute_c5(h)

            break

        hb = np.zeros(n)
        hb[np.random.choice(n, n//2, replace=False)] = 1.0
        inits.append(project_h(hb, ts))
        # Binary with structure (alternating blocks)
        for block in [4, 7, 11]:
            hb2 = np.tile([1.0]*min(block, n//2) + [0.0]*min(block, n//2), n//block + 1)[:n]
            # Adjust to have exactly n/2 ones
            n_ones = int(hb2.sum())
            if n_ones > n//2:
                ones_idx = np.where(hb2 == 1)[0]
                hb2[np.random.choice(ones_idx, n_ones - n//2, replace=False)] = 0
            elif n_ones < n//2:
                zeros_idx = np.where(hb2 == 0)[0]
                hb2[np.random.choice(zeros_idx, n//2 - n_ones, replace=False)] = 1
            inits.append(hb2.copy())
        
        for h0 in inits:
            if rem() < 15:
                break
            
            h = h0.copy()
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n
            
            # Phase: Min-max with exponential weights
            w = np.ones(num_lags) / num_lags
            eta0 = 0.2
            lr0 = 0.3
            total_iters = 3000
            stagnation = 0
            
            for it in range(total_iters):
                if rem() < 5:
                    break
                
                eta = eta0 / (1.0 + 0.001 * it)
                lr = lr0 / (1.0 + 0.005 * it)
                
                # Compute overlaps
                corr = np.correlate(h, 1.0 - h, mode="full")
                overlaps = corr * dx
                
                # Update weights
                w *= np.exp(overlaps / eta)
                w /= w.sum()
                
                # Gradient of weighted sum
                g = grad_weighted(h, w, dx)
                gn = np.linalg.norm(g)
                if gn < 1e-14:
                    break
                
                # Projected gradient step
                h_new = project_h(h - lr * g / gn, ts)
                c5_new = compute_c5(h_new)
                
                if c5_new < c5 - 1e-14:
                    h = h_new
                    c5 = c5_new
                    stagnation = 0
                    if c5 < best_c5:
                        best_h, best_c5, best_n = h.copy(), c5, n
                else:
                    stagnation += 1
                    if stagnation > 200:
                        # Kick
                        h = project_h(h + 0.2 * np.random.randn(n), ts)
                        c5 = compute_c5(h)
                        w = np.ones(num_lags) / num_lags
                        if c5 < best_c5:
                            best_h, best_c5, best_n = h.copy(), c5, n
                        stagnation = 0
            
            # SLSQP polish
            if rem() > 10:
                bounds = Bounds(np.zeros(n), np.ones(n))
                def _obj(x):
                    c = np.correlate(x, 1.0 - x, mode="full")
                    return smooth_max(c, 400.0) * dx
                def _con(x):
                    return x.sum() - ts
                def _jac(x):
                    return np.ones(n)
                try:
                    res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                                   constraints=[{'type':'eq','fun':_con,'jac':_jac}],
                                   options={'maxiter':3000, 'ftol':1e-15})
                    hs = project_h(res.x, ts)
                    c5s = compute_c5(hs)
                    if c5s < best_c5:
                        best_h, best_c5, best_n = hs.copy(), c5s, n
                except:
                    pass
    
    # Final fine-tuning
    if rem() > 8:
        nf = len(best_h)
        tsf = nf / 2.0
        dxf = 2.0 / nf
        w = np.ones(2*nf-1) / (2*nf-1)
        
        for it in range(5000):
            if rem() < 3:
                break
            eta = 0.05 / (1 + 0.002*it)
            lr = 0.05 / (1 + 0.005*it)
            
            corr = np.correlate(best_h, 1.0-best_h, mode="full")
            ov = corr * dxf
            w *= np.exp(ov / eta)
            w /= w.sum()
            
            g = grad_weighted(best_h, w, dxf)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            h_try = project_h(best_h - lr * g / gn, tsf)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
    
    return best_h, best_c5, best_n

This is a fundamentally different approach from the previous code. The key difference is using the exponential weights / online gradient method for the min-max problem instead of smooth max + SLSQP.

Let me review potential issues:
1. The gradient computation `grad_weighted` loops over all lags (O(n)), each doing O(n) work → O(n²) total. For n=240, that's ~115K operations, which should be fast.
2. The projection `project_h` does up to 200 iterations of clip-and-adjust. This is O(n) per iteration, so O(n) total. Fine.
3. Each main loop iteration: O(n²) for gradient + O(n) for projection + O(n²) for correlation (in compute_c5). So O(n²) per iteration.

Let me also think about whether the exponential weights approach is theoretically sound here.

The problem is: min_h max_k f_k(h)
The "follow the leader" / "exponential weights" approach gives us:
min_h min_w max_k (f_k(h) + (1/eta) log sum_k w_k exp(eta * f_k(h)))

Hmm, that's not quite right. Let me think again.

Actually, the standard approach for min-max is:
min_h max_k f_k(h) = min_h min_{p in simplex} max_k (f_k(h) / p_k)

No, that's not right either. The correct duality is:
min_h max_k f_k(h) = min_h max_{p in simplex} sum_k p_k f_k(h)

By Sion's minimax theorem (if the problem is convex-concave, which it's not here), we can swap:
= max_{p in simplex} min_h sum_k p_k f_k(h)

But since the problem isn't convex, we can't guarantee the swap. However, the "exponential weights" heuristic is:
- Maintain p_t
- Minimize sum_k p_k(t) f_k(h) → get h_t
- Update p_t(k) ∝ p_{t-1}(k) * exp(f_k(h_t) / eta)

This is a heuristic, not a guaranteed algorithm, but it often works well in practice.

The convergence rate for the "follow the leader" regret is O(sqrt(T)), so after T iterations, the weighted average of the max f_k is within O(1/sqrt(T)) of the optimal. But we don't get a guarantee on the actual max f_k at the final iterate.

In practice, the approach works well because:
1. The weights quickly concentrate on the few worst lags
2. Optimizing for those few lags tends to reduce all lags
3. The h dynamics (gradient descent) are stable

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_weighted(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for li in range(len(w)):
        wi = w[li] * dx
        if wi < 1e-14:
            continue
        lag = li - (n - 1)
        if lag >= 0:
            vl = n - 1 - lag
            if vl >= 0:
                grad[:vl+1] += wi * (1.0 - h[lag:])
                grad[lag:] -= wi * h[:vl+1]
        else:
            ma = -lag
            vl = n - 1 - ma
            if vl >= 0:
                grad[ma:] += wi * (1.0 - h[:vl+1])
                grad[:vl+1] -= wi * h[ma:]
    return grad


def minmax_exp_weights(h, n, ts, dx, max_iters, t_fn, kick_interval=500):
    num_lags = 2 * n - 1
    w = np.ones(num_lags) / num_lags
    h = h.copy()
    c5 = compute_c5(h)
    best_local = (h.copy(), c5)
    stagnation = 0

    for it in range(max_iters):
        if t_fn() < 5:
            break
        eta = 0.15 / (1.0 + 0.002 * it)
        lr = 0.25 / (1.0 + 0.008 * it)

        corr = np.correlate(h, 1.0 - h, mode="full")
        ov = corr * dx
        w *= np.exp(ov / eta)
        w /= w.sum()

        g = grad_weighted(h, w, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = project_h(h - lr * g / gn, ts)
        c5_new = compute_c5(h_new)

        if c5_new < c5 - 1e-14:
            h = h_new
            c5 = c5_new
            stagnation = 0
            if c5 < best_local[1]:
                best_local = (h.copy(), c5)
        else:
            stagnation += 1
            if stagnation >= kick_interval:
                h = project_h(h + 0.15 * np.random.randn(n), ts)
                c5 = compute_c5(h)
                w = np.ones(num_lags) / num_lags
                stagnation = 0
    return best_local[0], best_local[1]


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 140
        h_parent = np.full(n_parent, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    sizes = sorted(set([n_parent, 100, 120, 140, 160, 200, 240]))

    for n in sizes:
        if rem() < 50:
            break
        ts = n / 2.0
        dx = 2.0 / n

        inits = []
        x_old = np.linspace(0, 2, n_parent)
        x_new = np.linspace(0, 2, n)
        inits.append(project_h(np.interp(x_new, x_old, h_parent), ts))
        if len(best_h) != n:
            xb = np.linspace(0, 2, len(best_h))
            inits.append(project_h(np.interp(x_new, xb, best_h), ts))

        hb = np.zeros(n)
        hb[np.random.choice(n, n // 2, replace=False)] = 1.0
        inits.append(hb.copy())

        for block in [5, 8, 13]:
            period = block * 2
            pat = np.zeros(period)
            pat[:block] = 1.0
            filled = np.tile(pat, n // period + 1)[:n]
            n_ones = int(filled.sum())
            if n_ones > n // 2:
                oi = np.where(filled == 1)[0]
                filled[np.random.choice(oi, n_ones - n // 2, replace=False)] = 0
            elif n_ones < n // 2:
                zi = np.where(filled == 0)[0]
                filled[np.random.choice(zi, n // 2 - n_ones, replace=False)] = 1
            inits.append(filled.copy())

        for h0 in inits:
            if rem() < 15:

            h_init = h0.copy()
            c5_init = compute_c5(h_init)

                    return smooth_max(c, 500.0) * _dx

                    return x.sum() - _ts

                    return np.ones(_n)

                except Exception:
                    pass

    # Final refinement with high-precision OGR
    if rem() > 8:
        nf = len(best_h)
        tsf = nf / 2.0
        dxf = 2.0 / nf
        _, c5_ref = minmax_exp_weights(best_h, nf, tsf, dxf, 5000, rem, kick_interval=800)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h, best_n = best_h.copy(), nf

        # SLSQP final polish
        if rem() > 5:
            bounds_f = Bounds(np.zeros(nf), np.ones(nf))

            def _fobj(x, _dx=dxf):
                c = np.correlate(x, 1.0 - x, mode="full")
                return smooth_max(c, 1000.0) * _dx

            def _fcon(x, _ts=tsf):
                return x.sum() - _ts

            def _fjac(x, _nf=nf):
                return np.ones(_nf)

            try:
                res = minimize(_fobj, best_h, method='SLSQP', bounds=bounds_f,
                               constraints=[{'type': 'eq', 'fun': _fcon, 'jac': _fjac}],
                               options={'maxiter': 5000, 'ftol': 1e-16})
                hs = project_h(res.x, tsf)
                c5s = compute_c5(hs)
                if c5s < best_c5:
                    best_h, best_c5 = hs.copy(), c5s
            except Exception:
                pass

    return best_h, best_c5, best_n
