    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    h = np.clip(h, 0.0, 1.0)

        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)

            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    #   (1-h[j+k]) if j+k is valid (i.e., 0<=j+k<n)
    #   -h[j-k] if j-k is valid (i.e., 0<=j-k<n)

    lo, hi = 0.0, 1.0

        elif sg < 0:

        else:

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):

def grad_c5(h, n_active=3):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for idx in top_idx:
        k = int(idx) - (n - 1)
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
        else:

            grad[ka:] += (1.0 - h[:n - ka])

        g = g / gn
        mom = mu * mom + g
        mn = np.linalg.norm(mom)
        if mn < 1e-12:
            break
        d = -mom / mn
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002]:
            ht = project_h(h + step * d, target)
            ct = compute_c5(ht)
            if ct < h_c5 - 1e-15:
                h, h_c5 = ht, ct
                improved = True
                break
        if not improved and it % 80 == 0:
            h = project_h(h + 0.003 * np.random.randn(n), target)
            h_c5 = compute_c5(h)
    return h, h_c5

_p = {'beta': 1.0, 'dx': 1.0, 'tgt': 0.0}

def _obj(x):
    return smooth_max(compute_corr(x), _p['beta']) * _p['dx']

def _con(x):
    return np.sum(x) - _p['tgt']

def _jac(x):
    return np.ones(x.shape)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    else:
        n0 = 140
        hp = np.full(n0, 0.5)

    best_h = hp.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    n_opts = sorted(set([n0] + [n0 * m for m in range(1, 10) if 80 <= n0 * m <= 560]))

    for n in n_opts:
        if budget_s - (time.time() - t0) < 50:
            break
        tgt = n / 2.0
        dx = 2.0 / n
        off = n // 2

        if n == n0:
            cur = project_h(hp.copy(), tgt)
        else:
            xo = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, n)
            cur = project_h(np.interp(xn, xo, hp), tgt)

        if len(best_h) != n:
            xb = np.linspace(0, 2, len(best_h))
            xn2 = np.linspace(0, 2, n)
            c2 = project_h(np.interp(xn2, xb, best_h), tgt)
            if compute_c5(c2) < compute_c5(cur):
                cur = c2

        c5c = compute_c5(cur)
        if c5c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c5c, n

        # SLSQP phases
        for beta in [10, 30, 80, 200, 500, 1000]:
            if budget_s - (time.time() - t0) < 8:
                break
            _p['beta'] = beta
            _p['dx'] = dx
            _p['tgt'] = tgt
            try:
                res = minimize(_obj, cur, method='SLSQP',
                               bounds=Bounds(np.zeros(n), np.ones(n)),
                               constraints=[{'type': 'eq', 'fun': _con, 'jac': _jac}],
                               options={'maxiter': 5000, 'ftol': 1e-14})
                hn = project_h(res.x, tgt)
                cn = compute_c5(hn)
                if cn < c5c - 1e-14:
                    cur, c5c = hn.copy(), cn
                if c5c < best_c5:
                    best_h, best_c5, best_n = cur.copy(), c5c, n
            except Exception:
                pass

        # Pair coordinate descent (main local optimizer)
        cur, c5c = pair_cd(cur, max_sweeps=500, offset=off)
        if c5c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c5c, n

        # Subgradient to escape potential plateaus
        cur, c5c = subgrad_loop(cur, tgt, 2000, t0, budget_s, n_act=3)
        if c5c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c5c, n

        # Another round of pair CD from new point
        cur, c5c = pair_cd(cur, max_sweeps=500, offset=off)
        if c5c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c5c, n

        # Try different offset
        if budget_s - (time.time() - t0) > 30 and n > 4:
            off2 = n // 3
            cur2, c5c2 = pair_cd(cur.copy(), max_sweeps=300, offset=off2)
            if c5c2 < c5c:
                cur, c5c = cur2, c5c2
                if c5c < best_c5:
                    best_h, best_c5, best_n = cur.copy(), c5c, n

        # Binary projection + polish
        if budget_s - (time.time() - t0) > 20:
            for th in np.linspace(0.35, 0.65, 7):
                hb = (cur > th).astype(float)
                hb = project_h(hb, tgt)
                cb = compute_c5(hb)
                if cb < c5c:
                    cur, c5c = hb.copy(), cb
                    if c5c < best_c5:
                        best_h, best_c5, best_n = cur.copy(), c5c, n
                    cur, c5c = pair_cd(cur, max_sweeps=300, offset=off)
                    if c5c < best_c5:
                        best_h, best_c5, best_n = cur.copy(), c5c, n

    # Final high-precision polish
    if budget_s - (time.time() - t0) > 10:
        nf = len(best_h)
        tf = nf / 2.0
        df = 2.0 / nf
        _p['beta'] = 3000.0
        _p['dx'] = df
        _p['tgt'] = tf
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=Bounds(np.zeros(nf), np.ones(nf)),
                           constraints=[{'type': 'eq', 'fun': _con, 'jac': _jac}],
                           options={'maxiter': 20000, 'ftol': 1e-15})
            hn = project_h(res.x, tf)
            cn = compute_c5(hn)
            if cn < best_c5:
                best_h, best_c5 = hn.copy(), cn
        except Exception:
            pass

        # Final pair CD polish
        best_h, best_c5 = pair_cd(best_h, max_sweeps=1000, offset=nf // 2)

        # Final subgradient
        best_h, best_c5 = subgrad_loop(best_h, tf, 5000, t0, budget_s, n_act=2)

        # One more pair CD
        best_h, best_c5 = pair_cd(best_h, max_sweeps=500, offset=nf // 2)

    return best_h, best_c5, best_n
