import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.interpolate import interp1d
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    """Compute smooth max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    shifted = corr - m
    # Numerical stability
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12: continue
        s = k - (n - 1)
        js = j - s; jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any(): grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any(): grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_solve(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = LinearConstraint(np.ones(n), target, target)
    
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h, target)

def subgradient_opt(h, n_iter, lr0, decay, top_k, momentum=True):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    v = x.copy()  # velocity for momentum
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    
    for t in range(n_iter):
        lr = lr0 * (decay ** t)
        corr = compute_corr_full(x)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n-1)
            js = j - s; jp = j + s
            ms = (js>=0)&(js<n); mp = (jp>=0)&(jp<n)
            if ms.any(): grad[ms] += (1.0 - x[js[ms]])
            if mp.any(): grad[mp] += (-x[jp[mp]])
        grad = grad / top_k * dx
        
        if momentum:
            v = 0.9 * v - lr * grad
        else:
            v = -lr * grad
        
        x_new = project_feasible(x + v, target)
        c = compute_c5(x_new)
        if c < best_v:
            best_v = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.interpolate import interp1d
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad


def slsqp_solve(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = LinearConstraint(np.ones(n), target, target)

    def obj(x):
        return smoothmax_val_grad(x, beta)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta)[1]

    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def trust_constr_solve(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = LinearConstraint(np.ones(n), target, target)

    def obj(x):
        return smoothmax_val_grad(x, beta)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta)[1]

    try:
        res = minimize(obj, h, jac=objg, method='trust-constr',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'gtol': 1e-9,
                                'xtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def subgradient_opt(h, n_iter, lr0, decay, top_k, mom=0.9):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(n_iter):
        lr = lr0 * (decay ** t)
        corr = compute_corr_full(x)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - x[js[ms]])
            if mp.any():
                grad[mp] += (-x[jp[mp]])
        grad = grad / top_k * dx
        v = mom * v - lr * grad
        x_new = project_feasible(x + v, target)
        c = compute_c5(x_new)
        if c < best_v:
            best_v = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def fourier_init(n, K, seed=0):
    """Generate initial h from a low-frequency Fourier ansatz."""
    np.random.seed(seed)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    a = np.zeros(n)
    for k in range(1, K + 1):
        ak = 0.35 * np.random.randn()
        bk = 0.35 * np.random.randn()
        a += ak * np.cos(np.pi * k * x) + bk * np.sin(np.pi * k * x)
    h = 0.5 + a
    return project_feasible(h, n / 2.0)


def fourier_refine(h0, n, K, maxiter=300):
    """Refine h using Fourier parameterization with L-BFGS-B."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    # Extract Fourier coefficients from h0
    h0c = h0 - 0.5
    params = np.zeros(2 * K)
    for k in range(1, K + 1):
        cos_k = np.cos(np.pi * k * x)
        sin_k = np.sin(np.pi * k * x)
        params[2 * (k - 1)] = (2.0 / n) * np.dot(h0c, cos_k)
        params[2 * (k - 1) + 1] = (2.0 / n) * np.dot(h0c, sin_k)

    def obj(p):
        a = np.zeros(n)
        for k in range(1, K + 1):
            a += p[2 * (k - 1)] * np.cos(np.pi * k * x)
            a += p[2 * (k - 1) + 1] * np.sin(np.pi * k * x)
        h = project_feasible(np.clip(0.5 + a, 0.0, 1.0), n / 2.0)
        return compute_c5(h)

    bounds = [(-0.48, 0.48)] * (2 * K)
    try:
        res = minimize(obj, params, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        a = np.zeros(n)
        for k in range(1, K + 1):
            a += res.x[2 * (k - 1)] * np.cos(np.pi * k * x)
            a += res.x[2 * (k - 1) + 1] * np.sin(np.pi * k * x)
        h = project_feasible(np.clip(0.5 + a, 0.0, 1.0), n / 2.0)
        return h
    except Exception:
        return h0


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:

    # Phase 1: Fourier refinement at current resolution (fast, low-dim)

        h2 = slsqp_solve(h_best, maxiter=400, beta=beta)

        if c2 < best_c5:

    # Phase 4: Trust-constr attempt

        n_cur = len(h_best)

        if tl() < 50:

        h_pert = h_best + 0.015 * np.random.randn(n)

            h_best, best_c5 = h_final, c_final

    return h_best, best_c5, len(h_best)
