from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, dx, beta):
    """Objective and gradient of smooth_max(corr*dz, beta)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    obj_val = smooth_max(corr, beta)
    
    # Gradient weights
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)

        w = weights[k]
        if w < 1e-18:
            continue

            grad[idx] += w * (1.0 - h[idx + lag])

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_and_grad(h, dx, beta):
    n = len(h)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    obj_val = m + np.log(np.sum(exps)) / beta

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    grad *= dx
    return obj_val, grad


def con_fun(h, ts):
    return h.sum() - ts


def con_jac(h, ts):
    return np.ones(len(h))


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    choice = seed % 6
    if choice == 0:
        a = rng.uniform(4, 12)
        c = rng.uniform(0.5, 1.5)
        h = 1.0 / (1.0 + np.exp(-a * (x - c)))
    elif choice == 1:
        A = rng.uniform(0.3, 0.9)
        phi = rng.uniform(0, 2 * np.pi)
        h = 0.5 + A * np.cos(np.pi * x + phi)
    elif choice == 2:
        h = np.zeros(n)
        c1 = rng.uniform(0.2, 0.8)
        c2 = rng.uniform(1.2, 1.8)
        w = rng.uniform(0.15, 0.4)
        h += np.exp(-((x - c1) / w) ** 2)
        h += np.exp(-((x - c2) / w) ** 2)
    elif choice == 3:
        h = rng.uniform(0.2, 0.8, n)
        kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
        h = np.convolve(h, kernel, mode='same')
    elif choice == 4:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    else:
        h = np.full(n, 0.5)
        h[:n // 3] = 0.8
        h[2 * n // 3:] = 0.8
    h = np.clip(h, 0, 1)
    if h.sum() > 0 and h.sum() < n:
        h *= (n / 2.0) / h.sum()
        h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)


def projected_gradient_phase(h, dx, target_sum, beta, max_iter, time_limit_fn):
    n = len(h)
    lr = 0.05
    for it in range(max_iter):
        if time_limit_fn() < 10:
            break
        obj_val, grad = obj_and_grad(h, dx, beta)
        for _ in range(30):
            h_trial = h - lr * grad
            h_trial = project_feasible(h_trial, target_sum)
            o, _ = obj_and_grad(h_trial, dx, beta)
            if o < obj_val - 1e-14:
                h = h_trial
                break
            lr *= 0.5
        lr = min(lr * 1.1, 0.1)
    return h

    t0 = time.time()

        return budget_s - (time.time() - t0)

                    best_c5 = c5
            except Exception:

        if c5 < best_c5:

            best_c5 = c5

                        best_c5 = c5

                               bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-17, 'iprint': 0})
                h_new = project_feasible(res.x, target_sum)

            pen = 100.0 * (hc.sum() - target_sum) ** 2
            corr = np.correlate(hc, 1.0 - hc, mode="full")
            return smooth_max(corr, 5000) * dx + pen

                best_c5 = c5

            pass
