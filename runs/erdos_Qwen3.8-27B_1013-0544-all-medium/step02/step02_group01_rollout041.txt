import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Numerical stability
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    j = np.arange(n)
    # Only compute for significant weights
    significant = w > 1e-10
    for k in np.where(significant)[0]:
        wk = w[k]
        s = int(k - (n - 1))
        if s >= 0:
            idx = np.arange(0, n - s)
            grad[idx] += wk * (1.0 - h[idx + s])
            grad[idx + s] += wk * (-h[idx])
        else:
            s2 = -s
            idx = np.arange(0, n - s2)
            grad[idx] += wk * (1.0 - h[idx + s2])
            grad[idx + s2] += wk * (-h[idx])
    grad *= dx
    return val, grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_smoothmax(h, beta, maxiter=5000, tol=1e-12):
    """Optimize smooth max using L-BFGS-B with penalty for sum constraint."""
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        # Add penalty for sum constraint
        s = x.sum() - target
        v += 100.0 * s * s
        return v
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        s = x.sum() - target
        g += 200.0 * s * np.ones(n)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol})
    return project_feasible(res.x, target)

def interpolate_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

        # Add small perturbation

        h_pert = h_best + np.random.randn(n) * eps

        # Quick optimization

        c = compute_c5(h_opt)
        if c < best_c5:

    # Stage 3: Try different resolutions

        if time_left() < 120:

        if n_try == n:

        h_interp = interpolate_to_n(h_best, n_try)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    significant = np.where(w > 1e-15)[0]
    for k in significant:
        wk = w[k]
        s = int(k - (n - 1))
        if s >= 0:
            idx = np.arange(n - s)
            grad[idx] += wk * (1.0 - h[idx + s])
            grad[idx + s] -= wk * h[idx]
        else:
            s2 = -s
            idx = np.arange(n - s2)
            grad[idx] += wk * (1.0 - h[idx + s2])
            grad[idx + s2] -= wk * h[idx]
    grad *= dx
    return val, grad

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            free = (h > 1e-12) & (h < 1.0 - 1e-12)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def opt_smoothmax(h, beta, maxiter=5000, penalty=1000.0):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        s = x.sum() - target
        return v + 0.5 * penalty * s * s
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        s = x.sum() - target
        return g + penalty * s
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def tl():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(100, 0.5)
    
    best_c5 = compute_c5(h_best)
    n = len(h_best)
    
    # Stage 1: Progressive beta increase at current resolution
    betas = [50, 100, 200, 400, 800, 1600, 3200]
    for beta in betas:
        if tl() < 80:
            break
        h = opt_smoothmax(h_best, beta, maxiter=5000)
        c = compute_c5(h)
        if c < best_c5:
            h_best, best_c5 = h, c
    
    # Stage 2: Random perturbations
    for attempt in range(20):
        if tl() < 60:
            break
        eps = 0.02 * (1.0 - attempt / 25.0)
        h_pert = h_best + np.random.randn(n) * eps
        h_pert = project_feasible(h_pert, n / 2.0)
        h = opt_smoothmax(h_pert, beta=1000, maxiter=3000)
        c = compute_c5(h)
        if c < best_c5:
            h_best, best_c5 = h, c
    
    # Stage 3: Try different resolutions
    for n_try in [100, 60, 150, 80]:
        if tl() < 100:
            break
        if n_try == n:
            continue
        h_sub = interp_to_n(h_best, n_try)
        for beta in [200, 500, 1000]:
            if tl() < 50:
                break
            h = opt_smoothmax(h_sub, beta, maxiter=4000)
            c = compute_c5(h)
            if c < best_c5:
                h_best, best_c5 = h, c
                h_sub = h.copy()
    
    # Stage 4: Final ultra-high-beta polish
    if tl() > 30:
        h = opt_smoothmax(h_best, beta=5000, maxiter=8000)
        c = compute_c5(h)
        if c < best_c5:
            h_best, best_c5 = h, c
    
    # Stage 5: One more perturbation + polish
    if tl() > 50:
        for _ in range(5):
            if tl() < 30:
                break
            eps = 0.005
            h_pert = h_best + np.random.randn(len(h_best)) * eps
            h_pert = project_feasible(h_pert, len(h_best) / 2.0)
            h = opt_smoothmax(h_pert, beta=3000, maxiter=5000)
            c = compute_c5(h)
            if c < best_c5:
                h_best, best_c5 = h, c
    
    return h_best, best_c5, len(h_best)

                    h_bin[idx_zero[j]] = 1.0
        h_bin = project_feasible(h_bin, n / 2.0)
        for beta in [200, 600, 1500]:
            if tl() < 40:
                break
            h = opt_smoothmax(h_bin, beta, maxiter=3000)
            c = compute_c5(h)
            if c < best_c5:
                h_best, best_c5 = h, c
                h_bin = h.copy()

    # Stage 4: Different resolutions
    for n_try in [100, 60, 150, 80, 120]:
        if tl() < 80:
            break
        if n_try == n:
            continue
        h_sub = interp_to_n(h_best, n_try)
        for beta in [300, 800, 2000]:
            if tl() < 40:
                break
            h = opt_smoothmax(h_sub, beta, maxiter=3000)
            c = compute_c5(h)
            if c < best_c5:
                h_best, best_c5 = h, c
                h_sub = h.copy()
                n = len(h_best)

    # Stage 5: Final ultra-high-beta polish
    if tl() > 30:
        h = opt_smoothmax(h_best, 5000, maxiter=5000)
        c = compute_c5(h)
        if c < best_c5:
            h_best, best_c5 = h, c
            n = len(h_best)

    # Stage 6: Fine perturbations near the optimum
    for i in range(10):
        if tl() < 25:
            break
        eps = 0.003 * (1.0 + i * 0.5)
        h_p = h_best + np.random.randn(n) * eps
        h_p = project_feasible(h_p, n / 2.0)
        h = opt_smoothmax(h_p, 3000, maxiter=3000)
        c = compute_c5(h)
        if c < best_c5:
            h_best, best_c5 = h, c
            n = len(h_best)

    return h_best, best_c5, len(h_best)
