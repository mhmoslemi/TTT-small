def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def compute_slopes(h, j, jp):
    """Compute slope vector for pair (j, jp) in the pair CD."""
    n = len(h)
    sl = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        fj = 0.0
        if 0 <= j - k < n:
            fj += 1.0 - h[j - k]
        if 0 <= j + k < n:
            fj -= h[j + k]
        fjp = 0.0
        if 0 <= jp - k < n:
            fjp += 1.0 - h[jp - k]
        if 0 <= jp + k < n:
            fjp -= h[jp + k]
        sl[m] = fj - fjp
    return sl

def pair_cd_pass(h, corr, rng, n_shuffle=3):
    """One pass of pair coordinate descent."""
    n = len(h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for _ in range(n_shuffle):
        rng.shuffle(pairs)
        for j, jp in pairs:
            s = h[j] + h[jp]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-14:
                continue
            
            sl = compute_slopes(h, j, jp)
            hj = h[j]
            intercepts = corr - sl * hj
            
            # Ternary search (works well in practice despite k=0 issue)
            a, b = t_lo, t_hi
            for _ in range(50):

                t2 = b - (b - a) / 3.0
                f1 = np.max(sl * t1 + intercepts)

                else:

            h[jp] = s - t_opt
            corr = intercepts + sl * t_opt
