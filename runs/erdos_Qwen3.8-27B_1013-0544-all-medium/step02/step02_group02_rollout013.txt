from scipy.optimize import minimize

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

    h = h.copy()
    h = np.clip(h, 0.0, 1.0)

        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:

            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h

def obj_smooth(h, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)

    return np.sum(h) - n / 2.0

    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})

    dx = 2.0 / n

    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step in steps:

                return h, best_c5
            improved = False

                        improved = True
                    else:

            if not improved:
                break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, t0, budget_s, n_passes=4):
    best_c5 = compute_c5(h, n)
    steps = [0.01, 0.004, 0.002, 0.001, 0.0004]
    for step in steps:
        for _ in range(n_passes):
            if budget_s - (time.time() - t0) < 15:
                return h, best_c5
            improved = False
            for i in range(n):
                if budget_s - (time.time() - t0) < 15:
                    return h, best_c5
                hi = h[i]
                best_move = None
                best_c = best_c5
                for s in [1.0, -1.0]:
                    new_hi = hi + s * step
                    if new_hi < -1e-10 or new_hi > 1.0 + 1e-10:
                        continue
                    new_hi = np.clip(new_hi, 0.0, 1.0)
                    for j in range(n):
                        if j == i:
                            continue
                        new_hj = h[j] - s * step
                        if new_hj < -1e-10 or new_hj > 1.0 + 1e-10:
                            continue
                        new_hj = np.clip(new_hj, 0.0, 1.0)
                        h[i] = new_hi
                        h[j] = new_hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                            best_move = (j, s)
                        h[i] = hi
                        h[j] = h[j] + s * step
                if best_move is not None:
                    j, s = best_move
                    h[i] = np.clip(hi + s * step, 0.0, 1.0)
                    h[j] = np.clip(h[j] - s * step, 0.0, 1.0)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break
    return h, best_c5

def sa_block_swap(h, n, t0, budget_s, rng, n_iter, e_start, e_min):
    E = e_start
    e_ratio = (e_min / e_start) ** (1.0 / n_iter)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5
    last_polish = time.time()
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 30:
            break
        if time.time() - last_polish > 30:
            last_polish = time.time()
            h = best_h.copy()
            current_c5 = best_c5
            h = slsqp_optimize(h, n, 10000, 400, 1e-14)
            c = compute_c5(h, n)
            if c < best_c5:
                best_h = h.copy()
                best_c5 = c
                current_c5 = c
            if budget_s - (time.time() - t0) < 30:
                break
        L = rng.randint(2, 9)
        max_start = n - L
        if max_start <= 0:
            E *= e_ratio
            continue
        s1 = rng.randint(0, max_start + 1)
        s2 = rng.randint(0, max_start + 1)
        if s1 == s2 or (s1 < s2 and s1 + L > s2):
            E *= e_ratio
            continue
        b1 = h[s1:s1+L].copy()
        b2 = h[s2:s2+L].copy()
        h[s1:s1+L] = b2
        h[s2:s2+L] = b1
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        if delta < 0 or (delta < E and rng.random() < np.exp(-delta / max(E, 1e-10))):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[s1:s1+L] = b1
            h[s2:s2+L] = b2
        E *= e_ratio
    return best_h, best_c5

def sa_2pt(h, n, t0, budget_s, rng, n_iter, e_start, e_min, d_start, d_min):
    E = e_start
    D = d_start
    e_ratio = (e_min / e_start) ** (1.0 / n_iter)
    d_ratio = (d_min / d_start) ** (1.0 / n_iter)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5
    last_polish = time.time()
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 30:
            break
        if time.time() - last_polish > 25:
            last_polish = time.time()
            h = best_h.copy()
            current_c5 = best_c5
            h = slsqp_optimize(h, n, 10000, 300, 1e-14)
            c = compute_c5(h, n)
            if c < best_c5:
                best_h = h.copy()
                best_c5 = c
                current_c5 = c
            if budget_s - (time.time() - t0) < 30:
                break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.001, D) * rng.choice([1.0, -1.0])
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i] = hi
        h[j] = hj
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        if delta < 0 or (delta < E and rng.random() < np.exp(-delta / max(E, 1e-10))):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = hi - d
            h[j] = hj + d
        E *= e_ratio
        D *= d_ratio
    return best_h, best_c5

def fourier_perturb(h, n, rng, scale):
    x = np.linspace(0, 2, n)
    pert = np.zeros(n)
    for m in range(1, 13):
        a = rng.normal(0, scale / m)
        b = rng.normal(0, scale / m)
        pert += a * np.cos(m * np.pi * x) + b * np.sin(m * np.pi * x)
    return project_feasible(h + pert, n)
