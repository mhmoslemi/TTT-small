import time

def compute_c5(h):

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    vals = beta * (corr - m)

    if max_val > 500:
        vals -= max_val
        m += max_val / beta
    exp_vals = np.exp(vals)
    sum_exp = np.sum(exp_vals)
    obj = (m + np.log(sum_exp) / beta) * dx

    # Gradient of smooth max
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        for k in range(j + 1):
            s1 += w[k] * (1.0 - h[j - k])
        s2 = 0.0
        for k in range(n - j):
            s2 += w[k] * h[j + k]
        grad[j] = dx * (s1 - s2)
    
    # Penalty gradient
    grad += 2.0 * mu * s
    
    return obj, grad


class Optimizer:
    def __init__(self, beta, dx, target_sum, mu):
        self.beta = beta
        self.dx = dx
        self.target_sum = target_sum
        self.mu = mu
    
    def f(self, h):
        return smooth_max_and_grad(h, self.beta, self.dx, self.target_sum, self.mu)[0]
    
    def g(self, h):
        return smooth_max_and_grad(h, self.beta, self.dx, self.target_sum, self.mu)[1]


def pair_swap_refine(h, target_sum, n_iter, rng):
    """Refine using pair swaps."""
    n = len(h)
    best_c5 = compute_c5(h)
    for _ in range(n_iter):
        i, j = rng.choice(n, 2, replace=False)
        # Try several step sizes
        for d in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.005, -0.005]:
            if h[i] + d < 0 or h[i] + d > 1:
                continue
            if h[j] - d < 0 or h[j] - d > 1:
                continue
            h_trial = h.copy()
            h_trial[i] += d
            h_trial[j] -= d
            c5 = compute_c5(h_trial)
            if c5 < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c5
                break
    return h, best_c5
