    return float(np.max(corr) * dx)

Now let me put it all together. I also want to add a "pairwise trade" local search as a final polish:

For pairs (i, j), try transferring ε from h[j] to h[i] (or vice versa). Keep the change if it reduces C₅. This is like a coordinate descent but with two coordinates at a time.

For efficiency, I won't check all O(n²) pairs. Instead, I'll check pairs where the two coordinates are "far apart" in the correlation structure (i.e., they appear in different lags' computations).

Actually, for n=70, checking all pairs (4900) with a quick C₅ evaluation (O(n) each) is only 4900*70 ≈ 343K operations per full pass. That's very fast. Let me include it.

But to be efficient, I'll use a "greedy" approach: scan pairs in random order, apply the first improving move, then restart the scan. This is like a stochastic coordinate descent.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_fun(x, n, beta, lam):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    ev = np.exp(beta * (corr - m))
    soft = m + np.log(ev.sum()) / beta
    return soft * (2.0 / n) + lam * (x.sum() - n / 2.0) ** 2

def _c5_jac(x, n, beta, lam):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    ev = np.exp(beta * (corr - m))
    sw = ev / ev.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = sw[k_idx]
        if w < 1e-15:
            continue
        L = k_idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - x[L:])
            grad[L:] -= w * x[:n - L]
        else:
            Lp = -L
            grad[Lp:] += w * (1.0 - x[:n - Lp])
            grad[:n - Lp] -= w * x[Lp:]
    grad *= dx
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def subgradient_polish(h, n, max_iters=300, top_k=3):
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    step = 0.05

        corr = np.correlate(h, 1.0 - h, mode="full")
        top_indices = np.argsort(corr)[-top_k:]
        
        grad = np.zeros(n)
        for k_idx in top_indices:
            L = k_idx - (n - 1)
            if L >= 0:
                grad[:n - L] += (1.0 - h[L:])
                grad[L:] -= h[:n - L]
            else:
                Lp = -L
                grad[Lp:] += (1.0 - h[:n - Lp])
                grad[:n - Lp] -= h[Lp:]
        
        h_new = h - step * dx * grad / top_k
        h_new = project_box_sum(h_new, target)
        
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            h = h_new
            best_c5 = c_new
            step = min(step * 1.03, 0.3)
        else:
            step *= 0.6
            if step < 1e-8:
                break
    
    return h, best_c5

def pairwise_polish(h, n, max_passes=20):
    """Try pairwise transfers to reduce C5."""
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.default_rng(12345)
    
    for _pass in range(max_passes):
        improved = False
        idx_order = rng.permutation(n)
        for i_pos in range(n - 1):
            i = idx_order[i_pos]
            j = idx_order[(i_pos + 1) % n]
            if i == j:
                continue
            # Try transferring small amount from j to i and vice versa
            for direction in [1, -1]:
                for eps in [0.02, 0.05]:
                    transfer = direction * eps
                    h2 = h.copy()
                    h2[i] = h[i] + transfer
                    h2[j] = h[j] - transfer
                    if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                        continue
                    # Sum is preserved
                    c = compute_c5(h2)
                    if c < best_c5 - 1e-12:
                        h = h2
                        best_c5 = c
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    
    return h, best_c5
