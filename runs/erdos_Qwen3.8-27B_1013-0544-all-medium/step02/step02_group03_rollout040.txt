import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def slp_solve(h_star, n, target_sum):
    """One SLP iteration."""
    corr = np.correlate(h_star, 1.0 - h_star, mode="full")
    n_corr = 2 * n - 1
    
    # Build gradient matrix: grad[m, k] = d(corr[m])/d(h[k])
    # grad[m, k] = (1-h[k+s]) * [0<=k+s<n] - h[k-s] * [0<=k-s<n] where s = m-(n-1)
    grad = np.zeros((n_corr, n))
    for m in range(n_corr):
        s = m - (n - 1)
        for k in range(n):
            g = 0.0
            ks = k + s
            if 0 <= ks < n:
                g += (1.0 - h_star[ks])
            ks2 = k - s
            if 0 <= ks2 < n:
                g -= h_star[ks2]
            grad[m, k] = g
    
    # LP: minimize t
    # subject to: corr[m] + grad[m,:] @ (h - h*) <= t  for all m
    #            => grad[m,:] @ h - t <= corr[m] - grad[m,:] @ h*
    #            sum(h) = target_sum
    #            0 <= h <= 1
    
    c_m = corr - grad @ h_star  # constant terms
    
    # Inequality: grad[m,:] @ h - t <= c_m  =>  [grad[m,:], -1] @ [h, t] <= c_m
    # Also: -h[k] <= 0 for h[k] >= 0
    A_ub = np.zeros((n_corr + n, n + 1))
    b_ub = np.zeros(n_corr + n)
    for m in range(n_corr):
        A_ub[m, :n] = grad[m, :]
        A_ub[m, n] = -1.0
        b_ub[m] = c_m[m]
    for k in range(n):
        A_ub[n_corr + k, k] = -1.0
        # b_ub[n_corr + k] = 0.0 (already)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    bounds_lp = [(0.0, 1.0)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    
    if not res.success:
        return None
    return res.x[:n]

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def compute_grad_matrix(h_star, n):
    n_corr = 2 * n - 1
    grad = np.zeros((n_corr, n))
    for m in range(n_corr):
        s = m - (n - 1)
        if s >= 0:
            grad[m, :n - s] += (1.0 - h_star[s:])
            grad[m, s:] += (-h_star[:n - s])
        else:
            d = -s
            grad[m, d:] += (1.0 - h_star[:n - d])
            grad[m, :n - d] += (-h_star[d:])
    return grad


def slp_solve(h_star, n, target_sum):
    corr = np.correlate(h_star, 1.0 - h_star, mode="full")
    n_corr = 2 * n - 1
    grad = compute_grad_matrix(h_star, n)
    c_m = corr - grad @ h_star
    A_ub = np.zeros((n_corr + n, n + 1))
    b_ub = np.zeros(n_corr + n)
    A_ub[:n_corr, :n] = grad
    A_ub[:n_corr, n] = -1.0
    b_ub[:n_corr] = c_m
    A_ub[n_corr:, np.arange(n)] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds_lp = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if not res.success:
        return None
    return res.x[:n]


def slp_optimize(h_init, n, target_sum, max_iter=300, time_limit=120):
    t0 = time.time()
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_c5_h = compute_c5(h)
    stuck_count = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        h_new = slp_solve(h, n, target_sum)
        if h_new is None:
            break
        h_new = project_feasible(h_new, target_sum)
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            h_try = (1.0 - alpha) * h + alpha * h_new
            h_try = project_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_h = h_try.copy()
                best_c5 = c5_try
                improved = True
                break
            if c5_try < compute_c5(h) - 1e-14:
                h = h_try.copy()
                improved = True
        if not improved:
            stuck_count += 1
            if stuck_count > 5:
                h = project_feasible(h + np.random.randn(n) * 0.01, target_sum)
                stuck_count = 0
            else:
                h = best_h.copy()
                h = project_feasible(h + np.random.randn(n) * 0.005, target_sum)
        else:
            stuck_count = 0
            h = best_h.copy() if compute_c5(best_h) < compute_c5(h) else h
    return best_h, best_c5


def slsqp_polish(h_init, n, target_sum, time_limit=30):
    t0 = time.time()
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in [500, 2000, 10000]:
        if time.time() - t0 > time_limit - 5:
            break
        def obj_func(hv):
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            return smooth_max(corr, beta) * dx
        def con_func(hv):
            return hv.sum() - target_sum
        def con_jac(hv):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
        try:
            res = minimize(obj_func, h_init, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < compute_c5(h_init) - 1e-12:
                h_init = h_new
        except Exception:
            pass
    return h_init, compute_c5(h_init)


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
