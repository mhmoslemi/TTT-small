At the current best_h, I can identify which lags are "active" (close to the max). Then I can compute the gradient of the max objective (as a convex combination of the active lag gradients) and do a projected gradient step. This is more efficient than SLSQP when the active set is small.

def active_gradient_step(h, n, step_size, tol=1e-4):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    c_max = c.max()
    
    # Active set
    active = c >= c_max - tol * c_max
    if not np.any(active):
        active[np.argmax(c)] = True
    
    # Compute gradient of c[k] with respect to h
    # c[k] = dx * sum_i h[i] * (1 - h[i + shift(k)])
    # where shift(k) depends on the correlation indexing
    
    # For "full" mode, c[k] = sum_i h[i] * v[i - (k - (n-1))]
    # where v = 1 - h
    # Let shift = k - (n-1)
    
    # dc[k]/dh[i] = dx * [v[i - shift] + h[i + shift] * (-1)]
    #              = dx * [(1-h[i-shift]) - h[i+shift]]
    # where indices are NOT wrapped (since it's "full" correlation, not circular)
    
    # This is complex to handle correctly. Let me use numerical gradient instead.
    eps = 1e-7
    grad = np.zeros(n)
    target_indices = np.where(active)[0]
    
    for i in range(n):
        h[i] += eps
        c_new = np.correlate(h, 1 - h, mode='full') * dx
        h[i] -= eps
        f_new = np.max(c_new[target_indices])  # max over active set
        # Actually, this isn't quite right because the active set might change
        # Let's just use the full max
        f_new = np.max(c_new)
        f_old = c_max
        grad[i] = (f_new - f_old) / eps
    
    # Projected gradient descent
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    
    c_new = compute_c5(h_new, n)
    if c_new < c_max:
        return h_new, c_new
    return h, c_max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i] -= s * d
                            h[j] += s * d
    return h, best_c
