    grad *= dx
    d = x.sum() - ts
    grad += 2.0 * pen * d
    return grad

def eq_fun(x):
    return x.sum() - _opt['ts']

def eq_jac(x):
    return np.ones(x.shape[0])

def lbfgsb_opt(hc, beta, bounds, maxiter=3000):
    _opt['b'] = beta
    _opt['pen'] = 1e5
    try:
        res = minimize(sg_val, hc, jac=sg_jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return res.x
    except Exception:
        return hc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    _opt['dx'] = dx
    _opt['ts'] = ts
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = project_h(h0, ts)
    best_c5 = compute_c5(best_h)

    # Diverse starts
    starts = [best_h.copy()]
    for _ in range(5):
        hs = best_h + rng.randn(n) * 0.05
        hs = 0.5 * (hs + hs[::-1])
        starts.append(project_h(hs, ts))
    for nb in [3, 5, 7]:
        hb = np.zeros(n)
        bs = n // nb
        for bi in range(nb):
            st = bi * bs
            en = st + bs if bi < nb - 1 else n
            hb[st:en] = rng.uniform(0.1, 0.9)
        starts.append(project_h(hb, ts))
    for freq in [1, 2, 3]:
        t = np.arange(n)
        starts.append(project_h(0.5 + 0.3 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 6.28)), ts))

    # Phase 1: L-BFGS-B on all starts with increasing beta
    beta_schedule = [15, 50, 200, 800, 3000]
    for si, s in enumerate(starts):
        if time.time() - t0 > budget_s - 180:
            break
        hc = s.copy()
        for beta in beta_schedule:
            if time.time() - t0 > budget_s - 60:
                break
            hc = lbfgsb_opt(hc, beta, bounds, maxiter=2000)
            hn = project_h(hc, ts)
            c5 = compute_c5(hn)
            if c5 < best_c5:
                best_h = hn.copy()
                best_c5 = c5
                hc = hn.copy()

    # Phase 2: Local search with gradient-based + random perturbations
    for it in range(120):
        if time.time() - t0 > budget_s - 40:
            break
        ht = best_h.copy()
        strat = it % 6
        if strat == 0:
            ht += rng.randn(n) * 0.006
        elif strat == 1:
            _opt['b'] = 500
            _opt['pen'] = 1e5
            g = sg_jac(ht)
            g -= g.mean()
            scale = 0.02 / (np.max(np.abs(g)) + 1e-10)
            ht -= scale * g
        elif strat == 2:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.05, 0.05)
            ht[i] += d
            ht[j] -= d
        elif strat == 3:
            bt = rng.randint(2, 7)
            bs = rng.randint(0, n - bt)
            ht[bs:bs + bt] += rng.randn(bt) * 0.012
        elif strat == 4:
            ht = 0.93 * ht + 0.07 * np.mean(ht)
        else:
            ht = 0.7 * ht + 0.3 * ht[::-1]
            ht += rng.randn(n) * 0.005

        ht = project_h(ht, ts)
        for beta in [500, 2000]:
            if time.time() - t0 > budget_s - 20:
                break
            hn = lbfgsb_opt(ht, beta, bounds, maxiter=4000)
            hn = project_h(hn, ts)
            c5 = compute_c5(hn)
            if c5 < best_c5:
                best_h = hn.copy()
                best_c5 = c5
                ht = hn.copy()

    # Phase 3: Try higher resolution if time permits
    if time.time() - t0 < budget_s - 200 and n < 200:
        n2 = min(2 * n, 200)
        ts2 = n2 / 2.0
        dx2 = 2.0 / n2
        # Upsample via linear interpolation
        x1 = np.linspace(0, 2, n)
        x2 = np.linspace(0, 2, n2)
        h2_init = np.interp(x2, x1, best_h)
        h2_init = project_h(h2_init, ts2)
        _opt['dx'] = dx2
        _opt['ts'] = ts2
        bounds2 = Bounds(np.zeros(n2), np.ones(n2))
        c5_2 = compute_c5(h2_init)
        h2_best = h2_init.copy()
        for beta in [30, 100, 500, 2000, 5000]:
            if time.time() - t0 > budget_s - 60:
                break
            h2_best = lbfgsb_opt(h2_best, beta, bounds2, maxiter=3000)
            h2_best = project_h(h2_best, ts2)
            c5_2 = compute_c5(h2_best)
            if c5_2 < best_c5:
                best_h = h2_best.copy()
                best_c5 = c5_2
                n = n2
                dx = dx2
                ts = ts2
                bounds = bounds2

    # Phase 4: Final high-precision polish
    if time.time() - t0 < budget_s - 15:
        for beta in [5000, 10000, 20000]:
            if time.time() - t0 > budget_s - 8:
                break
            _opt['pen'] = 1e6
            hn = lbfgsb_opt(best_h.copy(), beta, bounds, maxiter=8000)
            hn = project_h(hn, ts)
            c5 = compute_c5(hn)
            if c5 < best_c5:
                best_h = hn.copy()
                best_c5 = c5

        # SLSQP exact constraint polish
        if time.time() - t0 < budget_s - 5:
            _opt['b'] = 10000
            _opt['pen'] = 0.0
            cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
            try:
                res = minimize(sg_val, best_h, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15})
                hn = project_h(res.x, ts)
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_h = hn.copy()
                    best_c5 = c5
            except Exception:
                pass

    best_h = project_h(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
