import numpy as np
from scipy.optimize import minimize, Bounds
import time

_global_target_sum = 35.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def _con_fun(x):
    return x.sum() - _global_target_sum

def _con_jac(x):
    return np.ones(len(x))

def _obj_func(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, beta) * dx

def cd_sweep(h, target_sum):
    n = len(h)
    for i in range(n):
        old = h[i]
        h[i] = 0.0
        corr0 = np.correlate(h, 1.0 - h, mode="full")
        h[i] = 1.0
        corr1 = np.correlate(h, 1.0 - h, mode="full")
        h[i] = old
        delta = corr1 - corr0
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(corr0 + delta * m1) < np.max(corr0 + delta * m2):
                hi = m2
            else:
                lo = m1
        t_mid = (lo + hi) / 2.0
        f0 = np.max(corr0 + delta * lo)
        f1 = np.max(corr0 + delta * t_mid)
        f2 = np.max(corr0 + delta * hi)
        if f0 <= f1 and f0 <= f2:
            h[i] = lo
        elif f2 <= f0 and f2 <= f1:
            h[i] = hi
        else:
            h[i] = t_mid
    return project_h(h, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    global _global_target_sum
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    _global_target_sum = target_sum
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def _tleft():
        return budget_s - (time.time() - start_time)

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    no_improve = 0
    slsqp_every = 6

    while _tleft() > 80:
        # Coordinate descent sweeps
        h_cur = best_h.copy()
        for _ in range(4):
            h_cur = cd_sweep(h_cur, target_sum)
        c5_cur = compute_c5(h_cur)

        if c5_cur < best_c5 - 1e-13:
            best_h, best_c5 = h_cur, c5_cur
            no_improve = 0
        else:
            no_improve += 1

        # SLSQP refinement when CD stalls
        if no_improve >= slsqp_every and _tleft() > 120:
            beta = min(200 + no_improve * 40, 2000)
            try:
                res = minimize(_obj_func, best_h, args=(beta, dx),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 1000, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                    no_improve = 0
            except Exception:
                pass

        # Perturbation to escape local optima
        if no_improve >= 12 and _tleft() > 250:
            n_pert = np.random.randint(4, min(18, n_points // 3 + 1))
            idx = np.random.choice(n_points, n_pert, replace=False)
            scale = 0.04 + 0.08 * min(no_improve / 40.0, 1.0)
