    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    # Vectorized gradient
    grad = np.zeros(n)
    j = np.arange(n)
    # For each lag s = k - (n-1):
    # d(corr[k])/d(h[j]) = (1-h[j+s]) - h[j-s] for valid indices
    # grad[j] = dx * sum_k w[k] * ((1-h[j+s_k]) - h[j-s_k])
    #         = dx * (sum_k w[k] - sum_k w[k]*h[j+s_k] - sum_k w[k]*h[j-s_k])
    # where sums are over valid k for each j
    
    # Let's compute this using convolutions
    # Term 1: sum_k w[k] for all k where both j+s and j-s are... 
    # Actually, the valid condition for the first term (1-h[j+s]) is 0<=j+s<n
    # and for the second term h[j-s] is 0<=j-s<n
    # These are different conditions!
    # So we need to be careful.
    
    # Let me just loop but vectorize the inner operation
    # Actually for n=200, 399 iters is fine
    
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        # (1 - h[j+s]) for valid j+s
        js = j + s
        mask_p = (js >= 0) & (js < n)
        if mask_p.any():
            grad[mask_p] += wk * (1.0 - h[js[mask_p]])
        # -h[j-s] for valid j-s
        jm = j - s
        mask_m = (jm >= 0) & (jm < n)
        if mask_m.any():
            grad[mask_m] += wk * (-h[jm[mask_m]])
    
    grad *= dx
    return val, grad


def lbfhsb_optimize(h, maxiter, beta, lam):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v + lam * (x.sum() - target) ** 2
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g + 2 * lam * (x.sum() - target) * np.ones(n)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def slsqp_refine(h, maxiter, beta):

    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))

def subgrad_momentum(h, max_iters, lr0, decay, top_k, momentum=0.9):

    target = n / 2.0
    x = h.copy().astype(float)

    j = np.arange(n)
    
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        y = project_feasible(x, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]

        grad = np.zeros(n)

        val = compute_c5(x_new)
        if val < best_v:

            best_x = x_new.copy()
        x = x_new

        he0 = h0[:n0 // 2].copy()
        for beta in [60, 120, 250, 500]:
            if tl() < 20:
                break
            hf = opt_symmetric(he0, n0, maxiter=3000, beta=beta)
            c = compute_c5(hf)
            if c < best_c5:
                best_c5 = c
                best_h = hf.copy()
                he0 = hf[:n0 // 2].copy()

    # Phase 2: Free optimization at original resolution
    for beta in [100, 250]:
        if tl() < 25:
            break
        ho = opt_free(h0, n0, maxiter=3000, beta=beta, lam=8000.0)
        c = compute_c5(ho)
        if c < best_c5:
            best_c5 = c
            best_h = ho.copy()

    # Phase 3: SLSQP polish
    for beta in [200, 500]:
        if tl() < 20:
            break
        hs = slsqp_refine(best_h, maxiter=500, beta=beta)
        c = compute_c5(hs)
        if c < best_c5:
            best_c5 = c
            best_h = hs.copy()

    # Phase 4: Subgradient diversity
    if tl() > 40:
        hsg, csg = subgrad_momentum(best_h, 2000, 0.04, 0.9997, top_k=7)
        if csg < best_c5:
            best_c5 = csg
            best_h = hsg.copy()

    # Phase 5: Resolution sweep
    resolutions = [100, 140, 200, 260, 300, 400, 500]
    for n_try in resolutions:
        if tl() < 90:
            break
        if n_try % 2 != 0:
            n_try += 1
        h_interp = interp_to_n(best_h, n_try)
        he_interp = h_interp[:n_try // 2].copy()

        candidates_he = [he_interp]
        candidates_free = [h_interp]

        # Structured starts
        hb = make_block(n_try, 0.5)
        candidates_free.append(hb)
        candidates_he.append(hb[:n_try // 2].copy())

        hsin = make_sinusoidal(n_try, 1.0, 0.4)
        candidates_free.append(hsin)
        candidates_he.append(hsin[:n_try // 2].copy())

        # Random feasible
        hr = np.random.rand(n_try)
        hr = project_feasible(hr, n_try / 2.0)
        candidates_free.append(hr)
        candidates_he.append(hr[:n_try // 2].copy())

        for he_cand in candidates_he:
            if tl() < 40:
                break
            hf = opt_symmetric(he_cand, n_try, maxiter=4000, beta=120)
            c = compute_c5(hf)
            if c < best_c5:
                best_c5 = c
                best_h = hf.copy()
                if tl() > 20:
                    hf2 = slsqp_symmetric(hf[:n_try // 2], n_try, maxiter=400, beta=300)
                    c2 = compute_c5(hf2)
                    if c2 < best_c5:
                        best_c5 = c2
                        best_h = hf2.copy()

        for hcand in candidates_free:
            if tl() < 40:
                break
            ho = opt_free(hcand, n_try, maxiter=4000, beta=120, lam=8000.0)
            c = compute_c5(ho)
            if c < best_c5:
                best_c5 = c
                best_h = ho.copy()
                if tl() > 20:
                    ho2 = slsqp_refine(ho, maxiter=400, beta=300)
                    c2 = compute_c5(ho2)
                    if c2 < best_c5:
                        best_c5 = c2
                        best_h = ho2.copy()

    # Phase 6: Final high-precision polish
    if tl() > 30:
        n_best = len(best_h)
        if n_best % 2 == 0:
            hsp = slsqp_symmetric(best_h[:n_best // 2], n_best, maxiter=800, beta=600)
            c = compute_c5(hsp)
            if c < best_c5:
                best_c5 = c
                best_h = hsp.copy()
        hp = slsqp_refine(best_h, maxiter=600, beta=600)
        c = compute_c5(hp)
        if c < best_c5:
            best_c5 = c
            best_h = hp.copy()

    # Phase 7: Try one higher resolution
    if tl() > 80:
        n_up = len(best_h) + 40
        n_up = n_up - n_up % 2
        h_up = interp_to_n(best_h, n_up)
        he_up = h_up[:n_up // 2].copy()
        hf_up = opt_symmetric(he_up, n_up, maxiter=4000, beta=150)
        c = compute_c5(hf_up)
        if c < best_c5:
            c2 = compute_c5(slsqp_symmetric(hf_up[:n_up // 2], n_up, 500, 400))
            if c2 < best_c5:
                best_c5 = c2
                best_h = slsqp_symmetric(hf_up[:n_up // 2], n_up, 500, 400)
            else:
                best_c5 = c
                best_h = hf_up.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
