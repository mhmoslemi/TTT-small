    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 0 and abs(s - target_sum) > 1e-10:
        # Distribute the correction
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
        # Second pass if needed
        s2 = h.sum()
        if abs(s2 - target_sum) > 1e-8:
            free = (h > 0.01) & (h < 0.99)
            if free.sum() > 0:
                h[free] += (target_sum - s2) / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def active_lag_obj(h, top_lags_idx, penalty_beta, dx):

    top = corr[top_lags_idx]
    m = np.max(top)
    val = m + np.log(np.sum(np.exp(penalty_beta * (top - m)))) / penalty_beta

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12 and abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if abs(s2 - target_sum) > 1e-8:
            free = (h > 0.005) & (h < 0.995)
            if free.sum() > 0:
                h[free] += (target_sum - s2) / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def binary_project(h, n):
    sorted_idx = np.argsort(h)[::-1]
    binary = np.zeros(n)
    binary[sorted_idx[:n // 2]] = 1.0
    return binary

def get_top_lag_indices(h, k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.argsort(corr)[::-1][:k]

def slsqp_solve(h0, n, target_sum, dx, beta, maxiter, ftol, bounds, constraints):
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, float(beta)) * dx
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return None, float('inf')
