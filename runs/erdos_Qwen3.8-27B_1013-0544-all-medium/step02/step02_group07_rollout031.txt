def compute_c5(h):
    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)


def smooth_max(x, beta):
    m = np.max(x)
    if beta < 1:
        return m
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    if k >= len(x):
        return smooth_max(x, beta)
    idx = np.argpartition(x, -k)[-k:]
    return smooth_max(x[idx], beta)
