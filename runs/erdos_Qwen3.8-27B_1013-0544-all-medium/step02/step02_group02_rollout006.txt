def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - n / 2.0

    return np.ones(n)

        # Quick SLSQP to see if this direction helps
        h_opt = slsqp(h_try, n, 50000, maxiter=1500, ftol=1e-15)
        c = compute_c5(h_opt, n)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
            # Polish the new best
            if remaining() > 30:
                h_opt = slsqp(best_h, n, 500000, maxiter=4000, ftol=1e-18)
                c2 = compute_c5(h_opt, n)
                if c2 < best_c5:
                    best_h, best_c5 = h_opt, c2
        # Also try structured perturbations
        if attempts % 3 == 0 and remaining() > 80:
            # Swap two values
            idx = rng.permutation(n)
            a1, b1 = idx[0], idx[1]
            h_try = best_h.copy()
            h_try[a1], h_try[b1] = best_h[b1], best_h[a1]
            h_opt = slsqp(h_try, n, 50000, maxiter=1500, ftol=1e-15)
            c = compute_c5(h_opt, n)
            if c < best_c5:
                best_h, best_c5 = h_opt, c

    # Phase 7: More pair descent on best
    if remaining() > 60:
        h_work = best_h.copy()
        h_work, c = pair_descend(h_work, n, rng, 2000, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = h_work, c

    # Phase 8: Final SLSQP polish
    for a in [2000000, 10000000]:
        if remaining() < 15:
            break
        h_opt = slsqp(best_h, n, a, maxiter=10000, ftol=1e-20)
        c = compute_c5(h_opt, n)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
