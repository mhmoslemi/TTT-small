    m = np.max(x)
    safe = np.clip(x - m, -100.0 / beta, 0.0)
    return float(m + np.log(np.sum(np.exp(beta * safe))) / beta)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target_sum) > 1e-10:
        h = h + (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth_all(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def obj_smooth_crit(h, beta, dx, crit_idx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr[crit_idx], beta) * dx

def obj_weighted(h, weights, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.dot(weights, corr)) * dx

def con_sum(x, target):
    return float(np.sum(x) - target)

def con_sum_jac(x, target):
    return np.ones(len(x))

def get_crit_indices(h, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.argsort(corr)[::-1][:top_k]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_par = np.array(initial_h_values, dtype=float).copy()
    else:
        h_par = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    for n_points in [70, 80, 90]:
        if time.time() - t_start > budget_s - 150:
            break

        n_target = float(n_points) / 2.0
        dx = 2.0 / n_points
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        con = {'type': 'eq',
               'fun': functools.partial(con_sum, target=n_target),
               'jac': functools.partial(con_sum_jac, target=n_target)}

        # Starting point from parent
        if h_par is not None:
            if len(h_par) == n_points:
                h = project_h(h_par.copy(), n_target)
            else:
                xo = np.linspace(0, 2, len(h_par), endpoint=False)
                xn = np.linspace(0, 2, n_points, endpoint=False)
                h = project_h(np.interp(xn, xo, h_par, period=2.0), n_target)
        else:
            h = np.full(n_points, 0.5)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n_points

        # Phase A: Global SLSQP with increasing smooth_max beta
        for beta in [20, 50, 100, 200]:
            if time.time() - t_start > budget_s - 130:
                break
            obj = functools.partial(obj_smooth_all, beta=beta, dx=dx)
            try:
                res = minimize(obj, h, method='SLSQP', bounds=bounds,
                              constraints=con, options={'maxiter': 500, 'ftol': 1e-14})
                h = project_h(res.x, n_target)
            except Exception:
                pass
            c5_new = compute_c5(h)
            if c5_new < c5:
                h, c5 = h, c5_new
                if c5 < best_c5:
                    best_h, best_c5, best_n = h.copy(), c5, n_points

        # Phase B: Active-set refinement loop
        for iteration in range(25):
            if time.time() - t_start > budget_s - 100:
                break
            crit_idx = get_crit_indices(h, 5)
            for beta in [300, 600]:
                if time.time() - t_start > budget_s - 90:
                    break
                obj = functools.partial(obj_smooth_crit, beta=beta, dx=dx, crit_idx=crit_idx)
                try:
                    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                                  constraints=con, options={'maxiter': 400, 'ftol': 1e-15})
                    h_new = project_h(res.x, n_target)
                except Exception:
                    h_new = h
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-15:
                    h, c5 = h_new, c5_new
                    if c5 < best_c5:
                        best_h, best_c5, best_n = h.copy(), c5, n_points

        # Phase C: Nelder-Mead with exact objective (derivative-free local search)
        if time.time() - t_start < budget_s - 70:
            try:
                res = minimize(compute_c5, h, method='Nelder-Mead',
                              options={'maxiter': 5000, 'xatol': 1e-13, 'fatol': 1e-14,
                                       'initial_simplex': h + 0.01 * np.random.randn(n_points)})
                h_nm = project_h(res.x, n_target)
                c5_nm = compute_c5(h_nm)
                if c5_nm < c5:
                    h, c5 = h_nm, c5_nm
                    if c5 < best_c5:
                        best_h, best_c5, best_n = h.copy(), c5, n_points
            except Exception:
                pass

        # Phase D: Exponential weights iteration (diversification)
        if time.time() - t_start < budget_s - 50:
            n_corr = 2 * n_points - 1
            weights = np.ones(n_corr) / n_corr
            for it in range(20):
                if time.time() - t_start > budget_s - 40:
                    break
                obj = functools.partial(obj_weighted, weights=weights, dx=dx)
                try:
                    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                                  constraints=con, options={'maxiter': 300, 'ftol': 1e-13})
                    h_new = project_h(res.x, n_target)
                except Exception:
                    h_new = h
                c5_new = compute_c5(h_new)

                # Update weights with exponential rule
                vals = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
                eta = 4.0 / (it + 1)
                log_w = np.log(weights + 1e-30) + eta * vals
                log_w -= np.max(log_w)
                weights = np.exp(log_w)
                weights /= weights.sum()

                if c5_new < c5:
                    h, c5 = h_new, c5_new
                    if c5 < best_c5:
                        best_h, best_c5, best_n = h.copy(), c5, n_points

        # Phase E: Final precision polish on best for this n
        if time.time() - t_start < budget_s - 20:
            for top_k in [3, 4, 5, 6]:
                if time.time() - t_start > budget_s - 15:
                    break
                crit_idx = get_crit_indices(h, top_k)
                obj = functools.partial(obj_smooth_crit, beta=800, dx=dx, crit_idx=crit_idx)
                try:
                    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                                  constraints=con, options={'maxiter': 2000, 'ftol': 1e-16})
                    h_new = project_h(res.x, n_target)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5:
                        h, c5 = h_new, c5_new
                        if c5 < best_c5:
                            best_h, best_c5, best_n = h.copy(), c5, n_points
                except Exception:
                    pass

        # Phase F: Targeted restarts around best
        if time.time() - t_start < budget_s - 15:
            for restart in range(8):
                if time.time() - t_start > budget_s - 10:
                    break
                scale = 0.005 * (1.0 + restart * 0.5)
                h_pert = h + np.random.randn(n_points) * scale
                h_pert = project_h(h_pert, n_target)
                obj = functools.partial(obj_smooth_all, beta=150, dx=dx)
                try:
                    res = minimize(obj, h_pert, method='SLSQP', bounds=bounds,
                                  constraints=con, options={'maxiter': 200, 'ftol': 1e-13})
                    h_new = project_h(res.x, n_target)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5:
                        h, c5 = h_new, c5_new
                        if c5 < best_c5:
                            best_h, best_c5, best_n = h.copy(), c5, n_points
                        obj2 = functools.partial(obj_smooth_crit, beta=500, dx=dx,
                                               crit_idx=get_crit_indices(h, 4))
                        try:
                            res2 = minimize(obj2, h, method='SLSQP', bounds=bounds,
                                           constraints=con, options={'maxiter': 500, 'ftol': 1e-15})
                            h = project_h(res2.x, n_target)
                            c5 = compute_c5(h)
                            if c5 < best_c5:
                                best_h, best_c5, best_n = h.copy(), c5, n_points
                        except Exception:
                            pass
                except Exception:
                    pass

    # Ultimate final check
    if best_h is not None:
        n_f = best_n
        dx_f = 2.0 / n_f
        ts = float(n_f) / 2.0
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        con_f = {'type': 'eq',
                 'fun': functools.partial(con_sum, target=ts),
                 'jac': functools.partial(con_sum_jac, target=ts)}
        if time.time() - t_start < budget_s - 5:
            crit_idx = get_crit_indices(best_h, 4)
            obj_f = functools.partial(obj_smooth_crit, beta=1000, dx=dx_f, crit_idx=crit_idx)
            try:
                res = minimize(obj_f, best_h, method='SLSQP', bounds=bounds_f,
                              constraints=con_f, options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_h(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
            except Exception:
                pass

    return best_h, best_c5, best_n
