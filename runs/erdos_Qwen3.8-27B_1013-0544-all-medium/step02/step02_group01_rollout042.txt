```python
        return v

        return g
```

```python
def fista_smoothmax(h0, beta, max_iters, step_size):
    """FISTA for smooth max with projection."""
    n = len(h0)
    target = n / 2.0
    x = project_feasible(h0, target)
    z = x.copy()
    t_k = 1.0
    best_x = x.copy()
    best_val = compute_c5(x)
    
    for k in range(max_iters):
        # Gradient at z
        val, grad = smoothmax_val_grad(z, beta)
        
        # Gradient step
        x_new = z - step_size * grad
        
        # Projection
        x_new = project_feasible(x_new, target)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z = project_feasible(z, target)
        
        t_k = t_new
        x = x_new
        
        v = compute_c5(x)
        if v < best_val:
            best_val = v
            best_x = x.copy()
    
    return best_x, best_val
```

**Approach: Direct formulation as a sequence of LPs/QPs**

For a fixed set of "active" lags (those achieving the max), the problem becomes:
min_h t
s.t. f_s(h) ≤ t for s in active_set
     0 ≤ h ≤ 1
     sum(h)*dx = 1

This is a QP (quadratically constrained) which is hard. But if I linearize the quadratic terms around the current solution, I get an LP that I can solve with cvxpy.

This is essentially a "successive linear programming" approach. Let me try it:

At each iteration:
1. Compute the current active set (top few lags)
2. Linearize f_s(h) around current h: f_s(h) ≈ f_s(h_old) + ∇f_s(h_old)^T (h - h_old)
3. Solve the LP: min t s.t. f_s(h_old) + ∇f_s(h_old)^T (h-h_old) ≤ t, 0≤h≤1, sum(h)=target
4. Update h, go to 1

This is the "luSS" or "outer approximation" method. It's guaranteed to converge to a local optimum for non-convex problems (under some conditions).

This could be very effective! Let me implement it.

∇f_s(h)[i] = dx * [(1-h[i-k_s])*[0≤i-k_s<n] - h[i+k_s]*[0≤i+k_s<n]]

where k_s is the correlation index corresponding to lag s.

```python
import cvxpy as cp

def lp_refinement(h0, n_lags=5, max_iters=50):
    """Successive LP refinement for the min-max problem."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Get top n_lags
        top_k = np.argsort(corr)[-n_lags:]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for k in top_k:
            # Linearize: f_s(h) ≈ f_s(h) + grad_s(h)^T (h_var - h)
            s = k  # correlation index
            f_val = corr[k] * dx
            grad_s = np.zeros(n)
            i1 = max(0, k)
            i2 = n - 1
            if i1 <= i2:
                grad_s[i1:i2+1] = (1.0 - h[i1-k:i2-k+1])
            j1 = 0
            j2 = min(n - 1, n - 1 - k)
            if j1 <= j2:
                grad_s[j1:j2+1] -= h[j1+k:j2+k+1]
            grad_s *= dx
            
            constraints.append(f_val + grad_s @ (h_var - h) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.GLOP)  # or GLPK
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, target)
        
        if np.allclose(h_new, h):
            break
        h = h_new
    
    return h
```

```python
def slsqp_optimize(h0, beta, maxiter, max_lags=None):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta, max_lags=max_lags)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta, max_lags=max_lags)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)

                    c_p2 = compute_c5(h_p)
                    if c_p2 < best_c5:
                        best_h, best_c5 = h_p, c_p2
                    break

    # Phase 9: Final polish
    if tl() > 30:
        h_fin = slsqp_optimize(best_h, 500, 400, max_lags=10)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_h, best_c5 = h_fin, c_fin
        h_fin = lp_refine(best_h, n_active=15, max_iters=30)
        c_fin2 = compute_c5(h_fin)
        if c_fin2 < best_c5:
            best_h, best_c5 = h_fin, c_fin2

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
