```python
def coord_descent(h, n_sweeps, mu_init=0.0, mu_lr=0.01):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    mu = mu_init
    
    # Precompute correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    
    for sweep in range(n_sweeps):
        for i in range(n):
            # Compute slopes for coordinate i
            slopes = np.zeros(2*n-1)
            # For s = 0 to n-1:
            #   slope includes (1-h[i+s]) if i+s < n
            #   slope includes -h[i-s] if i-s >= 0
            # For s = -(n-1) to -1 (t = -s):
            #   slope includes (1-h[i-t]) if i >= t
            #   slope includes -h[i+t] if i+t < n
            
            # Vectorized computation:
            # Positive s = 0 to n-1-i: (1-h[i+s])
            # Positive s = 0 to i: -h[i-s]  → this is -h[0:i+1] reversed = -h[i:i-(-s):]...
            
            # Let me just do it directly
            # s = 0:
            slopes[n-1] = (1.0 - 2.0 * h[i]) * dx
            
            # s = 1 to n-1:
            for s in range(1, n):
                val = 0.0
                if i + s < n:
                    val += (1.0 - h[i+s])
                if i - s >= 0:
                    val += (-h[i-s])
                slopes[n-1+s] = val * dx
                
                val_neg = 0.0
                if i >= s:
                    val_neg += (1.0 - h[i-s])
                if i + s < n:
                    val_neg += (-h[i+s])
                slopes[n-1-s] = val_neg * dx
            
            # Minimize max_s ((slopes[s] + mu) * t + intercepts[s]) for t in [0,1]
            intercepts = g - slopes * h[i]
            
            # Ternary search
            lo, hi = 0.0, 1.0
            mod_slopes = slopes + mu
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(mod_slopes * m1 + intercepts) <= np.max(mod_slopes * m2 + intercepts):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            
            # Update h[i] and correlation
            delta = t_opt - h[i]
            if abs(delta) > 1e-15:
                # Update correlation: corr_s += delta * slopes_s / dx = delta * slopes_s * n / 2
                # Because slopes_s = (raw_slope) * dx, so raw_slope = slopes_s / dx
                # And corr_s = g_s / dx, so g_s += delta * raw_slope = delta * slopes_s / dx
                # Wait: g_s = slope_s * h[i] + intercept_s
                # After update: g_s_new = slope_s * t_opt + intercept_s = g_s + slope_s * (t_opt - h[i]) = g_s + slope_s * delta
                g += slopes * delta
                # Update corr
                corr = g / dx
                
                # Also need to update h for future slope computations
                h[i] = t_opt
        
        # Update mu (Lagrange multiplier for sum constraint)
        sum_diff = h.sum() - target
        mu += mu_lr * sum_diff
        # Project to feasible
        h = project_feasible(h, target)
        # Recompute correlation (in case projection changed things)
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = corr * dx
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_sh = np.max(shifted)
    exps = np.exp(shifted - max_sh)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + (max_sh + np.log(sw)) / beta) * dx
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        k = s + (n-1)
        if w[k] < 1e-12:
            continue
        if s >= 0:
            grad[:n-s] += w[k] * (1.0 - h[s:])
            grad[s:] += w[k] * (-h[:n-s])
        else:
            t = -s
            grad[t:] += w[k] * (1.0 - h[:n-t])
            grad[:n-t] += w[k] * (-h[t:])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def coord_descent(h, n_sweeps, mu_lr=0.05, tol=1e-12):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    mu = 0.0
    
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    g = corr * dx
    
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            # Compute slopes for coordinate i
            # slope_s for s from -(n-1) to n-1
            # For s=0: (1-2h[i])
            # For s>0: (1-h[i+s])*I(i+s<n) + (-h[i-s])*I(i-s>=0)
            # For s<0 (t=-s): (1-h[i-t])*I(i>=t) + (-h[i+t])*I(i+t<n)
            
            # Vectorized computation of slopes
            slopes = np.zeros(2*n-1)
            # s = 0
            slopes[n-1] = (1.0 - 2.0*h[i]) * dx
            
            # s = 1, ..., n-1
            # For s > 0:
            #   (1-h[i+s]) if i+s < n → s < n-i
            #   -h[i-s] if i-s >= 0 → s <= i
            # For s < 0 (t = -s > 0):
            #   (1-h[i-t]) if i >= t → t <= i → s >= -i
            #   -h[i+t] if i+t < n → t < n-i → s > -(n-i)
            
            for s in range(1, n):
                val_pos = 0.0
                if i + s < n:
                    val_pos += 1.0 - h[i+s]
                if s <= i:
                    val_pos -= h[i-s]
                slopes[n-1+s] = val_pos * dx
                
                val_neg = 0.0
                if s <= i:
                    val_neg += 1.0 - h[i-s]
                if i + s < n:
                    val_neg -= h[i+s]
                slopes[n-1-s] = val_neg * dx
            
            # Ternary search for optimal t
            intercepts = g - slopes * h[i]
            mod_slopes = slopes + mu
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) * 0.333333
                m2 = hi - (hi - lo) * 0.333333
                if np.max(mod_slopes * m1 + intercepts) <= np.max(mod_slopes * m2 + intercepts):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) * 0.5
            
            delta = t_opt - h[i]
            if abs(delta) > 1e-14:
                g += slopes * delta
                h[i] = t_opt
                max_change = max(max_change, abs(delta))
        
        # Update mu
        sum_diff = h.sum() - target
        mu += mu_lr * sum_diff
        
        if max_change < tol:
            break
    
    h = project_feasible(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def tl():
        return budget_s - (time.time() - t0)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    
    n = len(h_best)
    best_c5 = compute_c5(h_best)
    
    # Phase 1: Coordinate descent (fast)
    h_cd = coord_descent(h_best, n_sweeps=5000, mu_lr=0.1)
    c5_cd = compute_c5(h_cd)
    if c5_cd < best_c5:
        h_best, best_c5 = h_cd, c5_cd
    
    # Phase 2: More coordinate descent with smaller mu_lr
    if tl() > 60:
        h_cd2 = coord_descent(h_best, n_sweeps=5000, mu_lr=0.03)
        c5_cd2 = compute_c5(h_cd2)
        if c5_cd2 < best_c5:
            h_best, best_c5 = h_cd2, c5_cd2
    
    # Phase 3: SLSQP polish
    if tl() > 80:
        h_sls = slsqp_refine(h_best, maxiter=500, beta=100)
        c5_sls = compute_c5(h_sls)
        if c5_sls < best_c5:
            h_best, best_c5 = h_sls, c5_sls
    
    # Phase 4: Coordinate descent after SLSQP (to recover from SLSQP artifacts)
    if tl() > 60:
        h_cd3 = coord_descent(h_best, n_sweeps=3000, mu_lr=0.05)
        c5_cd3 = compute_c5(h_cd3)
        if c5_cd3 < best_c5:
            h_best, best_c5 = h_cd3, c5_cd3
    
    # Phase 5: Upscale to higher resolution
    for n_try in [300, 500, 800]:
        if tl() < 120:
            break
        h_up = interp_to_n(h_best, n_try)
        # Coordinate descent at new resolution
        if tl() > 80:
            h_u = coord_descent(h_up, n_sweeps=3000, mu_lr=0.08)
            c5_u = compute_c5(h_u)
            if c5_u < best_c5:
                h_best, best_c5 = h_u, c5_u
        
        # SLSQP at new resolution
        if tl() > 60:
            h_s = slsqp_refine(h_best, maxiter=300, beta=150)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                h_best, best_c5 = h_s, c5_s
    
    # Phase 6: Final polish on best resolution
    if tl() > 40:
        h_f = coord_descent(h_best, n_sweeps=3000, mu_lr=0.02)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            h_best, best_c5 = h_f, c5_f
    
    if tl() > 30:
        h_f2 = slsqp_refine(h_best, maxiter=600, beta=300)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < best_c5:
            h_best, best_c5 = h_f2, c5_f2
    
    # Final ensure feasibility
```
