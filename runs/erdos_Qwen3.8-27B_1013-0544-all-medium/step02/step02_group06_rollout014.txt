import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (x - m))))

def project_box_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            if d > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def active_lag_descent(h, n, target, max_rounds=20):
    """Exploit linearity of single-lag objective in each coordinate."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c_cur = np.max(corr) * dx
        top_k = min(4, len(corr))
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for kk in sorted(top_idx, key=lambda x: -corr[x]):
            k_shift = int(kk) - (n - 1)
            # For this lag, compute preferred direction per coordinate
            direction = np.zeros(n)
            for i in range(n):
                val = 0.0
                i1 = i + k_shift
                i2 = i - k_shift
                if 0 <= i1 < n:
                    val += (1.0 - h[i1])
                if 0 <= i2 < n:
                    val -= h[i2]
                direction[i] = val
            # Try moving against the worst lag
            for step in [0.03, 0.015, 0.007, 0.003]:
                h_try = project_box_sum(h - step * direction, target)
                c_try = compute_c5(h_try, n)
                if c_try < c_cur - 1e-15:
                    h = h_try
                    c_cur = c_try
                    improved = True
                    break
        if not improved:
            break
    c_final = compute_c5(h, n)
    return h, min(best_c, c_final)
