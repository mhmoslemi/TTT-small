```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        idx1 = np.arange(lag, n)
        grad[idx1] += (1.0 - h[idx1 - lag])
        idx2 = np.arange(0, n - lag)
        grad[idx2] -= h[idx2 + lag]
    else:
        la = -lag
        idx1 = np.arange(0, n - la)
        grad[idx1] += (1.0 - h[idx1 + la])
        idx2 = np.arange(la, n)
        grad[idx2] -= h[idx2 - la]
    return grad


def grad_smooth_max(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += weights[k] * grad_lag(h, lag, n)
    return grad


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def lbfgsb_step(h, n, ts, dx, beta, pen, maxiter=500):
    def obj(h_in):
        hi = np.clip(h_in, 0, 1)
        corr = np.correlate(hi, 1.0 - hi, mode='full')
        return smooth_max(corr, beta) * dx + pen * (hi.sum() - ts) ** 2

    def grad(h_in):
        hi = np.clip(h_in, 0, 1)
        g = grad_smooth_max(hi, beta) * dx
        g += 2.0 * pen * (hi.sum() - ts) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return project_feasible(res.x, ts)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    def upd(h_cand):
        nonlocal best_h, best_c5
        c = compute_c5(h_cand)
        if c < best_c5:
            best_h = h_cand.copy()
            best_c5 = c
            return True
        return False

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    ts = n / 2.0
    best_h = project_feasible(h0, ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B at current resolution with increasing beta
    for beta in [50, 200, 1000]:
        if tl() < 30:
            break
        h_new = lbfgsb_step(best_h, n, ts, 2.0/n, beta, 50.0, maxiter=300)
        upd(h_new)

    # Phase 2: Active-lag SLSQP (optimize single worst lag)
    for _ in range(3):
        if tl() < 30:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_worst = int(np.argmax(corr))
        lag = k_worst - (n - 1)

        def obj_single(h_in):
            if lag >= 0:
                idx = np.arange(n - lag)
                return np.sum(h_in[idx] * (1.0 - h_in[idx + lag])) * (2.0/n)
            else:
                la = -lag
                idx = np.arange(n - la)
                return np.sum(h_in[idx + la] * (1.0 - h_in[idx])) * (2.0/n)

        def grad_single(h_in):
            return grad_lag(h_in, lag, n) * (2.0/n)

        bounds = [(0.0, 1.0)] * n
        pen = 50.0
        def obj_pen(h_in):
            return obj_single(h_in) + pen * (h_in.sum() - ts)**2
        def grad_pen(h_in):
            return grad_single(h_in) + 2.0*pen*(h_in.sum()-ts)*np.ones(n)

        h_new = project_feasible(best_h, ts) + np.random.randn(n) * 0.002
        h_new = project_feasible(h_new, ts)
        try:
            res = minimize(obj_pen, h_new, jac=grad_pen, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-15})
            upd(project_feasible(res.x, ts))
        except:
            pass

    # Phase 3: Momentum subgradient
    if tl() > 80:
        h = best_h.copy()
        v = np.zeros(n)
        mom = 0.8
        for it in range(100000):
            if tl() < 40:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = int(np.argmax(corr))
            lag = k_worst - (n - 1)
            g = grad_lag(h, lag, n)
            lr = 0.004 / (1.0 + it * 0.0003)
            v = mom * v + (1.0 - mom) * g
            h = h - lr * v
            h = project_feasible(h, ts)
            if it % 10000 == 9999:
                upd(h)
        upd(h)

    # Phase 4: Multi-resolution exploration
    for n_try in [n - 15, n + 15, n + 30, n + 50, n - 30]:
        if n_try < 30 or tl() < 50:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = project_feasible(resample_h(best_h, n_try), ts_t)

        for beta in [100, 500, 2000]:
            if tl() < 25:
                break
            h_new = lbfgsb_step(h_t, n_try, ts_t, dx_t, beta, 50.0, maxiter=400)
            if compute_c5(h_new) < best_c5:
                best_h = h_new.copy()
                best_c5 = compute_c5(best_h)
                n = n_try
                ts = ts_t

    # Phase 5: High-beta L-BFGS-B polish
    for beta in [5000, 20000, 50000]:
        if tl() < 20:
            break
        h_new = lbfgsb_step(best_h, n, ts, 2.0/n, beta, 100.0, maxiter=500)
        upd(h_new)

    # Phase 6: Final active-lag refinement loop
    for _ in range(5):
        if tl() < 15:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_worst = int(np.argmax(corr))
        lag = k_worst - (n - 1)

        def obj_pen2(h_in):
            if lag >= 0:
                idx = np.arange(n - lag)
                val = np.sum(h_in[idx] * (1.0 - h_in[idx + lag])) * (2.0/n)
            else:
                la = -lag
                idx = np.arange(n - la)
                val = np.sum(h_in[idx + la] * (1.0 - h_in[idx])) * (2.0/n)
            return val + 100.0 * (h_in.sum() - ts)**2

        def grad_pen2(h_in):
            g = grad_lag(h_in, lag, n) * (2.0/n)
            g += 200.0 * (h_in.sum() - ts) * np.ones(n)
            return g

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_pen2, best_h, jac=grad_pen2, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-16})
            upd(project_feasible(res.x, ts))
        except:
            pass

    return best_h, best_c5, len(best_h)
```
