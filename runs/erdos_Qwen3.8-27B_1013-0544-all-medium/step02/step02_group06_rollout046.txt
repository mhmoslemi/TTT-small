            starts.append(project_box_sum(h_s, ts))

    for a in [0.2, 0.35, 0.5]:
        f1, f2 = rng.randint(1, 5), rng.randint(5, 12)
        p1, p2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
        h_s = 0.5 + a * np.cos(np.pi * f1 * x + p1) + 0.3 * a * np.cos(np.pi * f2 * x + p2)
        starts.append(project_box_sum(h_s, ts))

    for _ in range(4):
        h_r = rng.uniform(0.15, 0.85, n)
        starts.append(project_box_sum(h_r, ts))

    # Step function starts
    for n_steps in [2, 3, 4]:
        edges = sorted(rng.uniform(0, 1, n_steps - 1))
        levels = rng.uniform(0.1, 0.9, n_steps)
        # Symmetrize: h(x) + h(2-x) should average ~1
        h_step = np.zeros(n)
        for xi in range(n):
            xv = 2.0 * xi / n
            for s in range(n_steps):
                lo = 2.0 * (0 if s == 0 else edges[s-1])
                hi = 2.0 * (1.0 if s == n_steps-1 else edges[s])
                if lo <= xv < hi:
                    h_step[xi] = levels[s]
                    break
            # Symmetric component
            xr = 2.0 - xv
            for s in range(n_steps):
                lo = 2.0 * (0 if s == 0 else edges[s-1])
                hi = 2.0 * (1.0 if s == n_steps-1 else edges[s])
                if lo <= xr < hi:
                    h_step[xi] = 1.0 - levels[s]
                    break
        starts.append(project_box_sum(h_step, ts))

    return starts


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = 1.0
    best_n = 70

    n_values = [70]
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_init = len(initial_h_values)
        if n_init not in n_values:
            n_values.insert(0, n_init)
    for n_try in [50, 60, 80, 90, 100, 56, 64]:
        if n_try not in n_values:
            n_values.append(n_try)

    for n in n_values:
        if tl() < 50:
            break
        ts = n / 2.0
        bnds = [(0.0, 1.0)] * n

        starts = generate_starts(n, ts, seed)

        # Prepend initial_h_values if it matches
        if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) == n:
            starts = [np.array(initial_h_values, dtype=float)] + starts

        # Limit starts based on remaining time
        max_starts = min(len(starts), max(3, int(tl() / 40)))
        starts = starts[:max_starts]

        for h_s in starts:
            if tl() < 35:
                break

            h_curr = project_box_sum(h_s, ts)
            c_curr = compute_c5(h_curr)

            # Multi-stage L-BFGS-B
            stages = [(40, 30, 250), (100, 80, 350), (250, 150, 400), (600, 400, 500)]
            for beta, lam, maxit in stages:
                if tl() < 20:
                    break
                try:
                    r = minimize(objective_fn, h_curr, args=(n, ts, beta, lam),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-9})
                    h_new = project_box_sum(r.x, ts)
                    c_new = compute_c5(h_new)
                    if c_new < c_curr:
                        h_curr, c_curr = h_new, c_new
                except Exception:
                    pass

            # Coordinate descent polish
            if tl() > 15:
                h_curr = coordinate_descent(h_curr, n, ts, 400, 400, n_passes=2)
                c_curr = compute_c5(h_curr)

            # Second L-BFGS-B pass at higher beta
            if tl() > 15:
                try:
                    r = minimize(objective_fn, h_curr, args=(n, ts, 1000, 800),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': 400, 'ftol': 1e-17, 'gtol': 1e-10})
                    h_new = project_box_sum(r.x, ts)
                    c_new = compute_c5(h_new)
                    if c_new < c_curr:
                        h_curr, c_curr = h_new, c_new
                except Exception:
                    pass

            # Final coordinate descent
            if tl() > 12:
                h_curr = coordinate_descent(h_curr, n, ts, 1000, 1000, n_passes=3)
                c_curr = compute_c5(h_curr)

            if c_curr < best_c5:
                best_c5 = c_curr
                best_h = h_curr.copy()
                best_n = n

        # If already below target, stop
        if best_c5 < 0.3803:
            break

    # Final high-precision polish on global best
    if best_h is not None and tl() > 25:
        n = best_n
        ts = n / 2.0
        bnds = [(0.0, 1.0)] * n
        for beta in [2000, 4000]:
            if tl() < 10:
                break
            try:
                r = minimize(objective_fn, best_h, args=(n, ts, beta, 3000),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12})
                h_new = project_box_sum(r.x, ts)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5 = h_new, c_new
            except Exception:
                pass

        if tl() > 15:
            best_h = coordinate_descent(best_h, n, ts, 3000, 3000, n_passes=4)
            best_c5 = compute_c5(best_h)

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_n = n
        best_c5 = compute_c5(best_h)

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
