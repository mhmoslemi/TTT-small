```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        # Distribute among non-saturated coordinates
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            # All saturated, can't fix
            break
    return np.clip(h, 0.0, 1.0)

def corr_gradients(h, n, active_lags):
    """Compute gradients of correlation values at given lags"""
    grads = np.zeros((len(active_lags), n))
    for idx, d in enumerate(active_lags):
        g = np.zeros(n)
        if d >= 0:
            # c(d) = sum_{i=0}^{n-1-d} h[i+d](1-h[i])
            for j in range(n):
                val = 0.0
                if d <= j <= n - 1:
                    val += 1.0
                if j <= n - 1 - d:
                    val -= h[j + d]
                if j >= d:
                    val -= h[j - d]
                g[j] = val
        else:
            dp = -d
            for j in range(n):
                val = 0.0
                if j <= n - 1 - dp:
                    val += 1.0
                if j >= dp:
                    val -= h[j - dp]
                if j <= n - 1 - dp:
                    val -= h[j + dp]
                g[j] = val
        grads[idx] = g
    return grads

def projected_descent_step(h, n, target_sum, alpha=None):
    """Take one projected descent step for max_k c_k"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    max_val = corr.max()
    # Active lags: those within 1% of max
    threshold = max_val - 0.01 * max_val
    active_idx = np.where(corr >= threshold)[0]
    active_lags = active_idx - center
    
    # Compute negative sum of gradients as descent direction
    grads = corr_gradients(h, n, active_lags)
    v = -np.sum(grads, axis=0)
    
    # Project onto sum-zero subspace
    v = v - np.mean(v)
    
    if alpha is None:
        alpha = np.sqrt(n) / (np.linalg.norm(v) + 1e-12)
    
    # Backtracking line search
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(30):
        h_trial = h + alpha * v
        h_trial = project_h(h_trial, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
        alpha *= 0.7
    return best_h, best_c5
```

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

    corr = get_corr(h)
    max_val = corr.max()
    # Active set: lags within 5% of max
    threshold = max_val * 0.95
    active = corr >= threshold
    
    # Subgradient of max: average gradient of active lags
    # For lag d, grad[j] = [d<=j<=n-1] - [j<=n-1-d]*h[j+d] - [j>=d]*h[j-d] (for d>=0)
    # For simplicity, use numerical subgradient
    # Actually, let's use the analytical form
    center = n - 1
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        if not active[center + d]:
            continue
        for j in range(n):
            if d >= 0:
                g = 0.0
                if d <= j <= n - 1:
                    g += 1.0
                if j <= n - 1 - d:
                    g -= h[j + d]
                if j >= d:
                    g -= h[j - d]
            else:
                dp = -d
                g = 0.0
                if j <= n - 1 - dp:
                    g += 1.0
                if j >= dp:
                    g -= h[j - dp]
                if j <= n - 1 - dp:
                    g -= h[j + dp]
            grad[j] += g
    
    # Normalize
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-12:
        return h.copy(), compute_c5(h)
    grad = grad / grad_norm
    
    # Project to sum-zero (feasible direction)
    grad = grad - np.mean(grad)
    
    # Try a few step sizes (backtracking)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for _ in range(20):

        c5_new = compute_c5(h_new)

            best_h = h_new
            best_c5 = c5_new
            alpha *= 1.5  # try bigger step
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                break
    return best_h, best_c5

For a given d (lag), the gradient ∂c(d)/∂h[j] can be computed as:

For d ≥ 0:
