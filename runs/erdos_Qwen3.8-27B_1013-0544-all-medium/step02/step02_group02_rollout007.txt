        length = rng.choice([2, 3, 4, 5, 6, 8])
        shift = rng.choice([1, 2, 3], size=1)[0] * rng.choice([1, -1])
        block_idx = [(start + i) % n for i in range(length)]
        new_vals = [h[(start + shift + i) % n] for i in range(length)]
        old_vals = [h[idx] for idx in block_idx]
        for idx, nv in zip(block_idx, new_vals):
            h[idx] = nv
        h = project_feasible(h, n)
        nc5 = compute_c5(h, n)
        if nc5 < best_c5 - 1e-14:
            best_h = h.copy()
            best_c5 = nc5
        else:
            for idx, ov in zip(block_idx, old_vals):
                h[idx] = ov
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 80
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n0)
    best_n = n0
    h = h0.copy()
    # Phase 1: SLSQP continuation
    for alpha in [200, 2000, 20000, 200000, 1000000]:
        if budget_s - (time.time() - t0) < 60:
            break
        h = do_slsqp(h, n0, alpha, 5000, 1e-16)
        c = compute_c5(h, n0)
        if c < best_c5:
            best_h, best_c5 = h.copy(), c
    h = best_h.copy()
    # Phase 2: Gradient-guided coordinate descent
    h, c = coord_2pt(h, n0, t0, budget_s, rounds=4000, seed=seed)
    if c < best_c5:
        best_h, best_c5 = h.copy(), c
    # Phase 3: Block moves
    if budget_s - (time.time() - t0) > 60:
        h, c = block_move_search(best_h, n0, t0, budget_s, max_iters=2000)
        if c < best_c5:
            best_h, best_c5 = h.copy(), c
    # Phase 4: Try larger n
    for n_try in [100, 120, 160]:
        if budget_s - (time.time() - t0) < 200:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, n_try)
        # Also try structured inits at this n
        x = np.linspace(0, 2, n_try, endpoint=False)
        for freq in [2, 3, 4, 5, 6]:
            for ph in [0.0, np.pi / 2.0, np.pi]:
                s = 0.5 + 0.45 * np.sin(freq * np.pi * x + ph)
                sp = project_feasible(s, n_try)
                if compute_c5(sp, n_try) < compute_c5(h_try, n_try):
                    h_try = sp
        for alpha in [5000, 50000, 500000]:
            if budget_s - (time.time() - t0) < 60:
                break
            h_try = do_slsqp(h_try, n_try, alpha, 4000, 1e-16)
        h_try, c = coord_2pt(h_try, n_try, t0, budget_s, rounds=1500, seed=seed + n_try)
        if c < best_c5:
            best_h, best_c5, best_n = h_try.copy(), c, n_try
    # Phase 5: Coord descent on best_n again with lower temp
    if budget_s - (time.time() - t0) > 30:
        h, c = coord_2pt(best_h, best_n, t0, budget_s, rounds=2000, seed=seed + 999)
        if c < best_c5:
            best_h, best_c5 = h.copy(), c
    # Phase 6: Final SLSQP polish
    for alpha in [1000000, 3000000]:
        if budget_s - (time.time() - t0) < 15:
            break
        h_pol = do_slsqp(best_h, best_n, alpha, 6000, 1e-18)
        c = compute_c5(h_pol, best_n)
        if c < best_c5:
            best_h, best_c5 = h_pol.copy(), c
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
