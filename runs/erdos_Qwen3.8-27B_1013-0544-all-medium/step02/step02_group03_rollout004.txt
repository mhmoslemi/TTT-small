from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        idx = np.arange(len(h))
        # Add to elements not at boundary
        boundary = np.isclose(h, 0) | np.isclose(h, 1)
        free = ~boundary
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
        else:
            h += diff / len(h)
            h = np.clip(h, 0, 1)
    return h

def weighted_gradient(h, w):
    """Compute gradient of sum_m w[m] * corr[m] where corr = correlate(h, 1-h, 'full')"""
    n = len(h)
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if w[m] < 1e-10:
            continue
        d = m - (n - 1)
        # corr[m] = sum_{i: 0<=i<n, 0<=i+d<n} h[i]*(1-h[i+d])
        # = sum h[i] over valid i - sum h[i]*h[i+d] over valid i
        # Valid range for i: max(0,-d) <= i <= min(n-1, n-1-d)
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        # First term: d(h[i])/dh[j] = 1 for i=j in valid range
        grad[i_lo:i_hi+1] += w[m]
        # Second term: d(h[i]*h[i+d])/dh[j]
        # If j = i (first factor): contribution is h[j+d] for j in valid range
        # If j = i+d (second factor): contribution is h[j-d] for j-d in valid range, i.e., j in valid range + d
        for i in range(i_lo, i_hi+1):
            grad[i] -= w[m] * h[i + d]  # derivative w.r.t. h[i]
            grad[i + d] -= w[m] * h[i]  # derivative w.r.t. h[i+d]
    return grad

from scipy.optimize import minimize, Bounds
import time

# Helper functions (top-level)
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def overlaps_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / beta

def project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = ts - h.sum()
        if abs(d) < 1e-15: break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0: break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def wgrad(h, w):
    """Gradient of sum_m w[m]*corr[m] w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    for m in range(2*n-1):
        wm = w[m]
        if wm < 1e-12: continue
        d = m - (n-1)
        lo = max(0, -d)
        hi = min(n-1, n-1-d)
        g[lo:hi+1] += wm
        idx = np.arange(lo, hi+1)
        g[idx] -= wm * h[idx+d]
        g[idx+d] -= wm * h[idx]
    return g

def weighted_obj(h, w, dx):
    return np.dot(w, np.correlate(h, 1-h, mode='full')) * dx

def weighted_obj_grad(h, w, dx):
    return wgrad(h, w) * dx

    return m + np.log(np.sum(np.exp(s))) / beta

def project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break

    g = np.zeros(n)
    for m in range(2 * n - 1):

        if wm < 1e-12:

        d = m - (n - 1)
        lo = max(0, -d)
        hi = min(n - 1, n - 1 - d)
        g[lo:hi + 1] += wm
        idx = np.arange(lo, hi + 1)
        g[idx] -= wm * h[idx + d]
        g[idx + d] -= wm * h[idx]
    return g

    def obj(h):
        return smooth_max(np.correlate(h, 1 - h, mode='full'), beta) * dx
    def objg(h):
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.max(corr)
        s = np.clip(beta * (corr - m), -500.0, 500.0)
        probs = np.exp(s) / np.sum(np.exp(s))
        return dx * wgrad(h, probs)

        return h.sum() - ts

        return np.ones(n)

    h_cur = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    n_lags = 2 * n - 1
    t_start = time.time()
    for outer in range(40):
        if time.time() - t_start > time_limit or (time_limit - (time.time() - t_start)) < 15:
            break
        ov = overlaps_of(h_cur)
        temp = 3.0 / (1.0 + 0.05 * outer)
        w = np.exp(ov / (temp * max(ov.max(), 1e-10)))
        w /= w.sum()
        h_new = slsqp_weighted(h_cur, w, dx, n, ts, bounds, maxiter=400)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_cur = h_new
        # Occasionally also run smooth-max
        if outer % 3 == 2 and time.time() - t_start < time_limit - 15:
            h_sm = slsqp_smooth(h_cur, dx, 300, n, ts, bounds, maxiter=400)
            c5s = compute_c5(h_sm)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = h_sm.copy()
            h_cur = h_sm
    return best_h, best_c5

def subgrad_polish(h0, n, ts, time_limit):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    t_start = time.time()
    lr = 0.003
    for it in range(50000):
        if time.time() - t_start > time_limit:
            break
        ov = overlaps_of(h)
        w = np.exp(ov / max(ov.max(), 1e-10) * 2.0)
        w /= w.sum()
        grad = wgrad(h, w) * dx
        lr_t = lr / (1.0 + 0.0003 * it)
        h = project(h - lr_t * grad, ts)
        if it % 2000 == 1999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(70, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Quick SLSQP from diverse starts at multiple resolutions
    starts = [(h_parent, len(h_parent))]
    for nt in [56, 64, 70, 80, 96, 112, 128]:
        starts.append((init_cosine(nt, 1), nt))
        starts.append((init_mixed(nt, seed + nt), nt))
        starts.append((init_random_binary(nt, seed + 100 + nt), nt))

    for h_s, ns in starts:
        if remaining() < 25:
            break
        ts = ns / 2.0
        bounds = Bounds(np.zeros(ns), np.ones(ns))
        for beta in [50, 200]:
            if remaining() < 12:
                break
            try:
                hn = slsqp_smooth(h_s, 2.0 / ns, beta, ns, ts, bounds, maxiter=400)
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hn.copy()
                    n_best = ns
                h_s = hn
            except Exception:
                pass

    # Phase 2: Adversarial optimization (key innovation)
    if remaining() > 150:
        n_a = n_best
        ts_a = n_a / 2.0
        dx_a = 2.0 / n_a
        bounds_a = Bounds(np.zeros(n_a), np.ones(n_a))
        time_alloc = min(remaining() * 0.35, 300)
        best_h, best_c5 = adversarial_optimize(best_h, n_a, ts_a, bounds_a, time_alloc)
        n_best = n_a

    # Phase 3: Subgradient polish
    if remaining() > 60:
        time_alloc2 = min(remaining() * 0.25, 150)
        n_s = len(best_h)
        best_h, best_c5 = subgrad_polish(best_h, n_s, n_s / 2.0, time_alloc2)

    # Phase 4: Multi-resolution refinement
    if remaining() > 50:
        for n_try in [n_best + 8, n_best - 8, n_best + 16, n_best + 32, n_best - 16]:
            if n_try < 32 or remaining() < 20:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = project(resample_h(best_h, n_try), ts_t)
            bounds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta in [200, 1000, 5000]:
                if remaining() < 10:
                    break
                try:
                    hn = slsqp_smooth(h_t, dx_t, beta, n_try, ts_t, bounds_t, maxiter=800)
                    c5 = compute_c5(hn)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hn.copy()
                        n_best = n_try
                    h_t = hn
                except Exception:
                    pass
            # Also try adversarial at this resolution
            if remaining() > 30:
                best_h_n, best_c5_n = adversarial_optimize(h_t, n_try, ts_t, bounds_t,
                                                           min(remaining() * 0.3, 60))
                if best_c5_n < best_c5:
                    best_c5 = best_c5_n
                    best_h = best_h_n.copy()
                    n_best = n_try

    # Phase 5: Final high-beta polish
    if remaining() > 15:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        for beta in [5000, 20000, 100000]:
            if remaining() < 8:
                break
            try:
                hn = slsqp_smooth(best_h, dx_f, beta, n_f, ts_f, bounds_f, maxiter=5000)
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hn.copy()
            except Exception:
                pass

    # Phase 6: Final subgradient nudge
    if remaining() > 10:
        n_g = len(best_h)
        best_h, best_c5 = subgrad_polish(best_h, n_g, n_g / 2.0, remaining() - 5)

    return best_h, best_c5, len(best_h)
