def grad_lag(h, d):
    """Gradient of correlation at lag d w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad = 1 - 2*h
    elif d > 0:
        # (1-h[j-d]) for j in [d, n-1]
        grad[d:] += (1 - h[:n-d])
        # -h[j+d] for j in [0, n-1-d]
        grad[:n-d] -= h[d:]
    else:
        lag = -d
        # (1-h[j+lag]) for j in [0, n-1-lag]
        grad[:n-lag] += (1 - h[lag:])
        # -h[j-lag] for j in [lag, n-1]
        grad[lag:] -= h[:n-lag]
    return grad

            h2, c5_2 = slsqp_polish(best_h, n2 / 2.0, 0.13 * budget_s)
            if c5_2 < best_c5:
                best_h, best_c5 = h2.copy(), c5_2

    # Phase 4: Upsample to 3*n0 (20% time)
    n3 = 3 * n0
    if remaining() > 0.15 * budget_s:
        h3 = upsample(best_h, n3)
        h3 = project_h(h3, n3 / 2.0)
        alloc3 = 0.12 * budget_s
        h3, c5_3 = subgrad_optimize(h3, alloc3)
        if c5_3 < best_c5:
            best_h, best_c5 = h3.copy(), c5_3
        if remaining() > 50:
            h3, c5_3 = slsqp_polish(best_h, n3 / 2.0, 0.08 * budget_s)
            if c5_3 < best_c5:
                best_h, best_c5 = h3.copy(), c5_3

    # Phase 5: Multi-start perturbation on best (10% time)
    if remaining() > 60:
        nbest = len(best_h)
        target_b = nbest / 2.0
        for attempt in range(6):
            if remaining() < 25:
                break
            h_p = best_h.copy()
            n_pert = np.random.randint(8, max(9, nbest // 5))
            idx = np.random.choice(nbest, n_pert, replace=False)
            h_p[idx] += np.random.randn(n_pert) * 0.06
            h_p = project_h(h_p, target_b)
            # Short subgradient
            t_short = min(12, remaining() / 3.0)
            h_p, c5_p = subgrad_optimize(h_p, t_short)
            if c5_p < best_c5:
                best_h, best_c5 = h_p.copy(), c5_p
            if remaining() < 20:
                break
            # Quick SLSQP
            h_p, c5_p = slsqp_polish(best_h, target_b, min(15, remaining() / 3.0))
            if c5_p < best_c5:
                best_h, best_c5 = h_p.copy(), c5_p

    # Final polish
    if remaining() > 15:
        nf = len(best_h)
        best_h, best_c5 = slsqp_polish(best_h, nf / 2.0, max(5, remaining() - 5))

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
