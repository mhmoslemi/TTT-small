    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def grad_multi(h, top_indices, n, weights):
    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        grad += weights[i] * grad_for_lag(h, k, n)
    return grad

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    c = np.correlate(h_flat, 1.0 - h_flat, mode='full')

def con_eq_jac(h_flat, target_sum):

def find_descent_direction(h, n, top_k_indices):
    """Find direction d that minimizes max_k grad_k · d, ||d||_inf ≤ 1, sum(d)=0."""
    # This is: min_t s.t. grad_k · d ≤ t for all k, -1≤d≤1, sum(d)=0
    # We can solve this with scipy.optimize.linprog or just use the weighted average
    # For simplicity, use the minimum norm solution in the subdifferential
    grads = np.array([grad_for_lag(h, k, n) for k in top_k_indices])
    # The direction is -projection of 0 onto the convex hull of grads
    # Simple approximation: negative weighted average with weights = exp(-grad_norm)
    # Better: just use -grads.mean() if all are similar
    # Even better: solve the LP
    from scipy.optimize import linprog
    # min t
    # s.t. grad_k · d ≤ t for each k
    #      -1 ≤ d_i ≤ 1
    #      sum(d) = 0
    m = len(top_k_indices)
    # Variables: d_0,...,d_{n-1}, t (total n+1 variables)
    # Objective: minimize t => c = [0,...,0,1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    # Inequality constraints: A_ub @ x ≤ b_ub
    # grad_k · d ≤ t => grad_k · d - t ≤ 0
    A_ub = np.zeros((m, n+1))
    for i, k in enumerate(top_k_indices):
        A_ub[i, :n] = grads[i]
        A_ub[i, -1] = -1.0
    b_ub = np.zeros(m)
    # -1 ≤ d_i ≤ 1 => d_i ≤ 1, -d_i ≤ 1
    A_box = np.zeros((2*n, n+1))
    A_box[:n, :n] = np.eye(n)
    A_box[n:, :n] = -np.eye(n)
    b_box = np.concatenate([np.ones(n), np.ones(n)])
    A_ub_full = np.vstack([A_ub, A_box])
    b_ub_full = np.concatenate([b_ub, b_box])
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    res = linprog(c, A_ub=A_ub_full, b_ub=b_ub_full, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n]
    return -grads.mean(axis=0)

            corr = get_corr(h)
            top_k = np.argsort(corr)[-n_top:]
            top_k = top_k[np.argsort(corr[top_k])[::-1]]
            d, t_val = find_lp_direction(h, n, top_k)
            if t_val >= 0:
                d = -grad_for_lag(h, top_k[0], n)
            # Normalize direction
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                d = d / dn
            # Line search
            best_alpha = base_lr / (1.0 + it * 0.0001)
            best_obj = compute_c5(h)
            for alpha_mult in [1.0, 0.5, 0.2, 2.0]:
                alpha = best_alpha * alpha_mult
                h_try = project_feasible(h + v + alpha * d, target_sum)
                obj_try = compute_c5(h_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_alpha = alpha
            v = mom * v + best_alpha * d
            h = project_feasible(h + v, target_sum)
            if it % 500 == 499:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 2: Momentum subgradient (simpler, robust)
    if time_left() > 150:
        h = best_h.copy()
        v = np.zeros(n)
        lr = 0.006
        for it in range(80000):
            if time_left() < 80:
                break
            corr = get_corr(h)
            top_k = np.argsort(corr)[-3:]
            top_k = top_k[np.argsort(corr[top_k])[::-1]]
            cv = corr[top_k]
            w = np.exp(10.0 * (cv - cv[0]))
            w /= w.sum()
            g = sum(w[i] * grad_for_lag(h, top_k[i], n) for i in range(len(top_k)))
            v = 0.8 * v - lr * g
            h = project_feasible(h + v, target_sum)
            lr *= 0.9997
            if it % 1000 == 999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 3: SLSQP with smooth max continuation
    if time_left() > 120:
        for beta in [20, 100, 500, 2000, 10000]:
            if time_left() < 50:
                break
            for trial in range(3):
                if time_left() < 25:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * (0.008 / trial)
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 4: Penalty-based SLSQP
    if time_left() > 80:
        for pen_w in [500.0, 5000.0]:
            if time_left() < 30:
                break
            for beta in [1000, 10000]:
                if time_left() < 15:
                    break
                try:
                    res = minimize(obj_smooth_pen, best_h, args=(dx, beta, target_sum, pen_w),
                                   method='SLSQP', bounds=bounds,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 5: Resolution exploration
    if time_left() > 120:
        for n_try in [n - 10, n + 10, n - 20, n + 20, n + 40, n - 5, n + 5]:
            if n_try < 30 or time_left() < 60:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (ts_t,)}]
            # Quick subgradient
            h_sw = h_t.copy()
            v_sw = np.zeros(n_try)
            lr_sw = 0.005
            for it in range(4000):
                if time_left() < 50:
                    break
                corr_sw = get_corr(h_sw)
                top_k_sw = np.argsort(corr_sw)[-3:]
                top_k_sw = top_k_sw[np.argsort(corr_sw[top_k_sw])[::-1]]
                g_sw = sum(grad_for_lag(h_sw, top_k_sw[i], n_try) for i in range(len(top_k_sw)))
                v_sw = 0.85 * v_sw - lr_sw * g_sw
                h_sw = project_feasible(h_sw + v_sw, ts_t)
                lr_sw *= 0.999
                if it % 500 == 499:
                    c5_sw = compute_c5(h_sw)
                    if c5_sw < best_c5:
                        best_h = h_sw.copy()
                        best_c5 = c5_sw
                        n, dx, target_sum, bounds, cons = n_try, dx_t, ts_t, bnds_t, cons_t
            # SLSQP
            if time_left() < 30:
                continue
            for beta in [2000, 20000]:
                if time_left() < 20:
                    break
                try:
                    res = minimize(obj_smooth, h_sw, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n, dx, target_sum, bounds, cons = n_try, dx_t, ts_t, bnds_t, cons_t
                except Exception:
                    pass

    # Phase 6: Final ultra-fine single-lag subgradient
    if time_left() > 50:
        h = best_h.copy()
        v = np.zeros(n)
        lr = 0.0015
        for it in range(200000):
            if time_left() < 10:
                break
            corr = get_corr(h)
            k_worst = np.argmax(corr)
            g = grad_for_lag(h, k_worst, n)
            v = 0.93 * v - lr * g
            h = project_feasible(h + v, target_sum)
            lr *= 0.99997
            if it % 5000 == 4999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 7: Final high-beta SLSQP
    if time_left() > 15:
        for beta in [50000, 200000]:
            if time_left() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
