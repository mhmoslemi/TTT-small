    n = len(h)
    # For lag 0: coefficient is (1 - 2*h[j])
    # For lag d (d>0): coefficient is (1 - h[j-d] - h[j+d]) with boundary handling

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -200, 200)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def obj_func(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / len(h))

def make_con(n):
    target = n / 2.0
    def f(x):
        return np.sum(x) - target
    def j(x):
        return np.ones(n)
    return {'type': 'eq', 'fun': f, 'jac': j}

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def gen_sine(n, freq, amp, phase=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    return 0.5 + amp * np.sin(np.pi * freq * x + phase)

def gen_multisine(n):
    x = np.linspace(0, 2, n, endpoint=False)
    return 0.5 + 0.2 * np.sin(np.pi * x) + 0.15 * np.sin(2 * np.pi * x) + 0.1 * np.cos(3 * np.pi * x)

def gen_step(n, n_blocks=5, on_frac=0.4):
    h = np.zeros(n)
    block_size = n // n_blocks
    n_on = max(1, int(n * on_frac))
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        n_on_b = max(1, int((end - start) * on_frac))
        perm = (np.arange(start, end) + np.random.permutation(end - start)) % n
        h[perm[:n_on_b]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def tleft():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(70, 0.5)
    n_parent = len(h_parent)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coarse optimization at small n to find structure
    if tleft() > 800:
        for n_coarse in [20, 30]:
            if tleft() < 780:
                break
            h_c = interpolate_h(h_parent, n_coarse)
            h_c = project_h(h_c, n_coarse / 2.0)
            bounds_c = [(0.0, 1.0)] * n_coarse
            con_c = make_con(n_coarse)
            for beta in [50, 150, 400]:
                if tleft() < 770:
                    break
                try:
                    res = minimize(obj_func, h_c, args=(beta,), method='SLSQP',
                                   bounds=bounds_c, constraints=[con_c],
                                   options={'maxiter': 1500, 'ftol': 1e-14})
                    h_c = project_h(res.x, n_coarse / 2.0)
                except Exception:
                    pass
            # Upsample and use as candidate
            h_up = interpolate_h(h_c, n_parent)
            h_up = project_h(h_up, n_parent / 2.0)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()

    # Phase 2: SLSQP from diverse starting points
    n_candidates = [n_parent]
    for n_try in [50, 60, 80, 100]:
        if n_try != n_parent:
            n_candidates.append(n_try)

    for n_pts in n_candidates:
        if tleft() < 200:
            break
        target_sum = n_pts / 2.0
        bounds = [(0.0, 1.0)] * n_pts
        con = make_con(n_pts)

        # Generate diverse starts for this n
        starts = []
        if n_pts == n_parent:
            starts.append(h_parent.copy())
        else:
            starts.append(interpolate_h(h_parent, n_pts))
        starts.append(np.full(n_pts, 0.5))
        starts.append(gen_sine(n_pts, 1, 0.35))
        starts.append(gen_sine(n_pts, 2, 0.3))
        starts.append(gen_multisine(n_pts))
        starts.append(gen_step(n_pts, 5, 0.4))
        starts.append(np.random.uniform(0.2, 0.8, n_pts))

        for h0 in starts:
            if tleft() < 160:
                break
            h0 = project_h(h0, target_sum)
            for beta in [40, 80, 160, 320, 600]:
                if tleft() < 140:
                    break
                try:
                    res = minimize(obj_func, h0, args=(beta,), method='SLSQP',
                                   bounds=bounds, constraints=[con],
                                   options={'maxiter': 2000, 'ftol': 1e-14})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                    h0 = h_new
                except Exception:
                    pass
        # Keep best for this n as candidate for SA
        if len(starts) > 0:
            h0_final = project_h(h0, target_sum)
            c5_final = compute_c5(h0_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h0_final.copy()

    # Phase 3: Simulated annealing with worst-lag targeting
    if tleft() > 150:
        n_pts = len(best_h)
        target_sum = n_pts / 2.0
        h_cur = best_h.copy()
        c5_cur = best_c5
        T = 0.012
        no_improve = 0
        max_sa_iters = 40000

        for it in range(max_sa_iters):
            if tleft() < 40:
                break

            h_new = h_cur.copy()
            move = np.random.random()

            if move < 0.20:
                # Gaussian perturbation on random subset
                frac = 0.1 + 0.3 * (T / 0.012)
                mask = np.random.random(n_pts) < frac
                nm = int(mask.sum())
                if nm > 0:
                    h_new[mask] += np.random.randn(nm) * (0.03 + 0.09 * T / 0.012)
            elif move < 0.40:
                # Block swap
                blen = np.random.randint(2, min(10, n_pts // 5 + 1))
                i = np.random.randint(0, n_pts - blen)
                j = np.random.randint(0, n_pts - blen)
                tries = 0
                while abs(i - j) < blen and tries < 20:
                    j = np.random.randint(0, n_pts - blen)
                    tries += 1
                if abs(i - j) >= blen:
                    tmp = h_new[i:i + blen].copy()
                    h_new[i:i + blen] = h_new[j:j + blen]
                    h_new[j:j + blen] = tmp
            elif move < 0.65:
                # Worst-lag balancing (most targeted)
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                k_max = int(np.argmax(corr))
                d = k_max - (n_pts - 1)
                n_adj = np.random.randint(3, min(12, n_pts // 3 + 1))
                if d > 0:
                    idx = np.random.choice(n_pts - d, min(n_adj, n_pts - d), replace=False)
                    for i in idx:
                        diff = h_cur[i] - h_cur[i + d]
                        if abs(diff) > 0.15:
                            step = 0.03 + 0.04 * np.random.random()
                            h_new[i] -= np.sign(diff) * step
                            h_new[i + d] += np.sign(diff) * step
                elif d < 0:
                    da = -d
                    idx = np.random.choice(da, min(n_adj, da), replace=False)
                    for i in idx:
                        diff = h_cur[i + da] - h_cur[i]
                        if abs(diff) > 0.15:
                            step = 0.03 + 0.04 * np.random.random()
                            h_new[i + da] -= np.sign(diff) * step
                            h_new[i] += np.sign(diff) * step
                else:
                    idx = np.random.choice(n_pts, n_adj, replace=False)
                    for i in idx:
                        h_new[i] += 0.05 if h_cur[i] < 0.5 else -0.05
            elif move < 0.80:
                # Second-worst lag targeting
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                k_vals = np.argsort(corr)[::-1][:3]
                k_pick = k_vals[np.random.randint(0, len(k_vals))]
                d = int(k_pick) - (n_pts - 1)
                n_adj = np.random.randint(2, min(8, n_pts // 4 + 1))
                if d > 0 and d < n_pts:
                    idx = np.random.choice(n_pts - d, min(n_adj, n_pts - d), replace=False)
                    for i in idx:
                        step = 0.03 * np.random.random()
                        if h_cur[i] > h_cur[i + d]:
                            h_new[i] -= step
                            h_new[i + d] += step
                elif d < 0 and -d < n_pts:
                    idx = np.random.choice(-d, min(n_adj, -d), replace=False)
                    for i in idx:
                        step = 0.03 * np.random.random()
                        if h_cur[i - d] > h_cur[i]:
                            h_new[i - d] -= step
                            h_new[i] += step
            elif move < 0.90:
                # Circular block shift
                blen = np.random.randint(3, min(9, n_pts // 4 + 1))
                i = np.random.randint(0, n_pts - blen)
                shift = np.random.randint(-4, 5)
                if shift != 0:
                    block = h_new[i:i + blen].copy()
                    h_new[i:i + blen] = np.roll(block, shift)
            else:
                # Multi-point
                nm = np.random.randint(5, min(15, n_pts // 2 + 1))
                idx = np.random.choice(n_pts, nm, replace=False)
                h_new[idx] += np.random.randn(nm) * 0.05

            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)

            delta = c5_new - c5_cur
            if delta < 0 or np.random.random() < np.exp(-min(300.0, delta / T)):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            T *= 0.9992
            if T < 0.0003:
                T = 0.0003
            if no_improve > 400:
                T = max(T, 0.008)
                no_improve = 0
                # Occasionally jump
                if np.random.random() < 0.3:
                    idx = np.random.choice(n_pts, min(10, n_pts // 3), replace=False)
                    h_cur[idx] += np.random.randn(len(idx)) * 0.1
                    h_cur = project_h(h_cur, target_sum)
                    c5_cur = compute_c5(h_cur)

    # Phase 4: Multi-restart SLSQP from perturbed best
    if tleft() > 80:
        n_pts = len(best_h)
        target_sum = n_pts / 2.0
        bounds = [(0.0, 1.0)] * n_pts
        con = make_con(n_pts)

        n_restarts = min(8, int(tleft() // 15))
        for r in range(n_restarts):
            if tleft() < 20:
                break
            h0 = best_h.copy()
            n_pert = np.random.randint(5, max(6, n_pts // 4))
            idx = np.random.choice(n_pts, n_pert, replace=False)
            h0[idx] += np.random.randn(n_pert) * 0.06
            h0 = project_h(h0, target_sum)
            beta = [100, 200, 400, 600][r % 4]
            try:
                res = minimize(obj_func, h0, args=(beta,), method='SLSQP',
                               bounds=bounds, constraints=[con],
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 5: Final high-precision polish
    if tleft() > 15:
        n_pts = len(best_h)
        target_sum = n_pts / 2.0
        bounds = [(0.0, 1.0)] * n_pts
        con = make_con(n_pts)
        for beta in [500, 1000, 2000]:
            if tleft() < 10:
                break
            try:
                res = minimize(obj_func, best_h, args=(beta,), method='SLSQP',
                               bounds=bounds, constraints=[con],
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
