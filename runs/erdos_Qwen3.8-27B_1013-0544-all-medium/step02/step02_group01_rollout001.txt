```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Proportional adjustment on free variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            adj = diff / free.sum()
            h[free] = h[free] + adj
            h = np.clip(h, 0.0, 1.0)
        # Final correction
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            h = h + diff2 / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_val_grad(h, beta):
    """Log-sum-exp of correlation values * dx, and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    grad *= dx
    return val, grad

def lbfgs_lse(h, beta, maxiter=500):
    """L-BFGS-B optimization with log-sum-exp objective and projection."""
    n = len(h)
    target = n / 2.0
    x = h.copy()
    
    def obj(x):
        v, _ = lse_val_grad(x, beta)
        return v
    
    def objg(x):
        _, g = lse_val_grad(x, beta)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, x, jac=objg, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16})
    x = project_feasible(res.x, target)
    return x

def nesterov(h, max_iters, lr0, decay, top_k):
    """Nesterov accelerated subgradient method."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():

            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] = h[free] + diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            h = h + diff2 / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp_ = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp_.any():
            grad[mp_] += (-h[jp[mp_]])
    grad *= dx
    return val, grad


def lbfgs_optimize(h, beta, maxiter=400):
    n = len(h)
    target = n / 2.0
    x = h.copy()

    def obj(x):
        v, _ = lse_val_grad(x, beta)
        return v

    def objg(x):
        _, g = lse_val_grad(x, beta)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, x, jac=objg, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        x = project_feasible(res.x, target)
    except Exception:
        pass
    return x


def nesterov_opt(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp_ = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp_.any():
                grad[mp_] += (-y[jp[mp_]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def slsqp_polish(h, beta, maxiter=400):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = lse_val_grad(x, beta)
        return v

    def objg(x):
        _, g = lse_val_grad(x, beta)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h


def multi_scale_optimize(h_start, t0, budget_s):
    """Optimize at current scale, then try higher scales."""
    best = h_start.copy()
    best_c = compute_c5(best)
    n = len(best)

    # L-BFGS with increasing beta
    for beta in [30, 80, 200, 500]:
        if time.time() - t0 > budget_s - 60:
            break
        h_l = lbfgs_optimize(best, beta, maxiter=300)
        c = compute_c5(h_l)
        if c < best_c:
            best, best_c = h_l, c

    # Try higher resolutions
    for n_new in [300, 500, 800]:
        if time.time() - t0 > budget_s - 150:
            break
        if n_new <= n:
            continue
        h_i = interp_to_n(best, n_new)
        # Nesterov at new resolution
        if time.time() - t0 < budget_s - 100:
            h_n, c_n = nesterov_opt(h_i, 1500, 0.015, 0.9998, 5)
            if c_n < best_c:
                h_i = h_n
        # L-BFGS polish
        for beta in [100, 300, 600]:
            if time.time() - t0 > budget_s - 40:
                break
            h_l = lbfgs_optimize(h_i, beta, maxiter=200)
            c_l = compute_c5(h_l)
            if c_l < best_c:
                h_i = h_l
        c_final = compute_c5(h_i)
        if c_final < best_c:
            best, best_c = h_i, c_final

    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(100, 0.5)

    best_c5 = compute_c5(h_best)

    # Phase 1: Nesterov with several parameter settings
    nesterov_params = [
        (0.04, 0.9997, 5, 3000),
        (0.02, 0.9999, 10, 3000),
        (0.06, 0.9995, 3, 2000),
    ]
    for lr0, decay, top_k, iters in nesterov_params:
        if time.time() - t0 > budget_s - 80:
            break
        h_n, c_n = nesterov_opt(h_best, iters, lr0, decay, top_k)
        if c_n < best_c5:
            h_best, best_c5 = h_n, c_n

    # Phase 2: Multi-scale L-BFGS optimization
    h_ms, c_ms = multi_scale_optimize(h_best, t0, budget_s)
    if c_ms < best_c5:
        h_best, best_c5 = h_ms, c_ms

    # Phase 3: Perturbation restart - try to escape local minimum
    if time.time() - t0 < budget_s - 120:
        n = len(h_best)
        for attempt in range(3):
            if time.time() - t0 > budget_s - 60:
                break
            # Small random perturbation preserving sum
            noise = np.random.randn(n) * 0.02
            noise -= noise.mean()
            h_pert = h_best + noise
            h_pert = project_feasible(h_pert, n / 2.0)
            # Quick optimize
            h_p, c_p = nesterov_opt(h_pert, 1500, 0.03, 0.9998, 5)
            h_l = lbfgs_optimize(h_p, 200, maxiter=200)
            c_l = compute_c5(h_l)
            if c_l < best_c5:
                h_best, best_c5 = h_l, c_l

    # Phase 4: Final SLSQP polish at current best resolution
    if time.time() - t0 < budget_s - 25:
        h_s = slsqp_polish(h_best, 800, maxiter=600)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            h_best, best_c5 = h_s, c_s

    # Phase 5: One more L-BFGS with very high beta if time allows
    if time.time() - t0 < budget_s - 15:
        h_f = lbfgs_optimize(h_best, 1000, maxiter=300)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    return h_best, best_c5, len(h_best)
```
