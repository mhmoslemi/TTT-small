            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (target_sum,)}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def anti_obj(g, t, dx):
    n_half = len(g)
    h = np.concatenate([g, 1.0 - g])
    return smooth_max_obj(h, t, dx)

def anti_slsqp(g, dx, t, maxiter):
    n_half = len(g)
    try:
        res = minimize(
            anti_obj, g.copy(), args=(t, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_half,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return g

def sub_descent_obj(x, h_full, idx, t, dx):
    h_sub = h_full.copy()
    h_sub[idx] = x
    return smooth_max_obj(h_sub, t, dx)

def sub_anti_obj(x, g_full, idx, t, dx):
    g_sub = g_full.copy()
    g_sub[idx] = x
    h_sub = np.concatenate([g_sub, 1.0 - g_sub])
    return smooth_max_obj(h_sub, t, dx)

def group_descent(h, dx, target_sum, group_size, n_rounds, t):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for _ in range(n_rounds):
        idx = np.random.choice(n, group_size, replace=False)
        sub_sum = np.sum(h[idx])
        try:
            res = minimize(
                sub_descent_obj, h[idx].copy(),
                args=(h, idx, t, dx),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * group_size,
                constraints=[{'type': 'eq', 'fun': sum_eq_sub, 'args': (sub_sum,)}],
                options={'maxiter': 60, 'ftol': 1e-10}
            )
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return h, best_c5

def anti_group_descent(g, dx, group_size, n_rounds, t):

    h_full = np.concatenate([g, 1.0 - g])
    best_c5 = compute_c5(h_full, dx)
    for _ in range(n_rounds):
        idx = np.random.choice(n_half, group_size, replace=False)
        try:
            res = minimize(
                sub_anti_obj, g[idx].copy(),

                method='SLSQP',
                bounds=[(0.0, 1.0)] * group_size,
                options={'maxiter': 80, 'ftol': 1e-10}
            )
            g_new = g.copy()
            g_new[idx] = np.clip(res.x, 0.0, 1.0)
            h_new = np.concatenate([g_new, 1.0 - g_new])
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                g = g_new
                best_c5 = c5_new
        except Exception:
            pass
    return g, best_c5

def fourier_anti_init(n_half, n_harm, amp, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 1.0, n_half)
    g = np.full(n_half, 0.5)
    for k in range(1, n_harm + 1):
        a = rng.randn() * amp / np.sqrt(float(k))
        b = rng.randn() * amp / np.sqrt(float(k))
        g += a * np.cos(2.0 * np.pi * k * x) + b * np.sin(2.0 * np.pi * k * x)
    return np.clip(g, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, new_n)
    return np.interp(x_new, x_old, h)

def project_anti(h):
    n_half = len(h) // 2
    g = (h[:n_half] + 1.0 - h[n_half:]) / 2.0
    return np.concatenate([g, 1.0 - g])
