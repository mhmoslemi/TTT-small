```python
    dx = 2.0 / n
    n_half = n - 1
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    coeffs = np.zeros(n_corr)
    consts = corr.copy()
    
    for k in range(n_corr):
        L = k - n_half
        if L >= 0:
            if idx <= n - 1 - L:
                consts[k] -= h[idx] * (1.0 - h[idx + L])
                coeffs[k] += (1.0 - h[idx + L])
            if idx >= L:
                consts[k] -= h[idx - L] * (1.0 - h[idx])
                coeffs[k] += -h[idx - L]
        else:
            L2 = -L
            if idx >= L2:
                i = idx - L2
                consts[k] -= h[idx] * (1.0 - h[i])
                coeffs[k] += (1.0 - h[i])
            if idx <= n - 1 - L2:
                consts[k] -= h[idx + L2] * (1.0 - h[idx])
                coeffs[k] += -h[idx + L2]
    
    a_adj = coeffs * dx
    c_adj = consts * dx
    
    # Generate breakpoints vectorized
    # x_ij = (c_j - c_i) / (a_i - a_j) for all pairs where |a_i - a_j| > eps
    diff_a = a_adj[:, None] - a_adj[None, :]  # shape (n_corr, n_corr)
    diff_c = c_adj[None, :] - c_adj[:, None]  # shape (n_corr, n_corr)
    # x = diff_c / diff_a where diff_a != 0
    mask = np.abs(diff_a) > 1e-12
    # Only upper triangle to avoid duplicates
    triu_mask = np.triu(np.ones((n_corr, n_corr), dtype=bool), k=1)
    mask = mask & triu_mask
    x_vals = np.zeros(mask.shape)
    x_vals[mask] = diff_c[mask] / diff_a[mask]
    valid = mask & (x_vals > 0.0) & (x_vals < 1.0)
    candidates = x_vals[valid]
    
    # Add boundaries and current value
    candidates = np.concatenate([candidates, [0.0, 1.0, h[idx]]])
    
    # Evaluate f(x) = max_k (c_k + a_k * x) for all candidates at once
    # Shape: (n_candidates, n_corr)
    f_all = np.max(c_adj[None, :] + a_adj[None, :] * candidates[:, None], axis=1)
    
    best_idx = int(np.argmin(f_all))
    h[idx] = candidates[best_idx]
    return h
```
