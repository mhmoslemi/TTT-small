Actually, this is getting complicated with the linearization. Let me simplify.

The key insight: at the current point h, the objective value is C5(h) = max_k c[k] * dx. The subgradient of this objective is any convex combination of the gradients of the active constraints (those k that achieve the max).

The cutting plane approach is:
1. At point h0, compute c0[k] for all k.
2. The linear approximation of c[k] at h0 is: c[k](h) ≈ c0[k] + g_k · (h - h0) where g_k = ∇c[k](h0).
3. Solve: min t s.t. c0[k] + g_k · (h - h0) ≤ t/dx for all k, and feasibility.
4. Move to the solution.
5. Repeat.

This is the "outer approximation" or "progressive hedging" style method.

        h_pert = best_h.copy()
        h_pert += np.random.normal(0, 0.05, n)
        h_pert = project_constraint(h_pert, ts)
        for t_val in [100, 300]:
            if time.time() > dl - 120:
                break
            h_new, c5_new = adam_run(h_pert, dx, ts, t_val, 0.3 / t_val, 25000, dl - 120)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                h_pert = h_new.copy()

    # Phase 3: Binary local search
    if time.time() < dl - 60:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(h_bin.sum())
        if n_ones != n // 2:
            oi = np.where(h_bin == 1)[0]
            zi = np.where(h_bin == 0)[0]
            if n_ones > n // 2:
                h_bin[oi[n_ones - n // 2:]] = 0.0
            elif n_ones < n // 2:
                h_bin[zi[:n // 2 - n_ones]] = 1.0
        h_b, c5_b = binary_search(h_bin, dx, 200000, dl - 60)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b

    # Phase 4: Continuous refinement from best binary
    if time.time() < dl - 30:
        h_start = best_h.copy()
        for t_val in [50, 150, 400]:
            if time.time() > dl - 30:
                break
            h_new, c5_new = adam_run(h_start, dx, ts, t_val, 0.5 / t_val, 30000, dl - 30)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                h_start = h_new.copy()

    # Phase 5: SLSQP final polish
    if time.time() < dl - 10:
        def sm_obj(x):
            hx = np.clip(x, 0.0, 1.0)
            c = np.correlate(hx, 1.0 - hx, mode="full")
            cm = c.max()
            tt = 150.0
            return (cm + (1.0 / tt) * np.log(np.sum(np.exp(tt * (c - cm))))) * dx

        def sm_sum(x):
            return np.sum(x) - ts

        try:
            res = minimize(sm_obj, best_h.copy(), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sm_sum}],
                           options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_r = project_constraint(res.x, ts)
                c5_r = compute_c5(h_r, dx)
                if c5_r < best_c5:
                    best_h = h_r
                    best_c5 = c5_r
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_constraint(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
