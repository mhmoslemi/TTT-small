    quad = -cross
    d_lo = t_lo - hi_old
    d_hi = t_hi - hj_old * 0.0 - hi_old + hj_old  # = t_lo - hi_old
    d_lo = t_lo - hi_old
    d_hi = t_hi - hi_old
    d_vals = np.linspace(d_lo, d_hi, n_grid)
    new_corr = corr[:, None] + slope[:, None] * d_vals[None, :] + quad[:, None] * (d_vals[None, :]) ** 2
    vals = np.max(new_corr, axis=0)
    best_idx = int(np.argmin(vals))
    best_d = d_vals[best_idx]
    h[i] += best_d
    h[j] -= best_d
    new_corr = corr + slope * best_d + quad * best_d * best_d
    return new_corr, h

def coord_descent(h, corr, n, rng, max_sweeps=300, deadline=None, n_grid=60):
    best_c5 = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_corr = corr.copy()
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        for idx in range(0, n - 1, 2):
            i = order[idx]
            j = order[idx + 1]
            corr, h, = pair_optimize_fast(corr, h, i, j, n, n_grid)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_corr = corr.copy()
    return best_h, best_corr, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def smooth_max_grad_step(h, lr, alpha, n):
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        return h
    w /= s
    grad = np.zeros(n)
    shifts = np.arange(-(n - 1), n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        sh = shifts[k_idx]
        im = np.arange(n) - sh
        ip = np.arange(n) + sh
        v1 = (im >= 0) & (im < n)
        v2 = (ip >= 0) & (ip < n)
        g = np.zeros(n)
        g[v1] += (1.0 - h[im[v1]])
        g[v2] -= h[ip[v2]]
        grad += w[k_idx] * g
    grad *= (2.0 / n)
    h_new = h - lr * grad
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(200, 0.5)
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    # Phase 1: Coordinate descent at original resolution
    deadline1 = t0 + budget_s * 0.35
    best_h, corr, best_c5 = coord_descent(best_h, corr, n, rng, max_sweeps=800, deadline=deadline1, n_grid=60)
    best_n = len(best_h)
    # Phase 2: Smooth gradient refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_ref = best_h.copy()
        lr = 0.003
        alpha_sm = 100.0
        deadline2 = t0 + budget_s * 0.5
        stall = 0
        while time.time() < deadline2:
            h_new = smooth_max_grad_step(h_ref, lr, alpha_sm, len(h_ref))
            h_new = project_onto_box_sum(h_new, len(h_new) / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                h_ref = h_new.copy()
                stall = 0
                if lr < 0.02:
                    lr *= 1.05
            else:
                stall += 1
                lr *= 0.95
                alpha_sm = min(alpha_sm * 1.1, 500.0)
                if stall > 60:
                    break
    # Phase 3: Multi-resolution search
    remaining = budget_s - (time.time() - t0) - 10
    for n_try in [100, 150, 250]:
        if remaining < 25:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_interp = project_onto_box_sum(h_interp, n_try / 2.0)
        corr_i = np.correlate(h_interp, 1 - h_interp, mode='full')
        phase_dl = time.time() + min(remaining * 0.3, 180.0)
        h_opt, corr_opt, c5_opt = coord_descent(h_interp, corr_i, n_try, rng, max_sweeps=400, deadline=phase_dl, n_grid=60)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
        remaining = budget_s - (time.time() - t0) - 10
    # Phase 4: Random restarts at best resolution
    remaining = budget_s - (time.time() - t0) - 5
    n_final = len(best_h)
    restart_count = 0
    while remaining > 15 and restart_count < 5:
        h_re = best_h + rng.randn(n_final) * 0.05
        h_re = project_onto_box_sum(h_re, n_final / 2.0)
        corr_re = np.correlate(h_re, 1 - h_re, mode='full')
        h_r, _, c5_r = coord_descent(h_re, corr_re, n_final, rng, max_sweeps=150, deadline=None, n_grid=50)
        if c5_r < best_c5:
            best_h = h_re
            best_c5 = c5_r
        remaining = budget_s - (time.time() - t0) - 5
        restart_count += 1
    # Phase 5: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        n_fin = len(best_h)
        corr_fin = np.correlate(best_h, 1 - best_h, mode='full')
        best_h, _, best_c5 = coord_descent(best_h, corr_fin, n_fin, rng, max_sweeps=500, deadline=t0 + budget_s - 3, n_grid=100)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
