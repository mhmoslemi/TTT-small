    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    h += (target_sum - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def ensure_binary(h, target):
    """Ensure h is binary with exactly target ones."""
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n_ones]] = 1
    return h

def sa_and_ls(h, target, time_fn, time_limit, sa_iters=30000, T0=3.0, T1=0.005):
    """SA followed by local search on binary sequence."""
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    cur = float(np.max(corr))
    best_h = h.copy()
    best_val = cur
    
    # SA phase
    for it in range(sa_iters):
        if time_fn() > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / sa_iters)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        h[i] = 0
        h[j] = 1
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        delta = new_max - cur
        if delta < 0 or (T > 1e-10 and np.random.random() < np.exp(-delta / T)):
            cur = new_max
            if cur < best_val:
                best_val = cur
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    # Local search phase
    h = best_h.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    
    no_improve = 0
    while time_fn() < time_limit and no_improve < 50:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        
        improved = False
        for i in ones:
            if improved:
                break
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_max = float(np.max(compute_corr(h)))
                if new_max < cur_max:
                    cur_max = new_max
                    improved = True
                    no_improve = 0
                    break
                h[i] = 1
                h[j] = 0
        if not improved:
            no_improve += 1
    
    return h, cur_max
