```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_all_shifts(h):
    """Compute f_m(h) for all m = 0, ..., n-1"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # mode='full' gives 2n-1 values, centered at n-1
    # corr[n-1+m] = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m]) * dx for m >= 0
    # corr[n-1-m] = sum_{i=m}^{n-1} h[i]*(1-h[i-m]) * dx for m >= 0
    # But we want f_m = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m]) * dx
    # which is corr[n-1+m] for m = 0, 1, ..., n-1
    return corr[n-1:n-1+n]

def compute_gradient(h):
    """Compute the subgradient of max_m f_m(h) at h"""
    n = len(h)
    dx = 2.0 / n
    shifts = compute_all_shifts(h)
    max_val = np.max(shifts)
    # Active set: shifts within a small tolerance of max
    active = np.where(np.abs(shifts - max_val) < 1e-3)[0]

    # df_m/dh[i] = dx * [(1-h[i+m]) * 1_{i+m < n} - h[i-m] * 1_{i-m >= 0}]
    grad = np.zeros(n)

            if i + m < n:

            if i - m >= 0:
                g -= h[i - m]
            grad[i] += g
    grad *= dx / len(active)

        else:

def smooth_objective(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # Log-sum-exp for numerical stability
    shifted = (corr - m) / temp
    max_shifted = np.max(shifted)
    lse = m + temp * (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted))))
    return lse

def smooth_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    max_c = np.max(corr)
    weights = np.exp((corr - max_c) / temp)
    weights /= np.sum(weights)

    grad = np.zeros(n)

    return grad
```

Actually, I realize there might be a much better approach: **use the dual formulation**.

The problem is: min_h max_{k} f_k(h) subject to constraints.

By the minimax theorem (since the problem is convex in the dual variable and the constraints make it well-behaved), we can write:
min_h max_k f_k(h) = max_{p ∈ Δ} min_h ∑_k p_k f_k(h)

where Δ is the simplex over k.

The inner problem: min_h ∑_k p_k f_k(h) subject to 0≤h≤1, ∑h=n/2.

This is a quadratic program (possibly non-convex), but for a fixed p, it's a finite-dimensional problem we can solve with SLSQP or similar.

The outer problem: max_p (value of inner problem) over the simplex.

We can use subgradient ascent for the outer problem:
- ∂/∂p_k (inner value) = f_k(h*(p)) (by envelope theorem)
- So the subgradient of the outer objective w.r.t. p is the vector of f_k values at the optimal h.
- We do: p_new = project_simplex(p + eta * f(h*(p)))

This is essentially the multiplicative weights / exponential weights algorithm!

1. Initialize p = uniform over k (or p concentrated on k=0)
2. For each iteration:
   a. Solve inner QP: h* = argmin ∑ p_k f_k(h) s.t. constraints
   b. Update p: p_k ← p_k * exp(f_k(h*) / τ) (normalized)
3. Return h*

```python
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_all_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr[n-1:]  # shifts 0 to n-1

def project_h(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h0)
    dx = 2.0 / n
    active = p > 1e-10
    k_act = np.where(active)[0]

        s2 = 0.0

            else:
                g[:n-k] -= pk * h[k:]
                g[k:] -= pk * h[:n-k]
        return dx * g
```

```python
        g[:] = dx * (1 - 2*h)
    elif d > 0:
        # f_d = dx * sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # d/dh[m]: (1-h[m-d])*1_{d<=m} - h[m+d]*1_{m<=n-1-d}
        g[d:] += dx * (1 - h[:n-d])    # (1-h[m-d]) for m=d,...,n-1
        g[:n-d] -= dx * h[d:]           # -h[m+d] for m=0,...,n-1-d
    else:
        k = -d
        # f_{-k} = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # d/dh[m]: (1-h[m+k])*1_{m<=n-1-k} - h[m-k]*1_{m>=k}
        g[:n-k] += dx * (1 - h[k:])     # (1-h[m+k]) for m=0,...,n-1-k
        g[k:] -= dx * h[:n-k]           # -h[m-k] for m=k,...,n-1
    return g
```

Now, for the minimax approach with 2n-1 = 359 lags, the p vector has 359 components. The inner QP still has n=180 variables. The objective for the inner QP becomes:

min_h ∑_{d=-(n-1)}^{n-1} p_d * f_d(h)

This is still a quadratic in h, just with more terms. The computation is O(n * number_of_active_lags) per evaluation.

If p is concentrated on, say, 10 lags, then each eval is O(1800) = very fast. If p is spread over all 359 lags, each eval is O(180*359) ≈ 65K, still fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_h(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def inner_qp(h0, p, max_iter=200, ftol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    n_lags = len(p)
    active = p > 1e-10
    k_acts = np.where(active)[0]
    p_acts = p[active]
    c = np.zeros(n)
    for idx in range(len(k_acts)):
        l = k_acts[idx]
        d = l - (n - 1)
        if d >= 0:
            c[d:] += p_acts[idx]
        else:
            c[:n + d] += p_acts[idx]
    def obj(h):
        val = c @ h
        for idx in range(len(k_acts)):
            l = k_acts[idx]
            d = abs(l - (n - 1))
            if d == 0:
                val -= p_acts[idx] * np.sum(h * h)
            else:
                val -= p_acts[idx] * np.dot(h[:n-d], h[d:])
        return dx * val
    def grad(h):
        g = c.copy()
        for idx in range(len(k_acts)):
            l = k_acts[idx]
            d = l - (n - 1)
            pk = p_acts[idx]
            if d == 0:
                g -= 2.0 * pk * h
            elif d > 0:
                g[d:] -= pk * (1.0 - h[:n-d])
                g[:n-d] += pk * h[d:]
            else:
                k = -d
                g[:n-k] -= pk * (1.0 - h[k:])
                g[k:] += pk * h[:n-k]
        return dx * g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: np.sum(v) - n * 0.5, 'jac': lambda v: np.ones(n)}]
    res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    return project_h(np.clip(res.x, 0.0, 1.0), n * 0.5)

def smooth_opt(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    dl = 2 * n - 1
    def obj(h):
        corr = compute_all_lags(h)
        m = np.max(corr)
        s = (corr - m) / temp
        ms = np.max(s)
        return m + temp * (ms + np.log(np.sum(np.exp(s - ms))))
    def grad(h):
        corr = compute_all_lags(h)
        m = np.max(corr)
        w = np.exp((corr - m) / temp)
        w /= np.sum(w)
        g = np.zeros(n)
        for idx in range(dl):
            if w[idx] < 1e-12:
                continue
            d = idx - (n - 1)
            wk = w[idx]
            if d == 0:
                g += wk * (1.0 - 2.0 * h)
            elif d > 0:
                g[d:] += wk * (1.0 - h[:n-d])
                g[:n-d] -= wk * h[d:]
            else:
                k = -d
                g[:n-k] += wk * (1.0 - h[k:])
                g[k:] -= wk * h[:n-k]
        return (2.0 / n) * g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: np.sum(v) - n * 0.5, 'jac': lambda v: np.ones(n)}]
    res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(np.clip(res.x, 0.0, 1.0), n * 0.5)
    return h_new, compute_c5(h_new)

def t_var_opt(h0, max_iter=500, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    def objf(v):
        return v[-1]
    def ineqf(v):
        hh = v[:n]
        return v[-1] - np.max(np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n))
    def eqf(v):
        return np.sum(v[:n]) - n * 0.5
    def eqj(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    v0 = np.append(h0.copy(), c5_cur + 0.05)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': ineqf}, {'type': 'eq', 'fun': eqf, 'jac': eqj}]
    res = minimize(objf, v0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(np.clip(res.x[:n], 0.0, 1.0), n * 0.5)
    return h_new, compute_c5(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5
    dl = 2 * n - 1

    # Phase 1: Minimax alternating (multiplicative weights)
    p = np.ones(dl) / dl
    temp_mm = 0.03
    for outer in range(50):
        if tl() < 250:
            break
        h = inner_qp(h, p, max_iter=180)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()
        lags = compute_all_lags(h)
        mx = np.max(lags)
        p = p * np.exp((lags - mx) / temp_mm)
        ps = np.sum(p)
        if ps < 1e-300:
            p = np.ones(dl) / dl
        else:
            p /= ps
        temp_mm = max(0.008, temp_mm * 0.82)

    # Phase 2: Smooth SLSQP refinement
    for temp in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
        if tl() < 120:
            break
        h_new, c5_new = smooth_opt(h_best, temp, max_iter=250)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 3: t-variable SLSQP
    if tl() > 150:
        h_new, c5_new = t_var_opt(h_best, max_iter=600, ftol=1e-11)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 4: More minimax with concentrated p
    temp_mm = 0.008
    for outer in range(20):
        if tl() < 150:
            break
        lags = compute_all_lags(h_best)
        mx = np.max(lags)
        p = p * np.exp((lags - mx) / temp_mm)
        p /= np.sum(p)
        h = inner_qp(h_best, p, max_iter=180)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 5: Perturbation + refinement
    for scale in [0.04, 0.02, 0.01, 0.005, 0.002]:
        if tl() < 150:
            break
        for trial in range(3):
            if tl() < 100:
                break
            h_p = h_best + rng.randn(n) * scale
            h_p = project_h(h_p, n * 0.5)
            h_new, c5_new = t_var_opt(h_p, max_iter=300, ftol=1e-10)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Phase 6: Final smooth polish
    if tl() > 60:
        h_new, c5_new = smooth_opt(h_best, 0.001, max_iter=300)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
    if tl() > 40:
        h_new, c5_new = t_var_opt(h_best, max_iter=400, ftol=1e-12)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    h_best = project_h(h_best, n * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, n
```
