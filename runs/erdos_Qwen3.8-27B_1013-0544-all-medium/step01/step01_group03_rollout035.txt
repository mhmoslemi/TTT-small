import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_init, max_flips=10000):
    """Find a good binary sequence by local search."""
    n = len(h_init)
    target = n / 2
    # Start from a threshold
    h = (h_init >= 0.5).astype(float)
    h = fix_sum(h, target)
    h = (h >= 0.5).astype(float)  # Make truly binary
    # Ensure exactly n/2 ones
    if h.sum() != target:
        idx = np.argsort(h_init)
        h = np.zeros(n)
        h[idx[target.astype(int):]] = 1.0
    
    best_corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(best_corr)
    
    for _ in range(max_flips):
        # Randomly try flipping a bit
        j = np.random.randint(n)
        if h[j] == 1.0:
            # Turn off: check if sum would be < target
            if h.sum() - 1 < target:
                continue
        else:
            if h.sum() + 1 > target:
                continue
        
        h[j] = 1.0 - h[j]
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(corr)
        
        if new_max < best_max:
            best_max = new_max
            best_corr = corr
        else:
            h[j] = 1.0 - h[j]
    
    return h

def coordinate_step(h, j, corr):
    """Solve the 1D minimax problem for coordinate j."""
    n = len(h)
    offset = n - 1
    old_j = h[j]
    
    # Compute coefficient a[m] for m from -(n-1) to n-1
    a = np.ones(2 * n - 1)
    # a[m + offset] = 1 - h[j+m] (if valid) - h[j-m] (if valid)
    for m in range(-(n-1), n):
        if 0 <= j + m < n:
            a[m + offset] -= h[j + m]
        if 0 <= j - m < n:
            a[m + offset] -= h[j - m]
    
    # Find t in [-old_j, 1-old_j] that minimizes max_k (corr[k] + t * a[k])
    t_lo = -old_j
    t_hi = 1.0 - old_j
    
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2.0
        vals = corr + t_mid * a
        k_max = np.argmax(vals)
        if a[k_max] > 1e-15:
            t_hi = t_mid
        elif a[k_max] < -1e-15:
            t_lo = t_mid
        else:
            t_lo = t_hi = t_mid
            break
    
    return np.clip(old_j + (t_lo + t_hi) / 2.0, 0.0, 1.0)

def coord_descent(h, max_iters=50, random_order=False, seed=0):
    """Coordinate descent for the C5 problem."""
    n = len(h)
    h = np.array(h, dtype=float).copy()
    h = fix_sum(h, n / 2.0)
    rnd = np.random.RandomState(seed)
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = rnd.permutation(n) if random_order else np.arange(n)
        for j in order:
            new_val = coordinate_step(h, j, corr)
            if abs(new_val - h[j]) > 1e-14:
                h[j] = new_val
                # Update corr incrementally (or just recompute)
                corr = np.correlate(h, 1.0 - h, mode='full')
        # Fix sum periodically
        if it % 5 == 4:
            h = fix_sum(h, n / 2.0)
    
    h = fix_sum(h, n / 2.0)
    return h

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Coordinate descent from initial
    if rem() > 30:
        h_cd = coord_descent(best_h, max_iters=30, random_order=False, seed=seed)
        c5 = evaluate_c5(h_cd)
        if c5 < best_c5:
            best_h, best_c5 = h_cd, c5
    
    # Phase 2: Binary local search at various resolutions
    for n in [60, 80, 100, 120, 140, 160]:
        if rem() < 20:
            break
        # Interpolate to this resolution
        h_n = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_h)), best_h)
        h_n = fix_sum(h_n, n / 2.0)
        
        # Binary local search
        h_bin = binary_local_search(h_n, max_flips=5000)
        c5_bin = evaluate_c5(h_bin)
        
        # Then continuous refinement
        h_ref = coord_descent(h_bin, max_iters=20, seed=seed + n)
        c5_ref = evaluate_c5(h_ref)
        
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
    
    # Phase 3: SLSQP refinement on best
    if rem() > 20:
        n = len(best_h)
        target = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max_val(corr, 2000)
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:

            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_h, best_c5 = h_final, c5_final

    # Phase 6: Try pushing toward binary and back
    if rem() > 20:
        n = len(best_h)
        idx_sorted = np.argsort(best_h)
        h_bin = np.zeros(n)
        h_bin[idx_sorted[n // 2:]] = 1.0
        h_back = coord_descent(h_bin, max_sweeps=40, lam=500.0, seed=seed + 777)
        c5_back = evaluate_c5(h_back)
        if c5_back < best_c5:
            best_h, best_c5 = h_back, c5_back

    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_final
