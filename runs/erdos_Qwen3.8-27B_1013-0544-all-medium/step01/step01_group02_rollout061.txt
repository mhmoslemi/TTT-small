from scipy.optimize import linprog

def slp_iteration(h, n, t, step_size=1.0):
    """One SLP iteration."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    n_lags = len(c)
    
    # Compute Jacobian J[k, j] = dc_k/dh_j
    # c_ell = sum_i h[i]*(1-h[i+ell]) for valid i
    # J[ell+n-1, j] = (1-h[j+ell])*I(0<=j+ell<n) - h[j-ell]*I(0<=j-ell<n)
    J = np.zeros((n_lags, n))
    for ell in range(-(n-1), n):
        k = ell + n - 1
        for j in range(n):
            val = 0.0
            if 0 <= j + ell < n:
                val += (1 - h[j + ell])
            if 0 <= j - ell < n:
                val -= h[j - ell]
            J[k, j] = val
    
    # Linearize: c_k ≈ c_k + J[k,:] @ (h' - h)
    # Constraint: c_k + J[k,:] @ (h' - h) <= t
    # => J[k,:] @ h' <= t - c_k + J[k,:] @ h
    
    rhs_J = t - c + J @ h
    
    # LP: minimize t (or minimize sum of h' or something)
    # Variables: [h'(0), ..., h'(n-1), t]
    # Objective: minimize t (index n)
    # Constraints:
    #   J @ h' - t <= rhs_J  (i.e., J @ [h', -t/1] <= rhs_J... let me set up properly)
    
    # Actually: J @ h' <= rhs_J + t
    # => J @ h' - t <= rhs_J
    # In standard form for linprog (minimize c^T x s.t. A x <= b):
    # x = [h'(0),...,h'(n-1), t]
    # A row for each k: [J[k,:], -1] <= rhs_J[k]
    # A row for sum: [1,...,1, 0] <= n/2 (and >= n/2)
    # Bounds: 0 <= h'[i] <= 1, t free (or t >= 0)
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.zeros((n_lags + 1, n+1))
    b_ub = np.zeros(n_lags + 1)
    for k in range(n_lags):
        A_ub[k, :n] = J[k, :]
        A_ub[k, n] = -1.0
        b_ub[k] = rhs_J[k]
    # Sum constraint: sum(h') = n/2
    # This is an equality, so use A_eq
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[n]
        # Damping
        h_damped = (1 - step_size) * h + step_size * h_new
        h_damped = project_feasible(h_damped, n)
        return h_damped, t_new
    return h, t

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(d) < 1e-14:
            break

            if t < 1e-15:

            h += (d / t) * room
        else:
            t = h.sum()
            if t < 1e-15:

            h += (d / t) * h

from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_c_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h += (d / t) * room
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def compute_jacobian(h, n):
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    jr = np.arange(n)
    for ell in range(-(n - 1), n):
        k = ell + n - 1
        v1 = (jr + ell >= 0) & (jr + ell < n)
        J[k, v1] = (1.0 - h[jr[v1] + ell])
        v2 = (jr - ell >= 0) & (jr - ell < n)
        J[k, v2] -= h[jr[v2] - ell]
    return J

def smooth_obj(x, n, alpha, top_m):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_m < len(c):
        idx = np.argpartition(c, -top_m)[-top_m:]
        c = c[idx]
    m = c.max()
    x_a = alpha * (c - m)
    xm = x_a.max()
    lse = xm + np.log(np.sum(np.exp(x_a - xm)))
    return m + lse / alpha
