```python
    corr = np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-boundary elements
        mask = (h > 0.01) & (h < 0.99)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_corr(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    # Soft max approximation
    shifted = t * (corr - max_c)
    val = (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx
    return val
```

```python
def active_set_optimization(h, dx, target_sum, n_active=5, n_rounds=50):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    
    for rnd in range(n_rounds):
        # Identify active lags
        corr = np.correlate(h, 1-h, mode="full")
        max_val = np.max(corr)
        
        # Active lags: those within 5% of max
        threshold = max_val * 0.95
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            active = [np.argmax(corr)]
        
        # Optimize to minimize the max of active lags
        # Use smooth max over just the active lags
        def obj(x):
            c = np.correlate(x, 1-x, mode="full")
            a_vals = c[active]
            t = 200
            m = np.max(a_vals)
            s = t * (a_vals - m)
            return (m + (1.0/t) * np.log(np.sum(np.exp(s)))) * dx
        
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)]*n,
                      constraints=[{'type':'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      options={'maxiter': 500, 'ftol': 1e-14})
        
        h_new = fix_sum(res.x, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    
    return h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, t, dx, active_idx=None):
    corr = np.correlate(h, 1 - h, mode="full")
    if active_idx is not None:
        corr = corr[active_idx]
    m = np.max(corr)
    s = t * (corr - m)
    return (m + (1.0/t) * np.log(np.sum(np.exp(s)))) * dx

def penalty_obj(h, theta, beta, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    excess = np.maximum(0, corr - theta)
    return np.sum(excess**2) * dx**2
