    n = len(h)
    s = m - (n - 1)
    g = np.zeros(n)

    """Local search on binary sequences."""
    n = len(h0)

    # Start from current h0 rounded to binary

    h = np.zeros(n)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def grad_corr(h, m, dx):
    n = len(h)
    s = m - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1.0 - h[s:]) * dx
        g[s:] -= h[:n - s] * dx
    else:
        t = -s
        g[t:] += (1.0 - h[:n - t]) * dx
        g[:n - t] -= h[t:] * dx
    return g

def project_h(h, target):
    h = np.asarray(h, dtype=float)

        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 3.0

        else:
