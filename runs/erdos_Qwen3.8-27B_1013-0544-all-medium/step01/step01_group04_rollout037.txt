    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_all_corrs(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.sum(free)
        if n_free > 0:
            h[free] -= diff / n_free
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        # Distribute among all non-bound variables
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) > 0:
            adj = diff / np.sum(free)
            h[free] -= adj
            h = np.clip(h, 0.0, 1.0)

    c = compute_all_corrs(h, dx)
    c_max = np.max(c)
    s = beta * (c - c_max)
    s = np.clip(s, -500, 500)
    val = (c_max + np.log(np.sum(np.exp(s)))) / beta
    val += pen_w * (np.sum(h) - target_sum) ** 2
    return val

def smooth_max_grad(h, dx, beta, target_sum, pen_w):
    n = len(h)
    c = compute_all_corrs(h, dx)
    c_max = np.max(c)
    s = beta * (c - c_max)
    s = np.clip(s, -500, 500)
    p = np.exp(s)
    p /= np.sum(p)
    
    grad = np.zeros(n)

    for idx in range(2 * n - 1):

        shift = idx - offset
        w *= dx
        if shift >= 0:

                grad[:n - shift] += w * (1.0 - h[shift:])
                grad[shift:] -= w * h[:n - shift]
        else:
            s2 = -shift
            if s2 < n:
                grad[s2:] += w * (1.0 - h[:n - s2])
                grad[:n - s2] -= w * h[s2:]
    
    grad += 2.0 * pen_w * (np.sum(h) - target_sum)
    return grad

    s = beta * (c - c_max)
    s = np.clip(s, -500.0, 500.0)
    val = (c_max + np.log(np.sum(np.exp(s)))) / beta
    val += pen_w * (np.sum(x) - target_sum) ** 2
    return val

def smooth_max_grad(x, dx, beta, target_sum, pen_w):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = np.max(c)
    s = beta * (c - c_max)
    s = np.clip(s, -500.0, 500.0)
    p = np.exp(s)
    p /= np.sum(p)
    grad = np.zeros(n)
    offset = n - 1
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-15:
            continue
        shift = idx - offset
        w *= dx
        if shift >= 0:
            if shift < n:
                ln = n - shift
                grad[:ln] += w * (1.0 - x[shift:])
                grad[shift:] -= w * x[:ln]
        else:
            s2 = -shift
            if s2 < n:
                ln = n - s2
                grad[s2:] += w * (1.0 - x[:ln])
                grad[:ln] -= w * x[s2:]
    grad += 2.0 * pen_w * (np.sum(x) - target_sum)
    return grad

def obj_wrapper(x, dx, beta, target_sum, pen_w):
    return smooth_max_val(x, dx, beta, target_sum, pen_w)

def grad_wrapper(x, dx, beta, target_sum, pen_w):
    return smooth_max_grad(x, dx, beta, target_sum, pen_w)

def lbfgs_optimize(h_start, dx, target_sum, beta, pen_w, max_iter=300):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_wrapper, h_start,
                   args=(dx, beta, target_sum, pen_w),
                   jac=grad_wrapper,
                   method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': max_iter, 'gtol': 1e-12, 'ftol': 1e-16})
    return project_feasible(res.x, n)

def upsample_linear(h, n_new):
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)
    n_best = n
    dx_best = dx

    # Phase 1: Optimize at original n (45% time)
    t1_end = t0 + budget_s * 0.45
    beta_list = [5, 10, 20, 50, 100, 200, 500]
    pen_list = [1, 10, 100, 1000]
    it = 0
    while time.time() - t0 < t1_end:
        it += 1
        beta = beta_list[(it * 7) % len(beta_list)]
        pen = pen_list[(it * 3) % len(pen_list)]
        progress = min((time.time() - t0 - t0) / (t1_end - t0), 1.0)
        scale = 0.12 * (1.0 - progress) + 0.005
        if it % 4 == 1:
            h_try = best_h.copy()
        elif it % 4 == 2:
            h_try = project_feasible(best_h + rng.randn(n) * scale, n)
        elif it % 4 == 3:
            h_try = project_feasible(best_h + rng.randn(n) * scale * 0.3, n)
        else:
            h_try = project_feasible(h0 + rng.randn(n) * 0.05, n)
        h_new = lbfgs_optimize(h_try, dx, target, beta, pen, max_iter=400)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
            n_best = n
            dx_best = dx

    # Phase 2: High-beta refinement at original n (20% time)
    t2_end = t0 + budget_s * 0.65
    for beta in [200, 500, 1000, 2000, 5000]:
        for pen in [100, 1000, 10000]:
            if time.time() - t0 > t2_end:
                break
            h_new = lbfgs_optimize(best_h, dx_best, n_best / 2.0, beta, pen, max_iter=500)
            c5_new = compute_c5(h_new, dx_best)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: Upsample to 2n and optimize (25% time)
    t3_end = t0 + budget_s * 0.90
    if time.time() - t0 < t3_end:
        n2 = 2 * n_best
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h_up = project_feasible(upsample_linear(best_h, n2), n2)
        c5_up = compute_c5(h_up, dx2)
        it2 = 0
        while time.time() - t0 < t3_end:
            it2 += 1
            beta3_list = [10, 20, 50, 100, 200, 500]
            pen3_list = [10, 100, 1000, 10000]
            beta = beta3_list[(it2 * 5) % len(beta3_list)]
            pen = pen3_list[(it2 * 3) % len(pen3_list)]
            progress2 = min((time.time() - t0 - t1_end) / max(t3_end - t1_end, 1e-6), 1.0)
            scale2 = 0.06 * (1.0 - progress2) + 0.005
            if it2 % 5 == 1:
                h_try2 = h_up.copy()
            elif it2 % 5 == 2:
                h_try2 = project_feasible(h_up + rng.randn(n2) * scale2, n2)
            elif it2 % 5 == 3:
                h_try2 = project_feasible(upsample_linear(best_h, n2) + rng.randn(n2) * 0.02, n2)
            elif it2 % 5 == 4:
                h_try2 = project_feasible(h_up + rng.randn(n2) * scale2 * 0.3, n2)
            else:
                h_try2 = project_feasible(h_up + rng.randn(n2) * 0.15, n2)
            h_new2 = lbfgs_optimize(h_try2, dx2, target2, beta, pen, max_iter=300)
            c5_new2 = compute_c5(h_new2, dx2)
            if c5_new2 < best_c5 - 1e-13:
                best_h = h_new2
                best_c5 = c5_new2
                n_best = n2
                dx_best = dx2
            if c5_new2 < c5_up - 1e-13:
                h_up = h_new2
                c5_up = c5_new2

    # Phase 4: Final polish with very high beta (remaining time)
    t4_end = t0 + budget_s * 0.97
    target_f = len(best_h) / 2.0
    while time.time() - t0 < t4_end:
        for beta_f in [2000, 5000, 10000]:
            if time.time() - t0 > t4_end:
                break
            h_new = lbfgs_optimize(best_h, dx_best, target_f, beta_f, 10000, max_iter=500)
            c5_new = compute_c5(h_new, dx_best)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                break
        else:
            break

    best_h = project_feasible(best_h, len(best_h))
    final_dx = 2.0 / len(best_h)
    best_c5 = compute_c5(best_h, final_dx)
    return (best_h, best_c5, len(best_h))
