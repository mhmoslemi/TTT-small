            grad /= float(len(active))
            mom = 0.3 * mom - ss * grad
            h_new = project_feasible(h + mom, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                ss *= 0.93
                if ss < 1e-13:
                    break
            h = h_new

    # Phase 5: Random perturbation search
    if tleft() > 20:
        for _ in range(300):
            if tleft() < 3:
                break
            scale = 0.003 + 0.005 * rng.random()
            hp = best_h + rng.normal(0, scale, n_points)
            hp = project_feasible(hp, target_sum)
            c5p = compute_c5(hp, dx)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = hp.copy()
            else:
                # Quick local refinement
                h = hp.copy()
                ss2 = 0.0001
                for _ in range(100):
                    if tleft() < 3:
                        break
                    corr2 = np.correlate(h, 1.0 - h, mode='full')
                    cm2 = float(np.max(corr2))
                    act2 = np.where(corr2 >= cm2 - 1e-4)[0]
                    if len(act2) < 1:
                        act2 = np.array([int(np.argmax(corr2))])
                    if len(act2) > 5:
                        act2 = np.argsort(corr2)[-5:]
                    g2 = np.zeros(n_points)
                    for idx in act2:
                        L = int(idx) - n_half
                        g2 += compute_grad_lag(h, L)
                    g2 /= float(len(act2))
                    h = project_feasible(h - ss2 * g2, target_sum)
                    c5t = compute_c5(h, dx)
                    if c5t < best_c5:
                        best_c5 = c5t
                        best_h = h.copy()
                    else:
                        ss2 *= 0.85
                        if ss2 < 1e-12:
                            break

    return (best_h, best_c5, n_points)
