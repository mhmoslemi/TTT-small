def compute_subgradient(h, active_lags):
    """Compute weighted sum of gradients for active lags.
    active_lags: list of (index_in_output, weight) pairs"""
    n = len(h)
    grad = np.zeros(n)
    for m, w in active_lags:
        k = m - n + 1
        # term 1: (1-h[j-k]) for valid j-k
        # term 2: -h[j+k] for valid j+k
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if k >= 0:
            # j-k valid means j >= k
            term1[k:] = 1 - h[:n-k]
            # j+k valid means j < n-k
            term2[:n-k] = h[k:]
        else:

            term1[:n+k] = 1 - h[-k:]
            # j+k valid means j >= -k
            term2[-k:] = h[:n+k]
        grad += w * (term1 - term2)
    return grad

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(t * (corr - m), -500, 500)
    return (m + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def project_sum_box(y, target_sum):
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def active_subgradient(h, n_active=5, eps=0.001):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    cmax = np.max(corr)
    active = np.where(corr >= cmax - eps)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = np.argsort(corr[active])[::-1][:n_active]
        active = active[idx]
    w = np.ones(len(active)) / len(active)
    grad = np.zeros(n)
    for m, wi in zip(active, w):
        k = m - n + 1
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if k >= 0:
            t1[k:] = 1.0 - h[:n - k]
            t2[:n - k] = h[k:]
        else:
            jk = -k
            t1[:n - jk] = 1.0 - h[jk:]
            t2[jk:] = h[:n - jk]
        grad += wi * (t1 - t2)
    return grad

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

        g = active_subgradient(h, n_active=5, eps=0.0005)
        h_new = project_sum_box(h - step * g, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5

def slsqp_opt(h, dx, target_sum, t=50, maxiter=3000):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def group_descent(h, dx, target_sum, group_size=10, n_rounds=60, t=100, seed=99):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    for r in range(n_rounds):
        for trial in range(4):
            idx = rng.choice(n, group_size, replace=False)
            sub_sum = h[idx].sum()
            h_store = h.copy()
            def sub_obj(x):
                hs = h_store.copy()
                hs[idx] = x
                return smooth_max_obj(hs, t, dx)
            try:
                res = minimize(
                    sub_obj, h[idx].copy(),
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * group_size,
                    constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - sub_sum}],
                    options={'maxiter': 150, 'ftol': 1e-13}

                h_new = h_store.copy()
                h_new[idx] = np.clip(res.x, 0.0, 1.0)

                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
            except Exception:
                pass
    return h, best_c5
