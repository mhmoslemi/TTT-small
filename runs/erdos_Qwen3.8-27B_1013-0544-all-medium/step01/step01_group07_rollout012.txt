def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    if max_c * t > 500:
        # numerical safety
        shifted = t * (corr - max_c)
        logsum = np.log(np.sum(np.exp(shifted)))
    else:
        shifted = t * (corr - max_c)
        logsum = np.log(np.sum(np.exp(shifted)))
    return (max_c + (1.0/t) * logsum) * dx

def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # grad of smooth_max w.r.t. h
    # = sum_m weights[m] * dcorr[m]/dh
    # dcorr[m]/dh[j] = (1-h[j+s_m]) - h[j-s_m] (with bounds)

        grad += w * contrib

def project_onto_simplex_box(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-12:

        if diff > 0:

            if total_room > 1e-15:

            else:

            if total_room > 1e-15:

        h = np.clip(h, 0.0, 1.0)

```python
def project_box_sum(v, lo, hi, target_sum):
    """Project v onto {x: lo<=x[i]<=hi[i], sum(x)=target_sum}."""
    n = len(v)
    # Bisection on lambda: x[i] = clip(v[i] - lambda, lo[i], hi[i])
    # Find lambda such that sum(x) = target_sum
    
    lo_arr = np.full(n, lo)
    hi_arr = np.full(n, hi)
    
    # The function sum(clip(v - lambda, lo, hi)) is continuous and decreasing in lambda
    # Find lambda such that it equals target_sum
    
    def f(lam):
        x = np.clip(v - lam, lo_arr, hi_arr)
        return x.sum() - target_sum
    
    # Find bounds for lambda
    lam_lo = v.min() - hi
    lam_hi = v.max() - lo
    
    # Bisection
    for _ in range(100):
        lam_mid = (lam_lo + lam_hi) / 2
        if f(lam_mid) > 0:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    
    lam = (lam_lo + lam_hi) / 2
    x = np.clip(v - lam, lo_arr, hi_arr)
    # Fix any numerical issues
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # Adjust the element with the most room
        room_hi = hi_arr - x
        room_lo = x - lo_arr
        if diff > 0:
            idx = np.argmax(room_hi)
            x[idx] += min(diff, room_hi[idx])
        else:
            idx = np.argmax(-room_lo)  # most room below
            x[idx] += max(diff, -room_lo[idx])
    return x
```

Wait, this bisection approach works because:
- As lambda → -∞: all x[i] → hi, sum → n*hi
- As lambda → +∞: all x[i] → lo, sum → n*lo
- The function is continuous and non-increasing in lambda

So we need sum = target_sum, and we find the lambda where this happens.

This is O(n * 100) per projection, which is very fast for n=160.

```python
def subgradient_of_max_corr(h, epsilon=0.001):
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    max_c = corr.max()
    active = np.where(corr >= max_c - epsilon)[0]
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n-1)
        j = np.arange(n)
        jp = j + s
        jm = j - s
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        g = np.zeros(n)
        g[mask_p] = 1.0 - h[jp[mask_p]]
        g[mask_m] -= h[jm[mask_m]]
        grad += g
    grad /= len(active)
    return grad
```

        else:
            total = x.sum()
            if total > 1e-15:
                x += diff * x / total
        x = np.clip(x, lo, hi)
    return x

def subgradient_of_corr(h, epsilon=0.002):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = corr.max()
    active = np.where(corr >= max_c - epsilon)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        j = np.arange(n)
        jp = j + s
        jm = j - s
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        g = np.zeros(n)
        g[mask_p] += 1.0 - h[jp[mask_p]]
        g[mask_m] -= h[jm[mask_m]]
        grad += g
    grad /= max(1, len(active))
    return grad

def subgradient_descent(h, dx, target_sum, n_steps=20000, alpha0=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_h = h.copy()
    curr_c5 = best_c5
    alpha = alpha0
    no_improve = 0
    for step in range(n_steps):
        g = subgradient_of_corr(curr_h, epsilon=0.002)
        g = g - g.mean()
        v = curr_h - alpha * g
        new_h = project_box_sum(v, target_sum)
        new_c5 = compute_c5(new_h, dx)
        if new_c5 < curr_c5:
            curr_h = new_h
            curr_c5 = new_c5
            no_improve = 0
            alpha = min(alpha * 1.05, 0.5)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            no_improve += 1
            alpha *= 0.9
            if no_improve > 100:
                alpha = alpha0 * 0.5
                no_improve = 0
        if alpha < 1e-6:
            break
    return best_h, best_c5

def pairwise_cd(h, dx, target_sum, n_sweeps=30):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    hw = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i_idx in range(n - 1):
            i = order[i_idx]
            for j_idx in range(i_idx + 1, n):
                j = order[j_idx]
                s = hw[i] + hw[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-13:
                    continue
                a, b = lo, hi
                x1 = b - gr * (b - a)
                x2 = a + gr * (b - a)
                hw[i] = x1; hw[j] = s - x1
                f1 = np.max(np.correlate(hw, 1 - hw, mode="full"))
                hw[i] = x2; hw[j] = s - x2
                f2 = np.max(np.correlate(hw, 1 - hw, mode="full"))
                for _ in range(35):
                    if f1 < f2:
                        b = x2; x2 = x1; f2 = f1
                        x1 = b - gr * (b - a)
                        hw[i] = x1; hw[j] = s - x1
                        f1 = np.max(np.correlate(hw, 1 - hw, mode="full"))
                    else:
                        a = x1; x1 = x2; f1 = f2
                        x2 = a + gr * (b - a)
                        hw[i] = x2; hw[j] = s - x2
                        f2 = np.max(np.correlate(hw, 1 - hw, mode="full"))
                x_opt = (a + b) / 2.0
                hw[i] = x_opt; hw[j] = s - x_opt
                c5 = np.max(np.correlate(hw, 1 - hw, mode="full")) * dx
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return hw, best_c5

def simulated_annealing(h, dx, target_sum, n_iter=30000, t_start=0.005, t_end=0.00005):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_h = h.copy()
    curr_c5 = best_c5
    t = t_start
    cooling = (t_end / t_start) ** (1.0 / n_iter)
    rng = np.random.RandomState(12345)
    for it in range(n_iter):
        n_move = rng.randint(2, 5)
        idx = rng.choice(n, n_move, replace=False)
        old_vals = curr_h[idx].copy()
        old_sum = old_vals.sum()
        new_vals = old_vals + rng.normal(0, 0.15, n_move)
        new_vals = np.clip(new_vals, 0, 1)
        diff = old_sum - new_vals.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - new_vals
                tr = room.sum()
                if tr > 1e-15:
                    new_vals += diff * room / tr
            else:
                tr = new_vals.sum()
                if tr > 1e-15:
                    new_vals += diff * new_vals / tr
            new_vals = np.clip(new_vals, 0, 1)
            diff2 = old_sum - new_vals.sum()
            new_vals += diff2 / n_move
            new_vals = np.clip(new_vals, 0, 1)
        curr_h[idx] = new_vals
        new_c5 = compute_c5(curr_h, dx)
        if new_c5 < curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(t, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr_h.copy()
        else:
            curr_h[idx] = old_vals
        t *= cooling
    return best_h, best_c5

def make_structured_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    pattern_len = n // 4
    for start in range(0, n, pattern_len):
        end = min(start + pattern_len, n)
        block = h[start:end]
        n_ones = rng.randint(max(1, len(block) // 4), max(2, len(block) * 3 // 4))
        n_ones = min(n_ones, len(block))
        positions = rng.choice(len(block), n_ones, replace=False)
        block[positions] = 1.0
        h[start:end] = block
    ones_needed = n // 2 - int(h.sum())
    if ones_needed > 0:
        zeros = np.where(h < 0.5)[0]
        if len(zeros) >= ones_needed:
            h[rng.choice(zeros, ones_needed, replace=False)] = 1.0
    elif ones_needed < 0:
        ones = np.where(h > 0.5)[0]
        h[rng.choice(ones, -ones_needed, replace=False)] = 0.0
    h = np.clip(h, 0, 1)
    return fix_sum(h, n // 2)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: SLSQP polish with high t
    for t in [200, 500, 1000, 2000]:
        if elapsed() > budget_s - 400:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=1000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Subgradient descent
    if elapsed() < budget_s - 300:
        h_sg, c5_sg = subgradient_descent(best_h, dx, target_sum, n_steps=30000, alpha0=0.08)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg
            # Polish after subgradient
            h_new = slsqp_optimize(best_h, dx, target_sum, t=1000, maxiter=500)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: Pairwise coordinate descent
    if elapsed() < budget_s - 200:
        h_cd, c5_cd = pairwise_cd(best_h, dx, target_sum, n_sweeps=20)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 4: Simulated annealing
    if elapsed() < budget_s - 150:
        h_sa, c5_sa = simulated_annealing(best_h, dx, target_sum, n_iter=20000, t_start=0.008, t_end=0.0001)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa
            # Polish
            h_new = slsqp_optimize(best_h, dx, target_sum, t=1000, maxiter=800)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: More pairwise CD on best
    if elapsed() < budget_s - 100:
        h_cd2, c5_cd2 = pairwise_cd(best_h, dx, target_sum, n_sweeps=25)
        if c5_cd2 < best_c5:
            best_h = h_cd2
            best_c5 = c5_cd2

    # Phase 6: Try structured starts
    if elapsed() < budget_s - 80:
        for ss in range(5):
            if elapsed() > budget_s - 50:
                break
            h_s = make_structured_start(n, seed + 1000 + ss * 7)
            h_s = slsqp_optimize(h_s, dx, target_sum, t=300, maxiter=500)
            c5_s = compute_c5(h_s, dx)
            if c5_s < best_c5:
                best_h = h_s
                best_c5 = c5_s
                h_cd_s, c5_cd_s = pairwise_cd(best_h, dx, target_sum, n_sweeps=10)
                if c5_cd_s < best_c5:
                    best_h = h_cd_s
                    best_c5 = c5_cd_s

    # Phase 7: More subgradient on best
    if elapsed() < budget_s - 60:
        h_sg2, c5_sg2 = subgradient_descent(best_h, dx, target_sum, n_steps=20000, alpha0=0.03)
        if c5_sg2 < best_c5:
            best_h = h_sg2
            best_c5 = c5_sg2

    # Phase 8: Final pairwise CD
    if elapsed() < budget_s - 40:
        h_cd3, c5_cd3 = pairwise_cd(best_h, dx, target_sum, n_sweeps=20)
        if c5_cd3 < best_c5:
            best_h = h_cd3
            best_c5 = c5_cd3

    # Final SLSQP polish
    if elapsed() < budget_s - 10:
        h_final = slsqp_optimize(best_h, dx, target_sum, t=3000, maxiter=2000)
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
