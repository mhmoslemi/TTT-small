def subgrad_step(h, n, alpha=100.0):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    
    # Softmax weights
    shifted = (c - max_c) * alpha / max(1e-10, max_c)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute subgradient
    subgrad = np.zeros(n)
    n_full = 2 * n - 1
    for m in range(n_full):
        w = weights[m]
        if w < 1e-12:
            continue
        shift = m - n + 1
        # dc[m]/dh[j] = dx * [(1 - h[j + shift]) * [0<=j+shift<n] - h[j - shift] * [0<=j-shift<n]]
        # Vectorized:

        # j + shift in [0, n-1] => j in [-shift, n-1-shift]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(y, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        if float(h.sum()) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    a = alpha * (c - m)
    amax = a.max()
    if amax > 500:
        return float(m)
    return float(m + np.log(np.exp(a - amax).sum() + np.exp(amax)) / alpha)

def subgrad_refine(h, n, max_iter=50000, alpha=50.0, t_end=0):
    dx = 2.0 / n
    target_sum = n / 2.0
    best = compute_c5(h, n)
    step = 0.05
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        if m < best:
            best = m
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.9
        # Softmax weights
        a = alpha * (c - m)
        w = np.exp(a)
        w /= w.sum()
        # Subgradient = dx * correlate(w, 1-2h, 'valid')
        g = dx * np.correlate(w, 1.0 - 2.0 * h, mode='valid')
        h = h - step * g
        h = project_box_sum(h, target_sum)
    return h, compute_c5(h, n)

def slsqp_opt(h, n, alpha, t_end):
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def con_fun(x):
        return float(x.sum()) - target_sum
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project_box_sum(res.x, target_sum)
        return h_new, compute_c5(h_new, n)
    except:
        return h, compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    return np.interp(x_new, x_old, h, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(y, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(c.max())
    a = alpha * (c - m)
    amax = float(a.max())
    if amax > 500:
        return m
    return m + float(np.log(np.exp(a - amax).sum() + np.exp(amax))) / alpha

def subgrad_refine(h, n, t_end, alpha=100.0, max_iter=200000):
    dx = 2.0 / n
    target_sum = n / 2.0
    best = compute_c5(h, n)
    best_h = h.copy()
    step = 0.02
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = float(c.max())
        if m < best - 1e-12:
            best = m
            best_h = h.copy()
        a = alpha * (c - m)
        a -= a.max()
        w = np.exp(a)
        w /= w.sum()
        g = dx * np.correlate(w, 1.0 - 2.0 * h, mode='valid')
        h = h - step * g
        h = project_box_sum(h, target_sum)
        # Adaptive step
        if m < best:
            step = min(step * 1.05, 0.3)
        elif it % 500 == 0:
            step *= 0.9
    return best_h, best

def slsqp_opt(h, n, alpha, t_end):
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def con_fun(x):
        return float(x.sum()) - target_sum
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-18})
        h_new = project_box_sum(res.x, target_sum)
        c = compute_c5(h_new, n)
        return h_new, c
    except Exception:
        return h, compute_c5(h, n)

    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)
