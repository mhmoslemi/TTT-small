def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    max_s = np.max(shifted)
    lse = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    return (max_c + (1.0/t) * lse) * dx

def slsqp(h, dx, target_sum, t, maxiter=500):
    n = len(h)
    _opt_state['t'] = t
    _opt_state['dx'] = dx
    _opt_state['target_sum'] = target_sum
    try:
        res = minimize(
            _obj_func, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _con_func, 'jac': _conjac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return np.clip(res.x, 0.0, 1.0), np.abs(np.sum(res.x) - target_sum) < 0.5
    except:
        return h, False

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

        c = compute_c5(h_final, dx)
        if c < best_c5:
            best_h, best_c5 = h_final, c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
