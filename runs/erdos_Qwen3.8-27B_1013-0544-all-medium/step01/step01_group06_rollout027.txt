def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = np.exp(alpha * (corr - m))
    w = ex / np.sum(ex)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-10:
            continue
        k = j - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += wj * (1.0 - h[:n-k])
                grad[:n-k] -= wj * h[k:]
        else:
            p = -k
            if p < n:
                grad[:n-p] += wj * (1.0 - h[p:])
                grad[p:] -= wj * h[:n-p]
    val = (m + np.log(np.sum(ex)) / alpha) * dx
    return val, grad * dx

    h1, c1 = gradient_optimize(best_h, target_sum, d1, rng, alpha_init=120)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: upsample to n=400 and optimize (40%)
    n_hi = 400
    h_hi = interpolate_to_n(best_h, len(best_h), n_hi)
    h_hi = project_box_sum(h_hi, n_hi / 2.0)
    d2 = t0 + budget_s * 0.50
    h2, c2 = gradient_optimize(h_hi, n_hi / 2.0, d2, rng, alpha_init=200)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: block coordinate descent (25%)
    d3 = t0 + budget_s * 0.72
    h3, c3 = block_cd(best_h, len(best_h) / 2.0, d3, block_size=12)
    if c3 < best_c5:
        best_h, best_c5 = h3, c3

    # Phase 4: high-alpha gradient polish (18%)
    d4 = t0 + budget_s * 0.90
    h4, c4 = gradient_optimize(best_h, len(best_h) / 2.0, d4, rng, alpha_init=600)
    if c4 < best_c5:
        best_h, best_c5 = h4, c4

    # Phase 5: final micro-steps (10%)
    d5 = t0 + budget_s * 0.97
    h5, c5v = gradient_optimize(best_h, len(best_h) / 2.0, d5, rng, alpha_init=1500)
    if c5v < best_c5:
        best_h, best_c5 = h5, c5v

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
