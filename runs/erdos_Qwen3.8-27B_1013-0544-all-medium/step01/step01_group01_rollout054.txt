    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-14:
        # Distribute the difference
        # Can only add to elements < 1 or subtract from elements > 0
        if diff > 0:
            # Need to add: only to elements < 1
            room = 1.0 - h
            if room.sum() > 0:
                h += diff * room / room.sum()
            else:
                h += diff / len(h)
        else:
            # Need to subtract: only from elements > 0
            h += diff * h / h.sum() if h.sum() > 0 else diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    return h

def gradient_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # softmax weights
    shifted = beta * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Gradient: sum_k weights[k] * grad_c_k * dx
    # grad_c_k[m] = (1-h[m+k-(n-1)]) * valid1 - h[m-k+(n-1)] * valid2
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        # (1-h[m+k-(n-1)]) for valid m: m+k-(n-1) in [0,n-1]
        # i.e., m in [0-n+1-k, n-1-k] ∩ [0, n-1]
        # j = m + lag, need 0 <= j < n
        # So m in [max(0, -lag), min(n-1, n-1-lag)]
        if lag >= 0:
            m_start, m_end = 0, n - 1 - lag
        else:
            m_start, m_end = -lag, n - 1
        if m_start <= m_end:
            grad[m_start:m_end+1] += weights[k] * (1 - h[m_start+lag:m_end+1+lag])
        
        # -h[m-k+(n-1)] = -h[m-lag] for valid m: m-lag in [0,n-1]
        # i.e., m in [max(0, lag), min(n-1, n-1+lag)]
        if lag >= 0:
            m_start2, m_end2 = lag, n - 1
        else:
            m_start2, m_end2 = 0, n - 1 + lag
        if m_start2 <= m_end2:
            grad[m_start2:m_end2+1] -= weights[k] * h[m_start2-lag:m_end2+1-lag]
    
    return grad * dx

    return np.correlate(h, 1.0 - h, mode="full")

    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            rsum = room.sum()
            if rsum > 1e-14:
                h += diff * room / rsum
            else:
                h += diff / len(h)
        else:
            hsum = h.sum()
            if hsum > 1e-14:
                h += diff * h / hsum
            else:
                h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return smooth_max_val(corr, beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    shifted = beta * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)

        if lag >= 0:
            ms, me = 0, n - 1 - lag
        else:

        if ms <= me:

        # -h[m-lag] for valid m
        if lag >= 0:
            ms2, me2 = lag, n - 1
        else:
            ms2, me2 = 0, n - 1 + lag
        if ms2 <= me2:
            grad[ms2:me2+1] -= w * h[ms2-lag:me2+1-lag]

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            rsum = room.sum()
            if rsum > 1e-14:
                h += diff * room / rsum
            else:
                h += diff / len(h)
        else:
            hsum = h.sum()
            if hsum > 1e-14:
                h += diff * h / hsum
            else:
                h += diff / len(h)
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return smooth_max_val(corr, beta) * dx


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    shifted = beta * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            ms, me = 0, n - 1 - lag
        else:
            ms, me = -lag, n - 1
        if ms <= me:
            grad[ms:me + 1] += w * (1.0 - h[ms + lag:me + 1 + lag])
        if lag >= 0:
            ms2, me2 = lag, n - 1
        else:
            ms2, me2 = 0, n - 1 + lag
        if ms2 <= me2:
            grad[ms2:me2 + 1] -= w * h[ms2 - lag:me2 + 1 - lag]
    return grad * dx


def cfun(x, target_sum):
    return x.sum() - target_sum


def cjac(x, n):
    return np.ones(n)


def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and initial_h_values is not None:

        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

        return (time.time() - start_time) < budget_s - margin

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    constr = [{'type': 'eq', 'fun': cfun, 'jac': cjac, 'args': (target_sum, n_points)}]

    beta = 20.0
