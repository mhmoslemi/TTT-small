The max of affine functions is convex. The max of a convex function and a concave function is not necessarily convex or concave.

    # coeffs: [a1, b1, a2, b2, ..., aK, bK]

        bk = coeffs[2*(k-1)+1]
        h += ak * np.cos(np.pi * k * x) + bk * np.sin(np.pi * k * x)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    s = np.sum(exp_vals)
    w = exp_vals / s
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for k_idx in active:
        k = k_idx - (n - 1)
        if k >= 0:
            grad[0:n-k] += w[k_idx] * (1.0 - h[k:n])
            grad[k:n] -= w[k_idx] * h[0:n-k]
        else:
            p = -k
            grad[p:n] += w[k_idx] * (1.0 - h[0:n-p])
            grad[0:n-p] -= w[k_idx] * h[p:n]
    return grad * (2.0 / n)

def fista_optimize(h0, target_sum, deadline, rng, alpha_start=40, alpha_end=250):
    n = len(h0)
    x = project_simplex_box(h0.astype(float).copy(), target_sum)
    x_prev = x.copy()
    best_x = x.copy()
    best_c5 = compute_c5(x)
    t_mom = 1.0
    lr = 0.01
    alpha = alpha_start
    stall = 0
    it = 0
    total_iters = 0

    while time.time() < deadline:
        it += 1
        total_iters += 1
        # FISTA momentum
        factor = (t_mom - 1.0) / (t_mom + 1.0)
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x + factor * (x - x_prev)
        y = project_simplex_box(y, target_sum)

        grad = smooth_max_grad(y, alpha)
        c5_y = compute_c5(y)

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(18):
            x_new = project_simplex_box(x - step * grad, target_sum)
            c5_new = compute_c5(x_new)
            if c5_new < c5_y - 1e-13 or c5_new < best_c5 - 1e-13:
                improved = True
                break
            step *= 0.5

        if improved:
            x_prev = x.copy()
            x = x_new
            if c5_new < best_c5:
                best_x = x.copy()
                best_c5 = c5_new
            stall = 0
            lr = min(step * 1.8, 0.1)
        else:
            stall += 1
            lr = max(step, 1e-5)
            if stall > 50:
                # Structured perturbation to escape
                strategy = stall % 3
                if strategy == 0:
                    pert = rng.randn(n) * 0.008
                elif strategy == 1:
                    pert = np.zeros(n)
                    s0 = rng.randint(0, max(1, n - 30))
                    s1 = min(n, s0 + rng.randint(10, 40))
                    pert[s0:s1] = rng.randn(s1 - s0) * 0.015
                else:
                    xx = np.arange(n) * 2.0 / n
                    pert = np.zeros(n)
                    for kk in range(1, 5):
                        pert += rng.randn() * 0.012 * np.cos(np.pi * kk * xx)
                        pert += rng.randn() * 0.012 * np.sin(np.pi * kk * xx)
                x = project_simplex_box(x + pert, target_sum)
                x_prev = x.copy()
                t_mom = 1.0
                lr = 0.01
                alpha = alpha_start
                stall = 0
                if stall == 0 and it > 100:
                    it = 0

        # Adapt alpha
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / 3000.0)

    return best_x, best_c5

def downsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.concatenate([h, h[:1]])
    x_ext = np.linspace(0, 2.0 + 1.0 / n_old, n_old + 1, endpoint=False)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(200, 0.5)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_simplex_box(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: FISTA from initial point, moderate alpha
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = fista_optimize(h0, target_sum, d1, rng, alpha_start=30, alpha_end=200)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Continue with higher alpha for sharper max
    remaining = budget_s - (time.time() - t0) - 10
    if remaining > 40:
        d2 = time.time() + min(remaining * 0.45, 200.0)
        h2, c5_2 = fista_optimize(best_h, target_sum, d2, rng, alpha_start=150, alpha_end=400)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: Try at lower resolution (easier to optimize)
    remaining = budget_s - (time.time() - t0) - 10
    if remaining > 50:
        for n_try in [120, 160]:
            if time.time() > t0 + budget_s - 60:
                break
            h_down = downsample(best_h, n, n_try)
            h_down = project_simplex_box(h_down, n_try / 2.0)
            d3 = time.time() + min(remaining * 0.3, 100.0)
            h3, c5_3 = fista_optimize(h_down, n_try / 2.0, d3, rng, alpha_start=30, alpha_end=200)
            if c5_3 < best_c5:
                best_h, best_c5, n, target_sum = h3, c5_3, n_try, n_try / 2.0
            remaining = budget_s - (time.time() - t0) - 10
            if remaining < 30:
                break

    # Phase 4: Final polish at best resolution
    remaining = budget_s - (time.time() - t0) - 5
    if remaining > 20:
        h4, c5_4 = fista_optimize(best_h, target_sum, time.time() + remaining, rng,
                                   alpha_start=200, alpha_end=500)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    best_h = project_simplex_box(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
