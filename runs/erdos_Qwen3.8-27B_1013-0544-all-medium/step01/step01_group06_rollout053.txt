def project(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid

            hi = mid
    lam = (lo + hi) / 2

def grad_at_k(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    # Only iterate over significant weights
    threshold = 1e-10
    active = np.where(w > threshold)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_at_k(h, k, n)
    return grad * (2.0 / n)
