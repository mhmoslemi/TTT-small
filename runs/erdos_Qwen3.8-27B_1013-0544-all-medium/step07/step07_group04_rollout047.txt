def coord_step(h, j, corr):
    """Find optimal h[j] given all other h fixed."""
    n = len(h)
    v = h[j]
    L = 2 * n - 1
    # Compute effect[m] = change in corr[m] per unit change in h[j]
    # corr[m] = sum_n h[n+m] * (1-h[n])
    # Changing h[j] by delta:
    #   - terms where n+m = j (n = j-m): delta * (1-h[n]) = delta*(1-h[j-m])
    #   - terms where n = j: -h[n+m]*delta = -h[j+m]*delta
    effect = np.zeros(L)
    # For m where j-m is in [0, n-1]: m in [j-(n-1), j]
    m_lo = max(0, j - (n - 1))
    m_hi = min(L - 1, j)
    # n = j - m, so 1-h[n] = 1-h[j-m]
    effect[m_lo:m_hi+1] += 1.0 - h[j - (np.arange(m_hi+1) - m_lo)]
    
    # For m where j+m is in [0, n-1]: m in [-j, n-1-j]
    m_lo2 = max(0, -j)
    m_hi2 = min(L - 1, n - 1 - j)
    effect[m_lo2:m_hi2+1] -= h[j + np.arange(m_hi2 - m_lo2 + 1)]
    
    # Now find delta in [-v, 1-v] minimizing max_m (corr[m] + delta * effect[m])
    # This is: minimize max_m (A_m + delta * B_m) where A_m = corr[m], B_m = effect[m]
    lo = -v
    hi = 1.0 - v
    if hi - lo < 1e-15:
        return v
    
    # Ternary search (the function is convex in delta)
    a = lo
    b = hi
    for _ in range(60):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + m1 * effect)
        f2 = np.max(corr + m2 * effect)
        if f1 < f2:
            b = m2
        else:
            a = m1
    delta = 0.5 * (a + b)
    return v + delta

import numpy as np
from scipy.optimize import minimize
import time


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(60):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v


def lag_gradient(h, n, k):
    """Gradient of corr lag k w.r.t. h, scaled by dx."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = (1.0 - 2.0 * h)
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n - kk])
        g[:n - kk] -= h[kk:]
    return dx * g


def prox_grad_step(h, corr, n, step, top_k=5, momentum_vec=None, mu=0.7):
    cmax = float(corr.max())
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:min(top_k, len(corr))]
    avg_k = active - (n - 1)
    g = np.zeros(n)
    for kk in avg_k:
        g += lag_gradient(h, n, int(kk))
    g /= len(avg_k)
    if momentum_vec is not None:
        g = mu * momentum_vec + g
    new_h = h - step * g
    new_h = project_box_sum(new_h, n)
    return new_h, g, float(corr.max())


def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return m
    return m + np.log(s) / alpha


def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:
        q = e / s
    g = np.zeros(n)
    off = n - 1
    for idx in np.flatnonzero(q > 1e-15):
        k = int(idx - off)
        w = float(q[idx])
        g += w * lag_gradient(x, n, k)
    return g


def h_slice_h(h, a, b):
    return h[a:b]


def pair_xs(lo, hi, x0):
    xs = np.array([lo, hi, x0], dtype=float)
    if hi - lo > 1e-12:
        xs = np.concatenate((xs, np.linspace(lo, hi, 32)))
    a2 = max(lo, x0 - 0.03)
    b2 = min(hi, x0 + 0.03)
    if b2 - a2 > 1e-12:
        xs = np.concatenate((xs, np.linspace(a2, b2, 128)))
    xs = np.unique(xs)
    xs = xs[(xs >= lo - 1e-12) & (xs <= hi + 1e-12)]
    return np.clip(xs, lo, hi)


def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i + n] += ax1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += ax2[:, None] * h[None, :]
    delta[:, j:j + n] += ax2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += ax1[:, None] * h[None, :]
    idx0 = off
    base_self = ax1 * (1.0 - x0) + x0 * ax2 + ax2 * (1.0 - y0) + y0 * ax1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, idx0] += exact_self - base_self
    d = j - i
    idxp = off - d
    base_p = ax1 * (1.0 - y0) + x0 * ax1
    exact_p = xs * (1.0 - yv) - x0 * (1.0 - y0)
    delta[:, idxp] += exact_p - base_p
    idxm = off + d
    base_m = ax2 * (1.0 - x0) + y0 * ax2
    exact_m = yv * (1.0 - xs) - y0 * (1.0 - x0)
    delta[:, idxm] += exact_m - base_m
    return np.max(c_base[None, :] + dx * delta, axis=1)


def pair_search(h, c_base, i, j, n):
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = pair_xs(lo, hi, x0)
    vals = pair_eval(h, c_base, i, j, xs, n, x0, y0, S)
    for _ in range(3):
        if len(xs) <= 1:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k - 1], xs[k + 1]
        if b - a < 1e-12:
            break
        xs2 = np.linspace(a, b, 80)
        vals = pair_eval(h, c_base, i, j, xs2, n, x0, y0, S)
        xs = xs2
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c_base.max()) - 1e-14:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(S - xb)
        if yb < 0.0:
            yb = 0.0
            xb = S
        elif yb > 1.0:
            yb = 1.0
            xb = S - 1.0
        return xb, yb, bv
    return None

def coord_descent(h, n, max_sweeps=30, deadline=None):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:

        for j in range(n):

                continue
            effect = np.zeros(L)
            m_lo = max(0, j - (n - 1))
            m_hi = min(L - 1, j)
            effect[m_lo:m_hi + 1] += 1.0 - h[j - (np.arange(m_hi + 1) - m_lo)]
            m_lo2 = max(0, 0)
            m_hi2 = min(L - 1, n - 1 - j)
            if m_hi2 >= m_lo2:
                effect[m_lo2:m_hi2 + 1] -= h[j + np.arange(m_hi2 - m_lo2 + 1)]
            delta_lo = -v

                if f1 < f2:
                    b = m2
                else:
                    a = m1

            new_v = v + delta
            new_v = float(np.clip(new_v, 0.0, 1.0))
            if abs(new_v - v) < 1e-15:
                continue
            h[j] = new_v
            c = c + (new_v - v) * effect
            vnow = float(c.max())
            if vnow < bestc:
                bestc = vnow
                besth = h.copy()
        h = project_box_sum(h, n)
        c = compute_c(h, n)
        vnow = float(c.max())
        if vnow < bestc:
            bestc = vnow
            besth = h.copy()
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def prox_gradient(h, n, deadline, top_k=8):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    step = 0.01
    momentum = np.zeros(n)
    mu = 0.6
    it = 0
    stagnation = 0
    while time.time() < deadline:
        c = compute_c(h, n)
        cur = float(c.max())
        if cur < bestc:
            bestc = cur
            besth = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 500:
            step *= 0.5
            if step < 1e-5:
                break
            stagnation = 0
        new_h, g, _ = prox_grad_step(h, c, n, step, top_k, momentum, mu)
        new_c = compute_c(new_h, n)
        new_val = float(new_c.max())
        if new_val < cur + 0.001:
            h = new_h
            momentum = g
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.7
        it += 1
        if it > 50000:
            break
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def lbfgsb_phase(h, n, alpha, mu_penalty, maxiter, deadline):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        def obj(x):
            return lse_obj(x, n, alpha) + 0.5 * mu_penalty * (x.sum() - 0.5 * n) ** 2
        def grad(x):
            return lse_grad(x, n, alpha) + mu_penalty * (x.sum() - 0.5 * n)
        res = minimize(obj, h, method="L-BFGS-B", jac=grad, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 3 * maxiter, "ftol": 1e-16, "gtol": 1e-10})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h


def slsqp_phase(h, n, alpha, maxiter, deadline):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        def sf(x):
            return float(x.sum() - 0.5 * n)
        def sj(x):
            return np.ones(n)
        cons = [{"type": "eq", "fun": sf, "jac": sj}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP", jac=lse_grad,
                       bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-15})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h


def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

    bestc = float(compute_c(h, n).max())
    besth = h.copy()

    if n < 4 or time.time() > deadline:

    # Phase 1: Proximal gradient

        if v2 < bestc:

        if v2 <= workc + 1e-7:
            work, workc = h2, v2
    # Phase 2: L-BFGS-B with increasing alpha
    phases = [(50, 20, 400), (200, 100, 300), (1000, 500, 250), (5000, 5000, 200)]
    for alpha, mu_p, maxit in phases:
        if time.time() > deadline - 8:

        h2 = lbfgsb_phase(work, n, alpha, mu_p, maxit, deadline)
        v2 = float(compute_c(h2, n).max())
        if v2 < bestc:

        if v2 <= workc + 1e-7:

        else:

    # Phase 3: Coordinate descent
    dl3 = time.time() + 0.15 * (deadline - time.time())

    return besth, bestc
