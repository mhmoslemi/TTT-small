def greedy_binary(h_b, n, deadline, max_iters=10000, n_crit=20):
    dx = 2.0 / n
    k_arr = np.arange(-(n-1), n)  # all lags
    
    # Precompute D matrix: D[a, k_idx] = h[a+k] - h[a-k] with bounds
    # k_idx goes from 0 to 2n-2, corresponding to lag k_arr[k_idx]
    
    def compute_D(h):
        D = np.zeros((n, 2*n-1))
        for k_idx, k in enumerate(k_arr):
            a_k = np.arange(n) + k
            a_ks = np.arange(n) - k
            m1 = (a_k >= 0) & (a_k < n)
            m2 = (a_ks >= 0) & (a_ks < n)
            D[:, k_idx] = 0
            D[m1, k_idx] = h[a_k[m1]]
            D[m2, k_idx] -= h[a_ks[m2]]
        return D
    
    D = compute_D(h_b)
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_max = float(corr.max())
    best_h = h_b.copy()
    best_max = cur_max
    
    ones = set(np.where(h_b == 1)[0])
    zeros = set(np.where(h_b == 0)[0])
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Find critical lags
        crit_idx = np.where(corr >= corr.max() - 2)[0]
        if len(crit_idx) > n_crit:
            crit_idx = np.argsort(corr)[::-1][:n_crit]
        
        # For each (i, j) where i in zeros, j in ones:
        # new_max = max(corr[crit_idx] + D[j, crit_idx] - D[i, crit_idx])
        # We want to minimize this
        
        zero_list = np.array(list(zeros))
        one_list = np.array(list(ones))
        
        # Vectorized: for each i in zeros, compute D[i, crit_idx]
        # For each j in ones, compute D[j, crit_idx]
        # new_maxs[i_idx, j_idx] = max(corr[crit_idx] - D[i, crit_idx] + D[j, crit_idx])
        
        Di = D[zero_list[:, None], crit_idx]  # shape (n_zeros, n_crit)
        Dj = D[one_list[None, :], crit_idx]  # shape (n_crit, n_ones)
        base = corr[crit_idx] - Di  # shape (n_zeros, n_crit)
        new_vals = base[:, :, None] + Dj[None, :, :]  # shape (n_zeros, n_crit, n_ones)
        new_maxs = new_vals.max(axis=1)  # shape (n_zeros, n_ones)
