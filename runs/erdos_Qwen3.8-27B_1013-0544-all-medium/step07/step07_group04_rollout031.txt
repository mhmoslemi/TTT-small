def compute_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_box_sum(v, n):
    tgt = 0.5 * n
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        d = tgt - v.sum()
        if abs(d) < 1e-15: break
        m = (v > 1e-14) & (v < 1.0 - 1e-14)
        if m.any(): v[m] += d / m.sum()
        else: v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = c.max()
    if alpha <= 1: return m
    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = c.max()
    if alpha <= 1:
        act = c >= m - 1e-15
        w = np.zeros(2*n-1); w[act] = 1.0 / act.sum()
    else:
        sh = alpha * (c - m); e = np.exp(sh); w = e / e.sum()
    dx = 2.0 / n; g = np.zeros(n)
    for lag in range(-(n-1), n):
        wl = w[n-1+lag]
        if wl < 1e-14: continue
        if lag == 0: g += wl * (1.0 - 2.0 * x)
        elif lag > 0:

            g[lag:] -= wl * x[:-lag]
        else:
            s = -lag
            g[s:] += wl * (1.0 - x[:-s])
            g[:n-s] -= wl * x[s:]

def pair_refine(h, c, i, j, n):
    """Optimize pair (i,j) maintaining h[i]+h[j] = S."""
    S = h[i] + h[j]
    x0, y0 = h[i], h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-12: return None
    old_cmax = c.max()
    grid = np.linspace(lo, hi, 64)
    grid = np.unique(np.concatenate([grid, [lo, hi, x0]]))
    best_t = x0
    best_val = old_cmax
    for t in grid:

        c_new = compute_c(h, n)
        v = c_new.max()
        if v < best_val:
            best_val = v; best_t = t
    h[i] = best_t; h[j] = S - best_t
    c = compute_c(h, n)
    return c.max() if c.max() < old_cmax - 1e-14 else None

            noimp = 0
            c = compute_c(h, n)
        else:

**New idea: "Gradient matching" approach**

At the optimum, by the KKT conditions, the gradient of the objective should be "balanced" in some sense. Specifically, if we denote by A the set of active lags (those achieving the max), then at the optimum:

-∑_{k∈A} λ_k ∇c_k + μ * 1 + ν = 0 (projected onto the feasible set)

where λ_k ≥ 0, ∑λ_k = 1, μ is the multiplier for the sum constraint, and ν handles the box constraints.

This suggests an algorithm:
1. Compute c, find active set A
2. For each k in A, compute ∇c_k
3. Find λ that minimizes ||∑ λ_k ∇c_k|| (the "most canceling" combination)
4. Take a step in the direction -∑λ_k ∇c_k (projected)
5. Repeat

This is essentially the "minimum norm subgradient" method, which is known to be more effective than the standard subgradient method.

def min_norm_subgrad_step(h, n, top_k=10):
    c = compute_c(h, n)
    cmax = c.max()
    # Get top-k lags
    idx = np.argsort(c)[-top_k:]
    active_c = c[idx]
    
    # Compute gradients for these lags
    grads = np.zeros((len(idx), n))
    dx = 2.0 / n
    off = n - 1
    for m, k_idx in enumerate(idx):
        lag = k_idx - off
        g = np.zeros(n)
        if lag == 0: g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:-lag]
        else:
            s = -lag
            g[s:] += (1.0 - h[:-s])
            g[:n-s] -= h[s:]
        grads[m] = dx * g
    
    # Find λ ≥ 0, sum λ = 1, minimizing ||∑ λ_k g_k||²
    # This is a QP: min λ' G λ s.t. λ ≥ 0, 1'λ = 1
    # where G = grads @ grads.T
    # For small top_k, we can solve this exactly
    
    G = grads @ grads.T  # (top_k x top_k)
    
    # Solve the QP using a simple approach
    # Since top_k is small (10), we can use enumeration of active sets
    # Or use a simple projected gradient on the simplex
    
    # Actually, for small k, let's just use scipy
    from scipy.optimize import minimize as min_opt
    def qf(lam):
        return lam @ G @ lam
    def qf_grad(lam):
        return 2.0 * G @ lam

    lam = np.clip(res.x, 0, 1)
    lam /= lam.sum()
    
    # Compute the combined gradient
    g = grads.T @ lam  # shape (n,)
    
    # Project the negative gradient onto the feasible set
    # ... (this requires a line search)
    
    return -g, cmax

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_box_sum(v, n):
    tgt = 0.5 * n
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        d = tgt - v.sum()
        if abs(d) < 1e-15:
            break
        m = (v > 1e-14) & (v < 1.0 - 1e-14)
        if m.any():
            v[m] += d / m.sum()
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = c.max()
    if alpha <= 1:
        return m
    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = c.max()
    if alpha <= 1:
        act = c >= m - 1e-15
        w = np.zeros(2 * n - 1)
        w[act] = 1.0 / act.sum()
    else:
        sh = alpha * (c - m)
        e = np.exp(sh)
        w = e / e.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for lag in range(-(n - 1), n):
        wl = w[n - 1 + lag]
        if wl < 1e-14:
            continue
        if lag == 0:
            g += wl * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n - lag] += wl * (1.0 - x[lag:])
            g[lag:] -= wl * x[:-lag]
        else:
            s = -lag
            g[s:] += wl * (1.0 - x[:-s])
            g[:n - s] -= wl * x[s:]
    return dx * g

def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(n)

def lbfqsb(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        res = minimize(lse_obj, h, args=(n, alpha), method="L-BFGS-B",
                       jac=lse_grad, bounds=[(0.0, 1.0)] * n,
                       options={"maxiter": maxiter, "ftol": 1e-17, "gtol": 1e-11})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h

def slsqp(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        cons = [{"type": "eq", "fun": sum_fun, "args": (n,), "jac": sum_jac}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lse_grad, bounds=[(0.0, 1.0)] * n,
                       constraints=cons, options={"maxiter": maxiter, "ftol": 1e-15})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h

def cyclic_1var(h, n, deadline, grid_sz=60):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    sweeps = 0
    while time.time() < deadline:
        sweeps += 1
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            lo = max(0.0, old - 0.4)
            hi = min(1.0, old + 0.4)
            grid = np.linspace(lo, hi, grid_sz)
            endpoints = np.array([0.0, 1.0, old])
            grid = np.unique(np.concatenate([grid, endpoints]))
            grid = grid[(grid >= 0.0) & (grid <= 1.0)]
            best_t = old
            best_v = float(c.max())
            for t in grid:
                h[i] = t
                cn = compute_c(h, n)
                v = float(cn.max())
                if v < best_v:
                    best_v = v
                    best_t = t
            h[i] = best_t
            h = project_box_sum(h, n)
            c = compute_c(h, n)
            cm = float(c.max())
            if cm < bestc:
                bestc = cm
                besth = h.copy()
            if sweeps % 10 == 0 and time.time() > deadline:
                break
    return project_box_sum(besth, n), bestc

def pair_eval_fast(h, n, i, j, t, S):
    old_i = h[i]
    old_j = h[j]
    c = compute_c(h, n)
    h[i] = t
    h[j] = S - t
    cn = compute_c(h, n)
    h[i] = old_i
    h[j] = old_j
    return float(cn.max())

def pair_search(h, n, i, j):
    S = h[i] + h[j]
    x0 = h[i]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-12:
        return None
    old_cmax = float(compute_c(h, n).max())
    g1 = np.linspace(lo, hi, 48)
    best_t = x0
    best_v = old_cmax
    for t in g1:
        h[i] = t
        h[j] = S - t
        v = float(compute_c(h, n).max())
        if v < best_v:
            best_v = v
            best_t = t
    if best_t == x0 and best_v >= old_cmax - 1e-14:
        h[i] = x0
        h[j] = S - x0
        return None
    lo2 = max(lo, best_t - (hi - lo) / 12.0)
    hi2 = min(hi, best_t + (hi - lo) / 12.0)
    g2 = np.linspace(lo2, hi2, 48)
    for t in g2:
        h[i] = t
        h[j] = S - t
        v = float(compute_c(h, n).max())
        if v < best_v:
            best_v = v
            best_t = t
    h[i] = best_t
    h[j] = S - best_t
    if best_v < old_cmax - 1e-14:
        return best_v
    return None

def pair_sweep(h, n, deadline, seed=1):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1
    noimp = 0
    max_noimp = 600
    while time.time() < deadline:
        if noimp > max_noimp:
            break
        r = rng.random()
        if r < 0.7:
            pos = c[off + 1:]
            neg = c[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-10
            w = w / w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
            i = int(rng.randint(n - d))
            j = i + d
        elif r < 0.85:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            i = int(rng.randint(n // 2))
            j = n - 1 - i
            if i == j:
                j = (j + 1) % n
        res = pair_search(h, n, i, j)
        c = compute_c(h, n)
        cm = float(c.max())
        if res is not None and cm < bestc:
            bestc = cm
            besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    return project_box_sum(besth, n), float(compute_c(besth, n).max())

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)
