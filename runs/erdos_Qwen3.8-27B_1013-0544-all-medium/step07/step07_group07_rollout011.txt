def lp_interate(h, n, max_iters=100, tol=1e-10):
    """Cutting plane / outer approximation for min-max."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iters):
        c = compute_corr(h, n)  # shape (2n-1,)
        cmax = c.max()
        if cmax < best_c - tol:
            best_c = cmax
            best_h = h.copy()
        
        # Compute gradients
        grads = correlation_gradients(h, n)  # shape (2n-1, n)
        
        # Set up LP: minimize t
        # s.t. c_k(h0) + grads[k] @ (h - h0) <= t  for all k
        #      sum(h) = n/2
        #      0 <= h <= 1
        #
        # Rewrite: grads[k] @ h - t <= grads[k] @ h0 - c_k(h0)
        
        var_h = cp.Variable(n)
        var_t = cp.Variable()
        obj = cp.Minimize(var_t)
        constraints = []
        # Linearized correlation constraints
        # c_k + g_k^T (h - h0) <= t
        # g_k^T h - t <= g_k^T h0 - c_k
        rhs = grads @ h - c  # shape (2n-1,)
        constraints += [grads @ var_h - var_t <= rhs]
        constraints += [cp.sum(var_h) == n/2.0]
        constraints += [var_h >= 0]
        constraints += [var_h <= 1]
        
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if var_h.value is None:
            break
        h_new = np.clip(var_h.value, 0, 1)
        # Project to ensure sum constraint
        h_new = project_feasible(h_new, n)

            best_c = c_new
            best_h = h_new.copy()
        
        if abs(c_new - compute_c5(best_h, n)) < tol:
            break
        h = h_new
    
    return best_h, best_c

    best_c = compute_c5(h, n)

        c = np.correlate(h, 1-h, mode='full') * dx
        # LSE gradient
        shifted = alpha * (c - c.max())
        weights = np.exp(shifted - shifted.max())
        weights /= weights.sum()

        t = lr
        while t > 1e-10:

            c_trial = np.correlate(h_trial, 1-h_trial, mode='full') * dx
            shifted_t = alpha * (c_trial - c_trial.max())

            best_c = c_curr
            best_h = h.copy()
        
        if it % 1000 == 0:

    return best_h, best_c
