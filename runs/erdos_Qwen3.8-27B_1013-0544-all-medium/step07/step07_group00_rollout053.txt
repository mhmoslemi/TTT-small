import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=100, deadline=None, grid_size=100):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline - 1:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            # Compute coefficient vector a for this coordinate
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2*n-2, 2*n-2-p)
            m_b_lo = max(0, p)
            m_b_hi = min(2*n-2, p+n-1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi+1)
            jf_arr = p + ms - (n-1)
            jb_arr = p - ms + (n-1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            # Grid search for best d
            d_grid = np.linspace(-h0, 1.0-h0, grid_size)
            best_d = 0.0
            best_val = current_max
            diag_idx = (ms == (n-1))
            for d in d_grid:
                if abs(d) < 1e-14: continue
                nv = c_aff + d * a
                if np.any(diag_idx):
                    nv[diag_idx] -= d * d  # quadratic correction
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        val += 0.1 * (m + lse/alpha)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

            break

        current_max = float(c.max())
        improved = False
        order = range(n) if pidx % 2 == 0 else range(n-1, -1, -1)
        for p in order:
            if deadline is not None and time.time() > deadline - 1:
                break

            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n-1-p)
            m_f_hi = min(2*n-2, 2*n-2-p)
            m_b_lo = max(0, p)
            m_b_hi = min(2*n-2, p+n-1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi+1)
            jf_arr = p + ms - (n-1)
            jb_arr = p - ms + (n-1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a[valid_b] -= h[jb_arr[valid_b]]

            if m_lo > 0:

            if m_hi < 2*n-2:

            diag = (ms == (n-1)).astype(float)
            # Two-stage: coarse then fine
            d_coarse = np.linspace(-h0, 1.0-h0, 60)
            vals = c_aff[None,:] + d_coarse[:,None]*a[None,:] - d_coarse[:,None]**2*diag[None,:]
            max_vals = vals.max(axis=1)
            total_vals = np.maximum(max_vals, unaff_max)

            max_vals2 = vals2.max(axis=1)

        if not improved:
