            break

            tot = room.sum()

            h += (diff / tot) * room

            if tot < 1e-14:

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = float(mx + np.log(np.sum(np.exp(shifted - mx))))
    return (m + lse / alpha) * dx

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = shifted.max()
    ew = np.exp(shifted - mx)
    w = ew / ew.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-8:

        r1_lo = max(0, lag)
        r1_hi = min(n - 1, n - 1 + lag)
        if r1_lo <= r1_hi:
            idx = np.arange(r1_lo, r1_hi + 1)
            grad[idx] += wk * (1.0 - x[idx - lag])
        r2_lo = max(0, -lag)
        r2_hi = min(n - 1, n - 1 - lag)
        if r2_lo <= r2_hi:
            idx = np.arange(r2_lo, r2_hi + 1)
            grad[idx] -= wk * x[idx + lag]
    return grad * dx / alpha

def sum_con(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

                if r1_lo <= p <= r1_hi:
                    a[k - k_lo] += (1.0 - h[p - lag])
                # Range 2: p in [max(-lag,0), min(n-1-lag,n-1)]
                r2_lo = max(0, -lag)
                r2_hi = min(n - 1, n - 1 - lag)
                if r2_lo <= p <= r2_hi:
                    a[k - k_lo] -= h[p + lag]
            
            c_aff = c[k_lo:k_hi + 1]
            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(c[:k_lo].max()))
            if k_hi < 2*n - 2:
                unaff_max = max(unaff_max, float(c[k_hi+1:].max()))
            
            # Find optimal delta via grid search
            # Two-level: coarse then fine
            diag_k = n - 1  # k where lag=0
            diag_idx = diag_k - k_lo if k_lo <= diag_k <= k_hi else -1
            
            def eval_delta(d):
                nv = c_aff + d * a
                if diag_idx >= 0:
                    nv[diag_idx] -= d * d
                return max(unaff_max, float(nv.max()))
            
            # Coarse search
            deltas = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                val = eval_delta(d)
                if val < best_val:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-14:
                # Fine search around best_d
                span = (1.0 - h0 - (-h0)) / grid_size
                fine = np.linspace(best_d - span, best_d + span, grid_size)
                fine = fine[(fine >= -h0) & (fine <= 1.0 - h0)]
                for d in fine:
                    if abs(d) < 1e-14:
                        continue
                    val = eval_delta(d)
                    if val < best_val:
                        best_val = val
                        best_d = d
                
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                if eval_delta(best_d) < best_c:
                    best_c = eval_delta(best_d)
                improved = True
        
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

    x_new = np.linspace(0, 2, n_new)

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = float(mx + np.log(np.sum(np.exp(shifted - mx))))
    return (m + lse / alpha) * dx

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = shifted.max()
    ew = np.exp(shifted - mx)
    w = ew / ew.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-8:
            continue
        lag = k - (n - 1)
        r1_lo = max(0, lag)
        r1_hi = min(n - 1, n - 1 + lag)
        if r1_lo <= r1_hi:
            idx = np.arange(r1_lo, r1_hi + 1)
            grad[idx] += wk * (1.0 - x[idx - lag])
        r2_lo = max(0, -lag)

1 - lag)
        if r2_lo <= r2_hi:

            grad[idx] -= wk * x[idx + lag]

                           jac=smooth_grad, bounds=bounds, constraints=cons,

            new_c = compute_c5(new_h, n)
