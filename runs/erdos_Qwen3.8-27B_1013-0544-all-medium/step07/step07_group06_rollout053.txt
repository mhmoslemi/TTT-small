```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def get_correlations(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)[-int(np.ceil(abs(diff))):-1] if diff < 1 else np.argsort(1.0 - h)[-1:]
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            idx = np.argsort(h)[:int(np.ceil(abs(diff)))] if abs(diff) < 1 else np.argsort(h)[:1]
            for i in idx:
                rem = min(h[i], -diff)
                h[i] -= rem
                diff += rem
                if diff >= -1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_obj(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def lse_minimize(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 5.0:
            break
        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.7 * (deadline - now))))
        _stop_time = stage_dl
        try:
            res = minimize(
                lbfgs_obj, work, method="L-BFGS-B", jac=True,
                bounds=bounds, args=(n, float(beta), float(pen)),
                callback=stop_callback,
                options={"maxiter": 1000000, "maxls": 50, "ftol": 1e-14, "gtol": 1e-9}
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
    return best, bc

def lp_direction(h, n, c, threshold, max_iter=50):
    """Find direction delta that reduces all near-max correlations."""
    dx = 2.0 / float(n)
    v = 1.0 - h
    m = c.max()
    # Active set: lags where c[k] > m - threshold
    active = np.where(c > m - threshold)[0]
    if len(active) == 0:
        return None
    if len(active) > 100:
        active = active[np.argsort(-c[active])[:100]]
    
    # Build constraint matrix: for each active lag k, D[k] · delta <= -1
    # D[k][i] = v[i+k-(n-1)] - h[i+(n-1)-k] (with bounds check)
    nlag = len(active)
    nvar = n
    A_ub = np.zeros((nlag, nvar))
    for idx, k in enumerate(active):
        offset = k - (n - 1)
        # term 1: sum_i delta[i] * v[i+offset] for valid i
        for i in range(n):
            j1 = i + offset
            j2 = i - offset
            coeff = 0.0
            if 0 <= j1 < n:
                coeff += v[j1]
            if 0 <= j2 < n:
                coeff -= h[j2]
            A_ub[idx, i] = coeff
    b_ub = np.full(nlag, -1.0)
    
    # Equality: sum delta = 0
    A_eq = np.ones((1, nvar))
    b_eq = np.array([0.0])
    
    bounds = [(-1.0, 1.0)] * nvar
    
    try:
        res = linprog(np.zeros(nvar), A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            delta = res.x
            if np.max(np.abs(delta)) > 1e-10:
                return delta
    except Exception:
        pass
    return None

def lp_descent(h, n, deadline, max_rounds=200, threshold_scale=0.02):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        c = get_correlations(work, n)
        m = c.max()
        threshold = threshold_scale * m + 1e-6
        
        # Try LP direction
        delta = lp_direction(work, n, c, threshold)
        
        if delta is not None:
            # Find max step size that keeps h in [0,1]
            pos = delta > 1e-10
            neg = delta < -1e-10
            step = 1.0
            if np.any(pos):
                step = min(step, ((1.0 - work[pos]) / delta[pos]).min())
            if np.any(neg):
                step = min(step, ((work[neg]) / (-delta[neg])).min())
            step = max(step, 1e-10) * 0.9  # safety margin
            
            new_h = project_feasible(work + step * delta, n)
            nc = compute_c5(new_h, n)
            if nc < bc - 1e-12:
                bc = nc
                best = new_h.copy()
                work = new_h
            elif nc < compute_c5(work, n) + 1e-10:
                work = new_h
        else:
            # LP infeasible, try steepest descent on active set
            active = np.where(c > m - threshold)[0]
            if len(active) == 0:
                break
            # Average gradient over active lags
            grad = np.zeros(n)
            for k in active:
                offset = k - (n - 1)
                for i in range(n):
                    j1 = i + offset
                    j2 = i - offset
                    g = 0.0
                    if 0 <= j1 < n:
                        g += (1.0 - work[j1])
                    if 0 <= j2 < n:
                        g -= work[j2]
                    grad[i] += g
            grad /= len(active)
            grad -= grad.mean()  # maintain zero sum
            norm = np.linalg.norm(grad)
            if norm < 1e-12:
                break
            grad /= norm
            work = project_feasible(work + 0.01 * grad, n)
            nc = compute_c5(work, n)
            if nc < bc - 1e-12:
                bc = nc
                best = work.copy()
    
    return best, bc
```

```python
    D = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        off = k - (n-1)
        if off == 0:
            D[k] = v - h
        elif off > 0:
            D[k, :n-off] += v[off:]
            D[k, off:] -= h[:n-off]
        else:
            o = -off
            D[k, o:] += v[:n-o]
            D[k, :n-o] -= h[o:]
    return D
```

This is O(n²) total which for n=400 is about 160K operations - very fast.

Now for the LP, I want to find δ such that:
- D[active, :] @ δ < 0 (reduce active correlations)
- sum(δ) = 0
- |δ[i]| ≤ 1 (box constraint for normalization)

I'll use linprog with HiGHS solver (fast and available in scipy).

The objective is 0 (feasibility problem), or I could minimize the maximum reduction (i.e., maximize the minimum of -D[active] @ δ).

Actually, let me formulate it as: maximize t subject to D[active, :] @ δ + t ≤ 0, sum δ = 0, |δ| ≤ 1.

This maximizes the minimum reduction across active lags. If t > 0, we have a descent direction.

Let me reformulate as LP:
- Variables: δ[0..n-1], t (total n+1 variables)
- Maximize t
- Subject to:

  - -1 ≤ δ[i] ≤ 1

In linprog form (minimize c @ x subject to A_ub @ x ≤ b_ub, A_eq @ x = b_eq, bounds):
- c = [0, 0, ..., 0, -1] (minimize -t)
- A_ub rows: [D[k,:], 1] with b = 0
- A_eq: [ones(1,n), 0] with b = 0
- bounds: δ[i] in [-1, 1], t in [-inf, inf) or [0, inf)

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h = np.clip(h + diff * room / s, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-15: break
            h = np.clip(h + diff * h / s, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            for i in np.argsort(1.0 - h):
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-13: break
        else:
            for i in np.argsort(h):
                rem = min(h[i], -diff)
                h[i] -= rem
                diff += rem
                if diff > -1e-13: break
    return h

def lse_obj_grad(h, n, beta, pen):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1.0 - h, "full") * dx, 0.0)
    m = c.max()
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = e.sum()
    if not np.isfinite(s) or s < 1e-300:
        p = np.ones_like(c) / len(c)
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    A = np.convolve(P[::-1], 1.0 - h, "full")[n-1:2*n-1]
    B = np.convolve(P, h, "full")[n-1:2*n-1]
    g = dx * (A - B)
    dev = h.sum() - 0.5 * n
    if pen:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    F += float(pen * 0.5) * 0  # no-op
    if not np.isfinite(F):
        return 1e100, np.zeros(n)
    return F, g

def lse_lbfgs(h, n, betas, pen, deadline, stage_time=40.0):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 5:
            break
        dl = min(deadline - 3, time.time() + stage_time)
        _stop_time = dl
        try:
            r = minimize(lse_obj_grad, work, method="L-BFGS-B", jac=True,
                         bounds=bounds, args=(n, beta, pen),
                         callback=stop_callback,
                         options={"maxiter": 500000, "maxls": 50, "ftol": 1e-15, "gtol": 1e-10})
            work = project_feasible(r.x, n)
        except Exception:
            pass
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
    return best, bc

def build_D(h, n):
    v = 1.0 - h
    D = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        off = k - (n-1)
        if off == 0:
            D[k] = v - h
        elif off > 0:
            D[k, :n-off] += v[off:]
            D[k, off:] -= h[:n-off]
        else:
            o = -off
            D[k, o:] += v[:n-o]
            D[k, :n-o] -= h[o:]
    return D

def lp_descent(h, n, deadline, max_rounds=500, max_active=80):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    D_cache = None
    stale = True
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        c = get_corr(work, n)
        m = c.max()
        thresh = 0.01 * m + 1e-8
        active = np.where(c > m - thresh)[0]
        if len(active) > max_active:
            active = active[np.argsort(-c[active])[:max_active]]
        
        if stale or D_cache is None:
            D_cache = build_D(work, n)
            stale = False
        
        # LP: max t s.t. D[active] @ delta + t <= 0, sum(delta)=0, |delta|<=1, t>=0
        na = len(active)
        A_ub = np.hstack([D_cache[active], np.ones((na, 1))])
        b_ub = np.zeros(na)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        c_obj = np.zeros(n+1)
        c_obj[n] = -1.0
        bounds_lp = [(-1.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        
        if not res.success or res.x[n] < 1e-10:
            # Try with steeper threshold
            thresh = 0.005 * m + 1e-8
            active = np.where(c > m - thresh)[0]
            if len(active) == 0 or len(active) > max_active:
                active = active[np.argsort(-c[active])[:max_active]] if len(active) > 0 else active
            if len(active) == 0:
                break
            A_ub = np.hstack([D_cache[active], np.ones((len(active), 1))])
            b_ub = np.zeros(len(active))
            bounds_lp2 = [(-1.0, 1.0)] * n + [(0.0, None)]
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds_lp2, method='highs')
            except Exception:
                break
            if not res.success or res.x[n] < 1e-10:
                break
        
        delta = res.x[:n]
        # Step size
        pos = delta > 1e-10
        neg = delta < -1e-10
        step = 1.0
        if np.any(pos):
            step = min(step, ((1.0 - work[pos]) / delta[pos]).min())
        if np.any(neg):

        step = max(step * 0.95, 1e-8)

        nc = compute_c5(new_h, n)
        if nc < bc - 1e-12:
            bc, best = nc, new_h.copy()
        work = new_h
        stale = True  # D matrix is stale after h changes
    
    return best, bc

def top_n_binary(h, n):
    hb = np.zeros(n)
    hb[np.argsort(-h)[:n//2]] = 1.0
    return hb

def binary_greedy(h, n, deadline, max_sweeps=50, chunk=2048):
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, "full").astype(float)
    M = int(round(corr.max()))
    besth, bestM = h.copy(), M
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        # Build base matrix
        D = build_D(h, n)
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            end = min(start + chunk, zi.size)
            i = zi[start:end]
            j = oj[start:end]
            # delta_c = D[i] - D[j] + corr_update
            arr = D[i] - D[j]
            # Add current corr
            arr = arr + corr[None, :]
            # Fix diagonal terms
            arr[:, n-1] -= 2.0
            d1 = n-1 + (j - i)
            d2 = n-1 + (i - j)
            valid_d1 = (d1 >= 0) & (d1 < 2*n-1)
            valid_d2 = (d2 >= 0) & (d2 < 2*n-1)
            arr[np.arange(len(i))[valid_d1], d1[valid_d1]] += 1.0
            arr[np.arange(len(i))[valid_d2], d2[valid_d2]] += 1.0
            mx = arr.max(axis=1)
            idx = np.where(mx <= M - 1 + 1e-8)[0]
            if idx.size:
                loc = idx[np.argmin(mx[idx])]
                h[i[loc]] = 1.0
                h[j[loc]] = 0.0
                corr = arr[loc].astype(float)
                M = int(round(corr.max()))
                if M < bestM:
                    bestM, besth = M, h.copy()
                improved = True
                break
        if not improved:
            break
    return besth, bestM * (2.0/n)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(np.min(a)) - 1.0, float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff * room / s, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff * h / s, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            for i in np.argsort(1.0 - h):
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-13:
                    break
        else:
            for i in np.argsort(h):
                rem = min(h[i], -diff)
                h[i] -= rem
                diff += rem
                if diff > -1e-13:
                    break
    return h

def lse_obj_grad(h, n, beta, pen):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1.0 - h, "full") * dx, 0.0)
    m = float(c.max())
    y = min(beta * (c - m), 0.0)
    e = np.exp(y)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        p = np.ones_like(c) / len(c)
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    A = np.convolve(p[::-1], 1.0 - h, "full")[n-1:2*n-1]
    B = np.convolve(p, h, "full")[n-1:2*n-1]
    g = dx * (A - B)
    dev = h.sum() - 0.5 * n
    if pen:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros(n)
    return float(F), g

def lse_lbfgs(h, n, betas, pen, deadline, stage_time=35.0):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 4:
            break
        dl = min(deadline - 2, time.time() + stage_time)
        _stop_time = dl
        try:
            r = minimize(lse_obj_grad, work, method="L-BFGS-B", jac=True,
                         bounds=bounds, args=(n, beta, pen),
                         callback=stop_callback,
                         options={"maxiter": 500000, "maxls": 50, "ftol": 1e-15, "gtol": 1e-10})
            work = project_feasible(r.x, n)
        except Exception:
            pass
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
    return best, bc

def build_D(h, n):
    v = 1.0 - h
    D = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        off = k - (n-1)
        if off == 0:
            D[k] = v - h
        elif off > 0:
            D[k, :n-off] += v[off:]
            D[k, off:] -= h[:n-off]
        else:
            o = -off
            D[k, o:] += v[:n-o]
            D[k, :n-o] -= h[o:]
    return D

def lp_descent(h, n, deadline, max_rounds=400, base_active=10, grow=2, max_active=100):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    failed = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if failed > 30:
            work = project_feasible(work + 0.005 * np.random.randn(n), n)
            failed = 0
            continue
        c = get_corr(work, n)
        m = float(c.max())
        na = min(base_active + (r // 15) * grow, max_active)
        thresh = 0.008 * m + 1e-8
        active = np.where(c > m - thresh)[0]
        if len(active) > na:
            active = active[np.argsort(-c[active])[:na]]
        elif len(active) == 0:
            active = np.argsort(-c)[:na]
        D = build_D(work, n)
        A_ub = np.hstack([D[active], np.ones((len(active), 1))])
        b_ub = np.zeros(len(active))
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        c_obj = np.zeros(n+1)
        c_obj[n] = -1.0
        bnds = [(-1.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bnds, method='highs')
        except Exception:
            failed += 1
            continue
        if not res.success or res.x[n] < 1e-9:
            failed += 1
            continue
        delta = res.x[:n]
        pos = delta > 1e-8
        neg = delta < -1e-8
        step = 1.0
        if np.any(pos):
            step = min(step, ((1.0 - work[pos]) / delta[pos]).min())
        if np.any(neg):
            step = min(step, (work[neg] / (-delta[neg])).min())
        step = max(step * 0.9, 1e-10)
        new_h = project_feasible(work + step * delta, n)
        nc = compute_c5(new_h, n)
        if nc < bc - 1e-12:
            bc, best = nc, new_h.copy()
            failed = 0
        else:
            failed += 1
        work = new_h
    return best, bc

def top_n_binary(h, n):
    hb = np.zeros(n)
    hb[np.argsort(-h)[:n//2]] = 1.0
    return hb

def binary_greedy(h, n, deadline, max_sweeps=40, chunk=2048):
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, "full").astype(float)
    M = int(round(corr.max()))
    besth, bestM = h.copy(), M
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        D = build_D(h, n)
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            end = min(start + chunk, zi.size)
            i = zi[start:end]
            j = oj[start:end]
            arr = D[i] - D[j] + corr[None, :]
            arr[:, n-1] -= 2.0
            d1 = n-1 + (j - i)
            d2 = n-1 + (i - j)
            ridx = np.arange(len(i))
            v1 = (d1 >= 0) & (d1 < 2*n-1)
            v2 = (d2 >= 0) & (d2 < 2*n-1)
            arr[ridx[v1], d1[v1]] += 1.0
            arr[ridx[v2], d2[v2]] += 1.0
            mx = arr.max(axis=1)
            idx = np.where(mx <= M - 1 + 1e-8)[0]
            if idx.size:
                loc = idx[np.argmin(mx[idx])]
                h[i[loc]] = 1.0
                h[j[loc]] = 0.0
                corr = arr[loc].copy()
                M = int(round(corr.max()))
                if M < bestM:
                    bestM, besth = M, h.copy()
                improved = True
                break
        if not improved:
            break
    return besth, bestM * (2.0 / float(n))

def variants(h, n):
    out = [h, h[::-1], 1.0 - h[::-1], 0.5*(h + h[::-1])]
    if n % 2 == 0:
        out.append(top_n_binary(h, n))
    return [project_feasible(x, n) for x in out]
```
