    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()

            break

            s = room.sum()
            if s < 1e-14:

            if s < 1e-14:

def compute_subgrad(c_idx, h, n):
    """Subgradient of c_k at h, where c_k = sum h[i]*(1-h[i+k])"""
    k = c_idx - (n - 1)  # convert from full correlate index to lag
    grad = np.zeros(n)
    if k >= 0:
        # c_k = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])

        # d(c_k)/d(h[i+k]) = -h[i] for i in [0, n-1-k], i.e., h[j] for j in [k, n-1]
        if k == 0:

        else:
            grad[:n-k] += (1.0 - h[k:])

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(c_idx, h, n):
    k = c_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
    else:
        kk = -k
        if kk == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[kk:] += (1.0 - h[:n - kk])
            grad[:n - kk] -= h[kk:]
    return grad

def slsqp_step(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

            h = project_feasible(h, n)

        top_idx = np.argsort(c)[::-1][:top_k]

            grad += compute_subgrad(int(idx), h, n)
