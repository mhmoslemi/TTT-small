import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    a = alpha * (corr - m)
    am = a.max()
    log_sum = am + np.log(np.exp(a - am).sum())
    return m + log_sum / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    a = alpha * (corr - m)
    am = a.max()
    p = np.exp(a - am)
    p /= p.sum()
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        w = p[li]
        if w < 1e-13:
            continue
        if 0 <= l < n:
            grad[:n-l] += w * (1.0 - x[l:])
            grad[l:] -= w * x[:n-l]
        elif -n < l < 0:
            ll = -l
            grad[ll:] += w * (1.0 - x[:n-ll])
            grad[:n-ll] -= w * x[ll:]
    grad *= dx
    return grad

def _eq_fun(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_step(h, n, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h
    obj_fn = partial(_smooth_obj, n=n, alpha=alpha)
    jac_fn = partial(_smooth_grad, n=n, alpha=alpha)
    con_fn = partial(_eq_fun, n=n)
    con_jac = partial(_eq_jac, n=n)
    cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h, jac=jac_fn, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_descent(h, n, deadline, mu=0.91):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = dx * 1.2
    momentum = np.zeros(n)
    stall = 0
    step = 0
    while time.time() < deadline:
        step += 1
        corr = np.correlate(h, 1 - h, mode='full') * dx
        m = corr.max()
        max_mask = corr >= m - 1e-12 * m
        count = max_mask.sum()
        if count == 0:
            break
        grad = np.zeros(n)
        idxs = np.where(max_mask)[0]
        for li in idxs:
            l = int(li) - (n - 1)
            if 0 <= l < n:
                grad[:n-l] += (1.0 - h[l:])
                grad[l:] -= h[:n-l]
            elif -n < l < 0:
                ll = -l
                grad[ll:] += (1.0 - h[:n-ll])
                grad[:n-ll] -= h[ll:]
        grad /= count
        grad *= dx
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.02, dx * 25.0)
        else:
            stall += 1
            if stall > 200:
                lr *= 0.88
                mu = max(mu - 0.003, 0.5)
                if lr < dx * 0.01:
                    lr = dx * 0.5
                    stall = 0
                    mu = 0.91
        h = h_new
    return best_h, best_c

    else:
        h = np.full(n, 0.5)
