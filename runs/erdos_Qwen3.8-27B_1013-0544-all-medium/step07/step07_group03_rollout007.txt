def project_feasible(h,n):
    h=np.asarray(h,float).copy(); S=n/2.0
    h=np.clip(h,0,1)
    for _ in range(100):
        diff=S-h.sum()
        if abs(diff)<1e-13: break
        if diff>0:
            room=1-h
            s=room.sum()
            if s<=1e-15: break
            h += diff/s*room
        else:
            s=h.sum()
            if s<=1e-15: break
            h += diff/s*h
    h=np.clip(h,0,1)
    # exact adjust on least saturated
    diff=S-h.sum()
    if abs(diff)>1e-15:
       if diff>0:
          order=np.argsort(1-h)
       else: order=np.argsort(h)
       for idx in order:
          if diff==0: break
          if diff>0:
             add=min(1-h[idx], diff)
             h[idx]+=add; diff-=add
          else:
             sub=min(h[idx], -diff)
             h[idx]-=sub; diff+=sub
    return h

def lag_gradient(h,n,lag):
   g=np.zeros(n)
   if lag==0: return 1-2*h
   if lag>0:
      g[lag:]+=1.0-h[:n-lag]
      g[:n-lag]-=h[lag:]
   else:
      q=-lag
      g[:n-q]+=1.0-h[q:]
      g[q:]-=h[:n-q]

import numpy as np
import time
from scipy.optimize import minimize, lsq_linear


def quick_project(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    S = 0.5 * n
    for _ in range(30):
        diff = S - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(h, n):
    h = quick_project(h, n)
    S = 0.5 * n
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 0:
                    break
                add = min(1.0 - h[idx], diff)
                h[idx] += add
                diff -= add
        else:
            need = -diff
            order = np.argsort(h)
            for idx in order:
                if need <= 0:
                    break
                sub = min(h[idx], need)
                h[idx] -= sub
                need -= sub
    return np.clip(h, 0.0, 1.0)


def cvec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def cmax(h, n):
    return float(np.max(cvec(h, n)))


def cons_fun(x, n):
    return x.sum() - 0.5 * n


def cons_jac(x, n):
    return np.ones_like(x)


def lag_gradient(h, n, lag):
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        q = -lag
        g[:n - q] += 1.0 - h[q:]
        g[q:] -= h[:n - q]
    return g


def weighted_grad(h, n, w):
    g = np.zeros(n)
    tol = 1e-14
    w0 = w[n - 1]
    if abs(w0) > tol:
        g += w0 * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[n - 1 + p]
        if abs(wp) > tol:
            g[p:] += wp * (1.0 - h[:n - p])
            g[:n - p] -= wp * h[p:]
        wn = w[n - 1 - p]
        if abs(wn) > tol:
            g[:n - p] += wn * (1.0 - h[p:])
            g[p:] -= wn * h[:n - p]
    return g


def fun_lse(x, n, alpha, beta):
    c = cvec(x, n)
    m = float(c.max())
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    s = np.exp(z).sum()
    val = m + np.log(s) / alpha
    if beta:
        val += beta * float(np.sum(c * c))
    return val


def jac_lse(x, n, alpha, beta):
    c = cvec(x, n)
    m = float(c.max())
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    ez = np.exp(z)
    p = ez / ez.sum()
    w = p if not beta else p + 2.0 * beta * c
    return (2.0 / n) * weighted_grad(x, n, w)


def slsqp_step(x, n, alpha, beta, maxiter):
    x0 = quick_project(x, n)
    if n < 4:
        return x0
    bounds = [(0.0, 1.0)] * n
    con = {"type": "eq", "fun": cons_fun, "jac": cons_jac, "args": (n,)}
    try:
        res = minimize(
            fun_lse,
            x0,
            method="SLSQP",
            jac=jac_lse,
            bounds=bounds,
            constraints=con,
            args=(n, alpha, beta),
            options={"maxiter": max(20, int(maxiter)), "ftol": 1e-11},
        )
        y = res.x
        if y is None or not np.all(np.isfinite(y)):
            return x0
        return quick_project(y, n)
    except Exception:
        return x0


def solve_kkt(A):
    m, n = A.shape
    ones = np.ones(n)
    Am = np.column_stack((A.T, -ones))
    b = np.zeros(n)
    lb = np.zeros(n + 1)
    ub = np.full(n + 1, np.inf)
    ub[n] = 100.0
    lb[n] = -100.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :m] = 1.0
    beq = np.array([1.0])
    try:
        res = lsq_linear(
            Am,
            b,
            bounds=(lb, ub),
            A_eq=Aeq,
            b_eq=beq,
            method="trf",
            check_finite=False,
        )
        if res.success or np.all(np.isfinite(res.x)):
            return np.maximum(res.x[:m], 0.0), float(res.x[m])
    except Exception:
        pass
    q = np.full(m, 1.0 / m)
    G = A.T.dot(q)
    return q, float(G.mean())


def active_descent(h, n, deadline, rng, active_m=32):
    h = quick_project(h, n).copy()
    c = cvec(h, n)
    cur = float(c.max())
    best = h.copy()
    best_c = cur
    base = 0.45
    stall = 0
    failed = 0
    L = 2 * n - 1
    while time.time() < deadline:
        m = min(active_m, L)
        if stall > 120:
            m = min(2 * m, L)
        if stall > 280:
            m = min(3 * m, L)
        idx = np.argsort(c)[::-1][:m]
        A = np.empty((m, n))
        for a in range(m):
            A[a] = lag_gradient(h, n, int(idx[a] - (n - 1)))
        A *= 2.0 / n
        q, lam = solve_kkt(A)
        G = A.T.dot(q) - lam
        d = G - G.mean()
        norm = float(np.linalg.norm(d))
        if norm < 1e-10:
            amp = 0.004 + 0.012 * (stall > 200)
            rn = rng.randn(n)
            hn = quick_project(h + amp * rn, n)
            cn = cvec(hn, n)
            cm = float(cn.max())
            if cm < cur + 1e-4:
                h = hn
                c = cn
                cur = cm
                if cur < best_c:
                    best = hn.copy()
                    best_c = cur
                    failed = 0
                base = 0.15
            else:
                h = best.copy()
                c = cvec(h, n)
                cur = float(c.max())
                failed += 1
            base = 0.08
            stall += 1
            if failed > 220 or stall > 4000:
                break
            continue
        dirn = -d / norm
        acc = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.04):
            step = base * mult
            if step <= 1e-7:
                continue
            hn = quick_project(h + step * dirn, n)
            cn = cvec(hn, n)
            cm = float(cn.max())
            if cm < cur - 1e-13:
                h = hn
                c = cn
                cur = cm
                if cur < best_c:
                    best = hn.copy()
                    best_c = cur
                    failed = 0
                base = min(1.0, step * 1.25)
                stall = 0
                acc = True
                break
        if not acc:
            base *= 0.58
            stall += 1
            if base < 1e-6 or stall > 240:
                amp = 0.003 + 0.01 * (stall > 480)
                rn = rng.randn(n)
                hn = quick_project(h + amp * rn, n)
                cn = cvec(hn, n)
                cm = float(cn.max())
                if cm < cur + 2e-4:
                    h = hn
                    c = cn
                    cur = cm
                    if cur < best_c:
                        best = hn.copy()
                        best_c = cur
                        failed = 0
                else:
                    h = best.copy()
                    c = cvec(h, n)
                    cur = float(c.max())
                    failed += 1
                base = 0.09
                stall = 0
                if failed > 260:
                    break
    return project_feasible(best, n), float(best_c)


def resample(h, n, m):
    h = np.asarray(h, dtype=float).copy()
    if n == m:
        return h
    if n > m and n % m == 0:
        b = n // m
        h = h.reshape(b, m).mean(axis=0)
        return quick_project(h, m)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(m) + 0.5) * (2.0 / m)
    return quick_project(np.interp(xn, xo, h), m)


def make_start(m, typ, rng):
    x = (np.arange(m) + 0.5) / m
    h = np.full(m, 0.5)
    a = 0.45
    if typ == 1:
        h += a * np.cos(2.0 * np.pi * x)
    elif typ == 2:
        h += a * np.sin(2.0 * np.pi * x)
    elif typ == 3:
        h += a * np.cos(4.0 * np.pi * x)
    elif typ == 4:
        h += a * np.tanh(4.0 * (x - 0.5))
    elif typ == 5:
        h += a * np.exp(-5.0 * (x - 0.5) ** 2)
    elif typ == 6:
        h += 0.6 * a * (np.exp(-5.0 * x * x) + np.exp(-5.0 * (1.0 - x) ** 2))
    elif typ == 7:
        d = rng.randn(m)
        k = 5
        h += a * np.convolve(d, np.ones(k) / k, mode="same")
    else:
        h = np.zeros(m)
        idx = rng.choice(m, m // 2, replace=False)
        h[idx] = 1.0
    return project_feasible(h, m)


def refine_candidate(h, n, end, rng):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = cmax(h, n)
    cur_c = best_c
    cur_h = h.copy()
    if n > 250:
        alphas = [200.0, 2000.0, 20000.0, 200000.0]
        iters = 70
    elif n > 100:
        alphas = [100.0, 1000.0, 10000.0, 100000.0, 1000000.0]
        iters = 130
    else:
        alphas = [50.0, 500.0, 5000.0, 50000.0, 200000.0, 1000000.0]
        iters = 220
    for alpha in alphas:
        rem = end - time.time()
        if rem < 4:
            break
        mi = max(18, min(iters, int(rem * 12)))
        h2 = slsqp_step(cur_h, n, alpha, 0.0, mi)
        c2 = cmax(h2, n)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
        if c2 < cur_c + 1e-8:
            cur_h = h2
            cur_c = c2
        else:
            cur_h = best_h.copy()
            cur_c = best_c
    if time.time() < end - 2:
        am = 48 if n >= 250 else (32 if n >= 100 else min(2 * n, 64))
        h3, c3 = active_descent(cur_h, n, end, rng, active_m=am)
        if c3 < best_c:
            best_h = h3.copy()
            best_c = c3
    return project_feasible(best_h, n), float(best_c)
