import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    arg = alpha * (c - m)
    amax = arg.max()
    e = np.exp(arg - amax)
    es = e.sum()
    p = e / es
    obj = m + (amax + np.log(es)) / alpha + penalty * (x.sum() - target) ** 2
    F = np.zeros(n)
    B = np.zeros(n)
    W = np.zeros(n)
    for mi in range(2 * n - 1):
        w = p[mi]
        if w < 1e-15:
            continue
        l = mi - (n - 1)
        if l >= 0:
            F[:n - l] += w * x[l:]
            B[l:] += w * x[:n - l]
            W[:n - l] += w
        else:
            ll = -l
            F[ll:] += w * x[:n - ll]
            B[:n - ll] += w * x[ll:]
            W[ll:] += w
    grad = dx * (W - F - B) + 2.0 * penalty * (x.sum() - target)
    return obj, grad

def lbfgs_optimize(h, n, alpha, penalty, maxiter=5000):
    def fun(x):
        return obj_grad(x, n, alpha, penalty)[0]
    def fprime(x):
        return obj_grad(x, n, alpha, penalty)[1]
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h, jac=fprime, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-14})
    return project_feasible(res.x, n)

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def downsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def anneal_alphas(h, n, alphas, penalties, maxiter, deadline):
    for i in range(len(alphas)):
        if time.time() > deadline - 10:
            break
        a = alphas[i]
        p = penalties[i] if i < len(penalties) else max(100.0, a * 0.05)
        h = lbfgs_optimize(h, n, a, p, maxiter=maxiter)
    return h
