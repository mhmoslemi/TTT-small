def lse_objective_grad(h,n,beta):
    h=np.clip(np.asarray(h,dtype=float).ravel(),0,1)
    dx=2.0/float(n); v=1.0-h
    c=np.correlate(h,v,mode='full')*dx
    c=np.maximum(c,0.0); m=float(c.max())
    y=beta*(c-m); e=np.exp(np.minimum(y,0.0)); s=float(e.sum())
    if not np.isfinite(s) or s<=0:
       P=np.full(2*n-1,1.0/float(2*n-1)); F=m
    else:
       P=e/s; F=m+np.log(s)/beta
    A=np.convolve(P[::-1],v,mode='full')[n-1:2*n-1]
    B=np.convolve(P,h,mode='full')[n-1:2*n-1]
    g=dx*(A-B)
    if not np.all(np.isfinite(g)): g=np.zeros(n)
    return float(F),g

    h = np.asarray(h, dtype=float).ravel()

                improved = True
                if M < bestM or (M == bestM and S < bestS):
                    bestM, bestS = M, S; bestx = x.copy()
        if not improved:
            break
    return bestx, bestM

def pattern_bits(m, p):
    p = max(2, int(p))
    half = p // 2
    pat = np.concatenate([np.ones(half), np.zeros(p - half)])
    return np.tile(pat, (m + p - 1) // p)[:m].astype(np.uint8)

def find_best_pair(x, c, n, zeros, ones, curM, curS, deadline):
    if not zeros or not ones:
        return None
    L = 2 * n - 1
    zarr = np.asarray(zeros, dtype=int)
    oarr = np.asarray(ones, dtype=int)
    base = np.zeros((n, L), dtype=np.int32)
    lag = np.arange(-(n - 1), n, dtype=int)
    for p in range(n):
        f = p + lag
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1 - x[f[mf]]
        b = p - lag
        mb = (b >= 0) & (b < n)
        base[p, mb] -= x[b[mb]]
    zi = np.repeat(zarr, oarr.size)
    oj = np.tile(oarr, zarr.size)
    corr = c.astype(np.int32)
    big = n * n * 4
    bestsc = curM * big + curS
    best = None
    chunk = 512
    for st in range(0, zi.size, chunk):
        if time.time() > deadline:
            break
        en = min(st + chunk, zi.size)
        cnt = en - st
        i = zi[st:en]
        j = oj[st:en]
        arr = base[i] - base[j] + corr[None, :]
        arr[:, n - 1] -= 2
        rows = np.arange(cnt)
        d1 = n - 1 + (j - i)
        d2 = n - 1 + (i - j)
        arr[rows, d1] += 1
        arr[rows, d2] += 1
        mx = arr.max(axis=1).astype(np.int64)
        good = mx <= curM
        if not np.any(good):
            continue
        sm = arr[good].sum(axis=1)
        sc = mx[good] * big + sm
        lg = int(np.argmin(sc))
        if sc[lg] < bestsc:
            loc = int(np.flatnonzero(good)[lg])
            best = (int(i[loc]), int(j[loc]), int(mx[loc]), int(sm[loc]))
            bestsc = int(sc[lg])
    return best

def full_binary_optimize(x, n, deadline, rng):
    x = np.asarray(x, dtype=np.uint8).ravel().copy()
    D, E, M, S = binary_de(x, n)
    bestx = x.copy()
    bestM, bestS = M, S
    dd = np.zeros(n, dtype=np.int32)
    ee = np.zeros(n, dtype=np.int32)
    big = n * n * 4
    stuck = 0
    while time.time() < deadline:
        zeros = [int(i) for i in range(n) if x[i] == 0]
        ones = [int(i) for i in range(n) if x[i] == 1]
        c = np.zeros(2 * n - 1, dtype=np.int32)
        if n > 1:
            idx = np.arange(1, n)
            c[n - 1 + idx] = D[1:]
            c[n - 1 - idx] = E[1:]
        bp = find_best_pair(x, c, n, zeros, ones, M, S, deadline)
        if bp is not None:
            z, o, nm, ns = bp
            x[z] = 1; x[o] = 0
            D, E, M, S = binary_de(x, n)
            if M < bestM or (M == bestM and S < bestS):
                bestM, bestS = M, S; bestx = x.copy()
            stuck = 0
            continue
        cur = M * big + S
        moved = False
        for _ in range(4500):
            if time.time() > deadline or not zeros or not ones:
                break
            z = int(zeros[int(rng.randint(len(zeros)))])
            o = int(ones[int(rng.randint(len(ones)))])
            pair_delta(x, z, o, n, dd, ee)
            nm, ns = new_score_binary(D, E, dd, ee, n)
            sc = nm * big + ns
            acc = sc < cur
            if not acc and nm <= M + 1:
                if nm == M + 1:
                    pr = 0.75 if ns < S else np.exp(-1.0 / 0.7)
                else:
                    pr = np.exp(-0.001 * max(0, ns - S))
                if rng.random() < pr:
                    acc = True
            if acc:
                D += dd; E += ee
                x[z] = 1; x[o] = 0
                if z in zeros:
                    zeros.remove(z)
                zeros.append(o)
                if o in ones:
                    ones.remove(o)
                ones.append(z)
                M, S, cur = nm, ns, nm * big + ns
                moved = True
                if M < bestM or (M == bestM and S < bestS):
                    bestM, bestS = M, S; bestx = x.copy()
        if moved:
            stuck = 0
        else:
            D, E, M, S = binary_de(x, n)
            stuck += 1
            if stuck > 4:
                break
    return bestx, bestM

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        P = np.full(2 * n - 1, 1.0 / float(2 * n - 1))
        F = m
    else:
        P = e / s
        F = m + float(np.log(s)) / beta
    A = np.convolve(P, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P[::-1], h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return float(F), g

def full_lse_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def anti_objective(a, n, m, beta):
    a = np.clip(np.asarray(a, dtype=float).ravel(), 0.0, 1.0)
    x = build_anti(a, n)
    F, gx = lse_objective_grad(x, n, beta)
    ga = gx[:m] - gx[m:][::-1]
    return F, ga

def full_polish(h, n, betas, deadline, stage_time=45.0):
    global _stop_time
    h = np.clip(project_feasible(h, n), 0.0, 1.0)
    work = h.copy()
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    opts = {"maxiter": 1000000, "maxls": 40, "ftol": 1e-13, "gtol": 1e-8}
    for beta in betas:
        now = time.time()
        if now > deadline - 25:
            break
        stage_dl = min(deadline - 15, now + max(4.0, min(stage_time, 0.55 * max(5.0, deadline - now))))
        _stop_time = stage_dl
        try:
            res = minimize(full_lse_objective, work, method="L-BFGS-B", jac=True, bounds=bounds,
                           args=(n, float(beta), 1e4), callback=stop_callback, options=opts)
            x = res.x if res is not None and hasattr(res, "x") else work
        except Exception:
            x = work
        work = np.clip(project_feasible(x, n), 0.0, 1.0)
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
        for cand in (work, 1.0 - work[::-1], 0.5 * (work + work[::-1]), build_anti(project_anti(work, n), n)):
            cand = np.clip(project_feasible(cand, n), 0.0, 1.0)
            cc = compute_c5(cand, n)
            if cc < bc:
                bc, best = cc, cand.copy()
        if bc < cw - 1e-11:
            work = best.copy()
    return best, bc

def anti_polish(a, n, betas, deadline, stage_time=45.0):
    global _stop_time
    m = n // 2
    a = np.asarray(a, dtype=float).ravel()
    if a.size < m:
        a = np.concatenate([a, np.zeros(m - a.size)])
    else:
        a = a[:m]
    a = np.clip(a, 0.0, 1.0)
    besta = a.copy()
    bestx = build_anti(a, n)
    bc = compute_c5(bestx, n)
    bounds = [(0.0, 1.0)] * m
    opts = {"maxiter": 1000000, "maxls": 40, "ftol": 1e-13, "gtol": 1e-8}
    for beta in betas:
        now = time.time()
        if now > deadline - 25:
            break
        stage_dl = min(deadline - 15, now + max(4.0, min(stage_time, 0.55 * max(5.0, deadline - now))))
        _stop_time = stage_dl
        try:
            res = minimize(anti_objective, a, method="L-BFGS-B", jac=True, bounds=bounds,
                           args=(n, m, float(beta)), callback=stop_callback, options=opts)
            na = res.x if res is not None and hasattr(res, "x") else a
        except Exception:
            na = a
        na = np.clip(na, 0.0, 1.0)
        nx = build_anti(na, n)
        cx = compute_c5(nx, n)
        if cx < bc:
            bc = cx
            bestx = nx.copy()
            besta = na.copy()
        a = besta.copy()
    return bestx, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    end = t0 + max(30.0, total - 12.0)
    rng = np.random.RandomState(seed)
    N = 400
    try:
        ip = np.asarray(initial_h_values, dtype=float).ravel()
        if ip.size >= 2 and np.all(np.isfinite(ip)):
            h = np.interp(np.linspace(0.0, 2.0, N), np.linspace(0.0, 2.0, ip.size), ip) if ip.size != N else ip.copy()
        else:
            raise ValueError
    except Exception:
        h = ((np.arange(N) // 40) % 2).astype(float)
    h = project_feasible(h, N)
    best_h = h.copy()
    best_n = N
    best_c = compute_c5(best_h, best_n)

    m = N // 2
    starts = [(project_anti(best_h, N) >= 0.5).astype(np.uint8)]
    starts.append(np.asarray(rng.randint(0, 2, m), dtype=np.uint8))
    for p in (16, 32, 48, 64, 96, 128, 160, 200):
        starts.append(pattern_bits(m, p))
    adi = min(end - 520.0, t0 + 170.0)
    for st in starts:
        if time.time() > adi - 5.0:
            break
        x = build_anti(st, N).astype(np.uint8)
        bx, bm = anti_hill(x, N, m, adi, rng)
        c = float(bm) * 2.0 / float(N)
        c = compute_c5(bx.astype(float), N)
        if c < best_c:
            best_c, best_h, best_n = c, bx.astype(float), N

    if time.time() < end - 350.0:
        fdi = min(end - 340.0, time.time() + 300.0)
        bx = top_n_binary(best_h, best_n)
        fy = full_binary_optimize(bx, best_n, fdi, rng)
        bx, bm = fy
        c = compute_c5(bx.astype(float), best_n)
        if c < best_c:
            best_c, best_h, best_n = c, bx.astype(float), best_n

    if time.time() < end - 230.0:
        cdl = min(end - 220.0, time.time() + 210.0)
        fh, fc = full_polish(best_h, best_n, [30, 100, 400, 1500, 6000, 20000], cdl, 42.0)
        if fc < best_c:
            best_c, best_h, best_n = fc, fh, best_n

    if time.time() < end - 180.0:
        cdl = min(end - 170.0, time.time() + 160.0)
        ax, ac = anti_polish(project_anti(best_h, best_n), best_n, [50, 200, 800, 3000, 10000], cdl, 40.0)
        if ac < best_c:
            best_c, best_h, best_n = ac, ax, best_n

    if time.time() < end - 150.0:
        cdl = min(end - 140.0, time.time() + 70.0)
        ax, ac = anti_polish(pattern_bits(project_anti(best_h, best_n).size, 64), best_n, [200, 1000, 5000], cdl, 35.0)
        if ac < best_c:
            best_c, best_h, best_n = ac, ax, best_n
        if time.time() < end - 90.0:
            ax, ac = anti_polish(np.asarray(rng.rand(best_n // 2), dtype=float), best_n, [200, 1000, 5000], min(end - 90.0, time.time() + 50.0), 35.0)
            if ac < best_c:
                best_c, best_h, best_n = ac, ax, best_n

    if time.time() < end - 160.0:
        n2 = 800
        x0 = np.linspace(0.0, 2.0, best_n)
        x1 = np.linspace(0.0, 2.0, n2)
        h2 = project_feasible(np.interp(x1, x0, best_h), n2)
        if time.time() < end - 120.0:
            fh, fc = full_polish(h2, n2, [100, 500, 2000], min(end - 100.0, time.time() + 80.0), 30.0)
            if fc < best_c:
                best_c, best_h, best_n = fc, fh, n2
        if time.time() < end - 70.0:
            ax, ac = anti_polish(project_anti(best_h, best_n), best_n, [200, 1000, 4000], min(end - 40.0, time.time() + 45.0), 30.0)
            if ac < best_c:
                best_c, best_h, best_n = ac, ax, best_n

    if time.time() < end - 45.0:
        bx = top_n_binary(best_h, best_n)
        bx, bm = full_binary_optimize(bx, best_n, min(end - 25.0, time.time() + 35.0), rng)
        c = compute_c5(bx.astype(float), best_n)
        if c < best_c:
            best_c, best_h, best_n = c, bx.astype(float), best_n

    best_h = np.clip(project_feasible(best_h, best_n), 0.0, 1.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
