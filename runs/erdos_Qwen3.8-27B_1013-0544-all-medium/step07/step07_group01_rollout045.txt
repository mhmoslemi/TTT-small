Let me also consider: what if the optimal solution has h values that are exactly 0 or 1 (a step function)? Then the problem becomes combinatorial. But the constraint sum(h) = n/2 means exactly n/2 entries are 1 and n/2 are 0 (for n even). In that case, f_k = dx * (number of i where h[i]=1 and h[i+k]=0).

For a binary solution with n/2 ones, the maximum overlap is at most n/2 * dx = 1 (when all ones are at one end and all zeros at the other, the lag that maximizes overlap would give n/2 pairs). Actually no, let me think more carefully.

If h is binary with n/2 ones and n/2 zeros, then h(i)*(1-h(i+k)) = 1 if h(i)=1 and h(i+k)=0. The sum over i counts the number of "1 followed by 0 at distance k" transitions. For k=0, this is 0 (since h(i)*(1-h(i)) = 0 always). For k>0, this depends on the arrangement.

The maximum possible value of this for any arrangement is at most n/2 (when all the 1s are in the first half and all the 0s are in the second half, then for k = n/2, all n/2 ones have a 0 at distance k). So C5 could be as high as n/2 * dx = 1.

But we want to minimize the maximum, so we want to arrange the 1s and 0s so that for every k, the number of "1-0 at distance k" pairs is as balanced as possible. This is related to the problem of designing a binary sequence with minimal maximum autocorrelation difference.

The theoretical minimum for binary sequences is related to the "minimum overlap" problem that Erdős studied. For n/2 ones and n/2 zeros, the minimum possible maximum overlap is approximately sqrt(n)/2 * something... I don't remember the exact asymptotic.

Anyway, for our purposes, the optimal continuous solution is what we're after, and it's likely not purely binary.

Actually, one more thought: the problem with very high alpha in log-sum-exp is numerical. When alpha is very large (like 50000000), the exp(alpha * (corr - m)) will underflow for all but the maximum element. This means the gradient only points in the direction of the single worst lag, which is effectively a subgradient step. This is fine for the optimization but the "smoothness" is lost.

A better approach might be to use a moderate alpha (like 1000-10000) for the bulk of the optimization, and then switch to a subgradient method for the final polishing.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    p = np.exp(sh - sm)
    p /= p.sum()
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        w = p[n - 1 + l]
        if w < 1e-12:
            continue
        if l >= 0:
            g[:n - l] += w * (1.0 - x[l:])
            g[l:] -= w * x[:n - l]
        else:
            ll = -l
            g[ll:] += w * (1.0 - x[:n - ll])
            g[:n - ll] -= w * x[ll:]
    g *= dx
    return g

def slbq_opt(h, n, alpha, maxiter=300, ftol=1e-15):
    def obj(x):
        return smooth_obj(x, n, alpha)
    def grad(x):
        return smooth_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-6, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def subgrad_holy(h, n, deadline, base_lr=None):
    dx = 2.0 / n
    if base_lr is None:
        base_lr = dx * 3.0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = base_lr
    stall = 0
    for step in range(1000000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        l = k_star - (n - 1)
        grad = np.zeros(n)
        if l >= 0 and l < n:
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] -= h[:n - l]
        elif l < 0 and -l < n:
            ll = -l
            grad[ll:] += (1.0 - h[:n - ll])
            grad[:n - ll] -= h[ll:]
        grad *= dx
        h = project_feasible(h - lr * grad, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, dx * 50.0)
        else:
            stall += 1
            if stall > 200:
                lr *= 0.9
                if lr < dx * 0.005:
                    lr = dx * 0.5
                    stall = 0
    return best_h, best_c

def multi_alpha_slbq(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    cur_h = best_h.copy()
    for a in alphas:
        if time.time() > deadline - 30:
            break
        cur_h = slbq_opt(cur_h, n, a, maxiter=400, ftol=1e-14)
        c = compute_c5(cur_h, n)
        if c < best_c:
            best_c = c
            best_h = cur_h.copy()
    return best_h, best_c, best_c

def coarse_init(rng, n_target, t_limit):
    best_h = np.full(n_target, 0.5)
    best_c = compute_c5(best_h, n_target)
    sizes = [6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192]
    h_cur = None
    for cs in sizes:
        if time.time() > t_limit:
            break
        if h_cur is not None:
            x_old = np.linspace(0, 2, h_cur.size)
            x_new = np.linspace(0, 2, cs)
            h_tmp = np.interp(x_new, x_old, h_cur)
        else:
            h_tmp = rng.uniform(0, 1, cs)
        h_tmp = project_feasible(h_tmp, cs)
        for a in [50, 500, 5000]:
            if time.time() > t_limit:
                break
            h_tmp = slbq_opt(h_tmp, cs, a, maxiter=150, ftol=1e-12)
        h_cur = h_tmp.copy()
    if h_cur is not None:
        x_old = np.linspace(0, 2, h_cur.size)
        x_new = np.linspace(0, 2, n_target)
        h_fine = np.interp(x_new, x_old, h_cur)
        h_fine = project_feasible(h_fine, n_target)
        c_fine = compute_c5(h_fine, n_target)
        if c_fine < best_c:
            best_c, best_h = c_fine, h_fine
    return best_h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:

        n = len(h_init)

    h_init = project_feasible(h_init, n)

    # Phase 1: Coarse-to-fine initialization (try to find better basin)
    t1 = min(t0 + 120, deadline - 300)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
