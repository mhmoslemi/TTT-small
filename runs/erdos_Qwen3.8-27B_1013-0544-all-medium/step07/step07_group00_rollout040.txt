def quad_feas(t, beta, e0, dlo, dhi, line_s, line_c, diag_c, A0):
    lo=dlo; hi=dhi
    if line_s.size:
       const = 2.0*beta*e0
       q0 = beta*e0*e0 - t
       b = line_s + const
       c0 = line_c + q0
       D = b*b - 4.0*beta*c0
       if np.any(D < -1e-9): return None
       D = np.maximum(D, 0.0)
       r = np.sqrt(D)
       inv = 1.0/(2.0*beta)
       left = (-b - r)*inv
       right = (-b + r)*inv
       max_left = float(np.max(left))
       min_right = float(np.min(right))
       if max_left > lo: lo=max_left
       if min_right < hi: hi=min_right
       if lo > hi + 1e-12: return None
    a = beta - 1.0
    b = A0 + 2.0*beta*e0
    c0 = diag_c + beta*e0*e0 - t
    D = b*b - 4.0*a*c0
    if D < -1e-9: return None
    if D < 0: D=0.0

    if a > 1e-12:

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(3):
        s = float(h.sum())
        if s < target:
            denom = float(n) - s
            if denom <= 1e-15:
                break
            alpha = (target - s) / denom
            if alpha > 0.0:
                h = h + alpha * (1.0 - h)
        elif s > target:
            if s <= 1e-15:
                h = np.full(n, target / n)
            else:
                h = h * (target / s)
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 1e-15:
                h = h + (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot > 1e-15:
                h = h * (target / tot)  # since h positive, reduces sum exactly
        h = np.clip(h, 0.0, 1.0)
    # final hard fix at most a few coords
    diff = target - float(h.sum())
    if abs(diff) > 0.0:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(room)
            for idx in order[::-1]:
                if diff <= 1e-15: break
                add = min(diff, room[idx])
                if add > 0: h[idx] += add; diff -= add
        else:
            vals = h.copy()
            order = np.argsort(vals)
            for idx in order:   # largest first? order[::-1]
                if diff >= -1e-15: break
                sub = min(-diff, vals[idx])
                if sub > 0: h[idx] -= sub; diff += sub
    return h

    h = project_feasible(h, n)

import time

from scipy.optimize import minimize

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_from_corr(c, n):
    return float(np.max(c) * (2.0 / float(n)))

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(3):
        s = float(h.sum())
        if s < target:
            denom = float(n) - s
            if denom <= 1e-15:
                break
            alpha = (target - s) / denom
            if alpha > 0.0:
                h = h + alpha * (1.0 - h)
        elif s > target:
            if s <= 1e-15:
                h = np.full(n, target / float(n))
            else:
                h = h * (target / s)
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 1e-15:
                h = h + (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot > 1e-15:
                h = h * (target / tot)
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())

def smooth_fun(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * c
    amax = float(a.max())
    e = np.exp(a - amax)
    se = float(e.sum())
    return (amax + np.log(se)) * (2.0 / (alpha * float(n)))

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * c
    amax = float(a.max())
    e = np.exp(a - amax)
    se = float(e.sum())
    w = e / se
    dx = 2.0 / float(n)
    w_rev = w[::-1]
    g1 = np.convolve(1.0 - x, w_rev, mode='full')[n-1:2*n-1]
    g2 = np.convolve(x, w, mode='full')[n-1:2*n-1]
    return dx * (g1 - g2)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alphas, maxiter, deadline):
    h = project_feasible(h, n)
    cc = c5_from_corr(compute_corr(h, n), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_fun, h, args=(n, float(alpha)), method='SLSQP',
                           bounds=bounds, constraints=cons, jac=smooth_jac,
                           options={'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': 0})
            x = project_feasible(np.asarray(res.x, dtype=float), n)
            val = c5_from_corr(compute_corr(x, n), n)
            if val < cc:
                h = x
                cc = val
        except Exception:
            pass
    return h, cc

def quad_feas(t, beta, e0, dlo, dhi, line_s, line_c, diag_c, A0):
    lo = dlo
    hi = dhi
    if line_s.size:
        const = 2.0 * beta * e0
        q0 = beta * e0 * e0 - t
        b = line_s + const
        c0 = line_c + q0
        D = b * b - 4.0 * beta * c0
        if np.any(D < -1e-8):
            return None
        D = np.maximum(D, 0.0)
        r = np.sqrt(D)
        inv = 1.0 / (2.0 * beta)
        left = (-b - r) * inv
        right = (-b + r) * inv
        ml = float(np.max(left))
        mr = float(np.min(right))
        if ml > lo:
            lo = ml
        if mr < hi:
            hi = mr
        if lo > hi + 1e-11:
            return None
    a = beta - 1.0
    b = A0 + 2.0 * beta * e0
    c0 = diag_c + beta * e0 * e0 - t
    D = b * b - 4.0 * a * c0
    if D < -1e-8:
        return None
    if D < 0.0:
        D = 0.0
    r = float(np.sqrt(D))
    if a > 1e-12:
        inv = 1.0 / (2.0 * a)
        left = (-b - r) * inv
        right = (-b + r) * inv
        if left > lo:
            lo = left
        if right < hi:
            hi = right
    else:
        if abs(b) < 1e-14:
            if c0 > 1e-10:
                return None
        else:
            bound = -c0 / b
            if b > 0:
                if bound < hi:
                    hi = bound
            else:
                if bound > lo:
                    lo = bound
    if lo > hi + 1e-11:
        return None
    return (lo, hi)

import time
import numpy as np
from scipy.optimize import minimize


def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_from_corr(c, n):
    return float(np.max(c) * (2.0 / float(n)))


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(3):
        s = float(h.sum())
        if s < target:
            denom = float(n) - s
            if denom <= 1e-15:
                break
            alpha = (target - s) / denom
            if alpha > 0.0:
                h = h + alpha * (1.0 - h)
        elif s > target:
            if s <= 1e-15:
                h = np.full(n, target / float(n))
            else:
                h = h * (target / s)
        else:
            break
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 1e-15:
                h = h + (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot > 1e-15:
                h = h * (target / tot)
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 0.0:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(room)
            need = diff
            for idx in order[::-1]:
                if need <= 1e-15:
                    break
                add = room[idx]
                if add > need:
                    add = need
                if add > 0.0:
                    h[idx] += add
                    need -= add
        else:
            order = np.argsort(h)
            need = -diff
            for idx in order[::-1]:
                if need <= 1e-15:
                    break
                sub = h[idx]
                if sub > need:
                    sub = need
                if sub > 0.0:
                    h[idx] -= sub
                    need -= sub
    return h


def smooth_fun(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    a = alpha * c
    amax = float(a.max())
    e = np.exp(a - amax)
    se = float(e.sum())
    return (amax + np.log(se)) * (2.0 / (alpha * float(n)))


def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    a = alpha * c
    amax = float(a.max())
    e = np.exp(a - amax)
    se = float(e.sum())
    w = e / se
    dx = 2.0 / float(n)
    w_rev = w[::-1]
    g1 = np.convolve(1.0 - x, w_rev, mode="full")[n - 1:2 * n - 1]
    g2 = np.convolve(x, w, mode="full")[n - 1:2 * n - 1]
    return dx * (g1 - g2)


def sum_con(x, n):
    return float(x.sum() - 0.5 * n)


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp_refine(h, n, alphas, maxiter, deadline):
    h = project_feasible(h, n)
    cc = c5_from_corr(compute_corr(h, n), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                smooth_fun,
                h,
                args=(n, float(alpha)),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                jac=smooth_jac,
                options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": 0},
            )
            x = project_feasible(np.asarray(res.x, dtype=float), n)
            val = c5_from_corr(compute_corr(x, n), n)
            if val < cc:
                h = x
                cc = val
        except Exception:
            pass
    return h, cc


def pen_value(d, line_s, line_c, diag_c, A0, e0, beta):
    if line_c.size:
        fl = float(np.max(line_c + line_s * d))
    else:
        fl = -1.0
    fd = diag_c + A0 * d - d * d
    f = fl if fl > fd else fd
    q = e0 + d
    return f + beta * q * q


def pen_cd_pass(h, n, c, S, beta, deadline, perm, gold_iters=26):
    target = 0.5 * float(n)
    M = float(c.max())
    e0 = S - target
    n1 = n - 1
    Mlen = 2 * n - 1
    improved = 0
    gr = 0.6180339887498949

    for p in perm:
        if time.time() > deadline:
            break
        p = int(p)
        h0 = float(h[p])

        m_f_lo = max(0, n1 - p)
        m_f_hi = min(Mlen - 1, Mlen - 1 - p)
        m_b_lo = max(0, p)
        m_b_hi = min(Mlen - 1, p + n1)

        m_lo = min(m_f_lo, m_b_lo)
        m_hi = max(m_f_hi, m_b_hi)
        L = m_hi - m_lo + 1

        slopes = np.zeros(L)

        if m_f_hi >= m_f_lo:
            ar = np.arange(m_f_lo, m_f_hi + 1)
            idx = ar - m_lo
            j = p + ar - n1
            slopes[idx] += 1.0 - h[j]

        if m_b_hi >= m_b_lo:
            ar = np.arange(m_b_lo, m_b_hi + 1)
            idx = ar - m_lo
            j = p - ar + n1
            slopes[idx] -= h[j]

        diag_off = n1 - m_lo if m_lo <= n1 <= m_hi else -1
        c_aff = c[m_lo:m_hi + 1]
        mask = np.ones(L, dtype=bool)
        if diag_off >= 0:
            mask[diag_off] = False

        line_s = slopes[mask]
        line_c = c_aff[mask]

        has_U = False
        Uval = -1.0
        if m_lo > 0:
            Uval = float(c[:m_lo].max())
            has_U = True
        if m_hi < Mlen - 1:
            u2 = float(c[m_hi + 1:].max())
            if not has_U or u2 > Uval:
                Uval = u2
                has_U = True

        if has_U:
            line_s = np.concatenate((line_s, np.array([0.0])))
            line_c = np.concatenate((line_c, np.array([Uval])))

        if has_U and Uval >= M - 1e-14:
            continue

        diag_c = float(c[n1])
        A0 = 1.0 - 2.0 * h0
        dlo = -h0
        dhi = 1.0 - h0
        if dhi - dlo < 1e-15:
            continue

        a = dlo
        b = dhi
        cgt = b - gr * (b - a)
        dgt = a + gr * (b - a)
        fc = pen_value(cgt, line_s, line_c, diag_c, A0, e0, beta)
        fd = pen_value(dgt, line_s, line_c, diag_c, A0, e0, beta)

        for _ in range(gold_iters):
            if fc < fd:
                b = dgt
                dgt = cgt
                fd = fc
                cgt = b - gr * (b - a)
                fc = pen_value(cgt, line_s, line_c, diag_c, A0, e0, beta)
            else:
                a = cgt
                cgt = dgt
                fc = fd
                dgt = a + gr * (b - a)
                fd = pen_value(dgt, line_s, line_c, diag_c, A0, e0, beta)

        lo = float(a)
        hi = float(b)
        mid = 0.5 * (lo + hi)

        cand = [dlo, dhi, lo, hi, mid]
        if dlo <= 0.0 <= dhi:
            cand.append(0.0)
        if beta > 0.0:
            dq = -e0 / beta
            if lo <= dq <= hi:
                cand.append(float(dq))
            elif dlo <= dq <= dhi:
                cand.append(float(dq))
        if dhi - dlo > 1e-14:
            cand.extend(np.linspace(dlo, dhi, 7).tolist())
        if hi - lo > 1e-14:
            cand.extend(np.linspace(lo, hi, 5).tolist())

        best_d = 0.0
        best_F = M + 1.0
        best_e = 1e100

        for dd in cand:
            dd = float(np.clip(dd, dlo, dhi))
            if dd < dlo - 1e-12 or dd > dhi + 1e-12:
                continue
            if line_c.size:
                fl = float(np.max(line_c + line_s * dd))
            else:
                fl = -1.0
            fd = diag_c + A0 * dd - dd * dd
            F = fl if fl > fd else fd
            ae = abs(e0 + dd)
            if F < best_F - 1e-12:
                best_F = F
                best_d = dd
                best_e = ae
            elif abs(F - best_F) <= 1e-12 and ae < best_e:
                best_F = min(best_F, F)
                best_d = dd
                best_e = ae

        if best_F < M - 1e-11:
            d = best_d
            if m_f_hi >= m_f_lo:
                ar = np.arange(m_f_lo, m_f_hi + 1)
                j = p + ar - n1
                c[ar] += d * (1.0 - h[j])
            if m_b_hi >= m_b_lo:
                ar = np.arange(m_b_lo, m_b_hi + 1)
                j = p - ar + n1
                c[ar] += d * (-h[j])
            c[n1] -= d * d
            h[p] = h0 + d
            S += d
            e0 = S - target
            M = float(c.max())
            improved += 1

    return h, c, float(S), M, improved
