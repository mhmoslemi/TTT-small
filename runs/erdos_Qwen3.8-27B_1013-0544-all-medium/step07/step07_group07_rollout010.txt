```python
def compute_gradient(h, w, n):
    """
    Compute grad[j] = sum_offset w[n-1+offset] * ((1-h[j-offset]) - h[j+offset]) * dx
    where h[x]=0 for x<0 or x>=n
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for offset in range(-(n-1), n):
            w_val = w[n-1+offset]
            if w_val == 0:
                continue
            # term 1: (1-h[j-offset])
            idx1 = j - offset
            if 0 <= idx1 < n:
                s += w_val * (1 - h[idx1])
            else:
                s += w_val  # h = 0, so 1-h = 1
            
            # term 2: -h[j+offset]
            idx2 = j + offset
            if 0 <= idx2 < n:
                s -= w_val * h[idx2]
            # else h = 0, no contribution
        grad[j] = s * dx
    return grad
```

def compute_gradient_direct(h, w, n):
    """O(n^2) gradient computation, vectorized over the inner sum."""
    dx = 2.0 / n
    # For each j, grad[j] = sum_p w[n-1+p] * ((1-h[j-p]) - h[j+p]) * dx
    # = (w.sum() - sum_p w[n-1+p]*h[j-p] - sum_p w[n-1+p]*h[j+p]) * dx
    
    # Create index arrays
    # For A[j] = sum_p w[n-1+p] * h[j-p]:
    # j-p must be in [0,n-1], so p in [j-(n-1), j]
    # Let's use a different approach: create a (n, 2n-1) matrix of valid indices
    
    # Actually, the simplest vectorized approach:
    # A = np.zeros(n)
    # for p in range(-(n-1), n):
    #     idx = np.arange(n) - p  # j - p for j=0..n-1
    #     valid = (idx >= 0) & (idx < n)
    #     A[valid] += w[n-1+p] * h[idx[valid]]
    
    # This is O(n^2) but each step is vectorized. Total: 2n-1 steps of O(n) work.
    
    A = np.zeros(n)
    B = np.zeros(n)
    for p in range(n):
        # p goes from -(n-1) to (n-1)
        # Let's handle p and -p together
        pass
    
    # Better approach using rolling/cumsum:
    # A[j] = sum_p w[n-1+p] * h[j-p] (valid)
    # This is a convolution!
    # A = correlate(h, w, ...) with appropriate slicing
    
    # Let me just use the loop version but vectorize the inner:
    for p in range(-(n-1), n):
        wp = w[n-1+p]
        if wp == 0:
            continue
        # h[j-p] for j=0..n-1: valid when 0<=j-p<n, i.e., p<=j<p+n

        j_max = min(n, p+n)

            A[j_min:j_max] += wp * h[j_min-p:j_max-p]

        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h, best_c = h0.copy(), compute_c5(h0, n)

    # Phase 1: Binary local search
    p1 = t0 + 0.08 * total
    hb, cb = binary_search(h0, n, p1, 80)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: LSE-SLSQP with interleaved binary search
    p2 = t0 + 0.55 * total
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
    for alpha in alphas:
        if time.time() > p2 - 12:
            break
        mi = 2000 if alpha < 5000 else (3500 if alpha < 500000 else 5000)
        h = lse_slsqp(best_h, n, alpha, mi)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if time.time() < p2 - 12:
            hb2, cb2 = binary_search(h, n, min(time.time() + 3, p2 - 12), 20)
            if cb2 < best_c:
                best_c, best_h = cb2, hb2.copy()

    # Phase 3: Subgradient descent for broader exploration
    p3 = t0 + 0.72 * total
    h2, c2 = subgradient_descend(best_h, n, p3, 5000)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()

    # Phase 4: High-alpha LSE polish
    p4 = t0 + 0.88 * total
    for alpha in [100000000, 300000000, 800000000]:
        if time.time() > p4 - 8:
            break
        h = lse_slsqp(best_h, n, alpha, 5000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Perturbed restarts
    p5 = dl - 12
    x = np.linspace(0, 2, n)
    perts = [
        0.01 * rng.randn(n),
        0.02 * np.sin(2 * np.pi * 3.0 * x + 0.5),
        0.008 * np.cos(2 * np.pi * 8.0 * x),
        0.015 * rng.randn(n),
        0.01 * np.sin(2 * np.pi * 1.5 * x + 1.0),
    ]
    for pt in perts:
        if time.time() > p5 - 10:
            break
        h = project_feasible(best_h + pt, n)
        for a in [5000, 50000, 500000]:
            if time.time() > p5 - 10:
                break
            h = lse_slsqp(h, n, a, 2000)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
