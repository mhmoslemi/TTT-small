    c = np.correlate(h, 1.0 - h, mode='full') * dx  # shape (2n-1,)
    # Smooth max

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_objective(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return (m + logsum / alpha) * dx

def smooth_max_gradient(h, n, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    weights = np.exp(shifted - sm)
    weights /= weights.sum()
    
    grad = np.zeros(n)

    threshold = 1e-12
    active = weights > threshold

        k = k_idx - (n - 1)

            grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n-kp]

import numpy as np

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return (m + sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha * (2.0 / n)

def smooth_max_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k_idx in active:
        wi = w[k_idx]
        k = int(k_idx) - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        else:
            kp = -k
            grad[:n-kp] += wi * (1.0 - h[kp:])
            grad[kp:] -= wi * h[:n-kp]
    return grad * dx

def projected_gradient(h, n, alpha, niter, lr_init):
    dx = 2.0 / n
    f_cur = smooth_max_obj(h, n, alpha)
    lr = lr_init
    for i in range(niter):
        g = smooth_max_obj_grad(h, n, alpha)
        for _ in range(30):

        if f_try < f_cur - 1e-15:
            h = h_try
            f_cur = f_try
        lr = min(lr * 1.1, lr_init)
    return h

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def grad(x):
        return smooth_max_obj_grad(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': cons_fun, 'jac': cons_jac})
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def lp_step(h, n, eps, M=40):
    import cvxpy as cp
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    top_idx = np.argsort(C)[::-1][:M]
    G = np.zeros((n, M))
    for j, k_idx in enumerate(top_idx):
        k = int(k_idx) - (n - 1)
        if k >= 0:
            G[k:, j] = 1.0 - h[:n-k]
            G[:n-k, j] -= h[k:]
        else:
            kp = -k
            G[:n-kp, j] = 1.0 - h[kp:]
            G[kp:, j] -= h[:n-kp]
    C_top = C[top_idx]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C_top + G.T @ delta <= t,
        delta >= -eps, delta <= eps,
        delta >= -h, delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for sol in ['GLOP', 'ECOS', 'GLPK']:
        try:
            prob.solve(solver=sol)
            break
        except Exception:
            continue
    if delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.7, 0.5, 0.3]:

        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

        scale = 0.003 + 0.012 * rng.rand()
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        alpha_r = 1000 + int(5000 * rng.rand())
        h = projected_gradient(h, n, alpha_r, 200, 0.5)
        h = slsqp_smooth(h, n, 10000, 80, 1e-14)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Final polish
    if elapsed() < dl - 3:
        h = best_h.copy()
        h = slsqp_smooth(h, n, 50000, 200, 1e-16)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
