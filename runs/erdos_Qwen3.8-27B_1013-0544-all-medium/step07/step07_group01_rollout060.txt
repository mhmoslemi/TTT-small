def smooth_grad_fast(h, n, alpha):
    """Fast gradient of smooth max using FFT-based convolution."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    max_s = shifted.max()
    exps = np.exp(shifted - max_s)
    p = exps / exps.sum()
    
    s = p[::-1]
    A = np.convolve(1.0 - h, s, mode='full')[n-1::-1]
    B = np.convolve(h, p, mode='full')[n-1:n-1+n]
    
    obj_val = m + (max_s + np.log(exps.sum())) / alpha
    grad = dx * (A - B)
    return obj_val, grad

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    max_s = shifted.max()
    exps = np.exp(shifted - max_s)
    p = exps / exps.sum()
    obj_val = m + (max_s + np.log(exps.sum())) / alpha

    B = np.convolve(h, p, mode='full')[n - 1:n - 1 + n]
    grad = dx * (A - B)
    return obj_val, grad

                if time.time() > p2_end - 10:
                    break
                hf = lbfgs_step(hf, n, alpha, lam, 400, p2_end)
                hf = project_feasible(hf, n)
            c = compute_c5(hf, n)
            if c < best_c:
                best_c, best_h = c, hf.copy()

    # Phase 3: Diverse starts with L-BFGS-B
    p3_end = t0 + 600
    starts = diverse_starts(n, rng)
    for st in starts:
        if time.time() > p3_end - 20:
            break
        hs = st.copy()
        for alpha, lam in [(500, 50), (5000, 500), (50000, 5000)]:
            if time.time() > p3_end - 10:
                break
            hs = lbfgs_step(hs, n, alpha, lam, 400, p3_end)
            hs = project_feasible(hs, n)
        c = compute_c5(hs, n)
        if c < best_c:
            best_c, best_h = c, hs.copy()

    # Phase 4: Projected gradient fine-tuning on best
    p4_end = t0 + 850
    h4 = best_h.copy()
    for alpha in [10000, 100000, 500000]:
        if time.time() > p4_end - 30:
            break
        h4, c4 = projected_grad(h4, n, alpha, min(p4_end - 10, time.time() + 60))
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Very high alpha L-BFGS-B polish
    p5_end = deadline - 20
    h5 = best_h.copy()
    for alpha, lam in [(200000, 50000), (1000000, 200000), (5000000, 500000)]:
        if time.time() > p5_end - 5:
            break
        h5 = lbfgs_step(h5, n, alpha, lam, 1000, p5_end)
        h5 = project_feasible(h5, n)
        c = compute_c5(h5, n)
        if c < best_c:
            best_c, best_h = c, h5.copy()

    # Phase 6: Small noise perturbations
    for trial in range(20):
        if time.time() > deadline - 3:
            break
        noise = 0.005 * (1 + trial * 0.4)
        h6 = best_h + noise * rng.randn(n)
        h6 = project_feasible(h6, n)
        h6 = lbfgs_step(h6, n, 100000, 10000, 500, deadline - 2)
        h6 = project_feasible(h6, n)
        c = compute_c5(h6, n)
        if c < best_c:
            best_c, best_h = c, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
