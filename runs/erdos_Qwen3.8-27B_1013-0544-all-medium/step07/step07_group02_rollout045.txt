import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = build_gradient_matrix(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'CLARABEL', 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_optimize(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def downsample(h, n_from, n_to):
    xo = np.linspace(0, 2, n_from)
    xnew = np.linspace(0, 2, n_to)
    h_interp = np.interp(xnew, xo, h)
    return project_feasible(h_interp, n_to)
