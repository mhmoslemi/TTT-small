def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_func(h, n, alpha):
    """LSE objective value only."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    return m + np.log(np.exp(shifted).sum()) / alpha

def lse_grad_func(h, n, alpha, penalty):
    """LSE objective + penalty, with analytical gradient."""
    dx = 2.0 / n

    shifted = alpha * (c - m)
    shifted -= shifted.max()

    Z = exps.sum()

    v = h
    S1 = np.correlate(u, w, 'full')[n-1 : 2*n-1]
    S2 = np.convolve(w, v, 'full')[n-1 : 2*n-1]
    grad = (S1 - S2) * dx
