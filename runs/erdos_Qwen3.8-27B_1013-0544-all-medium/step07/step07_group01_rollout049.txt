    # Phase 5: Coordinate push (600-750s)
    p5_end = t0 + 750
    h5, c5 = coordinate_push(best_h, n, p5_end)
    if c5 < best_c:
        best_c = c5
        best_h = h5.copy()
    
    # Phase 6: SLSQP after discrete search (750-900s)
    p6_end = t0 + 900
    for a in [10000000, 1000000000]:
        if time.time() > p6_end - 10:
            break
        h6 = slsqp_optimize(best_h, n, a, 3000, 1e-16)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
    
    # Phase 7: Perturbation restarts (900-1050s)
    p7_end = t0 + 1050
    for trial in range(40):
        if time.time() > p7_end - 5:
            break
        noise = 0.001 * (1 + trial * 0.3)
        h7 = best_h + noise * rng.randn(n)
        h7 = project_feasible(h7, n)
        h7 = slsqp_optimize(h7, n, 10000000, 1500, 1e-15)
        c7 = compute_c5(h7, n)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
            # Quick swap after improvement
            if time.time() < p7_end - 20:
                h7b, c7b = swap_local_search(h7, n, p7_end - 10)
                if c7b < best_c:
                    best_c = c7b
                    best_h = h7b.copy()
    
    # Phase 8: Final high-alpha polish (1050-end)
    if time.time() < deadline - 10:
        h8 = slsqp_optimize(best_h, n, 5000000000, 5000, 1e-17)
        c8 = compute_c5(h8, n)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
