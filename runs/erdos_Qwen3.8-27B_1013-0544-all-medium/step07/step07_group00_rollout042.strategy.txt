def smooth_obj(x, n, alpha, dx=2.0/n):
    c = np.correlate(x, 1.0-x, mode='full')
    center = n-1
    c_centered = c[center:]  # Drop out-of-bound lags
    m = c_centered.max()
    shifted = alpha*(c_centered - m)
    mx = shifted.max()
    if abs(mx) < 1e-12:
        lse = 0.0
    else:
        unexp = np.log(np.sum(np.exp(shifted-mx)))
        lse = mx + unexp
    return (m + lse/alpha) * dx
