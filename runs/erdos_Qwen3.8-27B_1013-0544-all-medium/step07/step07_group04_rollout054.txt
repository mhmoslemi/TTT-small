    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):

        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return m
    return m + np.log(s) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:
        q = e / s
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = int(idx) - off
        w = float(q[idx])
        if k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            s2 = -k
            g[:n-s2] += w * (1.0 - x[s2:])
            g[s2:] -= w * x[:n-s2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def lbfgsb_round(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    def obj(x):
        return lse_obj(x, n, alpha) + 0.5 * mu * (x.sum() - 0.5*n)**2
    def grad(x):
        return lse_grad(x, n, alpha) + mu * (x.sum() - 0.5*n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, jac=grad, method="L-BFGS-B", bounds=bounds,
                       options={"maxiter": maxiter, "ftol": 1e-15, "gtol": 1e-9})
        return res.x if res.x is not None else h
    except Exception:
        return h

def pair_optimize(h, i, j, n, base_c_max):
    """Find best x for h[i]=x, h[j]=S-x. Returns (x_best, new_max_c) or None."""
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-12:
        return None
    
    a = h[i]
    b = h[j]
    
    def f(x):
        h[i] = x
        h[j] = S - x
        c = compute_c(h, n)
        h[i] = a
        h[j] = b
        return float(c.max())
    
    # Quick check: is the current point already optimal?
    # Try a few points first
    best_x = a
    best_val = base_c_max

    try:

                             options={'xatol': 1e-8, 'maxiter': 60})

            best_val = float(res.fun)

Let me also think about the **concurrent tangents method** (CTM) for minimizing the maximum of convex functions. This is a specialized algorithm for min-max problems. The idea is to maintain a set of "active" lags and solve a subproblem for each active lag, then combine the solutions.

For our problem: minimize max_k c_k(h) subject to 0≤h≤1, sum(h)=n/2.

The CTM would:

2. For each k in A, solve: minimize c_k(h) subject to constraints. This is a QP (quadratic objective, linear constraints).
3. Average the solutions and update A.
4. Repeat.

But solving QPs for each active lag is expensive. And there might be many active lags.
