        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def compute_base_binary(h, n):
    l = np.arange(-(n - 1), n, dtype=int)
    base = np.zeros((n, 2 * n - 1), dtype=float)
    for p in range(n):
        f = p + l
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1.0 - h[f[mf]]
        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

def greedy_search(h, n, deadline, max_sweeps=300, chunk=2048):
    if n % 2 != 0:
        return h, compute_c5(h, n)
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(corr.max()))
    besth, bestM = h.copy(), M
    for _ in range(max_sweeps):
        if time.time() > deadline: break
        base = compute_base_binary(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0: break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline: break
            end = min(start + chunk, zi.size)
            i, j = zi[start:end], oj[start:end]
            arr = base[i] - base[j]
            arr += corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            arr[:, d1] += 1.0
            d2 = n - 1 + (i - j)
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv, jv = int(i[loc]), int(j[loc])
                h[iv], h[jv] = 1.0, 0.0
                corr = arr[loc].copy()
                M = int(round(corr.max()))
                if M < bestM:
                    bestM, besth = M, h.copy()
                improved = True
                break
        if not improved:
            break
    return besth, bestM * (2.0 / float(n))

def sa_search(h, n, deadline, max_iter=30000, t0=1.0, t1=0.01):
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(corr.max()))
    besth, bestM = h.copy(), M
    base = compute_base_binary(h, n)
    ones = np.nonzero(h == 1)[0]
    zeros = np.nonzero(h == 0)[0]
    n_iter = max(max_iter, 1)
    for it in range(n_iter):
        if time.time() > deadline: break
        temp = t0 * (t1 / t0) ** (it / n_iter)
        oi = int(ones[np.random.randint(len(ones))])
        zi = int(zeros[np.random.randint(len(zeros))])
        vec = corr + base[zi] - base[oi]
        vec[n - 1] -= 2.0
        d1 = n - 1 + (oi - zi)
        if 0 <= d1 < 2 * n - 1: vec[d1] += 1.0
        d2 = n - 1 + (zi - oi)
        if 0 <= d2 < 2 * n - 1: vec[d2] += 1.0
        newM = int(round(vec.max()))
        delta = newM - M
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-10)):
            h[zi], h[oi] = 1.0, 0.0
            corr = vec.copy()
            M = newM
            ones[oi] = oi  # remove old one
            ones[zi] = zi  # add new one (approximate bookkeeping)
            ones = np.nonzero(h == 1)[0]
            zeros = np.nonzero(h == 0)[0]
            if M < bestM:
                bestM, besth = M, h.copy()
            if M == 51:
                break
    return besth, bestM * (2.0 / float(n))

def lbfgs_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if s < 1e-300:
        return 1e100, np.zeros_like(h)
    p = e / s
    F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    dev = float(h.sum() - 0.5 * n)
    F += 0.5 * 1e4 * dev * dev
    g = g + 1e4 * dev
    return float(F), g

def polish(h, n, deadline):
    h = project_feasible(h, n)
    best, bc = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    _dt = deadline + 30
    def cb(*a): return time.time() > _dt
    for beta in [50.0, 200.0, 800.0, 3000.0]:
        if time.time() > deadline - 5: break
        _dt = min(deadline - 3, time.time() + 40)
        try:
            res = minimize(lbfgs_obj, best, method="L-BFGS-B", jac=True,
                           bounds=bounds, args=(n, beta), callback=cb,
                           options={"maxiter": 500000, "maxls": 30, "ftol": 1e-12, "gtol": 1e-8})
            x = res.x if hasattr(res, "x") else best
        except Exception:
            x = best
        h2 = project_feasible(x, n)
        c2 = compute_c5(h2, n)
        if c2 < bc:
            bc, best = c2, h2.copy()
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    end = t0 + max(30.0, total - 15.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad")
    except Exception:
        n0 = 400
        h0 = ((np.arange(n0) // 50) % 2).astype(float)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Thue-Morse starting point
    tm = thue_morse(n0)
    tm_bin = top_n_binary(tm, n0)
    c_tm = compute_c5(tm_bin, n0)
    if c_tm < best_c:
        best_c, best_h, best_n = c_tm, tm_bin, n0

    # Phase 1: Intensive binary greedy from multiple starts
    starts = [best_h, tm_bin]
    for s in range(4):
        rb = rng.randint(0, 2, n0).astype(float)
        rb = top_n_binary(rb, n0)
        starts.append(rb)

    phase1_end = min(end - 300, t0 + 0.3 * (end - t0))
    for st in starts:
        if time.time() > phase1_end - 20: break
        hb, cb = greedy_search(st, n0, phase1_end, max_sweeps=200, chunk=2048)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 2: SA to escape local optima
    phase2_end = min(end - 200, t0 + 0.6 * (end - t0))
    for attempt in range(3):
        if time.time() > phase2_end - 30: break
        hb, cb = sa_search(best_h, n0, phase2_end - 30 * (3 - attempt - 1),
                           max_iter=20000, t0=2.0, t1=0.05)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
        # Follow up with greedy
        if time.time() < phase2_end - 10:
            hb2, cb2 = greedy_search(best_h, n0, phase2_end, max_sweeps=100, chunk=2048)
            if cb2 < best_c:
                best_c, best_h = cb2, hb2.copy()

    # Phase 3: L-BFGS refinement
    phase3_end = min(end - 60, t0 + 0.85 * (end - t0))
    if time.time() < phase3_end:
        hh, cc = polish(best_h, n0, phase3_end)
        if cc < best_c:
            best_c, best_h = cc, hh.copy()

    # Phase 4: Try other n values
    if time.time() < end - 40:
        for nnew in [402, 398, 404]:
            if time.time() > end - 25: break
            x1 = np.linspace(0.0, 2.0, nnew)
            x0 = np.linspace(0.0, 2.0, n0)
            hr = np.interp(x1, x0, best_h)
            hr = project_feasible(hr, nnew)
            # Binary at new n
            if nnew % 2 == 0:
                hr_bin = top_n_binary(hr, nnew)
                hb, cb = greedy_search(hr_bin, nnew, end - 20, max_sweeps=80, chunk=1024)
                if cb < best_c:
                    best_c, best_h, best_n = cb, hb, nnew
            # Continuous refinement
            if time.time() < end - 15:
                hh, cc = polish(hr, nnew, end - 10)
                if cc < best_c:
                    best_c, best_h, best_n = cc, hh, nnew

    # Final: try binary top-n of current best
    if best_n % 2 == 0:
        tb = top_n_binary(best_h, best_n)
        ct = compute_c5(tb, best_n)
        if ct < best_c:
            best_c, best_h = ct, tb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
