            best_mc = mc
            best_h = h_new.copy()
            best_c5 = mc * 2.0 / n
            h = h_new

    # Phase 2: Active-set LP with adaptive step (30% budget)
    p2_dl = t0 + 0.65 * total
    h = best_h.copy()
    eps = 0.04
    no_improve = 0
    while time.time() < p2_dl - 5:
        h, mc = active_lp_step(h, n, eps, top_k=50)
        if mc < best_mc - 1e-13:
            best_mc = mc
            best_h = h.copy()
            best_c5 = mc * 2.0 / n
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                eps *= 0.6
                if eps < 0.0003:
                    break
                no_improve = 0
        h = project_feasible(h, n)

    # Phase 3: SLSQP + FISTA alternation (15% budget)
    p3_dl = t0 + 0.80 * total
    h = best_h.copy()
    for alpha in [50000, 200000, 1000000]:
        if time.time() > p3_dl - 10:
            break
        h = slsqp_smooth(h, n, alpha, maxiter=2000)
        mc = float(get_corr(h).max())
        if mc < best_mc - 1e-13:
            best_mc = mc
            best_h = h.copy()
            best_c5 = mc * 2.0 / n
    if time.time() < p3_dl - 5:
        h = fista_smooth(best_h, n, 500000, niter=1000)
        mc = float(get_corr(h).max())
        if mc < best_mc - 1e-13:
            best_mc = mc
            best_h = h.copy()
            best_c5 = mc * 2.0 / n

    # Phase 4: Noise restarts with LP (10% budget)
    p4_dl = t0 + 0.90 * total
    for _ in range(15):
        if time.time() > p4_dl - 5:
            break
        h = best_h + 0.008 * rng.randn(n)
        h = project_feasible(h, n)
        eps_r = 0.02
        for _ in range(80):
            if time.time() > p4_dl - 5:
                break
            h, mc = active_lp_step(h, n, eps_r, top_k=50)
            if mc < best_mc - 1e-13:
                best_mc = mc
                best_h = h.copy()
                best_c5 = mc * 2.0 / n
            h = project_feasible(h, n)
            eps_r = max(eps_r * 0.99, 0.001)

    # Phase 5: Multi-resolution (5% budget)
    p5_dl = dl - 5
    for nh in [300, 500, 600]:
        if time.time() > p5_dl - 2:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        mc_h = float(get_corr(hh).max())
        c5_h = mc_h * 2.0 / nh
        if c5_h >= best_c5:
            continue
        # Optimize on this grid
        h2 = slsqp_smooth(hh, nh, 50000, maxiter=1000)
        mc2 = float(get_corr(h2).max())
        c5_2 = mc2 * 2.0 / nh
        eps_r = 0.02
        for _ in range(100):
            if time.time() > p5_dl - 2:
                break
            h2, mc_new = active_lp_step(h2, nh, eps_r, top_k=40)
            c5_new = mc_new * 2.0 / nh
            if c5_new < c5_2:
                c5_2 = c5_new
            h2 = project_feasible(h2, nh)
            eps_r = max(eps_r * 0.99, 0.001)
        if c5_2 < best_c5:
            best_mc = mc2 if c5_2 == mc2 * 2.0 / nh else float(get_corr(h2).max())
            best_h = h2.copy()
            best_c5 = c5_2
            n = nh

    # Phase 6: Final fine polish
    if time.time() < dl - 2:
        h = best_h.copy()
        h = slsqp_smooth(h, n, 1000000, maxiter=2000)
        mc = float(get_corr(h).max())
        if mc < best_mc - 1e-13:
            best_mc = mc
            best_h = h.copy()
            best_c5 = mc * 2.0 / n
        eps_f = 0.002
        for _ in range(100):
            if time.time() > dl - 2:
                break
            h, mc = active_lp_step(best_h, n, eps_f, top_k=60)
            c5_new = mc * 2.0 / n
            if c5_new < best_c5 - 1e-13:
                best_mc = mc
                best_h = h.copy()
                best_c5 = c5_new
            eps_f = max(eps_f * 0.995, 0.0001)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
