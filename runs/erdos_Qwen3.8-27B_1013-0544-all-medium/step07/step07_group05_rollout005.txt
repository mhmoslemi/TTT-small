import time

import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'target': 0.5, 'beta': 1.0, 'rho': 1.0, 'lam': 0.0}

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w

def topk_weights(c, k):
    """Hard top-k weights for active set gradient."""
    n = len(c)
    if k >= n:
        return np.ones(n) / n
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(n)
    w[idx] = 1.0 / k
    return w

def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def l_fg(x):

    n = int(st['n'])
    dx = float(st['dx'])
    target = float(st['target'])

    rho = float(st['rho'])
    lam = float(st['lam'])

    return float(f), g

    """Direct projected subgradient with momentum on the max."""
    dx = 2.0 / n
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    v = np.zeros(n)
    beta_mom = 0.7
    cur_c = erdos_c5(h, n)
    step_scale = 1.0
    for it in range(max_iter):
        if time.time() > deadline:
            break

        cmax = float(c.max())
        # Active set: lags within 1% of max
        threshold = cmax - 0.01 * cmax
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = smooth_grad_from_w(h, w, n, dx)
        gnorm = float(np.sqrt(np.sum(grad * grad)))
        if gnorm < 1e-14:
            break
        # Adaptive step: decrease if oscillating
        alpha = step_scale / (gnorm + 1e-10)
        alpha = min(alpha, 0.5)
        v = beta_mom * v + (1.0 - beta_mom) * grad
        vnorm = float(np.sqrt(np.sum(v * v)))
        if vnorm > 1e-14:
            v = v / vnorm * gnorm
        h_new = h - alpha * v

        c_new = erdos_c5(h_new, n)

            cur_c = c_new
            step_scale = min(2.0, step_scale * 1.05)
        else:
            step_scale = max(0.1, step_scale * 0.7)
        # Decay momentum
        if it > max_iter // 3:
            beta_mom = max(0.3, beta_mom * 0.995)
    return h, erdos_c5(h, n)

    """Generate a {0,1} sequence via golden ratio low-discrepancy."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0

def block_binary(n, period, seed=0):
    """Block pattern with 50% duty cycle."""
    h = np.zeros(n)
    for i in range(n):

def pair_obj(delta, h, i, j, n):

        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
