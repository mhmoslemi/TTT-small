def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def make_comp(h, n):
    lags = np.arange(-(n - 1), n)
    C = np.zeros((n, 2 * n - 1))
    for pos in range(n):
        p1 = pos + lags
        p2 = pos - lags
        m1 = (p1 >= 0) & (p1 < n)
        C[pos, m1] = (1.0 - h[p1[m1]])
        m2 = (p2 >= 0) & (p2 < n)
        C[pos, m2] -= h[p2[m2]]
        C[pos, n - 1] = 0.0
    return C

def binary_search_fast(h_init, n, deadline, max_iters=200):

    best_h, best_c = h.copy(), cur_c
    zeros = np.where(h == 0)[0].copy()
    ones = np.where(h == 1)[0].copy()
    C = make_comp(h, n)

            fi = scores.argmin()
            bi = fi // n_ones
            bj = fi % n_ones

        if best_i < 0:

        h[best_i], h[best_j] = 1.0, 0.0
        corr += C[best_i] - C[best_j]
        if best_score < best_c:
            best_c = best_score
            best_h = h.copy()
        zl = zeros.tolist(); zl.remove(best_i); zeros = np.array(zl)
        ol = ones.tolist(); ol.remove(best_j); ones = np.array(ol)
        ones = np.append(ones, best_i)
        zeros = np.append(zeros, best_j)
        C = make_comp(h, n)
    return best_h, best_c

def lse_fun(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = c.max()
    a = alpha * (c - M)
    ea = np.exp(a - a.max())
    lse_val = M + np.log(ea.sum()) / alpha
    s = x.sum()
    return lse_val + 0.5 * mu * (s - n / 2.0) ** 2

def lse_jac(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = c.max()
    a = alpha * (c - M)
    ea = np.exp(a - a.max())
    w = ea / ea.sum()
    s = x.sum()
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    for m in range(2 * n - 1):
        wi = w[m]
        if wi < 1e-20:
            continue
        lag = lags[m]
        j1 = max(0, -lag)
        e1 = min(n, n - lag)
        if j1 < e1:
            js = np.arange(j1, e1)
            grad[js] += wi * (1.0 - x[js + lag])
        j2 = max(0, lag)
        e2 = min(n, n + lag)
        if j2 < e2:
            js2 = np.arange(j2, e2)
            grad[js2] -= wi * x[js2 - lag]
    grad *= dx
    grad += mu * (s - n / 2.0)
    return grad

def lse_lbfgsb(h, n, alpha, mu, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_fun, h, args=(n, alpha, mu), method='L-BFGS-B',
                       jac=lambda x: lse_jac(x, n, alpha, mu),
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h
