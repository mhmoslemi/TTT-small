def vectorized_binary_swap(h, n, max_sweeps=100, deadline=None):
    dx = 2.0 / n
    h = h.copy()
    # Make binary
    idx = np.argsort(h)[::-1]
    h[idx[:n//2]] = 1.0
    h[idx[n//2:]] = 0.0
    
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        new_zeros = []
        for i in zeros:
            if deadline and time.time() > deadline:
                break
            # Compute delta for all j in ones, all lags
            # term1: contribution from i becoming 1
            # For lag l (relative), i+l must be in [0,n)
            # corr index = l + (n-1)
            
            # Let's work with the full correlation array of size 2n-1
            # corr[m] corresponds to lag l = m - (n-1)
            
            # delta at correlation index m (lag l = m-(n-1)):
            # +1 if h[i+l]==0 and 0<=i+l<n  [i contributes new 1-at-i, 0-at-i+l]
            # -1 if h[j+l]==0 and 0<=j+l<n  [j no longer contributes]
            # -1 if h[i-l]==1 and 0<=i-l<n  [i-l no longer pairs with i]
            # +1 if h[j-l]==1 and 0<=j-l<n  [j-l now pairs with j]
            
            n1 = 2*n - 1
            lags = np.arange(-(n-1), n)  # shape (2n-1,)
            
            # For position i:
            i_plus = i + lags  # position i+l for each lag
            i_minus = i - lags  # position i-l for each lag
            
            valid_ip = (i_plus >= 0) & (i_plus < n)
            valid_im = (i_minus >= 0) & (i_minus < n)
            
            term1 = np.zeros(n1)
            term1[valid_ip] = (h[i_plus[valid_ip]] == 0).astype(float)
            
            term3 = np.zeros(n1)
            term3[valid_im] = (h[i_minus[valid_im]] == 1).astype(float)
            
            # For all j in ones:
            # j_plus = j + lags, shape (len(ones), 2n-1)
            j_plus = ones[:, None] + lags[None, :]
            j_minus = ones[:, None] - lags[None, :]
            
            valid_jp = (j_plus >= 0) & (j_plus < n)
            valid_jm = (j_minus >= 0) & (j_minus < n)
            
            h_jp = np.where(valid_jp, h[np.clip(j_plus, 0, n-1)], 0)
            h_jm = np.where(valid_jm, h[np.clip(j_minus, 0, n-1)], 0)
            
            term2 = (h_jp == 0).astype(float) * valid_jp
            term4 = (h_jm == 1).astype(float) * valid_jm
            
            # delta[j, m] = term1[m] - term2[j, m] - term3[m] + term4[j, m]
            delta = term1[None, :] - term2 - term3[None, :] + term4
            
            new_corr = corr[None, :] + delta
            max_new = new_corr.max(axis=1)
            
            best_j_idx = np.argmin(max_new)
            best_new_max = max_new[best_j_idx]
            
            if best_new_max < cur_max - 1e-15:
                j = ones[best_j_idx]
                # Apply the swap
                h[i] = 1.0
                h[j] = 0.0
                corr = corr + delta[best_j_idx]
                cur_max = best_new_max
                # Update ones and zeros
                ones = np.delete(ones, best_j_idx)
                zeros = np.delete(zeros, np.where(zeros == i)[0][0])
                zeros = np.append(zeros, j)
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_max * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:

            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]

                       options={'maxiter': maxiter, 'ftol': ftol})

def binary_search(h_init, n, deadline, use_sa=False, T0=0.5, cooling=0.97):
    """Fast vectorized binary local search with optional simulated annealing."""
    dx = 2.0 / n
    h = h_init.copy()
    if int((h > 0.5).sum()) != n // 2:
        idx = np.argsort(h)[::-1]
        h = np.zeros(n)
        h[idx[:n // 2]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    lags = np.arange(-(n-1), n)
    n1 = 2 * n - 1
    T = T0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(int(time.time() * 1000) % 100000)
        rng.shuffle(zeros)
        improved = False
        
        for i in zeros:
            if time.time() > deadline:
                break
            if h[i] != 0:
                continue
            if len(ones) == 0:
                break
            
            # Compute delta for all j in ones
            ip = i + lags
            im = i - lags
            vip = (ip >= 0) & (ip < n)
            vim = (im >= 0) & (im < n)
            
            t1 = np.zeros(n1)
            t1[vip] = (h[ip[vip]] == 0).astype(float)
            t3 = np.zeros(n1)
            t3[vim] = (h[im[vim]] == 1).astype(float)
            
            jp = ones[:, None] + lags[None, :]
            jm = ones[:, None] - lags[None, :]
            vjp = (jp >= 0) & (jp < n)
            vjm = (jm >= 0) & (jm < n)
            
            hjp = np.where(vjp, h[np.clip(jp, 0, n-1)], 1.0)
            hjm = np.where(vjm, h[np.clip(jm, 0, n-1)], 0.0)
            
            t2 = (hjp == 0).astype(float) * vjp
            t4 = (hjm == 1).astype(float) * vjm
            
            delta = t1[None, :] - t2 - t3[None, :] + t4
            new_max = (corr[None, :] + delta).max(axis=1)
            
            bi = np.argmin(new_max)
            bm = float(new_max[bi])
            
            # Decide whether to accept
            if use_sa and bm >= cur_max:
                if np.random.random() > np.exp(-(bm - cur_max) / max(T, 0.01)):
                    continue
            else:
                if bm >= cur_max - 1e-10:
                    continue
            
            j = int(ones[bi])
            h[i] = 1.0
            h[j] = 0.0
            corr += delta[bi]
            cur_max = bm
            improved = True
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            
            # Update ones (remove j)
            ones = np.delete(ones, bi)
            # i is no longer zero, j is now zero
        
        T *= cooling
        if not improved and not use_sa:
            break
    
    return best_h, best_max * dx

def subgrad_optimize(h, n, deadline, max_iters=50000, top_k=3):
    """Subgradient with RMS adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    n1 = 2 * n - 1
    G = np.zeros(n)
    momentum = np.zeros(n)
    beta = 0.9
    t = 0
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 8000:
            rng = np.random.RandomState(t)
            h = best_h + 0.02 * rng.randn(n)
            h = project_feasible(h, n)
            G = np.zeros(n)
            momentum = np.zeros(n)
            stall = 0
        
        # Subgradient: use top_k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] = 1.0 - 2.0 * h
                else:
                    grad[k:] += (1.0 - h[:n-k])
                    grad[:n-k] -= h[k:]
            else:
                kk = -k
                if kk == 0:
                    grad[:] = 1.0 - 2.0 * h
                else:
                    grad[:n-kk] += (1.0 - h[kk:])
                    grad[kk:] -= h[:n-kk]
        grad = dx * grad / top_k
        
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum**2
        lr = 0.1 / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_check.max() < cur_c + 0.01:
            h = h_new
    
    return best_h, best_c
