def top_idx_weights(c,K,alpha):
    m=c.size

       order=np.argsort(c)[::-1]

import numpy as np
import time

def project_box_sum(x, target):
    x = np.asarray(x, float)
    if abs(x.sum() - target) < 1e-14:
        return np.clip(x, 0.0, 1.0)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        if np.clip(x - tau, 0.0, 1.0).sum() > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        d = target - y.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            i = int(np.argmin(y))
            y[i] = min(1.0, y[i] + d)
        else:
            i = int(np.argmax(y))
            y[i] = max(0.0, y[i] + d)
    return y

def corr_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def top_indices(c, K):
    K = max(1, min(int(K), c.size))
    return np.argsort(c)[::-1][:K]

def make_gradient(h, idx):
    n = len(h)
    g = np.zeros(n)
    scale = (2.0 / n) / max(1, len(idx))
    for k in idx:
        s = n - 1 - k
        if s >= 0:
            l = s
            g[:n - l] += 1.0 - h[l:]
            g[l:] -= h[:n - l]
        else:
            l = -s
            g[l:] += 1.0 - h[:n - l]
            g[:n - l] -= h[l:]
    return g * scale

def free_descent(h, deadline, rng, K=6):
    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    c = corr_values(h)
    best_c = float(c.max())
    best_h = h.copy()
    mom = np.zeros(n)
    step = 0.12
    last = time.time()
    while time.time() < deadline:
        c = corr_values(h)
        cur = float(c.max())
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
            last = time.time()
        g = make_gradient(h, top_indices(c, K))
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        mom = 0.85 * mom - g / gn
        mn = float(np.linalg.norm(mom))
        if mn > 2.0:
            mom *= 2.0 / mn
        ok = False
        for sc in (1.0, 0.6, 0.35, 0.2):
            y = project_box_sum(h + sc * step * mom, target)
            cy = float(corr_values(y).max())
            if cy < cur - 1e-13 or cy < best_c - 1e-13:
                h = y
                if cy < best_c:
                    best_c = cy
                    best_h = y.copy()
                    last = time.time()
                ok = True
                break
        if ok:

            mom *= 0.5
            if time.time() - last > 20:
                h = project_box_sum(best_h + 0.02 * rng.randn(n), target)
                mom = np.zeros(n)
                step = 0.06
                last = time.time()
    return best_h, float(best_c)

def anti_from_h(h):
    n = len(h)
    m = n // 2
    left = h[:m]
    right = h[n - 1:n - m - 1:-1]
    return np.clip(0.5 * (left - right), -0.5, 0.5)

def from_anti(a, n):
    m = n // 2
    h = np.empty(n, float)
    h[:m] = 0.5 + a
    h[m:] = 0.5 - a[::-1]
    return h

def anti_descent(a, n, deadline, rng, K=8):
    m = n // 2
    a = np.clip(np.asarray(a, float), -0.5, 0.5)
    h = from_anti(a, n)
    c = corr_values(h)
    best_c = float(c.max())
    best_a = a.copy()
    best_h = h.copy()
    cur = a.copy()
    cur_h = h.copy()
    mom = np.zeros(m)
    step = 0.12
    last = time.time()
    while time.time() < deadline:
        c = corr_values(cur_h)
        curc = float(c.max())
        if curc < best_c:
            best_c = curc
            best_a = cur.copy()
            best_h = cur_h.copy()
            last = time.time()
        gh = make_gradient(cur_h, top_indices(c, K))
        ga = gh[:m] - gh[n - 1:n - m - 1:-1]
        gn = float(np.linalg.norm(ga))
        if gn < 1e-14:
            break
        mom = 0.85 * mom - ga / gn
        mn = float(np.linalg.norm(mom))
        if mn > 2.0:
            mom *= 2.0 / mn
        ok = False
        for sc in (1.0, 0.6, 0.35, 0.2):
            at = np.clip(cur + sc * step * mom, -0.5, 0.5)
            ht = from_anti(at, n)
            ct = float(corr_values(ht).max())
            if ct < curc - 1e-13 or ct < best_c - 1e-13:
                cur = at
                cur_h = ht
                if ct < best_c:
                    best_c = ct
                    best_a = at.copy()
                    best_h = ht.copy()
                    last = time.time()
                ok = True
                break
        if ok:
            step = min(step * 1.04, 0.35)
        else:
            step *= 0.65
            mom *= 0.5
            if time.time() - last > 20:
                cur = np.clip(best_a + 0.02 * rng.randn(m), -0.5, 0.5)
                cur_h = from_anti(cur, n)
                mom = np.zeros(m)
                step = 0.06
                last = time.time()
    return best_h, float(best_c)

def anti_flip_polish(a, n, deadline, rng):
    m = n // 2
    a = np.clip(np.asarray(a, float), -0.5, 0.5)
    best_a = a.copy()
    best_h = from_anti(a, n)
    best_c = float(corr_values(best_h).max())
    cur = a.copy()
    cur_c = best_c
    now = time.time()
    det_dead = now + 0.4 * (deadline - now) if deadline - now > 15 else now
    while time.time() < det_dead:
        imp = False
        for i in rng.permutation(m):
            if time.time() > det_dead:
                break
            o = cur[i]
            cur[i] = -o if abs(o) > 0.2 else (0.5 if rng.rand() < 0.5 else -0.5)
            c = float(corr_values(from_anti(cur, n)).max())
            if c < cur_c - 1e-14:
                cur_c = c
                imp = True
            else:
                cur[i] = o
            if c < best_c - 1e-14:
                best_c = c
                best_a = cur.copy()
                best_h = from_anti(cur, n)
        if not imp:
            break
    if time.time() >= deadline:
        return best_h, float(best_c)
    temp = 0.012
    stuck = 0
    while time.time() < deadline:
        i = int(rng.randint(m))
        o = cur[i]
        if abs(o) < 0.2:
            cur[i] = 0.5 if rng.rand() < 0.5 else -0.5
        elif rng.rand() < 0.7:
            cur[i] = -o
        else:
            cur[i] = 0.5 if o < 0 else -0.5
        c = float(corr_values(from_anti(cur, n)).max())
        accept = c < cur_c or (c - cur_c < 0.05 and rng.rand() < np.exp(-(c - cur_c) / max(temp, 1e-6)))
        if accept:
            cur_c = c
            if c < best_c - 1e-14:
                best_c = c
                best_a = cur.copy()
                best_h = from_anti(cur, n)
                stuck = 0
            else:
                stuck += 1
        else:
            cur[i] = o
            stuck += 1
        temp *= 0.9995
        if stuck > 4000:
            cur = best_a.copy()
            cur_c = best_c
            stuck = 0
            temp *= 0.7
    return best_h, float(best_c)

def anti_coord_polish(a, n, deadline, rng, step=0.05, grid=7):
    m = n // 2
    best_a = np.clip(np.asarray(a, float), -0.5, 0.5)
    best_c = float(corr_values(from_anti(best_a, n)).max())
    while time.time() < deadline and step > 1e-5:
        imp = False
        for i in rng.permutation(m):
            if time.time() > deadline:
                break
            old = best_a[i]
            lo = max(-0.5, old - step)
            hi = min(0.5, old + step)
            if hi - lo < 1e-8:
                continue
            changed = False
            for v in np.linspace(lo, hi, grid):
                if abs(v - old) < 1e-8:
                    continue
                best_a[i] = float(v)
                c = float(corr_values(from_anti(best_a, n)).max())
                if c < best_c - 1e-13:
                    best_c = c
                    imp = True
                    changed = True
                    break
            if not changed:
                best_a[i] = old
        if not imp:
            step *= 0.6
    return from_anti(best_a, n), float(best_c)

def polish_anti(a, n, seconds, rng, K=8, grad_frac=0.5, flip_frac=0.3, coord_frac=0.2):
    start = time.time()
    seconds = max(0.0, float(seconds))
    end = start + seconds
    h, c = anti_descent(a, n, start + seconds * grad_frac, rng, K)
    if flip_frac > 0 and time.time() < start + seconds * (1.0 - coord_frac) - 1:
        h2, c2 = anti_flip_polish(anti_from_h(h), n, start + seconds * (1.0 - coord_frac), rng)
        if c2 < c:
            h, c = h2, c2
    if coord_frac > 0 and time.time() < end - 1:
        h3, c3 = anti_coord_polish(anti_from_h(h), n, end, rng)
        if c3 < c:
            h, c = h3, c3
    return project_box_sum(h, 0.5 * n), float(c)

def resample_h(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, m, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.clip(np.interp(x1, x0, h), 0.0, 1.0)

def anti_sine_start(n, p, amp=0.5):
    i = np.arange(n, dtype=float)
    c = (n - 1) / 2.0
    val = amp * np.sin(2.0 * np.pi * p * (i - c) / (n - 1))
    m = n // 2
    a = 0.5 * (val[:m] - val[n - 1:n - m - 1:-1])
    return np.clip(a, -0.5, 0.5)

def better_candidate(h, c, best_h, best_c, best_n):
    c = float(c)
    if c < best_c:
        return np.array(h, float, copy=True), c, len(h)
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(20, int(budget_s)) - 6
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, float).copy()
        if h0.size < 8:
            raise ValueError
    except Exception:
        h0 = from_anti(anti_sine_start(256, 7), 256)
    n0 = len(h0)
    h0 = project_box_sum(h0, 0.5 * n0)
    best_h = h0.copy()
    best_c = float(corr_values(h0).max())
    best_n = n0

    wn = best_n if best_n % 2 == 0 and 48 <= best_n <= 700 else 256
    sh = project_box_sum(resample_h(best_h, wn), 0.5 * wn)

    rem = deadline - time.time()
    if rem > 40:
        hf, cf = free_descent(sh, time.time() + min(rem * 0.18, 80), rng, K=6)
        best_h, best_c, best_n = better_candidate(hf, cf, best_h, best_c, best_n)
        sh = project_box_sum(resample_h(best_h, wn), 0.5 * wn)

    rem = deadline - time.time()
    if rem > 60 and wn % 2 == 0:
        h, c = polish_anti(anti_from_h(sh), wn, min(rem * 0.35, 180), rng, K=10)
        best_h, best_c, best_n = better_candidate(h, c, best_h, best_c, best_n)

    for n in (32, 48, 64, 96, 128, 160, 200, 256, 320, 384):
        rem = deadline - time.time()
        if rem < 45 or n == best_n or n > 600:
            continue
        secs = 40 if n < 128 else (30 if n < 256 else 25)
        secs = min(secs, max(15, rem * 0.16))
        hs = project_box_sum(resample_h(best_h, n), 0.5 * n)
        h, c = polish_anti(anti_from_h(hs), n, secs, rng, K=8)
        best_h, best_c, best_n = better_candidate(h, c, best_h, best_c, best_n)
        rem = deadline - time.time()
        if rem > 25 and n <= 192:
            hf, cf = free_descent(hs, time.time() + min(rem * 0.2, 30), rng, K=6)
            best_h, best_c, best_n = better_candidate(hf, cf, best_h, best_c, best_n)

    for n in (128, 256, 400):
        rem = deadline - time.time()
        if rem < 70 or n <= best_n:
            continue
        secs = min(50, max(15, rem * 0.2))
        hs = project_box_sum(resample_h(best_h, n), 0.5 * n)
        h, c = polish_anti(anti_from_h(hs), n, secs, rng, K=10)
        best_h, best_c, best_n = better_candidate(h, c, best_h, best_c, best_n)

    rem = deadline - time.time()
    if rem > 120:
        bn = best_n if best_n % 2 == 0 and 48 <= best_n <= 512 else 256
        a0 = anti_from_h(project_box_sum(resample_h(best_h, bn), 0.5 * bn))
        for lev in (0.01, 0.02, 0.05, 0.08):
            rem = deadline - time.time()
            if rem < 50:
                break
            secs = 30 if bn < 250 else 20
            secs = min(secs, max(12, rem * 0.18))
            a = np.clip(a0 + lev * rng.randn(len(a0)), -0.5, 0.5)
            h, c = polish_anti(a, bn, secs, rng, K=10)
            best_h, best_c, best_n = better_candidate(h, c, best_h, best_c, best_n)

    for n in (96, 160, 256):
        rem = deadline - time.time()
        if rem < 90:
            break
        for p in (1, 3, 5):
            rem = deadline - time.time()
            if rem < 50:
                break
            secs = 25 if n < 200 else 18
            secs = min(secs, max(10, rem * 0.15))
            h, c = polish_anti(anti_sine_start(n, p), n, secs, rng, K=8)
            best_h, best_c, best_n = better_candidate(h, c, best_h, best_c, best_n)

    rem = deadline - time.time()
    if rem > 50:
        fn = best_n if best_n % 2 == 0 and 48 <= best_n <= 700 else 256
        fh = project_box_sum(resample_h(best_h, fn), 0.5 * fn)
        secs = min(rem - 15, 100)
        h, c = polish_anti(anti_from_h(fh), fn, secs, rng, K=14,
                           grad_frac=0.4, flip_frac=0.35, coord_frac=0.25)
        best_h, best_c, best_n = better_candidate(h, c, best_h, best_c, best_n)
        rem = deadline - time.time()
        if rem > 35:
            hf, cf = free_descent(best_h, time.time() + min(rem - 10, 45), rng, K=10)
            best_h, best_c, best_n = better_candidate(hf, cf, best_h, best_c, best_n)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = float(corr_values(best_h).max())
    return best_h, best_c, best_n
