       ew = np.exp(z)

       maxc = np.max(c)

       c = np.correlate(h, 1.0 - h, mode='full') * dx
       w = softmax_weights(c, beta)
       # Gradient: for each position j, dC/dh[j]
       # c[k] = sum_i h[i]*(1-h[i+k-(n-1)]) * dx (approximately)
       # This is complex; use the formula from the previous code

       w_neg = np.zeros(n)
       w_pos[1:] = w[mid+1:]
       w_neg[1:] = w[:mid][::-1]

       pref_n = np.cumsum(w_neg)

       consts = pref_p + pref_n[n-1-idx]
       kern = np.zeros(2*n-1)
       kern[mid+1:] = s[1:]
       kern[:mid][::-1] = s[1:]
       neigh = np.convolve(h, kern, 'full')[mid:mid+n]
       return (w[mid]*(1.0 - 2.0*h) + consts - neigh) * dx

           return base + lam*res + 0.5*rho*res*res

           res = x.sum() - target
           return g + (lam + rho*res) * np.ones(n)

                continue
            delta = rng.uniform(0, dmax)
        else:
            i = rng.randint(n)
            j = rng.randint(n - 1)
            if j >= i:
                j += 1
            if i == j:
                continue
            dlo = max(-h[j], -(1.0 - h[i]))
            dhi = min(1.0 - h[i], h[j])
            if dhi - dlo < 1e-11:
                continue
            delta = rng.uniform(-step, step)
            delta = min(max(delta, dlo), dhi)
            if abs(delta) < 1e-12:
                continue
        h[i] += delta
        h[j] -= delta
        nc = erdos_c5(h, n)
        if nc < cur:
            cur = nc
            if nc < best:
                best = nc
                best_h = h.copy()
            stuck = 0
        else:
            ap = np.exp(-(nc - cur) / max(T, 1e-12))
            if rng.random() < ap:
                stuck += 1
            else:
                h[i] -= delta
                h[j] += delta
                stuck += 1
        if stuck > 3000:
            h = project_box_sum(h + 0.03 * rng.randn(n), target)
            cur = erdos_c5(h, n)
            stuck = 0
    return best_h, best

def resample(h, n_old, n_new):
    x = np.linspace(0.0, 2.0, n_old)
    y = np.linspace(0.0, 2.0, n_new)
    return project_box_sum(np.interp(y, x, h), 0.5 * n_new)

def pair_polish(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    cur = erdos_c5(h, n)
    for rnd in range(4):
        if time.time() > deadline - 5.0:
            break
        g = smax_grad(h, n, 500.0)
        order = np.argsort(g)
        inc = order[:max(8, n // 8)]
        dec = order[-max(8, n // 8):]
        improved = False
        count = 0
        for ii in inc:
            if time.time() > deadline - 5.0 or count > 600:
                break
            for jj in dec:
                i, j = int(ii), int(jj)
                if i == j:
                    continue
                count += 1
                dlo = max(-h[j], -(1.0 - h[i]))
                dhi = min(1.0 - h[i], h[j])
                if dhi - dlo < 1e-11:
                    continue
                bd, bc = 0.0, cur
                for fr in (0.15, 0.35, 0.5, 0.7, 0.85):
                    d = dlo + fr * (dhi - dlo)
                    if abs(d) < 1e-11:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = erdos_c5(h, n)
                    if c < bc:
                        bc = c
                        bd = d
                    h[i] -= d
                    h[j] += d
                if bc < cur - 1e-14:
                    h[i] += bd
                    h[j] -= bd
                    cur = bc
                    improved = True
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n < 2:
        n = 400
        h = np.full(n, 0.5)
    if n > 1000:
        no = n
        n = 1000
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, no), h)
    h = project_box_sum(h, 0.5 * n)
    best_h, best_n, best_c = h.copy(), n, erdos_c5(h, n)

    # Phase 1: L-BFGS convergence
    e1 = t0 + 0.18 * total
    for b in (100.0, 1000.0, 10000.0, 100000.0):
        if time.time() > e1 - 2.0:
            break
        cnd, c = lbfgs(best_h, best_n, b, 60, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cnd, c

    # Phase 2: Anneal at current n
    e2 = t0 + 0.50 * total
    if time.time() < e2 - 10.0:
        ha, ca = anneal(best_h, best_n, e2, seed)
        if ca < best_c:
            best_h, best_c = ha, ca
        if time.time() < e2 - 2.0:
            cnd, c = lbfgs(best_h, best_n, 50000.0, 30, deadline)
            if c < best_c - 1e-15:
                best_h, best_c = cnd, c

    # Perturb and re-optimize to escape local optima
    e3 = t0 + 0.62 * total
    rng = np.random.RandomState(seed + 77)
    for trial in range(3):
        if time.time() > e3 - 10.0:
            break
        hh = project_box_sum(best_h + 0.04 * rng.randn(best_n), 0.5 * best_n)
        hh, c = lbfgs(hh, best_n, 5000.0, 40, deadline)
        if c < best_c:
            hh2, c2 = anneal(hh, best_n, min(e3, time.time() + 0.04 * total), seed + 200 + trial)
            if c2 < c:
                hh, c = hh2, c2
            if c < best_c:
                best_h, best_c = hh, c

    # Phase 3: Higher resolution
    e4 = t0 + 0.85 * total
    for nn in (800, 1200):
        if time.time() > e4 - 25.0 or nn == best_n:
            continue
        hh = resample(best_h, best_n, nn)
        hh, c = lbfgs(hh, nn, 3000.0, 45, deadline)
        if c < best_c + 0.003:
            if time.time() < e4 - 15.0:
                hh2, c2 = anneal(hh, nn, min(e4 - 8.0, time.time() + 0.10 * total), seed + 500)
                if c2 < c:
                    hh, c = hh2, c2
            if time.time() < e4 - 5.0:
                hh, c = lbfgs(hh, nn, 80000.0, 25, deadline)
            if c < best_c - 1e-15:
                best_h, best_n, best_c = hh, nn, c

    # Phase 4: Final polish
    e5 = deadline - 3.0
    if time.time() < e5 - 12.0:
        best_h, best_c = pair_polish(best_h, best_n, e5 - 8.0, seed + 900)
    if time.time() < e5 - 4.0:
        cnd, c = lbfgs(best_h, best_n, 300000.0, 15, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cnd, c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
