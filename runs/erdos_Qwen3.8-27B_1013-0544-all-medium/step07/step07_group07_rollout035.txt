def lse_slsqp(h, n, alpha, maxiter, ftol):

        zc = g[z_cand, :]
        oc = g[o_cand, :]
        base = corr[None, None, :] + dx * zc[:, None, :] - dx * oc[None, :, :]
        new_max = base.max(axis=2)
        flat_idx = int(np.argmin(new_max))
        ii, jj = flat_idx // no, flat_idx % no
        best_val = new_max[ii, jj]
        if best_val >= cur_c - 1e-14:
            no_improve += 1
            if no_improve > 200:
                break
            if no_improve % 20 == 0:
                z_cand = zeros[rng.randint(0, len(zeros), size=min(100, len(zeros)))]
                o_cand = ones[rng.randint(0, len(ones), size=min(100, len(ones)))]
                zc = g[z_cand, :]
                oc = g[o_cand, :]
                base = corr[None, None, :] + dx * zc[:, None, :] - dx * oc[None, :, :]
                new_max = base.max(axis=2)
                flat_idx = int(np.argmin(new_max))
                ii2, jj2 = flat_idx // len(o_cand), flat_idx % len(o_cand)
                best_val = new_max[ii2, jj2]
                if best_val >= cur_c - 1e-14:
                    continue
                best_i, best_j = z_cand[ii2], o_cand[jj2]
            else:
                continue
        else:
            best_i, best_j = z_cand[ii], o_cand[jj]
            no_improve = 0
        h[best_i] = 1.0
        h[best_j] = 0.0
        corr += dx * (g[best_i, :] - g[best_j, :])
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def lse_sweep(h, n, alphas, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline - 10:
            break
        mi = 2000 if a < 1e4 else (4000 if a < 1e6 else 6000)
        h = lse_slsqp(h, n, a, mi)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

def barrier_push(h, n, c_cur, deadline, n_iters=15):
    best_h, best_c = h.copy(), c_cur
    for i in range(n_iters):
        if time.time() > deadline - 5:
            break
        t = best_c * (1.0 - 0.005 * (i + 1))
        h = barrier_slsqp(h, n, t, 2500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 30
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h, best_c = h0.copy(), compute_c5(h0, n)

    # Phase 1: Quick LSE to refine continuous solution
    p1_end = t0 + 0.08 * total
    h = lse_sweep(h0, n, [1000, 10000, 50000, 200000], p1_end)
    if h[1] < best_c:
        best_c, best_h = h[1], h[0].copy()

    # Phase 2: Binary local search (main improvement driver)
    p2_end = t0 + 0.35 * total
    hb, cb = binary_search(best_h, n, p2_end, top_k=8, n_cand=100, max_iters=80000, rng=rng)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 3: Continuous optimization from binary solution
    p3_end = t0 + 0.60 * total
    h = best_h.copy()
    # Slight smoothing to give continuous optimizer room
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = project_feasible(h_smooth, n)
    h, c = lse_sweep(h_smooth, n, [5000, 50000, 500000, 5000000, 50000000], p3_end)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Barrier polish
    p4_end = t0 + 0.72 * total
    h, c = barrier_push(best_h, n, best_c, p4_end, n_iters=18)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: Try perturbed restarts
    p5_end = t0 + 0.85 * total
    x = np.linspace(0, 2, n)
    perts = [
        0.01 * rng.randn(n),
        0.02 * np.sin(2 * np.pi * 3.0 * x + 0.5),
        0.015 * np.cos(2 * np.pi * 6.0 * x),
        0.03 * rng.randn(n),
        0.008 * np.sin(2 * np.pi * 11.0 * x + 1.3),
    ]
    for pert in perts:
        if time.time() > p5_end - 15:
            break
        h = project_feasible(best_h + pert, n)
        h, c = lse_sweep(h, n, [50000, 500000, 5000000], p5_end - 10)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if time.time() > p5_end - 10:
            break
        h2, c2 = barrier_push(h, n, c, p5_end - 5, n_iters=8)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 6: Second binary search from best continuous
    p6_end = t0 + 0.92 * total
    hb, cb = binary_search(best_h, n, p6_end, top_k=10, n_cand=120, max_iters=40000, rng=rng)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 7: Final LSE + barrier polish
    h, c = lse_sweep(best_h, n, [100000, 2000000, 20000000, 100000000], dl - 20)
    if c < best_c:
        best_c, best_h = c, h.copy()
    h, c = barrier_push(best_h, n, best_c, dl - 10, n_iters=10)
    if c < best_c:
        best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
