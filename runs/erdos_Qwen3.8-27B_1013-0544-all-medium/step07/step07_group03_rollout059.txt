            if s < 1e-15:

def subgrad_at_lag(s, h, n):
    """Subgradient of correlation at shift s (where s can be negative)"""
    grad = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    elif s > 0:
        if s < n:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
    else:
        ss = -s
        if ss < n:
            grad[:n-ss] += (1.0 - h[ss:])
            grad[ss:] -= h[:n-ss]
    return grad

    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    x = h.copy()
    v = np.zeros(n)
    beta = 0.9
    base_lr = 0.3
    for t in range(max_iters):

        cur_c = c.max()
        if cur_c < best_c:

            best_h = x.copy()
        # Softmax weights

        w = np.exp(c_shifted - c_shifted.max())

        # Gradient: only top-k terms

            w[np.argmax(c)] = 1.0

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

    while time.time() < deadline:
        t += 1
        c = full_corr(x) * dx
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = x.copy()
        cs = alpha * (c - cur_c)
        w = np.exp(cs - cs.max())
        wsum = w.sum()
        if wsum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            wsum = 1.0
        w /= wsum
        grad = np.zeros(n)
        top_idx = np.argsort(c)[::-1][:top_k]
        for ii in range(len(top_idx)):
            s = s_range[top_idx[ii]]
            wi = w[top_idx[ii]]
            if s == 0:
                grad += wi * (1.0 - 2.0 * x)
            elif 0 < s < n:
                grad[s:] += wi * (1.0 - x[:n - s])
                grad[:n - s] -= wi * x[s:]
            elif 0 < -s < n:
                ss = -s
                grad[:n - ss] += wi * (1.0 - x[ss:])
                grad[ss:] -= wi * x[:n - ss]
        lr = 0.4 / (np.sqrt(t) + 1.0)
        x = project_feasible(x - lr * grad, n)
    return best_h, best_c

def compute_G(hb, n):
    G = np.zeros((n, 2 * n - 1))
    G[:, n - 1] = 2.0 * hb
    for s in range(1, n):
        si = n - 1 + s
        hp = np.zeros(n)
        hm = np.zeros(n)
        hp[:n - s] = hb[s:]
        hm[s:] = hb[:n - s]
        G[:, si] = hp + hm
        nsi = n - 1 - s
        hp2 = np.zeros(n)
        hm2 = np.zeros(n)
        hp2[s:] = hb[:n - s]
        hm2[:n - s] = hb[s:]
        G[:, nsi] = hp2 + hm2
    return G

def binary_search_fast(h_init, n, deadline, max_sweeps=10000):
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n // 2]] = 1.0
    corr = full_corr(h_b)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones_idx = np.where(h_b == 1)[0]
    zeros_idx = np.where(h_b == 0)[0]
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        G = compute_G(h_b, n)
        sweep_improved = 0
        for q_i in range(len(zeros_idx)):
            q = int(zeros_idx[q_i])
            if len(ones_idx) == 0:
                break
            D = corr - G[q, :]
            scores = np.max(D[None, :] + G[ones_idx, :], axis=1)
            best_i = int(np.argmin(scores))
            if scores[best_i] * dx < cur_c - 1e-12:
                p = int(ones_idx[best_i])
                old_c = cur_c
                h_b[p] = 0.0
                h_b[q] = 1.0
                corr = full_corr(h_b)
                cur_c = corr.max() * dx
                if cur_c < old_c - 1e-12:
                    ones_idx = np.delete(ones_idx, best_i)
                    ones_idx = np.append(ones_idx, q)
                    qpos = int(np.where(zeros_idx == q)[0][0])
                    zeros_idx = np.delete(zeros_idx, qpos)
                    zeros_idx = np.append(zeros_idx, p)
                    sweep_improved += 1
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h_b.copy()
                else:
                    h_b[p] = 1.0
                    h_b[q] = 0.0
                    cur_c = old_c
        if sweep_improved == 0:
            no_improve += 1
            if no_improve > 3 and len(ones_idx) > 2:
                for _ in range(3):
                    pi = int(np.random.randint(len(ones_idx)))
                    qi = int(np.random.randint(len(zeros_idx)))
                    p, q = int(ones_idx[pi]), int(zeros_idx[qi])
                    h_b[p], h_b[q] = 0.0, 1.0
                    ones_idx = np.delete(ones_idx, pi)
                    ones_idx = np.append(ones_idx, q)
                    zeros_idx = np.delete(zeros_idx, qi)
                    zeros_idx = np.append(zeros_idx, p)
                corr = full_corr(h_b)
                cur_c = corr.max() * dx
                no_improve = 0
        else:
            no_improve = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    np.random.seed(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: continuous (10%)
    p1 = t0 + 0.10 * total
    h, c = fast_proj_grad(best_h, n, p1, alpha=500, top_k=25)
    if c < best_c:
        best_c, best_h = c, h
    # Phase 2: binary (30%)
    p2 = t0 + 0.40 * total
    hb, cb = binary_search_fast(best_h, n, p2, max_sweeps=8000)
    if cb < best_c:
        best_c, best_h = cb, hb
    # Phase 3: continuous refine (12%)
    p3 = t0 + 0.52 * total
    h, c = fast_proj_grad(best_h, n, p3, alpha=2000, top_k=30)
    if c < best_c:
        best_c, best_h = c, h
    # Phase 4: binary (30%)
    p4 = t0 + 0.82 * total
    hb, cb = binary_search_fast(best_h, n, p4, max_sweeps=8000)
    if cb < best_c:
        best_c, best_h = cb, hb
    # Phase 5: continuous refine (10%)
    p5 = t0 + 0.92 * total
    h, c = fast_proj_grad(best_h, n, p5, alpha=5000, top_k=40)
    if c < best_c:
        best_c, best_h = c, h
    # Phase 6: final binary pass (8%)
    if time.time() < dl - 3:
        hb, cb = binary_search_fast(best_h, n, dl - 1, max_sweeps=4000)
        if cb < best_c:
            best_c, best_h = cb, hb
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
