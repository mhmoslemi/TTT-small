import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    """Softmax approximation of max(c)"""
    m = c.max()
    shifted = alpha * (c - m)
    return m + np.log(np.exp(shifted).sum()) / alpha

def compute_smooth_gradient(h, n, alpha):
    """Gradient of softmax-smoothed max of correlation"""
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            # c = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            j = np.arange(n)
            mask1 = j + lag < n
            mask2 = j >= lag
            dcdh = np.zeros(n)
            dcdh[mask1] += (1 - h[j[mask1] + lag])
            if lag > 0:
                dcdh[mask2] -= h[j[mask2] - lag]
            grad += w * dcdh
        else:
            L = -lag
            # c = sum_{i=0}^{n-1-L} (1-h[i])*h[i+L]
            j = np.arange(n)
            mask1 = j + L < n
            mask2 = j >= L
            dcdh = np.zeros(n)
            dcdh[mask1] += -h[j[mask1] + L]
            dcdh[mask2] += (1 - h[j[mask2] - L])
            grad += w * dcdh
    return grad

def adam_optimize(h, n, alpha, lr, max_iter, budget_dl):
    """Adam optimizer for the smooth max objective"""
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > budget_dl:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full')
        mval = c.max()
        shifted = alpha * (c - mval)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        grad = compute_smooth_gradient(h, n, alpha)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = h + step
        h_new = project_feasible(h_new, n)
        
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, weights):
    """Fast gradient computation using vectorized per-lag updates"""
    grad = np.zeros(n)
    for lag in range(n):
        w = weights[n-1+lag]
        if w < 1e-10:
            continue
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
    for L in range(1, n):
        w = weights[n-1-L]
        if w < 1e-10:
            continue
        grad[:n-L] -= w * h[L:]
        grad[L:] += w * (1.0 - h[:n-L])
    return grad

def adam_optimize(h, n, alpha, lr, max_iter, deadline, momentum_decay=1.0):
    """Adam optimization with softmax-smoothed max objective"""
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    current_lr = lr
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        c = compute_corr(h)
        mval = c.max()
        shifted = alpha * (c - mval)
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 50)
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-30:
            weights = np.ones_like(c) / len(c)
        else:
            weights /= ws
        
        grad = compute_grad(h, n, weights)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = current_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        h_new = h + step
        h_new = project_feasible(h_new, n)
        
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        
        # Decay learning rate slowly
        current_lr = lr * (0.999 ** t)
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, weights):
    grad = np.zeros(n)
    for lag in range(n):
        w = weights[n-1+lag]
        if w < 1e-10:
            continue
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
    for L in range(1, n):
        w = weights[n-1-L]
        if w < 1e-10:
            continue
        grad[:n-L] -= w * h[L:]
        grad[L:] += w * (1.0 - h[:n-L])
    return grad

def adam_opt(h, n, alpha, lr, max_iter, deadline, patience=500):

    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_t = 0
    curr_lr = lr
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it - best_t > patience:
            break
        t += 1
        c = compute_corr(h)
        mval = c.max()
        shifted = alpha * (c - mval)
        shifted = np.clip(shifted, -60, 60)
        wts = np.exp(shifted)
        wts /= wts.sum()
        
        grad = compute_grad(h, n, wts)
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad*grad
        mh = m / (1-b1**t)
        vh = v / (1-b2**t)
        h = h + curr_lr * mh / (np.sqrt(vh) + eps)
        h = project_feasible(h, n)
        
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            best_t = it
        curr_lr *= 0.9997
    
    return best_h, best_c5
