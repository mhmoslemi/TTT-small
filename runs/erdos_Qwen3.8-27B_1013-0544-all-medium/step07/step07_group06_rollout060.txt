def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    a = np.clip(np.asarray(a, dtype=float).ravel(), 0.0, 1.0)
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    target = 0.5 * n
    # Bisection on the threshold
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5*(lo+hi), 0.0, 1.0)
    # Fix sum exactly
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 0:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 0:
                h += diff * h / s
    h = np.clip(h, 0.0, 1.0)

        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))

    return h

def lse_obj_grad(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    v = 1.0 - h
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(P, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

            j = zeros[rng.randint(len(zeros))]
            cur[i], cur[j] = 0.0, 1.0
        
        # SA from perturbed
        sa_out, sa_c = binary_sa(cur, n, min(deadline, time.time() + 20), 
                                  max_iter=5000, seed=seed + iteration)
        sa_M = int(round(sa_c * n / 2.0))
        if sa_M < cur_M:
            cur = sa_out.copy()
            cur_M = sa_M
        if sa_c < bestM:
            bestM = int(round(sa_c * n / 2.0))
            besth = sa_out.copy()
    
    return besth, bestM * (2.0 / float(n))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + max(30.0, float(budget_s) - 15.0)
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    best_h, best_c, best_n = project_feasible(h0, n0), np.inf, n0
    
    # Initial continuous polish
    if time.time() < end - 300:
        pc, pnc = polish_continuous(h0, n0, end - 500)
        bc = compute_c5(pnc, n0)
        if bc < best_c:
            best_c, best_h = bc, pnc.copy()
    
    # Main binary optimization across multiple n values
    n_list = sorted(set([n0, 300, 400, 500, 600, 800]))
    n_list = [n for n in n_list if n % 2 == 0 and 100 <= n <= 1000]
    
    for n in n_list:
        if time.time() > end - 40:
            break
        remaining = end - time.time()
        if remaining < 30:
            break
        alloc = remaining / len([x for x in n_list if x >= n])
        
        # Get initial for this n
        if n == n0:
            h_init = best_h.copy()
        else:
            x0 = np.linspace(0.0, 2.0, len(best_h))
            x1 = np.linspace(0.0, 2.0, n)
            h_init = np.interp(x1, x0, best_h)
        
        # Quick continuous polish
        if time.time() < end - 60:
            pc, pnc = polish_continuous(h_init, n, min(end - 60, time.time() + 40),
                                         betas=(2000.0, 8000.0))
            h_init = pc
        
        # ILS binary search
        alloc_end = min(end - 20, time.time() + max(40.0, alloc * 0.9))
        bh, bc_new = ilds_binary(h_init, n, alloc_end, seed=seed + n)
        if bc_new < best_c:
            best_c, best_h, best_n = bc_new, bh.copy(), n
            print(f"  n={n}: C5={bc_new:.6f}")
        
        # Also try from a random binary start
        if time.time() < end - 50 and n % 2 == 0:
            h_rand = (np.arange(n) % 2).astype(float)
            bh2, bc2 = ilds_binary(h_rand, n, min(end - 30, time.time() + 30), seed=seed + 1000 + n)
            if bc2 < best_c:
                best_c, best_h, best_n = bc2, bh2.copy(), n
    
    # Final polish on best
    if time.time() < end - 10:
        fb, fc = binary_greedy(best_h, best_n, end - 5, max_swaps=100)
        if fc < best_c:
            best_c, best_h = fc, fb.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
