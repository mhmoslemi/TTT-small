def bin_stats(h):

    x=h.astype(np.float64)
    N=1
    while N < 2*n-1: N <<=1
    F=np.fft.rfft(x,N)
    A=np.fft.irfft(F*F,N)[:n]
    A=np.rint(A).astype(np.int32)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c(h,n):
    return np.correlate(h,1.0-h,mode="full")*(2.0/n)

def project_box_sum(v,n):
    v=np.nan_to_num(np.asarray(v,float).reshape(-1)); t=0.5*n
    lo=float(v.min())-1; hi=float(v.max())+1
    for _ in range(45):
        mid=0.5*(lo+hi); s=np.clip(v-mid,0,1).sum()
        if s>t: lo=mid
        else: hi=mid
    v=np.clip(v-0.5*(lo+hi),0,1)
    for _ in range(30):
        d=t-v.sum()
        if abs(d)<1e-13: break
        m=(v>1e-12)&(v<1-1e-12); c=int(m.sum())
        if c: v[m]+=d/c
        else: v+=d/n
        v=np.clip(v,0,1)
    return v

def interp_h(h,o,n):
    if o==n: return h.copy()
    return np.interp((np.arange(n)+0.5)*(2.0/n),(np.arange(o)+0.5)*(2.0/o),h)

def lse_w(c,a):
    m=float(c.max()); sh=np.clip(a*(c-m),-50,0); e=np.exp(sh); s=float(e.sum())
    if s<=0:
        q=np.zeros_like(c); q[int(np.argmax(c))]=1; return q
    return e/s

def lse_obj(x,n,a):
    c=compute_c(x,n); m=float(c.max())
    if a<=0: return m
    sh=np.clip(a*(c-m),-50,0); s=float(np.exp(sh).sum())
    return m+(np.log(s)/a if s>0 else 0.0)

def lse_grad(x,n,a):
    c=compute_c(x,n); q=lse_w(c,a); g=np.zeros(n); dx=2.0/n; off=n-1
    for idx in np.flatnonzero(q>1e-12):
        k=int(idx-off); w=float(q[idx])
        if k>0:
            s=k; m=n-s; g[s:]+=w*(1.0-x[:m]); g[:m]-=w*x[s:]
        elif k<0:
            s=-k; m=n-s; g[:m]+=w*(1.0-x[s:]); g[s:]-=w*x[:m]
        else:
            g+=w*(1.0-2.0*x)
    return dx*g

def pen_obj(x,n,a,mu):
    return lse_obj(x,n,a)+0.5*mu*float(x.sum()-0.5*n)**2

def pen_grad(x,n,a,mu):
    return lse_grad(x,n,a)+mu*float(x.sum()-0.5*n)

def lbfgsb(h,n,a,mu,it):
    h=project_box_sum(h,n)
    try:
        r=minimize(pen_obj,h,args=(n,a,mu),jac=pen_grad,method="L-BFGS-B",bounds=[(0,1)]*n,options={"maxiter":it,"maxfun":4*it,"ftol":1e-14,"gtol":1e-8})
        if r.x is not None: return r.x
    except Exception:
        pass
    return h

def tune_cont(h,n,seed,deadline):
    h=project_box_sum(h,n); besth=h.copy(); bestc=float(compute_c(h,n).max()); work=h.copy(); wc=bestc
    for a,mu,it in ((5,20,90),(25,120,70),(120,600,50),(500,2500,40)):
        if time.time()>deadline-8: break
        h2=lbfgsb(work,n,a,mu,it); h2=project_box_sum(h2,n); v=float(compute_c(h2,n).max())
        if v<bestc:
            bestc=bestc=v; besth=h2.copy()
        if v<=wc+1e-8: work=h2; wc=v
        else: work=besth.copy(); wc=bestc
    besth=project_box_sum(besth,n); return besth,float(compute_c(besth,n).max())

    f1=int(rng.randint(1,7)); f2=int(rng.randint(1,9)); p1=float(rng.uniform(0,2*np.pi)); p2=float(rng.uniform(0,2*np.pi))

def bin_stats(h):
    n=h.size; N=1
    while N<2*n-1: N <<= 1
    F=np.fft.rfft(h.astype(float),N); A=np.rint(np.fft.irfft(F*F,N)[:n]).astype(np.int32)
    pref=np.cumsum(h).astype(np.int32); m=int(pref[-1])
    P=np.maximum(pref[n-2::-1]-A[1:],0).astype(np.int32)

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, float).reshape(-1))
    t = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > t:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = t - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

    if old == new:
        return h.copy()
    xo = (np.arange(old) + 0.5) * (2.0 / old)
    xn = (np.arange(new) + 0.5) * (2.0 / new)
    return np.interp(xn, xo, h)


def lse_w(c, a):
    m = float(c.max())
    sh = np.clip(a * (c - m), -50.0, 0.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    return e / s


def lse_obj(x, n, a):
    c = compute_c(x, n)
    m = float(c.max())
    if a <= 0.0:
        return m
    sh = np.clip(a * (c - m), -50.0, 0.0)
    s = float(np.exp(sh).sum())
    return m + (np.log(s) / a if s > 0.0 else 0.0)


def lse_grad(x, n, a):
    c = compute_c(x, n)
    q = lse_w(c, a)
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q > 1e-12):
        k = int(idx - off)
        w = float(q[idx])
        if k > 0:
            s = k
            m = n - s
            g[s:] += w * (1.0 - x[:m])
            g[:m] -= w * x[s:]
        elif k < 0:
            s = -k
            m = n - s
            g[:m] += w * (1.0 - x[s:])
            g[s:] -= w * x[:m]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g


def pen_obj(x, n, a, mu):
    return lse_obj(x, n, a) + 0.5 * mu * float(x.sum() - 0.5 * n) ** 2


def pen_grad(x, n, a, mu):
    return lse_grad(x, n, a) + mu * float(x.sum() - 0.5 * n)


def lbfgsb(h, n, a, mu, it):
    h = project_box_sum(h, n)
    try:
        r = minimize(pen_obj, h, args=(n, a, mu), jac=pen_grad,
                     method="L-BFGS-B", bounds=[(0.0, 1.0)] * n,
                     options={"maxiter": it, "maxfun": 4 * it,
                              "ftol": 1e-14, "gtol": 1e-8})
        if r.x is not None:
            return r.x
    except Exception:
        pass
    return h


def tune_cont(h, n, seed, deadline):
    h = project_box_sum(h, n)
    besth = h.copy()
    bestc = float(compute_c(h, n).max())
    work = h.copy()
    wc = bestc
    for a, mu, it in ((5.0, 20.0, 80), (25.0, 120.0, 60),
                      (120.0, 600.0, 45), (500.0, 2500.0, 35)):
        if time.time() > deadline - 8:
            break
        h2 = lbfgsb(work, n, a, mu, it)
        h2 = project_box_sum(h2, n)
        v = float(compute_c(h2, n).max())
        if v < bestc:
            bestc = v
            besth = h2.copy()
        if v <= wc + 1e-8:
            work = h2
            wc = v
        else:
            work = besth.copy()
            wc = bestc
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())


def make_bin(v, n, rng=None):
    v = interp_h(v, v.size, n)
    if rng is not None:
        v = v + 1e-6 * rng.randn(n)
    k = n // 2
    idx = np.argpartition(v, n - k)[:n - k]
    h = np.zeros(n, dtype=np.int8)
    h[idx] = 1
    return h


def mk_start(n, rng, kind, par):
    if kind == 0 and par is not None:
        return make_bin(par, n, rng)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if kind == 1:
        h = np.zeros(n, dtype=np.int8)
        a = n // 4
        b = 3 * n // 4
        h[a:b] = 1
        return make_bin(h.astype(float), n, rng)
    f1 = int(rng.randint(1, 7))
    f2 = int(rng.randint(1, 9))
    p1 = float(rng.uniform(0.0, 2.0 * np.pi))
    p2 = float(rng.uniform(0.0, 2.0 * np.pi))
    v = 0.5 + 0.35 * np.sin(2.0 * np.pi * f1 * x + p1) + \
        0.25 * np.sin(2.0 * np.pi * f2 * x + p2)
    return make_bin(v, n, rng)


def bin_stats(h):
    n = h.size
    A = np.correlate(h.astype(float), h.astype(float), mode="full")
    A = np.rint(A[n - 1:]).astype(np.int32)
    pref = np.cumsum(h).astype(np.int32)
    m = int(pref[-1])
    P = np.maximum(pref[n - 2::-1] - A[1:], 0).astype(np.int32)
    Q = np.maximum(m - pref[:n - 1] - A[1:], 0).astype(np.int32)
    mx = int(max(int(P.max()), int(Q.max()))) if n > 1 else 0
    return P, Q, mx, float(2.0 * mx / n)


def p_delta(h, i1, i0):
    n = h.shape[0]
    dP = np.zeros(n - 1, dtype=np.int32)
    z = 1 - h
    m = n - 1 - i1
    if m > 0:
        dP[:m] -= z[i1 + 1:i1 + 1 + m]
    m = i1
    if m > 0:
        dP[:m] += h[m - 1::-1]
    m = n - 1 - i0
    if m > 0:
        dP[:m] += z[i0 + 1:i0 + 1 + m]
    m = i0
    if m > 0:
        dP[:m] -= h[m - 1::-1]
    dP[abs(i1 - i0) - 1] += 1
    return dP


def binary_local(h, n, seed, deadline):
    h = h.astype(np.int8).copy()
    z = 1 - h
    P, Q, mx, _ = bin_stats(h)
    besth = h.copy()
    bestmx = mx
    rng = np.random.RandomState(seed)
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(z)
    w = P + Q + 1
    wk = float(w.sum())
    wc = np.cumsum(w) if wk > 0.0 else np.arange(n, dtype=float)
    stuck = 0
    kicks = 0
    step = 0

    while time.time() < deadline:
        step += 1
        if step & 4095 == 0:
            w = P + Q + 1
            wk = float(w.sum())
            wc = np.cumsum(w) if wk > 0.0 else np.arange(n, dtype=float)

        if ones.size == 0 or zeros.size == 0:
            break

        a = -1
        b = -1
        if rng.random() < 0.65 or wk <= 0.0:
            a = int(ones[rng.randint(ones.size)])
            b = int(zeros[rng.randint(zeros.size)])
        else:
            u = float(rng.random()) * wk
            d = 1 + int(np.searchsorted(wc, u))
            d = min(n - 1, d)
            if P[d - 1] >= Q[d - 1]:
                for _ in range(6):
                    i = int(rng.randint(0, n - d))
                    if h[i] == 1 and h[i + d] == 0:
                        a = i
                        b = int(zeros[rng.randint(zeros.size)])
                        if b == i + d and zeros.size > 1:
                            for __ in range(4):
                                bb = int(zeros[rng.randint(zeros.size)])
                                if bb != b:
                                    b = bb
                                    break
                        break
            else:
                for _ in range(6):
                    i = int(rng.randint(0, n - d))
                    if h[i] == 0 and h[i + d] == 1:
                        b = i
                        a = int(ones[rng.randint(ones.size)])
                        if a == i + d and ones.size > 1:
                            for __ in range(4):
                                aa = int(ones[rng.randint(ones.size)])
                                if aa != a:
                                    a = aa
                                    break
                        break
            if a < 0 or b < 0:
                a = int(ones[rng.randint(ones.size)])
                b = int(zeros[rng.randint(zeros.size)])

        if a < 0 or b < 0 or h[a] != 1 or z[b] != 1:
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(z)
            if ones.size == 0 or zeros.size == 0:
                break
            a = int(ones[rng.randint(ones.size)])
            b = int(zeros[rng.randint(zeros.size)])

        dP = p_delta(h, a, b)
        dQ = p_delta(z, b, a)
        cand = max(int((P + dP).max()), int((Q + dQ).max()))

        if cand < mx or (cand == mx and rng.random() < 0.15) or \
                (cand == mx + 1 and rng.random() < 0.02):
            P += dP
            Q += dQ
            mx = cand
            h[a] = 0
            z[a] = 1
            h[b] = 1
            z[b] = 0
            pa = (ones == a)
            pb = (zeros == b)
            if pa.any():
                ones[pa] = b
            if pb.any():
                zeros[pb] = a
            if mx < bestmx:
                besth = h.copy()
                bestmx = mx
            stuck = 0
        else:
            stuck += 1

        if stuck > 6000:
            kc = 3 + int(rng.randint(4))
            for _ in range(kc):
                a = int(ones[rng.randint(ones.size)])
                b = int(zeros[rng.randint(zeros.size)])
                if h[a] != 1 or z[b] != 1:
                    break
                dP = p_delta(h, a, b)
                dQ = p_delta(z, b, a)
                P += dP
                Q += dQ
                h[a] = 0
                z[a] = 1
                h[b] = 1
                z[b] = 0
                pa = (ones == a)
                pb = (zeros == b)
                if pa.any():
                    ones[pa] = b
                if pb.any():
                    zeros[pb] = a
            mx = int(max(int(P.max()), int(Q.max())))
            stuck = 0
            kicks += 1
            if kicks > 8:
                h = besth.astype(np.int8).copy()
                z = 1 - h
                P, Q, mx, _ = bin_stats(h)
                ones = np.flatnonzero(h)
                zeros = np.flatnonzero(z)
                w = P + Q + 1
                wk = float(w.sum())
                wc = np.cumsum(w) if wk > 0.0 else np.arange(n, dtype=float)
                kicks = 0

    P, Q, mx, _ = bin_stats(besth.astype(np.int8))
    return besth.astype(float), float(2.0 * mx / n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    try:
        par = np.asarray(initial_h_values, float).copy()
        n0 = int(par.size)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        par = 0.5 + 0.5 * np.sin(3.0 * np.pi * x)

    par = project_box_sum(par, n0)
    besth = par.copy()
    bestc = float(compute_c(par, n0).max())
    bestn = n0

    span = deadline - t0
    if span < 60:
        return besth, bestc, bestn

    if 64 <= n0 <= 600:
        dl = t0 + span * 0.22
        h, c = tune_cont(par, n0, seed + 1, dl)
        if c < bestc:
            besth, bestc, bestn = h, c, n0

    ns = [n0 if n0 in (384, 400, 448, 480, 512, 560, 600) else 400]
    for m in (384, 400, 448, 512):
        if m not in ns and 16 <= m <= 700:
            ns.append(m)
    ns = sorted(set(ns))[:4]

    plans = []
    for m in ns:
        plans.append((m, 0))
        plans.append((m, 1))
        plans.append((m, 2))

    bin_dl = t0 + span * 0.86
    idx = 0
    for m, kind in plans:
        if time.time() >= bin_dl:
            break
        left = len(plans) - idx
        chunk = bin_dl - (bin_dl - time.time()) / float(left)
        if chunk < time.time() + 8:
            chunk = min(bin_dl, time.time() + 20)
        hstart = mk_start(m, np.random.RandomState(seed + 1000 * m + kind), kind, besth)
        hb, cb = binary_local(hstart, m, seed + 2000 * m + kind, chunk)
        if cb < bestc:
            besth, bestc, bestn = hb, cb, m
        idx += 1

    if time.time() < deadline - 25:
        if bestn > 600:
            hf, cf = tune_cont(interp_h(besth, bestn, 560), 560, seed + 999, deadline)
            if cf < bestc:
                besth, bestc, bestn = hf, cf, 560
        else:
            hf, cf = tune_cont(besth, bestn, seed + 999, deadline)
            if cf < bestc:
                besth, bestc, bestn = hf, cf, bestn

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, int(bestn)
