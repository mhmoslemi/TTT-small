import numpy as np
from scipy.optimize import minimize
import time

_stop_t = 0.0

def stop_cb(*a):
    return time.time() > _stop_t

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.full(n - a.size, 0.5)])
        else:
            a = a[:n]
    a = np.nan_to_num(a, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    lo = float(a.min()) - 1.0
    hi = float(a.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += diff * room / s
        else:
            s = h.sum()
            if s < 1e-15: break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            h[int(np.argmax(1.0 - h))] = min(1.0, h[int(np.argmax(1.0 - h))] + diff)
        else:
            h[int(np.argmax(h))] = max(0.0, h[int(np.argmax(h))] + diff)
    return h

def top_n_binary(h, n):
    hb = np.zeros(n)
    t = n // 2
    if t > 0:
        hb[np.argsort(-h)[:t]] = 1.0
    return hb

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full").astype(float)

def binary_local_search(h, n, deadline, max_rounds=200):
    h = top_n_binary(h, n).copy()
    c = compute_corr(h, n)
    M = float(c.max())
    besth, bestM = h.copy(), M
    lags = np.arange(-(n-1), n, dtype=int)
    nL = 2 * n - 1

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        # Compute G matrix: G[i,k] = dc[k]/dh[i]
        # G[i,k] = (1-h[i+lag_k])_{valid} - h[i-lag_k]_{valid}
        ip = zeros[:, None] + lags[None, :]
        im = zeros[:, None] - lags[None, :]
        Gz = np.zeros((zeros.size, nL))
        mp = (ip >= 0) & (ip < n)
        mm = (im >= 0) & (im < n)
        Gz[mp] += (1.0 - h[ip[mp]]) if mp.size > 0 else 0
        Gz[mm] -= h[im[mm]] if mm.size > 0 else 0

        ip = ones[:, None] + lags[None, :]
        im = ones[:, None] - lags[None, :]
        Go = np.zeros((ones.size, nL))
        mp = (ip >= 0) & (ip < n)
        mm = (im >= 0) & (im < n)
        Go[mp] += (1.0 - h[ip[mp]]) if mp.size > 0 else 0
        Go[mm] -= h[im[mm]] if mm.size > 0 else 0

        # For swap (i in zeros, j in ones): new_c = c + Gz[i] - Go[j]
        # new_max = max(c + Gz[i] - Go[j])
        # Find best swap: minimize new_max
        best_im = None
        best_jm = None
        best_newmax = M + 1.0
        chunk_z = 50
        for zi_start in range(0, zeros.size, chunk_z):
            if time.time() > deadline:
                break
            zi_end = min(zi_start + chunk_z, zeros.size)
            D = c[None, :] + Gz[zi_start:zi_end, :]  # shape (chunk, nL)
            # new_max[zi, oj] = max_k (D[zi, k] - Go[oj, k])
            nm = np.max(D[:, None, :] - Go[None, :, :], axis=2)  # (chunk, ones.size)
            flat_idx = int(np.argmin(nm))
            li, lj = divmod(flat_idx, ones.size)
            val = float(nm[li, lj])
            if val < best_newmax:
                best_newmax = val
                best_im = zeros[zi_start + li]
                best_jm = ones[lj]
        if best_im is None or best_newmax >= M - 1e-12:
            break
        h[best_im] = 1.0
        h[best_jm] = 0.0
        c = compute_corr(h, n)
        M = float(c.max())
        if M < bestM:
            bestM = M
            besth = h.copy()
    return besth, bestM * (2.0 / n)

def coord_descent(h, n, max_sweeps=300, deadline=None):
    h = project_feasible(h, n).copy()
    c = compute_corr(h, n)
    dx = 2.0 / n
    best_c = float(c.max()) * dx
    lags = np.arange(-(n-1), n, dtype=int)
    nL = 2 * n - 1
    rng = np.random.RandomState(0)

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        any_change = False
        for i in order:
            hi = h[i]
            if hi < 1e-14 or hi > 1.0 - 1e-14:
                continue
            ip = i + lags
            im = i - lags
            g = np.zeros(nL)
            mp = (ip >= 0) & (ip < n)
            mm = (im >= 0) & (im < n)
            g[mp] += (1.0 - h[ip[mp]])
            g[mm] -= h[im[mm]]
            dlo = -hi
            dhi = 1.0 - hi
            # Ternary search for min of max_k(c[k] + d*g[k])
            al, ar = dlo, dhi
            for _ in range(45):
                m1 = al + (ar - al) / 3.0
                m2 = ar - (ar - al) / 3.0
                f1 = float(np.max(c + m1 * g))
                f2 = float(np.max(c + m2 * g))
                if f1 < f2:
                    ar = m2
                else:
                    al = m1
            bm = 0.5 * (al + ar)
            f0 = float(np.max(c))
            fb = float(np.max(c + bm * g))
            flo = float(np.max(c + dlo * g))
            fhi = float(np.max(c + dhi * g))
            cands = [(0.0, f0), (bm, fb), (dlo, flo), (dhi, fhi)]
            bd, bf = min(cands, key=lambda x: x[1])
            if abs(bd) > 1e-12 and bf < f0 - 1e-14:
                h[i] = hi + bd
                c = c + bd * g
                if float(np.max(c)) * dx < best_c:
                    best_c = float(np.max(c)) * dx
                any_change = True
        if not any_change:
            break
        h = project_feasible(h, n)
        c = compute_corr(h, n)
        if float(np.max(c)) * dx < best_c:
            best_c = float(np.max(c)) * dx
    return h, best_c

def lse_minimize(h, n, beta, deadline, pen=1e4):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        hh = np.clip(x, 0.0, 1.0)
        c = np.correlate(hh, 1.0 - hh, mode="full") * (2.0 / n)
        c = np.maximum(c, 0.0)
        m = c.max()
        y = beta * (c - m)
        e = np.exp(np.minimum(y, 0.0))
        s = e.sum()
        if s <= 0 or not np.isfinite(s):
            return 1e6, np.zeros(n)
        F = m + np.log(s) / beta
        R = (e / s)[::-1]
        v = 1.0 - hh
        A = np.convolve(R, v, "full")[n-1:2*n-1]
        B = np.convolve(e / s, hh, "full")[n-1:2*n-1]
        g = (2.0 / n) * (A - B)
        dev = hh.sum() - 0.5 * n
        F += 0.5 * pen * dev * dev
        g += pen * dev
        return float(F), g
    _stop_t = deadline
    try:
        res = minimize(obj, h.copy(), method="L-BFGS-B", jac=True,
                       bounds=bounds, callback=stop_cb,
                       options={"maxiter": 1000000, "maxls": 40, "ftol": 1e-14, "gtol": 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
