```python
def lse_objective_and_grad(h, alpha):
    """Compute LSE approximation of max(c) and its gradient w.r.t. h"""
    c = np.correlate(h, 1-h, mode='full')
    # LSE: (1/alpha) * log(sum(exp(alpha*c)))
    c_max = c.max()
    exp_c = np.exp(alpha*(c - c_max))
    log_sum = c_max + np.log(exp_c.sum()) / alpha
    w = exp_c / exp_c.sum()  # weights for the max
    # Gradient w.r.t. h
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        # ∂c[k]/∂h[i] for all i
        # = (1-h)[i+k-n+1]·[0≤i+k-n+1<n] - h[i-k+n-1]·[0≤i-k+n-1<n]
        idx1 = np.arange(0, n) + k - n + 1
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = np.arange(0, n) - k + n - 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        g = np.zeros(n)
        g[mask1] += (1-h)[idx1[mask1]]
        g[mask2] -= h[idx2[mask2]]
        grad += w[k] * g
    return log_sum, grad / alpha
```

```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection."""
    lo = h.min() - target_sum
    hi = h.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_grad(h, alpha):
    """Compute LSE value and gradient w.r.t. h."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    s = exp_c.sum()
    val = (c_max + np.log(s) / alpha)
    w = exp_c / s
    # Gradient
    corr1 = np.correlate(1.0 - h, w, mode='full')
    first = corr1[3*n-3 - np.arange(n)]
    conv2 = np.convolve(h, w, mode='full')
    second = conv2[n-1 : 2*n-1]
    grad = (first - second) / alpha
    return val, grad

def lse_pg(h, alpha, n_iter, lr_init, target_sum):
    """Projected gradient on LSE."""
    h = h.copy()
    lr = lr_init
    for it in range(n_iter):
        val, grad = lse_grad(h, alpha)
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)

            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            val_new, _ = lse_grad(h_new, alpha)
        h = h_new
    return h
```

Actually, for the binary case, there's a simpler approach: for each pair (i, j) where h[i]=0 and h[j]=1, compute the change in max correlation if we swap them. The change is:
Δc[k] = (1-h)[i+k-n+1]·[valid] - h[i-k+n-1]·[valid] - (1-h)[j+k-n+1]·[valid] + h[j-k+n-1]·[valid]

But this is O(n) per pair and O(n⁴) total (n² pairs × n per pair × n for max). For n=400, that's 64 billion operations. Way too slow.

The previous program used a smarter approach: maintain the correlation vector and update it incrementally. When we swap h[i] and h[j]:
- h[i] goes from 0 to 1: adds (1-h) to the relevant positions
- h[j] goes from 1 to 0: subtracts (1-h) from the relevant positions

The delta to the correlation vector can be computed in O(n) time. Then finding the new max is O(n). So each swap candidate costs O(n), and with n/2 zeros and n/2 ones, one full sweep is O(n³) ≈ 64M for n=400. That's doable in a few seconds.

```python
def binary_search(h_init, n, deadline):
    """Greedy binary local search."""
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    c_vec = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = c_vec.max()
    best_h = h.copy()
    best_max = cur_max
    ones = set(idx[:n//2].tolist())
    zeros = set((idx[n//2:]).tolist())
    
    for sweep in range(50):
        if time.time() > deadline:
            break
        improved = False
        # Try each zero position
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            # Compute delta if h[i] becomes 1 (and some h[j] becomes 0)
            # But we need to try all j... that's O(n) per i
            # Better: for each i, find the best j
            best_j = -1
            best_new_max = cur_max + 1
            for j in ones:
                # Compute new max after swapping i and j
                # Delta to correlation:
                delta = np.zeros(2*n-1)
                # h[i]: 0 -> 1
                # Adds (1-h) at positions i+k-n+1 for valid k
                # i.e., c[k] += (1-h)[i+k-n+1] for 0<=i+k-n+1<n
                k1_lo = max(0, n-1-i)
                k1_hi = min(2*n-2, 2*n-2-i+1-1)  # hmm
                # Actually: 0<=i+k-n+1<n → n-1-i<=k<=n-1-i+n-1=2n-2-i
                # k in [max(0,n-1-i), min(2n-2, 2n-2-i)]
                lo1 = max(0, n-1-i)
                hi1 = min(2*n-2, 2*n-2-i)
                if lo1 <= hi1:
                    delta[lo1:hi1+1] += (1-h)[i+lo1-n+1 : i+hi1-n+2]
                
                # h[j]: 1 -> 0
                # Subtracts (1-h) at positions j+k-n+1
                lo2 = max(0, n-1-j)
                hi2 = min(2*n-2, 2*n-2-j)
                if lo2 <= hi2:
                    delta[lo2:hi2+1] -= (1-h)[j+lo2-n+1 : j+hi2-n+2]
                
                new_c = c_vec + delta
                new_max = new_c.max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_j = j
                    best_delta = delta.copy()
            
            if best_j >= 0 and best_new_max < cur_max - 1e-12:
                h[i] = 1.0
                h[best_j] = 0.0
                c_vec += best_delta
                cur_max = best_new_max
                ones.remove(best_j)
                zeros.remove(i)
                ones.add(i)
                zeros.add(best_j)
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break  # Restart sweep
    
        if not improved:
            break
    return best_h, best_max * dx
```

```python
def pg_solve(h, target_sum, val_grad_fn, n_iter, lr, deadline):
    h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        val, grad = val_grad_fn(h)
        if np.allclose(grad, 0, atol=1e-12):
            break
        step_lr = lr
        for _ in range(25):
            h_new = project_feasible(h - step_lr * grad, target_sum)
            val_new, _ = val_grad_fn(h_new)
            if val_new <= val + 1e-14:
                break
            step_lr *= 0.5
        h = h_new
        lr = max(step_lr * 1.3, lr * 0.85)
    return h

def sym_coord_descent(h, target_sum, alpha, n_sweeps, deadline):
    """Paired symmetric coordinate descent: change h[i]+h[n-1-i] keeping sum."""
    n = len(h)
    half = n // 2
    h = h.copy()
    h_sym = (h + h[::-1]) / 2.0
    h_sym = project_feasible(h_sym, target_sum)
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(half):
            if time.time() > deadline:
                break
            j = n - 1 - i
            if i == j:
                continue
            lo_d = -h_sym[i]
            hi_d = 1.0 - h_sym[j]
            if hi_d - lo_d < 1e-12:
                continue
            best_d = 0.0
            best_v = np.inf
            for _ in range(30):
                mid = (lo_d + hi_d) / 2.0
                h_test = h_sym.copy()
                h_test[i] += mid
                h_test[j] -= mid
                c_test = np.correlate(h_test, 1.0 - h_test, mode='full')
                c_max = c_test.max()
                exp_c = np.exp(alpha * (c_test - c_max))
                v = c_max + np.log(exp_c.sum()) / alpha
                if v < best_v:
                    best_v = v
                    best_d = mid
                # Gold
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                if (hi_d - mid) > gr * (mid - lo_d):
                    lo_d = mid
                else:
                    hi_d = mid
                if hi_d - lo_d < 1e-10:
                    break
            h_sym[i] += best_d
            h_sym[j] -= best_d
    h_sym = project_feasible(h_sym, target_sum)
    return h_sym

def barrier_push(h, target_sum, c_cur, deadline):
    h = h.copy()
    c = c_cur
    for i in range(15):
        if time.time() > deadline - 3:
            break
        t = c * (1.0 - 0.004 * (i + 1))
        val_grad_fn = lambda hh: barrier_val_grad(hh, t * (2.0 / len(hh)))
        h = pg_solve(h, target_sum, val_grad_fn, 200, 0.01, deadline - 3)
        c_new = compute_c5(h)
        if c_new < c:
            c = c_new
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    target_sum = n / 2.0
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Phase 1: LSE + PG increasing alpha (40% of time)
    p1_end = t0 + 0.40 * (budget_s - 10)
    alphas = [30, 60, 120, 300, 700, 1500, 3000, 7000, 15000, 30000, 60000, 120000]
    for alpha in alphas:
        if time.time() > p1_end - 8:
            break
        n_iter = 400 if alpha < 2000 else 250
        lr = 1.0 / (alpha * 6.0)
        val_grad_fn = lambda hh: lse_val_grad(hh, alpha)
        h = pg_solve(h, target_sum, val_grad_fn, n_iter, lr, p1_end - 8)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Symmetric coordinate descent (20% of time)
    p2_end = max(p1_end, t0 + 0.60 * (budget_s - 10))
    h_sym = (best_h + best_h[::-1]) / 2.0
    h_sym = project_feasible(h_sym, target_sum)
    for alpha in [5000, 20000, 80000, 200000]:
        if time.time() > p2_end - 5:
            break
        h_sym = sym_coord_descent(h_sym, target_sum, alpha, 8, p2_end - 5)
        c = compute_c5(h_sym)
        if c < best_c:
            best_c, best_h = c, h_sym.copy()

    # Phase 3: Barrier push on best (15% of time)
    p3_end = max(p2_end, t0 + 0.75 * (budget_s - 10))
    h_bar, c_bar = barrier_push(best_h, target_sum, best_c, p3_end - 5)
    if c_bar < best_c:
        best_c, best_h = c_bar, h_bar.copy()

    # Phase 4: Diversified restarts (15% of time)
    p4_end = max(p3_end, t0 + 0.90 * (budget_s - 10))
    x = np.linspace(0, 2, n)
    perts = [
        0.02 * rng.randn(n),
        0.03 * np.sin(2 * np.pi * 3.0 * x + 0.5),
        0.015 * np.cos(2 * np.pi * 8.0 * x + 1.0),
        0.025 * rng.randn(n),
        0.01 * np.sin(2 * np.pi * 15.0 * x),
    ]
    for pert in perts:
        if time.time() > p4_end - 20:
            break
        h_r = project_feasible(best_h + pert, target_sum)
        for alpha in [5000, 30000, 100000]:
            if time.time() > p4_end - 20:
                break
            val_grad_fn = lambda hh: lse_val_grad(hh, alpha)
            h_r = pg_solve(h_r, target_sum, val_grad_fn, 150, 1.0 / (alpha * 6.0), p4_end - 20)
        c_r = compute_c5(h_r)
        if c_r < best_c:
            best_c, best_h = c_r, h_r.copy()

    # Phase 5: Final polish (remaining time)
    h_f, c_f = barrier_push(best_h, target_sum, best_c, deadline - 5)
    if c_f < best_c:
        best_c, best_h = c_f, h_f.copy()
    for alpha in [100000, 300000]:
        if time.time() > deadline - 4:
            break
        val_grad_fn = lambda hh: lse_val_grad(hh, alpha)
        h_f = pg_solve(best_h, target_sum, val_grad_fn, 200, 1.0 / (alpha * 6.0), deadline - 4)
        c_f = compute_c5(h_f)
        if c_f < best_c:
            best_c, best_h = c_f, h_f.copy()

    best_h = project_feasible(best_h, target_sum)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
```
