def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_onto_set(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via bisection."""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_smooth(h, n, alpha, maxiter, step, time_limit):
    """Nesterov accelerated proximal gradient on smoothed max correlation."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    h_prev = h.copy()
    t = 1.0
    c_max_init = float(np.max(np.correlate(h, 1-h, mode='full')))
    
    for it in range(maxiter):
        if time.time() > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        
        # Compute correlation
        c = np.correlate(y, 1.0 - y, mode='full')
        
        # Softmax weights
        c_shifted = c - c.max()
        exp_c = np.exp(alpha * c_shifted)
        w = exp_c / exp_c.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        k = np.arange(-(n-1), n)
        for j in range(n):
            valid1 = (j + k >= 0) & (j + k < n)
            if valid1.any():
                idx1 = j + k[valid1]
                grad[j] += np.dot(w[valid1], 1.0 - y[idx1])
            valid2 = (j - k >= 0) & (j - k < n)
            if valid2.any():
                idx2 = j - k[valid2]
                grad[j] -= np.dot(w[valid2], y[idx2])
        grad *= alpha * dx
        
        # Proximal step
        h_new = project_onto_set(y - step * grad, n)
        h_prev = h
        h = h_new
        t = t_new
    
    return project_feasible(h, n)

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    # Compute gradient matrix efficiently
    # G[j, m] = ∂C[m]/∂h[j]
    # C[m] = sum_i h[i]*(1-h[i+k]) where k = m-(n-1)
    # ∂C[m]/∂h[j] = (1-h[j+k]) if valid - h[j-k] if valid
    G = np.zeros((n, 2*n-1))
    k = np.arange(-(n-1), n)
    for j in range(n):
        valid1 = (j + k >= 0) & (j + k < n)
        if valid1.any():
            G[j, valid1] += (1.0 - h[j + k[valid1]])
        valid2 = (j - k >= 0) & (j - k < n)
        if valid2.any():
            G[j, valid2] -= h[j - k[valid2]]
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
        h_try = h + lam * d
        h_try = np.clip(h_try, 0.0, 1.0)

        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def make_periodic(n, period, pattern):

import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_onto_set(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_gradient(y, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = np.arange(-(n - 1), n)
    for j in range(n):
        v1 = (j + k >= 0) & (j + k < n)
        if v1.any():
            idx = j + k[v1]
            grad[j] += np.dot(w[v1], 1.0 - y[idx])
        v2 = (j - k >= 0) & (j - k < n)
        if v2.any():
            idx = j - k[v2]
            grad[j] -= np.dot(w[v2], y[idx])
    return grad * dx

def nesterov_smooth(h, n, alpha, maxiter, step, deadline):
    dx = 2.0 / n
    h = h.astype(float).copy()
    h_prev = h.copy()
    t = 1.0
    for it in range(maxiter):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        c = np.correlate(y, 1.0 - y, mode='full')
        cs = c - c.max()
        exp_c = np.exp(alpha * cs)
        w = exp_c / exp_c.sum()
        grad = smooth_gradient(y, n, w)
        grad *= alpha
        h_new = project_onto_set(y - step * grad, n)
        h_prev = h
        h = h_new
        t = t_new
    return project_feasible(h, n)

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = np.zeros((n, 2 * n - 1))
    k = np.arange(-(n - 1), n)
    for j in range(n):
        v1 = (j + k >= 0) & (j + k < n)
        if v1.any():
            G[j, v1] += (1.0 - h[j + k[v1]])
        v2 = (j - k >= 0) & (j - k < n)
        if v2.any():
            G[j, v2] -= h[j - k[v2]]
    delta = cp.Variable(n)
    t = cp.Variable()
    cons = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    solved = False
    for sol in ['GLOP', 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()

        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc
