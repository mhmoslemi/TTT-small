def grad_from_w(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def _compute(h, n, mode, beta, gap, p):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if mode == 'smax':
        f, w = smooth_max(c, beta)
    else:  # topk
        cmax = float(np.max(c))
        thr = cmax - gap
        active = np.where(c >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        vals = np.maximum(c[active] - thr, 0.0)
        f = float(np.sum(vals ** p))
        w = np.zeros(len(c))
        w[active] = p * vals ** (p - 1)
    g = grad_from_w(h, w, n, dx)
    return f, g

                return h, erdos_c5(h, n)
            hi0, hj0 = h[i], h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-13:
                continue
            best_d, best_c = 0.0, cur
            for t in np.linspace(lo, hi, 9):
                c = pair_eval(t, h, i, j, n)
                if c < best_c:
                    best_c, best_d = c, t
            if best_c < cur - 1e-13:
                h[i] += best_d; h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)
                cur, improved = best_c, True
        if not improved:
            break
    return h, cur

def resample(h, n, nh):
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nh)
    return project_box_sum(np.interp(y, x, h), 0.5 * nh)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 10.0)
    deadline = t0 + total
    try:

        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n < 2:
        n = 400
        h = np.full(n, 0.5)
    h = project_box_sum(h, 0.5 * n)

            if c < best_c - 1e-15:

    if time.time() < deadline - 30.0:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 15.0, seed, max_passes=6)

    if time.time() < deadline - 12.0:

        if c < best_c - 1e-15:

    rng = np.random.RandomState(int(seed) + 31)
