            if fm < best_candidate:
                fp = int(np.argmin(row_max))
                best_candidate = fm
                best_p = p0 + fp // O
                best_q = fp % O
        if best_p < 0:
            stall += 1
            if stall > 300:
                break
            kick = min(4, Z // 2)
            kz = np.random.choice(Z, kick, replace=False)
            ko = np.random.choice(O, kick, replace=False)
            x[zeros[kz]] = 1.0
            x[ones[ko]] = 0.0
            c = np.correlate(x, 1.0 - x, mode='full').astype(float)
            cur_max = c.max()
            stall = 0
            continue
        x[zeros[best_p]] = 1.0
        x[ones[best_q]] = 0.0
        c += A[best_p, :] + B[best_q, :]
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_h = x.copy()
            stall = 0
    return best_h, best_max * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP
    p1 = t0 + 0.18 * total
    alphas = [500, 5000, 50000, 500000, 5e6, 5e7]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p1 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 2000, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 2: Vectorized binary local search
    p2 = t0 + 0.70 * total
    x_bin = binarize(best_h, n)
    xb, cb = binary_local_search_vec(x_bin, n, p2, max_sweeps=50000, chunk=16)
    if cb < best_c:
        best_c, best_h = cb, xb.copy()
    # Phase 3: Continuous polish
    p3 = t0 + 0.84 * total
    h = best_h.copy()
    for alpha in [1e5, 1e6, 1e7]:
        if time.time() > p3 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 1000, 1e-14)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 4: Lower resolution n=200
    p4 = t0 + 0.94 * total
    if time.time() < p4 - 30:
        nh = 200
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh = slsqp_optimize(hh, nh, 1e5, 500, 1e-13)
        ch = compute_c5(hh, nh)
        best2_h, best2_c = hh.copy(), ch
        xb2 = binarize(hh, nh)
        xb2, cb2 = binary_local_search_vec(xb2, nh, p4 - 5, max_sweeps=20000, chunk=16)
        if cb2 < best2_c:
            best2_h, best2_c = xb2.copy(), cb2
        if best2_c < best_c:
            best_c, best_h, n = best2_c, best2_h, nh
    # Final polish
    if time.time() < dl - 5:
        h = slsqp_optimize(best_h, n, 5e6, 800, 1e-14)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
