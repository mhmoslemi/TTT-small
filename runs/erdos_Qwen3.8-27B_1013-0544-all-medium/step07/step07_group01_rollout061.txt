def solve_single_lag(n, j):
    """Solve KKT for single active lag j."""
    # Unknowns: x[0], ..., x[n-1], lambda → total n+1
    # Equations:
    # For k=0..j-1: x[j+k] + lambda = 0  → coefficients: x[j+k]=1, lambda=1
    # For k=0..j-1: x[n-2j+k] - lambda = 1 → x[n-2j+k]=-1, lambda=1, rhs=1
    #   Wait: x[n-2j+k] = 1-lambda → x[n-2j+k] + lambda = 1
    # For i=j..n-1-j: x[i-j] + x[i+j] + lambda = 1
    # Sum: sum(x) = n/2
    
    # Build A (n+1) x (n+1), b (n+1)
    A = np.zeros((n+1, n+1))
    b = np.zeros(n+1)
    
    row = 0
    # Boundary equations 1: x[j+k] + lambda = 0
    for k in range(j):
        A[row, j+k] = 1.0
        A[row, n] = 1.0  # lambda coefficient
        b[row] = 0.0
        row += 1
    
    # Boundary equations 2: x[n-2j+k] + lambda = 1
    for k in range(j):
        A[row, n-2j+k] = 1.0
        A[row, n] = 1.0
        b[row] = 1.0
        row += 1
    
    # Interior equations: x[i-j] + x[i+j] + lambda = 1
    for i in range(j, n-j):
        A[row, i-j] = 1.0
        A[row, i+j] = 1.0
        A[row, n] = 1.0
        b[row] = 1.0
        row += 1
    
    # Sum constraint
    A[row, :n] = 1.0
    A[row, n] = 0.0
    b[row] = n / 2.0
    row += 1
    
    # Solve
    try:
        sol = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A, b, rcond=None)[0]
    
    x = sol[:n]
    lam = sol[n]
    
    # Check feasibility
    if x.min() < -1e-6 or x.max() > 1 + 1e-6:
        return None, None
    
    x_clipped = np.clip(x, 0, 1)
    # Project to satisfy sum constraint
    x_final = project_feasible(x_clipped, n)
    
    c = compute_c5(x_final, n)
    return x_final, c

    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h

            if s < 1e-13:

            if s < 1e-13:
                break
            h = h + diff / s * h

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Gradient of dx * logsumexp(corr, alpha) w.r.t. h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    sh = corr - m
    exp_sh = np.exp(alpha * sh)
    ps = exp_sh.sum()
    if ps < 1e-30:
        return np.zeros(n), m * dx
    p = exp_sh / ps
    f = (m + np.log(ps) / alpha) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        lag = k - (n - 1)
        w = p[k] * dx
        if 0 <= lag < n:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        elif -n < lag < 0:
            ll = -lag
            grad[:n - ll] += w * (1.0 - h[ll:])
            grad[ll:] -= w * h[:n - ll]
    return grad, f

def kkt_single_lag(n, j):
    """Try to solve KKT for single active lag j."""
    # Unknowns: x[0]..x[n-1], lambda → n+1 total
    # Build system A * [x; lambda] = b
    n_eq = 0  # count equations
    # Boundary 1: x[j+k] + lambda = 0  (j equations)
    # Boundary 2: x[n-2j+k] + lambda = 1  (j equations)
    # Interior: x[i-j] + x[i+j] + lambda = 1  (n-2j equations for i=j..n-1-j)
    # Sum: sum(x) = n/2  (1 equation)
    total_eq = j + j + (n - 2 * j) + 1  # = n + 1
    
    # But some equations might be redundant or conflicting.
    # Use lstsq to handle this.
    A = np.zeros((total_eq, n + 1))
    b = np.zeros(total_eq)
    
    row = 0
    for k in range(j):
        A[row, j + k] = 1.0
        A[row, n] = 1.0
        b[row] = 0.0
        row += 1
    
    for k in range(j):
        A[row, n - 2 * j + k] = 1.0
        A[row, n] = 1.0
        b[row] = 1.0
        row += 1
    
    for i in range(j, n - j):
        A[row, i - j] = 1.0
        A[row, i + j] = 1.0
        A[row, n] = 1.0
        b[row] = 1.0
        row += 1
