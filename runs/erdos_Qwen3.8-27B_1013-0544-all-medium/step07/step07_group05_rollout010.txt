```python
import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def softmax_w(c, beta):
    m = float(np.max(c))
    z = beta * (c - m)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    s = float(ew.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ew / s

def smooth_max_val(c, beta):
    m = float(np.max(c))
    z = -beta * (c - m)
    return m - float(np.log(float(np.sum(np.exp(z))))) / beta

def compute_grad(h, w, dx):
    g = 1.0 - 2.0 * h
    N = len(h)
    full = np.convolve(g, w, mode='full')
    return dx * full[N - 1:2 * N - 1]

def obj_grad(x, beta, dx, target, rho):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    w = softmax_w(c, beta)
    f = smooth_max_val(c, beta)
    g = compute_grad(x, w, dx)
    res = float(x.sum() - target)
    f += 0.5 * rho * res * res
    g = g + rho * res * np.ones(n)
    return f, g

def obj_only(x, beta, dx, target, rho):
    f, _ = obj_grad(x, beta, dx, target, rho)
    return f

def jac_only(x, beta, dx, target, rho):
    _, g = obj_grad(x, beta, dx, target, rho)
    return g

def refine_lbfgs(h, beta, rho, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    dx = 2.0 / n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    mx = int(min(maxiter, max(15, int(rem / 3.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(obj_only, h, jac=jac_only, method='L-BFGS-B',
                          bounds=bounds, args=(beta, dx, target, rho),
                          options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxls': 50, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

def active_lag_rebalance(h, n, deadline, K=8):
    """Try to equalize top-K lags by shifting mass between contributing indices."""
    h = np.asarray(h, dtype=float).copy()
    dx = 2.0 / n
    target = 0.5 * n
    c_all = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c_all)[::-1][:K]
    cur_max = float(c_all[top_idx[0]])
    
    for iteration in range(20):
        if time.time() > deadline:
            break
        c_all = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c_all)[::-1][:K]
        top_vals = c_all[top_idx]
        spread = float(top_vals.max() - top_vals.min())
        if spread < 1e-7:
            break
        # For each top lag, find which index pair swap would most reduce it
        best_improve = 0.0
        best_swap = None
        mid = n - 1
        for p in top_idx:
            k = int(p) - mid
            # c[p] = sum_i h[i]*(1-h[i+k])*dx for valid i (k>=0) or symmetric
            if k >= 0:
                valid = np.arange(n - k)
            else:
                valid = np.arange(-k, n)
            if len(valid) < 2:
                continue
            hi_idx = valid[np.argmax(h[valid])]
            lo_idx = valid[np.argmin(h[valid])]
            if hi_idx == lo_idx:
                continue
            delta = 0.01
            for sign in (1, -1):
                hh = h.copy()
                if sign > 0:
                    hh[hi_idx] = np.clip(h[hi_idx] - delta, 0, 1)
                    hh[lo_idx] = np.clip(h[lo_idx] + delta, 0, 1)
                else:
                    hh[hi_idx] = np.clip(h[hi_idx] + delta, 0, 1)
                    hh[lo_idx] = np.clip(h[lo_idx] - delta, 0, 1)
                hh = project_box_sum(hh, target)
                c_test = np.correlate(hh, 1.0 - hh, mode='full') * dx
                new_max = float(c_test[top_idx].max())
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (hi_idx, lo_idx, delta, sign, hh)
        if best_swap is None or best_improve < 1e-10:
            break
        h = best_swap[4]
    return h, erdos_c5(h, n)

def pair_polish(h, n, deadline, seed):
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    dx = 2.0 / n
    target = 0.5 * n
    rng = np.random.RandomState(int(seed))
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    m = min(20, max(2, len(order)))
    top = order[:m] if order.size >= 2 else np.arange(min(n, 30))
    for sweep in range(4):
        if time.time() > deadline - 2.0:
            break
        improved = False
        pairs = [(int(top[a]), int(top[b])) for a in range(len(top)) for b in range(a + 1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 2.0:
                break
            hi0, hj0 = h[i], h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-13:
                continue
            def line_obj(d):
                a = hi0 + d
                b = hj0 - d
                if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
                    return 1e100
                hh = h.copy()
                hh[i] = a
                hh[j] = b
                return erdos_c5(hh, n)
            try:
                rs = so.minimize_scalar(line_obj, bounds=(lo, hi), method='bounded',
                                       options={'maxiter': 30, 'xatol': 1e-11})
                if rs.x is not None and rs.fun < cur - 1e-13:
                    h[i] = hi0 + float(rs.x)
                    h[j] = hj0 - float(rs.x)
                    h = np.clip(h, 0.0, 1.0)
                    h = project_box_sum(h, target)
                    cur = float(rs.fun)
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, erdos_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n < 2:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n > 800:
        nold = n
        n = 800
        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, nold), h)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    end1 = t0 + 0.30 * total
    end2 = t0 + 0.55 * total
    end3 = t0 + 0.80 * total
    end4 = deadline - 4.0

    # Phase 1: L-BFGS-B with increasing beta
    lb_configs = [
        (200.0, 1e6, 100),
        (1000.0, 1e6, 90),
        (5000.0, 1e6, 80),
        (20000.0, 1e6, 70),
        (100000.0, 1e6, 60),
        (500000.0, 1e6, 50),
        (2000000.0, 1e6, 40),
        (10000000.0, 1e6, 35),
    ]
    for beta, rho, mx in lb_configs:
        if time.time() > end1 - 3.0:
            break
        cand, c = refine_lbfgs(best_h, beta, rho, mx, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 2: Active lag rebalancing
    if time.time() < end2 - 5.0:
        cand, c = active_lag_rebalance(best_h, best_n, end2 - 3.0, K=10)
        if c < best_c - 1e-14:
            best_h = cand.copy()
            best_c = c
            # Refine after rebalancing
            for beta in (5000.0, 100000.0):
                if time.time() > end2 - 3.0:
                    break
                cand, c = refine_lbfgs(best_h, beta, 1e6, 50, end2)
                if c < best_c - 1e-15:
                    best_h = cand.copy()
                    best_c = c

    # Phase 3: More L-BFGS-B at very high beta
    for beta in (50000.0, 200000.0, 1000000.0, 5000000.0):
        if time.time() > end3 - 5.0:
            break
        cand, c = refine_lbfgs(best_h, beta, 1e6, 45, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 4: Pair polish (aggressive)
    if time.time() < end4 - 20.0:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 10.0, seed)

    # Final L-BFGS-B sweeps
    if time.time() < end4 - 12.0:
        cand, c = refine_lbfgs(best_h, 1000000.0, 1e6, 30, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    if time.time() < end4 - 6.0:
        cand, c = pair_polish(best_h, best_n, end4 - 2.0, seed + 1)
        if c < best_c:
            best_h = cand.copy()
            best_c = c

    # Try small perturbation + refine
    rng = np.random.RandomState(int(seed) + 99)
    if time.time() < end4 - 8.0:
        for sig in (0.01, 0.03, 0.08):
            if time.time() > end4 - 4.0:
                break
            hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
            hh, c = refine_lbfgs(hh, 200000.0, 1e6, 30, deadline)
            if c < best_c - 1e-15:
                best_h = hh.copy()
                best_c = c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
```
