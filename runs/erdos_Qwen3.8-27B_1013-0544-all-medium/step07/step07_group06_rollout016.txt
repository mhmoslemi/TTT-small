```python
def binary_2swap(h, n, deadline, n_iters=50000):
    # h is binary with sum = n/2
    ones = np.nonzero(h == 1)[0]
    zeros = np.nonzero(h == 0)[0]
    out = np.correlate(h, 1-h, mode="full").astype(float)
    M = out.max()
    besth = h.copy()
    bestM = M
    
    rng = np.random.RandomState(0)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Pick two 1s and two 0s
        a, b = rng.choice(ones, 2, replace=False)
        c, d = rng.choice(zeros, 2, replace=False)
        
        # Compute change in correlation
        # For position p being flipped from 1 to 0: contributes -(1-h[p+k-(n-1)]) to out[k]
        # For position p being flipped from 0 to 1: contributes +(1-h[p+k-(n-1)]) to out[k]
        
        # The influence of flipping position p (from 1 to 0):
        # out[k] changes by -(1 - h[p+k-(n-1)]) for valid k
        # Valid k: n-1-p <= k <= n-1+p
        
        # Let's compute the delta for the full output
        delta = np.zeros(2*n-1)
        
        for p, sign in [(a, -1), (b, -1), (c, 1), (d, 1)] :
            lo = max(0, n-1-p)
            hi = min(2*n-2, n-1+p)
            for k in range(lo, hi+1):
                j = k - (n-1) + p  # index into h
                if sign == -1:  # flipping from 1 to 0
                    delta[k] -= (1.0 - h[j])
                else:  # flipping from 0 to 1
                    delta[k] += (1.0 - h[j])
        
        new_max = (out + delta).max()
        if new_max < M:
            h[a] = 0; h[b] = 0; h[c] = 1; h[d] = 1
            out = out + delta
            M = new_max
            if M < bestM:
                bestM = M
                besth = h.copy()
            ones = np.nonzero(h == 1)[0]
            zeros = np.nonzero(h == 0)[0]
    
    return besth, bestM
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_cb(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(np.min(a)) - 1.0, float(np.max(a)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15: break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15: break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            h[int(np.argmax(1.0 - h))] = min(1.0, h[int(np.argmax(1.0-h))] + diff)
        else:
            h[int(np.argmax(h))] = max(0.0, h[int(np.argmax(h))] + diff)
    return np.clip(h, 0.0, 1.0)

def top_n_binary(h, n):
    hb = np.zeros(n)
    idx = np.argsort(-h)[:n//2]
    hb[idx] = 1.0
    return hb

def lse_obj_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.full_like(c, 1.0 / (2*n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(P, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_fun(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_obj_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def variants(h, n):
    h = project_feasible(h, n)
    out = [h, h[::-1], 1.0-h[::-1], 0.5*(h+h[::-1]), 0.5*(h+1.0-h[::-1])]
    if n % 2 == 0:
        out.append(top_n_binary(h, n))
    return [project_feasible(x, n) for x in out]

def polish(h, n, betas, pen, deadline, stage_t):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 6.0: break
        sdl = min(deadline - 4.0, now + max(2.0, min(stage_t, 0.5*max(2.0, deadline-now))))
        _stop_time = sdl
        try:
            res = minimize(lbfgs_fun, work, method="L-BFGS-B", jac=True,
                          bounds=bounds, args=(n, float(beta), float(pen)),
                          callback=stop_cb,
                          options={"maxiter":500000,"maxls":30,"ftol":1e-14,"gtol":1e-9})
            x = res.x if res is not None and hasattr(res,"x") else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc: bc = cw; best = work.copy()
        for cand in variants(work, n):
            cc = compute_c5(cand, n)
            if cc < bc: bc = cc; best = cand.copy()
        if bc < cw - 1e-12: work = best.copy()
    return best, bc

def bin_1swap(h, n, deadline, max_sweeps=50, chunk=2048):
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0-h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    besth, bestM = h.copy(), M
    for _ in range(max_sweeps):
        if time.time() > deadline: break
        zeros = np.nonzero(h==0)[0]
        ones = np.nonzero(h==1)[0]
        if zeros.size==0 or ones.size==0: break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        l = np.arange(-(n-1), n, dtype=int)
        for start in range(0, zi.size, chunk):
            if time.time() > deadline: break
            end = min(start+chunk, zi.size)
            i = zi[start:end]; j = oj[start:end]
            rows = i[:,None] + l[None,:]
            valid1 = (rows>=0)&(rows<n)
            f = np.where(valid1, rows, 0)
            base_a = np.where(valid1, 1.0-h[f], 0.0).sum(axis=1)
            rows2 = j[:,None] + l[None,:]
            valid2 = (rows2>=0)&(rows2<n)
            g2 = np.where(valid2, rows2, 0)
            base_b = np.where(valid2, 1.0-h[g2], 0.0).sum(axis=1)
            d = base_a - base_b
            d2 = n-1+(j-i)
            d[d2>=0] -= 2.0  # correction for diagonal
            mx = d + corr[i,:]-corr[j,:]
            mx -= corr[None,n-1]  # not needed, let me use direct
            # Simpler: just compute new corr max for each candidate
            # This is getting complex, use the base matrix approach
            break  # fall through to simpler method
        # Use the proven base matrix approach
        base = np.zeros((n, 2*n-1), dtype=float)
        for p in range(n):
            f = p + l
            mf = (f>=0)&(f<n)
            base[p,mf] = 1.0 - h[f[mf]]
            b = p - l
            mb = (b>=0)&(b<n)
            base[p,mb] -= h[b[mb]]
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        for start in range(0, zi.size, chunk):
            if time.time() > deadline: break
            end = min(start+chunk, zi.size)
            i = zi[start:end]; j = oj[start:end]
            arr = base[i] - base[j]
            arr += corr[None,:]
            arr[:,n-1] -= 2.0
            d1 = n-1+(j-i)
            arr[:,d1] += 1.0
            d2 = n-1+(i-j)
            arr[:,d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M-1+1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv, jv = int(i[loc]), int(j[loc])
                h[iv]=1.0; h[jv]=0.0
                corr = arr[loc].copy()
                M = int(round(float(corr.max())))
                if M < bestM: bestM=M; besth=h.copy()
                improved = True
                break
        if not improved: break
    return besth, bestM*(2.0/float(n))

def bin_2swap(h, n, deadline, n_iters=30000, seed=0):
    h = h.copy().astype(float)
    out = np.correlate(h, 1.0-h, mode="full").astype(float)
    M = float(out.max())
    bestM = M; besth = h.copy()
    rng = np.random.RandomState(seed)
    l = np.arange(-(n-1), n, dtype=int)
    for it in range(n_iters):
        if time.time() > deadline: break
        if it % 500 == 0:
            ones = np.nonzero(h==1)[0]
            zeros = np.nonzero(h==0)[0]
            if ones.size < 2 or zeros.size < 2: break
        a, b = rng.choice(ones, 2, replace=False)
        c, d = rng.choice(zeros, 2, replace=False)
        # Compute delta for 4 flips
        delta = np.zeros(2*n-1)
        for p, sgn in [(a,-1),(b,-1),(c,1),(d,1)]:
            # First term: i=p, valid when n-1-p <= k <= n-1+p
            lo1 = max(0, n-1-p); hi1 = min(2*n-2, n-1+p)
            if lo1 <= hi1:
                kr = np.arange(lo1, hi1+1)
                jr = kr - (n-1) + p
                delta[lo1:hi1+1] += sgn * (-(1.0 - h[jr]))
            # Second term: i=p-lag, valid when p <= k <= n-1+p
            lo2 = max(0, p); hi2 = min(2*n-2, n-1+p)
            if lo2 <= hi2:
                kr = np.arange(lo2, hi2+1)
                jr = p - (kr - (n-1))
                mask = (jr>=0)&(jr<n)
                if mask.any():
                    kr_m = kr[mask]; jr_m = jr[mask]
                    delta[kr_m] += sgn * h[jr_m]
        new_M = float((out + delta).max())
        if new_M < M - 1e-12:
            h[a]=0; h[b]=0; h[c]=1; h[d]=1
            out = out + delta
            M = new_M
            if M < bestM: bestM=M; besth=h.copy()
        elif it % 2000 == 0 and rng.rand() < 0.3:
            # Occasional random 1-swap to explore
            a1 = rng.choice(ones); c1 = rng.choice(zeros)
            if a1 != c1:
                delta1 = np.zeros(2*n-1)
                for p, sgn in [(a1,-1),(c1,1)]:
                    lo1=max(0,n-1-p); hi1=min(2*n-2,n-1+p)
                    if lo1<=hi1:
                        kr=np.arange(lo1,hi1+1); jr=kr-(n-1)+p
                        delta1[lo1:hi1+1]+=sgn*(-(1.0-h[jr]))
                    lo2=max(0,p); hi2=min(2*n-2,n-1+p)
                    if lo2<=hi2:
                        kr=np.arange(lo2,hi2+1); jr=p-(kr-(n-1))
                        mask=(jr>=0)&(jr<n)
                        if mask.any():
                            delta1[kr[mask]]+=sgn*h[jr[mask]]
                nm1=float((out+delta1).max())
                if nm1 < M:
                    h[a1]=1-h[a1]; h[c1]=1-h[c1]
                    out=out+delta1; M=nm1
                    if M<bestM: bestM=M; besth=h.copy()
    return besth, bestM*(2.0/float(n))
```
