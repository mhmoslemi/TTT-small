def smooth_obj(x, n, power, penalty_lambda):

    c_pos = np.maximum(c, 0)

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def proj_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(h + mid, 0.0, 1.0)
        if x.sum() < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        if interior.sum() > 0:
            x[interior] += diff / interior.sum()
    return np.clip(x, 0.0, 1.0)

def ps_obj(x, n, p, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    cp = np.maximum(c, 0.0)
    val = np.sum(cp ** p) ** (1.0 / p) * (2.0 / n)
    val += lam * (x.sum() - n / 2.0) ** 2
    return val
