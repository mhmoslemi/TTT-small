def base_build(h):
    n = h.size

def coord_delta(base, q, delta, rows):

    ba = center + (a - b)

    c[ba] += t2

    h[b] += t

    if h[b] < 0.0:

def variants(h):

    raw = [
        h,
        h[::-1],
        1.0 - h[::-1],
        0.5 * (h + h[::-1]),
        0.5 * (h + 1.0 - h[::-1]),
    ]
    return [project(x, n) for x in raw]


def lse_fg(x, n, beta):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g


def lse_obj(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_fg(h, n, beta)
    dev = h.sum() - 0.5 * float(n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g


def smooth(h0, deadline, betas, stage=30.0, pen=1e4):

    n = h0.size
    work = project(h0, n)
    best = work.copy()
    bc = c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 12.0:
            break
        dl = min(deadline - 6.0, now + max(2.0, min(stage, 0.5 * max(2.0, deadline - now))))
        _stop = dl
        try:
            res = minimize(
                lse_obj, work, method="L-BFGS-B", jac=True, bounds=bounds,
                args=(n, float(beta), float(pen)), callback=_stop_cb,
                options={"maxiter": 1000000, "maxls": 40, "ftol": 1e-13, "gtol": 1e-9}
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project(x, n)
        cw = c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
        for cand in variants(work):
            cc = c5(cand, n)
            if cc < bc:
                bc = cc
                best = cand.copy()
        if bc < cw - 1e-12:

    return best, bc

        N = zi.size
        for start in range(0, N, 4096):
            if time.time() > deadline:
                break
            end = min(start + 4096, N)
            i = zi[start:end]
            j = oj[start:end]
            arr = c[None, :] + base[i] - base[j]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            d2 = n - 1 + (i - j)
            r = np.arange(end - start)
            arr[r, d1] += 1.0
            arr[r, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= target)[0]
            if idx.size:
                mi = mx[idx]
                loc = int(idx[int(np.argmin(mi))])
                iv = int(i[loc])
                jv = int(j[loc])
                pair_apply(h, base, c, jv, iv, 1.0, rows)
                M = float(c.max())
                found = True
                if M < bestM - 1e-9:
                    bestM = M
                    best = h.copy()
                break
        if not found:
            break
        sweeps += 1
        if sweeps % 50 == 0:
            base = base_build(h)
            c = carray(h)
            M = float(c.max())
            if M < bestM - 1e-9:
                bestM = M
                best = h.copy()
    best = project(best, n)
    return best, c5(best, n)


def make_start(n, kind, rng):
    t = np.linspace(0.0, 1.0, n)
    if kind == 0:
        h = np.full(n, 0.5)
        for f in range(1, 7):
            h += 0.12 * rng.randn() * np.sin(2.0 * np.pi * (float(f) * t + rng.rand()))
    elif kind == 1:
        h = np.zeros(n)
        pos = 0
        col = 1.0
        while pos < n:
            lim = max(4, n // 20)
            l = int(rng.randint(3, lim + 1))
            h[pos:pos + l] = col
            pos += l
            col = 1.0 - col
    elif kind == 2:
        lim = max(4, n // 20)
        p = int(rng.randint(3, lim + 1))
        ph = rng.rand() * p
        x = np.arange(n) * (2.0 / float(n)) + ph
        h = (np.mod(x, p) < 0.5 * p).astype(float)
    else:
        h = rng.rand(n)
    return project(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        B = float(budget_s)
    except Exception:
        B = 1100.0
    end = t0 + max(30.0, B - 25.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad initial")
    except Exception:
        n0 = 400
        h0 = ((np.arange(n0) // 50) % 2).astype(float)

    best_n = n0
    best_h = project(h0, best_n)
    best_c = c5(best_h, best_n)

    for cand in variants(best_h):
        cc = c5(cand, best_n)
        if cc < best_c:
            best_c = cc
            best_h = cand.copy()

    if time.time() < end - 190.0:
        hh, cc = pair_polish(best_h, end - 220.0, rng, 30, 700, 200)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    if time.time() < end - 220.0:
        hh, cc = smooth(best_h, end - 260.0, [20.0, 80.0, 300.0, 1200.0, 5000.0, 20000.0], 30.0, 1e4)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    if time.time() < end - 130.0:
        hh, cc = pair_polish(best_h, end - 160.0, rng, 40, 800, 220)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    if best_n % 2 == 0 and time.time() < end - 130.0:
        tb = top_binary(best_h, best_n)
        ct = c5(tb, best_n)
        if ct < best_c + 0.02:
            hb, cb = binary_greedy(tb, end - 150.0)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    for kind in range(4):
        now = time.time()
        dl = min(end - 90.0, now + 45.0)
        if dl > now + 25.0:
            hs = make_start(best_n, kind, rng)
            hh, cc = pair_polish(hs, dl, rng, 25, 400, 120)
            if cc < best_c:
                best_c = cc
                best_h = hh.copy()

    if time.time() < end - 75.0:
        hh, cc = smooth(best_h, end - 55.0, [2000.0, 8000.0, 30000.0], 20.0, 1e4)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    if time.time() < end - 35.0:
        hh, cc = pair_polish(best_h, end - 25.0, rng, 40, 850, 220)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    for cand in variants(best_h):
        cc = c5(cand, best_n)
        if cc < best_c:
            best_c = cc
            best_h = cand.copy()

    if best_n % 2 == 0:
        tb = top_binary(best_h, best_n)
        ct = c5(tb, best_n)
        if ct < best_c:
            best_c = ct
            best_h = tb.copy()

    best_h = project(best_h, best_n)
    best_c = c5(best_h, best_n)
    return best_h, best_c, int(best_n)
