def compute_sparse_gradient(h, c, alpha, n, top_k=30):
    """Compute gradient using only top-k significant lags."""
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    s = weights.sum()
    if s < 1e-300:
        return np.zeros(n)
    weights /= s
    grad = np.zeros(n)
    # Find top-k lags
    top_idx = np.argsort(weights)[::-1][:top_k]
    for m_idx in top_idx:
        w = weights[m_idx]
        if w < 1e-10:
            break

        if k >= 0:
            grad[k:] += w * (1 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            k2 = -k
            grad[:n-k2] += w * (1 - h[k2:])
            grad[k2:] -= w * h[:n-k2]
    return alpha * grad * dx
