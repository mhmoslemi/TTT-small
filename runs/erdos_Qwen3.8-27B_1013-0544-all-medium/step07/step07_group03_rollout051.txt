So it's not an LP. But we can linearize it using the fact that for binary h, c[k] = sum_t h[t+k-(n-1)] - h[t+k-(n-1)]*h[t].

The product h[t+k-(n-1)]*h[t] is the issue. For continuous h, we can use the "McCormick envelope" to get a convex relaxation, but that might be too loose.

Alternatively, we can use a "lifting" approach: introduce variables z[t,s] = h[t]*h[s] and add constraints. But this makes the problem very large.

For now, let me stick with the approach of:
1. Continuous optimization (SLSQP with log-sum-exp) to get a good continuous solution
2. Project to binary
3. Binary local search to improve

The key improvement over the previous code:
- More aggressive binary local search with simulated annealing
- Try multiple n values
- Better use of time (allocate more time to binary search since that's where the gains are)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_binary(h, n):
    k = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:k]] = 1.0
    return hb

def slsqp_logsumexp(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        if alpha < 1e10:
            shifted = alpha * (c - m)
            sm = shifted.max()
            logsum = sm + np.log(np.exp(shifted - sm).sum())
            return m + logsum / alpha
        else:
            return m
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def binary_sa_search(h_init, n, deadline, max_iters=200000, top_k=5):
    """Binary local search with simulated annealing."""
    h = project_binary(h_init, n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = float(c.max())
    best_h = h.copy()
    best_max = cur_max
    
    m_range = np.arange(-(n-1), n)
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    # Temperature schedule
    T = 5.0  # initial temperature (in units of correlation)
    T_min = 0.1
    decay = 0.9995
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Pick a random zero and a random one
        i = np.random.choice(list(zeros))
        j = np.random.choice(list(ones))
        
        # Compute delta correlation
        delta = np.zeros(2*n - 1)
        # Point 1: h[i] 0→1, first arg = i
        k_lo = max(0, i)
        k_hi = min(2*n-2, i+n-1)
        t_vals = i + n - 1 - np.arange(k_lo, k_hi+1)
        delta[k_lo:k_hi+1] += (1.0 - h[t_vals])
        
        # Point 2: (1-h[i]) 1→0, second arg = i
        k_lo2 = max(0, n-1-i)
        k_hi2 = min(2*n-2, 2*n-2-i)
        idx2 = i + np.arange(k_lo2, k_hi2+1) - (n-1)
        delta[k_lo2:k_hi2+1] -= h[idx2]
        
        # Point 3: h[j] 1→0, first arg = j
        k_lo3 = max(0, j)
        k_hi3 = min(2*n-2, j+n-1)
        t_vals3 = j + n - 1 - np.arange(k_lo3, k_hi3+1)
        delta[k_lo3:k_hi3+1] -= (1.0 - h[t_vals3])
        
        # Point 4: (1-h[j]) 0→1, second arg = j
        k_lo4 = max(0, n-1-j)
        k_hi4 = min(2*n-2, 2*n-2-j)
        idx4 = j + np.arange(k_lo4, k_hi4+1) - (n-1)
        delta[k_lo4:k_hi4+1] += h[idx4]
        
        new_c = c + delta
        new_max = float(new_c.max())
        
        # Accept if better or with probability exp(-(new_max-cur_max)/T)
        if new_max < cur_max or np.random.random() < np.exp(-(new_max - cur_max) / T):
            h[i], h[j] = 1.0, 0.0
            c = new_c
            cur_max = new_max
            ones.discard(i)
            ones.add(j)
            zeros.add(i)
            zeros.discard(j)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        
        T = max(T_min, T * decay)
    
    return best_h, best_max

def binary_steepest(h_init, n, deadline, max_sweeps=100):
    """Binary local search: steepest descent (find best swap each step)."""
    h = project_binary(h_init, n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = float(c.max())
    best_h = h.copy()
    best_max = cur_max
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Find top lags
        top_lags = np.argsort(c)[::-1][:top_k] if 'top_k' in dir() else np.argsort(c)[::-1][:5]
        improved = False
        for i in zeros_list:
            best_j = -1
            best_nm = cur_max + 1e-10
            for j in ones_list:
                # Compute delta
                delta = np.zeros(2*n - 1)
                k_lo = max(0, i); k_hi = min(2*n-2, i+n-1)
                tv = i + n - 1 - np.arange(k_lo, k_hi+1)
                delta[k_lo:k_hi+1] += (1.0 - h[tv])
                k2lo = max(0, n-1-i); k2hi = min(2*n-2, 2*n-2-i)
                ix = i + np.arange(k2lo, k2hi+1) - (n-1)
                delta[k2lo:k2hi+1] -= h[ix]
                k3lo = max(0, j); k3hi = min(2*n-2, j+n-1)
                tv3 = j + n - 1 - np.arange(k3lo, k3hi+1)
                delta[k3lo:k3hi+1] -= (1.0 - h[tv3])
                k4lo = max(0, n-1-j); k4hi = min(2*n-2, 2*n-2-j)
                ix4 = j + np.arange(k4lo, k4hi+1) - (n-1)
                delta[k4lo:k4hi+1] += h[ix4]
                
                nm = float((c + delta).max())
                if nm < best_nm:
                    best_nm = nm
                    best_j = j
            if best_j < 0:
                continue
            # Apply
            # Recompute delta with best_j
            delta = np.zeros(2*n - 1)
            k_lo = max(0, i); k_hi = min(2*n-2, i+n-1)
            tv = i + n - 1 - np.arange(k_lo, k_hi+1)
            delta[k_lo:k_hi+1] += (1.0 - h[tv])
            k2lo = max(0, n-1-i); k2hi = min(2*n-2, 2*n-2-i)
            ix = i + np.arange(k2lo, k2hi+1) - (n-1)
            delta[k2lo:k2hi+1] -= h[ix]
            k3lo = max(0, best_j); k3hi = min(2*n-2, best_j+n-1)
            tv3 = best_j + n - 1 - np.arange(k3lo, k3hi+1)
            delta[k3lo:k3hi+1] -= (1.0 - h[tv3])
            k4lo = max(0, n-1-best_j); k4hi = min(2*n-2, 2*n-2-best_j)
            ix4 = best_j + np.arange(k4lo, k4hi+1) - (n-1)
            delta[k4lo:k4hi+1] += h[ix4]
            
            h[i], h[best_j] = 1.0, 0.0
            c += delta
            cur_max = best_nm
            ones_list.remove(best_j)
            ones_list.append(i)
            zeros_list.remove(i)
            zeros_list.append(best_j)
            improved = True
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        if not improved:
            break
    return best_h, best_max
