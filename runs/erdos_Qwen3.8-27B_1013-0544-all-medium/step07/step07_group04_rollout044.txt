    idx0 = off

    exact_s = xs*(1.0-xs) + yv*(1.0-yv) - (x0*(1.0-x0) + y0*(1.0-y0))
    delta[:, idx0] += exact_s - base_s
    d = j - i
    idxp = off - d
    base_p = ax1*(1.0-y0) + x0*ax1
    exact_p = xs*(1.0-yv) - x0*(1.0-y0)
    delta[:, idxp] += exact_p - base_p
    idxm = off + d
    base_m = ax2*(1.0-x0) + y0*ax2
    exact_m = yv*(1.0-xs) - y0*(1.0-x0)
    delta[:, idxm] += exact_m - base_m
    return np.max(c_base[None, :] + dx * delta, axis=1)

def pair_search(h, c_base, i, j, n):
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = np.array([lo, hi, x0], dtype=float)
    if hi - lo > 1e-12:
        xs = np.concatenate((xs, np.linspace(lo, hi, 20)))
    a = max(lo, x0 - 0.05)
    b = min(hi, x0 + 0.05)
    if b - a > 1e-12:
        xs = np.concatenate((xs, np.linspace(a, b, 60)))
    xs = np.unique(np.clip(xs, lo, hi))
    vals = pair_eval(h, i, j, xs, n)
    for _ in range(3):
        if len(xs) <= 1:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k-1], xs[k+1]
        if b - a < 1e-12:
            break
        xs2 = np.linspace(a, b, 60)
        vals = pair_eval(h, i, j, xs2, n)
        xs = xs2
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c_base.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(S - xb)
        if yb < 0.0:
            yb = 0.0
            xb = S - yb
        elif yb > 1.0:
            yb = 1.0
            xb = S - yb
        return xb, yb, bv
    return None

def pair_local_search(h, deadline, seed=1):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    if n < 2 or time.time() > deadline:
        return project_box_sum(besth, n), bestc
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 2000
    off = n - 1
    while time.time() < deadline and noimp < maxno:
        if rng.random() < 0.8:
            pos = c[off+1:]
            neg = c[off-1::-1]
            w = np.maximum(pos, neg) + 1e-12
            w /= w.sum()
            d = int(rng.choice(n-1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search(h, c, i, j, n)
        if res is not None:
            x, y, _ = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())

def interp_gradient_step(h, n, alpha, mu, step):
    g = pen_grad(h, n, alpha, mu)
    h2 = h - step * g
    return project_box_sum(h2, n)

def optimize_single(h, n, seed, deadline):
    h = project_box_sum(h, n)
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    work = h.copy()
    workc = bestc
    phases = (
        (20.0, 50.0, 300),
        (100.0, 200.0, 250),
        (500.0, 1000.0, 200),
        (2000.0, 5000.0, 200),
        (10000.0, 50000.0, 150),
        (50000.0, 200000.0, 100),
    )
    for alpha, mu, maxit in phases:
        if time.time() > deadline - 8:
            break
        h2 = lbfgsb_opt(work, n, alpha, mu, maxit)
        h2 = project_box_sum(h2, n)
        v2 = float(compute_c(h2, n).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
        if v2 <= workc + 1e-8:
            work = h2
            workc = v2
        else:
            work = besth.copy()
            workc = bestc
    if time.time() < deadline - 20:
        besth, bestc = pair_local_search(besth, deadline, seed=seed)
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())
