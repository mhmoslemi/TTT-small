        best_h, best_c = h1, c1

    # Phase 2: Vectorized coordinate descent (fast)
    h2, c2 = coord_cd_vec(best_h, n, 400, deadline, grid_size=150)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-resolution refinement
    for n_try in [200, 300, 400, 600, 800, 1000]:
        if time.time() > deadline - 250:
            break
        n_cur = len(best_h)
        x_old = np.linspace(0.0, 2.0, n_cur)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [200, 2000, 20000, 100000, 500000], 800)
        h_opt, c_opt = coord_cd_vec(h_opt, n_try, 200, deadline, grid_size=100)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 4: Multi-restart with structured and random starts
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r % 5 == 4:
                hp = reappraisal_init(n_cur, rng)
            elif r % 5 == 3:
                hp = random_start_local(n_cur, rng)
            else:
                s = 0.003 + 0.03 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = slsqp_run(hp, n_cur, [1000, 10000, 80000], 500)
            h_o, c_o = coord_cd_vec(h_o, n_cur, 120, deadline, grid_size=80)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Final high-precision pass
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        hf, cf = slsqp_run(best_h, n_cur, [300000, 800000, 2000000], 3000)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd_vec(best_h, n_cur, 400, deadline, grid_size=200)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)

def random_start_local(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)
