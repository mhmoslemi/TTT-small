def project_feasible(a, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]

        a = np.full(n, 0.5)

    # Bisection on threshold
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)

    dx = 2.0 / float(n)

    return float(F), g

    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.clip(np.asarray(a, dtype=float).ravel(), 0.0, 1.0)
    if a.size != n:
        a = a[:n] if a.size > n else np.concatenate([a, np.zeros(n - a.size)])
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):

        if abs(d) < 1e-14: break

            if r.sum() < 1e-15: break

        else:

        h[i] = np.clip(h[i] + d, 0.0, 1.0)

    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))

    if s < 1e-300:
        p = np.full_like(c, 1.0 / len(c))
