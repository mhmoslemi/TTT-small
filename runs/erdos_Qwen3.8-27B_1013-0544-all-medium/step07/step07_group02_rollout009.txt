where c_m(h) is linear in h[j] (for fixed other coordinates). So C₅ as a function of h[j] is the max of linear functions, hence convex. We can find the minimum by:
- Evaluating at h[j] = 0 and h[j] = 1
- Finding the "kink" points where the active constraint changes
- The minimum is either at a kink or at a boundary

For n=400, there are 799 constraints (lags), so the function is the max of 799 linear functions in one variable. The minimum can be found by:

Actually, a simpler approach: since the function is the max of linear functions, and each linear function has slope a_m = dc_m/dh[j] and intercept b_m = c_m(h|_{j=0}), the max is a piecewise linear function. If all active slopes are positive, decrease v; if all are negative, increase v; if there's a sign change, we're at the minimum.

3. Find the minimum of the envelope

    def try_optimize(h, n, alphas=None, mu=500.0, label=""):
        nonlocal best_c5, h_best, n_best
        if alphas is None:
            alphas = [100, 300, 1000, 3000, 10000]
        h = h.copy()
        for alpha in alphas:
            if time.time() > dl - 5:
                break
            h = lbfaps_optimize(h, n, alpha, mu, maxiter=2000)
            c5 = compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                h_best = h.copy()
                n_best = n
        return h

    # Phase 1: Optimize from initial (30%)
    p1_dl = t0 + 0.30 * (budget_s - 10)
    h = h_best.copy()
    for alpha in [50, 150, 500, 1500, 5000, 15000]:
        if time.time() > p1_dl:
            break
        h = lbfaps_optimize(h, n_best, alpha, 500.0, maxiter=2500)
        c5 = compute_c5(h, n_best)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    # Phase 2: SLSQP polish (10%)
    if time.time() < dl - 100:
        h = slsqp_optimize(h_best, n_best, alpha=10000, maxiter=3000)
        c5 = compute_c5(h, n_best)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    # Phase 3: Random restarts (25%)
    p3_dl = t0 + 0.65 * (budget_s - 10)
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() > p3_dl:
            break
        if r == 0:
            h = h_best + 0.03 * rng.randn(n_best)
        elif r == 1:
            h = make_binary_init(n_best, rng)
        elif r == 2:
            h = make_smooth_init(n_best, rng)
        else:
            h = h_best + 0.05 * rng.randn(n_best)
        h = project_feasible(h, n_best)
        for alpha in [100, 500, 2000, 10000]:
            if time.time() > p3_dl:
                break
            h = lbfaps_optimize(h, n_best, alpha, 500.0, maxiter=1500)
            c5 = compute_c5(h, n_best)
            if c5 < best_c5:
                best_c5 = c5
                h_best = h.copy()

    # Phase 4: Multi-resolution - try n=800 (20%)
    p4_dl = t0 + 0.85 * (budget_s - 10)
    n2 = 800
    h2 = interpolate_h(h_best, n_best, n2)
    c5_2 = compute_c5(h2, n2)
    for alpha in [100, 500, 2000, 10000]:
        if time.time() > p4_dl:
            break
        h2 = lbfaps_optimize(h2, n2, alpha, 500.0, maxiter=1500)
        c5_2 = compute_c5(h2, n2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            h_best = h2.copy()
            n_best = n2

    # Phase 5: Final L-BFGS-B + SLSQP polish (15%)
    h = h_best.copy()
    n = n_best
    for alpha in [5000, 20000, 50000]:
        if time.time() > dl - 3:
            break
        h = lbfaps_optimize(h, n, alpha, 1000.0, maxiter=2000)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    if time.time() < dl - 5:
        h = slsqp_optimize(h_best, n_best, alpha=20000, maxiter=2000)
        c5 = compute_c5(h, n_best)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    h_best = project_feasible(h_best, n_best)
    best_c5 = compute_c5(h_best, n_best)
    return h_best, best_c5, n_best
