    A_p = np.zeros(N)
    A_p[:n] = A
    y_p = np.zeros(N)
    y_p[:n] = y
    w_p = np.zeros(N)

    Fy = np.fft.rfft(y_p)

            s = h.sum()
            if s < 1e-14:
                break

def compute_corr(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

class LSEProblem:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
        self.dx = 2.0 / n
        self.N = 2 ** int(np.ceil(np.log2(3 * n + 2)))
    
    def f(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = c.max()
        s = self.alpha * (c - m)
        s = np.clip(s, -700, 700)
        sm = s.max()
        lse = sm + np.log(np.exp(s - sm).sum())
        return m + lse / self.alpha
    
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = c.max()
        s = self.alpha * (c - m)
        s = np.clip(s, -700, 700)
        sm = s.max()
        w = np.exp(s - sm)
        w /= w.sum()
        # Grad via FFT
        n = self.n
        N = self.N
        A = 1.0 - x
        A_p = np.zeros(N); A_p[:n] = A

        w_p = np.zeros(N); w_p[:2*n-1] = w

        Q = np.fft.irfft(np.fft.rfft(x_p) * CFW, N)[:n]

To compute this efficiently:
M[a,b] = max_k (corr[k] + D[a,k] + E[b,k])

This can be written as: M[a,b] = max_k (corr[k] + D[a,k] + E[b,k])

For fixed a: M[a,:] = max_k (corr[k] + D[a,k] + E[:,k])
This is a (n_one,) vector, computed as the element-wise max of (n_one × (2n-1)) + (1 × (2n-1)) + (n_one × (2n-1)).

So: temp = E + (corr + D[a])[np.newaxis, :]  → shape (n_one, 2n-1)
M[a,:] = temp.max(axis=1)

Total work: n_zero × n_one × (2n-1) = (n/2) × (n/2) × 2n = n³/2.
For n=400: 32M operations. In numpy, this is a few seconds per sweep.

With 20-50 sweeps, that's 1-2 minutes. Manageable.

But I can do better: after each accepted swap, I only need to update D and E for the changed positions. And I can keep a "candidate list" of promising swaps.

Actually, let me simplify. I'll do:
1. Compute the initial binary solution
2. Compute corr
3. Precompute D for all zeros and E for all ones
4. In each sweep, for each zero i:
   a. Compute the best j (one position) to swap with
   b. If the swap improves, do it and update
5. Repeat until no improvement or time runs out

For step 4a, I'll use the vectorized approach above.

def lse_gradient(x, alpha, dx, N):

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    A_p = np.zeros(N); A_p[:n] = 1.0 - x
    x_p = np.zeros(N); x_p[:n] = x
    w_p = np.zeros(N); w_p[:2*n-1] = w

def compute_swap_delta(i, j, h, n):
    delta = np.zeros(2*n - 1)
    for k in range(-i, n - i):
        if k == 0: continue
        delta[n-1+k] += 1.0 - h[i+k]
    for k in range(i-n+1, i+1):
        if k == 0: continue
        delta[n-1+k] -= h[i-k]
    for k in range(-j, n - j):
        if k == 0: continue
        delta[n-1+k] -= 1.0 - h[j+k]
    for k in range(j-n+1, j+1):
        if k == 0: continue
        delta[n-1+k] += h[j-k]
    return delta

def binary_local_search(h_init, n, deadline, max_sweeps=60):
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n//2]] = 1.0
    corr = np.correlate(h_b, 1.0 - h_b, mode='full') * dx
    cur_c = float(corr.max())
    best_h = h_b.copy()
    best_c = cur_c
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = list(np.where(h_b == 0)[0])
        ones = list(np.where(h_b == 1)[0])
        for i in zeros:
            if h_b[i] == 1:
                continue
            best_j = -1
            best_nc = cur_c - 1e-15
            for j in ones:
                if h_b[j] == 0:
                    continue
                delta = compute_swap_delta(i, j, h_b, n)
                nc = float((corr + delta).max())
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j < 0:
                continue
            delta = compute_swap_delta(i, best_j, h_b, n)
            h_b[i], h_b[best_j] = 1.0, 0.0
            corr += delta
            cur_c = best_nc
            ones.remove(best_j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(best_j)
            improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c
