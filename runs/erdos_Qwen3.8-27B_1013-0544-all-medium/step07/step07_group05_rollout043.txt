def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = float(target - y.sum())
    if abs(diff) < 1e-14:
        return y
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5*(lo+hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target: lo = mid
        else: hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-13: break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = max(1, int(mask.sum()))
        y[mask] += diff/cnt
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def grad_smooth(h, n, beta):
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')*dx
    maxc = float(c.max())
    z = beta*(c - maxc)
    z = np.clip(z, -745.0, 0.0)
    ew = np.exp(z)
    ws = max(float(ew.sum()), 1e-300)
    w = ew/ws
    f = maxc + np.log(ws)/beta
    
    mid = n-1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid+1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pp = np.cumsum(w_pos)
    pn = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pp + pn[n-1-idx]
    kern = np.zeros(2*n-1)
    kern[mid+1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid+n]
    g = (w[mid]*(1.0-2.0*h) + consts - neigh)*dx
    return f, g

        if s >= 0:
            g[s:] = 1.0 - y_vec[:n-s]
            g[:n-s] -= y_vec[s:]
        else:
            s2 = -s
            g[:n-s2] = 1.0 - y_vec[s2:]
            g[s2:] -= y_vec[:n-s2]

            best_h = h.copy()

    c5y = erdos_c5(y_vec, n)

def coord_descent(h, n, rounds, deadline):
    """Coordinate descent: try moving mass between pairs."""
    target = 0.5*n
    h = h.astype(float).copy()
    cur = erdos_c5(h, n)

        if time.time() > deadline - 2.0:

        n_try = min(200, n*n//4)
        # Actually, let's try a subset of pairs

            break

        for _ in range(n_try):
            i = free[np.random.randint(len(free))]
            j = free[np.random.randint(len(free))]
            if i == j:
                continue
            # Binary search for best transfer amount
            lo = max(-h[i], h[j]-1.0)
            hi = min(1.0-h[i], h[j])
            # Try a few values

            best_c = cur
            for d in [0.5*(lo+hi), lo*0.5, hi*0.5, 0.0]:

                hh = h.copy()
                hh[i] += d
                hh[j] -= d
                c = erdos_c5(hh, n)

                    best_d = d
            if best_c < cur - 1e-14:

                h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)
                cur = best_c
                improved = True

def solve_lp_cutting_plane(h_init, n, max_iters, deadline):
    """Cutting plane method for min-max problem."""
    h = h_init.astype(float).copy()
    target = 0.5*n
    
    for it in range(max_iters):
        if time.time() > deadline - 2.0:
            break
        
        # Compute f_k and gradients at current h
        dx = 2.0/n
        c = np.correlate(h, 1.0-h, mode='full')*dx
        c5 = float(c.max())
        
        # Keep only constraints where c[k] > c5 - 0.01 (active or near-active)
        active = np.where(c > c5 - 0.005)[0]
        
        # Build LP: min t s.t. g_k^T h + d_k <= t, sum(h)=n/2, 0<=h<=1
        # where g_k = grad f_k at h, d_k = f_k(h) - g_k^T h
        vars_n = n + 1  # h[0..n-1] and t
        A_ub = []
        b_ub = []
        
        for k_idx in active:
            mid = n-1
            s = k_idx - mid
            g = np.zeros(n)
            if s >= 0:
                g[s:] = 1.0 - h[:n-s]
                g[:n-s] -= h[s:]
            else:
                s2 = -s
                g[:n-s2] = 1.0 - h[s2:]
                g[s2:] -= h[:n-s2]
            g *= dx
            f_k = float(c[k_idx])
            d = f_k - g @ h  # constant term (should be 0 for exact linearization)
            # Constraint: g^T h - t + d <= 0, i.e., g^T h - t <= -d
            row = np.concatenate([g, [-1.0]])
            A_ub.append(row)
            b_ub.append(-d)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Equality constraint: sum(h) = n/2
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0,1)]*n + [(None, None)]
        
        # Objective: minimize t (last variable)
        c_obj = np.zeros(n+1)
        c_obj[n] = 1.0
        
        res = so.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            t_new = res.x[n]
            if t_new < c5 - 1e-6:
                h = h_new
            else:
                # No improvement, break
                break
        else:
            break
    
    return h, erdos_c5(h, n)

This is a fundamentally different approach! The cutting plane method adds constraints one at a time (or in batches), and each LP solve is fast (using HiGHS). This should give a good lower bound on the optimal C₅ for the discretized problem.

The key advantage: each LP solve gives a point that satisfies the active linearized constraints, and we iteratively add more constraints. This is a form of **outer approximation** for the non-convex problem.

The disadvantage: it only gives a lower bound (since we're solving a relaxation). But combined with the original objective evaluation, we can check if the LP solution is actually good.

Wait, actually the LP solution is a feasible point for the linearized problem, not necessarily for the original problem. The actual C₅ at the LP solution might be higher than the LP optimal t. But it's a good starting point for local optimization.

Let me also think about whether the cutting plane approach can actually improve things. The key question: is the min-max landscape "convex enough" that linearization works well?

Since f_k(h) = sum_i h[i+s]*(1-h[i])*dx is a quadratic (specifically, a sum of bilinear terms), its linearization at a point h_t is:
f_k(h) = f_k(h_t) + ∇f_k(h_t)·(h-h_t) + (h-h_t)^T * H_k * (h-h_t) / 2

where H_k is the Hessian of f_k. Since f_k is bilinear (not convex or concave in general), the Hessian is indefinite. So the linearization can be an under-estimate or over-estimate depending on the direction.

This means the cutting plane approach might not converge to the true optimum. But it can still give good practical improvements.

import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = float(target - y.sum())
    if abs(diff) < 1e-14:
        return y
    lo, hi = -2.0, 2.0
    for _ in range(70):
        md = 0.5*(lo+hi)
        y = np.clip(x-md, 0.0, 1.0)
        if y.sum() > target: lo = md
        else: hi = md
    y = np.clip(x-0.5*(lo+hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-13: break
        mask = (y > 1e-12) & (y < 1.0-1e-12)
        cnt = max(1, int(mask.sum()))
        y[mask] += diff/cnt
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def grad_at_shift(h, s, n):
    """Gradient of overlap at shift s."""
    dx = 2.0/n
    g = np.zeros(n)
    if s >= 0:
        g[s:] = 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        s2 = -s
        g[:n-s2] = 1.0 - h[s2:]
        g[s2:] -= h[:n-s2]
    return g * dx

def grad_smoothed(h, n, beta):
    """Smoothed max gradient."""
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')*dx
    mc = float(c.max())
    z = np.clip(beta*(c-mc), -745.0, 0.0)
    ew = np.exp(z)
    ws = max(ew.sum(), 1e-300)
    w = ew/ws
    f = mc + np.log(ws)/beta
    mid = n-1
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid+1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n-1-idx]
    kern = np.zeros(2*n-1)
    kern[mid+1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid+n]
    g = (w[mid]*(1.0-2.0*h) + consts - neigh)*dx
    return f, g

def nesterov(h, n, steps, deadline, lr0=0.06):
    """Nesterov-projected subgradient."""
    target = 0.5*n
    h = h.astype(float).copy()
    yv = h.copy()
    best_h = h.copy()
    best_c = erdos_c5(h, n)
    t = 1.0
    for _ in range(steps):
        if time.time() > deadline-2.0: break
        dx = 2.0/n
        c = np.correlate(yv, 1.0-yv, mode='full')*dx
        k = int(c.argmax())
        g = grad_at_shift(yv, k-(n-1), n)
        lr = lr0/(1.0+0.05*t)
        z = yv + (t-1.0)/(t+2.0)*(yv-h)
        nh = project_box_sum(z - lr*g, target)
        mu = t/(t+2.0)
        yv = project_box_sum((1.0+mu)*nh - mu*h, target)
        h = nh
        t += 1.0
        c5 = erdos_c5(h, n)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
    yv = project_box_sum(yv, target)
    if erdos_c5(yv, n) < best_c:
        best_c = erdos_c5(yv, n)
        best_h = yv.copy()
    return best_h, best_c

def lbfgs_smooth(h, betas, maxiter, deadline):
    """L-BFGS with smooth max."""
    n = len(h)
    target = 0.5*n
    best = h.copy()
    best_c = erdos_c5(h, n)
    for beta in betas:
        if time.time() > deadline-3.0: break
        def obj(x): return grad_smoothed(x, n, beta)[0]
        def jac(x): return grad_smoothed(x, n, beta)[1]
        try:
            res = so.minimize(obj, h, method='L-BFGS-B', jac=jac,
                            bounds=[(0,1)]*n,
                            options={'maxiter':maxiter,'ftol':1e-18,'gtol':1e-14})
            cand = project_box_sum(res.x, target)
        except:
            cand = h
        c = erdos_c5(cand, n)
        if c < best_c - 1e-15:
            best_c = c
            best = cand
            h = cand
    return best, best_c

def lp_cutting(h, n, max_iters, deadline):
    """Cutting plane (outer approximation) via LP."""
    target = 0.5*n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = erdos_c5(h, n)
    mid = n-1
    for it in range(max_iters):
        if time.time() > deadline-2.0: break
        c = np.correlate(h, 1.0-h, mode='full')*(2.0/n)
        c5 = float(c.max())
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        # Active constraints
        thresh = c5 - max(0.005, 0.02*c5)
        active = np.where(c >= thresh)[0]
        if len(active) > 100:
            # Just take top 100
            top_idx = np.argsort(c)[-100:]
            active = top_idx
        # Build LP
        A_rows = []
        b_rows = []
        for k_i in active:
            s = k_i - mid
            g = grad_at_shift(h, s, n)
            fk = float(c[k_i])
            d = fk - g @ h
            A_rows.append(np.concatenate([g, [-1.0]]))
            b_rows.append(-d)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        A_eq = np.zeros((1, n+1))
        A_eq[0,:n] = 1.0
        b_eq = np.array([target])
        c_obj = np.zeros(n+1)
        c_obj[n] = 1.0
        bounds = [(0,1)]*n + [(None, 1.0)]
        try:
            res = so.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                            bounds=bounds, method='highs')
            if res.success:
                h = project_box_sum(res.x[:n], target)
                if erdos_c5(h, n) < best_c:
                    best_c = erdos_c5(h, n)
                    best_h = h.copy()
            else:
                break
        except:
            break
        # Check convergence
        c_new = np.correlate(h, 1.0-h, mode='full')*(2.0/n)
        if float(c_new.max()) > c5 + 1e-4:
            # Got worse, revert
            h = best_h
            break
    return best_h, best_c
