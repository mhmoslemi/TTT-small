    """Compute the change in correlation when h[i] 0->1 and h[j] 1->0."""
    delta = np.zeros(2*n-1)
    h_new_i = 1.0  # h'[i]
    h_new_j = 0.0  # h'[j]

        k = s + (n-1)  # correlation array index
        total = 0.0
        # t = i
        if 0 <= i + s < n:
            old = h[i] * (1 - h[i+s])
            new = h_new_i * (1 - (h_new_j if i+s == j else h[i+s]))
            total += new - old
        # t = j
        if 0 <= j + s < n:
            old = h[j] * (1 - h[j+s])
            new = h_new_j * (1 - (h_new_i if j+s == i else h[j+s]))
            total += new - old

            new = h[t] * (1 - h_new_i)

        t = j - s
        if 0 <= t < n and t != i and t != j:

            new = h[t] * (1 - h_new_j)

    return delta

        if abs(diff) < 1e-15:

            if s < 1e-15:

def slsqp_logsumexp(h, n, alpha, maxiter, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
