def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

            break

            tot = room.sum()

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(x.shape[0])

def slsqp_round(h, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h

def find_linear_min(c, a, lo, hi):
    for _ in range(45):
        if hi - lo < 1e-16:
            break
        d = 0.5 * (lo + hi)
        vals = c + d * a
        f_val = vals.max()
        active = (vals - f_val) > -1e-14
        if not active.any():
            idx = np.argmax(vals)
            active = np.zeros(a.shape, dtype=bool)
            active[idx] = True
        min_s = a[active].min()
        max_s = a[active].max()
        if min_s > 1e-15:
            hi = d
        elif max_s < -1e-15:
            lo = d
        else:
            break
    return 0.5 * (lo + hi)

def coord_cd(h, n, deadline=None, max_passes=10000, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    ncorr = 2 * n - 1
    nsub = n - 1
    ms = np.arange(ncorr)
    order = np.arange(n)
    pass_count = 0
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        pass_count += 1
        if random_order and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            a = np.zeros(ncorr)
            jf = p + ms - nsub
            vf = (jf >= 0) & (jf < n)
            a[vf] += 1.0 - h[jf[vf]]
            jb = p - ms + nsub
            vb = (jb >= 0) & (jb < n)
            a[vb] -= h[jb[vb]]
            lo, hi = -h0, 1.0 - h0
            d_star = find_linear_min(c, a, lo, hi)
            best_d = d_star
            vals = c + d_star * a
            vals[nsub] -= d_star * d_star
            best_val = vals.max()
            span = max(0.03, 0.3 * min(d_star - lo, hi - d_star))
            d_grid = np.linspace(d_star - span, d_star + span, 25)
            for d_try in d_grid:
                d_try = float(np.clip(d_try, lo, hi))
                vals = c + d_try * a
                vals[nsub] -= d_try * d_try
                v = vals.max()
                if v < best_val - 1e-16:
                    best_val = v
                    best_d = d_try
            for d_try in (lo, hi):
                vals = c + d_try * a
                vals[nsub] -= d_try * d_try
                v = vals.max()
                if v < best_val - 1e-16:
                    best_val = v
                    best_d = d_try
            if abs(best_d) > 1e-14:
                h[p] = h0 + best_d
                c += best_d * a
                c[nsub] -= best_d * best_d
                improved = True
        if pass_count % 50 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    return h, compute_c5(h, n)
