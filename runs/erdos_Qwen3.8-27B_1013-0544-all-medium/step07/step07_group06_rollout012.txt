```python
def compute_base_binary(h, n):
    # Vectorized version
    idx = np.arange(n)
    # For each p, base[p, n-1+s] = (1-h[p+s]) * I[0<=p+s<n] - h[p-s] * I[0<=p-s<n]
    # This is a circulant-like matrix
    # base[p, n-1+s] for s = -(n-1), ..., n-1
    # = (1-h[p+s]) if 0<=p+s<n else 0, minus h[p-s] if 0<=p-s<n else 0
    
    # Forward part: base[p, n-1+s] = 1-h[p+s] for p+s in [0,n-1]
    # This is like a Toeplitz matrix
    # forward[p, s] = 1-h[p+s] if 0<=p+s<n
    # = (1-h) shifted
    
    # Let's think of it as: forward_matrix[p, n-1+s] = 1-h[p+s] if valid
    # = row p of a matrix where entry (p, q) = 1-h[q] if q-p is in [-(n-1), n-1] and q in [0,n-1]
    # = just 1-h[q] for all valid q, placed at the right column
    
    # Actually, the correlation index n-1+s corresponds to the "shift" s.
    # For a fixed p, the shifts s such that p+s is valid are: s in [max(-(n-1), -p), min(n-1, n-1-p)]
    # And the value is 1-h[p+s]
    
    # Similarly for the backward part: -h[p-s] for p-s valid, s in [max(-(n-1), p-(n-1)), min(n-1, p)]
    
    # Let me just use numpy indexing
    p = np.arange(n)[:, None]  # (n, 1)
    s = np.arange(-(n-1), n)[None, :]  # (1, 2n-1)
    # Forward: p+s
    ps = p + s  # (n, 2n-1)
    fwd_valid = (ps >= 0) & (ps < n)
    fwd = np.where(fwd_valid, 1.0 - h[ps.clip(0, n-1)], 0.0)
    # Backward: p-s
    pm = p - s
    bwd_valid = (pm >= 0) & (pm < n)
    bwd = np.where(bwd_valid, -h[pm.clip(0, n-1)], 0.0)
    
    return fwd + bwd
```

```python
        if time.time() > deadline:
            break
        base = compute_base_binary(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            end = min(start + chunk, zi.size)
            i = zi[start:end]; j = oj[start:end]
            arr = base[i] - base[j] + corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            arr[:, d1] += 1.0
            d2 = n - 1 + (i - j)
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv, jv = int(i[loc]), int(j[loc])
                h[iv], h[jv] = 1.0, 0.0
                corr = arr[loc].astype(float, copy=True)
                M = int(round(corr.max()))
                if M < bestM:
                    bestM, besth = M, h.copy()
                improved = True
                break
        if not improved:
            break
    return besth, bestM * (2.0 / float(n))

def anneal_binary(h, n, deadline, t_init=5.0, t_final=0.01, seed=42, rebuild_every=80):
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(corr.max()))
    besth, bestM = h.copy(), M
    base = compute_base_binary(h, n)
    rng = np.random.RandomState(seed)
    ns = 2 * n - 1
    tmp = np.empty(ns)
    new_corr = np.empty(ns)
    total_steps = max(20000, n * 300)
    for step in range(total_steps):
        if time.time() > deadline:
            break
        frac = min(1.0, step / total_steps)
        T = t_init * (t_final / t_init) ** frac
        if step > 0 and step % rebuild_every == 0:
            base = compute_base_binary(h, n)
        i = rng.randint(n)
        j = rng.randint(n)
        while h[i] == h[j]:
            i = rng.randint(n)
            j = rng.randint(n)
        if h[i] == 1.0:
            i, j = j, i
        np.subtract(base[i], base[j], out=tmp)
        np.add(corr, tmp, out=new_corr)
        new_corr[n - 1] -= 2.0
        d1 = n - 1 + (j - i)
        if 0 <= d1 < ns:
            new_corr[d1] += 1.0
        d2 = n - 1 + (i - j)
        if 0 <= d2 < ns:
            new_corr[d2] += 1.0
        newM = int(round(new_corr.max()))
        delta = newM - M
        if delta <= 0 or rng.rand() < np.exp(-float(delta) / max(T, 1e-15)):
            h[i], h[j] = 1.0, 0.0
            corr, new_corr = new_corr, corr
            M = newM
            if M < bestM:
                bestM, besth = M, h.copy()
    return besth, bestM * (2.0 / float(n))

def lse_objective_grad(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    p = e / s if s > 0 else np.full_like(c, 1.0 / (2 * n - 1))
    F = m + np.log(s) / beta
    P = p; R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def continuous_refine(h, n, deadline, betas=(100.0, 500.0, 2000.0, 8000.0)):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 5.0:
            break
        _stop_time = min(deadline - 2.0, time.time() + 60.0)
        def obj(x, n=n, beta=beta):
            hh = np.clip(x, 0.0, 1.0)
            F, g = lse_objective_grad(hh, n, beta)
            dev = hh.sum() - 0.5 * n
            F += 0.5 * 1e4 * dev * dev
            g = g + 1e4 * dev
            return F, g
        try:
            res = minimize(obj, work, method="L-BFGS-B", jac=True,
                           bounds=bounds, callback=stop_callback,
                           options={"maxiter": 200000, "maxls": 30, "ftol": 1e-14, "gtol": 1e-8})
            x = res.x if res is not None and hasattr(res, "x") else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
        m2 = 0.5 * (work + work[::-1])
        m2 = project_feasible(m2, n)
        c2 = compute_c5(m2, n)
        if c2 < bc:
            bc, best = c2, m2.copy()
        inv = 1.0 - work[::-1]
        inv = project_feasible(inv, n)
        c3 = compute_c5(inv, n)
        if c3 < bc:
            bc, best = c3, inv.copy()
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + max(30.0, float(budget_s) - 12.0)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h0))
        if n < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n = 400
        h0 = ((np.arange(n) // 50) % 2).astype(float)

    best_h = top_n_binary(h0, n)
    best_c = compute_c5(best_h, n)

    # Quick greedy to establish local optimum
    if time.time() < end - 100.0:
        gh, gc = greedy_polish(best_h, n, min(end - 80.0, t0 + 60.0), max_sweeps=25)
        if gc < best_c:
            best_c, best_h = gc, gh.copy()

    # Multi-start annealing
    anneal_configs = [
        (5.0, 0.01, 42), (20.0, 0.05, 137), (1.0, 0.001, 999),
        (10.0, 0.02, 256), (50.0, 0.1, 777),
    ]
    phase1_end = min(end - 200.0, t0 + 0.55 * (float(budget_s) - 12.0))
    for ti, tf, sd in anneal_configs:
        if time.time() > phase1_end - 30.0:
            break
        a_deadline = min(phase1_end - 20.0, time.time() + max(30.0, 0.4 * (phase1_end - time.time())))
        ah, ac = anneal_binary(best_h, n, a_deadline, ti, tf, seed=sd)
        if ac < best_c:
            best_c, best_h = ac, ah.copy()
        if time.time() < phase1_end - 15.0:
            gh, gc = greedy_polish(ah, n, min(phase1_end - 10.0, time.time() + 40.0), max_sweeps=20)
            if gc < best_c:
                best_c, best_h = gc, gh.copy()

    # Try n*2 (upscale)
    n2 = 2 * n
    if n2 <= 1000 and time.time() < end - 180.0:
        h2 = np.repeat(best_h, 2)
        phase2_end = min(end - 120.0, time.time() + 0.25 * (float(budget_s) - 12.0))
        for ti, tf, sd in [(5.0, 0.01, 555), (3.0, 0.005, 666)]:
            if time.time() > phase2_end - 30.0:
                break
            a_end = min(phase2_end - 20.0, time.time() + max(20.0, 0.5 * (phase2_end - time.time())))
            ah2, ac2 = anneal_binary(h2, n2, a_end, ti, tf, seed=sd)
            if ac2 < best_c:
                best_c, best_h, n = ac2, ah2.copy(), n2
                h2 = np.repeat(best_h, 1) if n != n2 else h2
        if time.time() < phase2_end - 15.0:
            gh2, gc2 = greedy_polish(best_h if n == n2 else np.repeat(best_h, 2), n2,
                                     min(phase2_end - 10.0, time.time() + 30.0), max_sweeps=15)
            if gc2 < best_c:
                best_c, best_h, n = gc2, gh2.copy(), n2

    # Continuous refinement
    if time.time() < end - 50.0:
        rng = np.random.RandomState(seed)
        cont_end = min(end - 15.0, time.time() + 0.2 * (float(budget_s) - 12.0))
        starts = [best_h.copy(), best_h + 0.05 * rng.randn(n), 0.5 * (best_h + best_h[::-1])]
        for s_idx, sh in enumerate(starts):
            if time.time() > cont_end - 10.0:
                break
            hh, hc = continuous_refine(sh, n, min(cont_end - 5.0, time.time() + 80.0),
                                       betas=(200.0, 1000.0, 5000.0))
            if hc < best_c:
                best_c, best_h = hc, hh.copy()

    # Final greedy polish
    if time.time() < end - 10.0 and n % 2 == 0:
        gh, gc = greedy_polish(best_h, n, end - 5.0, max_sweeps=15)
        if gc < best_c:
            best_c, best_h = gc, gh.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, int(n)
```
