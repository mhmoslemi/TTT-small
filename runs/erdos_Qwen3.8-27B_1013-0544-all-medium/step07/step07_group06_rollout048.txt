    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Continuous optimization at n0 with beta annealing
    if time.time() < end - 300:
        betas1 = [50.0, 150.0, 400.0, 1000.0, 3000.0, 10000.0, 30000.0, 100000.0]
        hh, cc = continuous_optimize(best_h, n0, betas1, end - 300, stage_time=50.0)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    # Phase 2: Try other resolutions
    for n_alt in [300, 500, 350]:
        if time.time() > end - 200:
            break
        if n_alt % 2 != 0:
            continue
        x0 = np.linspace(0.0, 2.0, best_n)
        x1 = np.linspace(0.0, 2.0, n_alt)
        hr = np.interp(x1, x0, best_h)
        hr = project_feasible(hr, n_alt)
        cr = compute_c5(hr, n_alt)
        if time.time() < end - 100:
            betas2 = [200.0, 1000.0, 5000.0, 20000.0, 80000.0]
            hh, cc = continuous_optimize(hr, n_alt, betas2, end - 100, stage_time=35.0)
            if cc < best_c:
                best_c = cc
                best_h = hh.copy()
                best_n = n_alt
            # Binary search at this resolution
            if n_alt % 2 == 0 and time.time() < end - 60:
                hb, cb = binary_local_search(hh, n_alt, end - 80, max_sweeps=100)
                if cb < best_c:
                    best_c = cb
                    best_h = hb.copy()
                    best_n = n_alt
                # Re-optimize continuously from binary
                if time.time() < end - 50:
                    hh2, cc2 = continuous_optimize(hb, n_alt, [5000.0, 50000.0], end - 40, stage_time=25.0)
                    if cc2 < best_c:
                        best_c = cc2
                        best_h = hh2.copy()
                        best_n = n_alt

    # Phase 3: Binary search at best_n + continuous re-optimization
    if best_n % 2 == 0 and time.time() < end - 150:
        hb, cb = binary_local_search(best_h, best_n, end - 120, max_sweeps=200)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Re-optimize from binary
        if time.time() < end - 80:
            betas3 = [1000.0, 5000.0, 20000.0, 80000.0, 200000.0]
            hh, cc = continuous_optimize(hb, best_n, betas3, end - 50, stage_time=30.0)
            if cc < best_c:
                best_c = cc
                best_h = hh.copy()

    # Phase 4: Perturbation restarts
    for scale in (0.01, 0.03, 0.06):
        if time.time() > end - 40:
            break
        nh = project_feasible(best_h + scale * rng.randn(best_n), best_n)
        if time.time() < end - 25:
            hh, cc = continuous_optimize(nh, best_n, [10000.0, 50000.0, 200000.0], end - 15, stage_time=20.0)
            if cc < best_c:
                best_c = cc
                best_h = hh.copy()

    # Phase 5: Final high-beta polish
    if time.time() < end - 15:
        hh, cc = continuous_optimize(best_h, best_n, [100000.0, 500000.0], end - 5, stage_time=20.0)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    # Also try binary projection of final
    if best_n % 2 == 0:
        hb_final = project_binary(best_h, best_n)
        cf = compute_c5(hb_final, best_n)
        if cf < best_c:
            best_c = cf
            best_h = hb_final.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
