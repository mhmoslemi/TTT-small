def project_feasible(h, n):

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def precompute_struct(n):
    Ar, Ac, Am = [], [], []
    Br, Bc, Bj = [], [], []
    for p in range(2 * n - 1):
        q = p - (n - 1)
        for j in range(max(0, q), min(n, q + n)):
            Ar.append(p)
            Ac.append(j)
            Am.append(j - q)
        for m in range(max(0, -q), min(n, n - q)):
            Br.append(p)
            Bc.append(m)
            Bj.append(m + q)
    return (np.array(Ar), np.array(Ac), np.array(Am),
            np.array(Br), np.array(Bc), np.array(Bj))

def solve_lp(var_vals, struct, n, mode):
    """mode='h': fix v=var_vals, optimize h. mode='v': fix h=var_vals, optimize v."""
    if mode == 'h':
        rows, cols, idx = struct[0], struct[1], struct[2]
    else:
        rows, cols, idx = struct[3], struct[4], struct[5]
    vals = var_vals[idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    x = cp.Variable(n)
    t = cp.Variable()
    cons = [A @ x <= t, cp.sum(x) == n/2.0, x >= 0, x <= 1]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except Exception:
            return None, None
    if x.value is None:
        return None, None
    return np.array(x.value).flatten(), float(t.value)

def alternate_lp(h_init, n, max_iters=50, deadline=None):
    struct = precompute_struct(n)
    h = h_init.copy()
    v = 1.0 - h
    best_h = h.copy()
    best_t = compute_c5(h, n)
    for it in range(max_iters):
        if deadline and time.time() > deadline:

        # Optimize h given v

        if h_new is None:

        h_new = np.clip(h_new, 0, 1)
        # Optimize v given h_new
        v_new, t2 = solve_lp(h_new, struct, n, 'v')
        if v_new is None:

        v_new = np.clip(v_new, 0, 1)
        # Check consistency

        h = h_new
        v = v_new
        c = compute_c5(h, n)
        if c < best_t:
            best_t = c
            best_h = h.copy()
    return project_feasible(best_h, n), best_t

def binary_local_search(h_init, n, deadline, max_sweeps=100):

    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n//2]] = 1.0

    cur_c = corr.max() * dx

    best_c = cur_c
    ones = set(np.where(h_b == 1)[0])
    zeros = set(np.where(h_b == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros):
            best_j = -1
            best_nc = cur_c - 1e-15
            for j in ones:

                ones.add(i)
                zeros.discard(i)
                zeros.add(best_j)
                improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c

def compute_delta(i, j, h, n):
    """Delta in correlation when h[i]: 0->1, h[j]: 1->0"""
    d = np.zeros(2*n - 1)
    q_arr = np.arange(-(n-1), n)
    # h[i]: 0->1 adds (1-h)[i-q] for valid i-q, subtracts h[i+q] for valid i+q
    valid1 = (i - q_arr >= 0) & (i - q_arr < n)
    d[valid1] += (1.0 - h[i - q_arr[valid1]])
    valid2 = (i + q_arr >= 0) & (i + q_arr < n)
    d[valid2] -= h[i + q_arr[valid2]]
    # h[j]: 1->0 subtracts (1-h)[j-q] for valid j-q, adds h[j+q] for valid j+q
    valid3 = (j - q_arr >= 0) & (j - q_arr < n)
    d[valid3] -= (1.0 - h[j - q_arr[valid3]])
    valid4 = (j + q_arr >= 0) & (j + q_arr < n)
    d[valid4] += h[j + q_arr[valid4]]
    return d
