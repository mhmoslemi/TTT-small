        ones = np.where(h == 1)[0]
        # Remove random ones
        remove = rng.choice(ones, num_ones - target, replace=False)

    else:

        add = rng.choice(zeros, target - num_ones, replace=False)

def lse_obj_grad(h, n, beta, pen):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(P, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

_stop_time = 0.0
def stop_cb(*args):
    return time.time() > _stop_time

def polish_lbfgs(h, n, betas, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 5.0:
            break
        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.5 * (deadline - now))))
        _stop_time = stage_dl
        try:
            res = minimize(lse_obj_grad, work, method="L-BFGS-B", jac=True,
                          bounds=bounds, args=(n, float(beta), 1e4),
                          callback=stop_cb,
                          options={"maxiter": 500000, "maxls": 30, "ftol": 1e-14})
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
    return best, bc

def build_base(h, n):
    """Precompute base matrix for fast swap evaluation.
    base[i, k] = change in corr[k] when h[i] goes from 0 to 1 (or 1 to 0, sign matters).
    Actually let's define it as: if position i changes by delta (h[i] += delta),
    the change in corr[k] is delta * (contribution of position i to corr[k]).
    
    For corr[k] = sum_i h[i] * (1-h[i+d]) where d = k-(n-1):
    If h[i] changes by delta:
    - corr[k] changes by delta * (1-h[i+d]) for the term where first index is i
    - corr[k'] changes by -delta * h[i-d'] for the term where second index is i
      (i.e., where i = j + d', so d' = i - j, k' = n-1 + i - j)
    
    So base[i, k] = (1-h[i+d]) for valid i+d, where d = k-(n-1)
                    - h[i-d'] for valid i-d', where d' = i-j means k' = n-1+i-j
    """
    l = np.arange(-(n-1), n, dtype=int)  # lags d
    base = np.zeros((n, 2*n-1), dtype=float)
    for p in range(n):
        f = p + l
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1.0 - h[f[mf]]
        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

def project_feasible(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size != n:
        a = np.resize(a, n)
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = a.min()-1, a.max()+1
    for _ in range(50):

        if np.clip(a-mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    h = np.clip(a - 0.5*(lo+hi), 0, 1)
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-12: break
        if d > 0:
            r = 1.0-h; s = r.sum()
            if s < 1e-15: break
            h += d*r/s
        else:
            s = h.sum()
            if s < 1e-15: break
            h += d*h/s
    h = np.clip(h, 0, 1)

        h[i] = np.clip(h[i]+d, 0, 1)
    return h

        h8t = thue_morse_bin(n8, rng)
        # SA from both
        p3a = min(end - 90, time.time() + 100)
        h8b, c8b = sa_binary(hr, n8, p3a, rng, max_it=600000)
        if c8b < best_c:
            best_c, best_h, best_n = c8b, h8b, n8
        if time.time() < end - 90:
            p3b = min(end - 80, time.time() + 80)
            h8c, c8c = sa_binary(h8t, n8, p3b, rng, max_it=400000)
            if c8c < best_c:
                best_c, best_h, best_n = c8c, h8c, n8
        # Greedy on best n=800
        if best_n == n8 and time.time() < end - 40:
            hb, cb = greedy_bin(best_h, n8, min(end - 20, time.time() + 40))
            if cb < best_c:
                best_c, best_h, best_n = cb, hb, n8

    # Phase 4: Final L-BFGS polish
    if time.time() < end - 30:
        hh, cc = polish(best_h, best_n, [10000, 50000], end - 15, 20.0)
        if cc < best_c:
            best_c, best_h, best_n = cc, hh, best_n

    # Final: also try greedy on best
    if best_n % 2 == 0 and time.time() < end - 10:
        hb, cb = greedy_bin(best_h, best_n, end - 5)
        if cb < best_c:
            best_c, best_h = cb, hb

    # Also try rounding to binary directly
    if best_n % 2 == 0:
        tb = top_n_bin(best_h, best_n)
        ct = compute_c5(tb, best_n)
        if ct < best_c:
            best_c, best_h = ct, tb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
