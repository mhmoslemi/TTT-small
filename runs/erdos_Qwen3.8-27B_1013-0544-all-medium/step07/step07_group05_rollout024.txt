def softmax_w(c, beta):
    mc = float(np.max(c))
    z = np.clip(beta * (c - mc), -745.0, 0.0)
    ew = np.exp(z)
    s = float(ew.sum())
    if s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / s
    return w

def grad_weighted(h, w, n):
    dx = 2.0 / n
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid + 1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)

    kern[mid + 1:] = s[1:]

def hinge_grad(h, n, T, gamma):
    """Gradient of sum_k max(0, c[k]-T)^gamma / gamma"""
    c = get_corr(h, n)
    excess = np.maximum(0.0, c - T)
    if excess.max() <= 0:
        return np.zeros(n)
    # Weight = excess^(gamma-1) for active constraints
    if gamma > 1:
        w_active = excess ** (gamma - 1.0)
    else:
        w_active = np.ones_like(excess) * (excess > 0).astype(float)
    w = np.zeros_like(c)
    w[c > T] = w_active[c > T]
    return grad_weighted(h, w, n)

    w = softmax_w(c, beta)

    w = softmax_w(c, beta)

        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, nold), h)
    h = project_box_sum(h, 0.5 * n)
    best_h, best_n, best_c = h.copy(), n, erdos_c5(h, n)

    # Phase 1: Initial L-BFGS at current n
    end1 = t0 + 0.20 * total
    for beta in [200.0, 2000.0, 20000.0, 200000.0]:
        if time.time() > end1 - 5.0:
            break
        cand, c = refine_lbfgs(best_h, best_n, beta, 120, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    # Phase 2: Try different n values via resampling
    end2 = t0 + 0.42 * total
    for nh in [500, 600, 700, 350]:
        if time.time() > end2 - 20.0 or nh == best_n:
            continue
        hh = resample(best_h, best_n, nh)
        for beta in [1000.0, 10000.0, 100000.0]:
            if time.time() > end2 - 12.0:
                break
            cand, c = refine_lbfgs(hh, nh, beta, 80, deadline)
            hh = cand
        if c < best_c - 1e-15:
            best_h, best_n, best_c = hh, nh, c

    # Phase 3: Iterative refinement loop
    end3 = t0 + 0.78 * total
    iter_count = 0
    while time.time() < end3:
        iter_count += 1
        # L-BFGS with increasing beta
        beta = min(50000.0 * iter_count, 5000000.0)
        cand, c = refine_lbfgs(best_h, best_n, beta, 70, end3)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
        # Hinge penalty focusing on active constraints
        T = best_c * 0.995
        if time.time() < end3 - 30.0:
            cand, c = refine_hinge_lbfgs(best_h, best_n, T, 2.0, 60, end3)
            if c < best_c - 1e-15:
                best_h, best_c = cand, c
        # Pair polish
        if time.time() < end3 - 40.0:
            m_p = 50 if best_n <= 500 else 35
            best_h, best_c = pair_polish(best_h, best_n, end3 - 5.0,
                                         seed + iter_count * 37, m_p, 4)
        # Random perturbation + reoptimize occasionally
        if iter_count % 3 == 0 and time.time() < end3 - 60.0:
            sig = 0.01 + 0.01 * (iter_count % 5)
            hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
            cand, c = refine_lbfgs(hh, best_n, 50000.0, 60, end3)
            if c < best_c - 1e-15:
                best_h, best_c = cand, c

    # Phase 4: Final intensive polish
    end4 = deadline - 5.0
    if time.time() < end4 - 30.0:
        m_p = 60 if best_n <= 500 else 40
        best_h, best_c = pair_polish(best_h, best_n, end4 - 5.0,
                                     seed + 9999, m_p, 6)
    if time.time() < end4 - 15.0:
        cand, c = refine_lbfgs(best_h, best_n, 5000000.0, 40, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
    if time.time() < end4 - 8.0:
        T = best_c * 0.99
        cand, c = refine_hinge_lbfgs(best_h, best_n, T, 2.0, 40, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
    if time.time() < end4 - 4.0:
        best_h, best_c = pair_polish(best_h, best_n, end4, seed + 5555, 40, 4)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
