def obj_and_grad(h, n, alpha, dx):
    C = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max: f = C_max + (1/alpha)*log(sum(exp(alpha*(C-C_max))))
    C_max = C.max()
    shifted = alpha * (C - C_max)
    smax = shifted.max()
    log_sum = smax + np.log(np.exp(shifted - smax).sum())
    f = C_max + log_sum / alpha
    
    # Weights for gradient
    exp_w = np.exp(shifted - smax - log_sum)
    w_sum = exp_w.sum()
    weights = exp_w / w_sum
    
    grad = np.zeros(n)
    for k in range(len(weights)):
        wk = weights[k]
        if wk < 1e-8:
            continue
        shift = k - (n - 1)
        j1s = max(0, -shift)
        j1e = min(n, n - shift)
        if j1e > j1s:
            grad[j1s:j1e] += wk * (1.0 - h[j1s+shift:j1e+shift])
        j2s = max(0, shift)
        j2e = min(n, n + shift)
        if j2e > j2s:
            grad[j2s:j2e] -= wk * h[j2s-shift:j2e-shift]
    
    return f, grad

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        f, _ = obj_and_grad(x, n, alpha, dx)
        return f
    def grad(x):
        _, g = obj_and_grad(x, n, alpha, dx)
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        # Try without gradient
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h

def fw_step(h, n):
    """One Frank-Wolfe step"""
    C = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(C))
    f_val = C[m_star]
    k = m_star - (n - 1)
    
    # Subgradient
    g = np.zeros(n)
    j1s = max(0, -k)
    j1e = min(n, n - k)
    if j1e > j1s:
        g[j1s:j1e] = 1.0 - h[j1s+k:j1e+k]
    j2s = max(0, k)
    j2e = min(n, n + k)
    if j2e > j2s:
        g[j2s:j2e] -= h[j2s-k:j2e-k]
    
    # Linear minimization over {0≤s≤1, sum(s)=n/2}
    target = n / 2.0
    idx = np.argsort(g)
    s = np.zeros(n)
    n_full = int(target)
    s[idx[:n_full]] = 1.0
    frac = target - n_full
    if frac > 1e-10:
        s[idx[n_full]] = frac
    
    # Line search
    best_gamma = 0.0
    best_f = f_val
    for gamma in [1.0, 0.7, 0.5, 0.3, 0.1]:
        h_try = (1 - gamma) * h + gamma * s
        h_try = project_feasible(h_try, n)
        f_try = np.correlate(h_try, 1.0 - h_try, mode='full').max()
        if f_try < best_f - 1e-14:
            best_f = f_try
            best_gamma = gamma
    
    h_new = (1 - best_gamma) * h + best_gamma * s
    h_new = project_feasible(h_new, n)
    return h_new, best_f

def frank_wolfe(h, n, max_iter=3000, tol=1e-11):
    h_curr = h.copy()
    best_h = h.copy()
    best_f = np.correlate(h, 1-h, mode='full').max()
    
    for it in range(max_iter):
        h_new, f_new = fw_step(h_curr, n)
        if f_new < best_f - 1e-14:
            best_f = f_new
            best_h = h_new.copy()
        h_curr = h_new
        # Check convergence
        if it > 0 and f_new >= best_f - tol:
            if it > 100:
                break
    
    return best_h, best_f
