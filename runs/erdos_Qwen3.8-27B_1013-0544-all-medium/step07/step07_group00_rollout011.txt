def coord_cd(h, n, max_passes=100, deadline=None, grid_size=80):
    h = h.copy()
    total_lags = 2 * n - 1
    for iteration in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(total_lags - 1, total_lags - 1 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(total_lags - 1, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            a_coefs = np.zeros(len(ms))
            if np.any(valid_f):
                a_coefs[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a_coefs[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms].copy()
            diag_mask = (ms == (n - 1))
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < total_lags - 1:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            # Grid search
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a_coefs
                if np.any(diag_mask):
                    nv[diag_mask] = nv[diag_mask] - d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            # Parabolic refinement
            if abs(best_d) > 1e-12:
                step = (2.0 / n) / grid_size
                d_fine = np.linspace(max(-h0, best_d - step), min(1.0 - h0, best_d + step), 15)
                for d in d_fine:
                    if abs(d) < 1e-14:
                        continue
                    nv = c_aff + d * a_coefs
                    if np.any(diag_mask):
                        nv[diag_mask] = nv[diag_mask] - d * d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

    # Each c[m] is quadratic in h with max eigenvalue ~ n (roughly)

    step = 1.0 / (alpha + 1)

            break
        grad = smooth_grad(h, n, alpha)
        # Backtracking line search
        f_cur = smooth_obj(h, n, alpha)

            f_new = smooth_obj(h_new, n, alpha)
