    n = x.size

    for _ in range(80):

        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = float(target - y.sum())

        else:

def binary_push(h, n, deadline):
    """Push h toward binary by iteratively setting values closest to 0 or 1."""
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)

        if time.time() > deadline:
            break
        # Find values closest to 0 or 1 (but not already there)
        dist0 = np.where(h < 0.5, h, np.inf)
        dist1 = np.where(h >= 0.5, 1.0-h, np.inf)
        # Combine: for each position, distance to nearest boundary
        dist = np.minimum(dist0, dist1)
        # Sort by distance, try to round the closest ones
        order = np.argsort(dist)
        for idx in order[:10]:
            hh = h.copy()

            nn = int(np.sum(hh > 0.5))  # count of ones (approximate)

            if c < cur - 1e-14:

    n = x.size
