import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def current_c(h, n):
    return float(np.max(compute_c(h, n)))

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    if v.size < n:
        v = np.pad(v, (0, n - v.size), mode="constant", constant_values=0.5)
    elif v.size > n:
        v = v[:n]
    v = np.clip(v, 0.0, 1.0)
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = tgt - v.sum()
    if abs(d) > 1e-13:
        mask = (v > 1e-9) & (v < 1.0 - 1e-9)
        if mask.any():
            v[mask] += d / float(mask.sum())
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def project_sym(v, n):
    m = n // 2
    v = np.asarray(v, dtype=float).reshape(-1)
    y = np.clip(0.5 * (v[:m] + 1.0 - v[n - 1:m - 1:-1]), 0.0, 1.0)
    h = np.empty(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def lse_weights(c, alpha):
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    return e / s

def lse_obj(h, n, alpha):
    c = compute_c(h, n)
    m = float(c.max())
    if alpha <= 0.0:
        return m
    return float(m + np.log(float(np.exp(alpha * (c - m)).sum())) / alpha)

def lse_grad(h, n, alpha):
    c = compute_c(h, n)
    q = lse_weights(c, alpha)
    off = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    w = q[off]
    if w > 0.0:
        g += w * (1.0 - 2.0 * h)
    for r in range(1, n):
        s = n - r
        w = q[off + r]
        if w > 0.0:
            g[r:] += w * (1.0 - h[:s])
            g[:s] -= w * h[r:r + s]
        w = q[off - r]
        if w > 0.0:
            g[:s] += w * (1.0 - h[r:r + s])
            g[r:] -= w * h[:s]
    return g * dx

def free_obj(x, n, alpha, mu):
    d = x.sum() - 0.5 * n
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def free_jac(x, n, alpha, mu):
    return lse_grad(x, n, alpha) + mu * (x.sum() - 0.5 * n)

def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x)

def slsqp_free(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        r = minimize(lse_obj, h, args=(n, alpha), method="SLSQP", jac=lse_grad,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{"type": "eq", "fun": sum_fun, "args": (n,), "jac": sum_jac}],
                     options={"maxiter": maxiter, "ftol": 1e-12})
        if r.x is not None:
            return project_box_sum(r.x, n)
    except Exception:
        pass
    return h

def lbfgs_free(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        r = minimize(free_obj, h, args=(n, alpha, mu), method="L-BFGS-B", jac=free_jac,
                     bounds=[(0.0, 1.0)] * n,
                     options={"maxiter": maxiter, "maxfun": 4 * maxiter, "ftol": 1e-14, "gtol": 1e-8})
        if r.x is not None:
            return project_box_sum(r.x, n)
    except Exception:
        pass
    return h

def sym_build(y, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def sym_obj(y, n, alpha):
    return lse_obj(sym_build(y, n), n, alpha)

def sym_jac(y, n, alpha):
    h = sym_build(y, n)
    g = lse_grad(h, n, alpha)
    m = n // 2
    return g[:m] - g[n - 1:m - 1:-1]

def lbfgs_sym(y, n, alpha, maxiter):
    y = np.clip(np.asarray(y, dtype=float).reshape(-1), 0.0, 1.0)
    try:
        r = minimize(sym_obj, y, args=(n, alpha), method="L-BFGS-B", jac=sym_jac,
                     bounds=[(0.0, 1.0)] * len(y),
                     options={"maxiter": maxiter, "maxfun": 4 * maxiter, "ftol": 1e-14, "gtol": 1e-8})
        if r.x is not None:
            return np.clip(r.x, 0.0, 1.0)
    except Exception:
        pass
    return y

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def pair_xs(lo, hi, x0):
    xs = np.array([lo, hi, x0], dtype=float)
    if hi - lo > 1e-12:
        xs = np.concatenate((xs, np.linspace(lo, hi, 18)))
    a = max(lo, x0 - 0.04)
    b = min(hi, x0 + 0.04)
    if b - a > 1e-12:
        xs = np.concatenate((xs, np.linspace(a, b, 64)))
    xs = np.unique(xs)
    xs = xs[(xs >= lo - 1e-12) & (xs <= hi + 1e-12)]
    return np.clip(xs, lo, hi)

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i + n] += ax1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += ax2[:, None] * h[None, :]
    delta[:, j:j + n] += ax2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += ax1[:, None] * h[None, :]
    idx0 = off
    base_self = ax1 * (1.0 - x0) + x0 * ax2 + ax2 * (1.0 - y0) + y0 * ax1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, idx0] += exact_self - base_self
    d = j - i
    idxp = off - d
    base_p = ax1 * (1.0 - y0) + x0 * ax1
    exact_p = xs * (1.0 - yv) - x0 * (1.0 - y0)
    delta[:, idxp] += exact_p - base_p
    idxm = off + d
    base_m = ax2 * (1.0 - x0) + y0 * ax2
    exact_m = yv * (1.0 - xs) - y0 * (1.0 - x0)
    delta[:, idxm] += exact_m - base_m
    return np.max(c_base[None, :] + dx * delta, axis=1)

def pair_search(h, c_base, i, j, n):
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = pair_xs(lo, hi, x0)
    vals = pair_eval(h, c_base, i, j, xs, n, x0, y0, S)
    for _ in range(2):
        if len(xs) <= 1:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k - 1], xs[k + 1]
        if b - a < 1e-11:
            break
        xs2 = np.linspace(a, b, 48)
        vals = pair_eval(h, c_base, i, j, xs2, n, x0, y0, S)
        xs = xs2
    k = int(np.argmin(vals))
    if float(vals[k]) < float(c_base.max()) - 1e-14:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(np.clip(S - xb, 0.0, 1.0))
        xb = float(np.clip(S - yb, 0.0, 1.0))
        return xb, yb
    return None

def choose_pair(h, c, n, rng):
    off = n - 1
    if n < 2:
        return 0, 0
    if rng.random() < 0.88:
        pos = c[off + 1:off + n]
        neg = c[:off][::-1]
        w = np.maximum(pos, neg) + 1e-12
        w /= w.sum()
        d = int(rng.choice(n - 1, p=w)) + 1
    else:
        d = int(rng.randint(1, n))
    s = max(1, n - d)
    contrib = h[:s] * (1.0 - h[d:]) + h[d:] * h[:s]
    tot = float(contrib.sum())
    if tot > 1e-15:
        i = int(rng.choice(s, p=contrib / tot))
    else:
        i = int(rng.randint(0, s))
    return i, i + d

def pair_local_search(h, deadline, seed=1):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(np.max(c))
    besth = h.copy()
    if n < 2 or time.time() > deadline:
        return besth, current_c(besth, n)
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 900 if n > 250 else 1800
    cnt = 0
    while time.time() < deadline - 0.5:
        if noimp > maxno:
            break
        i, j = choose_pair(h, c, n, rng)
        res = pair_search(h, c, i, j, n)
        if res is not None:
            h[i] = res[0]
            h[j] = res[1]
            cnt += 1
            if cnt % 50 == 0:
                h = project_box_sum(h, n)
            c = compute_c(h, n)
            v = float(np.max(c))
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    return besth, current_c(besth, n)

def update_best(bh, bc, ch, cc, hv, v):
    if v < bc:
        bh, bc = hv.copy(), v
    if v <= cc + 1e-5:
        ch, cc = hv.copy(), v
    else:
        ch, cc = bh.copy(), bc
    return bh, bc, ch, cc

def local_refine(h, n, seed, deadline, use_sym=False):
    h = project_box_sum(h, n)
    bh = h.copy()
    bc = current_c(h, n)
    ch = h.copy()
    cc = bc
    now = time.time()
    total = max(0.0, deadline - now)
    if use_sym and n % 2 == 0 and total > 25.0:
        y = ch[:n // 2].copy()
        dl = now + 0.18 * total
        for alpha, mx in ((5, 80), (20, 60), (80, 50), (300, 40)):
            if time.time() > dl:
                break
            y = lbfgs_sym(y, n, alpha, mx)
            h2 = sym_build(y, n)
            v = current_c(h2, n)
            bh, bc, ch, cc = update_best(bh, bc, ch, cc, h2, v)
    dl = now + 0.42 * total
    for alpha, mu, mx in ((5, 1, 80), (20, 10, 60), (80, 100, 50), (300, 1000, 40)):
        if time.time() > dl:
            break
        h2 = lbfgs_free(ch, n, alpha, mu, mx)
        v = current_c(h2, n)
        bh, bc, ch, cc = update_best(bh, bc, ch, cc, h2, v)
    dl = now + 0.58 * total
    for alpha, mx in ((30, 40), (200, 30)):
        if time.time() > dl:
            break
        h2 = slsqp_free(ch, n, alpha, mx)
        v = current_c(h2, n)
        bh, bc, ch, cc = update_best(bh, bc, ch, cc, h2, v)
    reserve = 20.0 if total > 100.0 else max(1.0, 0.2 * total)
    pend = deadline - reserve
    if time.time() < pend - 3.0:
        h2, v = pair_local_search(ch, pend, seed)
        bh, bc, ch, cc = update_best(bh, bc, ch, cc, h2, v)
    if time.time() < deadline - 8.0:
        for alpha, mx in ((300, 30), (800, 20)):
            if time.time() > deadline - 3.0:
                break
            h2 = slsqp_free(ch, n, alpha, mx)
            v = current_c(h2, n)
            bh, bc, ch, cc = update_best(bh, bc, ch, cc, h2, v)
    if time.time() < deadline - 1.5:
        h2, v = pair_local_search(ch, deadline - 0.2, seed + 17)
        bh, bc, ch, cc = update_best(bh, bc, ch, cc, h2, v)
    bh = project_box_sum(bh, n)
    return bh, current_c(bh, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    span = max(10.0, float(budget_s) - 10.0)
    deadline = t0 + float(budget_s) - 3.0
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(hp))
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.45 * np.sin(np.pi * x) + 0.05 * np.sin(2.0 * np.pi * x)
    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = current_c(besth, n0)
    bestn = n0
    rng = np.random.RandomState(seed)

    dl1 = t0 + span * 0.42
    h2, c2 = local_refine(besth, bestn, seed, dl1, False)
    if c2 < bestc:
        besth, bestc = h2, c2

    if bestn % 2 == 0 and time.time() < t0 + span * 0.78:
        h2 = project_sym(besth, bestn)
        h2, c2 = local_refine(h2, bestn, seed + 1, t0 + span * 0.78, True)
        if c2 < bestc:
            besth, bestc = h2, c2

    if time.time() < t0 + span * 0.94:
        nalt = min(700, max(128, ((int(bestn * 0.7)) // 2) * 2))
        if nalt != bestn:
            h3 = interpolate_h(besth, bestn, nalt)
            h3 = project_box_sum(h3 + 0.004 * rng.randn(nalt), nalt)
            h3, c3 = local_refine(h3, nalt, seed + 2, t0 + span * 0.94, False)
            if c3 < bestc:
                besth, bestc, bestn = h3, c3, nalt

    if time.time() < deadline - 2.0:
        h4, c4 = local_refine(besth, bestn, seed + 3, deadline, False)
        if c4 < bestc:
            besth, bestc = h4, c4

    besth = project_box_sum(besth, bestn)
    return besth, current_c(besth, bestn), bestn
