    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd(h, n, max_passes=100, deadline=None):
    """Non-symmetric coordinate descent."""
    h = h.copy()
    best_c = compute_c5(h, n)
    n1 = 2*n - 1
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            lo = -h0
            hi = 1.0 - h0
            k_range = np.arange(n1)
            f = p + k_range - (n-1)
            b = p - k_range + (n-1)
            vf = (f >= 0) & (f < n)
            vb = (b >= 0) & (b < n)
            a = np.zeros(n1)
            a[vf] = 1.0 - h[f[vf]]
            a[vb] -= h[b[vb]]
            
            # Coarse
            dc = np.linspace(lo, hi, 25)
            vals = c[None,:] + dc[:,None]*a[None,:]
            vals[:, n-1] -= dc**2
            mc = vals.max(axis=1)
            bi = np.argmin(mc)
            if mc[bi] >= best_c:
                continue
            # Fine
            span = (hi-lo)/25.0
            db = dc[bi]
            df = np.linspace(max(lo, db-3*span), min(hi, db+3*span), 40)
            vf_ = c[None,:] + df[:,None]*a[None,:]
            vf_[:, n-1] -= df**2
            mf = vf_.max(axis=1)
            bf = np.argmin(mf)
            h[p] = np.clip(h0 + df[bf], 0.0, 1.0)
            improved = True
        
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    
    return h, best_c

def coord_cd_sym(h_half, n, max_passes=100, deadline=None):
    """Symmetric coordinate descent: h[i+n/2] = 1-h[i]."""
    n_half = n // 2
    h_full = np.empty(n)
    h_full[:n_half] = h_half
    h_full[n_half:] = 1.0 - h_half
    best_c = compute_c5(h_full, n)
    n1 = 2*n - 1
    k_range = np.arange(n1)
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h_full, 1.0 - h_full, mode='full')
        improved = False
        
        for p in range(n_half):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h_half[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            lo = -h0
            hi = 1.0 - h0
            p2 = p + n_half
            
            f1 = p + k_range - (n-1)
            b1 = p - k_range + (n-1)
            v1f = (f1 >= 0) & (f1 < n)
            v1b = (b1 >= 0) & (b1 < n)
            A1 = np.zeros(n1)
            A1[v1f] = 1.0 - h_full[f1[v1f]]
            A1[v1b] -= h_full[b1[v1b]]
            
            f2 = p2 + k_range - (n-1)
            b2 = p2 - k_range + (n-1)
            v2f = (f2 >= 0) & (f2 < n)
            v2b = (b2 >= 0) & (b2 < n)
            A2 = np.zeros(n1)
            A2[v2f] = 1.0 - h_full[f2[v2f]]
            A2[v2b] -= h_full[b2[v2b]]
            
            S = A1 - A2
            
            dc = np.linspace(lo, hi, 25)
            vals = c[None,:] + dc[:,None]*S[None,:]
            vals[:, n-1] -= 2.0*dc**2
            mc = vals.max(axis=1)
            bi = np.argmin(mc)
            if mc[bi] >= best_c:
                continue
            span = (hi-lo)/25.0
            db = dc[bi]
            df = np.linspace(max(lo, db-3*span), min(hi, db+3*span), 40)
            vfr = c[None,:] + df[:,None]*S[None,:]
            vfr[:, n-1] -= 2.0*df**2
            mfr = vfr.max(axis=1)
            bf = np.argmin(mfr)
            h_half[p] = np.clip(h0 + df[bf], 0.0, 1.0)
            h_full[p] = h_half[p]
            h_full[p2] = 1.0 - h_half[p]
            improved = True
        
        if not improved:
            break
    
    return h_half, best_c

def smooth_obj_sym(x, n, alpha):
    """Smoothed objective for symmetric parameterization."""
    n_half = n // 2
    h = np.empty(n)
    h[:n_half] = x
    h[n_half:] = 1.0 - x
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse/alpha) * (2.0/n)

def smooth_obj(x, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse/alpha) * (2.0/n)

def project_feasible(h, n):

            if tot < 1e-15:
