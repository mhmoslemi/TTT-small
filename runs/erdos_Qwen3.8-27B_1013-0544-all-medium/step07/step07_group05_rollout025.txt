def softmax_weights(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -700.0, 0.0, out=z)
    w = np.exp(z)
    ws = float(w.sum())
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    return w

def grad_max(h, w, n):
    """Gradient of sum_d w_d * c_d, using FFT convolution."""
    conv = np.fft.irfft(np.fft.rfft(h, 4 * n) * np.fft.rfft(w, 4 * n))
    q = conv[n - 1:2 * n - 1]
    return 1.0 - 2.0 * q

def proj_grad_optimize(h, n, beta_start, beta_end, max_iters, deadline, momentum=0.9):
    """Nesterov-accelerated projected gradient with adaptive softmax beta."""
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = 0.5 * n
    best_h = h.copy()
    best_c = erdos_c5(h, n)
    x = h.copy()
    y_vec = h.copy()
    for it in range(max_iters):
        rem = deadline - time.time()
        if rem < 2.0:
            break
        frac = (it + 1) / max(max_iters, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        c = get_c(x, n)
        w = softmax_weights(c, beta)
        g = grad_max(x, w, n)
        alpha = 0.4 / (1.0 + 0.1 * beta)
        y_new = y_vec - alpha * g
        y_new = project_box_sum(y_new, target)
        c_new = erdos_c5(y_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = y_new.copy()
        y_vec = y_new + momentum * (y_new - x)
        x = y_new
    return best_h, best_c

def lbfgs_objective(h_flat, n, beta):
    h = np.asarray(h_flat, dtype=float).reshape(-1)
    c = get_c(h, n)
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -700.0, 0.0, out=z)
    w = np.exp(z)
    ws = float(w.sum())
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    obj = maxc + np.log(ws) / beta
    g = grad_max(h, w, n)
    return obj, g

def lbfgs_refine(h, n, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    def fun(gv):
        return lbfgs_objective(gv, n, beta)[0]
    def jac(gv):
        return lbfgs_objective(gv, n, beta)[1]
    mx = int(min(maxiter, max(10, int(rem / 4.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(fun, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxls': 30, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

def pair_polish(h, n, deadline, seed):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    m = min(20, n)
    if h.size < 2:
        return h, cur
    frac_idx = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac_idx.size:
        score = np.minimum(h[frac_idx], 1.0 - h[frac_idx])
        order = frac_idx[np.argsort(score)[::-1]]
    else:
        order = np.arange(n)
    if order.size >= m:
        top = order[:m]
    else:
        extra = rng.choice(n, size=min(m - int(order.size), n), replace=False)
        top = np.unique(np.concatenate((order, extra)))
    for sweep in range(3):
        improved = False
        pairs = [(int(top[a]), int(top[b])) for a in range(len(top)) for b in range(a + 1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 3.0:
                return h, erdos_c5(h, n)
            hi0, hj0 = h[i], h[j]
            lo_d = max(-hi0, hj0 - 1.0)
            hi_d = min(1.0 - hi0, hj0)
            if hi_d - lo_d < 1e-12:
                continue
            best_d, best_c = 0.0, cur
            for frac in (0.25, 0.5, 0.75):
                d = lo_d + frac * (hi_d - lo_d)
                hh = h.copy()
                hh[i] += d
                hh[j] -= d
                cc = erdos_c5(hh, n)
                if cc < best_c:
                    best_c, best_d = cc, d
            if best_c < cur - 1e-14:
                h[i] += best_d
                h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)
                cur = best_c
                improved = True
        if not improved:
            break
    return h, erdos_c5(h, n)

def upsample(h, n, nh):
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nh)
    return project_box_sum(np.interp(y, x, h), 0.5 * nh)

def make_sinusoidal(n, freq, amp, phase=0.0):
    x = np.arange(n, dtype=float)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_box_sum(h, 0.5 * n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        h0 = np.full(400, 0.5)
        n0 = 400
    n_target = 800
    h = upsample(h0, n0, n_target) if n0 != n_target else project_box_sum(h0, 0.5 * n_target)
    n = n_target
    best_h = h.copy()
    best_c = erdos_c5(best_h, n)
    e1 = t0 + 0.30 * total
    e2 = t0 + 0.55 * total
    e3 = t0 + 0.78 * total
    e4 = deadline - 5.0
    # Phase 1: Nesterov projected gradient with increasing beta
    if time.time() < e1:
        for b_lo, b_hi, iters in [(50, 500, 200), (500, 5000, 200), (5000, 50000, 150)]:
            if time.time() > e1 - 2.0:
                break
            cand, c = proj_grad_optimize(best_h, n, b_lo, b_hi, iters, deadline, momentum=0.85)
            if c < best_c - 1e-15:
                best_h, best_c = cand, c
    # Phase 2: LBFGS polish with high beta
    if time.time() < e2:
        for beta in [1000.0, 10000.0, 100000.0, 1000000.0]:
            if time.time() > e2 - 3.0:
                break
            cand, c = lbfgs_refine(best_h, n, beta, 60, deadline)
            if c < best_c - 1e-15:
                best_h, best_c = cand, c
    # Phase 3: Pair polish
    if time.time() < e3 - 10.0:
        best_h, best_c = pair_polish(best_h, n, e3 - 5.0, seed)
    # Phase 4: Try alternative starting points
    rng = np.random.RandomState(int(seed) + 7)
    starts = []
    for freq in [2, 3, 5, 8]:
        for amp in [0.15, 0.3]:
            starts.append(make_sinusoidal(n, freq, amp, rng.uniform(0, 2 * np.pi)))
    for sig in [0.03, 0.08]:
        starts.append(project_box_sum(best_h + sig * rng.randn(n), 0.5 * n))
    for st in starts:
        if time.time() > e4 - 30.0:
            break
        cand, c = proj_grad_optimize(st, n, 200, 20000, 80, deadline, momentum=0.8)
        if c < best_c - 1e-12:
            cand, c = lbfgs_refine(cand, n, 50000.0, 40, deadline)
            if c < best_c - 1e-14:
                best_h, best_c = cand, c
    # Phase 5: Final polish
    if time.time() < e4 - 15.0:
        cand, c = lbfgs_refine(best_h, n, 200000.0, 30, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
    if time.time() < e4 - 8.0:
        best_h, best_c = pair_polish(best_h, n, e4 - 3.0, seed + 1)
    # Try downsampling if beneficial
    if time.time() < e4 - 10.0:
        for nh in [400, 600]:
            if time.time() > e4 - 5.0:
                break
            hh = upsample(best_h, n, nh)
            hh, c = lbfgs_refine(hh, nh, 50000.0, 30, deadline)
            if c < best_c - 1e-14:
                best_h, best_c, n = hh, c, nh
    best_h = project_box_sum(best_h, 0.5 * n)
    best_c = erdos_c5(best_h, n)
    return best_h, best_c, int(n)
