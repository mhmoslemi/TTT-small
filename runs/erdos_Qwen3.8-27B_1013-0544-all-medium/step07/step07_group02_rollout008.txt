    mc = float(C.max())

        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,

            solved = True

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -80, 80)
    return m + np.log(np.exp(shifted).sum()) / alpha

def lse_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -80, 80)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-13:
            continue
        k = k_idx - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            grad[j1s:j1e] += w[k_idx] * (1.0 - x[j1s + k:j1e + k])
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            grad[j2s:j2e] -= w[k_idx] * x[j2s - k:j2e - k]
    return grad

def lse_lbbfgs(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, jac=lambda x: lse_grad(x, n, alpha),
                       method='L-BFGS-B', bounds=bounds,
                       args=(n, alpha),
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps, delta <= eps,
        delta >= -h, delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.5, 0.25, 0.1]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

                stall = 0
        h = h_new
