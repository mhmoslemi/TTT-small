def coord_optimize(h, n, i, c):
    """Optimize coordinate i given current correlation c."""
    t0 = h[i]
    dx = 2.0 / n
    off = n - 1
    
    def eval_at(t):
        # Compute new max correlation
        # c_0 changes quadratically
        new_c0 = c[off] + t*(1-t) - t0*(1-t0)
        # For other lags, the change is linear
        # For lag k>0 (index off+k): 
        #   if i <= n-1-k: h[i](1-h[i+k]) -> t(1-h[i+k])
        #   if i >= k: h[i-k](1-h[i]) -> h[i-k](1-t)
        # For lag k<0 (index off+k, k negative, let s=-k):
        #   if i >= s: h[i](1-h[i-s]) -> t(1-h[i-s])  
        #   if i <= n-1-s: h[i+s](1-h[i]) -> h[i+s](1-t)
        
        # This is O(n) which is fine
        new_c = c.copy()
        # Lag 0
        new_c[off] = c[off] + t*(1-t) - t0*(1-t0)
        for k in range(1, n):
            # Positive lag k
            if i <= n-1-k:
                new_c[off+k] = c[off+k] + (t-t0)*(1-h[i+k])
            if i >= k:
                new_c[off+k] = c[off+k] + h[i-k]*(t0-t)
            # Negative lag -k
            if i >= k:
                new_c[off-k] = c[off-k] + (t-t0)*(1-h[i-k])
            if i <= n-1-k:
                new_c[off-k] = c[off-k] + h[i+k]*(t0-t)
        return float(new_c.max()) * dx
    
    # Evaluate at a few points
    candidates = [0.0, 0.25, 0.5, 0.75, 1.0, t0]
    best_t = t0
    best_val = eval_at(t0)
    for t in candidates:
        v = eval_at(t)
        if v < best_val:
            best_val = v
            best_t = t
    # Refine around best
    lo = max(0, best_t - 0.1)
    hi = min(1, best_t + 0.1)
    for _ in range(10):
        m1 = lo + (hi-lo)/3
        m2 = hi - (hi-lo)/3
        if eval_at(m1) < eval_at(m2):
            hi = m2
        else:
            lo = m1
    best_t = (lo+hi)/2
    return best_t

def coord_eval(h, n, i, t, c):
    """Evaluate new max correlation if h[i] changes to t."""
    t0 = h[i]
    if abs(t - t0) < 1e-15:
        return c.max()
    dx = 2.0 / n
    off = n - 1
    delta = t - t0
    # New c[off] (lag 0)
    new_c0 = c[off] + t*(1-t) - t0*(1-t0)
    # For k=1 to n-1:
    # new_c[off+k]: 
    #   contribution from first factor: delta * (1-h[i+k]) if i+k < n
    #   contribution from second factor: -delta * h[i-k] if i-k >= 0
    # new_c[off-k]:
    #   contribution from first factor: delta * (1-h[i-k]) if i-k >= 0
    #   contribution from second factor: -delta * h[i+k] if i+k < n

    for k in range(1, n):
        dp = 0.0
        dn = 0.0

            dn += -delta * h[i+k]

            dn += delta * (1.0 - h[i-k])

        v2 = c[off-k] + dn
        if v1 > new_max: new_max = v1
        if v2 > new_max: new_max = v2
    return new_max * dx

        # and D_k is the effect of changing h[j] by 1 unit
        # This is linear in t (delta_i)! Plus the k=0 term which is quadratic.
        
        # Effect of changing h[i] by delta:
        # c[off] += delta*(1-t0_i) + ... wait, this is getting complicated with two variables.
        # Let me just brute-force evaluate.
        
        # For a candidate t, the new h has h[i]=t, h[j]=S-t
        # The change in c[off] (lag 0): (t-x0)(1-t) + ((S-t)-y0)(1-(S-t)) - (x0-y0)(1-x0) ... no
        # c[off] = sum_k h[k](1-h[k]), so changing h[i] from x0 to t and h[j] from y0 to S-t:
        # delta_c0 = t(1-t) - x0(1-x0) + (S-t)(1-(S-t)) - y0(1-y0)
        # This is quadratic in t.
        
        # For c[off+k] (k≠0): linear in t (since h[i] and h[j] each appear at most once)
        # unless |i-j| = k, in which case they might interact.
        
        # Simplest: just evaluate at a few points
        ts = np.linspace(lo_t, hi_t, 50)
        best_t = x0
        best_v = float(c.max())
        for t in ts:
            h2 = h.copy()
            h2[i] = t
            h2[j] = S - t
            c2 = compute_c(h2, n)
            v = float(c2.max())
            if v < best_v:
                best_v = v
                best_t = t
        # Refine
        if best_t != x0:
            a = max(lo_t, best_t - 0.02)
            b = min(hi_t, best_t + 0.02)
            ts2 = np.linspace(a, b, 30)
            for t in ts2:
                h2 = h.copy()
                h2[i] = t
                h2[j] = S - t
                v = float(compute_c(h2, n).max())
                if v < best_v:
                    best_v = v
                    best_t = t
            h[i] = best_t
            h[j] = S - best_t
            h = project_box_sum(h, n)
            if best_v < bestc:
                bestc = best_v
                besth = h.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1
    
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc
