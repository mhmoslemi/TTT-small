import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_logsumexp(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n//2]] = 1.0
    return h_b

def compute_vecs(h, n):
    """Precompute set_vecs and clear_vecs for binary h."""
    set_vecs = np.zeros((n, 2*n-1))
    clear_vecs = np.zeros((n, 2*n-1))
    one_minus_h = 1.0 - h
    h_rev = h[::-1].copy()
    for p in range(n):
        # p as first index
        s1 = n - 1 - p
        e1 = 2 * n - p - 1
        set_vecs[p, s1:e1] += one_minus_h
        clear_vecs[p, s1:e1] -= one_minus_h
        # p as second index
        s2 = p
        e2 = n + p
        set_vecs[p, s2:e2] -= h_rev
        clear_vecs[p, s2:e2] += h_rev
    return set_vecs, clear_vecs

def binary_local_search(h_init, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h_b = to_binary(h_init, n)
    corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h_b.copy()
    
    ones_list = list(np.where(h_b == 1)[0])
    zeros_list = list(np.where(h_b == 0)[0])
    
    stall = 0
    stall_limit = 100
    sweep = 0
    while sweep < max_sweeps:
        if time.time() > deadline:
            break
        sweep += 1
        # Precompute vectors
        set_vecs, clear_vecs = compute_vecs(h_b, n)
        improved = False
        
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            d_i = corr + set_vecs[i]
            # For each j in ones_list, compute max(d_i + clear_vecs[j])
            new_maxes = (d_i[None, :] + clear_vecs[ones_list]).max(axis=1)
            best_idx = new_maxes.argmin()
            best_val = new_maxes[best_idx]
            
            if best_val < cur_max - 1e-12:
                j = ones_list[best_idx]
                # Perform swap
                corr = d_i + clear_vecs[j]
                cur_max = best_val
                h_b[i] = 1.0
                h_b[j] = 0.0
                ones_list.remove(j)
                ones_list.append(i)
                zeros_list.remove(i)
                zeros_list.append(j)
                improved = True
                # Recompute vectors (since h changed)
                set_vecs, clear_vecs = compute_vecs(h_b, n)
                if cur_max * dx < best_c:
                    best_c = cur_max * dx
                    best_h = h_b.copy()
        
        if not improved:
            stall += 1
            if stall > stall_limit:
                # Escape local optimum: random perturbation
                rng = np.random.RandomState(sweep)
                n_flips = max(1, n // 20)
                flip_zeros = rng.choice(zeros_list, size=min(n_flips, len(zeros_list)), replace=False)
                flip_ones = rng.choice(ones_list, size=min(n_flips, len(ones_list)), replace=False)
                for k in range(len(flip_zeros)):
                    i, j = flip_zeros[k], flip_ones[k]
                    h_b[i], h_b[j] = h_b[j], h_b[i]
                    ones_list.remove(j)
                    ones_list.append(i)
                    zeros_list.remove(i)
                    zeros_list.append(j)
                corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
                cur_max = corr.max()
                stall = 0
        else:
            stall = 0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_logsumexp(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h_init, n, deadline, seed=42):
    dx = 2.0 / n

    h_b = to_binary(h_init, n)

    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h_b.copy()
    ones = list(np.where(h_b == 1)[0])

    h_rev = h_b[::-1].copy()
    stall = 0
    stall_limit = 80
    sweep = 0
    max_sweeps = 5000
    while sweep < max_sweeps:
        if time.time() > deadline:
            break
        sweep += 1
        improved = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            if h_b[i] != 0.0:
                continue
            delta = np.zeros(ncorr)
            delta[n - 1 - i: 2 * n - i - 1] += (1.0 - h_b)
            delta[i: n + i] -= h_b[::-1]
            d = corr + delta
            best_j = -1
            best_val = cur_max
            for j in ones:
                if h_b[j] != 1.0:
                    continue
                total = d.copy()
                total[n - 1 - j: 2 * n - j - 1] -= (1.0 - h_b)
                total[j: n + j] += h_b[::-1]
                vm = total.max()
                if vm < best_val - 1e-12:
                    best_val = vm
                    best_j = j
            if best_j >= 0:

                corr[best_j: n + best_j] += h_b[::-1]
                cur_max = best_val
                h_b[i] = 1.0
                h_b[best_j] = 0.0

                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
                if cur_max * dx < best_c:
                    best_c = cur_max * dx
                    best_h = h_b.copy()
        if not improved:

            if stall > stall_limit:

                        ones.append(ii)

                    cur_max = corr.max()
                stall = 0
        else:
            stall = 0
    return best_h, best_c
