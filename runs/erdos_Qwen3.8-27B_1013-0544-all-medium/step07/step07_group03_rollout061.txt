import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_to_binary(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n//2]] = 1.0
    return h_b

The combined delta for swap (i,j) is:
delta(i,j)[k] = flip_delta_i[k] + unflip_delta_j[k] + 2*[k==j-i]

And new_corr[k] = corr[k] + delta(i,j)[k]

The new max is max_k new_corr[k].

For the greedy approach (one i at a time):
- Compute flip_delta_i for the current i
- Find j that minimizes max_k(corr[k] + flip_delta_i[k] + unflip_delta_j[k])
- This is: find j that minimizes max_k((corr[k] + flip_delta_i[k]) + unflip_delta_j[k])
- Let base[k] = corr[k] + flip_delta_i[k]
- Then: find j that minimizes max_k(base[k] + unflip_delta_j[k])

I can precompute unflip_delta for all j at once (it's the B matrix). Then for each i, I just need to find the j that minimizes max_k(base[k] + B[j,k]).

    c = np.correlate(h, 1 - h, mode='full')

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n//2]] = 1.0
    return h_b

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h
