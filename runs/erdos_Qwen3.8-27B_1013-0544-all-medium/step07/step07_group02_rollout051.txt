    return J - I + n - 1, I - J + n - 1

def soft_gradient(h, n, w, idx1, idx2):
    return w[idx1] @ (1.0 - h) - w[idx2] @ h

def adam_smooth(h, n, budget, idx1, idx2, lr=0.01, alpha_start=5.0, alpha_end=200.0):
    t_end = time.time() + budget
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999
    n_total = max(1, int((t_end - time.time()) / 0.02))
    while time.time() < t_end:
        progress = 1.0 - (t_end - time.time()) / max(budget, 1e-6)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        cur_lr = lr * (0.01 ** progress)
        c = np.correlate(h, 1.0 - h, mode='full')
        mc = c.max()
        w = np.exp(alpha * (c - mc))
        w /= w.sum()
        g = soft_gradient(h, n, w, idx1, idx2)
        t += 1
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1h = m1 / (1 - b1 ** t)
        m2h = m2 / (1 - b2 ** t)
        h = h - cur_lr * m1h / (np.sqrt(m2h) + 1e-8)
        h = project_feasible(h, n)
    return h

def build_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        m = m_idx - (n - 1)
        i_lo = max(0, -m)
        i_hi = min(n - 1, n - 1 - m)
        if i_hi >= i_lo:
            G[i_lo:i_hi + 1, m_idx] = 1.0 - h[i_lo + m:i_hi + m + 1]
        i_lo2 = max(0, m)
        i_hi2 = min(n - 1, m + n - 1)
        if i_hi2 >= i_lo2:
            G[i_lo2:i_hi2 + 1, m_idx] -= h[i_lo2 - m:i_hi2 - m + 1]
    return G

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = build_G(h, n)
    delta = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        C + G.T @ delta <= t_var,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0,
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solved = False
    for sol in ['GLOP', 'GLPK', 'ECOS', 'CLARABEL', 'SCS']:
        try:
            prob.solve(solver=sol)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
                break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = np.asarray(delta.value).flatten()
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_mc = float(np.correlate(h, 1.0 - h, mode='full').max())
    best_h = h.copy()
    best_c5 = best_mc * 2.0 / n
    idx1, idx2 = build_indices(n)

    # Phase 1: LP optimization with aggressive schedule (35%)
    p1_dl = t0 + 0.35 * total
    h_curr = best_h.copy()
    eps = 0.04
    no_improve = 0
    while time.time() < p1_dl - 2:
        h_new, mc_new = lp_step(h_curr, n, eps)
        if mc_new < best_mc - 1e-12:
            best_mc = mc_new
            best_h = h_new.copy()
            best_c5 = best_mc * 2.0 / n
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 12:
                eps *= 0.65
                if eps < 0.0003:
                    break
                no_improve = 0
        h_curr = h_new

    # Phase 2: Adam-based smooth optimization (20%)
    p2_dl = t0 + 0.55 * total
    for lr_try in [0.015, 0.005, 0.002]:
        if time.time() > p2_dl - 5:
            break
        budget_phase = min(50, p2_dl - time.time() - 3)
        if budget_phase < 8:
            break
        h_try = adam_smooth(best_h, n, budget_phase, idx1, idx2, lr=lr_try)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-10:
            best_mc = mc_try
            best_h = h_try.copy()
            best_c5 = best_mc * 2.0 / n

    # Phase 3: LP with noise restarts (20%)
    p3_dl = t0 + 0.75 * total
    for trial in range(30):
        if time.time() > p3_dl - 8:
            break
        noise_scale = 0.008 + 0.012 * rng.rand()
        h_try = best_h + noise_scale * rng.randn(n)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc:
            best_mc = mc_try
            best_h = h_try.copy()
            best_c5 = best_mc * 2.0 / n
        eps_r = 0.025
        for _ in range(80):
            if time.time() > p3_dl - 8:
                break
            h_new, mc_new = lp_step(h_try, n, eps_r)
            if mc_new < best_mc - 1e-12:
                best_mc = mc_new
                best_h = h_new.copy()
                best_c5 = best_mc * 2.0 / n
            h_try = h_new
            eps_r = max(eps_r * 0.996, 0.001)

    # Phase 4: Try n=600 (12%)
    p4_dl = t0 + 0.87 * total
    if time.time() < p4_dl - 10:
        nh = 600
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        mc_h = float(np.correlate(hh, 1.0 - hh, mode='full').max())
        c5_h = mc_h * 2.0 / nh
        if c5_h < best_c5 * 1.005:
            idx1n, idx2n = build_indices(nh)
            eps_r = 0.03
            for _ in range(100):
                if time.time() > p4_dl - 5:
                    break
                hh_new, mc_new = lp_step(hh, nh, eps_r)
                c5_new = mc_new * 2.0 / nh
                if c5_new < best_c5 - 1e-12:
                    best_mc = mc_new
                    best_h = hh_new.copy()
                    best_c5 = c5_new
                    n = nh
                hh = hh_new
                eps_r = max(eps_r * 0.997, 0.001)

    # Phase 5: Final polish (13%)
    idx1, idx2 = build_indices(n)
    if time.time() < deadline - 8:
        h_try = adam_smooth(best_h, n, min(30, deadline - time.time() - 6), idx1, idx2, lr=0.003)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-10:
            best_mc = mc_try
            best_h = h_try.copy()
            best_c5 = best_mc * 2.0 / n

    if time.time() < deadline - 3:
        h_curr = best_h.copy()
        eps_f = 0.004
        for _ in range(300):
            if time.time() > deadline - 3:
                break
            h_new, mc_new = lp_step(h_curr, n, eps_f)
            c5_new = mc_new * 2.0 / n
            if c5_new < best_c5 - 1e-12:
                best_mc = mc_new
                best_h = h_new.copy()
                best_c5 = c5_new
            h_curr = h_new
            eps_f = max(eps_f * 0.998, 0.0002)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
