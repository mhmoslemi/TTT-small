def project(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n: a = np.concatenate([a, np.zeros(n - a.size)])
        else: a = a[:n]

        if np.clip(a - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    h = np.clip(a - 0.5*(lo+hi), 0, 1)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-14: break

            if r.sum() < 1e-15: break
            h += d * r / r.sum()

def lse_fg(x, n, beta):

    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1-h, mode="full") * dx, 0)
    m = c.max()
    y = np.minimum(beta * (c - m), 0)
    e = np.exp(y)
    s = e.sum()

        F = m
    else:

    B = np.convolve(p, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)): g = np.zeros(n)
    return F, g

    bc = c5_of(best, n)

        except: pass
