import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

One more idea: **Block coordinate descent**. Instead of updating one coordinate at a time, update a small block (e.g., 2-3 adjacent coordinates). This can be more effective when the function is "flat" in the 1D direction but has a better direction in 2D.

For 2 coordinates (p, q), the change in c_all is:

The cross terms come from the fact that (1-h[p+k]) changes when h[p+k] changes, but if p+k ≠ q, there's no cross term. The only cross term is when the "1-h" part of one coordinate's effect overlaps with the "h" part of the other. Specifically:
- If q = p+k for some k: the forward coefficient of p at lag n-1+k includes (1-h[q]), which changes by -d_q
- Similarly for backward

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def exact_cd(h, n, max_passes=200, deadline=None, top_m=12):
    h = h.copy()
    c_all = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c_all.max())
    total_lags = 2 * n - 1

    for passed in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline - 0.5:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            k_min = min(n - 1 - p, p)
            k_max = max(2 * n - 2 - p, p + n - 1)
            k_range = np.arange(k_min, k_max + 1)
            n_aff = len(k_range)
            p_f = p + k_range - (n - 1)
            p_b = p - k_range + (n - 1)
            coeff = np.zeros(n_aff)
            mask_f = (p_f >= 0) & (p_f < n)
            mask_b = (p_b >= 0) & (p_b < n)
            coeff[mask_f] += (1.0 - h[p_f[mask_f]])
            coeff[mask_b] -= h[p_b[mask_b]]
            all_base = c_all[k_range].copy()
            if k_min > 0 or k_max < total_lags - 1:
                parts = []
                if k_min > 0:
                    parts.append(float(c_all[:k_min].max()))
                if k_max < total_lags - 1:
                    parts.append(float(c_all[k_max + 1:].max()))
                unaff_max = max(parts)
            else:
                unaff_max = -1e18
            diag_pos = np.where(k_range == n - 1)[0]
            diag_idx = int(diag_pos[0]) if len(diag_pos) > 0 else -1
            if n_aff > top_m:
                top_idx = np.argsort(all_base)[::-1][:top_m]
            else:
                top_idx = np.arange(n_aff)
            d_lo = -h0
            d_hi = 1.0 - h0
            cands = [d_lo, d_hi, 0.0]
            for i in range(len(top_idx)):
                ti = top_idx[i]

                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c_all[k_range] = all_base + best_d * coeff
                if diag_idx >= 0:
                    c_all[n - 1] -= best_d * best_d
                current_max = max(unaff_max, float(c_all[k_range].max()))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c_all = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_all.max())
    return h, compute_c5(h, n)
