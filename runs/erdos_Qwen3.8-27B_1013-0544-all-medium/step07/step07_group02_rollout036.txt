    h = np.asarray(h,dtype=float)

    target = 0.5*float(n)
    s = float(h.sum())
    if abs(s-target) < 1e-13:

    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5*(lo+hi)

        if val > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - 0.5*(lo+hi),0.0,1.0)
    for _ in range(100):
        diff = target - float(out.sum())
        if abs(diff) < 1e-12:

        if diff > 0.0:

            ss = float(room.sum())

import numpy as np
import time, math
from scipy.optimize import linprog, minimize

G_N = 400
G_ALPHA = 10.0
G_BETA = 0.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(80):
        diff = target - float(out.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - out
            ss = float(room.sum())
            if ss < 1e-14:
                break
            out = out + (diff / ss) * room
        else:
            ss = float(out.sum())
            if ss < 1e-14:
                break
            out = out + (diff / ss) * out
        out = np.clip(out, 0.0, 1.0)
    return out

def raw_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def logsum_data(h, n, alpha):
    C = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(C))
    if alpha <= 1e-10:
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
        return cm, 0.0, w
    s = alpha * (C - cm)
    sm = float(np.max(s))
    e = np.exp(s - sm)
    se = float(np.sum(e))
    if se <= 0.0:
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
        return cm, 0.0, w
    lse = sm + math.log(se)
    return cm, lse, e / se

def fast_gradient(h, n, w):
    b1 = np.convolve(w[::-1], h, mode='full')[n - 1:2 * n - 1]
    b2 = np.convolve(w, h, mode='full')[n - 1:2 * n - 1]
    pref = np.empty(2 * n)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    j = np.arange(n)
    return pref[2 * n - 1 - j] - pref[n - 1 - j] - b1 - b2

def soft_obj(x):
    x = np.asarray(x, dtype=float)
    n = int(G_N)
    cm, lse, _ = logsum_data(x, n, G_ALPHA)
    return cm + lse / G_ALPHA

def sum_eq(x):
    x = np.asarray(x, dtype=float)
    return float(np.sum(x) - 0.5 * float(G_N))

def soft_grad(x):
    x = np.asarray(x, dtype=float)
    n = int(G_N)
    cm, lse, w = logsum_data(x, n, G_ALPHA)
    return fast_gradient(x, n, w)

def pen_obj(x):
    x = np.asarray(x, dtype=float)
    n = int(G_N)
    cm, lse, _ = logsum_data(x, n, G_ALPHA)
    s = float(np.sum(x) - 0.5 * float(n))
    return cm + lse / G_ALPHA + 0.5 * G_BETA * s * s

def pen_grad(x):
    x = np.asarray(x, dtype=float)
    n = int(G_N)
    cm, lse, w = logsum_data(x, n, G_ALPHA)
    g = fast_gradient(x, n, w)
    s = float(np.sum(x) - 0.5 * float(n))
    return g + G_BETA * s

def lbfgs_smooth(h, n, alpha, beta, maxiter, deadline):
    global G_N, G_ALPHA, G_BETA
    if time.time() > deadline:
        return h
    n = int(n)
    old = raw_max(h)
    G_N, G_ALPHA, G_BETA = n, float(alpha), float(beta)
    try:
        res = minimize(pen_obj, h, method='L-BFGS-B', jac=pen_grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        hx = project_feasible(res.x, n)
        if raw_max(hx) < old - 1e-12:
            return hx
    except Exception:
        pass
    return h

def slsqp_smooth(h, n, alpha, maxiter, ftol, deadline):
    global G_N, G_ALPHA
    if time.time() > deadline:
        return h
    n = int(n)
    old = raw_max(h)
    G_N, G_ALPHA = n, float(alpha)
    try:
        res = minimize(soft_obj, h, method='SLSQP', jac=soft_grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_eq}],
                       options={'maxiter': maxiter, 'ftol': float(ftol)})
        hx = project_feasible(res.x, n)
        if raw_max(hx) < old - 1e-12:
            return hx
    except Exception:
        pass
    return h

def lp_step(h, n, eps, full=False, max_cons=150):
    L = 2 * n - 1
    C = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(C))
    if full:
        m_list = np.arange(L)
    else:
        m_list = np.argsort(-C)[:int(max_cons)]
    M = len(m_list)
    A_ub = np.zeros((M, n + 1))
    A_ub[:, -1] = -1.0
    b = np.empty(M)
    for ii, m in enumerate(m_list):
        b[ii] = float(C[m])
        k = int(m) - (n - 1)
        r = A_ub[ii]
        s1 = -k if k < 0 else 0
        e1 = n - k
        if e1 > n:
            e1 = n
        if e1 > s1:
            r[s1:e1] += 1.0 - h[s1 + k:e1 + k]
        s2 = k if k > 0 else 0
        e2 = n + k
        if e2 > n:
            e2 = n
        if e2 > s2:
            r[s2:e2] -= h[s2 - k:e2 - k]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    lb = np.maximum(-float(eps), -h)
    ub = np.minimum(float(eps), 1.0 - h)
    bounds = np.empty((n + 1, 2))
    bounds[:n, 0] = lb
    bounds[:n, 1] = ub
    bounds[n, 0] = -np.inf
    bounds[n, 1] = np.inf
    res = None
    for method in ('highs', 'interior-point'):
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=np.array([0.0]),
                          bounds=bounds, method=method)
            if res is not None and res.x is not None:
                break
        except Exception:
            res = None
    if res is None or res.x is None:
        return h, mc
    d = res.x[:n]
    best_h, best_mc = h, mc
    for lam in (1.0, 0.7, 0.5, 0.3, 0.1, 0.02):
        ht = project_feasible(h + lam * d, n)
        mct = raw_max(ht)
        if mct < best_mc - 1e-13:
            best_mc, best_h = mct, ht
    return best_h, best_mc

def pgm_improve(h, n, alpha, limit, deadline, steps):
    best_h = h.copy()
    best_mc = raw_max(h)
    cur_h = h.copy()
    stps = list(steps)
    for _ in range(limit):
        if time.time() > deadline:
            break
        cm, lse, w = logsum_data(cur_h, n, alpha)
        g = fast_gradient(cur_h, n, w)
        g = g - g.mean()
        if float(np.max(np.abs(g))) < 1e-14:
            break
        bt, btm = None, best_mc
        for st in stps:
            ht = project_feasible(cur_h - st * g, n)
            mc = raw_max(ht)
            if mc < btm - 1e-12:
                btm, bt = mc, ht
        if bt is not None:
            cur_h = bt
            best_mc, best_h = btm, bt.copy()
        else:
            stps = [s * 0.4 for s in stps]
            if stps[-1] < 1e-5:
                break
    return best_h, best_mc

def local_improve(h, n, start_eps, deadline, max_lp, max_pgm):

    best_mc = raw_max(h)
    cur_h = h.copy()
    eps = float(start_eps)
    reset = 0

        new_h, new_mc = lp_step(cur_h, n, eps, full=(it < 2), max_cons=150)

                if pmc < best_mc - 1e-12:

        if eps < 1e-4:

    if max_pgm > 0 and time.time() < deadline:

        if pmc < best_mc - 1e-12:

def resample(h, n0, n1):
    if n1 == n0:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n0)
    x1 = np.linspace(0.0, 2.0, n1)
    return np.interp(x1, x0, h)
