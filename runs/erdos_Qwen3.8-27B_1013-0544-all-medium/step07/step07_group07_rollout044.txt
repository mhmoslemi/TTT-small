def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

    best_c = cur_c

        T = t0 * (t_end / t0) ** (it / max(n_iter, 1))

            new_h[j] -= eps
            if new_h[i] > 1 or new_h[j] < 0:

        new_c = compute_c5(new_h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):

            cur_c = new_c
            if cur_c < best_c:

                best_h = cur_h.copy()
    return best_h, best_c

def mccorick_lb(h_init, n, K, solver):
    """Solve McCormick relaxation for first K shifts to get lower bound."""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0, t_var >= 0]
    
    # For each shift k = 1..K, add McCormick constraints
    for k in range(1, K + 1):
        u = cp.Variable(n - k)
        # u_i ≈ h[i] * h[i+k]
        for i in range(n - k):
            constraints.append(u[i] <= h_var[i])
            constraints.append(u[i] <= h_var[i + k])
            constraints.append(u[i] >= h_var[i] + h_var[i + k] - 1)
            constraints.append(u[i] >= 0)
        # Sum h[i] - u[i] <= t for valid i
        constraints.append(cp.sum(h_var[:n-k]) - cp.sum(u) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status == 'optimal':
            return t_var.value, h_var.value
    except:
        pass
    return None, None

The LSE objective f(h) = max_k c_k + (1/alpha) * log(Σ exp(alpha * c_k)) is smooth (for finite alpha). We can compute its gradient efficiently:
- ∂c_k/∂h[i] depends on whether i is in the sum for c_k

For the full correlation, c_k = Σ_i h[i] * (1-h[i+k]) (for the relevant range of i).
∂c_k/∂h[i] = (1-h[i+k]) - h[i] * (∂h[i+k]/∂h[i])

So for k ≠ 0: ∂c_k/∂h[i] = (1-h[i+k]) if i+k is in range, plus the contribution from the term where i is the "second" index: -h[i-k] if i-k is in range.

More precisely, for the full correlation:
c_{m} = Σ_i h[i] * (1-h[i+(m-(n-1))]) where the valid range ensures both indices are in [0, n-1].

Actually, for the analytical gradient:
Let d = m - (n-1) be the actual shift. Then:
c_m = Σ_i h[i] * (1-h[i+d]) for valid i (where 0 ≤ i < n and 0 ≤ i+d < n).

∂c_m/∂h[j] = Σ_i [δ_{i,j}(1-h[i+d]) - h[i]·δ_{i+d,j}]
= (1-h[j+d])·1_{valid} - h[j-d]·1_{valid}

where "valid" means the indices are in range.

For the LSE: f(h) = max(m) + (1/alpha) * log(Σ exp(alpha*(c_m - max)))
∂f/∂h[j] = Σ_m (exp(alpha*(c_m - max)) / Σ exp(alpha*(c_m - max))) * ∂c_m/∂h[j]
= Σ_m p_m * ∂c_m/∂h[j]

where p_m = exp(alpha*(c_m - max)) / Σ exp(alpha*(c_m - max)) are softmax weights.

This is a weighted average of the gradients of individual c_m's. We can compute this efficiently!

But the constraint Σ h[i] = n/2 makes it a constrained optimization. We can use projected gradient:
h_new = project(h - lr * grad)

def projected_gradient_lse(h, n, alpha, lr, n_iter, dl):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)  # shifts
    cur = h.copy()
    for it in range(n_iter):
        if time.time() > dl: break
        c = np.correlate(cur, 1.0 - cur, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        shift_max = shifted.max()
        weights = np.exp(shifted - shift_max)
        weights /= weights.sum()
        
        # Compute gradient
        # For each shift index m, the actual shift is d = m - (n-1)
        # grad[m, j] = (1-cur[j+d]) * 1[0<=j<n, 0<=j+d<n] - cur[j-d] * 1[0<=j<n, 0<=j-d<n]
        # weighted sum over m
        grad = np.zeros(n)
        for m_idx in range(len(c)):
            if weights[m_idx] < 1e-10: continue
            d = m_idx - (n - 1)
            w = weights[m_idx]
            if d >= 0:
                # valid i: 0 <= i < n-d
                # ∂c_m/∂h[j] = (1-h[j+d]) for j in [0, n-d-1]
                # plus -h[j-d] for j in [d, n-1]
                if d == 0:
                    pass  # c = 0 at d=0 for binary, but not for fractional
                else:
                    grad[:n-d] += w * (1.0 - cur[d:n])
                    grad[d:] -= w * cur[:n-d]
            else:
                d_pos = -d
                grad[d_pos:] += w * (1.0 - cur[:n-d_pos])
                grad[:n-d_pos] -= w * cur[d_pos:]
        
        # Projected gradient step
        new = cur - lr * grad
        new = project_feasible(new, n)
        
        # Line search (simple: accept if improved, else halve step)
        new_c = compute_c5(new, n)
        if new_c < compute_c5(cur, n):
            cur = new
        else:
            lr *= 0.5
        
    return cur, compute_c5(cur, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shift_max = shifted.max()
    weights = np.exp(shifted - shift_max)
    wsum = weights.sum()
    fval = m + np.log(wsum) / alpha
    # Gradient: for shift d, dc/du[j] involves (1-u[j+d]) and -u[j-d]
    grad = np.zeros(2 * n - 1)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx] / wsum
        if w < 1e-8:
            continue
        d = m_idx - (n - 1)
        if d >= 0:
            if d < n:
                grad[:n - d] += w * (1.0 - x[d:n])
                grad[d:] -= w * x[:n - d]
        else:
            dp = -d
            if dp < n:
                grad[dp:] += w * (1.0 - x[:n - dp])
                grad[:n - dp] -= w * x[dp:]
    # Sum over correlation indices to get gradient wrt x
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx] / wsum
        if w < 1e-8:
            continue
        d = m_idx - (n - 1)
        if d >= 0 and d < n:
            g[:n - d] += w * (1.0 - x[d:n])
            g[d:] -= w * x[:n - d]
        elif d < 0:
            dp = -d
            if dp < n:
                g[dp:] += w * (1.0 - x[:n - dp])
                g[:n - dp] -= w * x[dp:]
    return fval, g

def proj_grad_lse(h, n, alpha, lr, n_iter, dl):
    cur = h.copy()
    cur_c = compute_c5(cur, n)
    for it in range(n_iter):
        if time.time() > dl:
            break
        _, g = lse_obj_and_grad(cur, n, alpha)
        for _ in range(10):
            new = cur - lr * g
            new = project_feasible(new, n)
            new_c = compute_c5(new, n)
            if new_c <= cur_c:
                cur = new
                cur_c = new_c
                break
            lr *= 0.5
        else:
            lr = min(lr * 2.0, 0.5)
    return cur, cur_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_slsqp(h, n, t, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - t, 0)
        return 0.5 * np.sum(excess ** 2)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def simulated_annealing(h, n, t_start, t_end, n_iter, dl, seed):
    rng = np.random.RandomState(seed)
    cur = h.copy()
    cur_c = compute_c5(cur, n)
    best = cur.copy()
    best_c = cur_c
    block_sizes = [1, 2, 3, 5, 8]
    for it in range(n_iter):
        if time.time() > dl:
            break
        T = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        bs = block_sizes[rng.randint(len(block_sizes))]
        idx = rng.choice(n, size=bs, replace=False)
        new = cur.copy()
        for j in idx:
            new[j] += rng.uniform(-0.15, 0.15)
        new = project_feasible(new, n)
        if np.allclose(new, cur):
            continue
        new_c = compute_c5(new, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = new
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best = cur.copy()
    return best, best_c
