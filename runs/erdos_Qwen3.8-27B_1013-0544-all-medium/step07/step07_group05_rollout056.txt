import time
import numpy as np
import scipy.optimize as so


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def smooth_max_and_weights(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -700.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    w = ew / ws
    return f, w


def compute_grad_val(x, q, n, dx):
    # grad[i] = dx * (conv_q[1-h][:n][i] - corr_x_q[:n][i])
    A = np.convolve(q, 1.0 - x, mode='full')[:n]
    B = np.correlate(x, q, mode='full')[:n]
    return dx * (A - B)


def lbfgs_run(h, n, beta, maxiter, deadline):
    dx = 2.0 / n
    target = 0.5 * n
    mid = n - 1
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        f, _ = smooth_max_and_weights(c, beta)
        return f
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        _, w = smooth_max_and_weights(c, beta)
        q = np.zeros(n)
        q[0] = w[mid]
        if n > 1:
            q[1:] = w[mid + 1:] + w[mid - 1:0:-1]
        return compute_grad_val(x, q, n, dx)
    
    rem = deadline - time.time()
    if rem < 5.0:
        return h, erdos_c5(h, n)
    mx = int(min(maxiter, max(10, int(rem / 1.5))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 30, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def pair_step(h, n, i, j):
    # Find optimal delta for moving h[i]+=delta, h[j]-=delta
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, erdos_c5(h, n)
    
    def c5_at(delta):
        hh = h.copy()
        hh[i] = hi0 + delta
        hh[j] = hj0 - delta
        return erdos_c5(hh, n)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = c5_at(c)
    fd = c5_at(d)
    for _ in range(30):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = c5_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = c5_at(d)
    best_d = 0.5 * (a + b)
    best_c = c5_at(best_d)
    cur_c = erdos_c5(h, n)
    if best_c < cur_c - 1e-14:
        hh = h.copy()
        hh[i] = hi0 + best_d
        hh[j] = hj0 - best_d
        return hh, best_c
    return h, cur_c


def pair_polish_enhanced(h, n, deadline, seed, max_passes=10):

    h = np.asarray(h, dtype=float).copy()

    frac_idx = np.where(frac_mask)[0]
    
    for p in range(max_passes):

            break
        improved = False
        # Shuffle fractional indices for diversification
        order = frac_idx.copy()
        rng.shuffle(order)
        
        # Try pairs: for each i in order, try a few j's
        # Priority: pairs where |h[i] - h[j]| is large (more room to move)
        # and pairs at "active" distances
        c_vals = np.correlate(h, 1.0 - h, mode='full')
        active_k = int(np.argmax(c_vals))  # 0 to 2n-2
        active_dist = abs(active_k - (n - 1))
        
        n_try = min(len(order), 150)
        for idx_i in range(n_try):
            i = int(order[idx_i])
            # Choose j: prefer fractional, prefer distance = active_dist
            candidates = []
            for idx_j in range(n_try):
                if idx_j == idx_i:
                    continue
                j = int(order[idx_j])
                dist = abs(i - j)
                # Score: prefer active distance and fractional
                score = 0.0
                if dist == active_dist:
                    score += 10.0
                score += min(h[i], 1-h[i]) + min(h[j], 1-h[j])
                candidates.append((score, j))
            if not candidates:
                continue
            candidates.sort(key=lambda x: -x[0])
            # Try top few
            for score, j in candidates[:8]:
                if time.time() > deadline - 5.0:
                    return h, erdos_c5(h, n)
                hh, cc = pair_step(h, n, i, j)
                if cc < cur - 1e-14:
                    h = hh
                    cur = cc
                    improved = True
        if not improved:
            break
    return h, cur
