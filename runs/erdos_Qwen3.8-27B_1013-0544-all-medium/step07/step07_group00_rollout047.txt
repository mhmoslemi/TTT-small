    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

                m2 = p + offset

                has2 = (0 <= m2 < n)
                if not has1 and not has2:
                    continue
                if has1 and has2:
                    B = 1.0 - h[m1] - h[m2]
                elif has1:
                    B = 1.0 - h[m1]
                else:
                    B = -h[m2]
                A = c[m] - h0 * B
                B_list.append(B)
                A_list.append(A)
            A_center = c[n - 1] - h0 + h0 * h0

                h[p] = best_t
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def perturbed_local(h, n, rng, n_perturb, sigma, deadline, grid_K=200):
    for _ in range(n_perturb):
        if deadline is not None and time.time() > deadline:
            break
        idx = rng.choice(n, size=max(1, n // 20), replace=False)
        hp = h.copy()
        hp[idx] = np.clip(hp[idx] + rng.normal(0, sigma, len(idx)), 0, 1)
        hp = project_feasible(hp, n)
        ho, co = vec_coord_descent(hp, n, 40, grid_K, deadline)
        if co < compute_c5(h, n):
            h = ho
    return h, compute_c5(h, n)
