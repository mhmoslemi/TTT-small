def project_box_sum(x, target):

def erdos_c5(h):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size))


def resample_n(h, n, nh):
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * nh)


def softmax_log_w(c, beta):
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return cmax, w
    w = ew / ws
    return cmax + np.log(ws) / beta, w


def gradient_from_w(h, w, dx):
    n = h.size
    g = np.zeros(n, dtype=float)
    m = n - 1
    w0 = float(w[m])
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = float(w[m + k])
        if wp > 1e-15:
            g[:n - k] += wp * (1.0 - h[k:])
            g[k:] += -wp * h[:n - k]
        wn = float(w[m - k])
        if wn > 1e-15:
            r = k
            g[r:] += wn * (1.0 - h[:n - r])
            g[:n - r] += -wn * h[r:]
    return dx * g

    n = int(st['n'])

    target = float(st['target'])

    rho = float(st['rho'])

    res = float(x.sum() - target)
    if rho != 0.0:

        g = g + rho * res
    return float(f), g

                          options={'maxiter': mx, 'maxls': 20, 'ftol': 1e-18,

    s = m - j
    p[s:s + n] -= acr
    s = j
    p[s:s + n] += hr
    return p


def pair_eval(delta, base, p, idx1, idx2):
    d = float(delta)
    dd = d * d
    v = base + d * p
    am = float(np.max(v))
    x1 = float(base[idx1] + d * p[idx1] + dd)
    x2 = float(base[idx2] + d * p[idx2] + dd)
    if x1 > am:
        am = x1
    if x2 > am:
        am = x2
    return am


def golden_min(func, lo, hi, iters, *args):
    if hi <= lo:
        f = float(func(lo, *args))
        return lo, f
    a = float(lo)
    b = float(hi)
    c = b - GOLD * (b - a)
    d = a + GOLD * (b - a)
    fc = float(func(c, *args))
    fd = float(func(d, *args))
    if fc <= fd:
        best_x, best_f = c, fc
    else:
        best_x, best_f = d, fd
    for _ in range(int(iters)):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - GOLD * (b - a)
            fc = float(func(c, *args))
            if fc < best_f:
                best_f, best_x = fc, c
        else:
            a = c
            c = d
            fc = fd
            d = a + GOLD * (b - a)
            fd = float(func(d, *args))
            if fd < best_f:
                best_f, best_x = fd, d
    mid = 0.5 * (a + b)
    fm = float(func(mid, *args))
    if fm < best_f:
        best_f, best_x = fm, mid
    fa = float(func(a, *args))
    if fa < best_f:
        best_f, best_x = fa, a
    fb = float(func(b, *args))
    if fb < best_f:
        best_f, best_x = fb, b
    return best_x, best_f


def pair_polish(h, deadline, seed):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    n = h.size
    m = n - 1
    L = 2 * n - 1
    dx = 2.0 / n
    s = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(np.max(s))
    best_h = h.copy()
    best_cur = cur
    rng = np.random.RandomState(int(seed))
    beta_w = 8000.0 if n <= 250 else 4000.0

    for _ in range(2000):
        if time.time() > deadline - 3.0:
            break
        h = np.clip(h, 0.0, 1.0)
        s = np.correlate(h, 1.0 - h, mode='full').astype(float)
        cur = float(np.max(s))
        if cur < best_cur:
            best_cur = cur
            best_h = h.copy()

        w = softmax_log_w(s * dx, beta_w)[1]
        g = gradient_from_w(h, w, dx)
        frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        pool = frac if frac.size >= max(12, n // 4) else np.arange(n)
        if frac.size > 0:
            order = frac[np.argsort(g[frac])]
            kk = min(20, max(6, int(n // 20)))
            kk = min(kk, int(frac.size))
            low = order[:kk]
            high = order[-kk:]
        else:
            low = high = np.array([], dtype=int)

        maxp = 900 if n <= 200 else 1200
        pairs = []
        for ii in low:
            ii = int(ii)
            for jj in high:
                jj = int(jj)
                if ii != jj:
                    pairs.append((ii, jj))
        while len(pairs) < maxp:
            if pool.size < 2:
                break
            ii = int(rng.choice(pool))
            jj = int(rng.choice(pool))
            if ii != jj:
                pairs.append((ii, jj))
        if not pairs:
            break
        rng.shuffle(pairs)

        improved = 0
        for ii, jj in pairs:
            if time.time() > deadline - 3.0:
                break
            a = float(h[ii])
            b = float(h[jj])
            lo = max(-a, b - 1.0)
            hi = min(1.0 - a, b)
            if hi - lo < 1e-12:
                continue
            p = make_pair_p(h, ii, jj, L, m)
            d = abs(ii - jj)
            x1 = m - d
            x2 = m + d
            best_d, best_v = golden_min(pair_eval, lo, hi, 24, s, p, x1, x2)
            if best_v < cur - 1e-12:
                dt = float(best_d)
                h[ii] = min(1.0, max(0.0, a + dt))
                h[jj] = min(1.0, max(0.0, b - dt))
                s += dt * p
                s[x1] += dt * dt
                s[x2] += dt * dt
                cur = float(np.max(s))
                improved += 1
                if improved == 50:
                    h = project_box_sum(np.clip(h, 0.0, 1.0), 0.5 * n)
                    s = np.correlate(h, 1.0 - h, mode='full').astype(float)
                    cur = float(np.max(s))
                    if cur < best_cur:
                        best_cur = cur
                        best_h = h.copy()
                if improved >= 250:
                    break

        if improved == 0:
            break

    h = np.clip(h, 0.0, 1.0)
    s = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(np.max(s))
    if cur < best_cur:
        best_cur = cur
        best_h = h.copy()
    return best_h, best_cur * dx


def local_optimize(h, t_end, seed):
    h = project_box_sum(np.clip(h, 0.0, 1.0), 0.5 * h.size)
    c = erdos_c5(h)
    if time.time() < t_end - 25.0 and h.size <= 320:
        hh, cc = lbfgs_run(h, 1500.0, 5000.0, 30, min(t_end - 15.0, time.time() + 25.0))
        if cc < c:
            h, c = hh, cc
    pe = t_end - 8.0
    if time.time() < pe - 2.0:
        hh, cc = pair_polish(h, pe, int(seed))
        if cc < c:
            h, c = hh, cc
    if time.time() < t_end - 5.0 and h.size <= 320:
        hh, cc = lbfgs_run(h, 6000.0, 5000.0, 18, t_end)
        if cc < c:
            h, c = hh, cc
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    b = max(1.0, float(budget_s) - 3.0)
    deadline = t0 + b
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
    except Exception:
        h0, n0 = None, 0

    if h0 is not None and n0 >= 20:
        if n0 > 800:
            h0 = resample_n(h0, n0, 400)
            n0 = 400
        elif n0 < 80:
            h0 = resample_n(h0, n0, 200)
            n0 = 200
        best_h = h0
        best_n = n0
    else:
        best_n = 400
        best_h = make_initial_h(best_n, int(seed))

    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), 0.5 * best_n)
    best_c = erdos_c5(best_h)

    end_parent = t0 + 0.50 * b
    if time.time() < end_parent - 45.0:
        hh, cc = lbfgs_run(best_h, 2500.0, 5000.0, 35, min(deadline, end_parent - 20.0))
        if cc < best_c:
            best_h, best_c = hh, cc

    if time.time() < end_parent - 20.0:
        hh, cc = pair_polish(best_h, end_parent - 20.0, int(seed) + 1)
        if cc < best_c:
            best_h, best_c = hh, cc

    if time.time() < deadline - 15.0:
        end_global = t0 + 0.87 * b
        trials = []
        for nh in (320, 256, 200, 160, 120, 100, 80):
            trials.append((nh, 0))
            trials.append((nh, 1))
        trials.append((120, 2))
        trials.append((100, 2))
        trials.append((100, 3))
        trials.append((80, 3))

        if b > 350.0 and time.time() < end_global - 30.0:
            for idx, (nh, kind) in enumerate(trials):
                now = time.time()
                if now > end_global - 20.0:
                    break
                rem_trials = max(1, len(trials) - idx)
                slice_s = min(120.0, (end_global - now) / float(rem_trials))
                if slice_s < 8.0:
                    break
                t_end = now + slice_s
                if t_end > end_global:
                    t_end = end_global
                base = resample_n(best_h, best_n, nh)
                if kind == 0:
                    h0 = base
                elif kind == 1:
                    sig = 0.04 + 0.025 * float(idx % 5)
                    h0 = project_box_sum(base + sig * rng.randn(nh), 0.5 * nh)
                elif kind == 2:
                    h0 = make_initial_h(nh, int(seed) * 31 + idx)
                else:
                    h0 = make_binary_h(nh, int(seed) * 53 + idx)
                hh, cc = local_optimize(h0, t_end, int(seed) + 100 + idx)
                if cc < best_c - 1e-12:
                    best_h, best_c, best_n = hh, cc, nh

    if time.time() < deadline - 6.0:
        hh, cc = pair_polish(best_h, deadline - 6.0, int(seed) + 777)
        if cc < best_c:
            best_h, best_c = hh, cc

    if time.time() < deadline - 5.0 and best_n <= 400:
        hh, cc = lbfgs_run(best_h, 20000.0, 5000.0, 18, deadline)
        if cc < best_c:
            best_h, best_c = hh, cc

    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), 0.5 * int(best_n))
    best_c = erdos_c5(best_h)
    return best_h, float(best_c), int(best_n)
