import numpy as np
from scipy.optimize import minimize
import time


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v


def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    mx = sh.max()
    s = float(np.exp(sh - mx).sum())
    return m + (mx + np.log(s)) / alpha if s > 0 else m


def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = sh.max()
    e = np.exp(sh - mx)
    s = float(e.sum())
    q = e / s if s > 0 else np.zeros_like(c)
    if s <= 0:
        q[int(np.argmax(c))] = 1.0
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k == 0:
            g += w * (1.0 - 2.0 * x)
        elif k > 0:
            g[k:n] += w * (1.0 - x[0:n-k])
            g[0:n-k] -= w * x[k:n]
        else:
            jp = -k
            g[0:n-jp] += w * (1.0 - x[jp:n])
            g[jp:n] -= w * x[0:n-jp]
    return dx * g


def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d


def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d


def lbfgsb_opt(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=[(0.0, 1.0)] * n,
                       options={"maxiter": maxiter, "maxfun": 3 * maxiter,
                                "ftol": 1e-14, "gtol": 1e-8})
        if hasattr(res, "x") and res.x is not None:
            h = res.x
    except Exception:
        pass
    return project_box_sum(h, n)


def _sfun(x, n):
    return float(x.sum() - 0.5 * n)


def _sjac(x, n):
    return np.ones(n)


def slsqp_opt(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        cons = [{"type": "eq", "fun": _sfun, "args": (n,), "jac": _sjac}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lse_grad, bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-13})
        if hasattr(res, "x") and res.x is not None:
            h = res.x
    except Exception:
        pass
    return project_box_sum(h, n)


def pair_search(h, n, deadline, seed=42):

    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    c = compute_c(h, n)
    off = n - 1
    rng = np.random.RandomState(seed)
    k_w = np.maximum(c[off+1:], c[off-1::-1]) + 0.01
    k_w /= k_w.sum()
    noimp = 0
    maxno = 800
    it = 0
    while time.time() < deadline:
        if noimp > maxno:
            break
        d = int(rng.choice(n-1, p=k_w)) + 1
        i = int(rng.randint(n - d))
        j = i + d
        x0 = h[i]
        y0 = h[j]
        S = x0 + y0
        lo = max(0.0, S - 1.0)
        hi = min(1.0, S)
        if hi - lo < 1e-14:

        n_coarse = 24
        xs = np.linspace(lo, hi, n_coarse)
        best_x = x0
        best_cv = float('inf')
        for x in xs:

            h[i] = x
            h[j] = y
            cv = float(compute_c(h, n).max())
            if cv < best_cv:

        a = max(lo, best_x - (hi - lo) / (n_coarse - 1))

            if cv < bestc - 1e-15:

            else:

    besth = project_box_sum(besth, n)
