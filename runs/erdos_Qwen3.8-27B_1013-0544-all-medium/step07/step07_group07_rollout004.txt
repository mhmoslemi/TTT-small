def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    if n is None: n = len(h)
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    # bisection threshold t
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # residual correction
    res = target - x.sum()
    if abs(res) > 1e-14:
        # distribute to interior
        interior = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        for _ in range(5):
            if abs(res) <= 1e-14: break
            if len(interior) == 0:
                # adjust any not near bound
                if res > 0:
                    idx = np.argsort(1.0 - x)[::-1]
                else:
                    idx = np.argsort(x)
                add = res
            else:
                idx = interior
                add = res / len(idx)
            if res > 0:
                for ii in idx:
                    avail = 1.0 - x[ii]
                    step = min(add, avail)
                    if step <= 0: continue
                    x[ii] += step
                    res -= step
                    if abs(res) <= 1e-14: break
            else:
                for ii in idx:
                    avail = x[ii]
                    step = max(add, -avail)
                    if step >= 0: continue
                    x[ii] += step
                    res += step
                    if abs(res) <= 1e-14: break
            x = np.clip(x,0,1)
            res = target - x.sum()
            interior = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
    else:
        x = np.clip(x,0,1)
    return x

import numpy as np
from scipy.optimize import minimize
import time

_N_GLOBAL = 0
_TARGET_SUM = 0.0
_ACTIVE_IDX = np.array([], dtype=int)
_ACTIVE_WEIGHTS = np.array([], dtype=float)
_ACTIVE_T = 0.0
_PNORM_P = 64.0
_LSE_ALPHA = 1000.0


def set_global_n(n):
    global _N_GLOBAL, _TARGET_SUM
    _N_GLOBAL = int(n)
    _TARGET_SUM = 0.5 * float(_N_GLOBAL)


def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    if n is None:
        n = len(h)
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(x.sum())
    for _ in range(30):
        if abs(res) < 1e-13:
            break
        idx = np.flatnonzero((x > 1e-13) & (x < 1.0 - 1e-13))
        if len(idx) == 0:
            idx = np.flatnonzero((x > 0.0) & (x < 1.0))
        if len(idx) == 0:
            break
        x[idx] += res / float(len(idx))
        x = np.clip(x, 0.0, 1.0)
        res = target - float(x.sum())
    if res != 0.0:
        idx = np.flatnonzero((x > 1e-12) & (x < 1.0 - 1e-12))
        if len(idx):
            x[idx[0]] += res
            x = np.clip(x, 0.0, 1.0)
    return x


def c_vector(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    return float(np.max(c_vector(h)) * (2.0 / len(h)))


def choose_active(cvec, cm, step):
    n = len(cvec)
    order = np.argsort(cvec)[::-1]
    margin = max(0.5, 3.0 * max(step, 0.0))
    thr = cm - margin
    cand = np.where(cvec >= thr)[0]
    maxc = 100
    if len(cand) < 40:

    else:
        keep = min(maxc, len(cand))
        idx = cand[np.argsort(cvec[cand])[::-1][:keep]]
    vals = cvec[idx]
    w = np.exp(-np.maximum(cm - vals, 0.0) / max(margin, 1e-12))
    wm = float(w.max())
    if wm > 0.0:
        w = (w / wm) * 200.0
    return idx, w


def active_objective(x):

    if idx is None or len(idx) == 0:

    n = _N_GLOBAL
    t = float(_ACTIVE_T)

    if not np.isfinite(x).all():

        if wa == 0.0:
            continue
        m = int(idx[a])
        ell = n - 1 - m
        if ell >= 0:
            cnt = n - ell
            a0 = 0
            b0 = ell
        else:

            a0 = -ell
            b0 = 0

def active_grad(x):
    n = _N_GLOBAL
    idx = _ACTIVE_IDX
    ws = _ACTIVE_WEIGHTS
    t = float(_ACTIVE_T)
    g = np.zeros(n, dtype=float)

            cnt = n - ell
            a0 = 0
            b0 = ell

            a0 = -ell
            b0 = 0

def sum_constraint(x):

def full_grad_weights(x, w):

    g = np.zeros(n, dtype=float)

        return g
    for m in range(len(w)):

            a0 = -ell
            b0 = 0

        xb = x[b0:b0 + cnt]
        g[a0:a0 + cnt] += wm * (1.0 - xb)
        g[b0:b0 + cnt] -= wm * xa
    return g


def full_pnorm_objective(x):
    if not np.isfinite(x).all():
        return 1e100
    c = np.maximum(c_vector(x), 1e-12)
    p = float(_PNORM_P)
    logc = np.log(c)
    mc = float(logc.max())
    s = float(np.sum(np.exp(p * (logc - mc))))
    logF = mc + (np.log(s) - np.log(len(c))) / p
    return float(np.exp(logF))


def full_pnorm_grad(x):
    n = len(x)
    if not np.isfinite(x).all():
        return np.zeros(n, dtype=float)
    c = np.maximum(c_vector(x), 1e-12)
    p = float(_PNORM_P)
    logc = np.log(c)
    mc = float(logc.max())
    s = float(np.sum(np.exp(p * (logc - mc))))
    logF = mc + (np.log(s) - np.log(len(c))) / p
    logw = (p - 1.0) * (logc - logF) - np.log(len(c))
    lwmax = float(logw.max())
    w = np.exp(logw - lwmax)
    scale = float(np.exp(lwmax))
    if scale == 0.0:
        return np.zeros(n, dtype=float)
    return scale * full_grad_weights(x, w)


def full_lse_objective(x):
    if not np.isfinite(x).all():
        return 1e100
    c = np.maximum(c_vector(x), 0.0)
    m = float(c.max())
    alpha = float(_LSE_ALPHA)
    if alpha <= 1.0:
        return m
    thr = -60.0 / alpha
    act = c - m > thr
    e = np.exp(alpha * (c[act] - m))
    return m + float(np.log(e.sum()) / alpha)


def full_lse_grad(x):

    best_c = compute_c5(h)
    alphas = [10.0, 50.0, 250.0, 1500.0, 10000.0, 80000.0]
    for al in alphas:
        if time.time() > deadline - 12:
            break
        mi = safe_maxiter(deadline, 240, 0.18)
        if mi <= 0:
            break
        ht = minimize_full_lse(h, n, al, mi)
        ct = compute_c5(ht)
        if ct < best_c - 1e-12:
            best_c = ct
            best_h = ht.copy()
            h = ht
    for p in [16.0, 32.0, 64.0]:
        if time.time() > deadline - 12:
            break
        mi = safe_maxiter(deadline, 180, 0.22)
        if mi <= 0:
            break
        ht = minimize_full_pnorm(h, n, p, mi)
        ct = compute_c5(ht)
        if ct < best_c - 1e-12:
            best_c = ct
            best_h = ht.copy()
            h = ht
    return project_feasible(best_h, n), compute_c5(project_feasible(best_h, n))


def resample_h(h, n_new, x_old):
    if n_new == len(h):
        return h.copy()
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    reserve = 35.0
    dl = t0 + total - reserve
    target_c = 0.38079
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        x = np.linspace(0.0, 2.0, n0)
        h0 = np.where(np.sin(2.0 * np.pi * 5.0 * x) > 0.0, 1.0, 0.0)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h)

    if best_c > target_c and time.time() < t0 + 0.40 * total:
        h1, c1 = active_phase(best_h, best_n, t0 + 0.40 * total, target_c, cycles=3, rng=rng)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if best_c > target_c and time.time() < t0 + 0.58 * total:
        h2, c2 = full_polish(best_h, best_n, t0 + 0.58 * total)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if best_c > target_c and time.time() < t0 + 0.78 * total:
        h3, c3 = active_phase(best_h, best_n, t0 + 0.78 * total, target_c, cycles=3, rng=rng)
        if c3 < best_c:
            best_h, best_c = h3, c3

    if best_c > target_c and time.time() < t0 + 0.90 * total:
        x_best = np.linspace(0.0, 2.0, best_n)
        cand_ns = []
        for fr in [0.96, 0.98, 1.02, 1.04, 1.08]:
            nh = int(round(best_n * fr))
            if nh % 2:
                nh += 1
            if 220 <= nh <= 720 and nh != best_n and nh not in cand_ns:
                cand_ns.append(nh)
        for nh in cand_ns[:3]:
            if time.time() > dl - 60 or best_c <= target_c:
                break
            if len(cand_ns) > 1 and time.time() > t0 + 0.86 * total:
                break
            hh = project_feasible(resample_h(best_h, nh, x_best), nh)
            if compute_c5(hh) > best_c + 0.004:
                continue
            h_o, c_o = active_phase(hh, nh, dl - 45, target_c, cycles=1, rng=rng)
            if c_o < best_c:
                best_h, best_c, best_n = h_o, c_o, nh
                x_best = np.linspace(0.0, 2.0, best_n)

    if best_c > target_c and time.time() < dl - 10:
        h4, c4 = active_phase(best_h, best_n, dl, target_c, cycles=2, rng=rng)
        if c4 < best_c:
            best_h, best_c = h4, c4

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return best_h, best_c, best_n
