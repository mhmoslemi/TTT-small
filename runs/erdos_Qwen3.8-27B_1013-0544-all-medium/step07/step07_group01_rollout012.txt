from scipy.optimize import minimize
import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sh_max = sh.max()
    return m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sh_max = sh.max()
    p = np.exp(sh - sh_max)
    p /= p.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j] / alpha
        if w < 1e-15:

        else:
            ll = -lag
            grad[ll:] += w * (1.0 - x[:n - ll])
            grad[:n - ll] += w * (-x[ll:])
    return grad

    def obj(x):
        return smooth_obj(x, n, alpha)
    def objgrad(x):
        v = smooth_obj(x, n, alpha)
        g = smooth_grad(x, n, alpha)
        return v, g

        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_fun}]

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_feasible(res.x, n)

            if s < 1e-15:

            h = h + diff / s * h

        m2_hat = m2 / (1 - b2 ** t)
        step = lr * m1_hat / (np.sqrt(m2_hat) + 1e-8)
        h = project_feasible(h - step, n)
        if t % 50 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def subgrad_descent(h, n, deadline, lr=0.05, mu=0.92):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    momentum = np.zeros(n)
    stall = 0
    for step in range(500000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            ll = -lag
            grad[ll:] += (1.0 - h[:n - ll])
            grad[:n - ll] -= h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h = project_feasible(h + momentum, n)
        if step % 100 == 0:
            c = compute_c5(h, n)
            if c < best_c - 1e-10:
                best_c = c
                best_h = h.copy()
                stall = 0
                lr = min(lr * 1.01, 0.5)
            else:
                stall += 1
                if stall > 30:
                    lr = max(lr * 0.95, 0.01)
                    if lr <= 0.011:
                        lr = 0.1
                        stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B continuation (fast, handles many variables well)
    if time.time() < deadline - 500:
        for a in [50, 200, 1000, 5000, 20000]:
            if time.time() > deadline - 500:
                break
            h = lbbf_optimize(h, n, a, 300)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 2: Adam smooth descent with continuation
    if time.time() < deadline - 350:
        for a in [1000, 10000, 50000, 200000]:
            phase_end = min(time.time() + 60, deadline - 350)
            if time.time() > phase_end:
                break
            h2, c2 = adam_smooth(best_h.copy(), n, a, phase_end, lr=0.003)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()

    # Phase 3: SLSQP with high-alpha continuation
    if time.time() < deadline - 250:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
        for a in alphas:
            if time.time() > deadline - 100:
                break
            h3 = slsqp_optimize(best_h.copy(), n, a, 3000)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c, best_h = c3, h3.copy()

    # Phase 4: Subgradient for global exploration
    if time.time() < deadline - 150:
        h4, c4 = subgrad_descent(best_h.copy(), n, min(time.time() + 100, deadline - 150))
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Randomized restarts
    if time.time() < deadline - 100:
        for trial in range(15):
            if time.time() > deadline - 50:
                break
            if trial < 3:
                h5 = 1.0 - best_h
            elif trial < 6:
                h5 = best_h[::-1].copy()
            elif trial < 9:
                h5 = 0.5 * (best_h + best_h[::-1]) + 0.01 * rng.randn(n)
            else:
                nl = 0.003 * (1 + (trial - 9) * 0.4)
                h5 = best_h + nl * rng.randn(n)
            h5 = project_feasible(h5, n)
            h5 = lbbf_optimize(h5, n, 100000, 200)
            h5 = slsqp_optimize(h5, n, 2000000, 1500)
            c5 = compute_c5(h5, n)
            if c5 < best_c:
                best_c, best_h = c5, h5.copy()

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 10:
        for a in [10000000, 100000000, 1000000000]:
            if time.time() > deadline - 5:
                break
            h6 = slsqp_optimize(best_h.copy(), n, a, 3000)
            c6 = compute_c5(h6, n)
            if c6 < best_c:
                best_c, best_h = c6, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
