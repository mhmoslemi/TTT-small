```python
def lse_gradient(h, n, alpha):
    """Compute gradient of LSE-smoothed max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()
    
    # Gradient: for each lag k (0 to 2n-2), m = k-(n-1)
    # grad[j] = sum_k weights[k] * [(1-h[j-m]) - h[j+m]]
    # where indices are masked to [0,n)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        m = k - (n-1)
        w = weights[k]
        if w < 1e-15:
            continue
        # Term 1: (1-h[j-m]) for j-m in [0,n) => j in [m, m+n)
        # Term 2: -h[j+m] for j+m in [0,n) => j in [-m, n-m)
        # Combined: grad[j] += w * (1-h[j-m]) - w * h[j+m]
        
        # Vectorized:
        # (1-h) shifted by -m: position j gets (1-h[j-m]) if 0<=j-m<n
        if m >= 0:
            # j-m in [0,n) => j in [m, m+n), but j in [0,n) => j in [m, n)
            grad[m:n] += w * (1 - h[0:n-m])
            # j+m in [0,n) => j in [0, n-m)
            grad[0:n-m] -= w * h[m:n]
        else:
            # j-m in [0,n) => j in [m+n, n) = [0, -m) since m<0, j-m=j+|m| in [0,n) => j in [0, n-|m|)
            am = -m
            grad[0:n-am] += w * (1 - h[am:n])
            # j+m in [0,n) => j+(-am) in [0,n) => j in [am, n)
            grad[am:n] -= w * h[0:n-am]
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(v, n):
    target = n / 2.0
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def lse_score(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    sm = shifted - shifted.max()
    return c_max + np.log(np.exp(sm).sum()) / alpha

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    sm = shifted - shifted.max()
    w = np.exp(sm)
    w /= w.sum()
    grad = np.zeros(n)
    # Only process lags with significant weight
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k]
        m = int(k) - (n - 1)
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] -= wk * h[m:]
        else:
            am = -m
            grad[:n - am] += wk * (1.0 - h[am:])
            grad[am:] -= wk * h[:n - am]
    return grad * dx

def lbfgs_lse(h0, n, alpha, penalty, maxiter=5000):
    pen = penalty
    tgt = n / 2.0
    def f(x):
        return lse_score(x, n, alpha) + pen * (x.sum() - tgt) ** 2
    def df(x):
        g = lse_grad(x, n, alpha) + 2.0 * pen * (x.sum() - tgt)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=df, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return res.x
    except Exception:
        return h0

def binary_local_search(h_init, n, deadline, seed=42):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    rng = np.random.RandomState(seed)
    no_improve = 0
    max_no_improve = 50
    while time.time() < deadline and no_improve < max_no_improve:
        no_improve += 1
        improved = False
        n_try = min(40, len(zeros))
        for _ in range(n_try):
            i = int(zeros[rng.randint(len(zeros))])
            j = int(ones[rng.randint(len(ones))])
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            if 0 <= k1 < 2 * n - 1:
                delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            if 0 <= k2 < 2 * n - 1:
                delta[k2] += 1.0
            nc = (corr + delta).max() * dx
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                ones.remove(j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(j)
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def pgd_step(h, n, alpha, lr, momentum, v):
    grad = lse_grad(h, n, alpha)
    v = momentum * v - lr * grad
    h_new = project_box_sum(h + v, n)
    return h_new, v
