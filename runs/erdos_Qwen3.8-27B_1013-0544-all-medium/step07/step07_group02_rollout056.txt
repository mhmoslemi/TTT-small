```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-14:
                break
            h = h + (diff / rs) * room
        else:
            if s < 1e-14:
                break
            h = h + (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_c_k(h, n, k):
    """Gradient of c_k = sum_j h[j]*(1-h[j+k]) w.r.t. h."""
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    else:
        kp = -k
        g[kp:] += 1.0 - h[:n-kp]
        g[:n-kp] -= h[kp:]
    return g

def subgrad_step(h, n, alpha):
    """One subgradient step."""
    c = np.correlate(h, 1 - h, mode='full')
    max_c = c.max()
    # Find all active constraints (within tolerance)
    tol = 1e-6
    active = np.where(c >= max_c - tol)[0]
    
    # Compute combined gradient
    g = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        g += grad_c_k(h, n, k)
    g /= len(active)
    
    # Project gradient to sum-zero subspace (for the equality constraint)
    g -= g.mean()
    
    # Line search
    best_alpha = alpha
    best_c5 = max_c
    for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
        h_try = h - a * g
        h_try = project_feasible(h_try, n)
        c_try = np.correlate(h_try, 1 - h_try, mode='full').max()
        if c_try < best_c5:
            best_c5 = c_try
            best_alpha = a
        else:
            break
    h_new = h - best_alpha * g
    h_new = project_feasible(h_new, n)
    return h_new, float(np.correlate(h_new, 1 - h_new, mode='full').max())

def optimize_coord(h, n, j):
    """Optimize coordinate j to minimize max correlation."""
    c = np.correlate(h, 1 - h, mode='full')
    M = 2 * n - 1
    k_vals = np.arange(M) - (n - 1)
    
    # Compute alphas[m] = dc_m/dh[j]
    idx1 = j + k_vals
    idx2 = j - k_vals
    alphas = np.zeros(M)
    v1 = (idx1 >= 0) & (idx1 < n)
    alphas[v1] += 1.0 - h[idx1[v1]]
    v2 = (idx2 >= 0) & (idx2 < n)
    alphas[v2] -= h[idx2[v2]]
    
    d = c - h[j] * alphas
    
    # Bisection for optimal a in [0,1]
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = d + mid * alphas
        mv = vals.max()
        active = alphas[vals >= mv - 1e-9]
        if len(active) == 0:
            active = alphas[vals >= mv - 1e-6]
        if active.min() > 1e-12:
            hi = mid
        elif active.max() < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_sweep(h, n, rng):
    """One coordinate descent sweep."""
    # Optimize in random order
    order = rng.permutation(n)
    for j in order:
        h[j] = optimize_coord(h, n, j)
    return project_feasible(h, n)
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-14:
                break
            h = h + (diff / rs) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_ck(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    else:
        kp = -k
        g[kp:] += 1.0 - h[:n-kp]
        g[:n-kp] -= h[kp:]
    return g

def subgrad_step(h, n, alpha):

        g += grad_ck(h, n, k)

    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h, float(max_c)
    g = g / gn
    best_a = 0.0
```
