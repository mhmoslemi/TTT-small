```python
import numpy as np
import time
from scipy.optimize import minimize


def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def eval_c(h, n):
    return float(compute_corr(h, n).max() * (2.0 / n))


def project(v, n):
    v = np.asarray(v, dtype=float).ravel()
    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, 0.0, 1.0)
    tgt = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if float(np.clip(v - mid, 0.0, 1.0).sum()) > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v


def lse_weights(c, al):
    m = float(c.max())
    if al <= 0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    e = np.exp(al * (c - m))
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    return e / s


def grad_q(h, n, q):
    off = n - 1
    cs = np.empty(2 * n, dtype=float)
    cs[0] = 0.0
    cs[1:] = np.cumsum(q)
    w = cs[n:2 * n] - cs[:n]
    c2 = np.convolve(h, q, mode="full")[off:off + n]
    c3 = np.convolve(h, q[::-1], mode="full")[off:off + n]
    return (2.0 / n) * (w - c2 - c3)


def obj_lse(x, n, al, mu):
    c = compute_corr(x, n) * (2.0 / n)
    m = float(c.max())
    if al > 0:
        e = np.exp(al * (c - m))
        s = float(e.sum())
        if s <= 0.0:
            val = m
        else:
            val = m + float(np.log(s) / al)
    else:
        val = m
    if mu:
        d = float(x.sum() - 0.5 * n)
        val += 0.5 * mu * d * d
    return float(val)


def grad_lse(x, n, al, mu):
    c = compute_corr(x, n) * (2.0 / n)
    q = lse_weights(c, al)
    g = grad_q(x, n, q)
    if mu:
        d = float(x.sum() - 0.5 * n)
        g = g + mu * d
    return g


def obj_eq(x, n, al):
    return obj_lse(x, n, al, 0.0)


def grad_eq(x, n, al):
    return grad_lse(x, n, al, 0.0)


def con_eq(x, n):
    return float(x.sum() - 0.5 * n)


def con_jac(x, n):
    return np.ones(n)


def slsqp_phase(h, n, al, it):
    h = project(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": con_eq, "args": (n,), "jac": con_jac}]
        res = minimize(obj_eq, h, args=(n, al), method="SLSQP", jac=grad_eq,
                       bounds=bounds, constraints=cons,
                       options={"maxiter": int(it), "ftol": 1e-15})
        if res.x is not None:
            return project(res.x, n)
    except Exception:
        pass
    return h


def smooth(h, n, seed, dl):
    h = project(h, n)
    besth = h.copy()
    bestc = eval_c(h, n)
    work = h.copy()
    scale = 1.0 if n <= 450 else max(0.45, 450.0 / n)
    phases = ((8, 5, 160), (20, 50, 140), (60, 400, 120), (180, 2000, 85), (500, 8000, 60))
    for al, mu, it0 in phases:
        if n < 4 or time.time() > dl - 12:
            break
        it = max(25, int(it0 * scale))
        try:
            res = minimize(obj_lse, work, args=(n, al, mu), method="L-BFGS-B",
                           jac=grad_lse, bounds=[(0.0, 1.0)] * n,
                           options={"maxiter": it, "maxfun": 4 * it,
                                    "ftol": 1e-14, "gtol": 1e-9, "maxls": 20})
            if res.x is not None:
                work = res.x
        except Exception:
            pass
        work = project(work, n)
        c = eval_c(work, n)
        if c < bestc:
            bestc = c
            besth = work.copy()
        if c > bestc + 1e-6:
            work = besth.copy()
    work = project(besth, n)
    if n <= 520:
        for al, it in ((120, 70), (400, 45), (1000, 30)):
            if time.time() > dl - 5:
                break
            h2 = slsqp_phase(work, n, al, it)
            c = eval_c(h2, n)
            if c < bestc:
                bestc = c
                besth = h2.copy()
                work = h2.copy()
            elif c <= bestc + 1e-6:
                work = h2.copy()
            else:
                work = besth.copy()
    besth = project(besth, n)
    return besth, eval_c(besth, n)


def interp_h(h, n1, n2):
    if n1 == n2:
        return np.asarray(h, dtype=float).copy()
    if n2 <= 0:
        return np.zeros(0, dtype=float)
    xo = (np.arange(n1) + 0.5) * (2.0 / n1)
    xn = (np.arange(n2) + 0.5) * (2.0 / n2)
    return np.interp(xn, xo, np.asarray(h, dtype=float))


def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    v = 0.5 + 0.45 * np.sin(2.0 * np.pi * x) + 0.12 * np.sin(5.0 * np.pi * x + rng.rand() * 6.283)
    v += 0.03 * rng.randn(n)
    return project(v, n)


def pcoeff(h, n, i, j):
    off = n - 1
    C = np.zeros(2 * n - 1, dtype=float)
    outr = 1.0 - h[::-1]
    C[i:i + n] += outr
    C[off - i:off - i + n] -= h
    C[j:j + n] -= outr
    C[off - j:off - j + n] += h
    return C


def peval(corr, C, dx, dvec, off, D):
    v = corr[None, :] + (dx * dvec[:, None]) * C[None, :]
    if D:
        q = dx * (dvec * dvec)[:, None]
        v[:, off] -= 2.0 * q[:, 0]
        v[:, off + D] += q[:, 0]
        v[:, off - D] += q[:, 0]
    return np.max(v, axis=1)


def psearch(h, corr, i, j, n, cm, dx):
    if i == j:
        return None
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    dl = lo - x0
    dh = hi - x0
    if dh - dl < 1e-14:
        return None
    C = pcoeff(h, n, i, j)
    D = j - i
    off = n - 1
    g = np.linspace(dl, dh, 49)
    g = np.unique(np.clip(np.concatenate((g, np.array([0.0]))), dl, dh))
    vals = peval(corr, C, dx, g, off, D)
    bi = int(np.argmin(vals))
    bv = float(vals[bi])
    if bv >= cm - 1e-12:
        return None
    lg = len(g)
    if bi == 0:
        a, b = g[0], g[min(3, lg - 1)]
    elif bi == lg - 1:
        a, b = g[max(0, lg - 4)], g[-1]
    else:
        a, b = g[bi - 1], g[bi + 1]
    if b - a < 1e-12:
        return None
    g2 = np.linspace(a, b, 27)
    vals2 = peval(corr, C, dx, g2, off, D)
    bi2 = int(np.argmin(vals2))
    bv2 = float(vals2[bi2])
    if bv2 < cm - 1e-12 and abs(float(g2[bi2])) > 1e-13:
        return float(g2[bi2]), C
    return None


def pair_local_search(h, n, seed, dl, maxno=25000):
    if n < 3:
        h = project(h, n)
        return h, eval_c(h, n)
    h = project(h, n)
    corr = compute_corr(h, n)
    dx = 2.0 / n
    curc = float(corr.max() * dx)
    bestc = curc
    besth = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1
    dists = np.arange(1, n)
    topk = min(10, 2 * n - 1)
    noimp = 0
    moves = 0
    while time.time() < dl and noimp < maxno:
        if curc > bestc + 1e-8:
            h = besth.copy()
            corr = compute_corr(h, n)
            curc = float(corr.max() * dx)
            noimp = 0
        if rng.random() < 0.62 and topk > 1:
            idxs = np.argsort(corr)[-topk]
            w = corr[idxs] - float(np.min(corr[idxs])) + 1e-8
            idx = int(rng.choice(idxs, p=w / w.sum()))
            d = int(idx - off)
        else:
            pos = corr[off + 1:]
            neg = corr[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-8
            d = int(rng.choice(dists, p=w / w.sum()))
        ad = abs(d)
        m = n - ad
        if m <= 0:
            continue
        if rng.random() < 0.8:
            a = h[:m]
            b = h[ad:ad + m]
            term = a * (1.0 - b) + b * (1.0 - a) + 1e-10
            p = int(rng.choice(m, p=term / term.sum()))
        else:
            p = int(rng.randint(0, m))
        i = p
        j = p + ad
        res = psearch(h, corr, i, j, n, curc, dx)
        if res is None:
            noimp += 1
            continue
        delta, C = res
        x0 = float(h[i])
        y0 = float(h[j])
        S = x0 + y0
        xn = float(np.clip(x0 + delta, 0.0, 1.0))
        yn = S - xn
        if yn < 0.0:
            xn = S
            yn = 0.0
        elif yn > 1.0:
            xn = S - 1.0
            yn = 1.0
        actual = float(xn - x0)
        if abs(actual) < 1e-14:
            noimp += 1
            continue
        q2 = actual * actual
        corr += actual * C
        corr[off] -= 2.0 * q2
        D = j - i
        corr[off + D] += q2
        corr[off - D] += q2
        h[i] = xn
        h[j] = yn
        moves += 1
        curc = float(corr.max() * dx)
        if curc < bestc - 1e-13:
            bestc = curc
            besth = h.copy()
        noimp = 0
        if moves % 250 == 0:
            corr = compute_corr(h, n)
            curc = float(corr.max() * dx)
            if curc > bestc + 1e-9:
                h = besth.copy()
                corr = compute_corr(h, n)
                curc = float(corr.max() * dx)
    bh = project(besth, n)
    return bh, eval_c(bh, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(2.0, budget_s - 8.0)

    try:
        hp = np.asarray(initial_h_values, dtype=float).ravel()
        n0 = int(hp.size)
    except Exception:
        hp = make_initial(400, seed)
        n0 = 400

    if (not np.all(np.isfinite(hp))) or n0 < 80 or n0 > 900:
        if hp.size > 0:
            hp = interp_h(hp, n0, 400)
        else:
            hp = make_initial(400, seed)
        n0 = 400

    hp = project(hp, n0)
    besth = hp.copy()
    bestc = eval_c(hp, n0)
    bestn = n0

    now = time.time()
    if deadline - now > 330:
        sm_t = min(100.0, 0.12 * budget_s)
        if sm_t > 15.0:
            h2, c2 = smooth(besth, bestn, seed, now + sm_t)
            if c2 < bestc:
                bestc = c2
                besth = h2

    now = time.time()
    if deadline - now > 90:
        pair_t = min(620.0, 0.50 * budget_s, max(30.0, deadline - now - 240.0))
        if pair_t > 20.0:
            h2, c2 = pair_local_search(besth, bestn, seed + 101, now + pair_t, maxno=30000)
            if c2 < bestc:
                bestc = c2
                besth = h2

    rng = np.random.RandomState(seed + 17)
    now = time.time()
    for ep in (0.05, 0.03, 0.02):
        if deadline - now < 150:
            break
        alloc = min(115.0, deadline - now - 90.0)
        if alloc < 35.0:
            break
        hstar = project(besth + ep * rng.randn(bestn), bestn)
        cstar = eval_c(hstar, bestn)
        huse = hstar
        sm_t = min(18.0, alloc - 35.0)
        if sm_t > 6.0:
            hsm, csm = smooth(huse, bestn, seed + int(ep * 1000), now + sm_t)
            if csm < cstar:
                huse = hsm
                cstar = csm
        pdl = min(deadline - 30.0, time.time() + alloc - (sm_t if sm_t > 6.0 else 0.0))
        if pdl > time.time() + 10.0:
            h2, c2 = pair_local_search(huse, bestn, seed + 10000 + int(ep * 1000), pdl, maxno=9000)
            if c2 < bestc:
                bestc = c2
                besth = h2
        now = time.time()

    cands = []
    for m in (256, 300, 350, 400, 448, 512, 640, 768, 896):
        if 128 <= m <= 896 and m != bestn and m not in cands:
            cands.append(m)

    now = time.time()
    for m in cands:
        if deadline - now < 105:
            break
        alloc = min(135.0, deadline - now - 45.0)
        if alloc < 45.0:
            break
        if m == bestn:
            hinit = besth.copy()
        else:
            hinit = interp_h(besth, bestn, m)
            hinit = project(hinit + 0.0025 * np.random.RandomState(seed + 777 + m).randn(m), m)
        huse = hinit
        cinit = eval_c(huse, m)
        sm_t = min(16.0, alloc - 45.0)
        if alloc > 80.0 and sm_t > 6.0:
            hsm, csm = smooth(huse, m, seed + 90000 + m, now + sm_t)
            if csm < cinit:
                huse = hsm
                cinit = csm
        pdl = min(deadline - 25.0, time.time() + alloc - (sm_t if sm_t > 6.0 else 0.0))
        if pdl > time.time() + 12.0:
            h2, c2 = pair_local_search(huse, m, seed + 500000 + m, pdl, maxno=12000)
            if c2 < bestc:
                bestc = c2
                besth = h2
                bestn = m
        now = time.time()

    if time.time() < deadline - 4:
        h2, c2 = pair_local_search(besth, bestn, seed + 999999, deadline - 2.0, maxno=18000)
        if c2 < bestc:
            bestc = c2
            besth = h2

    besth = project(besth, bestn)
    bestc = eval_c(besth, bestn)
    return besth, float(bestc), int(bestn)
```
