from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps, solver='CLARABEL'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = compute_gradients(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLOP', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate') and delta.value is not None:
                solved = True
                break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.5, 0.3, 0.2, 0.1, 0.05]:
        h_try = h + lam * d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-15:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_polish(h, n, n_steps=500):
    best_c5 = compute_c5(h, n)
    for t in range(n_steps):
        C = np.correlate(h, 1 - h, mode='full')
        threshold = C.max() - 0.01 * C.std()
        active = np.where(C >= threshold)[0]
        g = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            j1s = max(0, -k)
            j1e = min(n, n - k)
            if j1e > j1s:
                g[j1s:j1e] += (1.0 - h[j1s + k:j1e + k])
            j2s = max(0, k)
            j2e = min(n, n + k)
            if j2e > j2s:
                g[j2s:j2e] -= h[j2s - k:j2e - k]
        g /= len(active)
        alpha_t = 0.02 / (1.0 + 0.01 * t)
        h_new = h - alpha_t * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
    return h, best_c5
