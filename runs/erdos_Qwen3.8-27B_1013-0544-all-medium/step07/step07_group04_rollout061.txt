def subgradient_descent(h, n, deadline, eps=0.005):
    h = project_box_sum(h, n)
    besth = h.copy()
    bestc = float(compute_c(h, n).max())
    dx = 2.0/n
    off = n-1
    
    step = 0.1
    for _ in range(10000):
        if time.time() > deadline: break
        c = compute_c(h, n)
        cm = c.max()
        if cm < bestc:
            bestc = cm
            besth = h.copy()
            step = min(0.5, step * 1.1)
        
        active = np.where(c >= cm - eps)[0]
        # Compute subgradient
        g = np.zeros(n)
        for idx in active:
            k = idx - off
            if k > 0:
                mlen = n - k
                g[k:] += (1.0 - h[:mlen])
                g[:mlen] -= h[k:]
            elif k < 0:
                s = -k
                mlen = n - s
                g[s:] += (1.0 - h[:mlen])
                g[:mlen] -= h[s:]
            else:
                g += (1.0 - 2.0*h)
        g *= dx / len(active)
        
        # Step
        h_new = h - step * g
        h_new = project_box_sum(h_new, n)
        h = h_new
    
    return project_box_sum(besth, n), float(compute_c(besth, n).max())

    off = n - 1
    L = 2*n - 1

    m_all = np.arange(L)
    k = m_all - off

    safe_i1 = np.clip(idx_i1, 0, n-1)
    mask_i1 = (idx_i1 >= 0) & (idx_i1 < n)
    v_i1 = np.where(mask_i1, 1.0 - h[safe_i1], 0.0)
    
    idx_i2 = i + k
    safe_i2 = np.clip(idx_i2, 0, n-1)
    mask_i2 = (idx_i2 >= 0) & (idx_i2 < n)
    v_i2 = np.where(mask_i2, -h[safe_i2], 0.0)

    idx_j1 = j - k
    safe_j1 = np.clip(idx_j1, 0, n-1)
    mask_j1 = (idx_j1 >= 0) & (idx_j1 < n)
    v_j1 = np.where(mask_j1, 1.0 - h[safe_j1], 0.0)
    
    idx_j2 = j + k
    safe_j2 = np.clip(idx_j2, 0, n-1)

    v_j2 = np.where(mask_j2, -h[safe_j2], 0.0)
    
    B = v_j1 + v_j2
    
    a = dx * A
    b = dx * B
    # Objective: max_m (c[m] + a[m]*(x-x0) + b[m]*(y-y0))
    # = max_m (c[m] - a[m]*x0 - b[m]*y0 + a[m]*x + b[m]*y)

    xv, yv = x0, y0
    for _ in range(iters_cd):

        by = c0 + a*xv

    cur_val = float(c.max())
    if val < cur_val - 1e-12:

    return None

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    tgt = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.clip(v-mid, 0, 1).sum() > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5*(lo+hi), 0, 1)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14: break
        m = (v > 1e-12) & (v < 1-1e-12)
        if m.any(): v[m] += d/m.sum()
        else: v += d/n
        v = np.clip(v, 0, 1)
    return v

def lse_obj(h, n, alpha):
    c = compute_c(h, n)
    m = c.max()
    if alpha < 10: return m
    sh = alpha * (c - m)
    return m + np.log(np.exp(sh).sum()) / alpha

def lse_grad(h, n, alpha):
    c = compute_c(h, n)
    m = c.max()
    sh = alpha * (c - m)
    e = np.exp(sh); q = e / e.sum()
    off = n - 1; dx = 2.0/n; g = np.zeros(n)
    nz = np.where(q > 1e-10)[0]
    for idx in nz:
        k = idx - off; w = q[idx]
        if k > 0:
            ml = n-k
            g[:ml] -= w*h[k:]; g[k:] += w*(1.0-h[:ml])
        elif k < 0:
            s = -k; ml = n-s
            g[:ml] += w*(1.0-h[s:]); g[s:] -= w*h[:ml]
        else:
            g += w*(1.0-2.0*h)
    return dx*g

def pen_obj(h, n, alpha, mu):
    d = h.sum()-0.5*n
    return lse_obj(h,n,alpha)+0.5*mu*d*d

def pen_grad(h, n, alpha, mu):
    d = h.sum()-0.5*n
    return lse_grad(h,n,alpha)+mu*d

def pl_min_1d(a, b, lo=0.0, hi=1.0, iters=40):
    if hi-lo < 1e-14:
        x = 0.5*(lo+hi); return x, float(np.max(a*x+b))
    ip = (np.sqrt(5.0)-1.0)/2.0
    x1 = hi-ip*(hi-lo); x2 = lo+ip*(hi-lo)
    f1 = float(np.max(a*x1+b)); f2 = float(np.max(a*x2+b))
    for _ in range(iters):
        if f1 <= f2:
            hi=x2; x2=x1; f2=f1; x1=hi-ip*(hi-lo); f1=float(np.max(a*x1+b))
        else:
            lo=x1; x1=x2; f1=f2; x2=lo+ip*(hi-lo); f2=float(np.max(a*x2+b))
    x = 0.5*(lo+hi); return x, float(np.max(a*x+b))

def pair_free_search(h, c, i, j, n, cd_iters=4):
    dx = 2.0/n; off = n-1; L = 2*n-1
    x0, y0 = h[i], h[j]
    ka = np.arange(L) - off
    ii1 = i-ka; si1 = np.clip(ii1,0,n-1); mi1 = (ii1>=0)&(ii1<n)
    ii2 = i+ka; si2 = np.clip(ii2,0,n-1); mi2 = (ii2>=0)&(ii2<n)
    j1 = j-ka; sj1 = np.clip(j1,0,n-1); mj1 = (j1>=0)&(j1<n)
    j2 = j+ka; sj2 = np.clip(j2,0,n-1); mj2 = (j2>=0)&(j2<n)
    A = np.where(mi1, 1.0-h[si1], 0.0) + np.where(mi2, -h[si2], 0.0)
    B = np.where(mj1, 1.0-h[sj1], 0.0) + np.where(mj2, -h[sj2], 0.0)
    a = dx*A; b = dx*B
    c0 = c - a*x0 - b*y0
    xv, yv = x0, y0
    for _ in range(cd_iters):
        bx = c0 + b*yv
        xv, _ = pl_min_1d(a, bx, 0.0, 1.0, 40)
        by = c0 + a*xv
        yv, _ = pl_min_1d(b, by, 0.0, 1.0, 40)
    val = float(np.max(c0 + a*xv + b*yv))
    if val < c.max() - 1e-12:
        return xv, yv, val
    return None

def local_search(h, deadline, seed=1, min_time=2.0):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max()); besth = h.copy()
    rng = np.random.RandomState(seed)
    off = n-1; noimp = 0; maxno = 5000
    while time.time() < deadline and noimp < maxno:
        if time.time() - max(deadline - (deadline-time.time()), 0) < 0:
            pass
        r = rng.random()
        if r < 0.7:
            pos = c[off+1:]; neg = c[off-1::-1]
            w = np.maximum(pos, neg) + 1e-12
            w /= w.sum()
            d = int(rng.choice(n-1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n-d)); j = i+d
        res = pair_free_search(h, c, i, j, n, cd_iters=4)
        if res is not None:
            xv, yv, _ = res
            h[i] = xv; h[j] = yv
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc - 1e-14:
                bestc = v; besth = h.copy(); noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())

def subgrad_polish(h, deadline, n_active=8):
    n = len(h)
    h = project_box_sum(h, n)
    besth = h.copy(); bestc = float(compute_c(h,n).max())
    dx = 2.0/n; off = n-1
    step = 0.05
    for _ in range(20000):
        if time.time() > deadline: break
        c = compute_c(h, n); cm = c.max()
        if cm < bestc - 1e-14:
            bestc = cm; besth = h.copy(); step = min(0.2, step*1.05)
        top_idx = np.argsort(c)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - off
            if k > 0:
                ml = n-k
                g[:ml] -= h[k:]; g[k:] += (1.0-h[:ml])
            elif k < 0:
                s = -k; ml = n-s
                g[:ml] += (1.0-h[s:]); g[s:] -= h[:ml]
            else:
                g += (1.0-2.0*h)
        g *= dx / n_active
        h_new = project_box_sum(h - step*g, n)
        cn = float(compute_c(h_new, n).max())
        if cn <= cm + 1e-15:
            h = h_new
        else:
            step *= 0.7
            if step < 1e-6: break
    return project_box_sum(besth, n), float(compute_c(besth, n).max())

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        hp = np.array(initial_h_values, dtype=float).copy()

    def do_lbfgs(h, n, alpha, mu, maxit):
