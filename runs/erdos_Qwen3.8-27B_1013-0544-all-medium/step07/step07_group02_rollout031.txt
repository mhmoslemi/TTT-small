def compute_c5(h, n):

def project_feasible(v, n):
    v = np.clip(np.array(v, dtype=float, copy=True), 0.0, 1.0)
    t = 0.5 * n

        if abs(d) < 1e-13:

        else:
            sv = float(v.sum())
            if sv < 1e-15:

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    q0 = max(0.0, float(np.dot(h, 1.0 - h)))
    if n <= 1:
        return q0, np.empty(0), np.empty(0)
    r = np.correlate(h, h, mode='full')
    m = np.arange(1, n, dtype=int)
    rp = r[n:2 * n - 1]
    qp = pref[n - m] - rp
    qn = pref[n] - pref[m] - rp
    qp = np.where(qp > 1e-12, qp, 0.0)
    qn = np.where(qn > 1e-12, qn, 0.0)
    return q0, qp, qn

def logsum_weights(q0, qpos, qneg, n, beta):
    dx = 2.0 / n
    f0 = q0 * dx
    if n == 1:
        return 1.0, np.empty(0), np.empty(0), f0
    fp = qpos * dx
    fq = qneg * dx
    maxf = max(float(f0), float(np.max(fp)), float(np.max(fq)))
    if beta <= 0.0:
        p = 1.0 / (2.0 * n - 1.0)
        return p, np.full(n - 1, p), np.full(n - 1, p), maxf
    e0 = np.exp(beta * (f0 - maxf))
    ep = np.exp(beta * (fp - maxf))
    eq = np.exp(beta * (fq - maxf))
    z = float(e0 + np.sum(ep) + np.sum(eq))
    if z <= 0.0:
        return 0.0, np.zeros(n - 1), np.zeros(n - 1), maxf
    p0 = e0 / z
    pp = ep / z
    pq = eq / z
    return p0, pp, pq, maxf + np.log(z) / beta

def gradient_from_weights(h, p0, pp, pq, n):
    dx = 2.0 / n
    idx = np.arange(n)
    pa = np.zeros(n)
    pb = np.zeros(n)
    c = pp + pq if n > 1 else np.empty(0)
    if n > 1:
        pa[1:] = np.cumsum(pp)
        pb[1:] = np.cumsum(pq)
    first = pa[n - 1 - idx] + pb[idx]
    cz = np.zeros(n)
    if n > 1:
        cz[1:] = c
    conv1 = np.convolve(h, cz, 'full')[:n]
    dr = np.zeros(n)
    if n > 1:
        dr[:-1] = c[::-1]
    conv2 = np.convolve(h, dr, 'full')[n - 1:2 * n - 1]
    gq = first - conv1 - conv2
    return (gq + p0 * (1.0 - 2.0 * h)) / dx

def gradient_row(h, n, typ, m):
    g = np.zeros(n)
    if typ == 0:
        g.fill(1.0)
        g -= 2.0 * h
    elif typ == 1 and 1 <= m < n:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    elif typ == 2 and 1 <= m < n:
        g[:n - m] -= h[m:]
        g[m:] += 1.0 - h[:n - m]
    return g

def select_active_lags(h, n, top_k):
    q0, qp, qn = lag_q(h)
    items = [(float(q0), 0, 0)]
    if n > 1:
        for i, v in enumerate(qp):
            items.append((float(v), 1, i + 1))
        for i, v in enumerate(qn):
            items.append((float(v), 2, i + 1))
    items.sort(reverse=True)
    maxq = float(items[0][0])
    tol = 0.05 * maxq + 1e-12
    active = []
    for v, t, m in items:
        if v >= maxq - tol:
            active.append((t, m, v))
        else:
            break
    if len(active) > top_k:
        active = active[:top_k]
    else:
        start = len(active)
        for v, t, m in items[start:]:
            active.append((t, m, v))
            if len(active) >= top_k:
                break
    return active, maxq

def active_lp_step(h, n, top_k, eps, current_c=None):
    if n == 0:
        return h, 1e6
    active, _ = select_active_lags(h, n, top_k)
    if current_c is None:
        current_c = compute_c5(h, n)
    if not active:
        return h, current_c
    k = len(active)
    aub = np.empty((k, n + 1), dtype=float)
    pub = np.empty(k, dtype=float)
    for row, (typ, m, qv) in enumerate(active):
        g = gradient_row(h, n, typ, m)
        aub[row, :n] = g
        aub[row, n] = -1.0
        pub[row] = -qv
    cv = np.zeros(n + 1)
    cv[n] = 1.0
    lo = np.maximum(-h, -eps)
    hi = np.minimum(1.0 - h, eps)
    bad = lo > hi
    if np.any(bad):
        mid = 0.5 * (lo[bad] + hi[bad])
        lo[bad] = mid
        hi[bad] = mid
    bounds = []
    for i in range(n):
        bounds.append((float(lo[i]), float(hi[i])))
    bounds.append((None, None))
    ae = np.zeros((1, n + 1), dtype=float)
    ae[0, :n] = 1.0
    be = np.zeros(1, dtype=float)
    sol = None
    try:
        sol = linprog(cv, A_ub=aub, b_ub=pub, A_eq=ae, b_eq=be, bounds=bounds, method='highs')
    except Exception:
        try:
            sol = linprog(cv, A_ub=aub, b_ub=pub, A_eq=ae, b_eq=be, bounds=bounds, method='interior-point')
        except Exception:
            sol = None
    d = None
    if sol is not None and sol.status == 0 and sol.x is not None:
        d = np.array(sol.x[:n], dtype=float)
    if d is None:
        g = np.sum([gradient_row(h, n, ty, m) for (ty, m, _q) in active], axis=0)
        g *= 1.0 / float(max(1, len(active)))
        d = -g
    d[np.abs(d) < 1e-12] = 0.0
    d -= d.mean()
    d = np.clip(d, -h - 1e-12, 1.0 - h + 1e-12)
    d = np.clip(d, -eps, eps)
    d -= d.mean()
    if float(np.max(np.abs(d))) < 1e-12:
        return h, current_c
    bh = h
    bc = current_c
    for lam in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02):
        ht = project_feasible(h + lam * d, n)
        c = compute_c5(ht, n)
        if c < bc - 1e-12:
            bc = c
            bh = ht
    return bh, bc

def smooth_step(h, n, beta, deadline, max_iter=120):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = (0.1, 0.05, 0.025, 0.0125, 0.00625, 0.003125, 0.0015625)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        q0, qp, qn = lag_q(h)
        p0, pp, pq, _s = logsum_weights(q0, qp, qn, n, beta)
        g = gradient_from_weights(h, p0, pp, pq, n)
        gm = float(np.max(np.abs(g)))
        if gm <= 1e-12:
            break
        u = -g / gm
        u -= u.mean()
        um = float(np.max(np.abs(u)))
        if um <= 1e-12:
            break
        u /= um
        improved = False
        for a in steps:
            ht = project_feasible(h + a * u, n)
            c = compute_c5(ht, n)
            if c < best_c - 1e-12:
                best_h = ht
                best_c = c
                h = ht
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def resample(h, new_n):
    n = len(h)
    if new_n <= 1:
        hh = np.full(max(1, new_n), float(h[0]) if n else 0.5)
        return project_feasible(hh, new_n)
    xo = np.linspace(0.0, 2.0, n)
    xn = np.linspace(0.0, 2.0, new_n)
    hh = np.interp(xn, xo, h)
    return project_feasible(hh, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(25.0, float(budget_s) - 15.0)
    deadline = t0 + total
    iv = globals().get('initial_h_values')
    if iv is not None:
        try:
            h = np.array(iv, dtype=float).copy()
            n = int(len(h))
        except Exception:
            n = 400
            h = np.full(n, 0.5)
    else:
        n = 400
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.40 * np.cos(np.pi * x) + 0.25 * np.cos(3.0 * np.pi * x) + 0.15 * np.cos(5.0 * np.pi * x)
    h = project_feasible(h, n)
    best = h.copy()
    best_c = compute_c5(h, n)
    top_k = int(min(112, max(32, n // 4)))

    p1 = t0 + 0.50 * total
    eps = 0.12
    no = 0
    while time.time() < p1 - 2.0:
        h2, c2 = active_lp_step(h, n, top_k, eps, best_c)
        if c2 < best_c - 1e-12:
            best = h2.copy()
            best_c = c2
            h = best.copy()
            no = 0
            eps = max(eps * 0.995, 0.015)
        else:
            h = best.copy()
            no += 1
            if no >= 12:
                eps *= 0.7
                no = 0
                if eps < 0.015:
                    break

    p2 = t0 + 0.68 * total
    for beta in (25.0, 50.0, 100.0, 200.0, 400.0):
        if time.time() > p2:
            break
        hh, cc = smooth_step(best, n, beta, min(time.time() + 15.0, p2), 100)
        if cc < best_c - 1e-12:
            best = hh
            best_c = cc

    p3 = t0 + 0.85 * total
    eps = 0.03
    no = 0
    while time.time() < p3 - 2.0:
        h2, c2 = active_lp_step(h, n, min(top_k, 72), eps, best_c)
        if c2 < best_c - 1e-12:
            best = h2.copy()
            best_c = c2
            h = best.copy()
            no = 0
            eps = max(eps * 0.9, 0.003)
        else:
            h = best.copy()
            no += 1
            if no >= 10:
                eps *= 0.6
                no = 0
                if eps < 0.003:
                    break

    if n <= 400 and time.time() < deadline - 120:
        cands = (600, 800)
    elif n < 600 and time.time() < deadline - 80:
        cands = (800,)
    else:
        cands = ()
    for nh in cands:
        if time.time() > deadline - 25:
            break
        hnew = resample(best, nh)
        cnew = compute_c5(hnew, nh)
        top_n = int(min(112, max(32, nh // 4)))
        sub = min(time.time() + 45.0, deadline - 20.0)
        if cnew <= best_c + 0.02:
            epsn = 0.08
            no = 0
            while time.time() < sub - 2.0:
                h3, c3 = active_lp_step(hnew, nh, top_n, epsn, cnew)
                if c3 < cnew - 1e-12:
                    hnew = h3
                    cnew = c3
                    no = 0
                    epsn = max(epsn * 0.95, 0.005)
                else:
                    no += 1
                    if no >= 8:
                        epsn *= 0.7
                        no = 0
                        if epsn < 0.005:
                            break
            if time.time() < sub - 4.0:
                h3, c3 = smooth_step(hnew, nh, 100.0, sub, 50)
                if c3 < cnew:
                    hnew = h3
                    cnew = c3
        if cnew < best_c - 1e-12:
            best = hnew
            best_c = float(cnew)
            n = nh
            top_k = top_n

    fin = deadline - 3.0
    if time.time() < fin:
        h = best.copy()
        eps = 0.01
        for _ in range(25):
            if time.time() > fin - 2.0:
                break
            h2, c2 = active_lp_step(h, n, min(top_k, 56), eps, best_c)
            if c2 < best_c - 1e-12:
                best = h2.copy()
                best_c = c2
                h = best.copy()
                eps = max(eps * 0.9, 0.001)
            else:
                break
    if time.time() < fin:
        hh, cc = smooth_step(best, n, 400.0, fin, 50)
        if cc < best_c - 1e-12:
            best = hh
            best_c = cc

    final = project_feasible(best, n)
    fc = compute_c5(final, n)
    if fc <= best_c:
        best = final
        best_c = fc
    return best, float(best_c), int(n)
