```python
def coord_descent_step(h, n, max_sweeps=1000):
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Compute coefficients for h[j] in each c[k]
            # c[k] = sum_i h[i]*(1-h[i+k])
            # Terms with h[j]: 
            #   i=j: h[j]*(1-h[j+k]) if 0<=j+k<n
            #   i=j-k: h[j-k]*(1-h[j]) if 0<=j-k<n
            
            # Current c[k] without the h[j] contribution, plus parametric form
            # c[k] + Δ*((1-h[j+k]) if valid) + Δ*(-h[j-k]) if valid)
            # where Δ = h[j]_new - h[j]_old
            
            delta_lo = -h[j]  # h[j]_new can go down to 0
            delta_hi = 1.0 - h[j]  # h[j]_new can go up to 1
            
            # coeff[k] for each k
            coeff = np.zeros(2*n-1)
            # For shift k (0-indexed in the correlation array, offset by n-1):
            # actual shift = k_idx - (n-1)
            for k_idx in range(2*n-1):
                shift = k_idx - (n-1)
                c_val = 0.0
                if 0 <= j + shift < n:
                    c_val += (1.0 - h[j+shift])
                if 0 <= j - shift < n:
                    c_val -= h[j-shift]
                coeff[k_idx] = c_val
            
            # Now minimize max_k (c[k] + delta * coeff[k]) over delta in [delta_lo, delta_hi]
            # This is a 1D convex problem
            
            # Use bisection / ternary search
            # f(delta) = max(c + delta * coeff)
            lo = delta_lo
            hi = delta_hi
            # Ternary search for convex function
            for _ in range(40):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                f1 = np.max(c + m1 * coeff)
                f2 = np.max(c + m2 * coeff)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            delta_opt = (lo + hi) / 2
            h_new_j = h[j] + delta_opt
            if abs(delta_opt) > 1e-12:
                h[j] = h_new_j
                # Update c
                c += delta_opt * coeff
                improved = True
            else:
                # Still need to update c to be consistent
                # Actually c is already correct if delta_opt is 0
                pass
        
        if not improved:
            break
    
    return h
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=100, time_limit=None, rng=None):
    """Coordinate descent for the overlap minimization problem."""
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        total_change = 0.0
        # Random order for diversity
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)

            # Compute coefficient of h[j] change in each correlation value
            coeff = np.zeros(2 * n - 1)
            for k_idx in range(2 * n - 1):
                shift = k_idx - (n - 1)
                cv = 0.0
                jp = j + shift
                jm = j - shift
                if 0 <= jp < n:
                    cv += (1.0 - h[jp])
                if 0 <= jm < n:
                    cv -= h[jm]
                coeff[k_idx] = cv
            
            # Minimize max(c + delta * coeff) over delta in [-h[j], 1-h[j]]
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-14:
                continue
            
            # Ternary search
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(c + m1 * coeff)
                f2 = np.max(c + m2 * coeff)
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1
            
            delta_opt = (lo + hi) / 2.0
            if abs(delta_opt) > 1e-12:
                h[j] += delta_opt
                c += delta_opt * coeff
                total_change += abs(delta_opt)
        
        if total_change < 1e-10:
            break
    
    return h
```

```python
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_coeff(h, n, j):
    """Compute the coefficient of h[j] in each correlation value."""
    shifts = np.arange(-(n-1), n)
    jp = j + shifts
    jm = j - shifts
    coeff = np.zeros(2*n-1)
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    coeff[vp] += (1.0 - h[jp[vp]])
    coeff[vm] -= h[jm[vm]]
    return coeff

def coord_descent(h, n, max_sweeps=200, time_limit=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    shifts = np.arange(-(n-1), n)
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        total_change = 0.0
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for j in order:
            jp = j + shifts
            jm = j - shifts
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            coeff = np.zeros(2*n-1)
            coeff[vp] += (1.0 - h[jp[vp]])
            coeff[vm] -= h[jm[vm]]
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-14:
                continue
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(c + m1 * coeff) <= np.max(c + m2 * coeff):
                    hi = m2
                else:
                    lo = m1
            delta = (lo + hi) / 2.0
            if abs(delta) > 1e-12:
                h[j] += delta
                c += delta * coeff
                total_change += abs(delta)
        if total_change < 1e-10:
            break
    return h

def build_G(h, n):
    """Build the Jacobian G where G[j,m] = dC[m]/dh[j]."""
    G = np.zeros((n, 2*n-1))
    shifts = np.arange(-(n-1), n)
    for j in range(n):
        jp = j + shifts
        jm = j - shifts
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        G[j, vp] += (1.0 - h[jp[vp]])
        G[j, vm] -= h[jm[vm]]
    return G

def lp_step(h, n, eps, time_limit=None):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    G = build_G(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in ['CLARABEL', 'GLOP', 'ECOS']:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if delta.value is None:
        return h, float(C.max())
    d = delta.value
    best_h = h.copy()
    best_mc = float(C.max())
    for lam in [1.0, 0.7, 0.5, 0.3, 0.1]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc < best_mc - 1e-14:
            best_mc = mc
            best_h = h_try
    return best_h, best_mc
```

```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=200, time_limit=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    shifts = np.arange(-(n - 1), n)
    m2n1 = 2 * n - 1
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        total_change = 0.0
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for j in order:
            jp = j + shifts
            jm = j - shifts
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            coeff = np.zeros(m2n1)
            coeff[vp] += (1.0 - h[jp[vp]])
            coeff[vm] -= h[jm[vm]]
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-14:
                continue
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(c + m1 * coeff) <= np.max(c + m2 * coeff):
                    hi = m2
                else:
                    lo = m1
            delta = (lo + hi) / 2.0
            if abs(delta) > 1e-13:
                h[j] += delta
                c += delta * coeff
                total_change += abs(delta)
        if total_change < 1e-11:
            break
    return h

def build_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    shifts = np.arange(-(n - 1), n)
    for j in range(n):
        jp = j + shifts
        jm = j - shifts
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        G[j, vp] += (1.0 - h[jp[vp]])
        G[j, vm] -= h[jm[vm]]
    return G

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    G = build_G(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps, delta <= eps,
        delta >= -h, delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for solver in ['CLARABEL', 'GLOP', 'ECOS']:
        try:
            prob.solve(solver=solver)
            if delta.value is not None:
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return h, float(C.max())
    d = delta.value
    best_h = h.copy()
    best_mc = float(C.max())
    for lam in [1.0, 0.7, 0.5, 0.3, 0.1]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc < best_mc - 1e-14:
            best_mc = mc
            best_h = h_try
    return best_h, best_mc
```
