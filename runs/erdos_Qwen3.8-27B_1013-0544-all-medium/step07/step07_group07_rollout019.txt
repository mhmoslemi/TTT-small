def projected_gradient_lse(h, n, alpha, lr, nsteps, deadline):
    dx = 2.0 / n
    m = 2 * n - 1
    h = h.copy()
    best_c = float('inf')
    best_h = h.copy()
    
    for step in range(nsteps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_max = c.max()
        if cur_max < best_c:
            best_c = cur_max
            best_h = h.copy()
        
        # Compute softmax weights
        shifted = alpha * (c - cur_max)
        # Clip for numerical stability
        shifted = np.clip(shifted, -700, 700)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        active = np.where(w > 1e-8)[0]
        for k in active:
            lag = k - (n - 1)
            if lag == 0:
                # Special case: d c_0 / d h[m] = 1 - 2*h[m]
                grad += w[k] * (1 - 2 * h)
            else:
                contrib = np.zeros(n)
                mp = np.arange(n) + lag
                vp = (mp >= 0) & (mp < n)
                contrib[vp] += (1 - h[mp[vp]])
                mm = np.arange(n) - lag
                vm = (mm >= 0) & (mm < n)
                contrib[vm] -= h[mm[vm]]
                grad += w[k] * contrib
        
        # Projected gradient step
        # Backtracking line search
        f_cur = cur_max + np.log(exp_s.sum()) / alpha
        s = lr
        for _ in range(20):
            h_try = project_feasible(h - s * grad, n)
            c_try = np.correlate(h_try, 1 - h_try, mode='full') * dx
            f_try = c_try.max() + np.log(np.exp(alpha * (c_try - c_try.max())).sum()) / alpha
            if f_try < f_cur - 1e-4 * s * np.sum(grad ** 2):
                h = h_try
                break
            s *= 0.5
        else:
            break
    
    return h, best_c, best_h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = np.clip(alpha * (g - m), -700, 0)
        logsum = np.log(np.exp(shifted).sum())
        return m + logsum / alpha
    def grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = np.clip(alpha * (g - m), -700, 0)
        es = np.exp(shifted)
        w = es / es.sum()
        gr = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            lag = k - (n - 1)
            if lag == 0:
                gr += w[k] * (1.0 - 2.0 * x)
            else:
                c = np.zeros(n)
                mp = np.arange(n) + lag
                vp = (mp >= 0) & (mp < n)
                c[vp] += (1.0 - x[mp[vp]])
                mm = np.arange(n) - lag
                vm = (mm >= 0) & (mm < n)
                c[vm] -= x[mm[vm]]
                gr += w[k] * c
        return gr
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

        w = es / es.sum()
        grad = np.zeros(n)
        active = np.where(w > 1e-10)[0]

                mm = np.arange(n) - lag
                vm = (mm >= 0) & (mm < n)

                grad += w[k] * c
        s = lr

            if f_try < f_cur - 1e-4 * s * gnorm:

            s *= 0.5
        else:

    return h, best_c, best_h
