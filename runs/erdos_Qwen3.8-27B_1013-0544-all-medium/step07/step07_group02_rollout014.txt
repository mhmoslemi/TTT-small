import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(h, n, alpha, lam=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    lse = m + (sm + np.log(exps.sum())) / alpha
    weights = exps / exps.sum()
    
    # Gradient of LSE w.r.t. h
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-15:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            js = np.arange(0, n - k)
            grad[js] += w * dx * (1.0 - h[js + k])
            grad[js + k] += w * dx * (-h[js])
        else:
            kk = -k
            js = np.arange(kk, n)
            grad[js] += w * dx * (1.0 - h[js + k])
            grad[js + k] += w * dx * (-h[js])
    
    if lam > 0:
        grad += 2.0 * lam * h  # gradient of lam * sum(h^2)
        lse += lam * np.sum(h**2)
    
    return lse, grad

def optimize_lbfqsb(h, n, alpha, lam, maxiter, ftol):
    def obj(x):
        v, _ = lse_obj_and_grad(x, n, alpha, lam)
        return v
    def grads(x):
        _, g = lse_obj_and_grad(x, n, alpha, lam)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grads, method='L-BFGS-B',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    s_exps = exps.sum()
    obj = m + (sm + np.log(s_exps)) / alpha
    w = exps / s_exps
    if lam > 0:
        obj += lam * np.dot(h, h)
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-18:
            continue
        k = mi - (n - 1)
        if k >= 0:
            js = range(0, n - k)
            # Can't use range with numpy ops directly in a loop-efficient way
            # Let's use array slicing
            a = 1.0 - h[:n-k]
            b = h[:n-k]
            grad[:n-k] += wi * dx * a
            grad[k:] += -wi * dx * b
        else:
            kk = -k
            a = 1.0 - h[kk:]
            b = h[kk:]
            grad[kk:] += wi * dx * a
            grad[:n-kk] += -wi * dx * b
    if lam > 0:
        grad += 2.0 * lam * h
    return obj, grad

def pgd_step(h, n, alpha, lam, step, momentum):
    """One step of projected gradient descent with momentum."""
    obj, grad = lse_obj_grad(h, n, alpha, lam)
    if not hasattr(pgd_step, 'vel'):

    pgd_step.vel = momentum * pgd_step.vel + (1 - momentum) * (-grad)
    h_new = h + step * pgd_step.vel
    h_new = project_feasible(h_new, n)
    return h_new, obj

def optimize_slsqp(h, n, alpha, lam, maxiter, ftol):
    """SLSQP optimization of LSE objective."""
    def obj(x):
        v, _ = lse_obj_grad(x, n, alpha, lam)
        return v
    def grad(x):
        _, g = lse_obj_grad(x, n, alpha, lam)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    s_exps = exps.sum()
    obj = m + (sm + np.log(s_exps)) / alpha
    w = exps / s_exps
    if lam > 0:
        obj += lam * np.dot(h, h)
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-18:
            continue
        k = mi - (n - 1)
        if k >= 0:
            grad[:n - k] += wi * dx * (1.0 - h[k:n])
            grad[k:] += -wi * dx * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wi * dx * (1.0 - h[:n - kk])
            grad[:n - kk] += -wi * dx * h[kk:]
    if lam > 0:
        grad += 2.0 * lam * h
    return obj, grad

def optimize_lbfgsb(h, n, alpha, lam, maxiter, ftol):
    def objf(x):
        v, _ = lse_obj_grad(x, n, alpha, lam)
        return v
    def gradf(x):
        _, g = lse_obj_grad(x, n, alpha, lam)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objf, h, jac=gradf, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pgd_optimize(h, n, alpha, n_iters, step_init, momentum):
    vel = np.zeros(n)
    step = step_init
    best_local = h.copy()
    best_obj = np.inf
    for i in range(n_iters):
        obj, grad = lse_obj_grad(h, n, alpha, 0.0)
        if obj < best_obj:
            best_obj = obj
            best_local = h.copy()
        vel = momentum * vel - (1.0 - momentum) * grad
        h = h + step * vel
        h = project_feasible(h, n)
        step *= 0.9995
    return best_local
