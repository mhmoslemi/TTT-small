        cand, c = lbfgs_opt(best_h, best_n, beta, 80, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 2: Try diverse inits at same resolution, keep best
    rng = np.random.RandomState(int(seed))
    candidates = [best_h.copy()]
    for s in range(4):
        hh = project_box_sum(best_h + 0.05 * rng.randn(best_n), 0.5 * best_n)
        cand, c = lbfgs_opt(hh, best_n, 5000.0, 50, deadline)
        if c < best_c + 0.005:
            candidates.append(cand)
            if c < best_c:
                best_h = cand.copy()
                best_c = c
    # Structured inits
    for s2 in range(3):
        hh = make_structured_init(best_n, int(seed) + 100 + s2)
        cand, c = lbfgs_opt(hh, best_n, 2000.0, 40, deadline)
        if c < best_c + 0.005:
            candidates.append(cand)
            if c < best_c:
                best_h = cand.copy()
                best_c = c

    # Phase 3: Upscale to n=800 and optimize
    n2 = 800
    if time.time() < deadline - 0.25 * (deadline - t0):
        hh = upsample(best_h, best_n, n2)
        for beta in [1000.0, 10000.0, 100000.0, 1000000.0]:
            if time.time() > deadline - 0.1 * (deadline - t0):
                break
            cand, c = lbfgs_opt(hh, n2, beta, 60, deadline)
            if c < best_c - 1e-15:
                best_h = cand.copy()
                best_n = n2
                best_c = c
        # Active set refinement
        if time.time() < deadline - 0.08 * (deadline - t0):
            cand, c = active_set_opt(best_h, best_n, 3, 40, deadline)
            if c < best_c - 1e-15:
                best_h = cand.copy()
                best_c = c

    # Phase 4: Try n=1200 if time permits
    n3 = 1200
    if time.time() < deadline - 0.15 * (deadline - t0):
        hh = upsample(best_h, best_n, n3)
        for beta in [5000.0, 50000.0, 500000.0]:
            if time.time() > deadline - 0.05 * (deadline - t0):
                break
            cand, c = lbfgs_opt(hh, n3, beta, 40, deadline)
            if c < best_c - 1e-15:
                best_h = cand.copy()
                best_n = n3
                best_c = c

    # Phase 5: Pair polish at best resolution
    if best_n <= 1200 and time.time() < deadline - 20.0:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 10.0, seed)

    # Phase 6: Final high-beta L-BFGS
    if time.time() < deadline - 8.0:
        cand, c = lbfgs_opt(best_h, best_n, 5000000.0, 30, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 7: One more active set pass
    if time.time() < deadline - 6.0:
        cand, c = active_set_opt(best_h, best_n, 2, 25, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
