    im = i + m; ms = (im >= 0) & (im < n); d[ms] += 1.0 - h[im[ms]]
    imn = i - m; ms = (imn >= 0) & (imn < n); d[ms] -= h[imn[ms]]
    jm = j + m; ms = (jm >= 0) & (jm < n); d[ms] -= 1.0 - h[jm[ms]]
    jmn = j - m; ms = (jmn >= 0) & (jmn < n); d[ms] += h[jmn[ms]]
    d[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        d[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        d[k2] += 1.0
    return d

def binary_ls(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur = corr.max()
    best_h, best_c = h.copy(), cur
    ones = list(np.where(h == 1)[0].astype(int))
    zeros = list(np.where(h == 0)[0].astype(int))
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        imp = False
        for i in zeros:
            done = False
            for j in ones:
                if time.time() > deadline:
                    break
                d = swap_delta(h, n, i, j)
                nc = (corr + d).max()
                if nc < cur - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur = nc
                    ones.remove(j); zeros.remove(i)
                    ones.append(i); zeros.append(j)
                    imp = True; done = True
                    break
            if done:
                break
        if cur < best_c:
            best_c = cur; best_h = h.copy()
        if not imp:
            break
    return best_h, best_c * dx

def binary_sa(h_init, n, deadline, max_iter=300000, seed=42):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur = corr.max()
    best_h, best_c = h.copy(), cur
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    rng = np.random.RandomState(seed)
    T0, T1 = 5.0, 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max_iter)
        ii = int(zeros[rng.randint(len(zeros))])
        jj = int(ones[rng.randint(len(ones))])
        d = swap_delta(h, n, ii, jj)
        new_max = (corr + d).max()
        de = (new_max - cur)
        if de < 0 or rng.random() < np.exp(-de * dx / max(T, 1e-10)):
            h[ii], h[jj] = 1.0, 0.0
            corr += d
            cur = new_max
            ones = np.append(np.delete(ones, np.where(ones == jj)[0]), ii)
            zeros = np.append(np.delete(zeros, np.where(zeros == ii)[0]), jj)
            if cur * dx < best_c:
                best_c = cur * dx; best_h = h.copy()
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sh = np.clip(sh, -500, 0)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def crit_lag_opt(h, n, dl, k=15):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    idx_top = np.argsort(c)[::-1][:k]
    best = h.copy()
    best_c = compute_c5(h, n)
    t = best_c * 0.995
    for _ in range(40):
        if time.time() > dl - 5:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
        captured_idx = idx_top.copy()
        captured_t = t
        def obj(x):
            cx = np.correlate(x, 1.0 - x, mode='full') * dx
            ex = np.maximum(cx[captured_idx] - captured_t, 0)
            return 0.5 * np.sum(ex ** 2) + 1e-8 * np.sum(cx ** 2)
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 1500, 'ftol': 1e-18})
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c, best = c2, h2
                c = np.correlate(best, 1 - best, mode='full') * dx
                idx_top = np.argsort(c)[::-1][:k]
        except Exception:
            pass
        t = max(t * 0.998, best_c * 0.99)
    return best, best_c

def refine_continuous(h, n, dl):
    best = h.copy()
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    for a in alphas:
        if time.time() > dl - 10:
            break
        mi = 2000 if a < 1e5 else 3000
        h2 = lse_slsqp(best, n, a, mi, 1e-18)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best = c2, h2
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary SA (explore different basins)
    p1 = t0 + 0.10 * budget_s
    hb, cb = binary_sa(h0, n, p1, max_iter=400000, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Binary local search (polish)
    p2 = t0 + 0.18 * budget_s
    hb, cb = binary_ls(best_h, n, p2, max_sweeps=150)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 3: Continuous LSE-SLSQP
    p3 = t0 + 0.50 * budget_s
    best_h, best_c = refine_continuous(best_h, n, p3)

    # Phase 4: Critical-lag focused
    p4 = t0 + 0.62 * budget_s
    h2, c2 = crit_lag_opt(best_h, n, p4, k=15)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()

    # Phase 5: Second SA with different seed + polish
    p5 = t0 + 0.72 * budget_s
    hb, cb = binary_sa(best_h, n, p5, max_iter=200000, seed=seed + 1000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    hb, cb = binary_ls(best_h, n, t0 + 0.78 * budget_s, max_sweeps=80)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 6: Try different resolution
    p6 = t0 + 0.88 * budget_s
    x = np.linspace(0, 2, n)
    for nh in [300, 500]:
        if time.time() > p6 - 10:
            break
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, x, best_h), nh)
        hh, ch = refine_continuous(hh, nh, p6 - 10)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
            x = xn

    # Phase 7: Final polish
    p7 = dl
    best_h, best_c = refine_continuous(best_h, n, p7)
    h2, c2 = crit_lag_opt(best_h, n, p7, k=20)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
