import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v


def build_A(h, n):
    L = 2 * n - 1
    off = n - 1
    A = np.zeros((L, n))
    A[off, :] = 1.0 - 2.0 * h
    for p in range(1, n):
        A[off + p, p:] += (1.0 - h[0:n - p])
        A[off + p, 0:n - p] += -h[p:n]
        A[off - p, 0:n - p] += (1.0 - h[p:n])
        A[off - p, p:] += -h[0:n - p]
    return A


def slp_optimize(h, n, deadline, max_iter=3000):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    dx = 2.0 / n
    L = 2 * n - 1
    noimp = 0
    damp = 1.0
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        if noimp > 150:
            break
        A = build_A(h, n)
        c_base = compute_c(h, n)
        A_ht = A @ h
        rhs = c_base - dx * A_ht
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([dx * A, -np.ones((L, 1))])
        b_ub = -rhs
        A_eq = np.hstack([np.ones((1, n)), np.zeros(1)])
        b_eq = np.array([0.5 * n])
        bounds = [(0, 1)] * n + [(0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method="highs",
                         options={"presolve": True, "time_limit": 3})
        except Exception:
            break
        if res.status != 0 or res.x is None:
            break
        h_lp = res.x[:n]
        h_new = (1.0 - damp) * h + damp * h_lp
        h_new = project_box_sum(h_new, n)
        v_new = float(compute_c(h_new, n).max())
        if v_new < bestc - 1e-12:
            bestc = v_new
            besth = h_new.copy()
            h = h_new
            noimp = 0
            damp = min(1.0, damp * 1.05)
        else:
            noimp += 1
            damp = max(0.05, damp * 0.6)
            h_try = project_box_sum(h_lp, n)
            v_try = float(compute_c(h_try, n).max())
            if v_try < bestc - 1e-12:
                bestc = v_try
                besth = h_try.copy()
                h = h_try
                noimp = 0
                damp = 1.0
    return project_box_sum(besth, n), float(compute_c(besth, n).max())


def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = np.clip(alpha * (c - m), -500, 500)
    s = float(np.exp(sh).sum())
    return m + np.log(max(s, 1e-300)) / alpha if s > 0 else m


def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(sh)
    s = float(e.sum())
    q = e / s if s > 0 else np.full_like(c, 1.0 / len(c))
    dx = 2.0 / n
    g = np.zeros(n)
    off = n - 1
    nz = np.flatnonzero(q > 1e-14)
    for idx in nz:
        k = idx - off
        w = q[idx]
        if k > 0:
            m2 = n - k
            g[k:n] += w * (1.0 - x[0:m2])
            g[0:m2] -= w * x[k:n]
        elif k < 0:
            m2 = n + k
            g[0:m2] += w * (1.0 - x[-k:n])
            g[-k:n] -= w * x[0:m2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g


def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones(len(x))


def lbfgsb_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(lse_obj, h, args=(n, alpha), method="L-BFGS-B",
                       jac=lambda x: lse_grad(x, n, alpha) + mu * (x.sum() - 0.5 * n),
                       bounds=bounds,
                       options={"maxiter": maxiter, "ftol": 1e-15, "gtol": 1e-9})
        if res.x is not None:
            return res.x
    except Exception:
        pass
    return h


def pair_eval_quick(h, c_base, i, j, xs, n, x0, y0):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    S = x0 + y0
    omh = 1.0 - h
    omhrev = omh[::-1]
    best_v = float(c_base.max())
    best_x = x0
    for xi in xs:
        yi = S - xi
        if yi < -1e-12 or yi > 1.0 + 1e-12:
            continue
        xi = float(np.clip(xi, 0.0, 1.0))
        yi = float(np.clip(S - xi, 0.0, 1.0))
        ax1 = xi - x0
        ax2 = x0 - xi
        yv = S - xi
        delta = np.zeros(L)
        delta[i:i + n] += ax1 * omhrev
        si = off - i
        if si >= 0 and si + n <= L:
            delta[si:si + n] += ax2 * h
        delta[j:j + n] += ax2 * omhrev
        sj = off - j
        if sj >= 0 and sj + n <= L:
            delta[sj:sj + n] += ax1 * h
        exact_self = xi * (1 - xi) + yv * (1 - yv) - (x0 * (1 - x0) + y0 * (1 - y0))
        base_self = ax1 * (1 - x0) + x0 * ax2 + ax2 * (1 - y0) + y0 * ax1
        delta[off] += exact_self - base_self
        d = j - i
        idxp = off - d
        if 0 <= idxp < L:
            delta[idxp] += (xi * (1 - yv) - x0 * (1 - y0)) - (ax1 * (1 - y0) + x0 * ax1)
        idxm = off + d
        if 0 <= idxm < L:
            delta[idxm] += (yv * (1 - xi) - y0 * (1 - x0)) - (ax2 * (1 - x0) + y0 * ax2)
        v = float(np.max(c_base + dx * delta))
        if v < best_v:
            best_v = v
            best_x = xi
    if best_v < float(c_base.max()) - 1e-13:
        return best_x, float(np.clip(S - best_x, 0.0, 1.0)), best_v
    return None


def pair_local_search(h, n, deadline, seed):

        j = i + d
        x0, y0 = float(h[i]), float(h[j])
        S = x0 + y0
        lo = max(0.0, S - 1.0)
        hi = min(1.0, S)
        if hi - lo < 1e-13:
            noimp += 1
            continue
        xs = np.concatenate([np.array([lo, hi, x0]), np.linspace(lo, hi, 32)])
        res = pair_eval_quick(h, c, i, j, xs, n, x0, y0)
        if res is not None:
            x, y, bv = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())


def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)
