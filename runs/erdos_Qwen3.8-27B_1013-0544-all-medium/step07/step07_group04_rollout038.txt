def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0

    for _ in range(80):

        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break

        c = int(m.sum())
        if c:

            v += d / n
        v = np.clip(v, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0:
        obj = m
    else:
        obj = m + np.log(s) / alpha
    d = float(x.sum() - 0.5 * n)
    return obj + 0.5 * mu * d * d


def lse_grad(x, n, alpha, mu):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:
        q = e / s
    q = np.maximum(q, 0.0)
    q /= q.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            g[:n+k] += w * (1.0 - x[-k:])
            g[-k:] -= w * x[:n+k]
        else:
            g += w * (1.0 - 2.0 * x)
    d = float(x.sum() - 0.5 * n)
    return dx * g + mu * d

    except Exception:
        pass
    return h

def local_search(h, deadline, seed=1):

                noimp = 0

        i = int(rng.randint(n - d))

            noimp = 0

def interpolate_h(h, nold, nnew):

    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

    bestn = n0
    # Resolve list: include n0 and a few others
    n_list = [n0]
    for m in (300, 256, 200, 400):
        if m not in n_list and 50 <= m <= 600:
            n_list.append(m)
    n_list = list(dict.fromkeys(n_list))  # dedupe preserving order
    L = len(n_list)
    # Time allocation: proportional to 1/n (smaller n = more time per unit work)
    tw = [1.0 / (nn + 50) for nn in n_list]
    tw_sum = sum(tw)
    tw = [w / tw_sum for w in tw]
    cum = 0.0
    total_avail = deadline - t0
    for k, nh in enumerate(n_list):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        cum += tw[k]
        if k == L - 1:
            dl_phase = deadline - 2
        else:
            dl_phase = t0 + total_avail * min(0.92, cum) - 3
        dl_phase = min(dl_phase, deadline - 15)
        if dl_phase <= time.time() + 5:
            continue
        if k == 0:
            hinit = besth.copy()
        else:
            hinit = interpolate_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)
        hopt, copt = optimize_one(hinit, nh, seed + 137 * k, dl_phase)
        if copt < bestc:
            bestc = copt
            besth = hopt.copy()
            bestn = nh
    # Try an alternative: binary-like init at bestn
    if time.time() < deadline - 30:
        hbin = np.zeros(bestn)
        pos = rng.choice(bestn, size=bestn // 2, replace=False)
        hbin[pos] = 1.0
        hbin = project_box_sum(hbin + 0.02 * rng.randn(bestn), bestn)
        dl2 = deadline - 10
        hopt2, copt2 = optimize_one(hbin, bestn, seed + 7777, dl2)
        if copt2 < bestc:
            bestc = copt2
            besth = hopt2.copy()
    # Final polish
    if time.time() < deadline - 8:
        besth, bestc = local_search(besth, deadline - 2, seed=seed + 9999)
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
