```python
import time
import numpy as np
import scipy.optimize as so

ST = {'n': 1, 'dx': 1.0, 'target': 0.5, 'ks': np.array([], dtype=int),
      'ws': np.array([], dtype=float), 'rho': 1.0}

SCHEDULE = ((128, 100.0, 120), (96, 400.0, 110), (64, 1500.0, 110),
            (48, 6000.0, 100), (32, 25000.0, 100), (16, 100000.0, 90),
            (8, 400000.0, 80), (4, 1500000.0, 70), (1, 1.0e9, 60))
QUICK = ((96, 500.0, 45), (64, 2000.0, 45), (32, 10000.0, 35), (16, 50000.0, 30))
FINAL = ((32, 100000.0, 45), (8, 1000000.0, 35), (2, 1.0e9, 30))


def erdos_c5(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def full_lags(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return c, float(np.max(c))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(4):
        if abs(diff) <= 1e-12:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_weights(v, beta):
    v = np.asarray(v, dtype=float)
    m = v.size
    if m <= 1:
        return np.ones(m, dtype=float) / max(m, 1)
    if beta < 1e-6:
        return np.full(m, 1.0 / m, dtype=float)
    mx = float(np.max(v))
    z = beta * (v - mx)
    np.clip(z, -745.0, 0.0, out=z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros(m, dtype=float)
        w[int(np.argmax(v))] = 1.0
        return w
    return e / s


def top_vals(x):
    st = ST
    n = int(st['n'])
    dx = float(st['dx'])
    target = float(st['target'])
    rho = float(st['rho'])
    ks = st['ks']
    ws = st['ws']
    cnt = 0.0
    g = np.zeros(n, dtype=float)
    v = 1.0 - x
    for ii in range(len(ks)):
        wi = float(ws[ii])
        if wi <= 0.0:
            continue
        k = int(ks[ii])
        if k == 0:
            t = float(np.dot(x, v))
            cnt += wi * t
            g += wi * (v - x)
        elif k > 0:
            m = n - k
            a = x[k:]
            b = v[:m]
            t = float(np.dot(a, b))
            cnt += wi * t
            g[k:] += wi * b
            g[:m] -= wi * a
        else:
            d = -k
            m = n - d
            a = x[:m]
            b = v[d:]
            t = float(np.dot(a, b))
            cnt += wi * t
            g[:m] += wi * b
            g[d:] -= wi * a
    res = float(x.sum() - target)
    f = dx * cnt
    g *= dx
    if rho != 0.0:
        f += 0.5 * rho * res * res
        g += rho * res
    return float(f), g


def top_fun(x):
    f, g = top_vals(x)
    return f


def top_jac(x):
    f, g = top_vals(x)
    return g


def top_round(h, n, K, beta, maxiter, deadline, rho=None):
    h = np.asarray(h, dtype=float)
    n = int(n)
    target = 0.5 * n
    dx = 2.0 / n
    if rho is None:
        rho = max(1.0, 800.0 / float(n))
    rem = deadline - time.time()
    if rem < 8.0:
        return h, erdos_c5(h, n)
    c, _ = full_lags(h, n)
    size = 2 * n - 1
    if K < 1:
        K = 1
    if K >= size:
        idx = np.argsort(c)[::-1]
    elif K == 1:
        idx = np.array([int(np.argmax(c))], dtype=int)
    else:
        idx = np.argpartition(c, size - K)[size - K:]
    vals = c[idx]
    w = softmax_weights(vals, beta)
    ST['n'] = n
    ST['dx'] = dx
    ST['target'] = target
    ST['ks'] = idx - (n - 1)
    ST['ws'] = w
    ST['rho'] = rho
    mx = int(min(maxiter, max(10, int(rem / 2.5))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(top_fun, h, jac=top_jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxls': 30, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    if cand is None or not np.all(np.isfinite(cand)):
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def resample_n(h, n, nh):
    h = np.asarray(h, dtype=float)
    n = int(n)
    nh = int(nh)
    if nh == n:
        return project_box_sum(h, 0.5 * nh)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nh)
    hh = np.interp(y, x, h)
    return project_box_sum(hh, 0.5 * nh)


def quick_refresh(h, n, limit):
    n = int(n)
    h = project_box_sum(h, 0.5 * n)
    cur = erdos_c5(h, n)
    best = h.copy()
    bc = cur
    for K, beta, it in QUICK:
        if time.time() > limit - 5.0:
            break
        cand, cc = top_round(h, n, K, beta, it, limit)
        if cc < cur - 1e-13:
            h = cand
            cur = cc
            if cc < bc:
                bc = cc
                best = cand.copy()
    return best, bc


def polish(h, n, deadline):
    n = int(n)
    h = project_box_sum(h, 0.5 * n)
    cur = erdos_c5(h, n)
    cur_h = h.copy()
    best = h.copy()
    bc = cur
    cycles = 0
    while time.time() < deadline - 15.0 and cycles < 4:
        no = 0
        for K, beta, it in SCHEDULE:
            if time.time() > deadline - 20.0:
                break
            rem = deadline - time.time()
            if rem < 15.0:
                break
            itp = int(min(it, max(12, int(rem / 3.0))))
            cand, cc = top_round(cur_h, n, K, beta, itp, deadline)
            if cc < cur - 1e-13:
                cur_h = cand
                cur = cc
                if cc < bc - 1e-15:
                    bc = cc
                    best = cand.copy()
                no = 0
            else:
                no += 1
            if no >= 4:
                break
        cycles += 1
        if no >= 4:
            if cycles < 2 and time.time() < deadline - 120.0:
                rng = np.random.RandomState(1000 + cycles * 101 + n)
                hh = project_box_sum(cur_h + 0.02 * rng.randn(n), 0.5 * n)
                if erdos_c5(hh, n) < cur + 0.004:
                    cand, cc = top_round(hh, n, 64, 2000.0, 55, deadline)
                    if cc < cur - 1e-13:
                        cur_h = cand
                        cur = cc
                        if cc < bc:
                            bc = cc
                            best = cand.copy()
                        continue
            break
    return best, bc


def pair_obj(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, int(n))


def pair_line_search(h, n, i, j, deadline):
    lo = max(float(-h[i]), float(h[j] - 1.0))
    hi = min(float(1.0 - h[i]), float(h[j]))
    if hi - lo < 1e-13:
        return h, erdos_c5(h, n)
    cur = erdos_c5(h, n)
    x0 = 0.0
    c0 = cur
    if time.time() < deadline - 3.0:
        try:
            mx = 10 if n > 600 else 20
            rs = so.minimize_scalar(pair_obj, args=(h, i, j, n),
                                    bounds=(lo, hi), method='bounded',
                                    options={'maxiter': mx, 'xatol': 1e-10})
            x0 = float(rs.x)
            c0 = pair_obj(x0, h, i, j, n)
        except Exception:
            x0 = 0.0
            c0 = cur
        for xd in (lo, hi):
            if time.time() > deadline - 2.0:
                break
            cd = pair_obj(xd, h, i, j, n)
            if cd < c0:
                x0 = xd
                c0 = cd
    if c0 < cur - 1e-14:
        hh = h.copy()
        hh[i] += x0
        hh[j] -= x0
        hh = np.clip(hh, 0.0, 1.0)
        return hh, c0
    return h, cur


def active_gradient(h, n, c, limit):
    n = int(n)
    mid = n - 1
    dx = 2.0 / n
    C = float(np.max(c))
    tol = max(1e-9, C * 1e-5)
    active = np.where(c >= C - tol)[0]
    if active.size > limit:
        active = np.argpartition(c, -limit)[-limit:]
    g = np.zeros(n, dtype=float)
    for idx in active:
        k = int(idx - mid)
        if k == 0:
            g += (1.0 - 2.0 * h)
        elif k > 0:
            m = n - k
            g[k:] += (1.0 - h[:m])
            g[:m] -= h[k:]
        else:
            d = -k
            m = n - d
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
    return g * dx


def pair_polish(h, n, deadline, seed):
    n = int(n)
    h = project_box_sum(h, 0.5 * n)
    cur = erdos_c5(h, n)
    for p in range(2):
        if time.time() > deadline - 25.0:
            break
        c, _ = full_lags(h, n)
        g = active_gradient(h, n, c, 8)
        pos = np.argsort(g)[-4:]
        neg = np.argsort(g)[:4]
        pairs = []
        for ii in pos:
            for jj in neg:
                if int(ii) != int(jj):
                    pairs.append((int(ii), int(jj)))
        pairs.append((int(np.argmax(g)), int(np.argmin(g))))
        rng = np.random.RandomState(int(seed) + 17 * p + 29)
        rng.shuffle(pairs)
        seen = set()
        lim = []
        maxp = 8 if n > 600 else 14
        for pa in pairs:
            if pa not in seen:
                seen.add(pa)
                lim.append(pa)
                if len(lim) >= maxp:
                    break
        for i, j in lim:
            if time.time() > deadline - 30.0:
                break
            hh, cc = pair_line_search(h, n, i, j, deadline - 30.0)
            if cc < cur - 1e-14:
                h = hh
                cur = cc
    return h, cur


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 5.0)
    rng = np.random.RandomState(int(seed) + 7)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
    except Exception:
        n0 = 400
        h0 = np.zeros(n0, dtype=float)
        p = np.random.RandomState(123).permutation(n0)
        h0[p[:n0 // 2]] = 1.0

    if n0 < 4:
        n0 = 400
        h0 = np.full(n0, 0.5, dtype=float)
    if n0 > 900:
        h0 = resample_n(h0, n0, 800)
        n0 = 800
    h0 = project_box_sum(h0, 0.5 * n0)

    cand = []
    seen = set()
    for nt in (n0, min(800, 2 * n0), 512, 320):
        nt = int(nt)
        if 6 <= nt <= 800 and nt not in seen:
            seen.add(nt)
            hh = resample_n(h0, n0, nt)
            cc = erdos_c5(hh, nt)
            cand.append((cc, nt, hh))

    order = []
    for it in range(len(cand)):
        order.append((float(cand[it][0]), int(cand[it][1]), it))
    order.sort()

    best_h = h0.copy()
    best_n = n0
    best_c = erdos_c5(best_h, best_n)

    if len(cand) and len(order):
        cc0, nt0, idx0 = cand[order[0][2]]
        if cc0 < best_c:
            best_c = cc0
            best_h = cand[order[0][2]].copy()
            best_n = nt0

    quick_end = t0 + min(240.0, max(60.0, (deadline - t0) * 0.20))
    for _, _, oi in order:
        c, nt, hh = cand[oi]
        if c > order[0][0] + 0.02:
            break
        if time.time() > quick_end - 35.0:
            break
        limit = min(quick_end - 5.0, time.time() + 90.0)
        hh2, c2 = quick_refresh(hh, nt, limit)
        if c2 < best_c:
            best_c = c2
            best_h = hh2.copy()
            best_n = nt

    main_end = deadline - 80.0
    if time.time() < main_end - 30.0:
        h2, c2 = polish(best_h, best_n, main_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    if time.time() < deadline - 120.0:
        for _, _, oi in order:
            c, nt, hh = cand[oi]
            if nt == best_n:
                continue
            if c > best_c + 0.02:
                break
            if time.time() > deadline - 100.0:
                break
            limit = min(deadline - 100.0, time.time() + 120.0)
            hh2, c2 = quick_refresh(hh, nt, limit)
            if c2 < best_c:
                best_c = c2
                best_h = hh2.copy()
                best_n = nt

    for sig in (0.01, 0.03, 0.08, 0.15):
        if time.time() > deadline - 80.0:
            break
        hh = best_h + sig * rng.randn(best_n)
        hh = project_box_sum(hh, 0.5 * best_n)
        if erdos_c5(hh, best_n) < best_c + 0.02:
            limit = min(deadline - 60.0, time.time() + 80.0)
            hh2, c2 = quick_refresh(hh, best_n, limit)
            if c2 < best_c:
                best_c = c2
                best_h = hh2.copy()

    for K, beta, it in FINAL:
        if time.time() > deadline - 15.0:
            break
        hh2, c2 = top_round(best_h, best_n, K, beta, it, deadline)
        if c2 < best_c:
            best_c = c2
            best_h = hh2.copy()

    if time.time() < deadline - 25.0 and best_n <= 900:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 10.0, int(seed))

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
