from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def lse_value_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    s = alpha * (c - cmax)
    e = np.exp(s)
    p = e / e.sum()
    val = cmax + np.log(e.sum()) / alpha
    
    g = np.zeros(n)
    for lag in range(n):
        w = p[n-1+lag] * dx
        if w > 1e-14:
            g[lag:] += w * (1.0 - h[:n-lag])
            g[:n-lag] -= w * h[lag:]
        w2 = p[n-1-lag] * dx if lag > 0 else 0.0
        if w2 > 1e-14:
            g[:n-lag] += w2 * (1.0 - h[lag:])
            g[lag:] -= w2 * h[:n-lag]
    return val, g

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):

        hp = np.clip(h - tau, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def nesterov_opt(h, n, alpha, max_iter, lr, deadline):
    h = h.copy()
    y = h.copy()
    best_f = lse_value_grad(h, n, alpha)[0]
    best_h = h.copy()
    lr = lr
    for k in range(max_iter):
        if time.time() > deadline:
            break
        beta = k / (k + 2.0)
        y = (1.0 + beta) * h - beta * y
        y = project_feasible(y, n)
        val, g = lse_value_grad(y, n, alpha)
        # Backtracking line search

            trial = project_feasible(y - step * g, n)
            trial_val = lse_value_grad(trial, n, alpha)[0]
            if trial_val <= val + 1e-12 * step * np.abs(g).max():
                break
            step *= 0.5
        h = trial
        if val < best_f:
            best_f = val
            best_h = h.copy()
    return best_h

def slsqp_polish(h, n, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = 1e7 * (c - m)
        return m + np.log(np.exp(s).sum()) / 1e7

        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': 3000, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    s = alpha * (c - cmax)
    e = np.exp(s)
    p = e / e.sum()
    val = cmax + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for lag in range(n):
        w = p[n - 1 + lag] * dx
        if w > 1e-14:
            g[lag:] += w * (1.0 - h[:n - lag])
            g[:n - lag] -= w * h[lag:]
        if lag > 0:
            w2 = p[n - 1 - lag] * dx
            if w2 > 1e-14:
                g[:n - lag] += w2 * (1.0 - h[lag:])
                g[lag:] -= w2 * h[:n - lag]
    return val, g

def topk_grad(h, n, k):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    idx = np.argpartition(c, -k)[-k:]
    val = c[idx].mean()
    g = np.zeros(n)
    for pos in idx:
        lag = pos - (n - 1)
        w = dx / k
        if lag >= 0:
            g[lag:] += w * (1.0 - h[:n - lag])
            g[:n - lag] -= w * h[lag:]
        else:
            t = -lag
            g[:n - t] += w * (1.0 - h[t:])
            g[t:] -= w * h[:n - t]
    return val, g

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        tau = (lo + hi) / 2.0
        hp = np.clip(h - tau, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista(grad_fn, h, n, lr, max_iter, deadline):
    h = h.copy()
    z = h.copy()
    t = 1.0
    best_val = grad_fn(h)[0]
    best_h = h.copy()
    for k in range(max_iter):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = h + (t - 1.0) / t_new * (h - z)
        z = project_feasible(z, n)
        val, g = grad_fn(z)
        # Backtracking
        step = lr
        for _ in range(15):
            trial = project_feasible(z - step * g, n)
            tv = grad_fn(trial)[0]
            if tv <= val + 1e-14:
                break
            step *= 0.5
        h = trial
        if tv < best_val:
            best_val = tv
            best_h = h.copy()
        t = t_new
    return best_h

def slsqp_refine(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        return m + np.log(np.exp(s).sum()) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
