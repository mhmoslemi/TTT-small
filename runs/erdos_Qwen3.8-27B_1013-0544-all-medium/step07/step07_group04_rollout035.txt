Better approach: For coordinate j, c as a function of h_j = t is:

where c_base[idx] is the current c[idx] with h_j removed (set to 0), and delta_idx is the coefficient of h_j in c[idx].

The coefficient of h_j in c[idx]:
- c[idx] = Σ_i h[i] * (1-h[i+k]) * dx where k = idx-(n-1)
- h_j appears when i=j: contribution is h_j * (1-h[j+k]) * dx = t * (1-h[j+k]) * dx
- h_j also appears when i+k=j, i.e., i=j-k: contribution is h[j-k] * (1-h_j) * dx = h[j-k] * (1-t) * dx

So delta_idx (coefficient of t) = (1-h[j+k])*dx - h[j-k]*dx = (1 - h[j+k] - h[j-k])*dx
And c_base[idx] = c[idx] - t_current * delta_idx (where t_current is the current h_j)

The function f(t) = max_idx (c_base[idx] + delta_idx * t) is the max of affine functions of t, hence convex. We can find its minimum over [0,1] by:
- Binary search on t (since the derivative is non-decreasing)
- Or: the minimum is at the point where the two largest "lines" cross

For efficiency, I'll use a ternary search or binary search on the derivative.

Actually, the derivative of f(t) = max_i (a_i + b_i * t) is the b_i of the active (maximum) function at t. Since f is convex, f' is non-decreasing. We can binary search for where f' changes sign.

```python
def coord_optimal_j(h, n, j, c):
    """Find optimal h_j given other coordinates fixed."""
    dx = 2.0 / n
    off = n - 1
    t_cur = h[j]
    
    # Compute delta for each correlation index
    # h_j appears in c[idx] with lag k = idx - off
    # Contribution: h_j*(1-h[j+k])*dx if j+k in [0,n-1]
    #              h[j-k]*(1-h_j)*dx if j-k in [0,n-1]
    
    # For each idx, the coefficient of t is:
    # (1-h[j+k])*dx if j+k valid, minus h[j-k]*dx if j-k valid
    # where k = idx - off
    
    deltas = np.zeros(2*n - 1)
    # h_j as first factor: i=j, k = idx-off, need j+k in [0,n-1]
    # idx = off + k, k = idx - off, j+k in [0,n-1] => k in [-j, n-1-j]
    # idx in [off-j, off+n-1-j] = [n-1-j, 2n-2-j]
    k_start = -j
    k_end = n-1-j
    for k in range(k_start, k_end+1):
        idx = off + k
        if 0 <= j+k < n:
            deltas[idx] += (1.0 - h[j+k]) * dx
    # h_j as second factor: i+k=j => i=j-k, need j-k in [0,n-1]
    # k = idx - off, j-k in [0,n-1] => k in [j-n+1, j]
    k_start = -(n-1-j)
    k_end = j
    for k in range(k_start, k_end+1):
        idx = off + k
        if 0 <= j-k < n:
            deltas[idx] -= h[j-k] * dx
    
    c_base = c - t_cur * deltas
    
    # Minimize max(c_base + deltas * t) over t in [0,1]
    # Binary search on t
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        val = np.max(c_base + deltas * mid)
        # Check derivative sign: compare val at mid-eps and mid+eps
        eps = 1e-8
        val_minus = np.max(c_base + deltas * (mid - eps))
        val_plus = np.max(c_base + deltas * (mid + eps))
        if val_minus > val_plus:
            # Function is decreasing at mid, search right
            lo = mid
        else:
            hi = mid
    t_opt = 0.5 * (lo + hi)
    
    return t_opt
```

```python
                if 0 <= j-k < n:
                    d[idx] -= h[j-k] * dx
            
            c_base = c - tcur * d
            # Minimize max(c_base + d*t) over t in [0,1]
            # Binary search
            lo_t, hi_t = 0.0, 1.0
            for _ in range(30):

                # Derivative sign: compare subgradients
                # f(t) = max_i (c_base[i] + d[i]*t)
                # f'(t) = d[argmax_i (c_base[i] + d[i]*t)]
                am = np.argmax(c_base + d*mid)
                slope = d[am]
                if slope > 0:
                    lo_t = mid
                elif slope < 0:
                    hi_t = mid
                else:
                    break
            t_new = 0.5*(lo_t+hi_t)
            # Update
            c += d*(t_new - tcur)
            h[j] = t_new
        
        # Project
        h = project_box_sum(h, n)
        c = compute_c(h, n)
        if c.max() < bestc:
            bestc = float(c.max())
            besth = h.copy()
    
    return besth, bestc
```

2. Find i2 = argmax(a_i + b_i * 1) = argmax(a_i + b_i) (value at t=1)
3. If i1 == i2, the minimum is at the boundary (t=0 or t=1, whichever gives lower value)
4. Otherwise, the crossing point is t* = (a_i2 - a_i1) / (b_i1 - b_i2)
5. Clamp t* to [0,1]
6. But this might not be the global minimum if there are other lines above both at t*. So we need to verify and potentially iterate.

Actually, the safe approach is just binary search on the slope (which is O(log(1/eps)) iterations, each O(n)). For 30 iterations and n=400, that's 12000 operations per coordinate, 4.8M per sweep. Fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1)).copy()
    tgt = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(60):
        d = tgt - v.sum()
        if abs(d) < 1e-15:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = m.sum()
        if c > 0:
            v[m] += d / c
        else:
            v[:int(n // 2)] += d / float(n // 2)
        v = np.clip(v, 0.0, 1.0)
    return v

def grad_c_k(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        g[:n - k] += dx * (1.0 - h[k:])
        g[k:] -= dx * h[:n - k]
    else:
        k2 = -k
        g[k2:] += dx * (1.0 - h[:n - k2])
        g[:n - k2] -= dx * h[k2:]
    return g

def lse_obj_grad(h, n, alpha):
    c = compute_c(h, n)
    m = float(c.max())
    sh = alpha * (c - m)
    sh -= sh.max()
    e = np.exp(sh)
    q = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    off = n - 1
    grad = np.zeros(n)
    active = np.where(q > 1e-8)[0]
    for idx in active:
        grad += q[idx] * grad_c_k(h, n, idx - off)
    return val, grad

def lbfgsb_opt(h, n, alpha, mu, maxiter, deadline):
    h = project_box_sum(h, n)

    def obj(x):
        d = x.sum() - 0.5 * n
        c = compute_c(x, n)
        m = c.max()
        sh = alpha * (c - m)
        sh -= sh.max()
        return m + np.log(np.exp(sh).sum()) / alpha + 0.5 * mu * d * d

    def grad(x):
        _, gl = lse_obj_grad(x, n, alpha)
        d = x.sum() - 0.5 * n
        return gl + mu * d

    try:
        res = minimize(obj, h, jac=grad, method="L-BFGS-B",
                       bounds=[(0.0, 1.0)] * n,
                       options={"maxiter": maxiter, "maxfun": 3 * maxiter,
                                "ftol": 1e-16, "gtol": 1e-10})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h

def coord_descent(h, n, sweeps, deadline, rng=None):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    dx = 2.0 / n
    off = n - 1
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for j in order:
            tcur = h[j]
            d = np.zeros(2 * n - 1)
            k_lo = max(-off, -j)
            k_hi = min(off, n - 1 - j)
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi + 1)
                d[off + ks] += (1.0 - h[j + ks]) * dx
            k_lo2 = max(-off, j - n + 1)
            k_hi2 = min(off, j)
            if k_lo2 <= k_hi2:
                ks2 = np.arange(k_lo2, k_hi2 + 1)
                d[off + ks2] -= h[j - ks2] * dx
            c_base = c - tcur * d
            lo_t, hi_t = 0.0, 1.0
            for _ in range(28):
                mid = 0.5 * (lo_t + hi_t)
                am = np.argmax(c_base + d * mid)
                sl = d[am]
                if sl > 1e-14:
                    lo_t = mid
                elif sl < -1e-14:
                    hi_t = mid
                else:
                    break
            t_new = 0.5 * (lo_t + hi_t)
            c += d * (t_new - tcur)
            h[j] = t_new
        h = project_box_sum(h, n)
        c = compute_c(h, n)
        if c.max() < bestc:
            bestc = float(c.max())
            besth = h.copy()
    return besth, bestc

def pair_search(h, c_base, i, j, n):
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0
    lo, hi = max(0.0, S - 1.0), min(1.0, S)
    if hi - lo < 1e-13:
        return None
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhr = omh[::-1]
    yr = S - x0

    def eval_at(xs):
        L = 2 * n - 1
        delta = np.zeros((len(xs), L))
        a1, a2 = xs - x0, x0 - xs
        yv = S - xs
        delta[:, i:i + n] += a1[:, None] * omhr[None, :]
        si = n - 1 - i
        delta[:, si:si + n] += a2[:, None] * h[None, :]
        delta[:, j:j + n] += a2[:, None] * omhr[None, :]
        sj = n - 1 - j
        delta[:, sj:sj + n] += a1[:, None] * h[None, :]
        b0 = a1 * (1.0 - x0) + x0 * a2 + a2 * (1.0 - y0) + y0 * a1
        e0 = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
        delta[:, off] += e0 - b0
        dd = j - i
        bp = a1 * (1.0 - y0) + x0 * a1
        ep = xs * (1.0 - yv) - x0 * (1.0 - y0)
        delta[:, off - dd] += ep - bp
        bm = a2 * (1.0 - x0) + y0 * a2
        em = yv * (1.0 - xs) - y0 * (1.0 - x0)
        delta[:, off + dd] += em - bm
        return np.max(c_base[None, :] + dx * delta, axis=1)

    xs = np.unique(np.concatenate([
        np.array([lo, hi, x0]),
        np.linspace(lo, hi, 20),
        np.linspace(max(lo, x0 - 0.05), min(hi, x0 + 0.05), 80)
    ]))
    xs = np.clip(xs, lo, hi)
    vals = eval_at(xs)
    for _ in range(3):
        if len(xs) <= 1:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k - 1], xs[k + 1]
        if b - a < 1e-12:
            break
        xs = np.linspace(a, b, 60)
        vals = eval_at(xs)
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c_base.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(np.clip(S - xb, 0.0, 1.0))
        return xb, yb, bv
    return None

def anneal_pair_search(h, c_base, n, deadline, seed, reheat_iter=600, noise_scale=0.02):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 2000
    while time.time() < deadline and noimp < maxno:
        if rng.random() < 0.8:
            pos = c[n:]
            neg = c[:n - 1][::-1]
            w = np.maximum(pos, neg) + 1e-12
            w /= w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search(h, c, i, j, n)
        if res is not None:
            x, y, _ = res
            h[i], h[j] = x, y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc, besth = v, h.copy()
            noimp = 0
        else:
            noimp += 1
            if noimp % reheat_iter == 0:
                h = project_box_sum(h + noise_scale * rng.randn(n), n)
                c = compute_c(h, n)
                noise_scale *= 0.8
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except NameError:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.49 * np.sin(3.0 * np.pi * x)
    hp = project_box_sum(hp, n0)
    besth, bestc, bestn = hp.copy(), float(compute_c(hp, n0).max()), n0

    def try_res(nh, t_start, t_end):
        nonlocal besth, bestc, bestn
        if time.time() > t_end - 10:
            return
        hi = interpolate_h(besth, bestn, nh) if nh != bestn else besth.copy()
        hi = project_box_sum(hi + 0.003 * rng.randn(nh), nh)
        # L-BFGS-B phases
        for al, mu, mi in [(20, 50, 200), (100, 500, 200), (500, 5000, 150)]:
            if time.time() > t_end - 30:
                break
            h2 = lbfgsb_opt(hi, nh, al, mu, mi, t_end)
            v2 = float(compute_c(h2, nh).max())
            if v2 < bestc:
                bestc, besth, bestn = v2, h2.copy(), nh
            if v2 <= float(compute_c(hi, nh).max()) + 1e-6:
                hi = h2
            else:
                hi = besth.copy() if bestn == nh else interpolate_h(besth, bestn, nh)
        # Coordinate descent
        if time.time() < t_end - 25:
            h3, v3 = coord_descent(hi, nh, 60, t_end - 20, rng)
            if v3 < bestc:
                bestc, besth, bestn = v3, h3.copy(), nh
            hi = h3
        # Pair search with annealing
        if time.time() < t_end - 15:
            h4, v4 = anneal_pair_search(hi, None, nh, t_end - 5, seed + nh, 500, 0.015)
            if v4 < bestc:
                bestc, besth, bestn = v4, h4.copy(), nh

    # Main resolution
    try_res(n0, t0, t0 + (deadline - t0) * 0.55)
    # Secondary resolutions
    for nh in [int(n0 * 0.8), int(n0 * 1.25), int(n0 * 1.5)]:
        if nh < 50 or nh > 800:
            continue
        rem = deadline - time.time()
        if rem < 40:
            break
        frac = 0.25 if nh != n0 else 0.4
        try_res(nh, time.time(), min(deadline - 30, time.time() + rem * frac))

    # Final polish at best resolution
    if time.time() < deadline - 20:
        hfin = besth.copy()
        for al, mi in [(500, 100), (2000, 80)]:
            if time.time() > deadline - 15:
                break
            h2 = lbfgsb_opt(hfin, bestn, al, 10000, mi, deadline)
            v2 = float(compute_c(h2, bestn).max())
            if v2 < bestc:
                bestc, besth, bestn = v2, h2.copy(), bestn
            hfin = besth.copy()
        hfin2 = coord_descent(besth, bestn, 40, deadline - 10, rng)
        if hfin2[1] < bestc:
            bestc, besth = hfin2[1], hfin2[0]
        if time.time() < deadline - 8:
            h3, v3 = anneal_pair_search(besth, None, bestn, deadline - 2, seed + 777, 400, 0.008)
            if v3 < bestc:
                bestc, besth = v3, h3.copy()

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
```
