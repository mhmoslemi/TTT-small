import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones_like(x)

def slsqp_run(h, n, alphas, maxiter_each=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    no_improve = 0
    for sweep in range(max_passes):
        if deadline is not None and time.time() > deadline:

        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        indices = range(n) if sweep % 2 == 0 else range(n-1, -1, -1)
        for p in indices:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            k_lo = max(p - n + 1, -p)
            k_hi = min(p, n - 1 - p)
            m_lo = k_lo + (n - 1)
            m_hi = k_hi + (n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            ks = ms - (n - 1)
            a = np.zeros(len(ms))
            for i, k in enumerate(ks):
                if k == 0:
                    a[i] = 1.0 - 2.0 * h0
                else:
                    coeff = 0.0
                    if 0 <= p - k < n:
                        coeff += 1.0 - h[p - k]
                    if 0 <= p + k < n:
                        coeff -= h[p + k]
                    a[i] = coeff
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff = max(unaff, float(c[m_hi+1:].max()))
            diag_mask = (ks == 0)
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            for d in d_grid:

                nv = c_aff + d * a
                if np.any(diag_mask):
                    nv[diag_mask] -= d * d
                val = max(unaff, float(nv.max()))
                if val < best_val:

                    best_d = d
            if abs(best_d) > 1e-10:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones_like(x)

def slsqp_run(h, n, alphas, maxiter_each=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=3000, grid_size=200, deadline=None, quiet=True):
    h = h.copy()
    no_improve = 0
    for sweep in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        indices = range(n) if sweep % 2 == 0 else range(n - 1, -1, -1)
        for p in indices:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            k_lo = max(p - n + 1, -p)
            k_hi = min(p, n - 1 - p)
            m_lo = k_lo + (n - 1)
            m_hi = k_hi + (n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            ks = ms - (n - 1)
            M = len(ms)
            a = np.zeros(M)
            diag_mask = (ks == 0)
            if np.any(diag_mask):
                a[diag_mask] = 1.0 - 2.0 * h0
            off = ~diag_mask
            if np.any(off):
                ko = ks[off]
                va = (p - ko >= 0) & (p - ko < n)
                vb = (p + ko >= 0) & (p + ko < n)
                if np.any(va):
                    a[off][va] += 1.0 - h[p - ko[va]]
                if np.any(vb):
                    a[off][vb] -= h[p + ko[vb]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            nv_all = c_aff[np.newaxis, :] + d_grid[:, np.newaxis] * a[np.newaxis, :]
            if np.any(diag_mask):
                nv_all[:, diag_mask] -= (d_grid[:, np.newaxis]) ** 2
            vals = np.maximum(unaff, nv_all.max(axis=1))
            bi = int(vals.argmin())
            if vals[bi] < current_max - 1e-11:
                h[p] = float(np.clip(h0 + d_grid[bi], 0.0, 1.0))
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 150:
                break
        else:
            no_improve = 0
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def perturb_quantize(h, n, beta, rng):
    center = 0.5
    shifted = h - center
    hq = center + np.tanh(beta * shifted) / np.tanh(beta / 2.0) * center
    hq = np.clip(hq, 0.0, 1.0)
    hq = hq + rng.normal(0, 0.01, n)
    return project_feasible(hq, n)

def perturb_sinusoidal(h, n, freq, amp, rng):
    x = np.linspace(0, 2, n)
    phase = rng.uniform(0, 2 * np.pi)
    h2 = h + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return project_feasible(h2, n)
