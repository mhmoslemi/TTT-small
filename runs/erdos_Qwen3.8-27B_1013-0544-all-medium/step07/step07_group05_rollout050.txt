import time
import numpy as np
import scipy.optimize as so

_CACHE = {'x': None, 'g': None}
_LD_CACHE = {'x': None, 'g': None}
_LB = {'n': 1, 'dx': 1.0, 'target': 0.5, 'beta': 1.0, 'rho': 1e6}
_LD = {'n': 1, 'dx': 1.0, 'target': 0.5, 'k_star': 0, 'rho': 1e6}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w


def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def _lbfgs_obj(x):
    n = int(_LB['n']); dx = float(_LB['dx']); beta = float(_LB['beta'])
    rho = float(_LB['rho']); target = float(_LB['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, beta)
    g = smooth_grad_from_w(x, w, n, dx)
    res = float(x.sum() - target)
    f += 0.5 * rho * res * res
    g = g + rho * res
    _CACHE['x'] = x.copy()
    _CACHE['g'] = g
    return float(f)


def _lbfgs_jac(x):
    ck = _CACHE['x']
    if ck is not None and np.array_equal(x, ck):
        return _CACHE['g'].copy()
    n = int(_LB['n']); dx = float(_LB['dx']); beta = float(_LB['beta'])
    rho = float(_LB['rho']); target = float(_LB['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, beta)
    g = smooth_grad_from_w(x, w, n, dx)
    res = float(x.sum() - target)
    g = g + rho * res
    _CACHE['x'] = x.copy()
    _CACHE['g'] = g
    return g


def lbfgs_refine(h, n, beta, maxiter, deadline):

    target = 0.5 * n
    rem = deadline - time.time()

    _LB['n'] = n; _LB['dx'] = 2.0 / n; _LB['beta'] = float(beta)
    _LB['rho'] = 1e6; _LB['target'] = target
    _CACHE['x'] = None; _CACHE['g'] = None
    mx = int(min(maxiter, max(10, int(rem / 2.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(_lbfgs_obj, h, method='L-BFGS-B', jac=_lbfgs_jac,
                          bounds=bounds, options={'maxiter': mx, 'ftol': 1e-18,
                                                  'gtol': 1e-14, 'maxls': 40})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def _ld_obj(x):
    n = int(_LD['n']); dx = float(_LD['dx']); ks = int(_LD['k_star'])
    rho = float(_LD['rho']); target = float(_LD['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx

def _pair_obj(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, n)


def _pair_line(h, n, i, j, cur):
    hi0 = h[i]; hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d = 0.0; best_c = cur
    c0 = _pair_obj(lo, h, i, j, n)
    if c0 < best_c:
        best_c = c0; best_d = lo
    c1 = _pair_obj(hi, h, i, j, n)
    if c1 < best_c:
        best_c = c1; best_d = hi
    try:
        rs = so.minimize_scalar(_pair_obj, args=(h, i, j, n),
                                bounds=(lo, hi), method='bounded',
                                options={'maxiter': 30, 'xatol': 1e-10})
        if rs.x is not None:
            c2 = _pair_obj(float(rs.x), h, i, j, n)
            if c2 < best_c:
                best_c = c2; best_d = float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d; hh[j] -= best_d
        hh = np.clip(hh, 0.0, 1.0)
        return hh, best_c
    return h, cur


def pair_polish(h, n, deadline, seed):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    m = 35
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= m:
        top = order[:m]
    else:
        need = m - int(order.size)
        if need > n - int(order.size):
            need = max(0, n - int(order.size))
        if need > 0:
            extra = rng.choice(n, size=need, replace=False)
            top = np.unique(np.concatenate((order, extra)))
        else:
            top = order
    if top.size < 2:
        return h, cur
    for _ in range(5):
        improved = False
        pairs = []
        tlen = len(top)
        for a in range(tlen):
            for b in range(a + 1, tlen):

            if cc < cur - 1e-14:

        if not improved:

    return h, cur
