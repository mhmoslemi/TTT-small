```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    """Compute gradient of smooth max of correlations."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    w = exp_c / exp_c.sum()
    
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    
    # For each k, w_k * (1 - h[i+k] - h[i-k])
    # Use the structure: grad = dx * (w_conv_1 - w_conv_h_shifted - w_conv_h_anti_shifted)
    # Simpler: loop over k
    for ki in range(n_corr):
        k = ki - (n - 1)
        wk = w[ki]
        if wk < 1e-12:
            continue
        if k >= 0:
            i1 = np.arange(0, n - k)  # valid i for h[i+k]
            i2 = np.arange(k, n)       # valid i for h[i-k]
        else:
            i1 = np.arange(0, n + k)   # valid i for h[i+k]
            i2 = np.arange(-k, n)      # valid i for h[i-k]
        if len(i1) > 0:
            grad[i1] += wk * (1.0 - h[i1 + k])
        if len(i2) > 0:
            grad[i2] -= wk * h[i2 - k]
    
    return grad * dx

def projected_gradient_step(h, n, alpha, lr):
    """One step of projected gradient descent on smooth objective."""
    g = smooth_grad(h, n, alpha)
    # Projected gradient: move in -grad direction, project to [0,1] and sum constraint
    h_new = h - lr * g
    h_new = project_feasible(h_new, n)
    return h_new

def lp_step(h, n, eps, solver='PDLP'):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    
    # Compute gradient matrix
    n_corr = 2 * n - 1
    G = np.zeros((n, n_corr))
    for m in range(n_corr):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'ECOS', 'CLARABEL', 'PDLP']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.5, 0.3, 0.1, 0.05]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc
```

```python
                G[:n-kp, m] -= h[kp:]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['CLARABEL', 'PDLP', 'GLOP', 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    best_mc = best_c5 * n / 2.0

    # Phase 1: Smooth gradient (40% time)
    p1_dl = t0 + 0.40 * total
    alphas = [10, 30, 100, 300, 1000, 3000, 10000]
    top_ks = [None, None, None, 60, 30, 15, 8]
    for ai, alpha in enumerate(alphas):
        if time.time() > p1_dl:
            break
        tk = top_ks[ai]
        h_cur = h.copy()
        for _ in range(100):
            if time.time() > p1_dl:
                break
            h_cur, c5_cur = gradient_phase(h_cur, n, alpha, max_iters=15, top_k=tk)
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_mc = c5_cur * n / 2.0
                h = h_cur.copy()

    # Phase 2: LP optimization (35% time)
    p2_dl = t0 + 0.75 * total
    h = best_h.copy()
    eps = 0.05
    no_improve = 0
    while time.time() < p2_dl:
        h, mc = lp_step(h, n, eps)
        c5 = mc * 2.0 / n
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            best_mc = mc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 12:
                eps *= 0.65
                if eps < 0.002:
                    break
                no_improve = 0

    # Phase 3: Restarts (15% time)
    p3_dl = dl - 25
    for trial in range(30):
        if time.time() > p3_dl:
            break
        ns = 0.008 + 0.015 * rng.rand()
        h_try = project_feasible(best_h + ns * rng.randn(n), n)
        for alpha in [200, 1000]:
            if time.time() > p3_dl:
                break
            h_try, _ = gradient_phase(h_try, n, alpha, max_iters=20, top_k=20)
        c5_try = compute_c5(h_try, n)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_mc = c5_try * n / 2.0
        for _ in range(40):
            if time.time() > p3_dl:
                break
            h_try, mc = lp_step(h_try, n, 0.025)
            c5_try = mc * 2.0 / n
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_mc = mc

    # Phase 4: Final LP polish
    h = best_h.copy()
    eps = 0.005
    while time.time() < dl - 2:
        h, mc = lp_step(h, n, eps)
        c5 = mc * 2.0 / n
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        eps = max(eps * 0.97, 0.0004)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
