```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(corr):
    return float(np.max(corr))

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def weighted_obj(h, w, n):
    """Compute sum_k w[k] * c_k(h)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return float(np.dot(w, corr) * dx), corr * dx

def weighted_grad(h, w, n):
    """Gradient of sum_k w[k] * c_k(h) with respect to h"""
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for m in range(ncorr):
        wm = w[m]
        if abs(wm) < 1e-18:
            continue
        k = m - (n - 1)
        j = np.arange(n)
        idx1 = j + k
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += wm * (1.0 - h[idx1[mask1]])
        idx2 = j - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] -= wm * h[idx2[mask2]]
    return dx * grad

def lse_slsqp(h, n, alpha, maxiter, ftol, active=None):
    """LSE approximation of max, optionally restricted to active lags"""
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        if active is not None:
            corr = corr[active]
        shifted = alpha * (corr - corr.max())
        sm = shifted.max()
        return corr.max() + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def obj_grad(x):
        # Gradient of LSE
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        if active is not None:
            # Compute gradient only for active lags
            weights = np.zeros(ncorr)
            weights[active] = 1.0
            # LSE gradient: sum of softmax weights * gradients
            corr_a = corr[active]
            shifted = alpha * (corr_a - corr_a.max())
            sm = shifted.max()
            probs = np.exp(shifted - sm) / np.exp(shifted - sm).sum()
            w = np.zeros(ncorr)
            w[active] = probs
            return weighted_grad(x, w, n)
        else:
            shifted = alpha * (corr - corr.max())
            sm = shifted.max()
            probs = np.exp(shifted - sm) / np.exp(shifted - sm).sum()
            return weighted_grad(x, probs, n)
    
    def cons(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(res.x, n)
    except:
        return h

def exp_weights(h, n, deadline, eta=0.5, inner_iters=30):

            best_h = h_cur.copy()

        w = w / w.sum()

            if time.time() > deadline - 5:

            grad = weighted_grad(h_cur, w, n)

            f_cur = float(np.dot(w, corr))

                f_try = float(np.dot(w, compute_corr(h_try, n)))

def subgrad_momentum(h, n, deadline, lr=0.001, momentum=0.9, n_iters=2000):

    v = np.zeros(n)
    best_h = h.copy()

        if c_max < best_c:
            best_c = c_max
            best_h = h_cur.copy()
        
        # Pick a random argmax (or all argmaxes averaged)
        amax = np.where(corr >= c_max - 1e-10)[0]
        k_star = amax[np.random.randint(len(amax))]
        k = k_star - (n - 1)

        grad *= dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(corr):
    return float(np.max(corr))

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def weighted_grad(h, w, n):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for m in range(ncorr):
        wm = w[m]
        if abs(wm) < 1e-18:
            continue
        k = m - (n - 1)
        idx1 = j + k
        msk1 = (idx1 >= 0) & (idx1 < n)
        grad[msk1] += wm * (1.0 - h[idx1[msk1]])
        idx2 = j - k
        msk2 = (idx2 >= 0) & (idx2 < n)
        grad[msk2] -= wm * h[idx2[msk2]]
    return dx * grad

def lse_slsqp(h, n, alpha, maxiter, ftol, active=None):
    dx = 2.0 / n
    ncorr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        if active is not None:
            cl = corr[active]
        else:
            cl = corr
        cm = cl.max()
        sh = alpha * (cl - cm)
        sm = sh.max()
        return cm + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        if active is not None:
            cl = corr[active]
        else:
            cl = corr
        sh = alpha * (cl - cl.max())
        sm = sh.max()
        probs = np.exp(sh - sm)
        probs /= probs.sum()
        w = np.zeros(ncorr)
        if active is not None:
            w[active] = probs
        else:
            w[:] = probs
        return weighted_grad(x, w, n)

    def cons(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(res.x, n)
    except Exception:
        return h

def exp_weights(h, n, deadline, eta=0.3, inner_iters=15):
    ncorr = 2 * n - 1
    w = np.ones(ncorr) / ncorr
    h_cur = h.copy()
    best_h = h.copy()
    corr = compute_corr(h_cur, n)
    best_c = c5_val(corr)

    for it in range(20000):
        if time.time() > deadline - 4:
            break
        corr = compute_corr(h_cur, n)
        c_max = corr.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_cur.copy()

        shifted = eta * (corr - c_max)
        w = w * np.exp(shifted)
        w = w / w.sum()

        for _ in range(inner_iters):
            if time.time() > deadline - 4:
                break
            grad = weighted_grad(h_cur, w, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = 0.5 / (gnorm * 0.5 + 0.01)
            step = min(step, 1.0)
            h_try = project(h_cur - step * grad, n)
            f_new = float(np.dot(w, compute_corr(h_try, n)))
            f_old = float(np.dot(w, corr))
            if f_new > f_old:
                step *= 0.25
                h_try = project(h_cur - step * grad, n)
            h_cur = h_try
            corr = compute_corr(h_cur, n)

        c_max = corr.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_cur.copy()

    return best_h, best_c

def subgrad_step(h, n, it, lr0, momentum, rng):

    ncorr = 2 * n - 1
    corr = compute_corr(h, n)
    c_max = corr.max()
    amax = np.where(corr >= c_max - 1e-10)[0]
    k_star = amax[rng.randint(len(amax))]
    k = k_star - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)

    m1 = (idx1 >= 0) & (idx1 < n)
    grad[m1] += (1.0 - h[idx1[m1]])
    idx2 = j - k
    m2 = (idx2 >= 0) & (idx2 < n)
    grad[m2] -= h[idx2[m2]]
    grad *= dx
    lr = lr0 / (1.0 + 0.01 * it)
    return grad, lr, c_max

def binary_search(h, n, deadline, max_sweeps=40):

                jmn = j - m_range

    p5 = t0 + 0.85 * total
    x = np.linspace(0, 2, n)
    perts = [0.005 * rng.randn(n), 0.01 * np.sin(2 * np.pi * 3.0 * x),
             0.008 * np.cos(2 * np.pi * 5.0 * x + 1.0), 0.012 * rng.randn(n)]
    for pert in perts:
        if time.time() > p5 - 6:
            break
        hp = project(best_h + pert, n)
        hp = lse_slsqp(hp, n, 200000, 1500, 1e-18)
        cp = c5_val(compute_corr(hp, n))
        if cp < best_c:
            best_c, best_h = cp, hp.copy()

    # Phase 6: Binary + final polish (15% time)
    p6 = t0 + 0.95 * total
    h6, c6 = binary_search(best_h, n, p6, max_sweeps=35)
    if c6 < best_c:
        best_c, best_h = c6, h6.copy()

    # Final high-alpha polish
    if time.time() < deadline - 4:
        hf = lse_slsqp(best_h, n, 100000000, 4000, 1e-18)
        cf = c5_val(compute_corr(hf, n))
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    if time.time() < deadline - 2:
        hf = lse_slsqp(best_h, n, 200000000, 3000, 1e-18)
        cf = c5_val(compute_corr(hf, n))
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    best_h = project(best_h, n)
    best_c = c5_val(compute_corr(best_h, n))
    return best_h, best_c, n
```
