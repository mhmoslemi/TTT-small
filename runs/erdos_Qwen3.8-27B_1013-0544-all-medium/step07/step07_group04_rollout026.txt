    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0

        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:

        else:

    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

        v = np.clip(v, 0.0, 1.0)

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return m
    return m + np.log(s) / alpha

def lse_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    return q

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            s = k
            m = n - s
            g[s:n] += w * (1.0 - x[:m])
            g[:m] -= w * x[s:n]
        elif k < 0:
            s = -k
            m = n - s
            g[:m] += w * (1.0 - x[s:n])
            g[s:n] -= w * x[:m]
        else:
            g += w * (1.0 - 2.0 * x)

= δ * [(1-h[i+d])*[i<n-d] - h[i-d]*[i>=d] - (1-h[j+d])*[j<n-d] + h[j-d]*[j>=d]]

Similarly for c[off+d] = sum_{t=d}^{n-1} h[t]*(1-h[t-d]) = sum_{s=0}^{n-d-1} h[s+d]*(1-h[s]):
- h[p] appears as LEFT factor when s+d = p, i.e., s = p-d (if p >= d and s < n-d, i.e., p < n)
- h[p] appears as RIGHT factor when s = p (if p < n-d)

Change in c[off+d] from h[p] → h[p]+δ_p:
= δ_p * (1-h[p-d]) * [p >= d]  -  δ_p * h[p+d] * [p < n-d]

For our pair move:
Change in c[off+d] = δ * [(1-h[i-d])*[i>=d] - h[i+d]*[i<n-d] - (1-h[j-d])*[j>=d] + h[j+d]*[j<n-d]]

And for c[off] (d=0): c[off] = sum_t h[t]*(1-h[t])
Change = δ*(1-2*h[i]) + (-δ)*(1-2*h[j]) = δ*(2*h[j] - 2*h[i])

OK so this gives me an efficient way to compute the change in all c[k] values from a pair move. Let me define for each coordinate p and each shift d, the "influence" coefficient:

For c[off-d] (d > 0): coeff[p, d] = (1-h[p+d])*[p < n-d] - h[p-d]*[p >= d]
For c[off+d] (d > 0): coeff2[p, d] = (1-h[p-d])*[p >= d] - h[p+d]*[p < n-d]
For c[off]: coeff0[p] = (1-2*h[p])

Then for a pair move (h[i] += δ, h[j] -= δ):
Δc[off-d] = δ * (coeff[i,d] - coeff[j,d]) for d > 0
Δc[off+d] = δ * (coeff2[i,d] - coeff2[j,d]) for d > 0
Δc[off] = δ * (coeff0[i] - coeff0[j])

So the new c values are: c_new[k] = c[k] + δ * Δ[k] where Δ is computed as above.

This means for a given pair (i,j), the function x → max_k (c[k] + (x-h[i]) * Δ[k]) is the max of linear functions in x, which is convex. We can minimize it by finding the point where the subgradient contains 0, i.e., by checking all the "breakpoints" (where two linear functions cross).

The breakpoints occur where c[k] + (x-h[i])*Δ[k] = c[l] + (x-h[i])*Δ[l] for some k, l.
This gives: x = (c[l] - c[k]) / (Δ[k] - Δ[l]) (if Δ[k] ≠ Δ[l]).

So the minimum of the max is at one of these breakpoints (or at the boundary).

For n=400, there are 2n-1 = 799 correlation values. The number of breakpoints is O(n^2) in the worst case, but in practice we can just do a ternary search or golden section search since the function is convex.

Actually, for a convex piecewise linear function in 1D, we can find the minimum by binary search on the subgradient. The subgradient at x is the set of slopes of the linear pieces that are active at x. The minimum is where 0 is in the subgradient.

Simpler approach: since we have 799 linear functions, the max is determined by at most a few of them at any point. We can:
1. Start at x = h[i] (current value)
2. Find which c[k] + (x-h[i])*Δ[k] is the maximum
3. If Δ[k] > 0 for the active k, we should decrease x
4. If Δ[k] < 0 for the active k, we should increase x
5. Move to the nearest breakpoint and repeat

This is essentially the "slope trick" or "convex function minimization" algorithm.

We want this to be negative, so we want g_proj[i] < g_proj[j].

The best pair is (i = argmin g_proj, j = argmax g_proj).

If the feasible range is very small (h[i] is near 0 or 1, or h[j] is near 0 or 1), the pair is less useful.

    g = lse_grad(h, n, alpha)
    gp = g - np.mean(g)
    # Score for each coordinate: how much it contributes to the objective
    # We want to pair a low gp with a high gp
    # But also weight by "feasibility" - how much room to move
    room = np.minimum(h, 1.0 - h)  # max step in either direction
    score = room * np.abs(gp)
    # Pick i from high negative gp (low gp) and j from high positive gp
    neg_scores = np.where(gp < 0, -gp * room, 0)
    pos_scores = np.where(gp > 0, gp * room, 0)
    if neg_scores.sum() > 0:
        neg_scores = neg_scores / neg_scores.sum()
    else:
        neg_scores = np.ones(n) / n
    if pos_scores.sum() > 0:

    else:
        pos_scores = np.ones(n) / n
    i = int(rng.choice(n, p=neg_scores))
    j = int(rng.choice(n, p=pos_scores))
    if i == j:
        j = (j + 1) % n
    return i, j

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):

    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]

    ax2 = x0 - xs

    delta[:, j:j + n] += ax2[:, None] * omhrev[None, :]

    d = j - i

def interpolate_h(h, nold, nnew):

    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.49 * np.sin(3.0 * np.pi * x)
        hp = project_box_sum(hp, n0)
    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = float(compute_c(hp, n0).max())
    bestn = n0
    # Phase 1: SLSQP
    for alpha, miter in ((50.0, 400), (200.0, 350), (800.0, 300)):
        if time.time() > deadline * 0.15:
            break
        h2 = slsqp_phase(besth, bestn, alpha, miter)
        h2 = project_box_sum(h2, bestn)
        v2 = float(compute_c(h2, bestn).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
    # Phase 2: Pair search (main workhorse)
    dl2 = t0 + (deadline - t0) * 0.55
    if time.time() < dl2:
        besth, bestc = pair_local_search(besth, dl2, seed + 1, alpha=500.0)
        bestn = len(besth)
    # Phase 3: SLSQP polish
    if time.time() < deadline * 0.7:
        for alpha, miter in ((2000.0, 200), (8000.0, 150)):
            if time.time() > deadline * 0.7:
                break
            h2 = slsqp_phase(besth, bestn, alpha, miter)
            h2 = project_box_sum(h2, bestn)
            v2 = float(compute_c(h2, bestn).max())
            if v2 < bestc:
                bestc = v2
                besth = h2.copy()
    # Phase 4: More pair search
    dl4 = deadline - 5
    if time.time() < dl4:
        besth, bestc = pair_local_search(besth, dl4, seed + 100, alpha=2000.0)
        bestn = len(besth)
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
