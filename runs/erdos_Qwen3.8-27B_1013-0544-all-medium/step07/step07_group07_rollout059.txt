import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-13: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_val_grad(h, alpha):
    n = len(h)
    c = compute_corr(h)
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    s = exps.sum()
    val = m + (sm + np.log(s)) / alpha
    w = exps / s
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wi = w[lag + n - 1]
        if wi < 1e-15: continue
        if lag >= 0:
            lo, hi = 0, n - lag
        else:
            lo, hi = -lag, n
        idx = np.arange(lo, hi)
        grad[idx] += wi * (1.0 - h[idx + lag])
        jlo = lo + lag
        jhi = hi + lag
        if jlo < 0: jlo = 0
        if jhi > n: jhi = n
        if jlo < jhi:
            idx2 = np.arange(jlo, jhi)
            grad[idx2] -= wi * h[idx2 - lag]
    return val, grad

def slsqp_opt(h, n, alpha, maxiter, ftol):
    def f(x):
        return lse_val_grad(x, alpha)
    def cons(x):
        return x.sum() - n / 2.0

    try:

                     constraints=[{'type': 'eq', 'fun': cons}],
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(r.x, n)
    except Exception:
        return h

Let h[n-1] = n/2 - sum(h[0:n-1]).
The bound 0 ≤ h[n-1] ≤ 1 becomes: n/2 - 1 ≤ sum(h[0:n-1]) ≤ n/2.

This is a linear constraint, not a box constraint. L-BFGS-B can't handle it directly.

- Let u[i] be free parameters.
