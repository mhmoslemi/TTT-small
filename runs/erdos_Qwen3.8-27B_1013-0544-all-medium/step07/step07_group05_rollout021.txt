def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def erdos_c5_float(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def resample_to(h, n):
    h = np.asarray(h, dtype=float)
    old = len(h)
    n = int(n)
    if old == n:
        return h.copy()
    x = np.linspace(0.0, 2.0, old)
    y = np.linspace(0.0, 2.0, n)
    hh = np.interp(y, x, h)
    return project_box_sum(hh, 0.5 * n)


def fix_binary_count(b, k, rng):
    b = np.asarray(b, dtype=np.int8).copy()
    k = int(k)
    for _ in range(2000):
        s = int(b.sum())
        if s == k:
            return b
        ones = np.flatnonzero(b)
        zeros = np.flatnonzero(1 - b)
        if s > k:
            if ones.size == 0 or zeros.size == 0:
                break
            b[int(rng.choice(ones))] = 0
            b[int(rng.choice(zeros))] = 1
        else:
            if ones.size == 0 or zeros.size == 0:
                break
            b[int(rng.choice(zeros))] = 1
            b[int(rng.choice(ones))] = 0
    return b


def topk_binary(h, noise, seed):
    h = np.asarray(h, dtype=float)
    n = len(h)
    k = n // 2
    rng = np.random.RandomState(int(seed))
    vals = h.copy()
    if noise > 0.0:
        vals += float(noise) * rng.randn(n)
    b = np.zeros(n, dtype=np.int8)
    b[np.argsort(vals)[::-1][:k]] = 1
    return b


def shuffled_binary(n, seed):
    rng = np.random.RandomState(int(seed))
    b = np.zeros(int(n), dtype=np.int8)
    b[:int(n) // 2] = 1
    rng.shuffle(b)
    return b


def markov_binary(n, seed, p):
    rng = np.random.RandomState(int(seed))
    n = int(n)
    b = np.empty(n, dtype=np.int8)
    val = 1 if rng.rand() < 0.5 else 0
    b[0] = val
    pv = float(p)
    for pos in range(1, n):
        if rng.rand() >= pv:
            val ^= 1
        b[pos] = val
    return fix_binary_count(b, n // 2, rng)


def block_binary(n, seed, mean_run):
    rng = np.random.RandomState(int(seed))
    n = int(n)
    b = np.zeros(n, dtype=np.int8)
    pos = 0
    val = int(rng.randint(2))
    m = float(mean_run)
    while pos < n:
        L = max(1, int(round(m + rng.randn() * max(1.0, 0.4 * m))))
        end = min(n, pos + L)
        b[pos:end] = val
        val ^= 1
        pos = end
    return fix_binary_count(b, n // 2, rng)

def binary_best_move(b, F, n, deadline, out):
    if n < 2:
        return None
    n1 = n - 1
    m = 2 * n - 1
    zeros = np.flatnonzero(1 - b)
    ones = np.flatnonzero(b)
    if zeros.size == 0 or ones.size == 0:
        return None
    Zrev = (1 - b).astype(np.int8)[::-1]
    O = b.astype(np.int8)
    B = np.empty((n, m), dtype=np.int32)
    P = np.empty((n, m), dtype=np.int32)
    for p in range(n):
        Bp = B[p]
        Bp[:] = F
        Bp[p:p + n] += Zrev
        Bp[n1 - p:n1 - p + n] -= O
        Bp[n1] -= 1
        Pp = P[p]
        Pp[:] = 0
        Pp[p:p + n] -= Zrev
        Pp[n1 - p:n1 - p + n] += O
        Pp[n1] -= 1

    best_nm = int(F.max())
    best_i = -1
    best_j = -1
    product = int(zeros.size * ones.size)
    check = 0

    if product <= 60000:
        for i in zeros:
            Bi = B[i]
            for j in ones:
                np.add(Bi, P[j], out=out)
                out[n1 + i - j] += 1
                out[n1 + j - i] += 1
                nm = int(out.max())
                if nm < best_nm:
                    best_nm = nm
                    best_i = int(i)
                    best_j = int(j)
                check += 1
                if (check & 2047) == 0 and time.time() > deadline:
                    if best_i >= 0:
                        break
                    return None
            if best_i >= 0 and time.time() > deadline:
                break
    else:
        take = max(8, 50000 // max(1, int(zeros.size)))
        for i in zeros:
            subset = _PAIR_RNG.choice(ones, size=min(take, int(ones.size)), replace=False)
            Bi = B[i]
            for j in subset:
                np.add(Bi, P[j], out=out)
                out[n1 + i - j] += 1
                out[n1 + j - i] += 1
                nm = int(out.max())
                if nm < best_nm:
                    best_nm = nm
                    best_i = int(i)

                    return None

def binary_polish(b0, time_limit, seed):
    b = np.asarray(b0, dtype=np.int8).copy()
    n = b.size
    F = np.correlate(b, 1 - b, mode='full').astype(np.int64)
    best_b = b.copy()
    best_max = int(F.max())
    out = np.empty(2 * n - 1, dtype=np.int32)
    rng = np.random.RandomState(int(seed))
    kicks = 0
    fails = 0
    while time.time() < time_limit and best_max > 0:
        old_max = int(F.max())
        res = binary_best_move(b, F, n, time_limit, out)
        if res is None:
            if kicks < 10 and time.time() < time_limit - 10:
                b = best_b.copy()
                kk = min(5, max(1, n // 80))
                for _ in range(kk):
                    ones = np.flatnonzero(b)
                    zeros = np.flatnonzero(1 - b)
                    if ones.size == 0 or zeros.size == 0:
                        break
                    oi = int(rng.choice(ones))
                    zi = int(rng.choice(zeros))
                    b[oi] = 0
                    b[zi] = 1
                F = np.correlate(b, 1 - b, mode='full').astype(np.int64)
                kicks += 1
                continue
            break
        i, j, mv = res
        b[i] = 1
        b[j] = 0
        F = np.correlate(b, 1 - b, mode='full').astype(np.int64)
        dmax = int(F.max())
        if dmax < old_max:
            if dmax < best_max:
                best_max = dmax
                best_b = b.copy()
                kicks = 0
            fails = 0
        else:
            b[j] = 1
            b[i] = 0
            F = np.correlate(b, 1 - b, mode='full').astype(np.int64)
            fails += 1
            if fails >= 3:
                break
    return best_b, best_max


def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w


def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def l_compute(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    target = float(st['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    g = smooth_grad_from_w(x, w, n, dx)
    res = float(x.sum() - target)
    rho = float(st['rho'])
    lam = float(st['lam'])
    if rho != 0.0 or lam != 0.0:
        f += lam * res + 0.5 * rho * res * res
        g = g + (lam + rho * res)
    return float(f), g


def l_obj(x):
    cx = LS.get('cache_x')
    if cx is not None and np.array_equal(x, cx):
        return float(LS.get('cache_f', 0.0))
    f, g = l_compute(x)
    LS['cache_x'] = x.copy()
    LS['cache_g'] = g
    LS['cache_f'] = f
    return f


def l_jac(x):

    f, g = l_compute(x)

def refine_lbfgs(h, beta, rho, maxiter, deadline):

    rem = deadline - time.time()

    LS['cache_f'] = None

def continuous_refine(h0, deadline):

    h = project_box_sum(h, 0.5 * n)

    lb = ((30.0, 1e3, 35), (100.0, 1e3, 35), (300.0, 1e3, 30),
          (1000.0, 1e4, 30), (3000.0, 1e4, 25), (10000.0, 1e5, 25),
          (30000.0, 1e5, 20), (100000.0, 1e6, 20))
    for beta, rho, mx in lb:
        if time.time() > deadline - 20:

            seen.add(key)
            top.append(b)
            if len(top) >= 5:
                break

        binary_time = max(60.0, 0.35 * total)
        binary_end = t0 + binary_time
        shares = (0.40, 0.22, 0.16, 0.11, 0.11)
        for idx, b0 in enumerate(top):
            if time.time() > binary_end - 8:
                break
            share = shares[idx] if idx < len(shares) else 0.1
            limit = min(binary_end, time.time() + max(12.0, binary_time * share))
            bb, cnt = binary_polish(b0, limit, seed + 50 + idx)
            c = float(cnt) * (2.0 / float(n))
            if c < binary_best_c - 1e-15:
                binary_best_h = bb.astype(float).copy()
                binary_best_c = c
            if c < best_c - 1e-15:
                best_c = c
                best_h = bb.astype(float).copy()
                best_n = n

    cont_end = t0 + 0.80 * total
    if time.time() < cont_end - 25:
        cand, c = continuous_refine(best_h, cont_end)
        if c < best_c - 1e-15:
            best_c = c
            best_h = cand.copy()
            best_n = len(cand)

    if binary_best_h is not None and time.time() < cont_end - 35:
        if not np.allclose(best_h, binary_best_h, atol=1e-5):
            cand, c = continuous_refine(binary_best_h, cont_end)
            if c < best_c - 1e-15:
                best_c = c
                best_h = cand.copy()
                best_n = len(cand)

    multi_end = deadline - max(40.0, 0.12 * total)
    if best_n != 300 and time.time() < multi_end - 35:
        hh = project_box_sum(resample_to(best_h, 300), 150.0)
        cc = erdos_c5_float(hh, 300)
        if cc < best_c + 0.004:
            limit = min(multi_end, time.time() + max(25.0, 0.45 * max(0.0, multi_end - time.time())))
            hh, cc = continuous_refine(hh, limit)
            if cc < best_c - 1e-15:
                best_c = cc
                best_h = hh.copy()
                best_n = 300
            if time.time() < multi_end - 20:
                b3 = topk_binary(hh, 0.0, seed + 800)
                bb, cnt = binary_polish(b3, multi_end - 10, seed + 801)
                cc = float(cnt) * (2.0 / 300.0)
                if cc < best_c - 1e-15:
                    best_c = cc
                    best_h = bb.astype(float).copy()
                    best_n = 300

    if best_n <= 512 and time.time() < deadline - 25:
        bfin = topk_binary(best_h, 0.0, seed + 1200)
        bfin, cnt = binary_polish(bfin, deadline - 20, seed + 1201)
        c = float(cnt) * (2.0 / float(best_n))
        if c < best_c - 1e-15:
            best_c = c
            best_h = bfin.astype(float).copy()
            best_n = int(best_n)

    if time.time() < deadline - 12:
        cand, c = continuous_refine(best_h, deadline)
        if c < best_c - 1e-15:
            best_c = c
            best_h = cand.copy()
            best_n = len(cand)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    diff = float(0.5 * best_n - best_h.sum())
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            best_h[mask] += diff / cnt
        else:
            best_h += diff / best_h.size
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c = erdos_c5_float(best_h, best_n)
    return best_h, float(best_c), int(best_n)
