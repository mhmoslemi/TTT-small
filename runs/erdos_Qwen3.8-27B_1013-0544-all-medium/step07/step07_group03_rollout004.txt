        T = T_init * np.exp(progress * log_ratio)

        i_list = list(zeros)
        j_list = list(ones)
        i = i_list[rng.randint(len(i_list))]
        j = j_list[rng.randint(len(j_list))]

        delta = np.zeros(2*n-1)
        vs = i + s

        delta[mask] -= h[vs[mask]]
        vs = j + s
        mask = (vs >= 0) & (vs < n)
        delta[mask] -= (1.0 - h[vs[mask]])
        vs = j - s
        mask = (vs >= 0) & (vs < n)
        delta[mask] += h[vs[mask]]
        
        new_corr = corr + delta
        new_max = float(new_corr.max())
        
        if new_max <= cur_max or (T > 1e-10 and rng.random() < np.exp(-(new_max - cur_max) / T)):

            zeros.add(j)
            ones.discard(j)
            ones.add(i)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

def project_feasible(h, n):

def subgrad_optimize(h, n, deadline, max_iters=50000, top_k=3, lr_base=0.08):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    G = np.zeros(n)
    momentum = np.zeros(n)
    beta = 0.9
    t = 0
    stall = 0
    stall_limit = 5000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            h = best_h + 0.01 * np.random.RandomState(t).randn(n)
            h = project_feasible(h, n)
            G = np.zeros(n)
            momentum = np.zeros(n)
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] = 1.0 - 2.0 * h
                else:
                    grad[k:] += (1.0 - h[:n - k])
                    grad[:n - k] -= h[k:]
            else:
                kk = -k
                if kk == 0:
                    grad[:] = 1.0 - 2.0 * h
                else:
                    grad[:n - kk] += (1.0 - h[kk:])
                    grad[kk:] -= h[:n - kk]
            grad /= 1.0
        grad = dx * grad / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum ** 2
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_check.max() < cur_c + 0.005:
            h = h_new
    return best_h, best_c
