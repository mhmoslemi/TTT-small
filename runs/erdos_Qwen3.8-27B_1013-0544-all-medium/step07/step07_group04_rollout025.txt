        if dl_phase <= time.time():
            continue
        if k == 0:
            hinit = best_h.copy()
        else:
            hinit = interp_h(best_h, best_n, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)
        hopt, copt = optimize_single(hinit, nh, seed + 13 * k, dl_phase)
        if copt < best_c:
            best_c = copt
            best_h = hopt.copy()
            best_n = nh
    # Final polish
    if time.time() < deadline - 10:
        hfin = best_h.copy()
        cfin = best_c
        for alpha, maxit in [(3000, 100), (10000, 80)]:
            if time.time() > deadline - 5:
                break
            h2 = slsqp_phase(hfin, best_n, alpha, maxit)
            h2 = project_box_sum(h2, best_n)
            v2 = float(compute_c(h2, best_n).max())
            if v2 < cfin:
                cfin = v2
                hfin = h2.copy()
        if time.time() < deadline - 3:
            hfin, cfin = pair_local_search(hfin, deadline - 2, seed=seed + 777,
                                           max_iter=3000)
        if cfin < best_c:
            best_c = cfin
            best_h = hfin.copy()
    best_h = project_box_sum(best_h, best_n)
    best_c = float(compute_c(best_h, best_n).max())
    return best_h, best_c, best_n
