        h4, c4 = adam_refine(h4, n, dl - 2, alpha0=10000, alpha1=200000, top_k=3, lr0=0.005)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
