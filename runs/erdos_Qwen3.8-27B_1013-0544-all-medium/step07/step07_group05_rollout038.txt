def softmax_values(c, beta):

    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    w = ew / ws if ws > 0 else np.zeros_like(c)
    return f, w

def smooth_grad(h, w, n, dx):
    mid = n - 1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    consts = np.cumsum(w_pos) + np.cumsum(w_neg)[n - 1 - np.arange(n)]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

        j = rng.randint(0, n)
        if i == j:
            continue

        lo = max(-hi0, hj0 - 1.0)
        hi = min(1.0 - hi0, hj0)
        if hi - lo < 1e-13:

        h[i] = hi0 + d
        h[j] = hj0 - d

            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            # Reidentified lags - occasionally refresh
        else:
            h[i] = hi0
            h[j] = hj0
    
    return best_h, best_c

def make_binary_relaxation(h, n):
    """Push h towards binary values, keeping sum constraint."""
    h = np.asarray(h, dtype=float)
    # Sort by distance from 0.5
    dist = np.abs(h - 0.5)
    order = np.argsort(dist)[::-1]  # Start with values already near 0 or 1
    # Threshold: values above 0.5 go to 1, below go to 0, with a gap
    frac_mask = (h > 0.05) & (h < 0.95)
    n_frac = int(frac_mask.sum())
    if n_frac == 0:
        return h
    # Push fractional values towards 0 or 1
    new_h = h.copy()
    for i in order:
        if not frac_mask[i]:
            continue
        if new_h[i] > 0.5:
            new_h[i] = 1.0
        else:
            new_h[i] = 0.0
    return project_box_sum(new_h, 0.5 * n)

        elif -n < offset < 0:
            idx = np.arange(-offset)
            prod = h[idx + offset] * (1.0 - h[idx])
            top_idx = np.argsort(prod)[::-1][:8]
            for ti in top_idx:
                moves.append((int(ti + offset), int(ti), 1))
    for _ in range(25):
        i, j = rng.randint(0, n), rng.randint(0, n)
        if i != j:
            moves.append((i, j, 1))
            moves.append((i, j, -1))
    return moves

def directed_sa(h, n, deadline, seed):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float)
    cur_c = erdos_c5(h, n)
    best_h = h.copy()
    best_c = cur_c
    moves = get_directed_moves(h, n, rng)
    temp = 0.004
    max_iter = int((deadline - time.time()) * 500)
    max_iter = min(max_iter, 80000)
    for it in range(max_iter):
        if time.time() > deadline - 5.0:
            break
        if it % 800 == 0 and temp > 1e-8:
            temp *= 0.96
        if it % 4000 == 0:
            moves = get_directed_moves(h, n, rng)
        i, j, sign = moves[rng.randint(0, len(moves))]
        if sign == 1:
            hi_max = min(h[i], 1.0 - h[j])
        else:
            hi_max = min(1.0 - h[i], h[j])
        if hi_max < 1e-13:
            continue
        delta = rng.uniform(0.0, hi_max)
        h[i] += sign * delta
        h[j] -= sign * delta
        new_c = erdos_c5(h, n)
        if new_c <= cur_c or (temp > 1e-9 and rng.random() < np.exp(-(new_c - cur_c) / temp)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] -= sign * delta
            h[j] += sign * delta
    return best_h, best_c

def binary_restart(h, n, rng, strength=0.7):
    """Create a restart candidate by pushing towards binary then adding noise."""
    h = np.asarray(h, dtype=float)
    b = (h > 0.5).astype(float)
    b = project_box_sum(b, 0.5 * n)
    noise = rng.randn(n) * 0.05
    return project_box_sum(b * (1.0 - strength) + h * strength + noise, 0.5 * n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 5.0)
    rng = np.random.RandomState(int(seed))
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n < 2:
        n = 400
        h = np.full(n, 0.5)
    if n > 600:
        h = np.interp(np.linspace(0, 2, 600), np.linspace(0, 2, n), h)
        n = 600
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    # Phase 1: L-BFGS-B sweep with increasing beta
    phase1_end = t0 + 0.12 * (deadline - t0)
    for beta in [100.0, 500.0, 2000.0, 8000.0, 30000.0]:
        if time.time() > phase1_end:
            break
        cand, c = refine_lbfgs(best_h, beta, 60, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    # Phase 2: Coarse-to-fine escape
    phase2_end = t0 + 0.25 * (deadline - t0)
    for nc in [80, 50]:
        if time.time() > phase2_end:
            break
        hc = resample(best_h, best_n, nc)
        for beta in [200.0, 2000.0, 10000.0]:
            if time.time() > phase2_end:
                break
            hc, _ = refine_lbfgs(hc, beta, 30, deadline)
        hf = resample(hc, nc, best_n)
        hf, c = refine_lbfgs(hf, 5000.0, 40, deadline)
        if c < best_c - 1e-6:
            best_h, best_c = hf, c

    # Phase 3: High-beta refinement
    phase3_end = t0 + 0.38 * (deadline - t0)
    for beta in [50000.0, 200000.0, 800000.0, 3000000.0]:
        if time.time() > phase3_end:
            break
        cand, c = refine_lbfgs(best_h, beta, 50, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    # Phase 4: Directed simulated annealing (main exploration)
    phase4_end = t0 + 0.65 * (deadline - t0)
    best_h, best_c = directed_sa(best_h, best_n, phase4_end, seed)

    # Phase 5: L-BFGS-B after SA
    phase5_end = t0 + 0.75 * (deadline - t0)
    for beta in [100000.0, 500000.0, 2000000.0]:
        if time.time() > phase5_end:
            break
        cand, c = refine_lbfgs(best_h, beta, 40, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    # Phase 6: Binary restart + optimize (explore different basin)
    phase6_end = t0 + 0.85 * (deadline - t0)
    for attempt in range(3):
        if time.time() > phase6_end - 30:
            break
        h_r = binary_restart(best_h, best_n, rng, strength=0.5 + 0.15 * attempt)
        h_r, c = refine_lbfgs(h_r, 5000.0, 40, deadline)
        h_r, c = refine_lbfgs(h_r, 100000.0, 30, deadline)
        if c < best_c - 1e-8:
            best_h, best_c = h_r, c

    # Phase 7: Second SA round
    phase7_end = deadline - 40.0
    if time.time() < phase7_end:
        best_h, best_c = directed_sa(best_h, best_n, phase7_end, seed + 999)

    # Phase 8: Final polish
    if time.time() < deadline - 15.0:
        cand, c = refine_lbfgs(best_h, 5000000.0, 30, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
