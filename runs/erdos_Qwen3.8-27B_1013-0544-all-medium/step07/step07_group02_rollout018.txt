def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):

        j2e = min(n, n + k)
        if j2e > j2s:

    return G

                stall = 0
            else:
                stall += 1
                if stall >= 15:
                    eps *= 0.6
                    if eps < 0.001:
                        break
                    stall = 0
            h = h_new

    # Phase 3: Diverse restarts on best grid
    p3_end = dl - 25
    work_n = best_n
    n_restarts = 200
    for trial in range(n_restarts):
        if time.time() > p3_end:
            break
        rem = p3_end - time.time()
        per_trial = max(2.0, rem / max(1, n_restarts - trial))

        r = rng.rand()
        if r < 0.4:
            h = best_h + 0.012 * rng.randn(work_n)
        elif r < 0.6:
            h = best_h[::-1].copy() + 0.008 * rng.randn(work_n)
        elif r < 0.8:
            f = rng.randint(1, work_n // 3)
            ph = rng.rand() * 2.0 * np.pi
            pert = 0.01 * np.sin(2.0 * np.pi * f * np.arange(work_n) / work_n + ph)
            h = best_h + pert
        else:
            idx = rng.choice(work_n, size=work_n // 4, replace=False)
            h = best_h.copy()
            h[idx] += 0.03 * rng.randn(len(idx))
        h = project_feasible(h, work_n)

        eps_r = 0.06
        for _ in range(300):
            if time.time() > p3_end or time.time() > t0 + per_trial:
                break
            h_new, mc_new = lp_step(h, work_n, eps_r)
            c5_new = mc_new * 2.0 / work_n
            if c5_new < best_c5 - 1e-13:
                best_mc = mc_new
                best_h = h_new.copy()
                best_c5 = c5_new
                best_n = work_n
            h = h_new
            eps_r = max(eps_r * 0.997, 0.003)

    # Phase 4: Final fine LP polish
    if time.time() < dl - 3:
        h = best_h.copy()
        cur_n = best_n
        eps_f = 0.003
        for _ in range(1000):
            if time.time() > dl - 3:
                break
            h_new, mc_new = lp_step(h, cur_n, eps_f)
            c5_new = mc_new * 2.0 / cur_n
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_mc = mc_new
                best_h = h_new.copy()
                best_n = cur_n
            h = h_new
            eps_f = max(eps_f * 0.9995, 0.0003)

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
