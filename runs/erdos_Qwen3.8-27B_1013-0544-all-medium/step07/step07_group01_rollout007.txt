```python
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def correlate_max(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def active_gradient(h, n, n_active=30):
    """Gradient of max correlation using active set (top n_active lags)"""
    dx = 2.0 / n
    corr = correlate_max(h, n)
    top_idx = np.argsort(corr)[-n_active:]

        elif ell < 0 and -ell < n:

    y = h.copy()

            break

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

            best_h = h_new.copy()
        if c_new >= compute_c5(h0, n) - 1e-10:
```

Actually, I realize there might be an even better approach. Let me think about the problem structure more carefully.

The correlation c_k = dx * sum_i h[i]*(1-h[i+k]). For fixed h, this is a function of k. We want to minimize the maximum over k.

At the optimum, multiple lags should be "active" (achieve the maximum). This is a saddle-point condition.

**KKT conditions** at the optimum:
- There exist weights λ_k ≥ 0, sum λ_k = 1, such that:
  - c_k = C for all k with λ_k > 0
  - c_k ≤ C for all k
  - The gradient of sum(λ_k * c_k) + (sum(h) - n/2)*μ is zero (for interior points) or the correct inequality holds (for boundary points)

This suggests an **iterative reweighting** approach:
1. Start with uniform weights on the top lags
2. Compute the weighted gradient
3. Take a step
4. Update weights based on new correlations
5. Repeat

This is essentially what the soft-max SLSQP does, but without the SLSQP overhead.

Let me implement this as a "dual ascent" method:

```python
    t2 = min(t0 + 450, deadline - 250)
    for a in [50, 200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > t2 - 25:
            break
        h2 = slsqp_smooth(best_h, n, a, 2000, 1e-14)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: CVXPY successive linearization
    if time.time() < deadline - 200:
        t3 = min(time.time() + 120, deadline - 150)
        h3, c3 = cvxpy_linearize(best_h, n, max_iter=25, deadline=t3)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Dual ascent refinement
    t4 = min(time.time() + 120, deadline - 120)
    if time.time() < t4:
        h4, c4 = dual_ascent(best_h, n, t4)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: High-alpha SLSQP polish
    t5 = deadline - 60
    for a in [500000, 2000000, 10000000, 50000000]:
        if time.time() > t5 - 12:
            break
        h5 = slsqp_smooth(best_h, n, a, 3000, 1e-16)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Noise restarts with short SLSQP
    for trial in range(20):
        if time.time() > deadline - 8:
            break
        noise = 0.001 + 0.0015 * trial
        h6 = best_h + noise * rng.randn(n)
        h6 = project_feasible(h6, n)
        h6 = slsqp_smooth(h6, n, 50000, 600, 1e-13)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
