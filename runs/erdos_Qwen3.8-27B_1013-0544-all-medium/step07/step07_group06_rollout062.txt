    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:

        F = m + np.log(s) / beta
    P = p

    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

    for p in range(n):
        f = p + l
        m1 = (f >= 0) & (f < n)
        base[p, m1] += 1.0 - h[f[m1]]
        b = p - l
        m2 = (b >= 0) & (b < n)
        base[p, m2] -= h[b[m2]]
    return base

def binary_descent(h, n, deadline, max_sweeps=50):
    if n % 2:
        return h, compute_c5(h, n)
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(corr.max()))
    best_h, best_M = h.copy(), M
    for _ in range(max_sweeps):
        if time.time() > deadline: break
        base = build_base(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0: break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        best_loc, best_mx = -1, M
        chunk = 4096
        for start in range(0, zi.size, chunk):
            if time.time() > deadline: break
            end = min(start + chunk, zi.size)
            i = zi[start:end]
            j = oj[start:end]
            arr = base[i] - base[j] + corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = (n - 1 + (j - i)).astype(int)
            d2 = (n - 1 + (i - j)).astype(int)
            dm = (d1 >= 0) & (d1 <= 2 * n - 2)
            arr[np.arange(end - start)[dm], d1[dm]] += 1.0
            dm2 = (d2 >= 0) & (d2 <= 2 * n - 2)
            arr[np.arange(end - start)[dm2], d2[dm2]] += 1.0
            mx = arr.max(axis=1)
            cand = mx < best_mx - 1e-8
            if cand.any():
                idx = np.nonzero(cand)[0]
                loc = int(idx[np.argmin(mx[idx])])
                if mx[loc] < best_mx:
                    best_mx = mx[loc]
                    best_loc = start + loc
        if best_loc >= 0:
            iv = int(zi[best_loc])
            jv = int(oj[best_loc])
            h[iv], h[jv] = 1.0, 0.0
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
            M = int(round(corr.max()))
            if M < best_M:
                best_M, best_h = M, h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_M * (2.0 / n)

def binary_sa(h, n, deadline, seed=42):
    if n % 2:
        return h, compute_c5(h, n)
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(corr.max())
    best_h, best_M = h.copy(), M
    rng = np.random.RandomState(seed)
    T = max(3.0, 0.08 * M)
    stall = 0
    while time.time() < deadline:
        if stall > 8000:
            T = max(2.0, 0.06 * M)
            stall = 0
        T = max(0.05, T * 0.9995)
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if h[i] == h[j]: continue
        if h[j] == 1:
            i, j = j, i
        h[i], h[j] = 0.0, 1.0
        nc = np.correlate(h, 1.0 - h, mode="full")
        nM = int(nc.max())
        if nM <= M or (T > 0.05 and rng.random() < np.exp(-(nM - M) / T)):
            corr, M = nc.astype(float), nM
            stall = 0
            if M < best_M:
                best_M, best_h = M, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
    return best_h, best_M * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    end = t0 + total - 15.0
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        h0 = (np.arange(n0) // (n0 // 40) % 2).astype(float)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0
    # Phase 1: continuous polish
    if time.time() < end - 350:
        hh, cc = polish(best_h, best_n, [50, 200, 1000, 4000, 15000], 1e4, end - 300, 45)
        if cc < best_c:
            best_c, best_h = cc, hh
    # Phase 2: binary descent (fast local search)
    if time.time() < end - 200:
        bh, bc2 = binary_descent(best_h, best_n, end - 180, max_sweeps=40)
        if bc2 < best_c:
            best_c, best_h, best_n = bc2, bh, best_n
    # Phase 3: binary SA (global search)
    if time.time() < end - 120:
        bh, bc3 = binary_sa(best_h, best_n, end - 100, seed=seed)
        if bc3 < best_c:
            best_c, best_h, best_n = bc3, bh, best_n
    # Phase 4: multi-resolution
    for nnew in (200, 300, 500, 600):
        if time.time() > end - 60: break
        x0 = np.linspace(0, 2, best_n)
        x1 = np.linspace(0, 2, nnew)
        hr = project_feasible(np.interp(x1, x0, best_h), nnew)
        hh, cc = polish(hr, nnew, [300, 1500, 6000], 1e4, end - 30, 15)
        if nnew % 2 == 0 and time.time() < end - 20:
            bh, bc4 = binary_descent(hh, nnew, end - 15, max_sweeps=15)
            if bc4 < cc:
                hh, cc = bh, bc4
        if cc < best_c:
            best_c, best_h, best_n = cc, hh, nnew
    # Phase 5: final polish + binary check
    if time.time() < end - 15:
        hh, cc = polish(best_h, best_n, [5000, 20000], 1e4, end - 8, 10)
        if cc < best_c:
            best_c, best_h = cc, hh
    if best_n % 2 == 0:
        tb = to_binary(best_h, best_n)
        ct = compute_c5(tb, best_n)
        if ct < best_c:
            best_c, best_h = ct, tb
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
