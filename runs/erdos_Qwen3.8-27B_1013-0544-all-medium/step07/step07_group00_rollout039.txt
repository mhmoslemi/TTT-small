def pair_update(h,c,p,q,lag,grid):
    n=len(h); L=2*n-1
    A=np.zeros(L); B=np.zeros(L)
    # p first
    valid = (p - lag >= 0) & (p - lag < n)
    if np.any(valid):
        idx = p - lag[valid]
        A[valid] += 1.0 - h[idx]
    # p second
    valid = (p + lag >=0) & (p + lag < n)
    if np.any(valid):
        idx = p + lag[valid]
        A[valid] -= h[idx]
    # q first
    valid = (q - lag >=0) & (q - lag < n)
    if np.any(valid):
        idx = q - lag[valid]
        A[valid] -= 1.0 - h[idx]
    # q second
    valid = (q + lag >=0) & (q + lag < n)
    if np.any(valid):
        idx = q + lag[valid]
        A[valid] += h[idx]
    # B diag
    B[n-1] += -2.0
    # cross
    d=q-p
    if - (n-1) <= d <= (n-1):
       B[d + n -1] += 1.0
    d=p-q
    if - (n-1) <= d <= (n-1):

    cur = float(c.max())

    hi = min(1.0 - hp, hq)
    if hi <= lo + 1e-14: return h.copy(), c.copy(), False
    # Make grid: combine negative and positive

def make_feasible(h, n):

        elif abs(r) > 1e-15:

def smooth_grad_base(h, p):
    n = int(h.size)
    ones = np.ones(n, dtype=float)
    A = np.convolve(ones, p, mode='full')[n - 1:2 * n - 1]
    B = np.convolve(h, p, mode='full')[n - 1:2 * n - 1]
    C = np.convolve(h, p[::-1], mode='full')[n - 1:2 * n - 1]
    return A - C - B


def smooth_val_grad(h, alpha):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    dx = 2.0 / n
    c = corr_of(h, n)
    m = float(np.max(c))
    z = alpha * (c - m)
    e = np.exp(z)
    se = float(np.sum(e))

    if se <= 0.0 or not np.isfinite(se):
        p = np.zeros_like(c)
        idx = np.flatnonzero(c == m)
        if idx.size == 0:
            p[n - 1] = 1.0
        else:
            p[idx] = 1.0
            p /= p.sum()
    else:
        p = e / se

    val = dx * (m + np.log(se) / alpha)
    g = dx * smooth_grad_base(h, p)
    return val, g

    return smooth_val_grad(h, alpha)[1]

def slsqp_jac(x, alpha):
    return smooth_grad_only(x, alpha)


def sum_con(x, n):
    return float(np.asarray(x).sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones(len(x), dtype=float)


def slsqp_whole(h, alpha, maxiter, deadline=None):
    n = int(h.size)
    old_c = compute_c5(h, n)
    old_v = smooth_val_only(h, alpha)
    if deadline is not None and time.time() > deadline - 2:
        return h, old_c, old_v

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    opts = {'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': 0}
    try:
        res = minimize(slsqp_obj, np.asarray(h, dtype=float).copy(),
                       args=(alpha,), jac=slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
        xx = np.asarray(res.x, dtype=float)
        if not np.all(np.isfinite(xx)):
            return h, old_c, old_v
    except Exception:
        return h, old_c, old_v

    h2 = make_feasible(xx, n)
    return h2, compute_c5(h2, n), smooth_val_only(h2, alpha)


def block_obj(x, base, idx, alpha):
    h = base.copy()
    h[idx] = x
    return smooth_val_only(h, alpha)


def block_jac(x, base, idx, alpha):
    h = base.copy()
    h[idx] = x
    return smooth_grad_only(h, alpha)[idx]


def block_sum_con(x, target):
    return float(np.asarray(x).sum() - target)


def block_sum_jac(x, m):
    return np.ones(m, dtype=float)


def slsqp_block(h, idx, alpha, maxiter, deadline=None):
    n = int(h.size)
    old_c = compute_c5(h, n)
    old_v = smooth_val_only(h, alpha)
    m = int(len(idx))
    if m < 2 or (deadline is not None and time.time() > deadline - 1):
        return h, old_c, old_v

    target = float(h[idx].sum())
    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': block_sum_con, 'jac': block_sum_jac,
             'args': (target, m)}]
    opts = {'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': 0}
    try:
        res = minimize(block_obj, h[idx].copy(),
                       args=(h, idx, alpha), jac=block_jac, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
        xx = np.asarray(res.x, dtype=float)
        if not np.all(np.isfinite(xx)):
            return h, old_c, old_v
    except Exception:
        return h, old_c, old_v

    h2 = h.copy()
    h2[idx] = np.clip(xx, 0.0, 1.0)
    h2 = make_feasible(h2, n)
    return h2, compute_c5(h2, n), smooth_val_only(h2, alpha)


def block_slsqp(h, n, alpha, b_size, stride, sweeps, maxiter, deadline, rng):
    n = int(h.size)
    b_size = min(int(b_size), n)
    if b_size < 2:
        return h, compute_c5(h, n)

    stride = max(1, min(b_size, int(stride)))
    if n <= b_size:
        starts = np.array([0], dtype=int)
    else:
        max_start = n - b_size
        starts = np.arange(0, max_start + 1, stride, dtype=int)
        if starts.size == 0 or starts[-1] != max_start:
            starts = np.append(starts, max_start)

    for sw in range(int(sweeps)):
        if deadline is not None and time.time() > deadline - 4:
            break
        order = starts.copy()
        if sw > 0 and rng is not None and order.size > 1:
            order = order[rng.permutation(order.size)]

        improved = False
        for st in order:
            if deadline is not None and time.time() > deadline - 2:
                break
            st = int(st)
            idx = np.arange(st, st + b_size)
            old_c = compute_c5(h, n)
            old_v = smooth_val_only(h, alpha)

            h2, c2, v2 = slsqp_block(h, idx, alpha, maxiter, deadline)
            if c2 < old_c - 1e-14 or (c2 <= old_c + 1e-8 and v2 < old_v - 1e-12):
                h = h2
                improved = True
        if not improved:
            break

    return h, compute_c5(h, n)


def interp_h(old, n_old, n_new):
    if n_old == n_new:
        return np.asarray(old, dtype=float).copy()
    x0 = np.linspace(0.0, 2.0, n_old)
    x1 = np.linspace(0.0, 2.0, n_new)
    return make_feasible(np.interp(x1, x0, np.asarray(old, dtype=float)), n_new)


def pair_update(h, c, p, q, lag, grid):
    n = int(h.size)
    L = 2 * n - 1
    A = np.zeros(L, dtype=float)
    B = np.zeros(L, dtype=float)

    valid = (p - lag >= 0) & (p - lag < n)
    if np.any(valid):
        A[valid] += 1.0 - h[p - lag[valid]]

    valid = (p + lag >= 0) & (p + lag < n)
    if np.any(valid):
        A[valid] -= h[p + lag[valid]]

    valid = (q - lag >= 0) & (q - lag < n)
    if np.any(valid):
        A[valid] -= 1.0 - h[q - lag[valid]]

    valid = (q + lag >= 0) & (q + lag < n)
    if np.any(valid):
        A[valid] += h[q + lag[valid]]

    B[n - 1] += -2.0
    d = q - p
    if -(n - 1) <= d <= (n - 1):
        B[d + n - 1] += 1.0
    d = p - q
    if -(n - 1) <= d <= (n - 1):
        B[d + n - 1] += 1.0

    cur = float(np.max(c))
    hp = float(h[p])
    hq = float(h[q])

    affected = (A != 0.0) | (B != 0.0)
    if not np.any(affected & (c >= cur - 1e-8)):
        return h, c, False

    lo = max(-hp, hq - 1.0)
    hi = min(1.0 - hp, hq)
    if hi <= lo + 1e-14:
        return h, c, False

    neg = np.linspace(lo, 0.0, int(grid) + 1)
    pos = np.linspace(0.0, hi, int(grid) + 1)
    deltas = np.concatenate((neg, pos))

    best_d = 0.0
    best_val = cur
    for d in deltas:
        if abs(d) < 1e-16:
            continue
        nc = c + A * d + B * d * d
        val = float(np.max(nc))
        if val < best_val:
            best_val = val
            best_d = d

    if best_d != 0.0 and best_val < cur - 1e-14:
        h2 = h.copy()
        h2[p] = np.clip(hp + best_d, 0.0, 1.0)
        h2[q] = np.clip(hq - best_d, 0.0, 1.0)
        c2 = c + A * best_d + B * best_d * best_d
        return h2, c2, True

    return h, c, False


def pair_cd(h, n, deadline, rng, grid=22, alpha=8000, max_iter=200000):
    dx = 2.0 / int(n)
    h = make_feasible(h, n)
    c = corr_of(h, n)
    best_score = float(np.max(c)) * dx
    lag = np.arange(2 * n - 1, dtype=int) - (n - 1)

    free_arr = np.arange(n, dtype=int)
    low = free_arr.copy()
    high = free_arr.copy()

    for it in range(int(max_iter)):
        if deadline is not None and time.time() > deadline - 3:
            break

        if it % 20 == 0:
            c = corr_of(h, n)
            score = float(np.max(c)) * dx
            if score < best_score:
                best_score = score

            _, g = smooth_val_grad(h, alpha)
            free = (h > 5e-5) & (h < 1.0 - 5e-5)
            if free.sum() < max(10, n // 10):
                free = (h > 0.0) & (h < 1.0)
            if free.sum() < 10:
                free = np.ones(n, dtype=bool)

            free_arr = np.flatnonzero(free)
            if free_arr.size < 2:
                break
            gf = g[free_arr]
            k = min(60, free_arr.size)
            order = np.argsort(gf)
            low = free_arr[order[:k]]
            high = free_arr[order[::-1][:k]]

        if low.size == 0:
            low = free_arr
        if high.size == 0:
            high = free_arr

        p = int(rng.choice(low))
        q = int(rng.choice(high))
        if p == q:
            if free_arr.size > 1:
                q = int(rng.choice(free_arr))
                if p == q:
                    continue
            else:
                continue

        h2, c2, ok = pair_update(h, c, p, q, lag, grid)
        if ok:
            h = h2
            c = c2
            score = float(np.max(c)) * dx
            if score < best_score:
                best_score = score

    h = make_feasible(h, n)
    c = corr_of(h, n)
    return h, float(np.max(c)) * (2.0 / n)


def refine_n(h, n, deadline, rng, full=True):
    n = int(h.size)
    h = make_feasible(h, n)
    cur = h.copy()
    cur_c = compute_c5(cur, n)

    best_h = cur.copy()
    best_c = cur_c

    if deadline is not None and time.time() > deadline - 12:
        return best_h, best_c

    if full:
        if time.time() < deadline - 12:
            b = min(n, 144)
            cur, cur_c = block_slsqp(cur, n, 3000, b, max(16, b // 2), 1, 20, deadline, rng)
            if cur_c < best_c - 1e-14:
                best_h, best_c = cur.copy(), cur_c

        if time.time() < deadline - 12:
            b = min(n, 96)
            cur, cur_c = block_slsqp(cur, n, 20000, b, max(12, b // 2), 1, 16, deadline, rng)
            if cur_c < best_c - 1e-14:
                best_h, best_c = cur.copy(), cur_c

        if time.time() < deadline - 20 and n < 1000:
            old_v = smooth_val_only(cur, 5000)
            h2, c2, v2 = slsqp_whole(cur, 5000, 55, deadline)
            old_c = compute_c5(cur, n)
            if c2 < best_c - 1e-14:
                best_h, best_c = h2.copy(), c2
            if c2 <= old_c + 1e-8 or (c2 <= old_c + 1e-4 and v2 < old_v - 1e-10):
                cur, cur_c = h2, c2

            if time.time() < deadline - 14:
                old_v = smooth_val_only(cur, 30000)
                h3, c3, v3 = slsqp_whole(cur, 30000, 35, deadline)
                old_c = compute_c5(cur, n)
                if c3 < best_c - 1e-14:
                    best_h, best_c = h3.copy(), c3
                if c3 <= old_c + 1e-8 or (c3 <= old_c + 1e-4 and v3 < old_v - 1e-10):
                    cur, cur_c = h3, c3
    else:
        if time.time() < deadline - 10:
            b = min(n, 112)
            cur, cur_c = block_slsqp(cur, n, 10000, b, max(16, b // 2), 1, 16, deadline, rng)
            if cur_c < best_c - 1e-14:
                best_h, best_c = cur.copy(), cur_c

        if time.time() < deadline - 18 and n < 950:
            old_v = smooth_val_only(cur, 30000)
            h2, c2, v2 = slsqp_whole(cur, 30000, 28, deadline)
            old_c = compute_c5(cur, n)
            if c2 < best_c - 1e-14:
                best_h, best_c = h2.copy(), c2
            if c2 <= old_c + 1e-8 or (c2 <= old_c + 1e-4 and v2 < old_v - 1e-10):
                cur, cur_c = h2, c2

    if time.time() < deadline - 8:
        h2, c2 = pair_cd(cur, n, deadline - 1, rng, grid=22, alpha=8000)
        if c2 < cur_c:
            cur, cur_c = h2, c2
        if cur_c < best_c - 1e-14:
            best_h, best_c = cur.copy(), cur_c

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s) - 5.0
    rng = np.random.RandomState(int(seed))

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 500
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * (0.7 * x + 1.0))
        h = make_feasible(h, n)

    h = make_feasible(h, int(h.size))
    best_h = h.copy()
    best_c = compute_c5(best_h, len(best_h))

    parent_deadline = min(deadline, t0 + min(0.36 * budget_s, 420.0))
    h1, c1 = refine_n(h, len(h), parent_deadline, rng, full=True)
    if c1 < best_c:
        best_h, best_c = h1.copy(), c1

    base_n = len(best_h)
    target_specs = [(700, 0.17), (850, 0.20), (950, 0.22)]

    for nt_alloc, frac in target_specs:
        if time.time() > deadline - 100:
            break
        nt = int(nt_alloc)
        if base_n >= 500:
            nt = min(1000, max(nt, base_n + 50))
        else:
            nt = min(950, max(nt, 500))
        if nt <= base_n:
            nt = min(1000, base_n + 100)

        stage_deadline = min(deadline, time.time() + max(50.0, budget_s * frac))
        st = interp_h(best_h, len(best_h), nt)
        if rng.random() < 0.65:
            st = st + rng.normal(0.0, 1.5e-4, nt)
        st = make_feasible(st, nt)

        h2, c2 = refine_n(st, nt, stage_deadline, rng, full=False)
        if c2 < best_c:
            best_h, best_c = h2.copy(), c2
            base_n = nt

    if time.time() < deadline - 20:
        n = len(best_h)
        h3, c3 = pair_cd(best_h, n, deadline - 2, rng, grid=30, alpha=12000)
        if c3 < best_c:
            best_h, best_c = h3.copy(), c3

    best_h = make_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
