import time
import numpy as np
import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'target': 0.5, 'beta': 1.0, 'rho': 1e5, 'lam': 0.0, 'topk': 0}
SS = {'n': 1, 'dx': 1.0, 'target': 0.5, 'beta': 1.0, 'topk': 0, 'cx': None, 'cg': None}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_values(c, beta, topk):
    c = np.asarray(c, dtype=float)
    if topk > 0 and len(c) > topk:
        idx = np.argsort(c)[-topk:]
        cc = c[idx].copy()
    else:
        idx = None
        cc = c.copy()
    maxc = float(np.max(cc))
    z = beta * (cc - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(cc)
        w[int(np.argmax(cc))] = 1.0
    else:
        w = ew / ws
    if idx is not None:
        wfull = np.zeros_like(c)
        wfull[idx] = w
        return f, wfull
    return f, w


def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def l_fg(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    target = float(st['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']), int(st['topk']))
    g = smooth_grad_from_w(x, w, n, dx)
    res = float(x.sum() - target)
    rho = float(st['rho'])
    lam = float(st['lam'])
    if rho != 0.0 or lam != 0.0:
        f += lam * res + 0.5 * rho * res * res
        g = g + (lam + rho * res)
    return float(f), g


def refine_lbfgs(h, beta, topk, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    LS['n'] = n
    LS['dx'] = 2.0 / n
    LS['target'] = target
    LS['beta'] = float(beta)
    LS['topk'] = int(topk)
    LS['rho'] = 1e5
    LS['lam'] = 0.0
    mx = int(min(maxiter, max(15, int(rem / 3.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(l_fg, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 40, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def s_obj(x):
    st = SS
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']), int(st['topk']))
    g = smooth_grad_from_w(x, w, n, dx)
    st['cx'] = x.copy()
    st['cg'] = g
    return float(f)


def s_jac(x):
    st = SS
    if st['cx'] is not None and np.array_equal(x, st['cx']):
        return st['cg'].copy()
    s_obj(x)
    return st['cg'].copy()


def s_con_fun(x):
    return float(x.sum() - SS['target'])


def s_con_jac(x):
    return np.ones(int(SS['n']), dtype=float)


def refine_slsqp(h, beta, topk, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 6.0:
        return h, erdos_c5(h, n)
    SS['n'] = n
    SS['dx'] = 2.0 / n
    SS['target'] = target
    SS['beta'] = float(beta)
    SS['topk'] = int(topk)
    SS['cx'] = None
    SS['cg'] = None
    mx = int(min(maxiter, max(8, int(rem / 8.0))))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': s_con_fun, 'jac': s_con_jac}]
    try:
        res = so.minimize(s_obj, h, method='SLSQP', jac=s_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'iprint': 0, 'ftol': 1e-14})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def topk_subgrad(h, k, n_steps, deadline):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    dx = 2.0 / n
    target = 0.5 * n
    step = dx * 0.5
    for it in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = float(np.max(c))
        threshold = cmax - 2e-5
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        k_eff = min(k, len(active))
        if k_eff < len(active):
            top_idx = np.argsort(c)[::-1][:k_eff]
            active = top_idx
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = smooth_grad_from_w(h, w, n, dx)
        for _ in range(3):
            h_new = project_box_sum(h - step * g, target)
            c_new = erdos_c5(h_new, n)
            if c_new < cmax - 1e-15:
                h = h_new
                cmax = c_new
                break
            step *= 0.5
        else:
            step = max(step * 1.2, dx * 0.01)
    return h, erdos_c5(h, n)


def pair_obj(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, int(n))


def pair_line(h, n, i, j, cur, deadline):
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d = 0.0
    best_c = cur
    if time.time() > deadline:
        return h, cur
    c0 = pair_obj(lo, h, i, j, n)
    if c0 < best_c:
        best_c = c0
        best_d = lo
    c1 = pair_obj(hi, h, i, j, n)
    if c1 < best_c:
        best_c = c1
        best_d = hi
    try:
        if time.time() < deadline - 0.5:
            rs = so.minimize_scalar(pair_obj, args=(h, i, j, n),
                                    bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 30, 'xatol': 1e-11})
            if rs.x is not None:
                c2 = pair_obj(float(rs.x), h, i, j, n)
                if c2 < best_c:
                    best_c = c2
                    best_d = float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        hh = np.clip(hh, 0.0, 1.0)
        return hh, best_c
    return h, cur


def pair_polish(h, n, deadline, seed):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    m = 35 if n <= 500 else 25
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= m:
        top = order[:m]
    else:
        need = m - int(order.size)
        if need > n - int(order.size):
            need = max(0, n - int(order.size))
        if need > 0:
            extra = rng.choice(n, size=need, replace=False)
            top = np.unique(np.concatenate((order, extra)))
        else:
            top = order
    if top.size < 2:
        return h, cur
    for outer in range(5):
        improved = False
        pairs = []
        tlen = len(top)
        for a in range(tlen):
            for b in range(a + 1, tlen):
                pairs.append((int(top[a]), int(top[b])))
        if not pairs:
            break
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 3.0:
                return h, erdos_c5(h, n)
            hh, cc = pair_line(h, n, i, j, cur, deadline - 3.0)
            if cc < cur - 1e-14:
                h = hh
                cur = cc
                improved = True
        if not improved:
            break
    return h, cur


def resample_n(h, n, nh):
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * nh)
