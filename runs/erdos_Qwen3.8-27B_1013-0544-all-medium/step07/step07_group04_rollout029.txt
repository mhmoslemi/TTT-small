    if bv < float(c_base.max()) - 5e-14:
        xb = float(np.clip(xs2[km2], lo, hi))
        yb = float(np.clip(S - xb, 0.0, 1.0))
        return xb, yb
    return None

def pair_search(h, c, i, j, n):
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-12:
        return None
    off = n - 1
    dx = 2.0 / n
    L = 2 * n - 1
    omh = 1.0 - h
    a = np.zeros(L)
    b = np.zeros(L)
    nk = n - i
    a[off:off+nk] += omh[i:i+nk]
    b[off:off+nk] -= x0 * omh[i:i+nk]
    if i > 0:
        a[off-i:off] -= h[0:i]
        b[off-i:off] += x0 * h[0:i]
    nk = n - j
    a[off:off+nk] -= omh[j:j+nk]
    b[off:off+nk] += (S - y0) * omh[j:j+nk]
    if j > 0:
        a[off-j:off] += h[0:j]
        b[off-j:off] -= (S - y0) * h[0:j]
    # Cross term correction: at lag d=j-i, term is x*(1-y)=x*(1-S+x)
    # Our linear model has: from i: x*omh[j], from j: -x*omh[i]... 
    # Actually at lag d: contribution from i is x*(1-h[j])_old = x*(1-y0)
    # contribution from j is -(x-(S-y0))*h[i-d]_old ... wait j is at position j, 
    # at lag d=j-i, j's "second factor" contribution is at position j, meaning
    # h[j-d] = h[i] is the first factor. So the product is h[i]*(1-h[j]) = x*(1-y).
    # Our model: i contributes x*(1-y0) at lag d (treating h[j]=y0 fixed)
    #             j contributes y*(1-x0)... no.
    # Let me re-derive. At lag d=j-i>0:
    # c[off+d] = sum_k h[k]*(1-h[k+d])*dx, valid k: 0<=k<n-d
    # h[i] appears as h[k] when k=i: term is h[i]*(1-h[i+d]) = x*(1-h[j])
    # h[j] appears as h[k+d] when k=i: same term
    # h[j] appears as h[k] when k=j: term is h[j]*(1-h[j+d]) = y*(1-h[j+d]) (if j+d<n)
    # So the only place both x and y appear is the term x*(1-y) at k=i.
    # Our linear model:
    # From i's "first factor" role: we added x*(1-h[j]) using old h[j]=y0 -> x*(1-y0)
    # But actual: x*(1-y) = x*(1-S+x) = x^2-(S-1)*x
    # Error: x^2-(S-1)*x - x*(1-y0) = x^2-(S-1)*x - x + x*y0 = x^2 - S*x + x - x + x*y0
    #       = x^2 - S*x + x*y0 = x*(x - S + y0) = x*(x - x0)  [since S-y0=x0]
    # So error = x*(x-x0) at lag d.
    # Similarly at lag -d=i-j<0: the term is h[j]*(1-h[i]) = y*(1-x)
    # Our model: from j's first factor: y*(1-h[i])_old = y*(1-x0)
    # Actual: y*(1-x) = (S-x)*(1-x)
    # Error: (S-x)*(1-x) - (S-x0)*(1-x0) ... this is getting messy
    # For the grid evaluation, the error is at most O((x-x0)^2) which is tiny for small steps.
    # Let's just accept the linear approximation and verify with full computation.
    base = c_base + dx * b
    ngrid = 40
    xs = np.linspace(lo, hi, ngrid)
    vals = np.array([float(np.max(base + dx * a * x)) for x in xs])
    km = int(np.argmin(vals))
    if vals[km] >= float(c_base.max()) - 5e-14:
        return None
    # Verify with exact computation
    xb = float(xs[km])
    yb = float(S - xb)
    h_test = h.copy()
    h_test[i] = xb
    h_test[j] = yb
    c_new = compute_c(h_test, n)
    vnew = float(c_new.max())
    if vnew < float(c_base.max()) - 5e-14:
        return xb, yb
    # If linear model was wrong, reject
    return None

def local_search(h, deadline, seed=1, max_iters=5000):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 300
    off = n - 1
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if noimp > maxno:
            break
        # Select lag: weight by correlation magnitude
        r = rng.random()
        if r < 0.7:
            mag = np.abs(c)
            mag = mag ** 2
            mag += 1e-10
            mag /= mag.sum()
            d_idx = int(rng.choice(2*n-1, p=mag))
            d = d_idx - off
            if d == 0:
                continue
        else:
            d = int(rng.randint(1, n))
        if d > n - 1:
            continue
        i = int(rng.randint(0, max(1, n - abs(d))))
        j = i + d if d > 0 else i - d
        if i < 0 or i >= n or j < 0 or j >= n:
            continue
        res = local_search_pair(h, c, i, j, n)
        if res is not None:
            x, y = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())

def local_search_pair(h, c, i, j, n):
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-12:
        return None
    off = n - 1
    dx = 2.0 / n
    L = 2 * n - 1
    omh = 1.0 - h
    a = np.zeros(L)
    b = np.zeros(L)
    nk = n - i
    if nk > 0:
        a[off:off+nk] += omh[i:i+nk]
        b[off:off+nk] -= x0 * omh[i:i+nk]
    if i > 0:
        a[off-i:off] -= h[0:i]
        b[off-i:off] += x0 * h[0:i]
    nk = n - j
    if nk > 0:
        a[off:off+nk] -= omh[j:j+nk]
        b[off:off+nk] += (S - y0) * omh[j:j+nk]
    if j > 0:
        a[off-j:off] += h[0:j]
        b[off-j:off] -= (S - y0) * h[0:j]
    base = c + dx * b
    ngrid = 32
    xs = np.linspace(lo, hi, ngrid)
    vals = np.array([float(np.max(base + dx * a * x)) for x in xs])
    km = int(np.argmin(vals))
    if vals[km] >= float(c.max()) - 1e-13:
        return None
    # Zoom
    if km == 0:
        xs2 = np.linspace(0, min(2*(hi-lo)/ngrid, hi), 32) + xs[0]
    elif km == ngrid - 1:
        xs2 = np.linspace(xs[-1] - min(2*(hi-lo)/ngrid, hi-xs[-1]), xs[-1], 32)
    else:
        xs2 = np.linspace(xs[km-1], xs[km+1], 32)
    vals2 = np.array([float(np.max(base + dx * a * x)) for x in xs2])
    km2 = int(np.argmin(vals2))
    xb = float(np.clip(xs2[km2], lo, hi))
    yb = float(np.clip(S - xb, 0.0, 1.0))
    # Verify
    h[i] = xb
    h[j] = yb
    c_new = compute_c(h, n)
    vnew = float(c_new.max())
    if vnew < float(c.max()) - 1e-13:
        c[:] = c_new
        return xb, yb
    else:
        h[i] = x0
        h[j] = y0
        return None

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.45 * np.sin(3.0 * np.pi * x + 0.5)
    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = float(compute_c(hp, n0).max())
    bestn = n0
    # Try a few resolutions
    candidates = [n0]
    for m in [n0 - 40, n0 + 40, n0 - 80, n0 + 80]:
        if 100 <= m <= 600 and m not in candidates:
            candidates.append(m)
    candidates.sort(reverse=True)
    n_cands = len(candidates)
    # Allocate time: first candidate gets 40%, rest split 60%
    times = []
    if n_cands == 1:
        times.append(deadline - t0 - 20)
    else:
        times.append(0.4 * (deadline - t0 - 20))
        rem = 0.6 * (deadline - t0 - 20) / (n_cands - 1)
        for _ in range(n_cands - 1):
            times.append(rem)
    for ci, nh in enumerate(candidates):
        if time.time() > deadline - 30:
            break
        dl = t0 + (deadline - t0) * (0.15 + 0.75 * (ci + 0.5) / n_cands) - 5
        dl = min(dl, deadline - 25)
        if dl <= time.time() + 5:
            continue
        if ci == 0:
            hinit = besth.copy()
        else:
            hinit = interpolate_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)
        # Quick L-BFGS-B
        hcur = hinit.copy()
        for alpha, mu, mi in [(20.0, 5.0, 80), (100.0, 50.0, 60), (500.0, 500.0, 50)]:
            if time.time() > dl - 40:
                break
            h2 = lbfgsb(hcur, nh, alpha, mu, mi)
            h2 = project_box_sum(h2, nh)
            v2 = float(compute_c(h2, nh).max())
            if v2 < float(compute_c(hcur, nh).max()):
                hcur = h2
        hcur = project_box_sum(hcur, nh)
        vcur = float(compute_c(hcur, nh).max())
        # Long pair search
        hcur, vcur = local_search(hcur, dl, seed=seed + 100 * ci, max_iters=8000)
        # Final L-BFGS-B polish
        if time.time() < deadline - 10:
            for alpha, mu, mi in [(1000.0, 1000.0, 40), (5000.0, 5000.0, 30)]:
                if time.time() > deadline - 10:
                    break
                h2 = lbfgsb(hcur, nh, alpha, mu, mi)
                h2 = project_box_sum(h2, nh)
                v2 = float(compute_c(h2, nh).max())
                if v2 < vcur:
                    hcur = h2
                    vcur = v2
        if vcur < bestc:
            bestc = vcur
            besth = hcur.copy()
            bestn = nh
    # Final pair search on best
    if time.time() < deadline - 5:
        besth, bestc = local_search(besth, deadline, seed=seed + 999, max_iters=3000)
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
