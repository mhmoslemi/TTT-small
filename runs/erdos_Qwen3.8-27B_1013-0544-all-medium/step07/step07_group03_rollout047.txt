    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h_binary, 1-h_binary, mode='full').astype(float)
    cur_energy = corr.max()
    best_energy = cur_energy
    best_h = h_binary.copy()
    
    ones = np.where(h_binary == 1)[0]
    zeros = np.where(h_binary == 0)[0]
    
    # Temperature schedule
    T_init = 0.5
    T_min = 1e-4
    cooling = 0.9995  # per iteration
    
    T = T_init
    it = 0
    no_improve = 0
    
    while time.time() < deadline:
        it += 1
        
        # Select a swap
        if rng.random() < 0.5 and len(ones) > 0:
            # Targeted: pick i from ones, j from zeros that are "near" the max lag
            max_idx = np.argmax(corr)
            max_lag = max_idx - (n-1)
            # Try swapping a 1 at position p with a 0 at position p+max_lag
            candidates = []
            for p in ones:
                q = p + max_lag
                if 0 <= q < n and h_binary[q] == 0:
                    candidates.append((p, q))
            if not candidates:
                # try p - max_lag

                    if 0 <= q < n and h_binary[q] == 1:
                        candidates.append((q, p))  # (one_pos, zero_pos)
            if candidates:
                i, j = candidates[rng.randint(len(candidates))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            grad += compute_subgrad(idx, h, n)
        grad = dx * grad / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum ** 2
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_check.max() < cur_c + 0.005:
            h = h_new
    return best_h, best_c

def compute_swap_delta(i, j, h, n):
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n-1] -= 2.0
    delta[n-1+(j-i)] += 1.0
    delta[n-1+(i-j)] += 1.0
    return delta

def binary_sa(h_init, n, deadline, rng, max_sweeps=100000):
    dx = 2.0 / n
    h_b = h_init.astype(float).copy()
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_e = float(corr.max())
    best_e = cur_e
    best_h = h_b.copy()
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    T = 3.0
    T_min = 0.01
    cooling = 0.9997
    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_sweeps:
        it += 1
        # Select move
        if rng.random() < 0.6 and len(ones) > 0:
            # Targeted: pick from top lags
            top_lags = np.argsort(corr)[::-1][:3]
            lag_idx = top_lags[rng.randint(3)]
            lag = lag_idx - (n-1)
            # Try to find a swap along this lag
            found = False
            for _ in range(20):
                if lag > 0:
                    p = ones[rng.randint(len(ones))]
                    q = p + lag
                    if 0 <= q < n and h_b[q] == 0:
                        i, j = q, p  # i is 0->1, j is 1->0
                        found = True
                        break
                elif lag < 0:
                    p = ones[rng.randint(len(ones))]
                    q = p + lag
                    if 0 <= q < n and h_b[q] == 0:
                        i, j = q, p
                        found = True
                        break
                else:
                    break
            if not found:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
        
        # Compute delta
        delta = compute_swap_delta(i, j, h_b, n)
        new_corr = corr + delta
        new_e = float(new_corr.max())
        dE = new_e - cur_e
        
        # SA acceptance
        if dE <= 0 or (T > T_min and rng.random() < np.exp(-dE / T)):
            h_b[i] = 1.0; h_b[j] = 0.0
            corr = new_corr
            cur_e = new_e
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            if cur_e < best_e:
                best_e = cur_e
                best_h = h_b.copy()
                no_improve = 0
            else:
                no_improve += 1
            T = max(T_min, T * cooling)

            no_improve += 1

            zeros = list(np.where(h_b == 0)[0])
            T = max(T * 0.5, T_min)
            no_improve = 0
    
    return best_h, best_e

def threshold_to_binary(h, n, rng):
    """Convert continuous h to binary with exactly n/2 ones."""
    h_b = np.zeros(n)
    threshold = np.sort(h)[n//2]
    h_b[h > threshold] = 1.0
    # Handle ties
    n_ones = int(h_b.sum())
    if n_ones != n // 2:
        # Adjust
        sorted_idx = np.argsort(h)[::-1]
        h_b = np.zeros(n)
        h_b[sorted_idx[:n//2]] = 1.0
    return h_b
