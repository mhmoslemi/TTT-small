def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_lbfgs(h, n, alpha, mu, maxiter, time_limit, t_start):
    dx = 2.0 / n
    n_c = 2 * n - 1
    
    def obj_and_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        s = shifted.max()
        exp_vals = np.exp(shifted - s)
        logsum = s + np.log(exp_vals.sum())
        obj = m + logsum / alpha
        
        # Add penalty for sum constraint
        penalty = mu * (x.sum() - n/2.0) ** 2
        obj += penalty
        
        w = exp_vals / exp_vals.sum()
        
        # Vectorized gradient
        conv_part = np.convolve(x, w, 'full')[n-1:2*n-1]
        corr_part = np.correlate(x, w, 'full')[n-1:2*n-1]

        grad_pen = 2.0 * mu * (x.sum() - n/2.0)
        grad = grad_lse + grad_pen
        
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n

        res = minimize(obj_and_grad, h, method='L-BFGS-B', 
                       bounds=bounds,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted.max()
    ev = np.exp(shifted - s)
    logsum = s + np.log(ev.sum())
    obj = m + logsum / alpha + mu * (x.sum() - n / 2.0) ** 2
    w = ev / ev.sum()
    n_c = 2 * n - 1
    cumw = np.concatenate([[0.0], np.cumsum(w)])
    s1 = cumw[n:] - cumw[:n]
    conv_p = np.convolve(x, w, 'full')[n-1:2*n-1]
    corr_wh = np.correlate(w, x, 'full')
    corr_p = corr_wh[np.arange(n-1, -1, -1)]
    grad = (1.0 + 1.0/alpha) * (s1 - conv_p - corr_p)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return obj, grad

def lse_lbfgs(h, n, alpha, mu, maxiter, deadline):

    try:
        res = minimize(lambda x: lse_obj_grad(x, n, alpha, mu), h,
                       method='L-BFGS-B', jac=True, bounds=bounds,

    w = act * (c - t)

        if time.time() > dl5 - 15: break
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, x, best_h), nh)
        for a in [10000, 100000, 1000000]:
            if time.time() > dl5 - 15: break
            hh = lse_lbfgs(hh, nh, a, 1e6, 3000, dl5 - 15)
            hh = project_feasible(hh, nh)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
                x = xn
        hh, ch = barrier_push(hh, nh, compute_c5(hh, nh), dl5 - 15)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), n
            x = np.linspace(0, 2, n)

    # Phase 6: final polish
    h, c = barrier_push(best_h, n, best_c, dl - 5)
    if c < best_c:
        best_c, best_h = c, h.copy()
    for a in [50000000, 200000000]:
        if time.time() > dl - 3: break
        h = lse_lbfgs(best_h, n, a, 1e7, 5000, dl - 3)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
