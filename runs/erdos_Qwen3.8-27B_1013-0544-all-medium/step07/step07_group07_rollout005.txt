def c5_of_h(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))


def adjust_sum(y, S):
    diff = float(S) - float(y.sum())
    if abs(diff) > 1e-12:
        idx = np.where((y > 1e-8) & (y < 1.0 - 1e-8))[0]
        if idx.size:
            k = int(idx[len(idx) // 2])
            y[k] = min(1.0, max(0.0, y[k] + diff))
        else:
            y = np.clip(y + diff / len(y), 0.0, 1.0)
    return y


def project_box_sum(x, S):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return x
    y = np.clip(x, 0.0, 1.0)
    if abs(y.sum() - float(S)) < 1e-13:
        return adjust_sum(y, S)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        v = np.clip(x - mid, 0.0, 1.0)
        if v.sum() > float(S):
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum(y, S)

        return project_box_sum(b, 0.25 * n)

def resample_h(h, n):
    h = np.asarray(h, dtype=float)
    if len(h) == n:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x1, x0, h)


def build_h(x, mode, n):
    x = np.asarray(x, dtype=float).ravel()
    if mode == 0:
        return x
    if mode == 1:
        m = n // 2
        return np.concatenate((x, x[::-1]))
    if mode == 2:
        m = n // 2
        return np.concatenate((x, 1.0 - x))
    q = n // 4
    return np.concatenate((x, 1.0 - x[::-1], 1.0 - x, x[::-1]))


def base_grad(g, mode, n):
    g = np.asarray(g, dtype=float)
    if mode == 0:
        return g
    if mode == 1:
        m = n // 2
        return g[:m] + g[n - 1:n - m - 1:-1]
    if mode == 2:
        m = n // 2
        return g[:m] - g[m:n]
    q = n // 4
    return (g[:q] - g[2 * q:3 * q]
            - g[2 * q - 1:q - 1:-1]
            + g[4 * q - 1:3 * q - 1:-1])


def compute_Q(h, n):
    r = np.correlate(h, h, mode='full')
    c = n - 1
    rpos = r[c:n]
    s = np.concatenate(([0.0], np.cumsum(h)))
    pos = s[n:0:-1]
    neg = s[-1] - s[:n]
    Q = np.empty(2 * n - 1, dtype=float)
    Q[c] = pos[0] - rpos[0]
    Q[c + 1:] = pos[1:] - rpos[1:]
    Q[:c] = (neg[1:] - rpos[1:])[::-1]
    return Q


def grad_Q_from_weights(h, w, n):
    g = w[n - 1] * (1.0 - 2.0 * h)
    for lag in range(1, n):
        a = n - lag
        wp = w[n - 1 + lag]
        wn = w[n - 1 - lag]
        if wp == 0.0 and wn == 0.0:
            continue
        if wp != 0.0:
            g[:a] += wp * (1.0 - h[lag:])
            g[lag:] -= wp * h[:a]
        if wn != 0.0:
            g[:a] -= wn * h[lag:]
            g[lag:] += wn * (1.0 - h[:a])
    return g


def compute_lse(h, alpha, n):
    Q = compute_Q(h, n)
    dx = 2.0 / n
    q = Q * dx
    m = float(np.max(q))
    z = alpha * (q - m)
    ex = np.exp(z)
    s = float(np.sum(ex))
    w = ex / s
    F = m + np.log(s) / alpha
    G = dx * grad_Q_from_weights(h, w, n)
    return float(F), G


def compute_barrier(h, target, k, n):
    Q = compute_Q(h, n)
    dx = 2.0 / n
    q = Q * dx
    z = k * (q - float(target))
    soft = np.logaddexp(0.0, z)
    sig = np.empty_like(z)
    pos = z > 0.0
    sig[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    sig[~pos] = ez / (1.0 + ez)
    F = float(np.sum(soft))
    G = dx * k * grad_Q_from_weights(h, sig, n)
    return F, G

def run_minimize(x0, bounds, obj_fun, jac_fun, args, method, constraints, opts):
    try:
        res = minimize(obj_fun, x0, jac=jac_fun, args=args, method=method,
                       bounds=bounds, constraints=constraints, options=opts)
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.asarray(res.x, dtype=float)
    except Exception:
        pass
    return x0
