def compute_c5(h, n):

    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def min_max_linear(A, B):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        if np.max(A + B * t1) <= np.max(A + B * t2):
            hi = t2
        else:
            lo = t1
    return 0.5 * (lo + hi)

def cd_sweep(h, n, order, lags):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    ncorr = len(corr)
    for j in order:
        jp = j + lags
        jm = j - lags
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        idx_p = np.where(vp, jp, 0)
        idx_m = np.where(vm, jm, 0)
