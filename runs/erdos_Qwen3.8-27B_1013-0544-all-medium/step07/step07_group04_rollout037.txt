def pair_search_fast(h, c, i, j, n):
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    dlo = lo - x0
    dhi = hi - x0
    if dhi - dlo < 1e-13:
        return None
    
    # Compute v_m = a_m - b_m where a_m = dc[m]/dh[i], b_m = dc[m]/dh[j]
    L = 2 * n - 1
    off = n - 1
    v = np.zeros(L)
    for m in range(L):
        k = m - off  # lag
        # a_m = (1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]
        a = 0.0
        if 0 <= i + k < n:
            a += 1.0 - h[i + k]
        if 0 <= i - k < n:
            a -= h[i - k]
        # b_m similarly for j
        b = 0.0
        if 0 <= j + k < n:
            b += 1.0 - h[j + k]
        if 0 <= j - k < n:
            b -= h[j - k]
        v[m] = a - b
    
    # f(d) = max_m (c[m] + d * v[m]) for d in [dlo, dhi]
    # f is convex piecewise linear
    # Find minimum by ternary search (or just dense grid since it's 1D)
    
    # Dense grid
    xs = np.linspace(dlo, dhi, 100)
    vals = np.array([float(np.max(c + d * v)) for d in xs])
    kmin = int(np.argmin(vals))
    
    if vals[kmin] >= float(c.max()) - 1e-14:
        return None
    
    # Refine
    a = xs[max(0, kmin-1)]
    b = xs[min(len(xs)-1, kmin+1)]

        c_cur = compute_c(h_cur, nh)
        dl2 = deadline - 25
        h_cur, c_cur = pair_sweep(h_cur, c_cur, nh, dl2, rng, max_passes=80)
        h_cur = project_box_sum(h_cur, nh)
        c_cur = compute_c(h_cur, nh)
        cv = float(c_cur.max())
        if cv < bestc:
            bestc = cv
            besth = h_cur.copy()
            bestn = nh
    if besth is None:
        besth = project_box_sum(hp, n0)
        bestc = float(compute_c(besth, bestn).max())
        bestn = n0
    return besth, bestc, bestn
