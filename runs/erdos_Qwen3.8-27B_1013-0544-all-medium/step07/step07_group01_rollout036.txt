def smooth_grad(h,n,alpha):
    dx=2.0/n
    corr=np.correlate(h,1.0-h,mode='full')
    F=corr*dx
    m=float(F.max())
    sh=alpha*(F-m)
    mx=sh.max()
    e=np.exp(sh-mx); s=e.sum()
    if s <= 1e-300:
       p=np.zeros(2*n-1); p[int(np.argmax(F))]=1; val=m
    else:
       p=e/s; val=m+(mx+np.log(s))/alpha
    p0=float(p[n-1])
    q=np.zeros(n); q[0]=p0
    U=np.full(n,p0)
    idx=np.arange(n)
    if n>1:
       right=p[n:2*n-1]
       left=p[n-2:0:-1]
       q[1:]=right+left
       pc=np.zeros(n); pc[1:]=np.cumsum(right)
       U += pc[n-1 - idx]

import time
import numpy as np
from scipy.optimize import minimize

_G_N = 1
_G_ALPHA = 1.0
_G_PEN = 1.0
_G_TARGET = 0.5
_G_BOUNDS = []
_RNG = np.random.RandomState(0)


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def project_feasible(h, n):
    target = 0.5 * n
    h = np.array(h, dtype=float, copy=False).reshape(-1)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        if diff > 0:
            h[mask] += diff * room[mask] / room[mask].sum()
        else:
            h[mask] += diff * h[mask] / h[mask].sum()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(12):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = int(np.argmin(1.0 - h))
            add = min(diff, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            add = max(diff, -h[idx])
        if add == 0:
            break
        h[idx] += add
    return np.clip(h, 0.0, 1.0)


def interp_values(src, n):
    m = len(src)
    if m == n:
        return src.copy()
    x = (np.arange(m) + 0.5) * (2.0 / m)
    y = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(y, x, src)


def random_h(n, kind, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if kind == 0:
        a = rng.uniform(0.20, 0.50)
        b = rng.uniform(0.10, 0.30)
        ph = rng.uniform(0.0, 6.28318)
        h = 0.5 + a * np.sin(np.pi * x + ph) + b * np.sin(2.0 * np.pi * x + ph)
    elif kind == 1:
        h = np.full(n, 0.5)
        for m in range(1, 7):
            h += rng.normal(0.0, 0.55 / m) * np.sin(np.pi * m * x)
            h += rng.normal(0.0, 0.55 / m) * np.cos(np.pi * m * x)
    elif kind == 2:
        B = int(rng.randint(3, min(18, max(4, n // 8))))
        B = max(2, min(B, n - 1))
        br = np.sort(rng.choice(np.arange(1, n), size=B - 1, replace=False))
        edges = np.concatenate((np.array([0]), br, np.array([n])))
        vals = np.clip(0.5 + rng.normal(0.0, 0.30, B), 0.0, 1.0)
        h = np.empty(n)
        for i in range(B):
            h[edges[i]:edges[i + 1]] = vals[i]
    else:
        base = 0.5 + 0.42 * np.sin(np.pi * x + rng.uniform(0.0, 6.28318))
        base += 0.18 * np.cos(2.0 * np.pi * x)
        r = max(1, n // 30)
        k = np.exp(-np.linspace(0.0, 3.0, 2 * r + 1) ** 2)
        k /= k.sum()
        pad = np.concatenate((base[:r][::-1], base, base[:r][::-1]))
        h = np.convolve(pad, k, mode='same')[r:2 * r + n]
    return 0.5 * (h + h[::-1])


def smooth_value(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    m = float(F.max())
    sh = alpha * (F - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    if s <= 1e-300:
        return m, m
    return m + (mx + np.log(s)) / alpha, m


def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    m = float(F.max())
    sh = alpha * (F - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    if s <= 1e-300:
        p = np.zeros(2 * n - 1, dtype=float)
        p[int(np.argmax(F))] = 1.0
        val = m
    else:
        p = e / s
        val = m + (mx + np.log(s)) / alpha

    p0 = float(p[n - 1])
    q = np.zeros(n)
    q[0] = p0
    U = np.full(n, p0)
    idx = np.arange(n)
    if n > 1:
        right = p[n:2 * n - 1]
        left = p[n - 2:0:-1]
        q[1:] = right + left
        pc = np.zeros(n)
        pc[1:] = np.cumsum(right)
        U += pc[n - 1 - idx]
        pm = np.zeros(n)
        pm[1:] = np.cumsum(left)
        U += pm[idx]

    conv_r = np.convolve(h, q[::-1], mode='full')
    H_right = conv_r[n - 1:2 * n - 1]
    H_left = np.convolve(h, q, mode='full')[:n]
    grad = dx * (U - H_left - H_right)
    return val, m, grad


def _smooth_obj(x):
    val, _ = smooth_value(x, _G_N, _G_ALPHA)
    r = x.sum() - _G_TARGET
    return float(val + 0.5 * _G_PEN * r * r)


def _smooth_jac(x):
    val, _, g = smooth_grad(x, _G_N, _G_ALPHA)
    r = x.sum() - _G_TARGET
    return g + (2.0 * _G_PEN * r) * np.ones(_G_N)


def lbfgs_step(h, n, alpha, pen, maxiter):
    global _G_N, _G_ALPHA, _G_PEN, _G_TARGET, _G_BOUNDS
    _G_N = int(n)
    _G_ALPHA = float(alpha)
    _G_PEN = float(pen)
    _G_TARGET = 0.5 * _G_N
    _G_BOUNDS = [(0.0, 1.0)] * _G_N
    x = project_feasible(h, n)
    try:
        res = minimize(
            _smooth_obj, x, jac=_smooth_jac, method='L-BFGS-B',
            bounds=_G_BOUNDS,
            options={'maxiter': int(maxiter), 'ftol': 1e-10, 'gtol': 1e-6, 'maxls': 20}
        )
        out = res.x
    except Exception:
        out = x
    return project_feasible(out, n)


def momentum_descent(h, n, alpha, end, lr=1.0, beta=0.90):
    global _RNG
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    val, mF, grad = smooth_grad(h, n, alpha)
    prev = val
    mom = np.zeros(n)
    step = lr
    stall = 0
    while time.time() + 0.05 < end:
        c = mF * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        mom = beta * mom + grad
        h_new = project_feasible(h - step * mom, n)
        val, mF, grad = smooth_grad(h_new, n, alpha)
        if val < prev - 1e-11:
            step = min(step * 1.02, 50.0)
            stall = 0
        else:
            step = max(step * 0.97, 0.001)
            stall += 1
        prev = val
        h = h_new
        c = mF * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if stall > 260:
            noise = 0.006 + 0.004 * min(1.0, stall / 800.0)
            h = project_feasible(h + noise * _RNG.randn(n), n)
            val, mF, grad = smooth_grad(h, n, alpha)
            prev = val
            mom *= 0.25
            step = lr
            stall = 0
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

                best_c = c1

        if n <= 256 and time.time() < deadline - 70:
            h2 = random_h(n, secondary_kind % 4, _RNG)
            secondary_kind += 1
            q2 = max(4.0, 0.35 * q)
            end2 = min(deadline - 70.0, time.time() + q2)
            if end2 > time.time() + 0.2:
                h2, c2 = optimize_candidate(h2, n, end2, final=False)
                if c2 < best_c:
                    best_h = h2
                    best_n = n
                    best_c = c2

    if best_h is not None and time.time() < deadline - 8:
        endf = deadline - 2.0
        hf, cf = optimize_candidate(best_h, best_n, endf, final=True)
        if cf < best_c:
            best_h, best_n, best_c = hf, best_n, cf

    if best_h is not None and time.time() < deadline - 12:
        for k in range(3):
            if time.time() > deadline - 10:
                break
            noise = 0.004 + 0.0025 * k
            h = project_feasible(best_h + noise * _RNG.randn(best_n), best_n)
            c = compute_c5(h, best_n)
            if c < best_c:
                best_h, best_n, best_c = h, best_n, c
            end = min(deadline - 4.0, time.time() + 25.0)
            if c < best_c and end > time.time() + 1.0:
                h2, c2 = optimize_candidate(h, best_n, end, final=False)
                if c2 < best_c:
                    best_h, best_n, best_c = h2, best_n, c2

    if best_h is None:
        best_n = 64
        best_h = project_feasible(random_h(best_n, 0, _RNG), best_n)
        best_c = compute_c5(best_h, best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = float(compute_c5(best_h, best_n))
    return best_h, best_c, int(best_n)
