def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

    return base

def lse_obj_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_obj(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_obj_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def polish(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 5.0:
            break
        _stop_time = min(deadline - 3.0, now + stage_time)
        try:
            res = minimize(lbfgs_obj, work, method="L-BFGS-B", jac=True,
                           bounds=bounds, args=(n, float(beta), float(pen)),
                           callback=stop_callback,
                           options={"maxiter": 500000, "maxls": 30, "ftol": 1e-12, "gtol": 1e-7})
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
    return best, bc

def eval_swaps(chunk_arr, corr, n, M, allow_neutral):
    """Given precomputed arr (delta correlations for a chunk of swaps), find best swap index."""
    mx = chunk_arr.max(axis=1)
    if allow_neutral:
        idx = np.nonzero(mx <= M + 1e-12)[0]
    else:
        idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
    return idx, mx

def binary_local_search(h, n, deadline, mode="greedy", max_plateau_moves=200, rng=None):
    """Binary local search: greedy, plateau, or combined."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(corr.max()))
    best_h = h.copy()
    best_M = M
    plateau_used = 0

    while time.time() < deadline:
        base = compute_base_binary(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break

        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        best_i, best_j = -1, -1
        best_val = M + 2
        best_corr_new = None
        found_improve = False
        found_neutral = False
        neutral_candidates = []

        for start in range(0, zi.size, 2048):
            if time.time() > deadline:
                break
            end = min(start + 2048, zi.size)
            i = zi[start:end]
            j = oj[start:end]
            arr = base[i] - base[j] + corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            v1 = (d1 >= 0) & (d1 < 2 * n - 1)
            if v1.any():
                arr[v1, d1[v1]] += 1.0
            d2 = n - 1 + (i - j)
            v2 = (d2 >= 0) & (d2 < 2 * n - 1)
            if v2.any():
                arr[v2, d2[v2]] += 1.0
            mx = arr.max(axis=1)
            imp = np.nonzero(mx < M - 0.5)[0]
            if imp.size:
                loc = imp[int(np.argmin(mx[imp]))]
                if mx[loc] < best_val:
                    best_val = float(mx[loc])
                    best_i, best_j = int(i[loc]), int(j[loc])
                    best_corr_new = arr[loc].copy()
                    found_improve = True
            if mode != "greedy":
                neu = np.nonzero((mx >= M - 0.5) & (mx <= M + 0.5))[0]
                if neu.size and plateau_used < max_plateau_moves:
                    neutral_candidates.extend([(int(i[k]), int(j[k]), float(mx[k]), arr[k].copy()) for k in neu[:10]])

        if found_improve:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = best_corr_new
            M = int(round(corr.max()))
            plateau_used = 0
            if M < best_M:
                best_M = M
                best_h = h.copy()
        elif mode != "greedy" and neutral_candidates and plateau_used < max_plateau_moves:
            # Pick a random neutral move
            pick = neutral_candidates[int(rng.randint(len(neutral_candidates)))]
            h[pick[0]] = 1.0
            h[pick[1]] = 0.0
            corr = pick[3]
            M = int(round(corr.max()))
            plateau_used += 1
        else:
            break

    return best_h, best_M * (2.0 / float(n))

def sa_binary(h, n, deadline, rng, n_restarts=4):
    """Simulated annealing on binary with n/2 ones."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_M = int(round(np.max(np.correlate(h, 1.0 - h, mode="full"))))

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart == 0:
            cur = best_h.copy()
        else:
            cur = best_h.copy()
            k = max(2, n // 15)
            oi = np.nonzero(cur == 1)[0]
            zi = np.nonzero(cur == 0)[0]
            fo = rng.choice(oi, min(k, len(oi)), replace=False)
            fz = rng.choice(zi, min(k, len(zi)), replace=False)
            cur[fo] = 0.0
            cur[fz] = 1.0

        cur_corr = np.correlate(cur, 1.0 - cur, mode="full").astype(float)
        cur_M = int(round(cur_corr.max()))
        T = 3.0
        T_min = 0.05
        cooling = 0.9995
        max_iter = max(500, int(200 * (deadline - time.time())))

        for it in range(max_iter):
            if time.time() > deadline - 2:
                break
            oi = np.nonzero(cur == 1)[0]
            zi = np.nonzero(cur == 0)[0]
            i = int(rng.choice(zi))
            j = int(rng.choice(oi))
            cur[i] = 1.0
            cur[j] = 0.0
            new_corr = np.correlate(cur, 1.0 - cur, mode="full")
            new_M = int(round(new_corr.max()))
            delta = new_M - cur_M
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_M = new_M
                cur_corr = new_corr
                if cur_M < best_M:
                    best_M = cur_M
                    best_c = best_M * 2.0 / n
                    best_h = cur.copy()
            else:
                cur[i] = 0.0
                cur[j] = 1.0
            T = max(T_min, T * cooling)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1100.0
    end = t0 + max(30.0, total_budget - 8.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n0 = 400
        h0 = ((np.arange(n0) // (n0 // 4)) % 2).astype(float)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Continuous polish
    if time.time() < end - 300:
        hh, cc = polish(best_h, best_n, [50.0, 200.0, 1000.0, 5000.0], 1e4, end - 280, 50)
        if cc < best_c:
            best_c, best_h = cc, hh.copy()

    # Phase 2: Greedy binary
    if best_n % 2 == 0 and time.time() < end - 250:
        hb, cb = binary_local_search(best_h, best_n, end - 260, mode="greedy", rng=rng)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 3: Plateau binary search (key innovation)
    if best_n % 2 == 0 and time.time() < end - 200:
        hb, cb = binary_local_search(best_h, best_n, end - 210, mode="plateau",
                                     max_plateau_moves=500, rng=rng)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 4: Simulated annealing
    if best_n % 2 == 0 and time.time() < end - 150:
        sa_h, sa_c = sa_binary(best_h, best_n, end - 155, rng, n_restarts=4)
        if sa_c < best_c:
            best_c, best_h = sa_c, sa_h.copy()

    # Phase 5: Different sizes
    for nnew in (300, 500):
        if nnew % 2 != 0 or time.time() > end - 100:
            break
        x0 = np.linspace(0.0, 2.0, best_n)
        x1 = np.linspace(0.0, 2.0, nnew)
        hr = np.interp(x1, x0, best_h)
        hr = project_feasible(hr, nnew)
        if time.time() < end - 80:
            hh, cc = polish(hr, nnew, [200.0, 1000.0], 1e4, end - 75, 25)
            if cc < best_c:
                best_c, best_h, best_n = cc, hh.copy(), nnew
        if time.time() < end - 60:
            hb, cb2 = binary_local_search(hr, nnew, end - 55, mode="plateau",
                                          max_plateau_moves=200, rng=rng)
            if cb2 < best_c:
                best_c, best_h, best_n = cb2, hb.copy(), nnew

    # Phase 6: Final polish on best
    if time.time() < end - 20:
        hh, cc = polish(best_h, best_n, [2000.0, 10000.0], 1e4, end - 10, 15)
        if cc < best_c:
            best_c, best_h = cc, hh.copy()

    # Final binary check
    if best_n % 2 == 0:
        tb = top_n_binary(best_h, best_n)
        ct = compute_c5(tb, best_n)
        if ct < best_c:
            best_c, best_h = ct, tb.copy()
        hb, cb = binary_local_search(best_h, best_n, end - 3, mode="greedy", rng=rng)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
