```python
        j1s = max(0, -k)
        j1e = min(n, n-k)
        if j1e > j1s:
            G[j1s:j1e, m] += (1.0 - h[j1s+k:j1e+k])
        j2s = max(0, k)
        j2e = min(n, n+k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s-k:j2e-k]
    return G
```

Wait, building G is O(n²) which for n=400 is 160K entries. That's fine. But doing it repeatedly in the LP loop is wasteful since h changes each step.

For the coordinate descent, at each step j, I only need G[j,:], which I can compute in O(n). So coordinate descent is O(n²) per sweep (n coordinates × O(n) to compute the row and O(n) to solve the 1D problem).

Let me write the final algorithm:

```python
import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lp_step(h, n, eps, solver='ECOS'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    # Build gradient matrix
    G = np.zeros((n, 2*n-1))
    for m in range(2*n-1):
        k = m - (n-1)
        j1s = max(0, -k)
        j1e = min(n, n-k)
        if j1e > j1s:
            G[j1s:j1e, m] += (1.0 - h[j1s+k:j1e+k])
        j2s = max(0, k)
        j2e = min(n, n+k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s-k:j2e-k]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps, delta <= eps,
        delta >= -h, delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for sol in [solver, 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            break
        except:
            continue
    if delta.value is None:
        return h, mc
    d = delta.value
    best_h, best_mc = h.copy(), mc
    for lam in [1.0, 0.5, 0.25]:
        h_try = np.clip(h + lam * d, 0, 1)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.max(np.correlate(h_try, 1-h_try, mode='full')))
        if mc_try < best_mc - 1e-14:
            best_mc, best_h = mc_try, h_try
    return best_h, best_mc

def coord_descent(h, n, num_sweeps=5, tol=1e-10):
    """Coordinate descent: for each j, minimize max_k c_k by varying h[j]."""
    best_c5 = float(np.max(np.correlate(h, 1-h, mode='full')))
    for sweep in range(num_sweeps):
        improved = False
        for j in range(n):
            c = np.correlate(h, 1-h, mode='full')
            M = float(c.max())
            old_val = h[j]
            # Compute slope of c_m w.r.t. h[j] for all m
            slopes = np.zeros(2*n-1)
            for m in range(2*n-1):
                k = m - (n-1)
                s = 0.0
                if 0 <= j+k < n:
                    s += (1.0 - h[j+k])
                if 0 <= j-k < n:
                    s -= h[j-k]
                slopes[m] = s
            # Minimize max(c + slopes * t) over t in [-old_val, 1-old_val]
            # Use grid search + refinement
            lo, hi = -old_val, 1.0 - old_val
            best_t = 0.0
            best_obj = M
            # Grid search
            grid = np.linspace(lo, hi, 50)
            for t in grid:
                obj = float(np.max(c + slopes * t))
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            # Refine around best_t
            span = (hi - lo) / 50
            lo2 = max(lo, best_t - span)
            hi2 = min(hi, best_t + span)
            grid2 = np.linspace(lo2, hi2, 50)
            for t in grid2:
                obj = float(np.max(c + slopes * t))
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            h[j] = np.clip(old_val + best_t, 0, 1)
            if abs(best_t) > tol:
                improved = True
        h = project_feasible(h, n)
        new_c5 = float(np.max(np.correlate(h, 1-h, mode='full')))
        if new_c5 < best_c5:
            best_c5 = new_c5
        if not improved:
            break
    return h, best_c5
```

Hmm, the coordinate descent as written is O(n³) per sweep (n coordinates × O(n) for slopes × O(n) for grid eval). For n=400, that's 64M per sweep. In Python this might be slow.

Let me vectorize the slope computation. For a fixed j:
slopes[m] = (1-h[j+k]) * I(0≤j+k<n) - h[j-k] * I(0≤j-k<n) where k = m-(n-1)

This is essentially:
- First term: (1-h) shifted by k, valid where j+k is in [0,n)
- Second term: -h shifted by -k, valid where j-k is in [0,n)

For a fixed j, the valid range of k:
- First term: -j ≤ k ≤ n-1-j
- Second term: j-n+1 ≤ k ≤ j

In terms of m = k + (n-1):
- First term valid: (n-1)-j ≤ m ≤ (n-1)+(n-1-j) = 2n-2-j
- Second term valid: (n-1)-(n-1-j) = j ≤ m ≤ (n-1)+j

So:
slopes[m] for m in [n-1-j, 2n-2-j]: (1 - h[m-(n-1)+j]) = (1 - h[m+j-(n-1)])
slopes[m] for m in [j, n-1+j]: -h[m-(n-1)-j] = -h[m-j-(n-1)]

Let me verify: if m = n-1 (k=0):
- First term: 1-h[j+0] = 1-h[j] (valid since 0≤j<n) ✓
- Second term: -h[j-0] = -h[j] (valid since 0≤j<n) ✓
- Slope = 1-h[j] - h[j] = 1-2h[j]

Makes sense: c_0 = sum_i h[i](1-h[i]) = sum h[i] - sum h[i]². dc_0/dh[j] = 1-2h[j]. ✓

OK so for the vectorized version, I can precompute the slopes for all j at once using the G matrix. But building G is O(n²) = 160K for n=400, which is fast.

```python
def coord_descent_fast(h, n, num_sweeps=5):

            old_val = h[j]

            slopes = np.zeros(2*n-1)
            k_lo = max(0, -j)  # min k such that j+k >= 0
            k_hi = min(n-1, n-1-j)  # max k such that j+k < n
            k_lo2 = max(0, j-(n-1))  # min k such that j-k >= 0  => k <= j
            k_hi2 = min(n-1, j)  # max k such that j-k < n => k > j-n
            
            # First term: for k in [k_lo, k_hi], slope[m] += (1-h[j+k]) where m = k+(n-1)
            if k_hi >= k_lo:

                m_hi = k_hi + (n-1)

            # Second term: for k in [0, k_hi2], slope[m] -= h[j-k] where m = k+(n-1)
            if k_hi2 >= 0:
                m_lo2 = 0 + (n-1)
                m_hi2 = k_hi2 + (n-1)
                # h[j-k] for k=0..k_hi2 is h[j], h[j-1], ..., h[j-k_hi2]
                # That's h[j-k_hi2 : j+1] reversed
                slopes[m_lo2:m_hi2+1] -= h[j-k_hi2 : j+1][::-1]

            lo, hi = -old_val, 1.0 - old_val

            best_t = 0.0
            best_obj = M
            grid = np.linspace(lo, hi, 40)

            idx = np.argmin(objs)
            best_obj = objs[idx]
            best_t = grid[idx]
            h[j] = np.clip(old_val + best_t, 0, 1)
        h = project_feasible(h, n)
    return h
```

Wait, this is still O(n) per coordinate for the grid search (evaluating max over 2n-1 values for each of 40 grid points). Total per sweep: n * 40 * 2n = 80n² = 12.8M for n=400. In numpy this should be fast (vectorized).

Actually, the line `objs = np.max(c[None,:] + slopes[None,:] * grid[:,None], axis=1)` creates a (40, 2n-1) array and takes max along axis 1. For n=400, that's 40×799 = 32K elements. Very fast.

Total per sweep: 400 coordinates × 32K = 12.8M floating point ops. Should take ~0.01 seconds per sweep in NumPy. 

But wait, I'm recomputing c = np.correlate(h, 1-h, mode='full') at each coordinate j, which is O(n²) = 160K. So per sweep: 400 × 160K = 64M just for the correlations. That's more like 0.1-0.5 seconds per sweep.

I can optimize by not recomputing c from scratch. When I change h[j] by δ, the new c is:
c_new = c_old + slopes[j] * δ

So I can maintain c incrementally!

```python
def coord_descent_fast(h, n, num_sweeps=10):
    c = np.correlate(h, 1-h, mode='full').copy()
    for sweep in range(num_sweeps):
        for j in range(n):
            old_val = h[j]
            # Compute slope for coordinate j
            slopes = np.zeros(2*n-1)
            k_lo = 0  # min k such that j+k >= 0 => k >= -j, but k >= 0 for correlate
            # Actually k ranges from -(n-1) to (n-1) in the correlate
            # First term: (1-h[j+k]) valid for 0 <= j+k < n => -j <= k <= n-1-j
            k1_lo = max(-(n-1), -j)
            k1_hi = min(n-1, n-1-j)
            if k1_hi >= k1_lo:
                m_lo = k1_lo + (n-1)
                m_hi = k1_hi + (n-1)
                slopes[m_lo:m_hi+1] += (1.0 - h[j+k1_lo : j+k1_hi+1])
            
            # Second term: -h[j-k] valid for 0 <= j-k < n => j-(n-1) <= k <= j
            k2_lo = max(-(n-1), j-(n-1))
            k2_hi = min(n-1, j)
            if k2_hi >= k2_lo:
                m_lo = k2_lo + (n-1)
                m_hi = k2_hi + (n-1)
                # h[j-k] for k = k2_lo..k2_hi
                # j-k ranges from j-k2_lo down to j-k2_hi
                # So it's h[j-k2_hi : j-k2_lo+1] reversed
                # = h[j-k2_hi : j-k2_lo+1][::-1]
                # Length = k2_hi - k2_lo + 1
                idxs = j - np.arange(k2_lo, k2_hi+1)
                slopes[m_lo:m_hi+1] -= h[idxs]
            
            # Minimize max(c + slopes*t) over t in [-old_val, 1-old_val]
            lo, hi = -old_val, 1.0 - old_val
            if hi - lo < 1e-12:
                continue
            grid = np.linspace(lo, hi, 50)
            vals = c + slopes * grid[0]  # just to check
            # Vectorized: (50, 2n-1) matrix, take max along axis 1
            objs = np.max(c[None,:] + slopes[None,:] * grid[:,None], axis=1)
            idx = np.argmin(objs)
            best_t = grid[idx]
            
            if abs(best_t) > 1e-10:
                # Update c incrementally
                c += slopes * best_t
                h[j] = np.clip(old_val + best_t, 0, 1)
        
        # Project to restore sum constraint
        h = project_feasible(h, n)
        # Recompute c to avoid drift
        c = np.correlate(h, 1-h, mode='full')
    return h
```

This is O(n²) per sweep (dominated by the correlation computation at the end). For n=400: 160K per sweep. Very fast! We can do hundreds of sweeps.
