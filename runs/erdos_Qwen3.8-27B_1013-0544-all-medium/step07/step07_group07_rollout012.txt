    """Soft-max objective and its gradient."""
    dx = 2.0 / n
    c = compute_corr(h) * dx
    # Numerically stable soft-max
    m = c.max()
    unshifted = alpha * (c - m)
    unshifted -= unshifted.max()
    w = np.exp(unshifted)
    wsum = w.sum()
    obj = m + np.log(wsum) / alpha

        return obj, np.zeros(n)

        # -h[j-sv] contribution  
        j_start = max(0, sv)
        j_end = min(n, n + sv)
        if j_start < j_end:
            grad[j_start:j_end] -= wk * h[j_start - sv:j_end - sv]
    return obj, grad

    """Projected gradient descent with backtracking line search."""
    dx = 2.0 / n

    return best_h

- This is like a multiplicative weights algorithm

Actually, that's essentially what the soft-max does when alpha is large. Let me just focus on making the soft-max approach work well.

One more idea: **dual approach**. The primal problem is:
min max_k c[k] * dx  subject to 0≤h≤1, sum(h)=n/2

The dual involves finding weights p_k ≥ 0, Σp_k = 1 that maximize:
Σ_k p_k * (min over feasible h of Σ_k p_k * c_k(h))

This is a minimax problem. The inner minimization is linear in h (c_k is bilinear in h, but for fixed p, Σ p_k c_k is... let me check).

c_k(h) = Σ_i h[i]*(1-h[i+k]) = Σ_i h[i] - Σ_i h[i]*h[i+k]

So Σ_k p_k * c_k(h) = Σ_k p_k * Σ_i h[i] - Σ_k p_k * Σ_i h[i]*h[i+k]
= (Σ_i h[i]) * (Σ_k p_k) - Σ_k p_k * Σ_i h[i]*h[i+k]
= n/2 - Σ_i Σ_k p_k * h[i]*h[i+k]
= n/2 - h^T (Σ_k p_k * S_k) h

where S_k is the shift matrix for shift k.

So the dual is:
max_{p≥0, Σp=1} [n/2 - min_{0≤h≤1, sum(h)=n/2} h^T Q(p) h]

where Q(p) = Σ_k p_k * S_k.

This is a maximin problem. The inner minimization over h of h^T Q h subject to box constraints and sum constraint is a quadratic program, which can be solved efficiently!

But this might be overkill. Let me stick with the primal approach but make it more efficient.

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15:
                break
            h = np.clip(h + diff / rs * room, 0, 1)
        else:
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def soft_max_obj(h, n, alpha):
    c = compute_corr(h) * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    wsum = np.exp(sh).sum()
    return m + np.log(wsum) / alpha

def soft_max_grad(h, n, alpha):
    c = compute_corr(h) * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    wsum = w.sum()
    p = w / wsum  # probability distribution over shifts
    
    grad = np.zeros(n)
    shifts = np.arange(len(c)) - (n - 1)
    # Only process significant weights
    thresh = 1e-12
    for k in range(len(c)):
        if p[k] < thresh:
            continue
        sv = int(shifts[k])
        # d(c_k)/d(h[j]) = (1-h[j+sv]) - h[j-sv]
        # First term: (1-h[j+sv]) for j where j+sv ∈ [0,n-1]
        i0 = max(0, -sv)
        i1 = min(n, n - sv)
        if i0 < i1:
            grad[i0:i1] += p[k] * (1.0 - h[i0+sv:i1+sv])
        # Second term: -h[j-sv] for j where j-sv ∈ [0,n-1]
        j0 = max(0, sv)
        j1 = min(n, n + sv)
        if j0 < j1:
            grad[j0:j1] -= p[k] * h[j0-sv:j1-sv]
    return grad

def projected_grad_descent(h, n, alpha, max_iter=3000, lr_init=0.05):
    best_h = h.copy()
    best_obj = soft_max_obj(h, n, alpha)
    lr = lr_init
    for it in range(max_iter):
        obj = best_obj
        grad = soft_max_grad(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        t = lr
        improved = False
        for _ in range(40):
            t *= 0.5
            new_h = project_feasible(h - t * grad, n)
            new_obj = soft_max_obj(new_h, n, alpha)
            if new_obj < obj - 1e-10:
                h = new_h
                best_h = new_h.copy()
                best_obj = new_obj
                lr = t * 1.3
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-12:
                break
    return best_h, compute_c5(best_h, n)

def binary_local_search(h, n, deadline, max_sweeps=100):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n//2]] = 1.0
    best_h = h_b.copy()
    best_c = compute_c5(h_b, n)
    ones = set(np.where(h_b == 1)[0].tolist())
    zeros = set(np.where(h_b == 0)[0].tolist())
    
    dx = 2.0 / n
    corr = compute_corr(h_b).astype(float)
    cur_c = corr.max() * dx
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try each zero position
        z_list = sorted(zeros)
        for i in z_list:
            if time.time() > deadline:
                break
            # Try each one position
            o_list = sorted(ones)
            for j in o_list:
                # Compute delta in correlation
                delta = np.zeros(len(corr))
                m_range = np.arange(-(n-1), n)
                # h[i] goes from 0 to 1
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h_b[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                # h[j] goes from 1 to 0
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h_b[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                # Self-interaction correction
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h_b[i] = 1.0
                    h_b[j] = 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.add(i)
                    zeros.add(j)
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h_b.copy()
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter, ftol=1e-16):
    def obj(x):
        return soft_max_obj(x, n, alpha)
    def grad(x):
        return soft_max_grad(x, n, alpha)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

Wait, I realize there might be an issue with my projected gradient method. The soft-max objective is:
f(h) = max(c) + (1/α) * log(Σ exp(α(c_k - max(c))))

This is a smooth approximation of max(c). As α → ∞, f → max(c). The gradient is:
∇f = Σ_k p_k * ∇c_k where p_k = exp(α(c_k - max(c))) / Σ exp(α(c_k - max(c)))

This is correct. The issue is that for large α, the distribution p becomes very concentrated on the max, and the gradient becomes essentially the subgradient of the max function. This is good for "targeting" the maximum, but the line search might struggle because the objective is nearly flat in most directions.

One trick: use the **top-K subgradient** instead of the full soft-max. This means:
- Find the top-K shifts with highest correlation
- Use equal weights on those K shifts
- This gives a "sharper" gradient that directly targets the worst shifts

        sv = int(k - (n-1))
        i0 = max(0, -sv)
        i1 = min(n, n - sv)
        if i0 < i1:

        j1 = min(n, n + sv)

FISTA:
- y_k = x_k + (t_{k-1}-1)/(t_k) * (x_k - x_{k-1})
- x_{k+1} = proj(y_k - ∇f(y_k)/L)
- t_{k+1} = (1+√(1+4t_k²))/2

    y = h.copy()
    t = 1.0

    best_c = compute_c5(h, n)

        grad = soft_max_grad(y, n, alpha)

        x_new = project_feasible(y - grad / L, n)

        t = t_new
        x = x_new
        
        # Track best
        c = compute_c5(x, n)
        if c < best_c:
            best_c = c
            best_h = x.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15:
                break
            h = np.clip(h + diff / rs * room, 0, 1)
        else:
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_corr(h) * dx
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    wsum = w.sum()
    p = w / wsum
    grad = np.zeros(n)
    shifts = np.arange(len(c)) - (n - 1)
    thresh = 1e-13
    for k in range(len(c)):
        if p[k] < thresh:
            continue
        sv = int(shifts[k])
        i0 = max(0, -sv)
        i1 = min(n, n - sv)
        if i0 < i1:
            grad[i0:i1] += p[k] * (1.0 - h[i0 + sv:i1 + sv])
        j0 = max(0, sv)
        j1 = min(n, n + sv)
        if j0 < j1:
            grad[j0:j1] -= p[k] * h[j0 - sv:j1 - sv]
    return grad

def softmax_obj(h, n, alpha):
    c = compute_corr(h) * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    return m + np.log(np.exp(sh).sum()) / alpha

def fist(h, n, alpha, max_iter, lr_init):
    L = lr_init
    x = h.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for k in range(max_iter):
        grad = softmax_grad(y, n, alpha)
        x_new = project_feasible(y - grad / L, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        t = t_new
        x = x_new
        if k % 50 == 0 or k == max_iter - 1:
            c = compute_c5(x, n)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter):
    def obj(x):
        return softmax_obj(x, n, alpha)
    def grad(x):
        return softmax_grad(x, n, alpha)
    def cons(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_push(h, n, c_target, maxiter):
    def obj(x):
        c = compute_corr(x) * (2.0 / len(x))
        excess = np.maximum(c - c_target, 0)
        return 0.5 * np.sum(excess * excess)
    def grad(x):
        nn = len(x)
        dx = 2.0 / nn
        c = compute_corr(x) * dx
        mask = c > c_target
        g = np.zeros(nn)
        shifts = np.arange(len(c)) - (nn - 1)
        for k in np.where(mask)[0]:
            sv = int(shifts[k])
            i0 = max(0, -sv)
            i1 = min(nn, nn - sv)
            if i0 < i1:
                g[i0:i1] += (c[k] - c_target) * (1.0 - x[i0 + sv:i1 + sv])
            j0 = max(0, sv)
            j1 = min(nn, nn + sv)
            if j0 < j1:
                g[j0:j1] -= (c[k] - c_target) * x[j0 - sv:j1 - sv]
        return g
    def cons(x):
        return x.sum() - len(x) / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_random_search(h, n, deadline, rng):

    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    best_h = h_b.copy()
    best_c = compute_c5(h_b, n)
    cur_c = best_c
    corr = compute_corr(h_b).astype(float)
    m_range = np.arange(-(n - 1), n)
    for _ in range(2000000):
        if time.time() > deadline:

        i = rng.randint(n)
        j = rng.randint(n)
        if h_b[i] == 1 or h_b[j] == 0:
            continue
        delta = np.zeros(len(corr))
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += (1.0 - h_b[im[msk]])
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h_b[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= (1.0 - h_b[jm[msk]])
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h_b[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < len(delta):
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < len(delta):
            delta[k2] += 1.0
        nc = (corr + delta).max() * (2.0 / n)
        if nc < cur_c - 1e-15:
            h_b[i] = 1.0
            h_b[j] = 0.0
            corr += delta
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
    return best_h, best_c
