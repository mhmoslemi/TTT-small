from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float))
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        d = target - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = m.sum()
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def grad_smooth_max(h, n, alpha):
    c = compute_c(h, n)
    m = c.max()
    sh = alpha * (c - m)
    w = np.exp(sh)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= sw
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    omh = 1.0 - h
    # Process each active lag
    for idx in np.nonzero(w > 1e-12)[0]:
        k = idx - off
        wi = w[idx] * dx
        if k > 0:
            m_ = n - k
            # g[i] for i in [0, m_): -h[i+k] * wi
            g[:m_] += -wi * h[k:]
            # g[i] for i in [k, n): (1-h[i-k]) * wi
            g[k:] += wi * omh[:m_]
        elif k < 0:
            m2 = -k
            m_ = n - m2
            # g[i] for i in [0, m_): (1-h[i+m2]) * wi
            g[:m_] += wi * omh[m2:]
            # g[i] for i in [m2, n): -h[i-m2] * wi
            g[m2:] += -wi * h[:m_]
        else:
            g += wi * (1.0 - 2.0 * h)
    return g, m

def subgrad_optimize(h, n, deadline, alpha=500, n_iter=5000):
    h = project_box_sum(h, n)
    bestc = compute_c(h, n).max()
    besth = h.copy()
    t = 1.0 / alpha  # initial step size
    for it in range(n_iter):

        v = c.max()

        h = h2

The maximum of the changed c values is a piecewise linear function of h[i] (since each c[lag] is linear in h[i] when we fix the sum). So we want to find h[i] that minimizes the maximum of a set of linear functions, i.e., a 1D LP.

Actually it's even simpler: the changed correlation values are linear in h[i] (given S is fixed). So the max of the changed values is a convex piecewise linear function of h[i], and we can find its minimum by checking the breakpoints.

But the unchanged correlation values might still be the maximum. So we need: min_{h[i]} max(max_unchanged, max_changed(h[i])).

The max_changed(h[i]) is convex piecewise linear, and max_unchanged is a constant. So the overall max is convex piecewise linear, and we can find its minimum.

This is what the previous code's pair_search does, but it uses a grid + zoom approach. Let me make it exact by finding the breakpoints.

        # For lag k: c[off+k] involves h[i] iff i+k in [0,n-1] or i in [0,n-1] and (1-h) at position i+k
        # Let me just compute the delta for each affected correlation
        # Affected correlations: those where i or j appears

        #   k > 0: i >= k (h[i] is first factor, need i-k < n) AND i < n-k... 
        #   Actually it's simpler: i appears if there exists j' such that {i, j'} = {a, a+k} for some a
        #   i appears as first factor (position a+k) if a = i-k, need 0 <= i-k < n-k, i.e., k <= i <= n-1
        #   i appears as second factor (position a) if a = i, need 0 <= i < n and i+k < n, i.e., 0 <= i < n-k
        #   For k < 0 (k=-m): i appears as first factor if a = i+m, need 0 <= i+m < n-m... 
        # This is getting complicated. Let me just recompute c after the change.
        
        # Grid evaluation
        ngrid = 32
        xs = np.linspace(lo, hi, ngrid)
        best_v = bestc
        best_x = x0
        for xv in xs:

            h[j] = S - xv
            c_new = compute_c(h, n)
            v = float(c_new.max())
            if v < best_v:
                best_v = v
                best_x = xv
            c = c_new  # keep updated

            lo2 = max(lo, best_x - margin)
            hi2 = min(hi, best_x + margin)
            xs2 = np.linspace(lo2, hi2, 32)
            for xv in xs2:

            h[j] = S - best_x
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
                noimp = 0
            else:
                noimp += 1
        else:

            h[j] = y0
            noimp += 1
    
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

This is not exactly the continuous integral - it's a Riemann sum approximation. The "valid i" constraint means we're integrating over the overlap of [0,2] and [-k, 2-k].

For the continuous problem, if h is periodic with period 2, then:
∫₀² h(x)(1-h(x+k)) dx = ∫₀² h(x) dx - ∫₀² h(x)h(x+k) dx = 1 - ⟨h, T_k h⟩

where T_k is the shift operator. If h has Fourier series h(x) = Σ c_n e^{2πinx/2} = Σ c_n e^{πinx}:
⟨h, T_k h⟩ = Σ |c_n|² e^{πink}

To minimize this, we want to maximize min_k Σ |c_n|² e^{πink}.

With the constraint ∫h = 1, which means c_0 = 1/2 (the DC component).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    t = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        d = t - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = m.sum()
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def fast_pair_search(h, n, deadline, seed=42):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()

def subgrad_polish(h, n, deadline, alpha=1000):
    h = project_box_sum(h, n)
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    off = n - 1
    dx = 2.0 / n
    for it in range(20000):
        if time.time() > deadline:
            break
        t = 4.0 / (alpha * np.sqrt(it + 1.0))
        c = compute_c(h, n)
        m = c.max()
        sh = alpha * (c - m)
        sh -= sh.max()
        w = np.exp(sh)
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= sw
        g = np.zeros(n)
        omh = 1.0 - h
        for idx in np.nonzero(w > 1e-8)[0]:
            k = idx - off
            wi = w[idx] * dx
            if k > 0:
                m_ = n - k
                g[:m_] += -wi * h[k:]
                g[k:] += wi * omh[:m_]
            elif k < 0:
                m2 = -k
                m_ = n - m2
                g[:m_] += wi * omh[m2:]
                g[m2:] += -wi * h[:m_]
            else:
                g += wi * (1.0 - 2.0 * h)
        h2 = project_box_sum(h - t * g, n)
        v = float(compute_c(h2, n).max())

        h = h2

    if c_cur < c_best:

    return h_best, c_best, n_best
