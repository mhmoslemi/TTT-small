def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_obj_grad(x, n, alpha, mu):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1-x, mode='full') * dx
    c_shifted = c - c.max()
    exps = np.exp(alpha * c_shifted)
    w = exps / exps.sum()
    f = c.max() + np.log(exps.sum()) / alpha + mu * (x.sum() - target)**2
    # gradient
    v = 1 - x
    w_pos = w[n-1:]
    term_plus = np.zeros(n)
    term_minus = np.zeros(n)
    for k in range(1, n):
        term_plus[:n-k] += w_pos[k] * v[k:]
        term_minus[k:] += w_pos[k] * x[:n-k]
    grad = dx * (w_pos[0] * (1-2*x) + 2*(term_plus - term_minus)) + 2*mu*(x.sum()-target)
    return f, grad

    bounds = [(0.0, 1.0)] * n

                      h, jac=lambda x: lse_obj_grad(x, n, alpha, mu)[1],
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

        if not improved:
            break
    return best_h, best_c

    target = n / 2.0
