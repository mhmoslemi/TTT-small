        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_G(h, n):
    nc = 2 * n - 1
    G = np.zeros((n, nc))
    omh = 1.0 - h
    for m in range(nc):
        k = m - (n - 1)
        js, je = max(0, -k), min(n, n - k)
        if je > js:
            G[js:je, m] = omh[js + k:je + k]
        js2, je2 = max(0, k), min(n, n + k)
        if je2 > js2:
            G[js2:je2, m] -= h[js2 - k:je2 - k]
    return G

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = build_G(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    cons = [C + G.T @ delta <= t, cp.abs(delta) <= eps,
            delta >= -h, delta <= 1.0 - h, cp.sum(delta) == 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    for sol in ['GLOP', 'CLARABEL', 'GLPK']:
        try:
            prob.solve(solver=sol)
            if delta.value is not None:
                break
        except Exception:
            pass
    if delta.value is None:
        return h, mc
    d = delta.value
    bh, bmc = h.copy(), mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2]:
        ht = project_feasible(h + lam * d, n)
        mct = float(np.correlate(ht, 1.0 - ht, mode='full').max())
        if mct < bmc:
            bmc, bh = mct, ht
    return bh, bmc

def directional_perturb(h, n, scale, rng):
    C = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(C)[::-1][:3]
    kstar = int(top[int(rng.randint(len(top)))])
    k = kstar - (n - 1)
    g = np.zeros(n)
    js, je = max(0, -k), min(n, n - k)
    if je > js:
        g[js:je] = 1.0 - h[js + k:je + k]
    js2, je2 = max(0, k), min(n, n + k)
    if je2 > js2:
        g[js2:je2] -= h[js2 - k:je2 - k]
    gn = np.linalg.norm(g)
    if gn < 1e-10:
        return h
    return project_feasible(h - scale * g / gn + 0.01 * rng.randn(n), n)

def smooth_polish(h, n, alpha, maxiter=600):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        s = alpha * (g - m)
        sm = s.max()
        return float(m + sm / alpha + np.log(np.exp(s - sm).sum()) / alpha)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_mc = float(np.correlate(h, 1 - h, mode='full').max())
    best_h = h.copy()
    best_c5 = best_mc * 2.0 / n

    # Phase 1: LP optimization with aggressive epsilon reduction
    p1dl = t0 + 0.38 * total
    eps = 0.07
    no_imp = 0
    while time.time() < p1dl:
        h, mc = lp_step(h, n, eps)
        if mc < best_mc - 1e-13:
            best_mc, best_h, best_c5 = mc, h.copy(), mc * 2.0 / n
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                eps *= 0.5
                no_imp = 0
                if eps < 0.0008:
                    break
    best_h = project_feasible(best_h, n)

    # Phase 2: Diverse restarts (directional + random + sinusoidal)
    p2dl = t0 + 0.67 * total
    x = np.linspace(0, 2, n)
    for trial in range(200):
        if time.time() > p2dl:
            break
        r = rng.rand()
        if r < 0.30:
            h = directional_perturb(best_h, n, 0.03 + 0.04 * rng.rand(), rng)
        elif r < 0.55:
            h = project_feasible(best_h + (0.015 + 0.04 * rng.rand()) * rng.randn(n), n)
        elif r < 0.80:
            f = 1.0 + 12.0 * rng.rand()
            ph = 2.0 * np.pi * rng.rand()
            h = project_feasible(best_h + (0.025 + 0.05 * rng.rand()) * np.sin(2.0 * np.pi * f * x / 2.0 + ph), n)
        else:
            nb = 4 + 40 * int(rng.rand())
            bv = rng.randn(nb) * 0.025
            h = project_feasible(best_h + np.interp(x, np.linspace(0, 2, nb + 1), np.concatenate([bv, [0]])), n)
        mc_h = float(np.correlate(h, 1 - h, mode='full').max())
        if mc_h > best_mc * 1.15:
            continue
        er = 0.05
        for _ in range(60):
            if time.time() > p2dl:
                break
            h, mc = lp_step(h, n, er)
            if mc < best_mc - 1e-13:
                best_mc, best_h, best_c5 = mc, h.copy(), mc * 2.0 / n
            er = max(er * 0.98, 0.003)
    best_h = project_feasible(best_h, n)

    # Phase 3: Multi-resolution (finer grids)
    p3dl = t0 + 0.82 * total
    for nh in [500, 600, 800]:
        if time.time() > p3dl - 60:
            break
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, np.linspace(0, 2, n), best_h), nh)
        mc_hh = float(np.correlate(hh, 1 - hh, mode='full').max())
        if mc_hh * 2.0 / nh > best_c5 + 0.005:
            continue
        er = 0.04
        for _ in range(120):
            if time.time() > p3dl - 30:
                break
            hh, mc = lp_step(hh, nh, er)
            if mc * 2.0 / nh < best_c5 - 1e-13:
                best_mc, best_h, best_c5, n = mc, hh.copy(), mc * 2.0 / nh, nh
            er = max(er * 0.98, 0.003)
    best_h = project_feasible(best_h, n)

    # Phase 4: Smooth polish via SLSQP
    p4dl = t0 + 0.92 * total
    for alpha in [3000, 15000, 80000]:
        if time.time() > p4dl:
            break
        h2 = smooth_polish(best_h, n, alpha, 1000)
        mc2 = float(np.correlate(h2, 1 - h2, mode='full').max())
        if mc2 < best_mc - 1e-14:
            best_mc, best_h, best_c5 = mc2, h2.copy(), mc2 * 2.0 / n

    # Phase 5: Final deep LP polish
    h = best_h.copy()
    ef = 0.002
    while time.time() < dl:
        h, mc = lp_step(h, n, ef)
        if mc < best_mc - 1e-14:
            best_mc, best_h, best_c5 = mc, h.copy(), mc * 2.0 / n
        ef = max(ef * 0.993, 0.0001)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
