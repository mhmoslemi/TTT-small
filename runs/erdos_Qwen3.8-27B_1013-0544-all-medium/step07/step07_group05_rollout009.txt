        best_t = 0.0
        best_c5 = cur
        for alpha in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            hh = project_box_sum(h + alpha * gp, target)
            c5 = erdos_c5(hh, n)
            if c5 < best_c5:
                best_c5 = c5
                best_t = alpha
        if best_t > 0 and best_c5 < cur - 1e-15:
            h = project_box_sum(h + best_t * gp, target)
            cur = best_c5
        else:
            break
    return h, cur

def pair_obj_fn(delta, h_i, h_j, h_full, n):
    a = h_i + delta
    b = h_j - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h_full.copy()
    hh[0] = a
    hh[1] = b
    c = np.correlate(hh, 1.0 - hh, mode='full')
    return float(np.max(c)) * (2.0 / n)

def pair_polish(h, n, deadline, seed):
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    rng = np.random.RandomState(int(seed))
    m = min(50, n)
    score = np.minimum(h, 1.0 - h)
    order = np.argsort(score)[::-1]
    top = order[:m]
    dx = 2.0 / n
    for sweep in range(5):
        if time.time() > deadline - 10.0:
            break
        improved = False
        pairs = []
        tlen = len(top)
        for a_i in range(tlen):
            for b_i in range(a_i + 1, tlen):
                pairs.append((int(top[a_i]), int(top[b_i])))
        if not pairs:
            break
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 10.0:
                break
            hi0, hj0 = h[i], h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-13:
                continue
            best_d = 0.0
            best_c = cur
            for frac in np.linspace(-1.0, 1.0, 9):
                d = frac * 0.5 * (hi - lo)
                a_v, b_v = hi0 + d, hj0 - d
                if a_v < 0 or a_v > 1 or b_v < 0 or b_v > 1:
                    continue
                hh = h.copy()
                hh[i] = a_v
                hh[j] = b_v
                c = erdos_c5(hh, n)
                if c < best_c:
                    best_c = c
                    best_d = d
            if best_d != 0.0:
                a_lo = lo
                a_hi = hi
                for _ in range(30):
                    if time.time() > deadline - 10.0:
                        break
                    m1 = a_lo + 0.382 * (a_hi - a_lo)
                    m2 = a_lo + 0.618 * (a_hi - a_lo)
                    a1, b1 = hi0 + m1, hj0 - m1
                    a2, b2 = hi0 + m2, hj0 - m2
                    f1 = 1e100 if (a1 < 0 or a1 > 1 or b1 < 0 or b1 > 1) else erdos_c5(np.concatenate(([a1, b1], h[np.arange(n) != i] if False else h[[x for x in range(n) if x not in (i,j)]])), n)
                    f2 = 1e100 if (a2 < 0 or a2 > 1 or b2 < 0 or b2 > 1) else 1e100
                    hh1 = h.copy(); hh1[i] = a1; hh1[j] = b1
                    f1 = erdos_c5(hh1, n) if (0 <= a1 <= 1 and 0 <= b1 <= 1) else 1e100
                    hh2 = h.copy(); hh2[i] = a2; hh2[j] = b2
                    f2 = erdos_c5(hh2, n) if (0 <= a2 <= 1 and 0 <= b2 <= 1) else 1e100
                    if f1 < f2:
                        a_hi = m2
                    else:
                        a_lo = m1
                    if a_hi - a_lo < 1e-10:
                        break
                d_best = 0.5 * (a_lo + a_hi)
                a_v, b_v = hi0 + d_best, hj0 - d_best
                if 0 <= a_v <= 1 and 0 <= b_v <= 1:
                    hh = h.copy()
                    hh[i] = a_v
                    hh[j] = b_v
                    c = erdos_c5(hh, n)
                    if c < best_c:
                        best_c = c
                        best_d = d_best
            if best_c < cur - 1e-15:
                h[i] += best_d
                h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)
                h = project_box_sum(h, 0.5 * n)
                cur = erdos_c5(h, n)
                improved = True
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n < 2:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    # Phase 1: LSE with moderate beta (coarse optimization)
    end1 = t0 + 0.25 * total
    betas1 = [200.0, 1000.0, 5000.0, 20000.0]
    iters1 = [80, 70, 60, 50]
    for beta, mx in zip(betas1, iters1):
        if time.time() > end1 - 10.0:
            break
        cand, c = refine_lbfgs(best_h, beta, 1e4, mx, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 2: LSE with high beta (fine optimization)
    end2 = t0 + 0.50 * total
    betas2 = [100000.0, 500000.0, 2000000.0, 8000000.0]
    iters2 = [50, 45, 40, 35]
    for beta, mx in zip(betas2, iters2):
        if time.time() > end2 - 10.0:
            break
        cand, c = refine_lbfgs(best_h, beta, 1e4, mx, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 3: Direct max reduction
    end3 = t0 + 0.60 * total
    if time.time() < end3 - 5.0:
        cand, c = direct_polish(best_h, best_n, end3)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 4: Try nearby n values
    end4 = t0 + 0.82 * total
    n_candidates = [350, 380, 420, 450, 500]
    for nh in n_candidates:
        if time.time() > end4 - 30.0:
            break
        hh = resample(best_h, best_n, nh)
        hh, c = refine_lbfgs(hh, 50000.0, 1e4, 50, deadline)
        hh, c = refine_lbfgs(hh, 2000000.0, 1e4, 35, deadline)
        if c < best_c - 1e-15:
            best_h = hh.copy()
            best_n = nh
            best_c = c
            if time.time() < end4 - 15.0:
                hh, c = direct_polish(hh, nh, deadline)
                if c < best_c - 1e-15:
                    best_h = hh.copy()
                    best_c = c

    # Phase 5: Pair polish on best
    end5 = deadline - 40.0
    if time.time() < end5 - 10.0:
        best_h, best_c = pair_polish(best_h, best_n, end5, seed)
        best_h = project_box_sum(best_h, 0.5 * best_n)
        best_c = erdos_c5(best_h, best_n)

    # Phase 6: Final direct polish
    if time.time() < deadline - 15.0:
        cand, c = direct_polish(best_h, best_n, deadline - 5.0)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 7: One last pair polish if time
    if time.time() < deadline - 8.0:
        cand, c = pair_polish(best_h, best_n, deadline - 2.0, seed + 99)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
