def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v


def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0:
        return m
    return m + np.log(s) / alpha


def lse_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    return q


def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            s = k
            m = n - s
            g[s:n] += w * (1.0 - x[0:m])
            g[:m] -= w * x[s:n]
        elif k < 0:
            s = -k
            m = n - s
            g[:m] += w * (1.0 - x[s:n])
            g[s:n] -= w * x[0:m]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pair_eval_fast(h, c_base, i, j, xs, n, x0, y0, S):

    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i + n] += ax1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += ax2[:, None] * h[None, :]
    delta[:, j:j + n] += ax2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += ax1[:, None] * h[None, :]
    idx0 = off
    base_self = ax1 * (1.0 - x0) + x0 * ax2 + ax2 * (1.0 - y0) + y0 * ax1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, idx0] += exact_self - base_self
    d = j - i
    idxp = off - d
    base_p = ax1 * (1.0 - y0) + x0 * ax1
    exact_p = xs * (1.0 - yv) - x0 * (1.0 - y0)
    delta[:, idxp] += exact_p - base_p
    idxm = off + d
    base_m = ax2 * (1.0 - x0) + y0 * ax2
    exact_m = yv * (1.0 - xs) - y0 * (1.0 - x0)
    delta[:, idxm] += exact_m - base_m
    return np.max(c_base[None, :] + dx * delta, axis=1)

            noimp = 0
        else:
            noimp += 1
            if noimp > 300:
                break
        thr = curmax - 0.005
        active = np.where(c >= thr)[0]
        if len(active) > n_active:

            active = idx
        g = np.zeros(n)
        dx = 2.0 / n
        off = n - 1
        for aidx in active:
            k = aidx - off
            if k > 0:
                s = k
                m = n - s
                g[s:n] += (1.0 - h[0:m])

            elif k < 0:

                g[s:n] -= h[0:m]
            else:

        g *= dx / len(active)
        v = mom * v + g
        h = project_box_sum(h - lr * v, n)

    return besth, bestc

            w = np.maximum(pos, neg) + 1e-12
            w = w / w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search(h, c, i, j, n)
        if res is not None:
            x, y, _ = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)


def optimize_n(h, n, seed, deadline):
    h = project_box_sum(h, n)
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    work = h.copy()
    workc = bestc
    if n < 4 or time.time() > deadline:
        return besth, bestc

    phases = (
        (15.0, 50.0, 400),
        (50.0, 200.0, 350),
        (200.0, 1000.0, 300),
        (1000.0, 10000.0, 250),
        (5000.0, 100000.0, 200),
        (20000.0, 500000.0, 150),
    )
    for alpha, mu, maxit in phases:
        if time.time() > deadline - 10:
            break
        h2 = lbfgsb_phase(work, n, alpha, mu, maxit)
        h2 = project_box_sum(h2, n)
        v2 = float(compute_c(h2, n).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
        if v2 <= workc + 1e-7:
            work = h2
            workc = v2
        else:
            work = besth.copy()
            workc = bestc

    h2 = binary_push(work, n)
    v2 = float(compute_c(h2, n).max())
    if v2 < bestc:
        bestc = v2
        besth = h2.copy()
        work = h2.copy()
        workc = v2
    else:
        work = besth.copy()
        workc = bestc

    sg_dl = deadline - 40
    if time.time() < sg_dl:
        h3, v3 = subgradient_momentum(work, n, sg_dl, lr_start=0.015, decay=0.99, mom=0.6, n_active=3)
        if v3 < bestc:
            bestc = v3
            besth = h3.copy()
        work = besth.copy()
        workc = bestc

    ph = lbfgsb_phase(work, n, 5000.0, 100000.0, 200)
    ph = project_box_sum(ph, n)
    vp = float(compute_c(ph, n).max())
    if vp < bestc:
        bestc = vp
        besth = ph.copy()

    h2 = binary_push(besth, n, thresh=0.04)
    v2 = float(compute_c(h2, n).max())
    if v2 < bestc:
        bestc = v2
        besth = h2.copy()

    if time.time() < deadline - 5:
        besth, bestc = local_search(besth, deadline, seed=seed)

    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.49 * np.sin(3.0 * np.pi * x)
        hp = project_box_sum(hp + 0.02 * rng.randn(n0), n0)

    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = float(compute_c(hp, n0).max())
    bestn = n0

    n_list = [n0]
    for m in (300, 350, 400, 450, 512):
        if m not in n_list and 16 <= m <= 700:
            n_list.append(m)

    L = len(n_list)
    if L == 1:
        fr = [0.85]
    else:
        fr = [0.40]
        rem = 0.42 / (L - 1)
        for _ in range(L - 1):
            fr.append(rem)

    span = max(1.0, deadline - t0)
    cum = 0.0

    for k, nh in enumerate(n_list):
        if time.time() > deadline - 20:
            break
        cum += fr[k]
        dl_phase = t0 + span * min(0.88, cum) - 5
        if dl_phase > deadline - 20:
            dl_phase = deadline - 20
        if dl_phase < time.time() + 8:
            continue

        if k == 0:
            hinit = besth.copy()
        else:
            hinit = interpolate_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)

        hopt, copt = optimize_n(hinit, nh, seed + 13 * k, dl_phase)
        if copt < bestc:
            bestc = copt
            besth = hopt.copy()
            bestn = nh

    if time.time() < deadline - 10:
        hfin, cfin = slsqp_phase(besth, bestn, 3000, 120)
        hfin = project_box_sum(hfin, bestn)
        cfin = float(compute_c(hfin, bestn).max())
        if cfin < bestc:
            bestc = cfin
            besth = hfin.copy()

    if time.time() < deadline - 5:
        besth, bestc = local_search(besth, bestn, deadline, seed=seed + 777)

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
