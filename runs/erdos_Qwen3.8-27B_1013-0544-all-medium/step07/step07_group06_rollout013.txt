from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

    h = np.asarray(h, dtype=float).ravel()

    hb = np.zeros(n)
    if target > 0:

        hb[idx] = 1.0

            zeros_list[ib] = a

def lse_grad(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    p = e / s
    F = m + np.log(s) / beta
    
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(P, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_obj(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    F, g = lse_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

    best = work.copy()

            break
        stage_dl = min(deadline - 3, time.time() + max(2, min(stage_time, 0.4 * max(2, deadline - time.time()))))
        _stop_time = stage_dl

                          bounds=bounds, args=(n, float(beta), float(pen)),

    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:

            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

    h = np.asarray(h, dtype=float).ravel()
    target = n // 2
    hb = np.zeros(n)
    if target > 0:
        idx = np.argsort(-h)[:target]
        hb[idx] = 1.0

def lse_grad(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    p = e / s
    F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(p, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_obj(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    F, g = lse_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(x)
    return float(F), g

def polish(h, n, betas, deadline, stage_time, pen=1e4):

    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 5:
            break

        try:
            res = minimize(lbfgs_obj, work, method="L-BFGS-B", jac=True,
                          bounds=bounds, args=(n, float(beta), float(pen)),
                          callback=stop_cb,

            x = res.x if res is not None and hasattr(res, "x") else work

            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:

            best = work.copy()
    return best, bc

    M = int(round(float(corr.max())))
    besth = h.copy()
    bestM = M

        v = 1.0 - h
        found = False
        ones_idx = np.nonzero(h == 1)[0]
