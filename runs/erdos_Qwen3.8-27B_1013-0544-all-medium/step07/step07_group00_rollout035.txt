import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def compute_all_lags(h, n):
    """Compute c_m for all lags m = -(n-1), ..., n-1"""
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_max(h, n, lags):

def fw_linear_min(s, n):
    """Solve min s.v s.t. 0<=v<=1, sum v = n/2.
    Optimal: v=1 for the n/2 smallest s components."""
    k = n // 2
    # Get indices of k smallest
    idx = np.argpartition(s, k)[:k]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def frank_wolfe(h, n, deadline, max_iter=5000):
    """Frank-Wolfe method for minimizing max overlap."""
    h = h.copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for t in range(max_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_val = float(c.max())
        
        # Find active lags
        c_max = c.max()
        tol = max(1e-8, 0.01 * c_max)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_subgrad(h, n, m):
    g = np.zeros(n)
    if m >= 0:
        if m < n:
            j1 = np.arange(m, n)
            g[j1] += (1.0 - h[j1 - m])
            j2 = np.arange(0, n - m)
            g[j2] -= h[j2 + m]
    else:
        p = -m
        if p < n:
            j1 = np.arange(0, n - p)
            g[j1] += (1.0 - h[j1 + p])
            j2 = np.arange(p, n)
            g[j2] -= h[j2 - p]
    return g

def fw_optimize(h, n, max_iter, deadline):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stale = 0
    for t in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c.max())
        tol = max(1e-10, 0.003 * max(1.0, c_max))
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            grad += lag_subgrad(h, n, m)
        grad /= len(active)
        k_half = n // 2
        idx = np.argpartition(grad, k_half)[:k_half]
        v = np.zeros(n)
        v[idx] = 1.0
        gap = np.dot(grad, v - h)
        if gap > -1e-12:
            break
        alpha = 2.0 / (t + 2.0)
        h_new = (1.0 - alpha) * h + alpha * v
        h_new = project_feasible(h_new, n)
        c_new_val = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')))
        if c_new_val < best_c - 1e-12:
            best_c = c_new_val
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        h = h_new
        if stale > 300:
            break
    return best_h, best_c

def coord_cd(h, n, max_passes, deadline, grid_size=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            diag_pos = np.where(ms == (n - 1))[0]
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if len(diag_pos) > 0:
                    nv = nv.copy()
                    nv[diag_pos] -= d * d
                val = max(unaff, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        cur = compute_c5(h, n)
        if cur < best_c:
            best_c = cur
    return h, best_c

def binary_sa(h, n, time_budget, rng):
    h_bin = np.zeros(n)
    n_ones_target = n // 2
    order = np.argsort(-h)[:n_ones_target]
    h_bin[order] = 1.0
    best_h = h_bin.copy()
    c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    best_c = float(c.max())
    cur_h = h_bin.copy()
    cur_c = best_c
    T = 0.03
    T_end = 0.0005
    cool = 0.99995
    t_start = time.time()
    one_indices = list(np.where(cur_h == 1.0)[0])
    zero_indices = list(np.where(cur_h == 0.0)[0])
    while time.time() - t_start < time_budget and T > T_end:
        i = one_indices[rng.randint(len(one_indices))]
        j = zero_indices[rng.randint(len(zero_indices))]
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c_new = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        cur_c_new = float(c_new.max())
        delta = cur_c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = cur_c_new
            one_indices[one_indices == i] = j
            one_indices[one_indices == i] = j
            zero_indices[zero_indices == j] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        T *= cool
    return best_h, best_c
