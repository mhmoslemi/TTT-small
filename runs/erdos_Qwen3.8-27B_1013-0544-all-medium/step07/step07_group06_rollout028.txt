    h = np.asarray(h, dtype=float).ravel()
    idx = np.argsort(-h)[:target]
    hb = np.zeros(n)
    hb[idx] = 1.0

    base = np.zeros((n, 2*n-1), dtype=np.int64)

        psm = p - s
        v2 = (psm >= 0) & (psm < n)
