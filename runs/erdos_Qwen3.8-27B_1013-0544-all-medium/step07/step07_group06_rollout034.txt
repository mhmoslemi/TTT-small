        if 0 <= t1 < 2*n-1:
            r1 = 2*q - p
            if 0 <= r1 < n:
                new_corr[t1] += 1.0 - 2.0*h[r1]
        k2 = p - q
        t2 = n - 1 + k2
        if 0 <= t2 < 2*n-1 and t2 != t1:
            r2 = 2*p - q
            if 0 <= r2 < n:
                new_corr[t2] += 1.0 - 2.0*h[r2]
        new_M = int(round(new_corr.max()))
        delta = new_M - M
        if delta <= 0 or rng.rand() < np.exp(-float(delta) / max(T, 1e-10)):
            h[p], h[q] = 0.0, 1.0
            corr = new_corr
            base_rows = build_base_rows(h, n, np.array([p, q]))
            base[p] = base_rows[0]
            base[q] = base_rows[1]
            M = new_M
            if M < best_M:
                best_M = M
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 50000 and T < 0.1:
            break
    return best_h, best_M

def binary_greedy(h, n, deadline):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(corr.max()))
    best_h, best_M = h.copy(), M
    base = build_base_rows(h, n, np.arange(n))
    for _ in range(50):
        if time.time() > deadline: break
        ones = np.nonzero(h == 1)[0]
        zeros = np.nonzero(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0: break
        improved = False
        chunk = 2048
        zi = np.repeat(ones, 1)
        oj = np.tile(zeros, 1)
        for start in range(0, len(zi), chunk):
            if time.time() > deadline: break
            end = min(start + chunk, len(zi))
            i = zi[start:end]
            j = oj[start:end]
            arr = base[i] + base[j]
            # Cross terms
            for idx in range(len(i)):
                p, q = i[idx], j[idx]
                t1 = n - 1 + (q - p)
                if 0 <= t1 < 2*n-1:
                    r1 = 2*q - p
                    if 0 <= r1 < n:
                        arr[idx, t1] += 1.0 - 2.0*h[r1]
                t2 = n - 1 + (p - q)
                if 0 <= t2 < 2*n-1 and t2 != t1:
                    r2 = 2*p - q
                    if 0 <= r2 < n:
                        arr[idx, t2] += 1.0 - 2.0*h[r2]
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-6)[0]
            if idx.size:
                loc = int(idx[int(np.argmin(mx[idx]))])
                iv, jv = int(i[loc]), int(j[loc])
                h[iv], h[jv] = 0.0, 1.0
                corr = arr[loc].astype(float, copy=True)
                M = int(round(corr.max()))
                base_rows = build_base_rows(h, n, np.array([iv, jv]))
                base[iv] = base_rows[0]
                base[jv] = base_rows[1]
                if M < best_M:
                    best_M = M
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_M

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1100.0
    end = t0 + max(30.0, total_budget - 15.0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad")
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_n, best_h, best_c = n0, h0.copy(), 999.0

    # Phase 1: LSE polish on n0
    if time.time() < end - 300:
        hh, cc = polish(h0, n0, [50.0, 200.0, 1000.0, 5000.0], end - 300, 60.0)
        if cc < best_c:
            best_c, best_h, best_n = cc, hh.copy(), n0

    # Phase 2: Binary search on n0
    if time.time() < end - 200:
        hb0 = to_binary(best_h, best_n)
        cg = compute_c5(hb0, best_n)
        if cg < best_c:
            best_c, best_h = cg, hb0.copy()
        # Greedy
        if time.time() < end - 150:
            hb_g, M_g = binary_greedy(hb0, best_n, end - 150)
            cg2 = M_g * 2.0 / best_n
            if cg2 < best_c:
                best_c, best_h = cg2, hb_g.copy()
        # SA
        if time.time() < end - 80:
            hb_sa, M_sa = binary_sa(hb_g, best_n, end - 80, T0=6.0, T1=0.05, max_iter=100000, seed=seed)
            cs = M_sa * 2.0 / best_n
            if cs < best_c:
                best_c, best_h = cs, hb_sa.copy()

    # Phase 3: Try other n values
    for nnew in (320, 256, 200):
        if time.time() > end - 60: break
        x0 = np.linspace(0, 2, best_n)
        x1 = np.linspace(0, 2, nnew)
        hr = np.interp(x1, x0, best_h)
        hr = project_feasible(hr, nnew)
        # Quick polish
        if time.time() < end - 40:
            hh, cc = polish(hr, nnew, [200.0, 2000.0], end - 35, 25.0)
            if cc < best_c:
                best_c, best_h, best_n = cc, hh.copy(), nnew
        hb = to_binary(best_h if best_n == nnew else hr, nnew)
        if time.time() < end - 30:
            hb_g, M_g = binary_greedy(hb, nnew, end - 30)
            cg = M_g * 2.0 / nnew
            if cg < best_c:
                best_c, best_h, best_n = cg, hb_g.copy(), nnew
        if time.time() < end - 20:
            hb_sa, M_sa = binary_sa(hb_g, nnew, end - 20, T0=4.0, T1=0.05, max_iter=50000, seed=seed+1)
            cs = M_sa * 2.0 / nnew
            if cs < best_c:
                best_c, best_h, best_n = cs, hb_sa.copy(), nnew

    # Final: ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
