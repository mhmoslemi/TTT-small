    # Softmax weights
    aC = alpha * C
    aC_shifted = aC - aC.max()
    w = np.exp(aC_shifted)
    w /= w.sum()
    # Gradient = weighted sum of individual gradients

    grad = G.T @ w  # shape (n,)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return logsum / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: SLSQP with increasing alpha
    alphas_s = [200, 1000, 5000, 25000, 100000]
    for a in alphas_s:
        if time.time() > deadline * 0.35:
            break
        h = slsqp_run(h, n, a, maxiter=4000)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: L-BFGS-B with penalty (faster, more iterations)
    alphas_l = [50000, 200000, 1000000]
    for a in alphas_l:
        if time.time() > deadline * 0.65:
            break
        h = lbfgs_run(best_h, n, a, maxiter=8000)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Subgradient with momentum
    if time.time() < deadline * 0.85:
        h, c5 = subgrad_polish(best_h, n, n_iter=20000, start_alpha=50000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Multi-start SLSQP at very high alpha
    for trial in range(5):
        if time.time() > deadline - 10:
            break
        if trial == 0:
            h_try = best_h.copy()
        else:
            h_try = best_h + 0.005 * rng.randn(n)
            h_try = project_feasible(h_try, n)
        h_try = slsqp_run(h_try, n, 5000000, maxiter=5000)
        c5 = compute_c5(h_try, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 5: Final L-BFGS-B polish
    if time.time() < deadline - 3:
        h = lbfgs_run(best_h, n, 5000000, maxiter=5000)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
