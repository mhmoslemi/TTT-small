def flip_delta(h, pos, n):
    """Vectorized computation of correlation change when h[pos] flips."""
    delta = np.zeros(2 * n - 1)
    # Direct: h[pos] * (1-h[pos+l]) for valid l
    # l range: max(-pos, -(n-1)) to min(n-pos-1, n-1)
    lo = max(-pos, -(n-1))
    hi = min(n - pos - 1, n - 1)
    ls = np.arange(lo, hi + 1)
    idx1 = n - 1 + ls
    vals = 1.0 - h[pos + ls]
    if h[pos] == 0:
        delta[idx1] += vals
    else:
        delta[idx1] -= vals
    
    # Indirect: h[pos-l] * (1-h[pos]) for valid l
    # l range: max(pos-n+1, -(n-1)) to min(pos, n-1)
    lo2 = max(pos - n + 1, -(n-1))
    hi2 = min(pos, n - 1)
    ls2 = np.arange(lo2, hi2 + 1)
    idx2 = n - 1 + ls2
    vals2 = h[pos - ls2]
    if h[pos] == 0:
        delta[idx2] -= vals2
    else:
        delta[idx2] += vals2
    
    return delta

                ones.remove(best_j)

    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    return h_b

def flip_delta(h, pos, n):
    delta = np.zeros(2 * n - 1)
    lo = max(-pos, -(n - 1))
    hi = min(n - pos - 1, n - 1)
    if lo <= hi:
        ls = np.arange(lo, hi + 1)
        idx1 = n - 1 + ls
        vals = 1.0 - h[pos + ls]
        if h[pos] == 0:
            delta[idx1] += vals
        else:
            delta[idx1] -= vals
    lo2 = max(pos - n + 1, -(n - 1))
    hi2 = min(pos, n - 1)
    if lo2 <= hi2:
        ls2 = np.arange(lo2, hi2 + 1)
        idx2 = n - 1 + ls2
        vals2 = h[pos - ls2]
        if h[pos] == 0:
            delta[idx2] -= vals2
        else:
            delta[idx2] += vals2
    return delta

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return float(m + logsum / alpha)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary(h_b, n, c_arr, max_sweeps=200, top_k=15):
    dx = 2.0 / n
    cur_c = float(c_arr.max() * dx)
    best_h = h_b.copy()
    best_c = cur_c
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    for _ in range(max_sweeps):
        improved = False
        for i in list(zeros):
            d_i = flip_delta(h_b, i, n)
            c_new_i = c_arr + d_i
            candidates = sorted(ones, key=lambda j: (c_new_i + flip_delta(h_b, j, n)).max())[:top_k]
            best_j = -1
            best_nc = cur_c
            for j in candidates:
                nc = float((c_new_i + flip_delta(h_b, j, n)).max() * dx)
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                d = d_i + flip_delta(h_b, best_j, n)
                h_b[i], h_b[best_j] = 1.0, 0.0
                c_arr = c_arr + d
                cur_c = best_nc
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
        if not improved:
            break
    return best_h, best_c, c_arr

def anneal_binary(h_b, n, c_arr, deadline, T0=0.004, cool=0.99997, reheat_every=20000, reheat_amt=0.002):
    dx = 2.0 / n
    rng = np.random.RandomState(77)
    cur_c = float(c_arr.max() * dx)
    best_h = h_b.copy()
    best_c = cur_c
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    T = T0
    it = 0
    while time.time() < deadline:
        it += 1
        if it % reheat_every == 0 and T < 0.0005:
            T += reheat_amt
        T *= cool
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        d = flip_delta(h_b, i, n) + flip_delta(h_b, j, n)
        new_max = float((c_arr + d).max() * dx)
        accept = new_max < cur_c
        if not accept and T > 1e-10:
            diff = (new_max - cur_c) / T
            if diff < 50:
                accept = rng.random() < np.exp(-diff)
        if accept:
            h_b[i], h_b[j] = 1.0, 0.0
            c_arr = c_arr + d
            cur_c = new_max
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
    return best_h, best_c, c_arr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Continuous optimization (12%)
    p1 = t0 + 0.12 * total
    for alpha in [50000, 500000, 5000000]:
        if time.time() > p1 - 3:
            break
        h = slsqp_optimize(best_h, n, alpha, 5000, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Binarize + greedy (8%)
    p2 = t0 + 0.20 * total
    h_b = binarize(best_h, n)
    c_arr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    h_b, c_b, c_arr = greedy_binary(h_b, n, c_arr, max_sweeps=300, top_k=20)
    if c_b < best_c:
        best_c, best_h = c_b, h_b.copy()

    # Phase 3: Annealing (35%)
    p3 = t0 + 0.55 * total
    h_b, c_a, c_arr = anneal_binary(h_b, n, c_arr, p3, T0=0.005, cool=0.99995)
    if c_a < best_c:
        best_c, best_h = c_a, h_b.copy()

    # Phase 4: Greedy polish (8%)
    p4 = t0 + 0.63 * total
    h_b2, c_g, c_arr = greedy_binary(h_b, n, c_arr, max_sweeps=300, top_k=25)
    if c_g < best_c:
        best_c, best_h = c_g, h_b2.copy()

    # Phase 5: Second annealing with different params (20%)
    p5 = t0 + 0.83 * total
    h_b3, c_a2, c_arr = anneal_binary(h_b2, n, c_arr, p5, T0=0.002, cool=0.9999, reheat_every=15000, reheat_amt=0.001)
    if c_a2 < best_c:
        best_c, best_h = c_a2, h_b3.copy()

    # Phase 6: Final continuous optimization (10%)
    p6 = t0 + 0.93 * total
    h = project_feasible(best_h, n)
    for alpha in [1e6, 1e7, 1e8]:
        if time.time() > p6 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 5000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 7: Try higher resolution if time permits and n is small
    if time.time() < deadline - 80 and n <= 400:
        nh = 600
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = slsqp_optimize(hh, nh, 500000, 4000, 1e-14)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
            h_b6 = binarize(best_h, n)
            c6 = np.correlate(h_b6, 1 - h_b6, mode='full').astype(float)
            h_b6, c6b, c6 = greedy_binary(h_b6, n, c6, max_sweeps=100, top_k=15)
            if c6b < best_c:
                best_c, best_h = c6b, h_b6.copy()
            hb7, c7, _ = anneal_binary(h_b6, n, c6, min(deadline - 20, time.time() + 100))
            if c7 < best_c:
                best_c, best_h = c7, hb7.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
