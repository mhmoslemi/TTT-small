            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    Z = exps.sum()
    obj = m + (sm + np.log(max(Z, 1e-300))) / alpha
    
    if Z < 1e-300:
        return obj, np.zeros(n)
    probs = exps / Z
    
    grad = np.zeros(n)
    # Only process significant probability lags
    thresh = max(1e-10, 5.0 / (2*n-1))
    active_idx = np.where(probs > thresh)[0]
    for k in active_idx:
        w = alpha * probs[k] * dx
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n-lag] += w * (1.0 - x[lag:])
            grad[lag:] -= w * x[:n-lag]
        else:
            ll = -lag
            grad[ll:] += w * (1.0 - x[:n-ll])
            grad[:n-ll] -= w * x[ll:]
    return obj, grad

def optimize_slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(softmax_obj_grad, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < 1e10:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def optimize_penalty_lbfgsb(h, n, alpha, penalty=1e6, maxiter=500):
    """L-BFGS-B with penalty for the sum constraint."""
    def obj_pen(x):
        obj, grad = softmax_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        obj_pen = obj + penalty * s * s
        grad_pen = grad + 2 * penalty * s * np.ones(n)
        return obj_pen, grad_pen
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coarse_to_fine(h_fine, n, n_coarse):
