def lbfabs_eval(x,n,alpha,pen,flat_diff,prefix_weights? no):
    c = np.correlate(x, 1.0-x, 'full')[::-1]

_DIFF_CACHE = {}

def get_flat_diff(n):

    idx = np.arange(n, dtype=np.int64)

def c5_bound(h,n):
    return float(np.max(np.correlate(h, 1.0-h, 'full')) * (2.0/n))

def project_box_sum(y,n):
    T=0.5*n
    y=np.asarray(y,dtype=float).copy()
    y=np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo=float(y.min())-1.0; hi=float(y.max())+1.0
    for _ in range(60):

        s=float(np.clip(y+mid,0.0,1.0).sum())
        if s < T: lo=mid
        else: hi=mid
    h=np.clip(y+0.5*(lo+hi),0.0,1.0)
    for _ in range(20):

        if abs(diff)<1e-13: break
        if diff > 0:

        else:
            idx=np.where(h > 1e-14)[0]
        if idx.size == 0: break
        h[idx] += diff/idx.size
        h=np.clip(h,0.0,1.0)

    x=np.asarray(x,dtype=float)

    cm=float(c.max())

        f=cm + float(c.mean()-cm)
        w=np.full(2*n-1,1.0/(2*n-1))
    else:

        ez=np.exp(z)
        s=ez.sum()
        f=cm + np.log(s)/alpha

def round_to_binary(h,n,seed=0):
    T=n//2
    rng=np.random.RandomState(seed)
    scores = np.asarray(h,dtype=float)
    if scores.size == n:
       scores = scores + 1e-8*rng.uniform(size=n)
    else: scores = np.zeros(n)
    idx=np.argsort(scores)[::-1]
    b=np.zeros(n,dtype=float)
    b[idx[:T]]=1.0
    return b

import time
import numpy as np
from scipy.optimize import minimize

_DIFF_CACHE = {}

def get_flat_diff(n):
    if n in _DIFF_CACHE:
        return _DIFF_CACHE[n]
    idx = np.arange(n, dtype=np.int64)
    d = (idx[None, :] - idx[:, None]) + (n - 1)
    flat = d.astype(np.int16).ravel()
    _DIFF_CACHE[n] = flat
    return flat

def c5_bound(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def project_box_sum(y, n):
    T = 0.5 * n
    y = np.asarray(y, dtype=float).reshape(-1).copy()
    if y.size != n:
        y = np.resize(y, n)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < T:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        diff = T - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if idx.size == 0:
            break
        h[idx] += diff / idx.size
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h, n_old, n_new):
    if n_old == n_new:
        return project_box_sum(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return project_box_sum(np.interp(x_new, x_old, h), n_new)

def soft_counts_eval(x, n, alpha, pen, flat_diff):
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode='full')[::-1]
    cm = float(c.max())
    if alpha < 1e-8:
        f = float(c.mean())
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        z = alpha * (c - cm)
        ez = np.exp(z)
        s = float(ez.sum())
        f = cm + float(np.log(s)) / alpha
        w = ez / s
    pref = np.empty(2 * n, dtype=float)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    i = np.arange(n)
    rowsum = pref[2 * n - 1 - i] - pref[n - 1 - i]
    W = np.take(w, flat_diff).reshape(n, n)
    g = rowsum - W.dot(x) - W.T.dot(x)
    perr = float(x.sum()) - 0.5 * n
    if pen:
        f += 0.5 * pen * perr * perr
        g = g + pen * perr
    return f, g

def eq_fun(x, n):
    return float(x.sum()) - 0.5 * n

def eq_jac(x, n):
    return np.ones(n, dtype=float)

def lbfgs_round(h, n, alpha, pen, maxiter, deadline, flat_diff, bounds):
    x0 = project_box_sum(h, n)
    try:
        res = minimize(soft_counts_eval, x0, args=(n, alpha, pen, flat_diff),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-5, 'maxls': 20})
        x = project_box_sum(res.x, n)
        return x, c5_bound(x, n)
    except Exception:
        return x0, c5_bound(x0, n)

def coord_cd(h, n, passes, deadline, grid=49, seed=0):
    h = project_box_sum(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float)
    c = np.maximum(c, 0.0)
    rng = np.random.RandomState(seed)
    total_lag = 2 * n - 2
    for _ in range(passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            x0 = float(h[p])
            lo = min(-p, p - (n - 1))
            hi = max(n - 1 - p, p)
            start = lo + n - 1
            end = hi + n - 1
            length = end - start + 1
            idxs = np.arange(start, end + 1)
            c_aff = c[idxs]
            ms = idxs - (n - 1)
            a = np.zeros(length, dtype=float)
            jf = p + ms
            v = (jf >= 0) & (jf < n)
            if np.any(v):
                a[v] += 1.0 - h[jf[v]]
            jb = p - ms
            v = (jb >= 0) & (jb < n)
            if np.any(v):
                a[v] -= h[jb[v]]
            diag = np.flatnonzero(ms == 0)
            unaff = 0.0
            if start > 0:
                unaff = max(unaff, float(c[:start].max()))
            if end < total_lag:
                unaff = max(unaff, float(c[end + 1:].max()))
            init_val = max(unaff, float(c_aff.max()))
            dgrid = np.linspace(-x0, 1.0 - x0, grid)
            aff_vals = c_aff[None, :] + dgrid[:, None] * a[None, :]
            if diag.size:
                aff_vals[:, diag] -= (dgrid * dgrid)[:, None]
            scores = np.maximum(unaff, aff_vals.max(axis=1))
            bi = int(np.argmin(scores))
            if scores[bi] < init_val - 1e-12:
                h[p] = float(np.clip(x0 + dgrid[bi], 0.0, 1.0))
                c[idxs] = aff_vals[bi]
                improved = True
        if improved:
            c = np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float)
            c = np.maximum(c, 0.0)
        else:
            break
    return h, float(c.max())

def optimize_continuous(h, n, deadline, cap_s, seed=0):
    t_start = time.time()
    limit = min(deadline, t_start + cap_s)
    h = project_box_sum(h, n)
    best_h = h.copy()
    best_c = c5_bound(h, n)
    flat_diff = get_flat_diff(n)
    bounds = [(0.0, 1.0)] * n
    schedule = ((10.0, 1.0), (30.0, 5.0), (100.0, 10.0), (400.0, 20.0))
    for alpha, pen in schedule:
        if time.time() > limit - 45:
            break
        maxit = 50 if n <= 700 else 30
        h, c = lbfgs_round(h, n, alpha, pen, maxit, limit, flat_diff, bounds)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        if time.time() < limit - 25:

            c2 = mc * (2.0 / n)

    c2 = mc * (2.0 / n)
    if c2 < best_c:
        best_h = h2
        best_c = c2
    return best_h, best_c

def round_to_binary(h, n, seed=0):
    T = n // 2
    rng = np.random.RandomState(seed)
    scores = np.asarray(h, dtype=float).reshape(-1).copy()
    if scores.size != n:
        scores = np.zeros(n)
    scores = scores + 1e-8 * rng.uniform(size=n)
    idx = np.argsort(scores)[::-1]
    b = np.zeros(n, dtype=float)
    b[idx[:T]] = 1.0
    return b

def binary_candidates(c, b, n, rand_count, seed, cap=140):
    rng = np.random.RandomState(seed)
    codes = []
    maxv = float(c.max())
    order = np.argsort(c)[::-1]
    n_shifts = 12
    for idx in order[:n_shifts]:
        val = float(c[idx])
        if val < maxv - 1.0:
            break
        m = int(idx - (n - 1))
        if m > 0:
            d = m
            pos = np.flatnonzero((b[:-d] == 1.0) & (b[d:] == 0.0))
            if pos.size:
                if pos.size > 8:
                    pos = rng.choice(pos, size=8, replace=False)
                for pi in pos:
                    codes.append(int(pi * n + (pi + d)))
        elif m < 0:
            d = -m
            pos = np.flatnonzero((b[d:] == 1.0) & (b[:-d] == 0.0)) + d
            if pos.size:
                if pos.size > 8:
                    pos = rng.choice(pos, size=8, replace=False)
                for pi in pos:
                    codes.append(int(pi * n + (pi - d)))
    ones = np.flatnonzero(b == 1.0)
    zeros = np.flatnonzero(b == 0.0)
    for _ in range(rand_count):
        if ones.size and zeros.size:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            codes.append(int(i * n + j))
    if not codes:
        return np.array([], dtype=np.int64)
    arr = np.array(codes, dtype=np.int64)
    arr = np.unique(arr)
    if arr.size > cap:
        arr = rng.choice(arr, size=cap, replace=False)
    return arr

def binary_local_search(h, n, deadline, seed=0, cand_cap=130):
    if n % 2 != 0:
        return h, c5_bound(h, n)
    b = round_to_binary(h, n, seed)
    c = np.correlate(b, 1.0 - b, mode='full')[::-1].astype(float)
    c = np.maximum(c, 0.0)
    best_b = b.copy()
    best_c = c5_bound(b, n)
    flat_moves = 0
    it = 0
    while time.time() < deadline and it < 20000:
        it += 1
        maxv = float(c.max())
        active = int(np.sum(c >= maxv - 0.5))
        codes = binary_candidates(c, b, n, 24, seed + it * 7919, cand_cap)
        if codes.size == 0:
            break
        best_code = -1
        best_key = (maxv, active)
        best_c_new = None
        for code in codes:
            i = int(code // n)
            j = int(code % n)
            if i == j or i < 0 or j < 0 or i >= n or j >= n:
                continue
            if b[i] != 1.0 or b[j] != 0.0:
                continue
            b[i] = 0.0
            b[j] = 1.0
            c_new = np.correlate(b, 1.0 - b, mode='full')[::-1].astype(float)
            nm = float(c_new.max())
            act = int(np.sum(c_new >= nm - 0.5))
            if nm < best_key[0] - 1e-12 or (abs(nm - best_key[0]) <= 1e-12 and act < best_key[1]):
                best_key = (nm, act)
                best_code = code
                best_c_new = c_new.copy()
            b[i] = 1.0
            b[j] = 0.0
        if best_code < 0:
            break
        i = int(best_code // n)
        j = int(best_code % n)
        b[i] = 0.0
        b[j] = 1.0
        c = best_c_new
        cur_c5 = float(np.max(c) * (2.0 / n))
        if cur_c5 < best_c - 1e-14:
            best_c = cur_c5
            best_b = b.copy()
            flat_moves = 0
        else:
            flat_moves += 1
            if flat_moves > 80:
                b = best_b.copy()
                c = np.correlate(b, 1.0 - b, mode='full')[::-1].astype(float)
                break
    cur_c5 = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * (2.0 / n))
    if cur_c5 < c5_bound(best_b, n):
        return b, cur_c5
    return best_b, c5_bound(best_b, n)

def slsqp_polish(h, n, deadline):
    h = project_box_sum(h, n)
    best_h = h.copy()

    flat_diff = get_flat_diff(n)

                best_h = x
        except Exception:

import time
import numpy as np
from scipy.optimize import minimize

_D = {}

def fdiff(n):
    if n in _D:
        return _D[n]
    a = np.arange(n)
    d = (a[None, :] - a[:, None] + n - 1).astype(np.int16).ravel()
    _D[n] = d
    return d

def c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def proj(y, n):
    T = 0.5 * n
    y = np.asarray(y, dtype=float).reshape(-1).copy()
    if y.size != n:
        y = np.resize(y, n)
    y = np.where(np.isfinite(y), y, 0.5)
    y = np.clip(y, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < T:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        d = T - float(h.sum())
        if abs(d) < 1e-13:
            break
        if d > 0:
            ix = np.where(h < 1.0 - 1e-14)[0]
        else:
            ix = np.where(h > 1e-14)[0]
        if ix.size:
            h[ix] += d / ix.size
            h = np.clip(h, 0.0, 1.0)
    return h

def res(h, na, nb):
    if na == nb:
        return proj(h, nb)
    xo = np.linspace(0.0, 2.0, na)
    xn = np.linspace(0.0, 2.0, nb)
    return proj(np.interp(xn, xo, h), nb)

def soft(x, n, al, pen, fl):
    x = np.asarray(x, dtype=float)
    cv = np.correlate(x, 1.0 - x, mode='full')[::-1]
    cm = float(cv.max())
    if al < 1e-8:
        f = float(cv.mean())
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        z = al * (cv - cm)
        e = np.exp(z)
        s = float(e.sum())
        f = cm + float(np.log(s)) / al
        w = e / s
    p = np.empty(2 * n, dtype=float)
    p[0] = 0.0
    np.cumsum(w, out=p[1:])
    i = np.arange(n)
    rs = p[2 * n - 1 - i] - p[n - 1 - i]
    W = np.take(w, fl).reshape(n, n)
    g = rs - W.dot(x) - W.T.dot(x)
    err = float(x.sum()) - 0.5 * n
    if pen:
        f += 0.5 * pen * err * err
        g = g + pen * err
    return f, g

def lb(h, n, al, pen, it, dep, fl, bd):
    x = proj(h, n)
    try:
        r = minimize(soft, x, args=(n, al, pen, fl), method='L-BFGS-B', jac=True,
                     bounds=bd, options={'maxiter': it, 'ftol': 1e-12, 'gtol': 1e-5, 'maxls': 15})
        x = proj(r.x, n)
        return x, c5(x, n)
    except Exception:
        return x, c5(x, n)

def cd(h, n, pa, dep, gr=49, seed=0):
    h = proj(h, n).copy()
    cv = np.maximum(np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float), 0.0)
    rng = np.random.RandomState(seed)
    L = 2 * n - 2
    for _ in range(pa):
        if time.time() > dep:
            break
        imp = False
        for p in rng.permutation(n):
            if time.time() > dep:
                break
            x0 = float(h[p])
            lo = min(-p, p - n + 1)
            hi = max(n - 1 - p, p)
            s = lo + n - 1
            e = hi + n - 1
            ix = np.arange(s, e + 1)
            ca = cv[ix]
            ms = ix - (n - 1)
            a = np.zeros(ix.size, dtype=float)
            jf = p + ms
            v = (jf >= 0) & (jf < n)
            a[v] += 1.0 - h[jf[v]]
            jb = p - ms
            v = (jb >= 0) & (jb < n)
            a[v] -= h[jb[v]]
            dg = np.flatnonzero(ms == 0)
            un = 0.0
            if s > 0:
                un = max(un, float(cv[:s].max()))
            if e < L:
                un = max(un, float(cv[e + 1:].max()))
            init = max(un, float(ca.max()))
            d = np.linspace(-x0, 1.0 - x0, gr)
            av = ca[None, :] + d[:, None] * a[None, :]
            if dg.size:
                av[:, dg] -= (d * d)[:, None]
            sc = np.maximum(un, av.max(axis=1))
            b = int(np.argmin(sc))
            if sc[b] < init - 1e-12:
                h[p] = float(np.clip(x0 + d[b], 0.0, 1.0))
                cv[ix] = av[b]
                imp = True
        if imp:
            cv = np.maximum(np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float), 0.0)
        else:
            break
    return h, float(cv.max())

def rab(h, n, seed=0):
    T = n // 2
    rng = np.random.RandomState(seed)
    s = np.asarray(h, dtype=float).reshape(-1).copy()
    if s.size != n:
        s = np.zeros(n)
    s = s + 1e-8 * rng.uniform(size=n)
    b = np.zeros(n, dtype=float)
    b[np.argsort(s)[::-1][:T]] = 1.0
    return b

def bcode(cv, b, n, rnd, seed, cap=120):
    rng = np.random.RandomState(seed)
    codes = []
    mv = float(cv.max())
    ord = np.argsort(cv)[::-1]
    for ix in ord[:10]:
        if cv[ix] < mv - 1.0:
            break
        m = int(ix - (n - 1))
        if m > 0:
            d = m
            pos = np.flatnonzero((b[:-d] == 1.0) & (b[d:] == 0.0))
            if pos.size:
                if pos.size > 6:
                    pos = rng.choice(pos, 6, replace=False)
                for q in pos:
                    codes.append(int(q * n + q + d))
        elif m < 0:
            d = -m
            pos = np.flatnonzero((b[d:] == 1.0) & (b[:-d] == 0.0)) + d
            if pos.size:
                if pos.size > 6:
                    pos = rng.choice(pos, 6, replace=False)
                for q in pos:
                    codes.append(int(q * n + q - d))
    ones = np.flatnonzero(b == 1.0)
    zeros = np.flatnonzero(b == 0.0)
    for _ in range(rnd):
        if ones.size and zeros.size:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            codes.append(int(i * n + j))
    if not codes:
        return np.array([], dtype=np.int64)
    arr = np.unique(np.array(codes, dtype=np.int64))
    if arr.size > cap:
        arr = rng.choice(arr, cap, replace=False)
    return arr

            nm = float(cn.max())

                key = (nm, ac)

            b[i] = 1.0

        i = int(best_code // n)
