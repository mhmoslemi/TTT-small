        if len(active) > 30:
            active = np.argsort(c)[-20:]
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            if k >= 0:
                grad[:n - k] += a[k:]
                grad[k:] += -h[:n - k]
            else:
                kn = -k
                grad[kn:] += a[:n - kn]
                grad[:n - kn] += -h[kn:]
        grad /= len(active)
        grad -= grad.mean()
        lr = lr_init / (1.0 + t * 0.0004)
        m_acc = beta1 * m_acc + (1 - beta1) * grad
        v_acc = beta2 * v_acc + (1 - beta2) * grad ** 2
        m_hat = m_acc / (1 - beta1 ** (t + 1))
        v_hat = v_acc / (1 - beta2 ** (t + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
    return best_h, best_c

def binary_sa(h0, n, deadline, max_iters=400000):
    rng = np.random.RandomState(123)
    h = to_binary(h0, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = 0.04
    T_min = 0.0003
    cool_rate = (T_min / T) ** (1.0 / max_iters)
    m_range = np.arange(-(n - 1), n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < len(delta):
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < len(delta):
            delta[k2] += 1.0
        new_c = (corr + delta).max() * dx
        dc = new_c - cur_c
        if dc < 0 or rng.random() < np.exp(-dc / T):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = new_c
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool_rate
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=80):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            best_delta = None
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def lse_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    log_ah = np.linspace(np.log(40), np.log(80000000), 28)
    for la in log_ah:
        if time.time() > deadline - 5:
            break
        alpha = int(np.exp(la))
        mi = 2000 if alpha < 10000 else (3500 if alpha < 1e6 else 5000)
        h = lse_slsqp(h, n, alpha, mi, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def barrier_push(h, n, c_cur, deadline):
    h = h.copy()
    c = c_cur
    for i in range(10):
        if time.time() > deadline - 4:
            break
        t = c * (1.0 - 0.004 * (i + 1))
        h = barrier_slsqp(h, n, t, 2500, 1e-16)
        c = compute_c5(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary SA
    p1 = t0 + 0.07 * total
    hb, cb = binary_sa(h0, n, p1)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Greedy binary best-swap
    p2 = t0 + 0.10 * total
    hb, cb = greedy_binary(best_h, n, p2)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 3: Subgradient Adam (new method)
    p3 = t0 + 0.32 * total
    h_sg, c_sg = subgrad_adam(best_h, n, p3, max_iters=100000, lr_init=0.01)
    if c_sg < best_c:
        best_c, best_h = c_sg, h_sg.copy()

    # Phase 4: LSE SLSQP main refinement
    p4 = t0 + 0.55 * total
    h, c = lse_refine(best_h, n, p4)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: Barrier push
    p5 = t0 + 0.65 * total
    h, c = barrier_push(best_h, n, best_c, p5)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 6: Diverse restarts
    p6 = t0 + 0.84 * total
    x = np.linspace(0, 2, n)
    perturbations = []
    perturbations.append(0.012 * rng.randn(n))
    perturbations.append(0.02 * np.sin(2 * np.pi * 3.0 * x + 0.5))
    perturbations.append(0.008 * np.cos(2 * np.pi * 9.0 * x + 1.1))
    perturbations.append(0.025 * rng.randn(n))
    perturbations.append(0.015 * np.sin(2 * np.pi * 5.5 * x + 2.0))
    for pert in perturbations:
        if time.time() > p6 - 12:
            break
        h = project_feasible(best_h + pert, n)
        h, c = subgrad_adam(h, n, p6 - 12, max_iters=30000, lr_init=0.008)
        if c < best_c:
            best_c, best_h = c, h.copy()
        h2, c2 = barrier_push(h, n, c, min(p6 - 8, dl - 60))
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 7: Higher resolution
    p7 = t0 + 0.92 * total
    for nh in [480, 600]:
        if time.time() > p7 - 25:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(x_new, x_old, best_h), nh)
        h2, c2 = lse_refine(hh, nh, p7 - 20)
        if c2 < best_c:
            best_c, best_h, n = c2, h2.copy(), nh

    # Final barrier polish
    h, c = barrier_push(best_h, n, best_c, dl - 5)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Final LSE micro-push
    for alpha in [50000000, 150000000]:
        if time.time() > dl - 3:
            break
        h = lse_slsqp(best_h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
