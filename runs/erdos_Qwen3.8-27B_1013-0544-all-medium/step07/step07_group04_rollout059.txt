    q /= q.sum()

        vals = c + a_mid * delta
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-12)[0]

        elif np.min(deltas) > 0:

            a_hi = a_mid
        else:
            # f is decreasing at a_mid, try larger a
            a_lo = a_mid

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    return m + np.log(e.sum()) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    q = np.exp(alpha * (c - m))
    q /= q.sum()
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    nz = np.flatnonzero(q > 1e-10)
    for idx in nz:
        k = idx - off
        w = q[idx]
        if k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            k2 = -k
            g[:n-k2] += w * (1.0 - x[k2:])
            g[k2:] -= w * x[:n-k2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def compute_pair_delta(h, n, i, j):
    dx = 2.0 / n
    off = n - 1
    delta = np.zeros(2 * n - 1)
    for pos, sign in ((i, 1.0), (j, -1.0)):
        delta[off] += sign * dx * (1.0 - 2.0 * h[pos])
        if pos > 0:
            ks = np.arange(1, pos + 1)
            delta[off + ks] += sign * dx * (1.0 - h[pos - 1::-1])
            delta[off - ks] += sign * (-dx) * h[pos - 1::-1]
        if pos < n - 1:
            ks = np.arange(1, n - pos)
            delta[off + ks] += sign * (-dx) * h[pos + 1:]
            delta[off - ks] += sign * dx * (1.0 - h[pos + 1:])
    return delta

def pair_minimize(c, delta, a_lo, a_hi):
    for _ in range(60):

        vals = c + a_mid * delta
        mv = float(np.max(vals))
        active = np.where(vals >= mv - 1e-14)[0]
        d_min = float(delta[active].min())
        d_max = float(delta[active].max())
        if d_min <= 0.0 <= d_max:
            break
        elif d_min > 0.0:
            a_hi = a_mid
        else:
            a_lo = a_mid
    return 0.5 * (a_lo + a_hi)

def try_pair(h, c, n, i, j):
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    a_lo, a_hi = lo - x0, hi - x0
    delta = compute_pair_delta(h, n, i, j)
