```python
def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w

def smooth_grad(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def l_fg(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    target = float(st['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    g = smooth_grad(x, w, n, dx)
    res = float(x.sum() - target)
    rho = float(st['rho'])
    lam = float(st['lam'])
    if rho != 0.0 or lam != 0.0:
        f += lam * res + 0.5 * rho * res * res
        g = g + (lam + rho * res)
    return float(f), g

def t_fg(x):
    """Targeted penalty: sum max(0, c_k - T)^2 + mu*(sum- target)^2"""
    st = TS
    n = int(st['n'])
    dx = float(st['dx'])
    T = float(st['T'])
    mu = float(st['mu'])
    target = 0.5 * n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    excess = np.maximum(0.0, c - T)
    f = float(np.sum(excess * excess))
    res = float(x.sum() - target)
    f += mu * res * res
    w = 2.0 * excess
    g = smooth_grad(x, w, n, dx)
    g = g + 2.0 * mu * res * np.ones(n)
    return f, g

def pair_obj(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, int(n))

def pair_line(h, n, i, j, cur, deadline):
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d = 0.0
    best_c = cur
    if time.time() > deadline:
        return h, cur
    c0 = pair_obj(lo, h, i, j, n)
    if c0 < best_c:
        best_c = c0
        best_d = lo
    c1 = pair_obj(hi, h, i, j, n)
    if c1 < best_c:
        best_c = c1
        best_d = hi
    try:
        if time.time() < deadline - 0.3:
            rs = so.minimize_scalar(pair_obj, args=(h, i, j, n),
                                    bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 25, 'xatol': 1e-10})
            if rs.x is not None:
                c2 = pair_obj(float(rs.x), h, i, j, n)
                if c2 < best_c:
                    best_c = c2
                    best_d = float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        hh = np.clip(hh, 0.0, 1.0)
        return hh, best_c
    return h, cur

def pair_polish(h, n, deadline, seed):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    m = 25 if n <= 500 else 15
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= m:
        top = order[:m]
    else:
        need = m - int(order.size)
        if need > n - int(order.size):
            need = max(0, n - int(order.size))
        if need > 0:
            extra = rng.choice(n, size=need, replace=False)
            top = np.unique(np.concatenate((order, extra)))
        else:
            top = order
    if top.size < 2:
        return h, cur
    for _ in range(3):
        improved = False
        pairs = []
        tlen = len(top)
        for a in range(tlen):
            for b in range(a + 1, tlen):
                pairs.append((int(top[a]), int(top[b])))
        if not pairs:
            break
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 4.0:
                return h, erdos_c5(h, n)
            hh, cc = pair_line(h, n, i, j, cur, deadline - 4.0)
            if cc < cur - 1e-14:
                h = hh
                cur = cc
                improved = True
        if not improved:
            break
    return h, cur
```

           sum(h+d) = n/2

This is essentially a **successive linear programming (SLP)** approach:
1. Linearize the nonconvex constraints at the current point
2. Solve the resulting LP
3. Take a step in the LP solution direction
4. Repeat

```python
def slp_step(h, n, active_lags, max_step):
    """One SLP iteration: linearize and solve LP."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    mid = n - 1
    # For each active lag, compute the gradient of f_k at h
    # and the current value
    # Variables: d[0..n-1] (change in h) and t
    # Minimize t
    # Subject to:
    #   f_k(h) + grad_k · d ≤ t  for k in active_lags
    #   0 ≤ h[i] + d[i] ≤ 1  for all i
    #   sum(d) = 0  (to maintain the sum constraint)
    #   |d[i]| ≤ max_step  (to ensure the linearization is valid)
    
    nvars = n + 1  # d[0..n-1] + t
    # Objective: minimize t
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0  # t is the last variable
    
    # Inequality constraints: f_k + grad_k · d - t ≤ 0
    A_ub = []
    b_ub = []
    for k in active_lags:
        grad = np.zeros(n)
        # Compute grad of f_k at h
        # f_k = sum_i h[i]*(1-h[i+k])*dx (for lag k > 0)
        # or use the smooth_grad formula with a delta weight
        # Actually, let me just use the correlation index
        # The full correlate index for lag k is mid+k (for positive k) or mid-k (for negative k)
        # Let me use the general formula:
        # For correlation index m, the gradient is:
        # grad[j] = [(1-h[j-(m-mid)]) * 1_{valid} + (-h[j+(m-mid)]) * 1_{valid}] * dx
        l = k - mid  # this is the actual lag
        for j in range(n):
            g = 0.0
            # h[j] appears as first factor when j = i+l, i.e., i = j-l
            if 0 <= j - l < n:
                g += (1 - h[j - l]) * dx
            # h[j] appears in (1-h[i]) when i = j
            if 0 <= j + l < n:
                g += (-h[j + l]) * dx
            grad[j] = g
        row = np.zeros(nvars)
        row[:n] = grad
        row[n] = -1.0  # -t
        A_ub.append(row)
        b_ub.append(-c[k])  # f_k + grad·d - t ≤ 0 → grad·d - t ≤ -f_k
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: 0 ≤ h[i]+d[i] ≤ 1 → -h[i] ≤ d[i] ≤ 1-h[i]
    # Also |d[i]| ≤ max_step
    bounds = []
    for i in range(n):
        lo = max(-h[i], -max_step)
        hi = min(1-h[i], max_step)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is unbounded
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            h_new = np.clip(h + d, 0, 1)
            return h_new
    except Exception:
        pass
    return h
```

    pref_n = np.cumsum(w_neg)
