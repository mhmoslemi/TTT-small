import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0

        else:

            if s > 1e-14:
                h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exps = np.exp(shifted)
    total = exps.sum()
    val = c_max + np.log(total) / alpha
    weights = exps / total
    grad = np.zeros_like(h)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            grad[j1s:j1e] += w * (1.0 - h[j1s + k:j1e + k])
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            grad[j2s:j2e] -= w * h[j2s - k:j2e - k]
    grad *= dx
    return val, grad

def optimize_smooth(h, n, alpha, maxiter=5000):
    def obj(x):
        return smooth_obj_and_grad(x, n, alpha)[0]
    def grad_func(x):
        return smooth_obj_and_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total  # length 2n-1

    # grad contribution = w_m * [(1-h) shifted by -k for valid positions] 
    #                    - w_m * [h shifted by k for valid positions]

    # computed as a convolution-like operation.

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0

            if s > 1e-14:
                h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def get_smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total
    grad = np.zeros_like(h)
    active = np.where(weights > 1e-10)[0]
    for m in active:
        w = weights[m]
        k = m - (n - 1)
        j1s, j1e = max(0, -k), min(n, n - k)
        if j1e > j1s:
            grad[j1s:j1e] += w * (1.0 - h[j1s + k:j1e + k])
        j2s, j2e = max(0, k), min(n, n + k)
        if j2e > j2s:
            grad[j2s:j2e] -= w * h[j2s - k:j2e - k]
    grad *= dx
    obj_val = c_max + np.log(total) / alpha
    return obj_val, grad

def projected_grad_loop(h, n, alpha, max_iter, time_limit, step_init=0.01, step_max=0.1):
    cur_obj, _ = get_smooth_gradient(h, n, alpha)
    cur_c5 = compute_c5(h, n)
    step = step_init
    best_h = h.copy()
    best_c5 = cur_c5
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        _, grad = get_smooth_gradient(h, n, alpha)
        grad_tan = grad - grad.mean()
        h_new = project_feasible(h - step * grad_tan, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            step = min(step * 1.03, step_max)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            step *= 0.7
            if step < 1e-7:
                break
    return best_h, best_c5, h

def slsqp_step(h, n, alpha, maxiter=2000):
    def obj(x):
        return get_smooth_gradient(x, n, alpha)[0]
    def grad(x):
        return get_smooth_gradient(x, n, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except NameError:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Projected gradient with increasing alpha
    phase1_dl = t0 + 0.35 * (budget_s - 15)
    alphas = [30, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    for alpha in alphas:
        if time.time() > phase1_dl:
            break
        h = best_h.copy()
        h, c5_this, h_end = projected_grad_loop(h, n, alpha, 30000, phase1_dl,
                                                 step_init=0.01, step_max=0.08)
        if c5_this < best_c5 - 1e-13:
            best_c5 = c5_this
            best_h = h.copy()
            h = h_end

    # Phase 2: Noise restarts with re-optimization
    phase2_dl = t0 + 0.65 * (budget_s - 15)
    n_restarts = 20
    for trial in range(n_restarts):
        if time.time() > phase2_dl:
            break
        scale = 0.015 + 0.02 * rng.rand()
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [500, 2000, 5000]:
            if time.time() > phase2_dl:
                break
            h, c5_this, h_end = projected_grad_loop(h, n, alpha, 15000, phase2_dl,
                                                     step_init=0.008, step_max=0.05)
            if c5_this < best_c5 - 1e-13:
                best_c5 = c5_this
                best_h = h.copy()
                h = h_end

    # Phase 3: Multi-resolution
    phase3_dl = t0 + 0.82 * (budget_s - 15)
    for nh in [200, 600, 800]:
        if time.time() > phase3_dl - 30:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, nh)
        hh = np.interp(x_new, x_old, best_h)
        hh = project_feasible(hh, nh)
        c5_hh = compute_c5(hh, nh)
        if c5_hh > best_c5 + 0.002:
            continue
        for alpha in [300, 1000, 3000]:
            if time.time() > phase3_dl - 10:
                break
            hh, c5_this, hh_end = projected_grad_loop(hh, nh, alpha, 10000, phase3_dl - 10,
                                                       step_init=0.01, step_max=0.06)
            if c5_this < best_c5 - 1e-13:
                best_c5 = c5_this
                best_h = hh.copy()
                n = nh
                hh = hh_end

    # Phase 4: SLSQP polish
    phase4_dl = dl - 30
    if time.time() < phase4_dl:
        for alpha in [5000, 20000, 50000]:
            if time.time() > phase4_dl:
                break
            h_pol = slsqp_step(best_h, n, alpha, maxiter=1000)
            c5_pol = compute_c5(h_pol, n)
            if c5_pol < best_c5 - 1e-13:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Final fine projected gradient
    if time.time() < dl - 5:
        for alpha in [10000, 30000, 100000]:
            if time.time() > dl - 5:
                break
            h_fin, c5_fin, _ = projected_grad_loop(best_h, n, alpha, 50000, dl - 5,
                                                    step_init=0.003, step_max=0.02)
            if c5_fin < best_c5 - 1e-13:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
