   L=c.size

def compute_lin(h,n):

import time
import numpy as np
from scipy.optimize import minimize

_stop_time = 0.0

def _stop_cb(x):
    return time.time() > _stop_time

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s <= 1e-15:
                break
            h += d * r / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += d * h / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    for _ in range(3):
        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - h
            if r.sum() > 0:
                h += d * r / r.sum()
            else:
                h[int(np.argmax(h))] += d
        else:
            s = h.sum()
            if s > 0:
                h += d * h / s
            else:
                h[0] -= d
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
    return h

def raw_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full").astype(float)

def c_vals(h, n):
    return raw_corr(h, n) * (2.0 / float(n))

def c5(h, n):
    return float(np.max(raw_corr(h, n))) * (2.0 / float(n))

def top_n_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    m = n // 2
    out = np.zeros(n, dtype=float)
    if m > 0:
        out[np.argsort(-h, kind="stable")[:m]] = 1.0
    return out

def weight_from_c(c, beta, K):
    c = np.asarray(c, dtype=float).ravel()
    m = float(c.max())
    L = c.size
    if K and K > 0:
        K = int(min(K, L))
        if K >= L:
            return np.full(L, 1.0 / L)
        idx = np.argpartition(c, -K)[-K:]
        p = np.zeros(L, dtype=float)
        p[idx] = 1.0 / K
        return p
    if beta is None or beta <= 0:
        return np.full(L, 1.0 / L)
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return np.full(L, 1.0 / L)
    return e / s

def grad_raw(h, n, p):
    v = 1.0 - h
    r = p[::-1]
    a = np.convolve(p, v, mode="full")[n - 1:2 * n - 1]
    b = np.convolve(h, r, mode="full")[n - 1:2 * n - 1]
    return a - b

def lse_obj(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    cv = raw_corr(h, n) * dx
    m = float(cv.max())
    y = beta * (cv - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        p = np.full(cv.size, 1.0 / cv.size)
        f = m
    else:
        p = e / s
        f = m + np.log(s) / beta
    g = grad_raw(h, n, p) * dx
    d = h.sum() - 0.5 * n
    if pen:
        f += 0.5 * pen * d * d
        g = g + pen * d
    if not np.isfinite(f):
        return 1e100, np.zeros_like(h)
    return float(f), g

def fixed_obj(x, n, p, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    cv = raw_corr(h, n) * dx
    f = float(np.dot(p, cv))
    g = grad_raw(h, n, p) * dx
    d = h.sum() - 0.5 * n
    if pen:
        f += 0.5 * pen * d * d
        g = g + pen * d
    if not np.isfinite(f):
        return 1e100, np.zeros_like(h)
    return float(f), g

def minimize_stage(x, obj, args, bounds, deadline, stage):
    now = time.time()
    dl = min(deadline, now + max(2.0, stage)) - 1.5
    if time.time() > dl:
        return x
    _stop_time = dl
    try:
        res = minimize(obj, x, method="L-BFGS-B", jac=True, bounds=bounds, args=args,
                       callback=_stop_cb,
                       options={"maxiter": 300000, "maxls": 24, "ftol": 1e-13, "gtol": 1e-7})
        if res is not None and hasattr(res, "x"):
            return res.x
    except Exception:
        pass
    return x

def variant_candidates(h, n):
    h = project_feasible(h, n)
    out = [
        h[::-1],
        1.0 - h,
        0.5 * (h + h[::-1]),
        0.5 * (h + 1.0 - h[::-1]),
    ]
    if n % 2 == 0:
        out.append(top_n_binary(h, n))
    return out

def comb_scores(n, p, w, phase, amp):
    p = max(2, int(p))
    w = max(1, min(p - 1, int(w)))
    x = np.arange(n, dtype=float)
    part = ((x + float(phase)) % float(p)) < float(w)
    return 0.5 + 0.5 * amp * (2.0 * part - 1.0)

def smooth_scores(n, rng):
    f = 1.0 + 2.0 * float(rng.randint(0, 3))
    ph = 2.0 * np.pi * float(rng.rand())
    ph2 = 2.0 * np.pi * float(rng.rand())
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return 0.5 + 0.35 * np.sin(np.pi * f * x + ph) + 0.25 * np.cos(2.0 * np.pi * f * x + ph2)

def make_candidates(n, parent_h, parent_n, rng):
    out = []
    if parent_n > 1:
        x0 = np.linspace(0.0, 2.0, parent_n)
        x1 = np.linspace(0.0, 2.0, n)
        out.append(np.interp(x1, x0, parent_h))
    for p in (max(4, n // 16), max(6, n // 10), max(8, n // 6), max(12, n // 4)):
        p = int(max(4, p))
        w = p // 2
        for amp in (1.0, 0.55):
            for ph in (0, p // 4):
                out.append(comb_scores(n, p, w, ph, amp))
    out.append(smooth_scores(n, rng))
    out.append(smooth_scores(n, rng))
    return out

def choose_from_candidates(cands, n):
    vals = []
    hs = []
    for idx, sc in enumerate(cands):
        h = project_feasible(sc, n)
        hs.append(h)
        vals.append((c5(h, n), idx))
    if not vals:
        return project_feasible(np.full(n, 0.5), n), c5(np.full(n, 0.5), n)
    vals.sort()
    best = hs[vals[0][1]].copy()
    bc = vals[0][0]
    for cv, idx in vals[:3]:
        h = hs[idx]
        if cv < bc:
            bc = cv
            best = h.copy()
        for var in variant_candidates(h, n):
            vv = project_feasible(var, n)
            xv = c5(vv, n)
            if xv < bc:
                bc = xv
                best = vv.copy()
    return best, bc

def continuous_polish(h, n, deadline, fast=False):
    work = project_feasible(h, n)
    best = work.copy()
    bc = c5(best, n)
    bounds = [(0.0, 1.0)] * n
    if fast:
        prof = [
            ("T", 128, 0.0, 1000.0, 1),
            ("L", 0, 2000.0, 10000.0, 2),
            ("L", 0, 50000.0, 100000.0, 3),
        ]
    else:
        prof = [
            ("T", 512, 0.0, 1000.0, 2),
            ("T", 256, 0.0, 1000.0, 2),
            ("L", 0, 100.0, 10000.0, 2),
            ("T", 128, 0.0, 10000.0, 2),
            ("L", 0, 2000.0, 100000.0, 3),
            ("T", 64, 0.0, 100000.0, 3),
            ("L", 0, 20000.0, 100000.0, 5),
            ("L", 0, 200000.0, 100000.0, 6),
        ]
    tot = 0.0
    for pr in prof:
        tot += pr[4]
    for mode, K, beta, pen, w in prof:
        rem = deadline - time.time()
        if rem < 20.0:
            break
        stage = min(120.0, max(5.0, rem * w / tot))
        cv = c_vals(work, n)
        p = weight_from_c(cv, beta if mode == "L" else 0.0, K if mode == "T" else 0)
        if mode == "T":
            x = minimize_stage(work, fixed_obj, (n, p, pen), bounds, deadline, stage)
        else:
            x = minimize_stage(work, lse_obj, (n, beta, pen), bounds, deadline, stage)
        work = project_feasible(x, n)
        cw = c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
        for var in variant_candidates(work, n):
            vv = project_feasible(var, n)
            xv = c5(vv, n)
            if xv < bc:
                bc = xv
                best = vv.copy()
        if cw > bc + 1e-7:
            work = best.copy()
        else:
            work = work.copy()
    return best, bc

def score_val(c, M):
    return float((np.abs(c - M) < 1e-6).sum()) * 100000.0 + float(np.dot(c, c))

def compute_lin(h, n):
    L = 2 * n - 1
    lin = np.zeros((n, L), dtype=float)
    lag = np.arange(-(n - 1), n, dtype=int)
    for p in range(n):
        f = p - lag
        mf = (f >= 0) & (f < n)
        s = p + lag
        ms = (s >= 0) & (s < n)
        if mf.any():
            lin[p, mf] += 1.0 - h[f[mf]]
        if ms.any():
            lin[p, ms] -= h[s[ms]]
        lin[p, n - 1] = 0.0
    return lin

def binary_local_search(h, n, deadline):
    h = np.asarray(h, dtype=float).ravel()
    if n % 2 != 0 or time.time() > deadline:
        return h, c5(h, n)
    hb = top_n_binary(h, n).copy()
    c = raw_corr(hb, n).astype(float)
    M = int(round(float(c.max())))
    cur_score = score_val(c, M)
    bestM = M
    best_score = cur_score
    besth = hb.copy()
    chunk = 1024 if n <= 450 else 512
    no_move = 0
    while time.time() < deadline:
        if no_move > 20:
            break
        zeros = np.nonzero(hb < 0.5)[0]
        ones = np.nonzero(hb > 0.5)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        lin = compute_lin(hb, n)
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        best_newM = M
        best_c = None
        best_i = -1
        best_j = -1
        best_sc = 1e100
        for st in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            en = min(st + chunk, zi.size)
            i = zi[st:en]
            j = oj[st:en]
            arr = c[None, :] + lin[i] - lin[j]
            arr[:, n - 1 + (j - i)] += 1.0
            arr[:, n - 1 + (i - j)] += 1.0
            mx = arr.max(axis=1)
            desc = mx <= M - 1 + 1e-6
            if desc.any():
                idx = np.nonzero(desc)[0]
                mn_i = int(round(float(np.min(mx[idx]))))
                rows = idx[mx[idx] <= mn_i + 1e-6]

                cnt = (np.abs(arr[rows] - mn_i) < 1e-6).sum(axis=1)

                loc = rows[int(np.argmin(sc))]

        if best_c is not None:
            hb[best_i] = 1.0
            hb[best_j] = 0.0
            c = best_c
            M = best_newM
            cur_score = score_val(c, M)
            no_move = 0
            if M < bestM or (M == bestM and cur_score < best_score):
                bestM = M
                best_score = cur_score
                besth = hb.copy()
            continue
        if deadline - time.time() < 12.0:
            break
        best_lat_i = -1
        best_lat_j = -1
        best_lat_c = None
        best_lat_sc = cur_score - 1e-9
        for st in range(0, zi.size, chunk):

                break
            en = min(st + chunk, zi.size)
            i = zi[st:en]
            j = oj[st:en]
            arr = c[None, :] + lin[i] - lin[j]
            arr[:, n - 1 + (j - i)] += 1.0
            arr[:, n - 1 + (i - j)] += 1.0
            mx = arr.max(axis=1)
            eq = mx <= M + 1e-6
            if not eq.any():
                continue
            idx = np.nonzero(eq)[0]
            if idx.size > 512:
                idx = idx[::max(1, idx.size // 512)][:512]
            sq = np.einsum("ij,ij->i", arr[idx], arr[idx])
            cnt = (np.abs(arr[idx] - M) < 1e-6).sum(axis=1)
            sc = cnt * 100000.0 + sq
            locl = int(np.argmin(sc))

            if minv < best_lat_sc:

                best_lat_i = int(i[idx[locl]])
                best_lat_j = int(j[idx[locl]])
