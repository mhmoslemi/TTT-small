    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = c.max()
        shifted = alpha * (c - mc)
        sm = shifted.max()

            if j2e > j2s:
                g[j2s:j2e] -= weights[m] * x[j2s - k:j2e - k]
