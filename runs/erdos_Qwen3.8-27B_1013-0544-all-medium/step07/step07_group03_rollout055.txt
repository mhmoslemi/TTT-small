```python
def linearized_local_search(h, n, deadline, max_rounds=1000):
    """
    Linearized local search: at each step, linearize the objective
    around the current h and solve a small LP to find a descent direction.
    """
    import cvxpy as cp
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1-h, mode='full') * dx
        cur_c = c.max()
        
        # Find active constraints (top few)
        top_k = 5
        active_idx = np.argsort(c)[::-1][:top_k]
        
        # Gradient of c_k with respect to h
        grads = np.zeros((top_k, n))
        for ii, idx in enumerate(active_idx):
            grads[ii] = compute_subgrad(idx, h, n) * dx
        
        # LP: minimize t subject to grads[ii] @ eps <= t - c[active_idx[ii]] + cur_c for all ii
        #     |eps| <= delta
        #     sum(eps) = 0
        delta = 0.01  # step size limit
        
        eps = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(eps) == 0,
            eps >= -delta,
            eps <= delta,
        ]
        for ii in range(top_k):
            constraints.append(grads[ii] @ eps <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=10000)
        except:
            break
        
        if eps.value is None:
            break
        
        eps_val = eps.value
        h_new = h + eps_val
        h_new = np.clip(h_new, 0, 1)
        # Re-project to maintain sum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < cur_c - 1e-10:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Reduce delta
            delta *= 0.5
            if delta < 1e-6:
                break
    
    return best_h, best_c
```

```python
            # We want all new_c_ii to be as small as possible

            constraints.append(c[top_idx[ii]] + G[ii] @ d <= t_var)

            break
```

Let me reconsider the LP approach. The key insight is:

At point h, the objective is f(h) = max_k c_k. The subdifferential is conv{g_k : c_k = f(h)}.

A first-order method would take a step in the direction -g_{k*} where c_{k*} = f(h). But this only considers one constraint.

The LP approach considers the top K constraints simultaneously and finds the best step that decreases all of them. This is like a "bundle method" step.

The LP:
minimize t
subject to: c_k + g_k^T d ≤ t, for k in active set
            -min(h_i, delta) ≤ d_i ≤ min(1-h_i, delta)
            sum(d) = 0

The optimal t* tells us the best possible value of the active constraints after the step. If t* < f(h), we make progress.

The key advantage over subgradient: we're solving the "worst case over K active constraints" exactly at each step.

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)


def compute_subgrad(c_idx, h, n):
    k = c_idx - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad


class _LSQPObj:
    def __init__(self, n, alpha):
        self._n = n
        self._alpha = alpha
        self._dx = 2.0 / n

    def __call__(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self._dx
        m = c.max()
        sh = self._alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / self._alpha


class _LSQPCons:
    def __init__(self, n):
        self._n = n
        self._target = n / 2.0

    def __call__(self, x):
        return x.sum() - self._target


def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _LSQPCons(n)}]
    try:
        res = minimize(_LSQPObj(n, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h


def lp_local_search(h, n, deadline, max_rounds=5000, top_k=12, delta_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    delta = delta_init
    no_improve = 0

    for r in range(max_rounds):
        if time.time() > deadline:
            break

        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_max = float(c.max())

        if cur_max < best_c:
            best_c = cur_max
            best_h = h.copy()

        top_idx = np.argsort(c)[::-1][:top_k]

        G = np.array([compute_subgrad(idx, h, n) * dx for idx in top_idx])

        d = cp.Variable(n)
        t_var = cp.Variable()
        lb = -np.minimum(h, delta)
        ub = np.minimum(1.0 - h, delta)
        constraints = [
            cp.sum(d) == 0,
            d >= lb,
            d <= ub,
        ]
        for ii in range(top_k):
            constraints.append(c[top_idx[ii]] + G[ii] @ d <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-9)
        except Exception:
            break

        if d.value is None:
            break

        d_val = np.asarray(d.value).flatten()
        h_new = project_feasible(h + d_val, n)
        c_new_val = compute_c5(h_new, n)

        if c_new_val < cur_max - 1e-11:
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            delta *= 0.7
            if delta < 1e-8 or no_improve > 100:
                break

    return best_h, best_c


def subgrad_optimize(h, n, deadline, max_iters=100000, top_k=3, lr_base=0.04):
```

```python
                                       max_rounds=3000, top_k=10, delta_init=0.02)
            if c_r < best_c:
                best_c, best_h = c_r, h_r.copy()

    # Final LP polish
    if time.time() < dl - 5:
        h_final, c_final = lp_local_search(best_h, n, dl - 2,
                                           max_rounds=5000, top_k=15, delta_init=0.005)
        if c_final < best_c:
            best_c, best_h = c_final, h_final.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
