    h_avg = h_trimmed.reshape(target_n, ratio).mean(axis=1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = corr - m
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def downsample(h, n, target_n):
    if n == target_n:
        return h.copy()
    ratio = n // target_n
    h_t = h[:target_n * ratio]
    return project_feasible(h_t.reshape(target_n, ratio).mean(axis=1), target_n)

def upsample(h, n, target_n):
    if n == target_n:
        return h.copy()
    return project_feasible(np.interp(np.linspace(0, 2, target_n), np.linspace(0, 2, n), h), target_n)

def adam_descent(h, n, deadline, lr=0.03, beta1=0.9, beta2=0.999, eps=1e-8):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    while time.time() < deadline:
        t += 1
        corr = np.correlate(h, 1 - h, mode='full') * dx
        m_corr = corr.max()
        sh = corr - m_corr
        p = np.exp(sh)
        p /= p.sum()
        S_plus = np.zeros(n)
        S_minus = np.zeros(n)
        for li in range(2 * n - 1):
            l = li - (n - 1)
            w = p[li]
            if w < 1e-12:
                continue
            if 0 <= l < n:
                S_plus[:n-l] += w * h[l:]
                S_minus[l:] += w * h[:n-l]
            elif -n < l < 0:
                ll = -l
                S_plus[ll:] += w * h[:n-ll]
                S_minus[:n-ll] += w * h[ll:]
        grad = dx * ((1.0 - S_plus) - S_minus)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

        return slsqp_smooth(h, n, alpha_base * 10, 500, 1e-13)
