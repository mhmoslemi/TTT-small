        pref_n = np.cumsum(w_neg)

def subgradient_polish(h, n, deadline, n_steps=50):
    """Direct subgradient steps on the max overlap."""
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    dx = 2.0 / n
    lr = 0.01  # learning rate
    
    for step in range(n_steps):
        if time.time() > deadline - 3.0:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(c))
        # Subgradient at k_star
        # dO_{k*}/dh[m] = dx * (1 - h[m + lag]) - dx * h[m - lag]
        # where lag = k_star - (n-1)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        # For each m, g[m] = dx * (1 - h[m+lag]) - dx * h[m-lag]
        # h[j] = 0 for j outside [0, n)
        for m in range(n):
            val = 1.0
            if 0 <= m + lag < n:
                val -= h[m + lag]
            if 0 <= m - lag < n:
                val -= h[m - lag]
            g[m] = dx * val
        
        # Projected gradient step
        h_new = project_box_sum(h - lr * g, 0.5 * n)
        c_new = erdos_c5(h_new, n)
        if c_new < cur:
            h = h_new
            cur = c_new
            lr = min(lr * 1.2, 0.1)
        else:
            lr *= 0.7
    
    return h, cur

            if time.time() < p2_end - 8.0:
                cand, c = opt_slsqp(best_h, 100000, 30, deadline)
                if c < best_c - 1e-15:
                    best_h, best_c = cand, c

    # Try upsample to 600
    if time.time() < p2_end - 15.0 and best_n < 600:
        nh_h = resample(best_h, best_n, 600)
        for beta in [5000, 50000]:
            if time.time() > p2_end - 5.0:
                break
            cand, c = opt_lbfgs(nh_h, beta, 60, deadline)
            nh_h = cand
        c_f = erdos_c5(nh_h, 600)
        if c_f < best_c - 5e-5:
            best_h, best_n, best_c = nh_h.copy(), 600, c_f

    # Phase 3: Subgradient polish
    if time.time() < deadline - 60.0:
        best_h, best_c = subgrad_polish(best_h, best_n, deadline - 40.0, n_steps=120)

    # Phase 4: Aggressive pair polish
    if time.time() < deadline - 20.0:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 10.0,
                                     n_pairs=70, rounds=8)

    # Phase 5: Final refinement
    if time.time() < deadline - 12.0:
        cand, c = opt_lbfgs(best_h, 2000000, 40, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
    if time.time() < deadline - 8.0:
        best_h, best_c = subgrad_polish(best_h, best_n, deadline - 4.0, n_steps=40)
    if time.time() < deadline - 4.0:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 1.0,
                                     n_pairs=40, rounds=3)

    # Random restarts if time permits
    rng = np.random.RandomState(int(seed) + 99)
    for sig in [0.03, 0.08, 0.15]:
        if time.time() > deadline - 10.0:
            break
        hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
        cand, c = opt_lbfgs(hh, 50000, 40, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
            if time.time() < deadline - 6.0:
                best_h, best_c = pair_polish(best_h, best_n, deadline - 2.0,
                                             n_pairs=30, rounds=2)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
