def compute_c(h, n):

    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    # Bisection on threshold
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual

    if abs(d) > 1e-14:

        c = int(m.sum())
        if c > 0:

    b[order[:k]] = 1.0

    c = compute_c(x, n)
    m = float(c.max())
    s = float(np.exp(alpha * (c - m)).sum())
    return m + np.log(s) / alpha if s > 0 else m

    c = compute_c(x, n)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    s = e.sum()
    q = e / s if s > 0 else np.zeros_like(c)
    g = np.zeros(n)
