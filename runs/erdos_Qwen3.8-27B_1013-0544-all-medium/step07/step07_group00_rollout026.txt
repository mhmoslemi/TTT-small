import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = (1.0 - h).sum()
            if room < 1e-15: break
            h += (diff / room) * (1.0 - h)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

Wait, the -d² term: it comes from the fact that when both i=p and i+s=p (i.e., s=0, k=n-1), the term h[p]*(1-h[p]) changes to (h[p]+d)*(1-h[p]-d) = h[p]*(1-h[p]) + d - d² - 2d*h[p]. The linear part is d*(1-h[p]) - d*h[p] = d*(1-2h[p]), which is captured by a. The quadratic part is -d².

So: c[m] += d * a[m] - d² * δ(m, n-1)

This is what the previous code does. Let me keep it.

But wait, there's a subtlety: the formula c[m] += d * a[m] assumes that a[m] was computed with the OLD h values. If h[p] was changed, then (1-h[p+s]) and h[p-s] might also change if s=0... but s=0 only happens at m=n-1, and we handle that with the -d² term.

For m ≠ n-1, s ≠ 0, so p+s ≠ p and p-s ≠ p, meaning h[p+s] and h[p-s] don't change. So the update c[m] += d*a[m] is exact.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = (1.0 - h).sum()
            if room < 1e-15:
                break
            h += (diff / room) * (1.0 - h)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_lp(fixed_vec, n):
    """Solve min t s.t. sum_i var[i]*fixed[i+s_k] <= t for all k, sum var = n/2, 0<=var<=1"""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        if i_lo <= i_hi:
            A[k, i_lo:i_hi+1] = fixed_vec[i_lo+s:i_hi+s+1]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    bounds = [(0.0, 1.0)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

            c2 = compute_c5(h2, n)

                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    return best_h, best_c

            break

                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2*n-2, n - 1 + p)

            m_b_hi = min(2*n-2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            for idx, m in enumerate(ms):
                s = m - (n-1)
                if 0 <= p + s < n:
                    a[idx] += (1.0 - h[p + s])
                if 0 <= p - s < n:
                    a[idx] -= h[p - s]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_mask = (ms == (n-1))
            d_grid = np.linspace(-h0, 1.0-h0, grid_size)
            best_d = 0.0
            best_val = current_max
            for d in d_grid:

            break
