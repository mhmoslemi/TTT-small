import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha, beta=0.0):
    """Compute smooth approximation value and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sm = corr.max()
    sh = alpha * (corr - sm)
    exp_sh = np.exp(sh)
    p = exp_sh / exp_sh.sum()
    val = sm + np.log(exp_sh.sum()) / alpha
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    threshold = 1e-10
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < threshold:
            continue
        l = k - (n - 1)
        if l >= 0:
            m = n - l
            if m <= 0:
                continue
            grad[:m] += w * (1.0 - h[l:n])
            if l > 0:
                grad[l:] -= w * h[:m]
        else:
            ll = -l
            m = n - ll
            if m <= 0:
                continue
            grad[ll:] += w * (1.0 - h[:m])
            if ll > 0:
                grad[:m] -= w * h[ll:]
    
    if beta > 0:
        s = h.sum() - n / 2.0
        val += beta * s * s
        grad += 2.0 * beta * s
    return val, grad

def lbfgsb_step(h, n, alpha, beta, max_iter):
    def obj(x):
        v, _ = smooth_grad(x, n, alpha, beta)
        return v
    def grad(x):
        _, g = smooth_grad(x, n, alpha, beta)
        return g
    bounds = [(0.0, 1.0)] * n
    try:

                       options={'maxiter': max_iter, 'gtol': 1e-8, 'maxls': 30})
        return project_feasible(res.x, n)
    except:
        return h

def nesterov_step(h, n, alpha, beta, n_iters, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    # Estimate Lipschitz constant
    L = alpha * dx * 2.0  # rough estimate
    lr = 1.0 / (L + 1e-10)
    t = 1.0
    y = h.copy()
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, grad = smooth_grad(y, n, alpha, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        y_new = project_feasible(y - lr * grad, n)
        x_new = y_new + theta * (y_new - y)
        x_new = project_feasible(x_new, n)
        y = x_new
        t = t_new
        if it % 200 == 0:
            c = compute_c5(y, n)
            if c < best_c:
                best_c = c
                best_h = y.copy()
    c = compute_c5(y, n)
    if c < best_c:
        best_c = c
        best_h = y.copy()
    return best_h, best_c

3. Find x ∈ [0,1] that minimizes max_l (C0_l + x * C1_l) (O(n log n) via sorting)
4. Update h[i] = x

Step 3 is the key: we're minimizing max of linear functions over x ∈ [0,1]. The optimum is either at x=0, x=1, or at an intersection point of two lines.

To find the optimum:
- At x=0: max_l C0_l
- At x=1: max_l (C0_l + C1_l)
- Interior: the x where the max transitions from one line to another

A simple approach: sort the lines by their value at x=0, and find the "upper envelope". The minimum of the upper envelope over [0,1] is at a vertex of the upper envelope or at the boundary.

Actually, a simpler approach for 1D: the function f(x) = max_l (C0_l + x*C1_l) is convex in x. So we can find the minimum by ternary search or by finding where the subgradient crosses 0.

The subgradient of f at x is: conv({C1_l : c_l(x) = f(x)}). f is minimized when 0 is in the subgradient, i.e., when there exist l+ and l- such that C1_{l+} > 0, C1_{l-} < 0, and c_{l+}(x) = c_{l-}(x) = f(x), and all other c_l(x) ≤ f(x).

A practical approach:
- Evaluate f at x=0, x=0.5, x=1
- Use ternary search to find the minimum (since f is convex)
- Or: binary search on the subgradient

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha, beta=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sm = corr.max()
    sh = alpha * (corr - sm)
    if sh.max() > 500:
        sh = sh - sh.max()
    exp_sh = np.exp(sh)
    psum = exp_sh.sum()
    p = exp_sh / psum
    val = sm + np.log(psum) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-10:
            continue
        l = abs(k - (n - 1))
        m = n - l
        if m <= 0:
            continue
        grad[l:] += w * (1.0 - h[:m])
        grad[:m] -= w * h[l:]
    if beta > 0:
        s = h.sum() - n / 2.0
        val += beta * s * s
        grad += 2.0 * beta * s
    return val, grad

def lbfgsb_step(h, n, alpha, beta, max_iter):
    def obj(x):
        return smooth_grad(x, n, alpha, beta)[0]
    def gradf(x):
        return smooth_grad(x, n, alpha, beta)[1]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=gradf, bounds=bounds,
                       options={'maxiter': max_iter, 'gtol': 1e-9, 'maxls': 40})
        return project_feasible(res.x, n)
    except Exception:
        return h

def nesterov_opt(h, n, alpha, beta, max_iters, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    L = alpha * dx * 3.0
    lr = 1.0 / (L + 1e-12)
    t = 1.0
    y = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        _, grad = smooth_grad(y, n, alpha, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        y_new = project_feasible(y - lr * grad, n)
        x_new = project_feasible(y_new + theta * (y_new - y), n)
        y = x_new
        t = t_new
        if it % 300 == 0:
            c = compute_c5(y, n)
            if c < best_c:
                best_c = c
                best_h = y.copy()
    c = compute_c5(y, n)
    if c < best_c:
        best_c = c
        best_h = y.copy()
    return best_h, best_c

def coord_polish(h, n, sweeps=5):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for sw in range(sweeps):
        improved = False
        for i in range(n):
            x_old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full') * dx
            diff = c1 - c0
            a, b = 0.0, 1.0
            for _ in range(25):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(c0 + m1 * diff)
                f2 = np.max(c0 + m2 * diff)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            x_opt = (a + b) / 2.0
            f_opt = np.max(c0 + x_opt * diff)
            if f_opt < best_c - 1e-13:
                h[i] = x_opt
                best_c = f_opt
                improved = True
            else:
                h[i] = x_old
        h = project_feasible(h, n)
        best_c = compute_c5(h, n)
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()

    alphas_phase1 = [50, 200, 1000, 5000, 20000]
    for i, a in enumerate(alphas_phase1):

        h1 = lbfgsb_step(best_h, n, a, 1e6, 400)
        c1 = compute_c5(h1, n)
        if c1 < best_c:

    # Phase 2: Nesterov accelerated PG
    if time.time() < deadline - 300:

        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: L-BFGS-B high alpha
    if time.time() < deadline - 200:
        for a in [50000, 200000, 500000]:
            if time.time() > deadline - 200:
                break
            h3 = lbfgsb_step(best_h, n, a, 1e8, 600)
            h3 = project_feasible(h3, n)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c, best_h = c3, h3.copy()

    # Phase 4: Coordinate descent polish (exact max)
    if time.time() < deadline - 150:
        h4, c4 = coord_polish(best_h, n, sweeps=3)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Diverse perturbation restarts
    if time.time() < deadline - 100:
        n_trials = 12
        for trial in range(n_trials):
            if time.time() > deadline - 60:

            if trial % 3 == 0:
                h5 = best_h + 0.004 * (1 + trial * 0.2) * rng.randn(n)
            elif trial % 3 == 1:
                mask = rng.rand(n) < 0.1
                h5 = best_h.copy()
                h5[mask] = 1.0 - best_h[mask]
            else:

                phase = rng.rand() * 2 * np.pi
                idx = np.arange(n)
                pert = 0.005 * np.sin(2 * np.pi * freq * idx / n + phase)
                h5 = best_h + pert
            h5 = project_feasible(h5, n)
            h5 = lbfgsb_step(h5, n, 50000, 1e7, 500)
            h5 = project_feasible(h5, n)
            h5, c5 = coord_polish(h5, n, sweeps=1)
            if c5 < best_c:

        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

        h7, c7 = coord_polish(best_h, n, sweeps=4)
        if c7 < best_c:
            best_c, best_h = c7, h7.copy()

    # Final L-BFGS-B polish
    if time.time() < deadline - 5:
        h8 = lbfgsb_step(best_h, n, 500000, 1e9, 2000)
        h8 = project_feasible(h8, n)
        h8, c8 = coord_polish(h8, n, sweeps=2)
        if c8 < best_c:
            best_c, best_h = c8, h8.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
