        cval = float(c.max() * dx)
        if cval < best_c:
            best_c = cval
            best_h = h.copy()
    
    return best_h, best_c

For a 0-1 valued h with integral 1 (so h=1 on a set of measure 1), the overlap is:
∫ h(x)(1-h(x+k)) dx = ∫_{S} (1-h(x+k)) dx = |S| - |S ∩ (S-k)| = 1 - |S ∩ (S-k)|

So C₅ = max_k (1 - |S ∩ (S-k)|) = 1 - min_k |S ∩ (S-k)|

To minimize C₅, we want to maximize min_k |S ∩ (S-k)|.

This is related to the problem of finding a set S ⊂ [0,2] of measure 1 that maximizes its minimum autocorrelation. This is a classical problem in combinatorics/additive number theory!

For S = [0,1] (an interval), |S ∩ (S-k)| = |{x: x∈[0,1] and x+k∈[0,1]}| = max(0, 1-|k|) for k ∈ [-1,1], and 0 for |k|>1.
So min_k |S∩(S-k)| = 0 (at k=±2, for example). C₅ = 1.

For S = [0, 0.5] ∪ [1.5, 2] (two intervals of length 0.5):
|S ∩ (S-k)| is more complex but might give a better minimum.

Actually, the best known constructions for this problem (in the 0-1 case) are related to "Nelson-Sobolev" constructions or "Ruzsa" constructions. But for our purposes, the continuous relaxation (h ∈ [0,1]) should give better results than the 0-1 case.

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_descent(h, n, deadline, mu=0.92):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = dx * 2.0
    momentum = np.zeros(n)

            break

        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n - lag]
        else:

                grad[ll:] += (1.0 - h[:n - ll])
                grad[:n - ll] -= h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)

    return best_h, best_c

    lr = 1.0 / (alpha * dx)
    momentum = np.zeros(n)
    for step in range(500000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        sh = corr - corr.max()
        p = np.exp(alpha * sh)
        ps = p.sum()
        if ps < 1e-30:
            break
        p /= ps
        S_plus = np.zeros(n)
        S_minus = np.zeros(n)
        for li in range(2 * n - 1):
            l = li - (n - 1)
            w = p[li]
            if w < 1e-10:
                continue
            if 0 <= l < n:
                S_plus[:n - l] += w * h[l:]
                S_minus[l:] += w * h[:n - l]
            elif -n < l < 0:
                ll = -l
                S_plus[ll:] += w * h[:n - ll]
                S_minus[:n - ll] += w * h[ll:]
        grad = dx * ((1.0 - S_plus) - S_minus)
        momentum = mu * momentum - lr * grad
        h = project_feasible(h + momentum, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Exact coordinate descent (fast exploitative)
    p1 = t0 + 180
    h1, c1 = exact_coord_descent(h, n, p1, max_sweeps=10000)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: SLSQP moderate smoothing
    p2 = t0 + 330
    if time.time() < p2:
        h2 = best_h.copy()
        for alpha in [500, 5000, 50000]:
            if time.time() > p2 - 15:
                break
            h2 = slsqp_smooth(h2, n, alpha, 2000, 1e-14)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()

    # Phase 3: Subgradient exploration
    p3 = t0 + 520
    if time.time() < p3:
        h3, c3 = subgrad_descent(best_h.copy(), n, p3, mu=0.93)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Coordinate descent after exploration
    p4 = t0 + 700
    if time.time() < p4:
        h4, c4 = exact_coord_descent(best_h, n, p4, max_sweeps=10000)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Aggressive SLSQP + smooth grad
    p5 = deadline - 120
    if time.time() < p5:
        h5 = best_h.copy()
        for alpha in [200000, 2000000, 20000000]:
            if time.time() > p5 - 25:
                break
            h5 = slsqp_smooth(h5, n, alpha, 2500, 1e-15)
            c5 = compute_c5(h5, n)
            if c5 < best_c:
                best_c, best_h = c5, h5.copy()
        if time.time() < p5 - 10:
            h5b, c5b = smooth_grad_descent(best_h.copy(), n, 100000, p5, mu=0.88)
            if c5b < best_c:
                best_c, best_h = c5b, h5b.copy()

    # Phase 6: Noise restarts with coord descent
    if time.time() < deadline - 80:
        for trial in range(6):
            if time.time() > deadline - 80:
                break
            nl = 0.004 * (1 + 0.5 * trial)
            h6 = best_h + nl * rng.randn(n)
            h6 = project_feasible(h6, n)
            h6, c6 = exact_coord_descent(h6, n, deadline - 80, max_sweeps=3000)
            if c6 < best_c:
                best_c, best_h = c6, h6.copy()

    # Phase 7: Multi-scale
    if time.time() < deadline - 50:
        for ratio in [2, 3]:
            nl = n // ratio
            if nl < 50:
                continue
            xo = np.linspace(0, 2, n)
            xl = np.linspace(0, 2, nl)
            hl = np.interp(xl, xo, best_h)
            hl = project_feasible(hl, nl)
            hl, ch = exact_coord_descent(hl, nl, deadline - 50, max_sweeps=5000)
            hl_hi = np.interp(xo, xl, hl)
            hl_hi = project_feasible(hl_hi, n)
            hl_hi, ch2 = exact_coord_descent(hl_hi, n, deadline - 40, max_sweeps=2000)
            ch2 = compute_c5(hl_hi, n)
            if ch2 < best_c:
                best_c, best_h = ch2, hl_hi.copy()
            if time.time() > deadline - 50:
                break

    # Final polish
    if time.time() < deadline - 5:
        hf, cf = exact_coord_descent(best_h, n, deadline - 3, max_sweeps=5000)
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
