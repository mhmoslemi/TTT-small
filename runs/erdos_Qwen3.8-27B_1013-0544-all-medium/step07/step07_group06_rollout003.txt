        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

def greedy_binary(h, n, deadline, max_sweeps=50):
    """Greedy improvement using single swaps."""
    if n % 2 != 0:
        return h.copy(), int(round(np.max(np.correlate(h, 1-h, mode='full'))))
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    best_h = h.copy()
    best_M = M
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        base = make_base_matrix(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        # Process in chunks to manage memory
        chunk = min(10000, zi.size)
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                return best_h, best_M
            end = min(start + chunk, zi.size)
            i_idx = zi[start:end]
            j_idx = oj[start:end]
            arr = base[i_idx].astype(float) - base[j_idx].astype(float)
            arr += corr[np.newaxis, :]
            d1 = n - 1 + (j_idx - i_idx)
            d2 = n - 1 + (i_idx - j_idx)
            valid1 = (d1 >= 0) & (d1 < 2*n-1)
            valid2 = (d2 >= 0) & (d2 < 2*n-1)
            for row in range(arr.shape[0]):
                if valid1[row]:
                    arr[row, d1[row]] += 1.0
                if valid2[row]:
                    arr[row, d2[row]] += 1.0
            arr[:, n-1] -= 2.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv = int(i_idx[loc])
                jv = int(j_idx[loc])
                h[iv] = 1.0
                h[jv] = 0.0
                corr = arr[loc].astype(float).copy()
                M = int(round(float(corr.max())))
                if M < best_M:
                    best_M = M
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_M

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        a = np.resize(a, n)
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(a.min()) - 1.0, float(a.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15: break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15: break
            h += diff * h / s

def top_n_binary(h, n):
    hb = np.zeros(n)
    hb[np.argsort(-h)[:n // 2]] = 1.0

def make_base(h, n):
    l = np.arange(-(n - 1), n, dtype=int)
    base = np.zeros((n, 2 * n - 1), dtype=np.float64)

        f = p + l
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1.0 - h[f[mf]]
        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

def greedy_binary(h, n, deadline, max_sweeps=80):
    if n % 2 != 0:
        return h.copy(), int(round(np.max(np.correlate(h, 1 - h, mode="full"))))
    h = h.astype(float).copy()

    M = int(round(float(corr.max())))
    best_h, best_M = h.copy(), M
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        base = make_base(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:

        zi = np.repeat(zeros, ones.size)

        chunk = min(20000, zi.size)

                return best_h, best_M
            en = min(st + chunk, zi.size)
            ii, jj = zi[st:en], oj[st:en]
            arr = base[ii].copy() - base[jj]
            arr += corr[np.newaxis, :]
            d1 = n - 1 + (jj - ii)
            d2 = n - 1 + (ii - jj)
            rows = np.arange(arr.shape[0])
            v1 = (d1 >= 0) & (d1 < 2 * n - 1)
            v2 = (d2 >= 0) & (d2 < 2 * n - 1)
            arr[rows[v1], d1[v1]] += 1.0

def ils_binary(h_init, n, deadline, rng, n_iters=200):

            best_h = h_new.copy()

def lse_obj_grad(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return m + 10.0, np.zeros(n)
    p = e / s
    F = m + np.log(s) / beta
    P, R = p, p[::-1]
    v = 1.0 - h
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.continuous_conv = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    return F, g

def continuous_polish(h, n, deadline, betas=(500.0, 2000.0, 8000.0)):
    work = project_feasible(h, n)
    best, bc = work.copy(), compute_c5(work, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 10:
            break
        _stop_time = min(deadline - 5, time.time() + 40.0)
        def obj(x):
            hh = np.clip(x, 0.0, 1.0)
            F, g = lse_obj_grad(hh, n, beta)
            dev = float(hh.sum() - 0.5 * n)
            F += 5e3 * dev * dev
            g = g + 1e4 * dev
            return F, g

                           bounds=bounds, callback=stop_callback,
                           options={"maxiter": 50000, "maxls": 30, "ftol": 1e-12})
            work = project_feasible(res.x, n)
        except Exception:
            pass
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
    return best, bc
