            if s < 1e-14:

        if c4 < best_c:
            best_c, best_h = c4, h4.copy()
    # Phase 5: Binary refinement + continuous re-optimize
    t5 = t0 + 850
    if time.time() < t5 - 30:
        h5b, c5b = binary_refine(best_h, n, t5 - 30)
        if c5b < best_c:
            best_c, best_h = c5b, h5b.copy()
        h5c = slsqp_active(h5b, n, 15, 50000, 1500, 1e-15)
        c5c = compute_c5(h5c, n)
        if c5c < best_c:
            best_c, best_h = c5c, h5c.copy()
    # Phase 6: Perturbation restarts + SLSQP
    t6 = deadline - 80
    for trial in range(6):
        if time.time() > t6 - 5:
            break
        noise = 0.002 * (1 + trial * 0.7)
        h6 = best_h + noise * rng.randn(n)
        h6 = project_feasible(h6, n)
        h6 = slsqp_active(h6, n, 10, 100000, 1500, 1e-15)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()
    # Phase 7: Final full SLSQP polish
    t7 = deadline - 10
    if time.time() < t7 - 5:
        h7 = slsqp_full(best_h, n, 20000000, 3000, 1e-17)
        c7 = compute_c5(h7, n)
        if c7 < best_c:
            best_c, best_h = c7, h7.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
