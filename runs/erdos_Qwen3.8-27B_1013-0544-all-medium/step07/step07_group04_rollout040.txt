This makes the pair search very efficient:
1. Compute d = v_i - v_j (O(n) vector)
2. Find ax1* = argmin_{x} max_k (c_old[k] + x * d[k]) subject to constraints on x

The constraint on x: new h[i] = h[i] + x ∈ [0,1] and new h[j] = h[j] - x ∈ [0,1]
So x ∈ [max(-h[i], -(1-h[j])), min(1-h[i], h[j])]
   = [max(-h[i], h[j]-1), min(1-h[i], h[j])]

And the 1D minimax: min_x max_k (a_k + b_k * x) where a_k = c_old[k], b_k = d[k].

Since this is a convex piecewise-linear function, the minimum is at a breakpoint or boundary.

Fast algorithm:
1. f(x) = max_k (a_k + b_k * x)
2. The subgradient at x is [min_{k∈argmax} b_k, max_{k∈argmax} b_k]
3. We want 0 in the subgradient

Bisection approach:
- x_lo = max(-h[i], h[j]-1), x_hi = min(1-h[i], h[j])
- At each midpoint, compute f and the argmax set
- If all b_k for k in argmax are > 0, the minimum is to the left
- If all b_k for k in argmax are < 0, the minimum is to the right
- Otherwise, we're at the minimum (or very close)

This converges in O(log(range/eps)) iterations, each O(n). For n=400 and ~50 iterations, that's 20000 ops per candidate. Very fast!

1. Evaluate f at x_lo and x_hi
2. If f is flat (all b_k for argmax are 0), any x is optimal
3. Otherwise, use the fact that the minimum is at a kink: find the x where the argmax changes

    c_new = c_base[None, :] + xs[:, None] * d[None, :]

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1)).copy()
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):

        if abs(d) < 1e-14: break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        if m.any(): v[m] += d / m.sum()
        else: v += d / n
        v = np.clip(v, 0.0, 1.0)

    m = float(c.max())
    if alpha <= 0: return m
    sh = alpha * (c - m)
    am = sh.max()
    return m + (am + np.log(np.exp(sh - am).sum())) / alpha

    m = float(c.max())

        idx = np.where(c == m)[0]

        c_pos = c[off+1:]  # positive lags
        c_neg = c[off-1::-1]  # negative lags (reversed)
        w = np.maximum(c_pos, c_neg) + 1e-10
        w = w ** 3  # emphasize high correlations
        w = w / w.sum()
        d = int(rng.choice(n-1, p=w)) + 1
        
        # Select position
        # Weight by contribution to correlation at lag d
        if d <= n-1:
            contrib = h[:n-d] * (1.0 - h[d:]) + h[d:] * (1.0 - h[:n-d])
        else:
            s = d
            contrib = h[:n-s] * (1.0 - h[s:]) + h[s:] * (1.0 - h[:n-s])  # same thing for d>n/2
        # Wait, for lag d (0 < d < n), c[off+d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])*dx
        # The "contribution" of position i (as the first factor) is h[i]*(1-h[i+d])
        # But we also want positions where the "rate" is high
        # Let's just use h[i]*(1-h[i+d]) as weight
        i_max = n - d
        if i_max > 0:
            wpos = h[:i_max] * (1.0 - h[d:d+i_max]) + 1e-10
            wpos /= wpos.sum()
            i = int(rng.choice(i_max, p=wpos))
        else:
            i = int(rng.randint(0, n))
        j = i + d
        
        # Try the move
        res = pair_best_move(h, c, i, j, n)
        if res is None:
            continue
        new_c5, x_i, x_j = res
        
        # Accept
        if new_c5 < curc:
            h[i] = x_i
            h[j] = x_j
            c = compute_c(h, n)
            curc = new_c5
            if curc < bestc:
                bestc = curc
                besth = h.copy()
        elif rng.random() < np.exp(-(new_c5 - curc) / max(T, 1e-10)):
            h[i] = x_i
            h[j] = x_j
            c = compute_c(h, n)
            curc = new_c5
    
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1)).copy()
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14: break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        if m.any(): v[m] += d / m.sum()
        else: v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0: return m
    sh = alpha * (c - m)
    am = sh.max()
    return m + (am + np.log(np.exp(sh - am).sum())) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        idx = np.where(c == m)[0]
        q = np.zeros(len(c)); q[idx] = 1.0 / len(idx)
    else:
        sh = alpha * (c - m)
        am = sh.max()
        e = np.exp(sh - am)
        q = e / e.sum()
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    for ii in np.flatnonzero(q > 1e-18):
        k = int(ii) - off
        w = float(q[int(ii)])
        if k == 0: g += w * (1.0 - 2.0 * x)
        elif k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        else:
            s = -k
            g[:n-s] += w * (1.0 - x[s:])
            g[s:] -= w * x[:n-s]
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d

def lbfgsb_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,
                       options={"maxiter": maxiter, "ftol": 1e-15, "gtol": 1e-9})
        if res.x is not None: return res.x
    except Exception: pass
    return h

def pair_optimal(h, c_base, i, j, n):
    """Find optimal (h[i], h[j]) keeping sum fixed. Returns (best_c5, new_i, new_j)."""
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0
    lo, hi = max(0.0, S-1.0), min(1.0, S)
    if hi - lo < 1e-14: return None
    L = 2*n - 1
    dvec = np.zeros(L)
    dvec[i:i+n] += 1.0 - 2.0*h
    dvec[j:j+n] -= 1.0 - 2.0*h
    a_lo, a_hi = lo - x0, hi - x0
    xs = np.linspace(a_lo, a_hi, 48)
    vals = (c_base[None,:] + xs[:,None]*dvec[None,:]).max(axis=1)
    k = int(np.argmin(vals))
    a, b = (xs[max(0,k-1)], xs[min(len(xs)-1,k+1)])
    xs2 = np.linspace(a, b, 48)
    vals2 = (c_base[None,:] + xs2[:,None]*dvec[None,:]).max(axis=1)
    k2 = int(np.argmin(vals2))
    xn = float(np.clip(x0 + xs2[k2], lo, hi))
    return float(vals2[k2]), xn, S - xn

def sa_search(h, n, deadline, seed, T0=0.0003):
    """Simulated annealing pair search."""
    rng = np.random.RandomState(seed)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max()); besth = h.copy(); curc = bestc
    off = n - 1; it = 0
    while time.time() < deadline:
        it += 1
        T = T0 * (1.0 - 0.00003 * it)
        if T < 1e-12: T = 1e-12
        # Select lag weighted by correlation
        w = c[1:2*n-1] + 1e-10  # all lags except center? No, include all
        # Actually use |k| weighted
        # Let's weight by c value cubed (emphasize high)
        w = np.maximum(c, 0.0) ** 3 + 1e-10
        w = w / w.sum()
        lag_idx = int(rng.choice(2*n-1, p=w))
        d = lag_idx - off
        if d == 0:
            d = int(rng.randint(1, n))
        s = abs(d)
        if s >= n: continue
        max_i = n - s
        if max_i <= 0: continue
        # Weight position by contribution
        if d > 0:
            wpos = h[:max_i]*(1.0-h[d:d+max_i]) + 1e-12
        else:
            wpos = h[s:s+max_i]*(1.0-h[:max_i]) + 1e-12
        wpos /= wpos.sum()
        if d > 0:
            i = int(rng.choice(max_i, p=wpos))
            j = i + d
        else:
            j = int(rng.choice(max_i, p=wpos))
            i = j + s
        res = pair_optimal(h, c, i, j, n)
        if res is None: continue
        nc, xi, xj = res
        if nc < curc - 1e-15:
            h[i] = xi; h[j] = xj
            c = compute_c(h, n); curc = nc
            if curc < bestc: bestc = curc; besth = h.copy()
        elif rng.random() < np.exp(-(nc-curc)/T):
            h[i] = xi; h[j] = xj
            c = compute_c(h, n); curc = nc
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())
