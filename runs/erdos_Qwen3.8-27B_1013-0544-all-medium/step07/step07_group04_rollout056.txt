import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        return m, np.zeros(n)
    obj = m + np.log(s) / alpha
    q = e / s
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    nz = np.flatnonzero(q > 1e-14)
    for idx in nz:
        w = q[idx]
        k = idx - off
        if k > 0:
            m2 = n - k
            g[k:] += dx * w * (1.0 - x[:m2])
            g[:m2] -= dx * w * x[k:]
        elif k < 0:
            s2 = -k
            m2 = n - s2
            g[s2:] += dx * w * (1.0 - x[:m2])
            g[:m2] -= dx * w * x[s2:]
        else:
            g += dx * w * (1.0 - 2.0 * x)
    return obj, g

    obj, g = lse_obj_grad(x, n, alpha)
    d = x.sum() - 0.5 * n
    return obj + 0.5 * mu * d * d

def pen_g(x, n, alpha, mu):
    obj, g = lse_obj_grad(x, n, alpha)
    d = x.sum() - 0.5 * n
    return g + mu * d

                       jac=pen_g, bounds=bounds,
                       options={"maxiter": maxiter, "ftol": 1e-16, "gtol": 1e-11})
        return res.x if res.x is not None else h
    except Exception:
        return h
