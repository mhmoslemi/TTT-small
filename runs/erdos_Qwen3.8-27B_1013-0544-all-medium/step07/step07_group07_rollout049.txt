import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_obj_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    p_raw = np.exp(shifted)
    S = p_raw.sum()
    p = p_raw / S
    grad = np.zeros(n)
    for l in range(-(n - 1), n):
        w = p[n - 1 + l]
        if w < 1e-18:
            continue
        js = max(0, -l)
        je = min(n, n - l)
        if js < je:
            grad[js:je] += w * (1.0 - x[js + l:je + l])
        js2 = max(0, l)
        je2 = min(n, n + l)
        if js2 < je2:
            grad[js2:je2] -= w * x[js2 - l:je2 - l]
    grad *= dx
    val = m + np.log(S) / alpha
    return val, grad

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        return lse_obj_grad(x, n, alpha)[0]
    def objgrad(x):
        return lse_obj_grad(x, n, alpha)
    def cons(x):
        return x.sum() - n / 2.0
    def consjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=objgrad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons, 'jac': consjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_slsqp(h, n, t, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - t, 0)
        return 0.5 * np.sum(excess ** 2)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_init, n, deadline, max_sweeps=200, sa_temp=0.0):
