import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def topk_smooth(obj_vals, alpha, k):
    """Smooth max of top-k values"""
    topk_idx = np.argsort(obj_vals)[-k:]
    topk = obj_vals[topk_idx]
    m = topk.max()
    sh = alpha * (topk - m)
    # Use log-sum-exp on top-k only
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def obj_grad(x):

        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(x, n, p):
    """Compute gradient given probability weights p over lags"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = p[i] * dx
        if w < 1e-15:
            continue
        lag = i - (n - 1)
        if lag >= 0 and lag < n:
            grad[:n - lag] += w * (1.0 - x[lag:])
            grad[lag:] -= w * x[:n - lag]
        elif lag < 0 and -lag < n:
            ll = -lag
            grad[ll:] += w * (1.0 - x[:n - ll])
            grad[:n - ll] -= w * x[ll:]
    return grad

def lsqb_smooth(x0, n, alpha, penalty, maxiter, topk=0):
    """L-BFGS-B with log-sum-exp smoothed max and penalty for sum constraint"""
    dx = 2.0 / n

    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        if topk > 0 and topk < len(g):
            part = np.argpartition(g, -topk)[-topk:]
            gk = g[part]
            m = gk.max()
            sh = alpha * (gk - m)
            sm = sh.max()
            p = np.exp(sh - sm)
            ps = p.sum()
            p = p / ps
            f = m + (sm + np.log(ps)) / alpha
            pp = np.zeros(len(g))
            pp[part] = p
        else:
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            p = np.exp(sh - sm)
            ps = p.sum()
            p = p / ps
            f = m + (sm + np.log(ps)) / alpha
            pp = p
        # Penalty for sum constraint
        diff = x.sum() - n / 2.0
        f += 0.5 * penalty * diff * diff
        grad = lag_gradient(x, n, pp)
        grad += penalty * diff
        return f, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, x0, method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8,
                                               'ftol': 1e-14, 'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return x0

def subgrad_descent(h, n, deadline, mu=0.92, topk=5):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = dx * 4.0
    momentum = np.zeros(n)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    stall = 0
    for step in range(2000000):
        if time.time() > deadline:
            break
        if topk < len(corr):
            topk_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            topk_idx = np.arange(len(corr))
        p = np.zeros(2 * n - 1)
        for idx in topk_idx:
            p[idx] = 1.0 / len(topk_idx)
        grad = lag_gradient(h, n, p)
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)
        corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
        c = float(corr.max())
        if c < best_c - 1e-11:
            best_c = c
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.003, dx * 25.0)
        else:
            stall += 1
            if stall > 150:
                lr *= 0.94
                mu = max(mu - 0.002, 0.4)
                if lr < dx * 0.003:
                    lr = dx * 0.8
                    stall = 0
        h = h_new
    return best_h, best_c

def slsqp_solve(h, n, alpha, maxiter, ftol, topk=0):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        if topk > 0 and topk < len(g):
            part = np.argpartition(g, -topk)[-topk:]
            g = g[part]
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def multi_resolution(h_best, n, deadline):
    best_h = h_best.copy()
    best_c = compute_c5(best_h, n)
    xo = np.linspace(0, 2, n)
    for nr in [200, 100]:
        if time.time() > deadline:
            break
        xl = np.linspace(0, 2, nr)
        hl = np.interp(xl, xo, best_h)
        hl = project_feasible(hl, nr)
        hl = lsqb_smooth(hl, nr, 5000, 50.0, 300, topk=10)
        hl = lsqb_smooth(hl, nr, 50000, 100.0, 300, topk=5)
        hu = np.interp(xo, xl, hl)
        hu = project_feasible(hu, n)
        hu = lsqb_smooth(hu, n, 100000, 200.0, 500, topk=10)
        cu = compute_c5(hu, n)
        if cu < best_c:
            best_c = cu
            best_h = hu.copy()
    return best_h, best_c
