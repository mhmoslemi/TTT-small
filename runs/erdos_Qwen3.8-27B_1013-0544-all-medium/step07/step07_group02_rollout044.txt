        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = np.array(delta.value).flatten()
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        h_try = h + lam * d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-15:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    G = compute_gradients(h, n)
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        return G.T @ w
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def gen_starts(n, rng, base_h):
    starts = []
    for scale in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
        ht = base_h + scale * rng.randn(n)
        starts.append(project_feasible(ht, n))
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.1, 0.2]:
            ht = base_h + amp * np.sin(2 * np.pi * freq * x + rng.rand() * 2 * np.pi)
            starts.append(project_feasible(ht, n))
    for level in [0.3, 0.5, 0.7]:
        ht = np.where(base_h > level, 0.8, 0.2)
        starts.append(project_feasible(ht, n))
    starts.sort(key=lambda x: compute_c5(x, n))
    return starts[:10]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 400
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_c5 = compute_c5(h_init, n)
    best_h = h_init.copy()
    best_n = n

    # Phase 1: Aggressive LP from initial
    p1_dl = t0 + 0.40 * (budget_s - 5)
    eps = 0.10
    stall = 0
    while time.time() < p1_dl:
        h_cur, mc = lp_step(h_init, n, eps)
        c5v = mc * 2.0 / n
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > 12:
                eps *= 0.7
                if eps < 0.002:
                    break
                stall = 0
        h_init = h_cur

    # Phase 2: Multi-start LP
    p2_dl = t0 + 0.75 * (budget_s - 5)
    starts = gen_starts(n, rng, best_h)
    for sh in starts:
        if time.time() > p2_dl:
            break
        hc = sh.copy()
        ec = 0.07
        st = 0
        while time.time() < p2_dl:
            hc, mc = lp_step(hc, n, ec)
            cv = mc * 2.0 / n
            if cv < best_c5 - 1e-14:
                best_c5 = cv
                best_h = hc.copy()
                st = 0
            else:
                st += 1
                if st > 8:
                    break
            ec = max(ec * 0.994, 0.002)

    # Phase 3: Resolution changes
    p3_dl = t0 + 0.88 * (budget_s - 5)
    for nh in [300, 500, 600]:
        if time.time() > p3_dl:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        ec = 0.05
        for _ in range(80):
            if time.time() > p3_dl - 5:
                break
            hh, mc = lp_step(hh, nh, ec)
            cv = mc * 2.0 / nh
            if cv < best_c5:
                best_c5 = cv
                best_h = hh.copy()
                best_n = nh
            hh = project_feasible(hh, nh)
            ec = max(ec * 0.996, 0.002)

    # Phase 4: SLSQP with gradient
    p4_dl = dl - 4
    hc = best_h.copy()
    for alpha in [500, 5000, 50000, 500000]:
        if time.time() > p4_dl:
            break
        hc = slsqp_smooth(hc, best_n, alpha, 1500)
        cv = compute_c5(hc, best_n)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc.copy()

    # Phase 5: Final fine LP
    if time.time() < dl - 2:
        hc = best_h.copy()
        ef = 0.004
        for _ in range(400):
            if time.time() > dl - 2:
                break
            hc, mc = lp_step(hc, best_n, ef)
            cv = mc * 2.0 / best_n
            if cv < best_c5:
                best_c5 = cv
                best_h = hc.copy()
            ef = max(ef * 0.998, 0.0002)

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, int(best_n)
