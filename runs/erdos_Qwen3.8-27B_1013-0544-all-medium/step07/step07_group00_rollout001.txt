        h_init = structured_start(n_init, 0, rng)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B with alpha annealing on initial n
    h1, c1 = lbfgs_opt(best_h, n_init, [500, 5000, 50000, 200000], 800, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Coordinate descent
    h2, c2 = coord_cd(best_h, n_init, 300, deadline, grid_size=80)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Multi-scale - try smaller n then interpolate back
    x_orig = np.linspace(0, 2, len(best_h))
    for n_small in [80, 120, 160, 240]:
        if time.time() > deadline - 100:
            break
        x_s = np.linspace(0, 2, n_small, endpoint=False)
        h_s = project_feasible(np.interp(x_s, x_orig, best_h) + 0.02 * rng.randn(n_small), n_small)
        h_s, c_s = lbfgs_opt(h_s, n_small, [1000, 10000, 100000], 400, deadline)
        h_s, c_s = coord_cd(h_s, n_small, 100, deadline, grid_size=60)
        # interpolate back to best_h size
        x_big = np.linspace(0, 2, len(best_h), endpoint=False)
        h_back = project_feasible(np.interp(x_big, x_s, h_s), len(best_h))
        h_back, c_back = coord_cd(h_back, len(best_h), 80, deadline, grid_size=60)
        if c_back < best_c:
            best_h, best_c = h_back, c_back
    # Phase 4: Diverse restarts
    if time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(4, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r % 4 == 3:
                hp = structured_start(n, r % 5, rng)
            elif r % 4 == 2:
                hp = project_feasible(rng.uniform(0, 1, n), n)
            else:
                s = 0.01 + 0.06 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = lbfgs_opt(hp, n, [2000, 20000], 400, deadline)
            h_o, c_o = coord_cd(h_o, n, 80, deadline, grid_size=60)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: High-alpha L-BFGS + fine coord descent
    if time.time() < deadline - 15:
        n = len(best_h)
        hf, cf = lbfgs_opt(best_h, n, [200000, 500000, 1000000], 1000, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd(best_h, n, 300, deadline, grid_size=120)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Final polish
    if time.time() < deadline - 2:
        n = len(best_h)
        hf, cf = coord_cd(best_h, n, 100, deadline, grid_size=150)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
