import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_delta(h, n, i, j):
    """Compute the delta in correlation when h[i]:0->1, h[j]:1->0."""
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_local_search(h_init, n, deadline, max_sweeps=300):
    """Best-improvement binary local search with kicks."""
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0].astype(int))
    zeros_list = list(np.where(h == 0)[0].astype(int))
    
    sweep = 0
    no_improve = 0
    while sweep < max_sweeps and time.time() < deadline:
        sweep += 1
        improved = False
        best_delta_c = 0
        best_i = -1
        best_j = -1
        best_delta = None
        
        # Evaluate all (i,j) pairs
        for i in zeros_list:
            for j in ones_list:
                d = compute_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                dc = nc - cur_c
                if dc < best_delta_c - 1e-15:
                    best_delta_c = dc
                    best_i, best_j = i, j
                    best_delta = d
                    improved = True
        
        if improved:
            h[best_i], h[best_j] = 1.0, 0.0
            corr += best_delta
            cur_c += best_delta_c
            ones_list.remove(best_j)
            zeros_list.remove(best_i)
            ones_list.append(best_i)
            zeros_list.append(best_j)
            no_improve = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 3:
                # Kick: randomly swap 3-5 pairs
                n_kicks = np.random.randint(3, 6)
                for _ in range(n_kicks):

                    d = compute_delta(h, n, i, j)

                    corr += d
                    cur_c = corr.max() * dx
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                no_improve = 0
            else:
                break
    
    return best_h, best_c

def precompute_half_deltas(h, n):
    """Precompute the half-delta for each position changing 0->1 and 1->0."""
    m_range = np.arange(-(n - 1), n)
    # delta_plus[i]: change in corr if h[i] goes 0->1 (ignoring interaction with other change)
    # This is: add (1-h[i+k]) at lag k, subtract h[i-k] at lag k, minus 1 at k=0
    delta_plus = np.zeros((n, 2*n-1))
    delta_minus = np.zeros((n, 2*n-1))
    
    for i in range(n):
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta_plus[i, msk] += (1.0 - h[im[msk]])
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta_plus[i, msk] -= h[imn[msk]]
        delta_plus[i, n-1] -= 1.0
        
    for j in range(n):
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta_minus[j, msk] -= (1.0 - h[jm[msk]])
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta_minus[j, msk] += h[jmn[msk]]
        delta_minus[j, n-1] -= 1.0
    
    return delta_plus, delta_minus

Let me also add the **2-swap move**: simultaneously swap two (i,j) pairs. This gives a much larger move space and can escape local optima that single swaps can't.

For a 2-swap (i1: 0→1, j1: 1→0, i2: 0→1, j2: 1→0):
delta = delta(i1,j1) + delta(i2,j2) + interaction terms

The interaction terms are non-zero only if the four positions are related by the lag. For random positions, the interaction is almost always zero (the probability of i1+k=j2 or similar is O(1/n)). So for most pairs, delta ≈ delta(i1,j1) + delta(i2,j2), and we can approximate:
new_max ≈ max(corr + delta(i1,j1) + delta(i2,j2))

This is a good approximation when the two swaps are "far apart" (no interaction). We can use this for a fast approximation and only verify the exact value for promising candidates.

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_search_with_kicks(h_init, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0].astype(int))
    zeros_list = list(np.where(h == 0)[0].astype(int))
    no_improve = 0
    kick_count = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        best_dc = -1e-18
        best_i, best_j, best_d = -1, -1, None
        for i in zeros_list:
            for j in ones_list:
                d = compute_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                dc = nc - cur_c
                if dc < best_dc:
                    best_dc, best_i, best_j, best_d = dc, i, j, d
                    improved = True
        if improved:
            h[best_i], h[best_j] = 1.0, 0.0
            corr += best_d
            cur_c += best_dc
            ones_list.remove(best_j)
            zeros_list.remove(best_i)
            ones_list.append(best_i)
            zeros_list.append(best_j)
            no_improve = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 2:
                kick_size = min(2 + kick_count // 3, 8)
                for _ in range(kick_size):
                    if time.time() > deadline:
                        break
                    ii = zeros_list[np.random.randint(len(zeros_list))]
                    jj = ones_list[np.random.randint(len(ones_list))]
                    d = compute_delta(h, n, ii, jj)
                    h[ii], h[jj] = 1.0, 0.0
                    corr += d
                    ones_list.remove(jj)
                    zeros_list.remove(ii)
                    ones_list.append(ii)
                    zeros_list.append(jj)
                cur_c = corr.max() * dx
                no_improve = 0
                kick_count += 1
                if kick_count > 40:
                    h = best_h.copy()
                    corr = np.correlate(h, 1 - h, mode='full').astype(float)
                    cur_c = corr.max() * dx
                    ones_list = list(np.where(h == 1)[0].astype(int))
                    zeros_list = list(np.where(h == 0)[0].astype(int))
                    kick_count = 0
    return best_h, best_c

def two_swap_search(h_init, n, deadline, max_iters=20000):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i1 = zeros_arr[np.random.randint(len(zeros_arr))]
        j1 = ones_arr[np.random.randint(len(ones_arr))]
        i2 = zeros_arr[np.random.randint(len(zeros_arr))]
        j2 = ones_arr[np.random.randint(len(ones_arr))]
        if i1 == i2 or j1 == j2:
            continue
        d1 = compute_delta(h, n, i1, j1)
        d2 = compute_delta(h, n, i2, j2)
        nc = (corr + d1 + d2).max() * dx
        if nc < cur_c - 1e-15:
            h[i1], h[j1] = 1.0, 0.0
            h[i2], h[j2] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def lse_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pn_optimize(h, n, p, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        c_pos = np.maximum(c, 1e-12)
        return (np.sum(c_pos ** p)) ** (1.0 / p)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_push(h, n, c_cur, dl, n_iters=15):
    dx = 2.0 / n
    best_h, best_c = h.copy(), c_cur
    for i in range(n_iters):
        if time.time() > dl - 3:
            break
        t = best_c * (1.0 - 0.005 * (i + 1))
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            excess = np.maximum(c - t, 0)
            return 0.5 * np.sum(excess ** 2)
        def cons(x):
            return x.sum() - n / 2.0
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons}],
                           options={'maxiter': 3000, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            break
    return best_h, best_c

def refine_continuous(h, n, dl):
    c = compute_c5(h, n)
    for alpha in [1000, 10000, 50000, 200000, 1000000, 5000000, 20000000]:
        if time.time() > dl - 8:
            break
        mi = min(3000, int((dl - time.time()) * 0.15) * 50)
        mi = max(mi, 500)
        h = lse_optimize(h, n, alpha, mi, 1e-16)
        c = compute_c5(h, n)
    for p in [50, 100, 300, 800, 2000]:
        if time.time() > dl - 5:
            break
        h = pn_optimize(h, n, p, 2000, 1e-16)
        c = compute_c5(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary local search with kicks (30% of time)
    p1 = t0 + 0.30 * total
    hb, cb = binary_search_with_kicks(h0, n, p1, max_sweeps=400)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Two-swap search from best (10%)
    p2 = t0 + 0.40 * total
    h2, c2 = two_swap_search(best_h, n, p2, max_iters=30000)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()

    # Phase 3: Continuous refinement (25%)
    p3 = t0 + 0.65 * total
    h = best_h.copy()
    h, c = refine_continuous(h, n, p3 - 10)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Barrier push (10%)
    p4 = t0 + 0.75 * total
    h, c = barrier_push(best_h, n, best_c, p4)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: Multi-start from best (12%)
    p5 = t0 + 0.87 * total
    x = np.linspace(0, 2, n)
    perts = [
        0.01 * rng.randn(n),
        0.02 * np.sin(2 * np.pi * 3.0 * x + 0.5),
        0.015 * np.cos(2 * np.pi * 7.0 * x),
        0.005 * rng.randn(n),
        0.025 * np.sin(2 * np.pi * 1.5 * x + 1.0),
    ]
    for pert in perts:
        if time.time() > p5 - 10:
            break
        hc = project_feasible(best_h + pert, n)
        hc = to_binary(hc, n)
        hc, cc = binary_search_with_kicks(hc, n, min(p5 - 10, time.time() + 8), max_sweeps=40)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()
        if time.time() > p5 - 10:
            break
        hc, cc = refine_continuous(hc, n, p5 - 8)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 6: Try other resolutions (6%)
    p6 = t0 + 0.93 * total
    for nh in [384, 416, 448]:
        if time.time() > p6 - 15:
            break
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, x, best_h)
        hh = project_feasible(hh, nh)
        hh = to_binary(hh, nh)
        hh, ch = binary_search_with_kicks(hh, nh, min(p6 - 15, time.time() + 10), max_sweeps=30)
        hh, ch = refine_continuous(hh, nh, p6 - 12)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
            x = xn

    # Phase 7: Final polish (7%)
    h, c = barrier_push(best_h, n, best_c, dl - 10)
    if c < best_c:
        best_c, best_h = c, h.copy()
    for alpha in [50000000, 200000000]:
        if time.time() > dl - 5:
            break
        best_h = lse_optimize(best_h, n, alpha, 4000, 1e-16)
        c = compute_c5(best_h, n)
        if c < best_c:
            best_c = c

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
