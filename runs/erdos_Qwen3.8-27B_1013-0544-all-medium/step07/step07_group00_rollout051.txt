import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h * (1.0 + diff / tot), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12 and n > 0:
        if diff > 0:
            room = 1.0 - h
            k = min(n, int(abs(diff)) + 1)
            idx = np.argpartition(-room, k)[:k]
            h[idx] = np.clip(h[idx] + diff / k, 0.0, 1.0)
        else:
            k = min(n, int(abs(diff)) + 1)
            idx = np.argpartition(h, k)[:k]
            h[idx] = np.clip(h[idx] + diff / k, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=800, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fast_coord_descent(h, n, max_passes=1000, deadline=None):
    h = h.copy()
    dx = 2.0 / n
    nc = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    G = 150
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            a = np.zeros(nc)
            m_lo = max(0, p)
            m_hi = min(nc - 1, p + n - 1)
            if m_lo <= m_hi:
                i_arr = p + n - 1 - np.arange(m_lo, m_hi + 1)
                a[m_lo:m_hi + 1] += (1.0 - h[i_arr])
            m_lo2 = max(0, n - 1 - p)
            m_hi2 = min(nc - 1, 2 * n - 1 - p)
            if m_lo2 <= m_hi2:
                j_arr = p + np.arange(m_lo2, m_hi2 + 1) - n + 1
                a[m_lo2:m_hi2 + 1] -= h[j_arr]
            d_grid = np.linspace(-h0, 1.0 - h0, G)
            aff = a[None, :] * d_grid[:, None]
            aff_max = (c + aff).max(axis=1)
            quad = c[n - 1] + a[n - 1] * d_grid - d_grid * d_grid
            F_vals = np.maximum(aff_max, quad)
            best_idx = F_vals.argmin()
            if 0 < best_idx < G - 1:
                lo = d_grid[best_idx - 1]
                hi = d_grid[best_idx + 1]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(20):
                    d1 = hi - gr * (hi - lo)
                    d2 = lo + gr * (hi - lo)
                    F1 = max(float(np.max(c + a * d1)), c[n-1] + a[n-1]*d1 - d1*d1)
                    F2 = max(float(np.max(c + a * d2)), c[n-1] + a[n-1]*d2 - d2*d2)
                    if F1 < F2:
                        hi = d2
                    else:
                        lo = d1
                best_d = (lo + hi) / 2.0
            else:
                best_d = d_grid[best_idx]
            if abs(best_d) > 1e-10:
                h[p] = h0 + best_d
                c += best_d * a
                c[n - 1] -= best_d * best_d
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgradient_step(h, n, eta, k_active=10):
    c = np.correlate(h, 1.0 - h, mode='full')
    nc = len(c)
    top_idx = np.argsort(c)[::-1][:k_active]
    top_avg = np.mean(top_idx)
    active = top_idx[top_idx >= top_avg - 1.0] if len(top_idx) > 2 else top_idx
    grad = np.zeros(n)
    for m in active:
        m_lo = max(0, int(m) - (n - 1))
        m_hi = min(n - 1, int(m))
        for i in range(max(0, n - 1 - int(m)), min(n, int(m) + 1)):
            j = i + int(m) - (n - 1)
            if 0 <= i < n and 0 <= j < n:
                grad[j] += (1.0 - h[i])
                grad[i] -= h[j]
    h_new = h * np.exp(-eta * grad)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new
