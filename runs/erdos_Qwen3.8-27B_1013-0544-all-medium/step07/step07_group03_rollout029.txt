    """Compute new max corr for a batch of swaps (i_arr[k]: 0->1, j_arr[k]: 1->0)."""
    m = len(i_arr)
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    batch_size = 500
    results = np.zeros(m)
    
    for start in range(0, m, batch_size):
        end = min(start + batch_size, m)
        ib = i_arr[start:end]
        jb = j_arr[start:end]
        b = end - start
        
        # For each swap (i, j), delta = A[i] - B[i] - A[j] + B[j]
        # where A[i][p] = 1-h[i+m_range[p]] (valid part) and B[i][p] = h[i-m_range[p]] (valid part)
        
        # A[i]: for position i, valid p are those where 0 <= i+m_range[p] < n
        # m_range[p] = p - (n-1), so valid when 0 <= i + p - (n-1) < n, i.e., (n-1)-i <= p <= n-1+i
        # In terms of the actual index: h[i + p - (n-1)] for p in [(n-1)-i, n-1+i]
        
        # Let's compute A for the batch:
        # A[b_idx, p] = 1 - h[ib[b_idx] + p - (n-1)] (when valid)
        
        # Vectorized: for b_idx in range(b), p in range(2n-1):
        #   idx = ib[b_idx] + p - (n-1)
        #   if 0 <= idx < n: A[b_idx, p] = 1 - h[idx]

        # Instead, compute the score directly:

        # (where each term is 0 when the index is out of range)

        # We can compute D for all ib and jb values:
        all_ids = np.unique(np.concatenate([ib, jb]))

The overlap at shift k is:
overlap(k) = ∑_i h[i] * (1-h[i+k]) * dx

For binary h with n/2 ones, this is:

For k=1: overlap(1) = (# of adjacent 1-0 pairs) * dx = (# of 1s) - (# of adjacent 1-1 pairs) * dx... no, that's not right.

= total # of 1s in [0, n-1-k] - # of 1s at positions i where h[i+k]=1

This gives a lower bound on M:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_bin(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def slsqp_opt(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_f(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints={'type': 'eq', 'fun': cons_f},
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_delta(h_b, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h_b[im[msk]])
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h_b[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h_b[jm[msk]])
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h_b[jmn[msk]]
    return delta
