    """Gradient of c[m] = correlate(h, 1-h, 'full')[m] w.r.t. h."""
    grad = np.zeros(n)

            grad[n_idx] += (1.0 - h[other])
            grad[other] -= h[n_idx]
        else:

    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    stall = 0
    stall_limit = 5000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            # Restart from best with perturbation
            h = best_h + 0.02 * np.random.randn(n)
            h = project_feasible(h, n)
            m_acc = np.zeros(n)
            v_acc = np.zeros(n)
            stall = 0
            continue
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.exp(c[top_idx])
        weights /= weights.sum()
        g = np.zeros(n)
        for w, m_idx in zip(weights, top_idx):
            if w > 1e-8:
                g += w * grad_c_at_m(h, n, m_idx)

        h_new = project_feasible(h_new, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
        if c_new.max() < cur_c + 0.01:
            h = h_new
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_c_at_m_vec(h, n, m):
    grad = np.zeros(n)
    lo = max(0, m - (n - 1))
    hi = min(n - 1, m)
    indices = np.arange(lo, hi + 1)
    others = m - indices
    valid = (others >= 0) & (others <= n - 1)
    grad[indices] += 1.0
    grad[indices[valid]] -= h[others[valid]]
    if np.any(valid):
        np.add.at(grad, others[valid], -h[indices[valid]])
    return grad

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        mval = c.max()
        return mval + np.log(np.exp(alpha * (c - mval)).sum()) / alpha
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

            stall = 0
            continue

        raw_w = c[top_idx] - cur_c
        w = np.exp(20.0 * raw_w)

def swap_delta_vec(h, n, i, j):
    m_range = np.arange(2 * n - 1)
    r = m_range - i
    s = m_range - j
    valid_r = (r >= 0) & (r < n)
    delta_i = np.ones(2 * n - 1)
    delta_i[valid_r] = 1.0 - 2.0 * h[r[valid_r]]
    mask_ri = (r == i)
    delta_i[mask_ri] = -1.0
    mask_rj = (r == j)
    delta_i[mask_rj] = 1.0
    valid_s = (s >= 0) & (s < n)
    delta_j = np.full(2 * n - 1, -1.0)
    delta_j[valid_s] = 2.0 * h[s[valid_s]] - 1.0
    mask_si = (s == i)
    delta_j[mask_si] = 0.0
    mask_sj = (s == j)
    delta_j[mask_sj] = 0.0
    return delta_i + delta_j

def binary_local_search(h_init, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n // 2]] = 1.0
    corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in zeros:
            for j in ones:
                delta = swap_delta_vec(h_b, n, i, j)
                nc = (corr + delta).max() * dx
                gain = cur_c - nc
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        if best_i >= 0:
            delta = swap_delta_vec(h_b, n, best_i, best_j)
            h_b[best_i], h_b[best_j] = 1.0, 0.0
            corr += delta
            cur_c = best_c - best_gain if False else (corr.max() * dx)
            ones.remove(best_j)
            ones.append(best_i)
            zeros.remove(best_i)
            zeros.append(best_j)
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
        if not improved:
            break
    return best_h, best_c
