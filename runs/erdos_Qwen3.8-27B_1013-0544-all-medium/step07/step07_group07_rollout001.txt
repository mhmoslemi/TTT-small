def project_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    if h.size != n:
        return h
    np.clip(h,0.0,1.0,out=h)
    target = n/2.0
    for _ in range(200):

            if sr < 1e-14: break

            h = np.pad(h, (0, n - h.size), mode="edge")
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    for _ in range(300):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            sr = float(room.sum())
            if sr < 1e-14:
                break
            h += (diff / sr) * room
        else:
            if s < 1e-14:
                break
            h += (diff / s) * h
        np.clip(h, 0.0, 1.0, out=h)
    return h


def to_binary(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.size), mode="edge")
    hb = np.zeros(n, dtype=float)
    hb[np.argsort(-h)[:n // 2]] = 1.0
    return hb


def compute_c(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def weighted_grad(h, w, dx):
    n = len(h)
    out = np.zeros(n, dtype=float)
    base = n - 1
    nz = np.flatnonzero(w > 1e-15)
    for t in nz:
        d = int(t - base)
        wt = float(w[t])
        if d >= 0:
            L = n - d
            if L <= 0:
                continue
            out[:L] += wt * (1.0 - h[d:])
            if d == 0:
                out[:] -= wt * h
            else:
                out[d:] -= wt * h[:L]
        else:
            e = -d
            L = n - e
            if L <= 0:
                continue
            out[e:] += wt * (1.0 - h[:L])
            out[:L] -= wt * h[e:]
    out *= dx
    return out


def lse_components(h, alpha):
    c = np.correlate(h, 1.0 - h, mode="full") * _G_DX
    m = float(c.max())
    a = float(alpha)
    if a < 1e8:
        sh = a * (c - m)
        e = np.exp(sh)
        s = float(e.sum())
        if s <= 0.0:
            e = (c - m >= -1e-15).astype(float)
            s = float(e.sum())
        p = e / s
        val = m + math.log(s) / a
    else:
        act = (c >= m - 1e-15).astype(float)
        s = float(act.sum())
        p = act / s
        val = m
    return float(val), p, c


def lbfgs_objective(x):
    val, p, _ = lse_components(x, _G_ALPHA)
    se = float(x.sum()) - _G_TARGET
    val += _G_LAM * se * se
    jac = weighted_grad(x, p, _G_DX)
    if _G_LAM != 0.0:
        jac += (2.0 * _G_LAM * se) * np.ones_like(x)
    return float(val), jac


def lbfgs_lse(h, n, alpha, lam, maxiter):
    global _G_DX, _G_TARGET, _G_ALPHA, _G_LAM
    h0 = project_feasible(h, n)
    _G_DX = 2.0 / float(n)
    _G_TARGET = 0.5 * n
    _G_ALPHA = float(alpha)
    _G_LAM = float(lam)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_objective, h0, method="L-BFGS-B", bounds=bnds, jac=True,
                       options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-5,
                                "maxls": 50, "disp": False})
        x = project_feasible(res.x, n)
    except Exception:
        x = h0
    return x, compute_c(x, n)


def slsqp_objective(x):
    val, _, _ = lse_components(x, _G_ALPHA)
    return val


def slsqp_jacobian(x):
    _, p, _ = lse_components(x, _G_ALPHA)
    return weighted_grad(x, p, _G_DX)


def slsqp_eq(x):
    return float(x.sum()) - _G_TARGET


def slsqp_lse(h, n, alpha, maxiter):
    global _G_DX, _G_TARGET, _G_ALPHA
    h0 = project_feasible(h, n)
    _G_DX = 2.0 / float(n)
    _G_TARGET = 0.5 * n
    _G_ALPHA = float(alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": slsqp_eq}]
    try:
        res = minimize(slsqp_objective, h0, method="SLSQP", bounds=bnds,
                       constraints=cons, jac=slsqp_jacobian,
                       options={"maxiter": maxiter, "ftol": 1e-12, "disp": False})
        x = project_feasible(res.x, n)
    except Exception:
        x = h0
    return x


def hinge_components(h, t):
    c = np.correlate(h, 1.0 - h, mode="full") * _G_DX
    e = np.maximum(c - float(t), 0.0)
    val = float(np.sum(e)) + 0.5 * float(np.dot(e, e))
    w = np.zeros_like(c)
    nz = e > 1e-15
    w[nz] = 1.0 + e[nz]
    return val, w


def hinge_objective(x):
    val, _ = hinge_components(x, _G_T)
    return val


def hinge_jacobian(x):
    _, w = hinge_components(x, _G_T)
    return weighted_grad(x, w, _G_DX)


def slsqp_hinge(h, n, t, maxiter):
    global _G_DX, _G_TARGET, _G_T
    h0 = project_feasible(h, n)
    _G_DX = 2.0 / float(n)
    _G_TARGET = 0.5 * n
    _G_T = float(t)
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": slsqp_eq}]
    try:
        res = minimize(hinge_objective, h0, method="SLSQP", bounds=bns,
                       constraints=cons, jac=hinge_jacobian,
                       options={"maxiter": maxiter, "ftol": 1e-11, "disp": False})
        x = project_feasible(res.x, n)
    except Exception:
        x = h0
    return x


def flip_inplace(h, corr, p, newbit, n):
    old = float(h[p])
    s = float(newbit) - old
    if abs(s) < 1e-12:
        return
    base = n - 1
    st = base - p
    corr[st:st + n] += s * (1.0 - h)
    stop = p - 1
    if stop < 0:
        stop = -1
    corr[base + p:stop:-1] -= s * h
    corr[base] -= s * (1.0 - 2.0 * old)
    h[p] = float(newbit)


def build_bit_lists(h, n):
    ones = []
    zeros = []
    po = np.full(n, -1, dtype=int)
    pz = np.full(n, -1, dtype=int)
    for i in range(n):
        if h[i]:
            po[i] = len(ones)
            ones.append(i)
        else:
            pz[i] = len(zeros)
            zeros.append(i)
    return ones, zeros, po, pz


def binary_phase(h0, n, deadline, seed):
    if n % 2 == 1:
        h = project_feasible(h0, n)
        return h, compute_c(h, n)
    h_arr = np.asarray(h0, dtype=float).reshape(-1)
    if h_arr.size != n:
        if h_arr.size > n:
            h_arr = h_arr[:n]
        else:
            h_arr = np.pad(h_arr, (0, n - h_arr.size), mode="edge")
    hn = np.round(h_arr)
    if np.all(np.abs(h_arr - hn) < 1e-8) and abs(float(h_arr.sum()) - 0.5 * n) < 1e-7:
        h = hn.astype(float).copy()
    else:
        h = to_binary(h_arr, n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    cur = int(np.round(corr.max()))
    best_h = h.copy()
    best_raw = cur
    ones, zeros, po, pz = build_bit_lists(h, n)
    rng = np.random.RandomState(seed)
    temp = 3.0
    min_temp = 0.25
    step = 0
    stalls = 0
    while True:
        if step % 32 == 0 and time.time() > deadline:
            break
        step += 1
        if not ones or not zeros:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        flip_inplace(h, corr, i, 1.0, n)
        flip_inplace(h, corr, j, 0.0, n)
        new = int(np.round(corr.max()))
        if new < best_raw:
            best_raw = new
            best_h = h.copy()
        if new < cur:
            accept = True
        elif new == cur:
            accept = rng.random() < 0.05
        else:
            d = new - cur
            if d <= max(1, int(2.0 * temp)):
                accept = rng.random() < math.exp(-d / temp)
            else:
                accept = False
        if accept:
            cur = new
            idxj = po[j]
            last = ones[-1]
            ones[idxj] = last
            po[last] = idxj
            ones.pop()
            po[j] = -1
            idxi = pz[i]
            last = zeros[-1]
            zeros[idxi] = last
            pz[last] = idxi
            zeros.pop()
            pz[i] = -1
            po[i] = len(ones)
            ones.append(i)
            pz[j] = len(zeros)
            zeros.append(j)
            stalls = 0
        else:
            flip_inplace(h, corr, j, 1.0, n)
            flip_inplace(h, corr, i, 0.0, n)
            if new > cur:
                stalls += new - cur
        if step % 20000 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
        if cur > best_raw + 10 or (step % 20000 == 0 and cur == best_raw and stalls > 20000):
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
            cur = best_raw
            ones, zeros, po, pz = build_bit_lists(h, n)
            temp = max(1.0, temp * 2.0)
            stalls = 0
        temp = max(min_temp, temp * 0.9999 - 1e-5)
    return best_h.copy(), float(best_raw) * (2.0 / float(n))


def interp_h(old_h, n_old, n_new):
    if n_old == n_new:
        return np.asarray(old_h, dtype=float).copy()
    if n_old <= 1:
        return np.full(n_new, float(np.mean(old_h)))
    xo = np.linspace(0.0, 2.0, n_old)
    xn = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(xn, xo, old_h), n_new)


def template_h(n, seed, kind):
    x = np.linspace(0.0, 2.0, n)
    rng = np.random.RandomState(seed + 1001 * kind)
    idx = (kind - 1) % 5
    m = idx + 1
    if m >= 2:
        ph = rng.uniform(0.0, 2.0 * np.pi)
        ph2 = rng.uniform(0.0, 2.0 * np.pi)
        h = 0.5 + 0.46 * np.cos(np.pi * m * x + ph) + 0.10 * np.cos(np.pi * (m + 2) * x + ph2)
    else:
        L = max(3, n // 30)
        F = np.zeros(n, dtype=complex)
        F[:L] = rng.randn(L) + 1j * rng.randn(L)
        v = np.real(np.fft.ifft(F) * n)
        sc = float(np.max(np.abs(v)))
        if sc > 1e-12:
            h = 0.5 + 0.45 * v / sc
        else:
            h = np.full(n, 0.5)
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def random_binary(n, seed):
    if n % 2:
        return project_feasible(np.full(n, 0.5), n)
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def fractional_polish(h_in, n, deadline, seed):
    h = project_feasible(h_in, n)
    c = compute_c(h, n)
    lb_iters = 260 if n < 200 else (170 if n < 350 else 110)
    alphas = [6, 15, 40, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]
    for a in alphas:
        if time.time() > deadline - 90:
            break
        x1, c1 = lbfgs_lse(h, n, a, 50.0, lb_iters)
        if np.isfinite(c1) and c1 < c - 1e-14:
            h, c = x1, c1
    sl_iter = 180 if n < 200 else (120 if n < 350 else 80)
    for a in [1000, 20000, 200000, 2000000]:
        if time.time() > deadline - 50:
            break
        x1 = slsqp_lse(h, n, a, sl_iter)
        c1 = compute_c(x1, n)
        if np.isfinite(c1) and c1 < c - 1e-14:
            h, c = x1, c1
    hg_iter = 160 if n < 200 else (100 if n < 350 else 60)
    decs = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    for _ in range(2):
        for dec in decs:
            if time.time() > deadline - 4:
                break
            t = c - dec
            x1 = slsqp_hinge(h, n, t, hg_iter)
            c1 = compute_c(x1, n)
            if np.isfinite(c1) and c1 < c - 1e-14:
                h, c = x1, c1
                break
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(60.0, float(budget_s) - 20.0)
    deadline = t0 + total

    parent = None
    iv = globals().get("initial_h_values", None)
    if iv is not None:
        try:
            parent = np.asarray(iv, dtype=float).copy()
        except Exception:
            parent = None
    if parent is None or parent.size == 0:
        parent = np.full(400, 0.5)
    n_parent = int(len(parent))
    parent = project_feasible(parent, n_parent)

    best_h = parent.copy()
    best_n = n_parent
    best_c = compute_c(best_h, best_n)

    res_list = [n_parent]
    for b in [400, 320, 240, 180, 140]:
        if b <= n_parent and b not in res_list:
            res_list.append(b)
    res_list = sorted(set(res_list), reverse=True)
    cleaned = []
    for n in res_list:
        if n >= 40 and n not in cleaned:
            cleaned.append(int(n))
    res_list = cleaned

    fracs = [0.44, 0.58, 0.70, 0.80, 0.88, 0.94]
    idx = 0
    for n in res_list:
        if idx >= len(fracs):
            break
        phase_dl = t0 + total * fracs[idx]
        if time.time() > phase_dl - 12:
            idx += 1
            continue

        if n == best_n:
            h0 = best_h.copy()
        else:
            h0 = interp_h(best_h, best_n, n)

        c0 = compute_c(h0, n)
        h0 = project_feasible(h0, n)
        if time.time() < phase_dl - 35:
            for k in (1, 2):
                ht = template_h(n, int(seed) + 97 * idx + k, k)
                ct = compute_c(ht, n)
                if ct < c0:
                    h0, c0 = ht, ct
                if n <= 280 and k == 1:
                    hr = random_binary(n, int(seed) + 123 * idx + 5)
                    cr = compute_c(hr, n)
                    if cr < c0:
                        h0, c0 = hr, cr

        if n % 2 == 0 and time.time() < phase_dl - 25:
            btime = max(10.0, 0.07 * total if n == best_n or n == 400 else 0.035 * total)
            bdl = min(phase_dl - 25, time.time() + btime)
            hb, cb = binary_phase(h0, n, bdl, int(seed) + 151 * idx + 11)
            if cb < c0:
                h0, c0 = hb, cb
            if cb < best_c:
                best_h, best_n, best_c = hb.copy(), n, cb

        if time.time() < phase_dl - 8:
            hf, cf = fractional_polish(h0, n, phase_dl, int(seed) + 211 * idx + 17)
            if np.isfinite(cf) and cf < best_c:
                best_h, best_n, best_c = hf.copy(), n, cf
        idx += 1

    if time.time() < deadline - 35:
        hf, cf = fractional_polish(best_h, best_n, deadline - 32, int(seed) + 1001)
        if np.isfinite(cf) and cf < best_c:
            best_h, best_n, best_c = hf.copy(), best_n, cf

    targets = [0.38070, 0.38075, 0.38078, 0.38080, 0.38082, 0.38084]
    for tg in targets:
        if time.time() > deadline - 8:
            break
        if tg < best_c - 1e-7:
            x1 = slsqp_hinge(best_h, best_n, max(0.0, tg), 55)
            c1 = compute_c(x1, best_n)
            if np.isfinite(c1) and c1 < best_c:
                best_h, best_c = x1, c1

    if best_n % 2 == 0 and time.time() < deadline - 4:
        hb, cb = binary_phase(best_h, best_n, deadline - 1, int(seed) + 7713)
        if cb < best_c:
            best_h, best_c = hb, cb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c(best_h, best_n)
    return best_h, float(best_c), int(best_n)
