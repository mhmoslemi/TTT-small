```python
def proj_box_hyperplane(x, target):
    """Project x onto {0<=y<=1, sum(y)=target}"""
    # Binary search on lambda
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2, 0.0, 1.0)
```

```python
        current_max = float(c.max())
        improved = False

            if np.any(valid_j):
                a[valid_j] -= h[j_arr[valid_j]]

            unaff = 0.0

            if k_hi < 2*n-2:

                        if d_lo < d < d_hi:

            # Intersections with diagonal
            if np.any(diag_in_aff):

                cd = c_aff[di]
                ad = a[di]
                # c_aff[i] + a[i]*d = cd + ad*d - d^2
                # d^2 + (a[i]-ad)*d + (c_aff[i]-cd) = 0
                for ii in range(len(active_idx)):
                    i_idx = active_idx[ii]
                    if ks[i_idx] == n1:
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

    best_c = compute_c5(h, n)

        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: SLSQP initial optimization
    t_slsqp = min(60.0, budget_s * 0.06)
    if time.time() < deadline - (budget_s - t_slsqp):
        h1, c1 = do_slsqp(h, n, [100, 500, 5000, 50000], t_slsqp)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Accelerated gradient with increasing alpha
    if time.time() < deadline - budget_s * 0.5:
        t_grad = min(120.0, (deadline - time.time()) - budget_s * 0.4)
        for alpha_try in [50, 200, 1000, 5000]:
            if time.time() > deadline - budget_s * 0.4:
                break
            h2, c2 = accelerated_smooth_gradient(best_h, n, alpha_try, 200,
                                                 min(time.time() + 30, deadline - budget_s * 0.4 + t0))
            if c2 < best_c:
                best_h, best_c = h2, c2
    # Phase 3: Main coordinate descent
    h3, c3 = coord_cd_exact(best_h, n, max_passes=5000, deadline=deadline - 80, M=30)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Perturbation and restart
    attempts = 0
    while time.time() < deadline - 40 and attempts < 10:
        attempts += 1
        eps_list = [0.005, 0.02, 0.08, 0.15]
        eps = eps_list[attempts % len(eps_list)]
        hp = project_feasible(best_h + rng.normal(0, eps, n), n)
        h4, c4 = coord_cd_exact(hp, n, max_passes=200, deadline=deadline - 80, M=25)
        if time.time() > deadline - 60:
            h4, c4 = coord_cd_exact(hp, n, max_passes=200, deadline=deadline - 40, M=20)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Try different n
    if time.time() < deadline - 120:
        for n_try in [400, 600, 300]:
            if time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            h5, c5 = do_slsqp(h_int, n_try, [500, 5000], 20)
            h5, c5 = coord_cd_exact(h5, n_try, max_passes=300, deadline=deadline - 30, M=25)
            if c5 < best_c:
                best_h, best_c = h5, c5
    # Phase 6: Final polish
    if time.time() < deadline - 3:
        n_f = len(best_h)
        hf, cf = coord_cd_exact(best_h, n_f, max_passes=1000, deadline=deadline - 1, M=35)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
