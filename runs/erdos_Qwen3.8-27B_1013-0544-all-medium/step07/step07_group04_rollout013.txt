        if np.clip(v - mid, 0, 1).sum() > tgt:

    off = n - 1
    g = np.zeros(n)
    nz = np.flatnonzero(q > 1e-8)
    for idx in nz:
        w = q[idx]
        k = idx - off
        if k > 0:
            g[k:] += w * (1.0 - h[:n-k])
            g[:n-k] -= w * h[k:]
        elif k < 0:
            s2 = -k
            g[:n-s2] += w * (1.0 - h[s2:])
            g[s2:] -= w * h[:n-s2]
        else:
            g[:] += w * (1.0 - 2.0 * h)
    return dx * g

def fw_line_search(h, s, n, t0=1.0):
    """Backtracking line search for Frank-Wolfe step."""
    c_curr = float(compute_c(h, n).max())
    t = t0
    for _ in range(30):
        h_try = (1-t)*h + t*s
        c_try = float(compute_c(h_try, n).max())
        if c_try < c_curr:
            return h_try, c_try, t
        t *= 0.5
    return h, c_curr, 0.0

def frank_wolfe(h, n, maxiter=2000, smooth_alpha=100.0):
    """Frank-Wolfe with smoothed subgradient."""
    h = project_box_sum(h, n).copy()
    c_curr = float(compute_c(h, n).max())
    best_h = h.copy()
    best_c = c_curr
    tgt = 0.5 * n
    for it in range(maxiter):
        g = grad_smooth(h, n, smooth_alpha)
        # Linear minimization: find s in constraint set minimizing g^T s
        idx = np.argsort(g)
        s = np.zeros(n)
        n_full = int(tgt)
        frac = tgt - n_full
        s[idx[:n_full]] = 1.0
        if frac > 0:
            s[idx[n_full]] = frac
        # Line search
        h_new, c_new, _ = fw_line_search(h, s, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new >= c_curr - 1e-12:
            # No improvement, try with pure subgradient
            g2 = grad_overlap(h, n)
            idx2 = np.argsort(g2)
            s2 = np.zeros(n)
            s2[idx2[:n_full]] = 1.0
            if frac > 0:
                s2[idx2[n_full]] = frac
            h_new2, c_new2, _ = fw_line_search(h, s2, n)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2.copy()
            h = h_new2 if c_new2 < c_curr else h
            c_curr = min(c_new2, c_curr)
        else:
            h = h_new
            c_curr = c_new
    return best_h, best_c

    # Restore

    h[j] = y0

But actually, the smarter approach from the previous solution computes the delta analytically, which is O(n) per candidate x (not O(n) per compute_c call since we only need the max, and we can update incrementally).

Let me use the analytical delta approach for efficiency:

When we change h[i] by dx_i and h[j] by dx_j (where dx_i + dx_j = 0 to preserve sum), the change in correlation at shift k is:
delta_c[k] = dx * (dx_i * (1-h[i-k mod n]) + dx_j * (1-h[j-k mod n]) - dx_i * h[i+k mod n] - dx_j * h[j+k mod n])

Wait, I need to be more careful. The correlation at "full" mode index p (where p = off + k, k being the shift) is:
c[p] = sum_{valid i} h[i] * (1-h[i-k]) * dx

When we change h[i] → h[i] + δ and h[j] → h[j] - δ:
The change in c[p] comes from:
- Terms where h[i] appears as the first factor: (1-h[i-k]) * δ (if i-k is valid)
- Terms where h[i] appears in (1-h): -h[i+k] * δ (if i+k is valid)  
- Similarly for h[j] with -δ

So delta_c[p] = dx * δ * [(1-h[i-k])_valid - h[i+k]_valid - (1-h[j-k])_valid + h[j+k]_valid]

where _valid means the index is in [0,n-1], otherwise 0.

For a pair search where x changes from x0 to x0+t (and y changes from y0 to y0-t):
delta_c[p] = dx * t * [(1-h[i-k])_valid - h[i+k]_valid - (1-h[j-k])_valid + h[j+k]_valid]

This is linear in t! So for each shift p, the new correlation is:
c_new[p] = c_base[p] + dx * t * D[p]

where D[p] is a fixed vector (independent of t) for a given pair (i,j).

So the maximum of c_new is max_p (c_base[p] + dx * t * D[p]), which is the maximum of a set of linear functions of t. This is a piecewise linear convex function in t, and we can find its minimum by finding where the subgradient contains 0, or by ternary search (since it's convex).

Wait, is it convex? max_p (a_p + b_p * t) where a_p = c_base[p]*dx and b_p = dx * D[p]... Yes, the maximum of linear functions is convex. So we're minimizing a convex piecewise linear function over an interval. The minimum is at a point where the subgradient contains 0, i.e., where the positive and negative slopes balance.

This can be solved by:
1. Finding all breakpoints (where two lines cross)
2. Evaluating at each breakpoint and the endpoints
3. Or: using the fact that the slope changes from negative to positive at the minimum

The slope of c_new[p] with respect to t is dx * D[p]. The maximum of these has slope = dx * D[argmax_p(c_new[p])] at any non-breakpoint.

To find the minimum of max_p(c_base[p] + dx*t*D[p]):
- At t=0, the active set is the set of p that achieve the maximum
- As t increases, the active p might change
- The minimum is where the weighted average slope of the active set equals 0

This is getting complex. Let me just use a simple approach: evaluate at many t values and pick the best.

Actually, the simplest efficient approach: since the function is convex piecewise linear, I can use ternary search or golden section search. Or I can just evaluate at 200 points in [lo, hi] and pick the best. For n=400 and 200 evaluations, each evaluation is O(n) to compute the max of c_base + t*D, so total is O(200*n) = 80000 per pair.                 c_pt = b - gr * (b - a)

            else:

                c_pt = d_pt
                fc = fd

                fd = float((c_base + dx * d_pt * D).max())

        h[j] = y0

    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

        mid = 0.5 * (lo + hi)

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    s = float(np.exp(alpha * (c - m)).sum())
    return m + np.log(max(s, 1e-300)) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    q = e / max(s, 1e-300)
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    nz = np.flatnonzero(q > 1e-10)
    for idx in nz:
        w = q[idx]
        k = int(idx) - off
        if k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            s2 = -k
            g[:n-s2] += w * (1.0 - x[s2:])
            g[s2:] -= w * x[:n-s2]
        else:
            g[:] += w * (1.0 - 2.0 * x)
    return dx * g

    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d
