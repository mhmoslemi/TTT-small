    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            if s < 1e-15:

        if not improved:
            break
        lr = min(lr * 1.15, 0.1)
    return best_h, best_c

def max_steepest(h, n, max_steps, dl):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
    for step in range(max_steps):
        if time.time() > dl:
            break
        grad = max_gradient(h, n)
        curr_c = compute_c5(h, n)
        improved = False
        for trial in range(40):
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < curr_c - 1e-16:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
        lr = min(lr * 1.1, 0.15)
    return best_h, best_c

def binary_local_search(h, n, dl, max_attempts=50000, seed=42):
    rng_local = np.random.RandomState(seed)
    dx = 2.0 / n
    h_b = to_binary(h, n)
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    for attempt in range(max_attempts):
        if time.time() > dl:
            break
        i = zeros[rng_local.randint(len(zeros))]
        j = ones[rng_local.randint(len(ones))]
        h_b[i] = 1.0
        h_b[j] = 0.0
        new_corr = np.correlate(h_b, 1 - h_b, mode='full')
        nc = new_corr.max() * dx
        if nc < cur_c - 1e-15:
            corr = new_corr.astype(float)
            cur_c = nc
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
        else:
            h_b[i] = 0.0
            h_b[j] = 1.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: LSE SLSQP with analytical gradient (main workhorse)
    p1 = t0 + 0.38 * total
    alphas_p1 = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000,
                 1000000, 3000000, 10000000, 30000000]
    for alpha in alphas_p1:
        if time.time() > p1 - 8:
            break
        mi = 3000 if alpha < 10000 else 5000
        h = lse_slsqp_fast(best_h, n, alpha, mi, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Smooth steepest descent (high alpha LSE gradient)
    p2 = t0 + 0.48 * total
    for alpha in [50000000, 100000000, 300000000]:
        if time.time() > p2 - 5:
            break
        h, c = smooth_steepest(best_h, n, alpha, 20000, p2 - 3)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: More LSE SLSQP at very high alpha
    p3 = t0 + 0.65 * total
    for alpha in [50000000, 100000000, 300000000, 800000000]:
        if time.time() > p3 - 8:
            break
        h = lse_slsqp_fast(best_h, n, alpha, 5000, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Max-specific steepest descent
    p4 = t0 + 0.78 * total
    h, c = max_steepest(best_h, n, 30000, p4)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: Binary local search
    p5 = t0 + 0.90 * total
    hb, cb = binary_local_search(best_h, n, p5, max_attempts=80000, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 6: Final LSE + steepest polish
    p6 = dl - 5
    if time.time() < p6:
        for alpha in [500000000, 1500000000]:
            if time.time() > p6 - 5:
                break
            h = lse_slsqp_fast(best_h, n, alpha, 4000, 1e-15)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        if time.time() < p6 - 2:
            h, c = smooth_steepest(best_h, n, 800000000, 10000, p6 - 2)
            if c < best_c:
                best_c, best_h = c, h.copy()
        if time.time() < p6 - 1:
            h, c = max_steepest(best_h, n, 10000, p6 - 1)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
