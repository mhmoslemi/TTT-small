def project_feasible(z,n):
    z=np.asarray(z,dtype=float)
    z=np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target=0.5*n
    if z.min() >= -1e-12 and z.max() <= 1.0+1e-12 and abs(z.sum()-target) <= 1e-9*max(1,n):
       return np.clip(z,0,1)
    low=float(z.min())-2.0
    high=float(z.max())+2.0
    for _ in range(50):
       tau=0.5*(low+high)
       s=np.clip(z-tau,0,1).sum()
       if s>target:
          low=tau
       else:
          high=tau
    y=np.clip(z-0.5*(low+high),0,1)
    res=target-y.sum()
    if res>0:
       mask=y<1.0
    else:
       mask=y>0.0
    if mask.any() and abs(res)>1e-13:
       room=(1.0-y) if res>0 else y
       tot=room.sum()
       if tot>1e-12:
          y[mask] += (res/tot)*room[mask]
    return np.clip(y,0,1)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + float(np.log(s)) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def pen_val(x, n, alpha, mu):
    val, p, _ = smooth_components(x, n, alpha)
    dev = x.sum() - 0.5 * n
    return val + 0.5 * mu * dev * dev

def pen_jac(x, n, alpha, mu):
    _, p, _ = smooth_components(x, n, alpha)
    g = grad_weights(x, n, p)
    dev = x.sum() - 0.5 * n
    g += mu * dev
    return g

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-14, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def lbfgs_stage(h, n, bounds, plans, mu, deadline):

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def eval_pair(h, c, A, i, j, n, grid):
    if i == j:
        return None
    hi, hj = h[i], h[j]
    S = hi + hj
    lo = max(0.0, S - 1.0)
    hiT = min(1.0, S)
    if hiT - lo <= 1e-14:
        return None
    us = np.linspace(lo - hi, hiT - hi, max(3, int(grid)))
    B = A[i] - A[j]
    lines = c + us[:, None] * B[None, :]
    off = n - 1
    u2 = us * us
    lines[:, off] -= 2.0 * u2
    lines[:, off + i - j] += u2
    lines[:, off + j - i] += u2
    vals = lines.max(axis=1)
    idx = int(np.argmin(vals))
    if vals[idx] < float(c.max()) - 1e-9:
        return float(us[idx])
    return None

def targeted_pair_search(h, n, max_tries, grid, deadline, rng):
    if n < 2 or max_tries <= 0:
        return h, compute_c5(h, n)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')

    accepts = 0
    no_improve = 0
    max_accept = min(120, max(40, max_tries // 5))
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.02, 0.0012 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 5 == 4 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n

                    continue
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - (n - 1)

                if m <= 0:

            else:

        if u is None:
            no_improve += 1
            continue
        old = float(h[i] + h[j])
        t = float(np.clip(h[i] + u, 0.0, 1.0))
        u = t - float(h[i])
        if abs(u) < 1e-15:
            no_improve += 1
            continue
        B = A[i] - A[j]
        u2 = u * u
        cn = c + u * B
        off = n - 1
        cn[off] -= 2.0 * u2
        cn[off + i - j] += u2
        cn[off + j - i] += u2
        nm = float(cn.max())
        if nm < cur_max - 1e-9:
            h[i] = t
            h[j] = old - t

            accepts += 1
            no_improve = 0

                c = np.correlate(h, 1.0 - h, mode='full')
        else:
            no_improve += 1
        if no_improve > max(300, max_tries // 2):
            break
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_smooth_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 8):
        a = rng.uniform(-0.45, 0.45)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    h += 0.08 * (x / 2.0 - 0.5)
    return project_feasible(np.clip(h, 0.0, 1.0), n)
