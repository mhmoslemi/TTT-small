                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                c += (t_new - h0) * A - (t_new*t_new - h0*h0) * D
                h[p] = t_new
                current_max = float(np.max(c))
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def pnorm_slsqp(h, n, p_list, maxiter_each=800, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0-x, mode='full')
            s = np.sum(c**p)
            return s**(1.0/p) * (2.0/n)
        def sum_con(x):
            return x.sum() - n/2.0
        def sum_con_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)]*n
        cons = [{'type':'eq','fun':sum_con,'jac':sum_con_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter':maxiter_each,'ftol':1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP p-norm on initial n
    h1, c1 = pnorm_slsqp(best_h, n_init, [10, 50, 200, 800], 600, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Exact coordinate descent
    h2, c2 = exact_cd(best_h, n_init, 300, deadline, grid_pts=30, gs_iters=30)
    if c2 < best_c:
        best_h, best_c = h2, c1  # typo guard
        best_h, best_c = h2, c2
    # Phase 3: Multi-scale exploration
    x_orig = np.linspace(0, 2, len(best_h))
    n_list = [300, 350, 400, 450, 500, 550, 600, 700, 800, 1000]
    if n_init in n_list:
        n_list.remove(n_init)
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        h_int = interpolate_h(best_h, len(best_h), n_try)
        h_opt, c_opt = pnorm_slsqp(h_int, n_try, [50, 300, 1000], 400, deadline)
        h_opt, c_opt = exact_cd(h_opt, n_try, 200, deadline, grid_pts=25, gs_iters=25)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 4: Multi-restart on best n
    n_best = len(best_h)
    if time.time() < deadline - 100:
        n_restarts = max(2, int((deadline - time.time() - 60) / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            if r % 3 == 2:
                hp = rng.uniform(0, 1, n_best)
                hp = project_feasible(hp, n_best)
            else:
                s = 0.01 + 0.06 * (r / max(1, n_restarts-1))
                hp = project_feasible(best_h + rng.normal(0, s, n_best), n_best)
            h_o, c_o = pnorm_slsqp(hp, n_best, [200, 1000], 500, deadline)
            h_o, c_o = exact_cd(h_o, n_best, 150, deadline, grid_pts=25, gs_iters=30)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: Final high-p SLSQP + exact CD
    if time.time() < deadline - 30:
        n = len(best_h)
        hf, cf = pnorm_slsqp(best_h, n, [5000, 20000, 100000], 500, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 10:
        hf, cf = exact_cd(best_h, len(best_h), 200, deadline, grid_pts=40, gs_iters=40)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 6: Ultra-fine CD
    if time.time() < deadline - 3:
        hf, cf = exact_cd(best_h, len(best_h), 100, deadline, grid_pts=60, gs_iters=50)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
