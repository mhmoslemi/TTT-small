def compute_autocorr(h, n):

    # Using FFT for efficiency
    H = np.fft.rfft(h, n=2*n)
    ac = np.fft.irfft(H * np.conj(H), n=2*n)[:n]
    return ac  # ac[0] = R_0, ac[k] = R_k (circular)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_autocorr(h, n):
    H = np.fft.rfft(h, n=2*n)
    ac = np.fft.irfft(H * np.conj(H), n=2*n)[:n]
    return ac

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def chase_min(h, n, deadline, mu=0.88, lr0=0.5):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = lr0
    momentum = np.zeros(n)
    stall = 0
    for step in range(1000000):
        if time.time() > deadline:
            break
        R = compute_autocorr(h, n)
        k_star = int(np.argmin(R[1:])) + 1
        c5 = 2.0/n * (n/2.0 - R[k_star])
        if c5 < best_c - 1e-13:
            best_c = c5
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, 5.0)
        else:
            stall += 1
            if stall > 300:
                lr *= 0.93
                mu_eff = max(mu - 0.002 * (stall // 300), 0.3)
                if stall > 5000:
                    lr = lr0 * 0.5
                    stall = 0
            else:
                mu_eff = mu
        grad = np.roll(h, -k_star) + np.roll(h, k_star)
        momentum = mu_eff * momentum + lr * grad
        h = project_feasible(h + momentum, n)
    return best_h, best_c

def soft_min_grad(h, n, deadline, alpha=50.0, mu=0.85, lr0=0.3):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = lr0
    momentum = np.zeros(n)
    stall = 0
    H = np.fft.fft(h)  # cache
    for step in range(500000):
        if time.time() > deadline:
            break
        R = compute_autocorr(h, n)
        Rs = R[1:]
        m = Rs.min()
        w = np.exp(-alpha * (Rs - m))
        ws = w.sum()
        if ws < 1e-30:
            break
        w = w / ws
        a = np.zeros(n)
        a[1:] = w
        A = np.fft.fft(a)
        H = np.fft.fft(h)
        grad = np.fft.ifft(2.0 * np.real(A) * H).real
        c5 = 2.0/n * (n/2.0 - Rs.min())
        if c5 < best_c - 1e-13:
            best_c = c5
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.02, 2.0)
        else:
            stall += 1
            if stall > 400:
                lr *= 0.92
                if stall > 8000:
                    lr = lr0 * 0.5
                    stall = 0
        momentum = mu * momentum + lr * grad
        h = project_feasible(h + momentum, n)
    return best_h, best_c

def slsqp_refine(h, n, alpha=1000, maxiter=2000):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        a = alpha
        m = np.max(corr) / a
        return np.sum(np.exp(corr - m)) / a / len(corr)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Chase min (fast, O(n) per step)
    p1 = t0 + 180
    h1, c1 = chase_min(best_h, n, p1, mu=0.85, lr0=0.8)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    
    # Phase 2: Soft min gradient (FFT-based, O(n log n) per step)
    p2 = t0 + 350
    h2, c2 = soft_min_grad(best_h, n, p2, alpha=30, mu=0.82, lr0=0.4)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()
    
    # Phase 3: Alternate
    p3 = t0 + 650
    h3 = best_h.copy()
    n_iter = 8
    for i in range(n_iter):
        sub_dl = p3 - (p3 - time.time()) * (n_iter - 1 - i) / n_iter
        if time.time() > sub_dl:
            break
        h3, _ = chase_min(h3, n, sub_dl, mu=0.88, lr0=0.4)
        h3, _ = soft_min_grad(h3, n, sub_dl, alpha=80, mu=0.8, lr0=0.2)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()
    
    # Phase 4: SLSQP polish
    p4 = deadline - 20
    h4 = best_h.copy()
    for a in [200, 1000, 5000]:
        if time.time() > p4 - 15:
            break
        h4 = slsqp_refine(h4, n, alpha=a, maxiter=1500)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()
    
    # Phase 5: Perturbation restarts
    p5 = deadline - 5
    for trial in range(5):
        if time.time() > p5 - 3:
            break
        h5 = best_h + 0.01 * rng.randn(n)
        h5 = project_feasible(h5, n)
        sub_dl = min(p5 - 2, time.time() + (p5 - time.time()) / (5 - trial))
        h5, _ = chase_min(h5, n, sub_dl, mu=0.9, lr0=0.3)
        h5, _ = soft_min_grad(h5, n, sub_dl, alpha=100, mu=0.8, lr0=0.15)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

def alternating_lp(h, n, maxit=50, deadline=None):

    best_h = h.copy()

    odd_idx = list(range(1, n, 2))

For the "chase the max" method:
1. Compute c = np.correlate(h, 1-h, mode='full') 
2. Find k_star = argmax(c)
3. The gradient of c[k_star] with respect to h can be computed directly

For step 3, let me derive it cleanly. Let's denote the output index as K (= k_star), and define lag = K - (n-1).

c_out[K] = Σ_n h[n + lag] * (1 - h[n]) where the sum is over all n such that 0 ≤ n ≤ n-1 and 0 ≤ n+lag ≤ n-1.

Valid n range: max(0, -lag) ≤ n ≤ min(n-1, n-1-lag)

The gradient:
∂c_out[K]/∂h[j] = Σ_n [∂h[n+lag]/∂h[j] * (1-h[n]) + h[n+lag] * ∂(1-h[n])/∂h[j]]
= Σ_n [δ(n+lag, j) * (1-h[n]) - h[n+lag] * δ(n, j)]
= (1-h[j-lag]) * [j-lag in valid range] - h[j+lag] * [j in valid range]

Where:
- j-lag in valid range means: max(0,-lag) ≤ j-lag ≤ min(n-1, n-1-lag)
- j in valid range means: max(0,-lag) ≤ j ≤ min(n-1, n-1-lag)

For simplicity, let me just code it up with the understanding that:
- The first term (1-h[j-lag]) is present when j-lag is a valid n index: 0 ≤ j-lag ≤ n-1-lag (i.e., lag ≤ j ≤ n-1-lag+lag = n-1)
  Wait, let me be more careful.
  
  n = j - lag must satisfy: max(0, -lag) ≤ n ≤ min(n-1, n-1-lag)
  → max(0, -lag) ≤ j-lag ≤ min(n-1, n-1-lag)
  → max(lag, -lag) ≤ j ≤ min(n-1+lag, n-1) ... this is getting messy.

```python
def grad_c_at_k(h, n, K):
    """Gradient of c_out[K] = correlate(h, 1-h, 'full')[K] with respect to h."""
    lag = K - (n - 1)
    grad = np.zeros(n)
    
    # Term 1: (1 - h[n]) for n = j - lag, i.e., h[j] = h[n+lag]
    # Valid n: 0 <= n <= n-1 and 0 <= n+lag <= n-1
    # So n ranges from max(0, -lag) to min(n-1, n-1-lag)
    n_min = max(0, -lag)
    n_max = min(n-1, n-1-lag)
    for n in range(n_min, n_max+1):
        j = n + lag
        grad[j] += (1 - h[n])
    
    # Term 2: -h[n+lag] for n = j
    # Valid n: same range
    for n in range(n_min, n_max+1):
        j = n
        grad[j] -= h[n+lag]
    
    return grad
```

This is O(n) which is fine, but the loop is slow in Python. Let me vectorize:

def grad_c_at_k(h, n, K):
    lag = K - (n - 1)
    grad = np.zeros(n)
    n_min = max(0, -lag)
    n_max = min(n-1, n-1-lag)
    if n_max < n_min:
        return grad
    # Term 1: for n in [n_min, n_max], j = n + lag gets (1 - h[n])
    j_indices = np.arange(n_min, n_max+1) + lag
    grad[j_indices] += (1 - h[n_min:n_max+1])
    # Term 2: for n in [n_min, n_max], j = n gets -h[n+lag]
    grad[n_min:n_max+1] -= h[n_min:n_max+1 + lag]
    return grad
```

This is vectorized and O(n).

Now, one more thing to consider: since c[-lag] > c[lag] (as I showed earlier), the maximum is likely at a negative lag. For the "chase" method, this means we're typically looking at lags where the function h is shifted to the left relative to (1-h).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_c_at_k(h, n, K):
    """Gradient of correlate(h, 1-h, 'full')[K] wrt h."""
    lag = K - (n - 1)
    grad = np.zeros(n)
    n_min = max(0, -lag)
    n_max = min(n - 1, n - 1 - lag)
    if n_max < n_min:
        return grad
    idx = np.arange(n_min, n_max + 1)
    grad[idx + lag] += (1.0 - h[idx])
    grad[idx] -= h[idx + lag]
    return grad

                lr *= 0.95
                if stall > 3000:
                    lr = lr0
                    stall = 0
        g = grad_c_at_k(h, n, k_star)
        momentum = mu * momentum - lr * g
