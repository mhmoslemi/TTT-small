def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for k_idx in range(len(p)):
        w = p[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        i1s, i1e = max(0, -k), min(n, n - k)
        if i1e > i1s:
            idx = np.arange(i1s, i1e)
            grad[idx] += w * (1.0 - h[idx + k])
        i2s, i2e = max(0, k), min(n, n + k)
        if i2e > i2s:
            idx2 = np.arange(i2s, i2e)
            grad[idx2] -= w * h[idx2 - k]
    grad *= dx
    return grad

def grad_descent(h, n, alpha, max_steps, lr):

    dx = 2.0 / n
    best_c5 = compute_c5(h, n)
    for it in range(max_steps):

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        # Try the step
        h_try = project_feasible(h - lr * g, n)
        c5_try = compute_c5(h_try, n)
        if c5_try < best_c5:
            h = h_try
            best_c5 = c5_try
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return h, best_c5

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    # Compute gradient G (Jacobian of each c_k w.r.t. delta)
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s, j1e = max(0, -k), min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s, j2e = max(0, k), min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.5, 0.3]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

For each i, the gradient is:
grad[i] = dx * sum_k p[k] * ([0<=i+k<n] * (1-h[i+k]) - [0<=i-k<n] * h[i-k])

Let me think of this as a correlation-like operation. 

Let me define:
- A[i] = sum_k p[k] * [0<=i+k<n] * (1-h[i+k])
- B[i] = sum_k p[k] * [0<=i-k<n] * h[i-k]
- grad[i] = dx * (A[i] - B[i])

A[i] = sum_k p[k] * (1-h[i+k]) where k ranges over values such that 0<=i+k<n

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    p = exp_s / exp_s.sum()
    idx_i = np.arange(n)[:, None]
    idx_j = np.arange(n)[None, :]
    P_idx = (n - 1) + idx_j - idx_i
    Q_idx = (n - 1) + idx_i - idx_j
    A = p[P_idx] @ (1.0 - h)
    B = p[Q_idx] @ h
    return dx * (A - B)

def grad_descent(h, n, alpha, max_steps, lr_init):
    best_c5 = compute_c5(h, n)
    lr = lr_init
    for _ in range(max_steps):
        g = compute_gradient(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_try = project_feasible(h - lr * g, n)
        c5_try = compute_c5(h_try, n)
        if c5_try < best_c5 - 1e-15:
            h = h_try
            best_c5 = c5_try
            lr = min(lr * 1.08, 1.0)
        else:
            lr *= 0.6
            if lr < 1e-7:
                break
    return h, best_c5

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s, j1e = max(0, -k), min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s, j2e = max(0, k), min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps, delta <= eps,
        delta >= -h, delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['GLOP', 'CLARABEL', 'ECOS', 'GLPK']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h, best_mc = h.copy(), mc
    for lam in [1.0, 0.8, 0.5, 0.3, 0.1]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc, best_h = mc_try, h_try
    return best_h, best_mc

def lbfgs_phase(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        sm = s.max()
        return m + np.log(np.sum(np.exp(s - sm))) / alpha
    def grad(x):
        return compute_gradient(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9, 'maxls': 30})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_h = project_feasible(best_h, len(best_h))
