import numpy as np
from scipy.optimize import minimize
import time


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v


def obj_lse(x, n, alpha, mu):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    base = m + (np.log(s) / alpha if s > 0 else 0.0)
    d = float(x.sum() - 0.5 * n)
    return base + 0.5 * mu * d * d


def grad_lse(x, n, alpha, mu):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    nz = np.flatnonzero(q > 1e-10)
    for idx in nz:
        k = int(idx) - off
        w = q[idx]
        if k == 0:
            g += w * (1.0 - 2.0 * x)
        elif k > 0:
            m2 = n - k
            g[k:] += w * (1.0 - x[:m2])
            g[:m2] -= w * x[k:]
        else:
            s2 = -k
            m2 = n - s2
            g[:m2] += w * (1.0 - x[s2:])
            g[s2:] -= w * x[:m2]
    d = float(x.sum() - 0.5 * n)
    return dx * g + mu * d


def obj_strict(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    return m + (np.log(s) / alpha if s > 0 else 0.0)


def grad_strict(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    nz = np.flatnonzero(q > 1e-10)
    for idx in nz:
        k = int(idx) - off
        w = q[idx]
        if k == 0:
            g += w * (1.0 - 2.0 * x)
        elif k > 0:
            m2 = n - k
            g[k:] += w * (1.0 - x[:m2])
            g[:m2] -= w * x[k:]
        else:
            s2 = -k
            m2 = n - s2
            g[:m2] += w * (1.0 - x[s2:])
            g[s2:] -= w * x[:m2]
    return dx * g


def con_fun(x, n):
    return float(x.sum() - 0.5 * n)


def con_jac(x, n):
    return np.ones(n)


def phase_lbfgsb(h, n, alpha, maxiter, deadline):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj_lse, h, args=(n, alpha, 1e8), method="L-BFGS-B",
                       jac=grad_lse, bounds=bounds,
                       options={"maxiter": maxiter, "ftol": 1e-15, "gtol": 1e-10})
        if res.x is not None:
            return res.x
    except Exception:

    tgt = 0.5 * n

    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:

        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:

        c = int(m.sum())

    base = m + (np.log(s) / alpha if s > 0 else 0.0)
    d = float(x.sum() - 0.5 * n)
    return base + 0.5 * mu * d * d
