import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    # Project to {0 <= h <= 1, sum(h) = n/2}
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def subgradient_method(h, n, T, deadline):

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for t in range(T):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        # Compute subgradient of c[k_star] w.r.t. h
        sg = np.zeros(n)
        # c[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # d/dc[k**] dh[i] = (1-h[i+k**-(n-1)]) for valid i (first factor)
        #                   - h[i- (n-1)-k**] for valid i (second factor, i.e., h appears as 1-h[j])
        for i in range(n):
            j = i + k_star - (n - 1)
            if 0 <= j < n:
                sg[i] += (1.0 - h[j])
            j2 = i - (n - 1) - k_star
            if 0 <= j2 < n:
                sg[i] -= h[j2]
        # Note: for k_star = n-1, this gives sg[i] = (1-h[i]) - h[i] = 1-2h[i]
        # Step size

        h_new = h - alpha_t * sg * dx
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_method(h, n, T, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for t in range(T):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        j1 = np.arange(n) + k_star - (n - 1)
        m1 = (j1 >= 0) & (j1 < n)
        sg[m1] += (1.0 - h[j1[m1]])
        j2 = np.arange(n) - (n - 1) - k_star
        m2 = (j2 >= 0) & (j2 < n)
        sg[m2] -= h[j2[m2]]
        alpha_t = 1.0 / (1.0 + 0.05 * t)
        h = project_feasible(h - alpha_t * sg * dx, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            j_arr = p + ms - (n - 1)
            vj = (j_arr >= 0) & (j_arr < n)
            a[vj] += (1.0 - h[j_arr[vj]])
            i_arr = p - ms + (n - 1)
            vi = (i_arr >= 0) & (i_arr < n)
            a[vi] -= h[i_arr[vi]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_idx = (ms == (n - 1))
            best_d = 0.0
            best_val = current_max
            lo, hi = -h0, 1.0 - h0
            for ref in range(3):
                d_grid = np.linspace(lo, hi, grid_size)
                for d in d_grid:
                    if abs(d) < 1e-14:
                        continue
                    nv = c_aff + d * a
                    if np.any(diag_idx):
                        di = np.where(diag_idx)[0][0]
                        nv[di] -= d * d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
                span = (hi - lo) / max(1, grid_size // 3)
                lo = max(-h0, best_d - 3 * span)
                hi = min(1.0 - h0, best_d + 3 * span)
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

        c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        k_star = int(np.argmax(c))
        preds = np.arange(n) - k_star + (n - 1)
        valid_pred = (preds >= 0) & (preds < n)
        pred_vals = np.zeros(n)
        pred_vals[valid_pred] = h_bin[preds[valid_pred]]
        flip_mask = (h_bin == 0) & valid_pred & (pred_vals == 1)
        flip_targets = np.where(flip_mask)[0]
        if len(flip_targets) == 0:
            flip_targets = np.where(h_bin == 0)[0]
        if len(flip_targets) == 0:
            break
        p = flip_targets[rng.randint(len(flip_targets))]
        ones = np.where(h_bin == 1)[0]
        if len(ones) == 0:
            break
        q = ones[rng.randint(len(ones))]
        h_bin[p] = 1.0
        h_bin[q] = 0.0
        new_c = compute_c5(h_bin, n)
        if new_c <= current_c:
            current_c = new_c
            no_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            temp = max(0.0005, 0.05 * (1.0 - it / float(max_iter)))
            if rng.random() < np.exp(-(new_c - current_c) / temp):
                current_c = new_c
            else:
                h_bin[p] = 0.0
                h_bin[q] = 1.0
            no_improve += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP warm start
    if time.time() < deadline - 400:
        h1 = slsqp_optimize(best_h, n, 3000, 800)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Subgradient (diversification)
    if time.time() < deadline - 400:
        h2, c2 = subgradient_method(best_h, n, T=800, deadline=deadline - 400)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Binary local search (new approach)
    if time.time() < deadline - 350:
        h3, c3 = binary_local_search(best_h, n, max_iter=80000,
                                     deadline=deadline - 350, seed=7)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: SLSQP with increasing alpha
    for alpha in [5000, 50000, 500000]:
        if time.time() > deadline - 250:
            break
        h4 = slsqp_optimize(best_h, n, alpha, 1500)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Coordinate descent
    if time.time() < deadline - 80:
        h5, c5v = coord_descent(best_h, n, 300, deadline - 80, grid_size=100)
        if c5v < best_c:
            best_h, best_c = h5, c5v
    # Phase 6: Try alternate n
    for n_try in [400, 600, 800]:
        if time.time() > deadline - 120:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h6 = slsqp_optimize(h_int, n_try, 20000, 600)
        h6, c6 = coord_descent(h6, n_try, 150, deadline - 80, grid_size=80)
        if c6 < best_c:
            best_h, best_c = h6, c6
    # Phase 7: Final polish
    if time.time() < deadline - 2:
        nf = len(best_h)
        hf, cf = coord_descent(best_h, nf, 400, deadline, grid_size=120)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
