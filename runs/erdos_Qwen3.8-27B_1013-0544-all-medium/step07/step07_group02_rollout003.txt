import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    """Return the raw correlation values (before dx scaling)."""
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    """Compute gradient matrix G where G[i,k] = d/dc_k h[i]."""
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps, solver='GLOP'):
    C = compute_corr(h, n).astype(float)
    mc = float(C.max())
    G = compute_gradients(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'CLARABEL', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.7, 0.5, 0.3, 0.1]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.max(compute_corr(h_try, n)))
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def flip_restart(h, n, rng, frac=0.3):
    """Structured restart: flip a random subset of values."""
    h_new = h.copy()
    n_flip = max(1, int(frac * n))
    indices = rng.choice(n, n_flip, replace=False)
    # For each flipped index, push towards the other extreme
    h_new[indices] = 1.0 - h_new[indices]
    return project_feasible(h_new, n)

def smooth_restart(h, n, rng, noise_scale):
    """Smooth restart with Gaussian noise."""
    h_new = h + noise_scale * rng.randn(n)
    return project_feasible(h_new, n)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = compute_corr(x, n) * dx
        m = g.max()
        shifted = alpha * (g - m)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

        return h

                break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

        delta <= eps,

        delta <= 1.0 - h,
