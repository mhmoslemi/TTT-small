def lse_lbfgs(h, n, alpha, maxiter, ftol, mu=100.0):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        penalty = mu * (x.sum() - target) ** 2
        return m + logsum / alpha + penalty
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    def _obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        ls = sm + np.log(np.exp(sh - sm).sum())
        return m + ls / alpha

        return x.sum() - target

                       constraints=[{'type': 'eq', 'fun': _cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def build_G(h, n):
    m_idx = np.arange(n)[:, None]
    s = np.arange(-(n - 1), n)[None, :]
    idx1 = m_idx + s
    idx2 = m_idx - s
    G1 = np.where((idx1 >= 0) & (idx1 < n), h[np.clip(idx1, 0, n - 1)], 0.0)
    G2 = np.where((idx2 >= 0) & (idx2 < n), h[np.clip(idx2, 0, n - 1)], 0.0)
    return G1 + G2

def update_G(G, pos, delta, n):
    s = np.arange(-(n - 1), n)
    k_vals = s + (n - 1)
    m1 = pos - s
    m2 = pos + s
    mask1 = (m1 >= 0) & (m1 < n)
    mask2 = (m2 >= 0) & (m2 < n)
    G[m1[mask1], k_vals[mask1]] += delta
    G[m2[mask2], k_vals[mask2]] += delta

def binary_search(h_init, n, deadline):

    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)

            break
        G = build_G(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            if h[i] != 0:
                continue
            base = corr - G[i]
            base[n - 1] = corr[n - 1]
            cands = base[None, :] + G[ones, :]
            nmaxes = cands.max(axis=1) * dx
            j_idx = int(nmaxes.argmin())
            j = int(ones[j_idx])
            if nmaxes[j_idx] < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                delta = G[j] - G[i]
                delta[n - 1] = 0.0
                corr += delta
                cur_c = float(nmaxes[j_idx])
                update_G(G, i, 1.0, n)
                update_G(G, j, -1.0, n)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    def _obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        ls = sm + np.log(np.exp(sh - sm).sum())
        return m + ls / alpha
    def _cons(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_slsqp(h, n, t, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - t, 0)
        return 0.5 * np.sum(excess ** 2)
    def _cons(x):
        return x.sum() - target

                       constraints=[{'type': 'eq', 'fun': _cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h
