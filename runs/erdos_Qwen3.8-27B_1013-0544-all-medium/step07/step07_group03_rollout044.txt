        sm = shifted.max()

from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_swap_delta(h_b, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h_b[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h_b[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h_b[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h_b[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n // 2]] = 1.0
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_h = h_b.copy()
    best_c = cur_c
    ones_list = list(np.where(h_b == 1)[0])
    zeros_list = list(np.where(h_b == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            best_j = -1
            best_nc = cur_c - 1e-15
            for j in ones_list:
                delta = binary_swap_delta(h_b, i, j, n, m_range)
                nc = float((corr + delta).max() * dx)
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j < 0:
                continue
            delta = binary_swap_delta(h_b, i, best_j, n, m_range)
            h_b[i], h_b[best_j] = 1.0, 0.0
            corr += delta
            cur_c = best_nc
            ones_list.remove(best_j)
            ones_list.append(i)
            zeros_list.remove(i)
            zeros_list.append(best_j)
            improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c

def binary_anneal(h_init, n, deadline, T0=0.08, Tmin=1e-5, cooling=0.9997):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n // 2]] = 1.0
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_h = h_b.copy()
    best_c = cur_c
    ones_arr = np.where(h_b == 1)[0].tolist()
    zeros_arr = np.where(h_b == 0)[0].tolist()
    rng = np.random.RandomState(77)
    T = T0
    it = 0
    plateau = 0
    while time.time() < deadline:
        it += 1
        i = zeros_arr[rng.randint(len(zeros_arr))]
        j = ones_arr[rng.randint(len(ones_arr))]
        delta = binary_swap_delta(h_b, i, j, n, m_range)
        new_corr = corr + delta
        new_c = float(new_corr.max() * dx)
        accept = new_c < cur_c or (T > Tmin and rng.random() < np.exp(-(new_c - cur_c) / T))
        if accept:
            h_b[i], h_b[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            ones_arr.remove(j)
            ones_arr.append(i)
            zeros_arr.remove(i)
            zeros_arr.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
                plateau = 0
            else:
                plateau += 1
        else:
            plateau += 1
        if plateau > 50000:
            T = min(T * 0.5, T0 * 0.5)
            plateau = 0
        T = max(T * cooling, Tmin)
    return best_h, best_c

def subgrad_opt(h, n, deadline, max_iters=40000, top_k=3, lr_base=0.08):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    G = np.zeros(n)
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0
    stall_limit = 4000
    rng = np.random.RandomState(99)
    t = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            h = best_h + 0.01 * rng.randn(n)
            h = project_feasible(h, n)
            G[:] = 0
            momentum[:] = 0
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] = 1.0 - 2.0 * h
                else:
                    grad[k:] += (1.0 - h[:n - k])
                    grad[:n - k] -= h[k:]
            else:
                kk = -k
                grad[:n - kk] += (1.0 - h[kk:])
                grad[kk:] -= h[:n - kk]
        grad = dx * grad / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum ** 2
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_check.max() < cur_c + 0.003:
            h = h_new
    return best_h, best_c
