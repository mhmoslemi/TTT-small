def soft_max_and_grad(h, n, beta):
    """Compute soft max of correlation and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = float(np.max(c))

        w[int(np.argmax(c))] = 1.0

    else:

def pair_line_search(h, n, i, j, deadline):
    """1D line search over delta for pair (i,j)."""
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h, cur
    best_d = 0.0
    best_c = cur
    # Evaluate at endpoints and use bounded scalar minimization
    try:

            rs = so.minimize_scalar(lambda d: erdos_c5(
                np.where(np.array([i, j]) == np.arange(n)[np.newaxis.T],
                         h + np.array([d, -d])[:, None], h), n),
                bounds=(lo, hi), method='bounded', options={'maxiter': 30, 'xatol': 1e-10})
            if rs.x is not None and rs.fun < best_c:
                best_c = float(rs.fun)
                best_d = float(rs.x)
    except Exception:

    if best_c < cur - 1e-15:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        return hh, best_c
    return h, cur

def pair_polish(h, n, deadline, seed, n_pairs=40, n_rounds=4):

    h = np.asarray(h, dtype=float).copy()

                cur = cc
                improved = True
        if not improved:

    return h, cur
