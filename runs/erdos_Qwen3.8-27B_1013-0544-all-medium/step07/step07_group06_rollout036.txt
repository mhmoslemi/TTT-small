def lse_obj(x, n, beta):
    h = np.clip(x, 0, 1)
    dx = 2.0/n
    c = np.correlate(h, 1-h, mode="full") * dx
    c = np.maximum(c, 0)
    m = c.max()
    y = beta*(c - m)
    e = np.exp(np.minimum(y, 0))
    s = e.sum()
    F = m + np.log(s)/beta if s > 0 else m
    # gradient
    p = e/s
    R = p[::-1]
    v = 1-h
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(p, h, mode="full")[n-1:2*n-1]
    g = dx*(A-B)
    return F, g
