For the coordinate descent, since the function in each coordinate is the max of affine functions (plus one concave quadratic), I can find the exact minimum:
- The minimum of max(affine functions, concave quadratic) occurs either at a kink point (where two affine functions are equal and dominate) or at the vertex of the quadratic (if it's below all affine functions at that point).
- I can find all kink points by solving pairs of linear equations, then evaluate the objective at each kink point and at the quadratic vertex, and take the minimum.

But there are O(n) kink points per coordinate, so this is O(n) per coordinate, O(n²) per pass. For n=500, that's 250K per pass. Very fast!

def coord_cd_exact(h, n, max_passes=100, deadline=None):
    """Coordinate descent with exact 1D optimization."""
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
        current_max = float(c.max())
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-15 or h0 > 1.0 - 1e-15:
                continue
            # Compute affected lags
            # For lag m (index in c), the change when h[p] changes by delta:
            # c[m] changes by delta * a[m] - delta^2 * I(m == n-1)
            # where a[m] = (1-h[p+m-(n-1)]) - h[p-m+(n-1)] (with bounds checking)
            
            m_lo = max(0, p - (n-1))
            m_hi = min(2*n-2, p + (n-1))
            ms = np.arange(m_lo, m_hi + 1)
            k = ms - (n-1)  # lag values
            jf = p + k  # index for (1-h[.]) term
            jb = p - k  # index for h[.] term
            a = np.zeros(len(ms))
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a[valid_f] += 1.0 - h[jf[valid_f]]
            a[valid_b] -= h[jb[valid_b]]
            
            c_aff = c[ms]
            # Unaffected max
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            
            # Find delta that minimizes max(unaff_max, c_aff + delta*a, c_diag - delta^2)
            # where c_diag is the diagonal element (if it exists in ms)
            diag_idx = np.where(ms == n-1)[0]
            has_diag = len(diag_idx) > 0
            
            # The function to minimize: g(delta) = max(unaff_max, max_j(c_aff[j] + delta*a[j]), c_diag - delta^2 if has_diag)
            # Domain: delta in [-h0, 1-h0]
            delta_lo = -h0
            delta_hi = 1.0 - h0
            
            # Find all kink points where two terms are equal:
            # c_aff[i] + delta*a[i] = c_aff[j] + delta*a[j] => delta = (c_aff[j]-c_aff[i])/(a[i]-a[j])
            # c_aff[i] + delta*a[i] = c_diag - delta^2 (quadratic, skip for now)
            # c_aff[i] + delta*a[i] = unaff_max => delta = (unaff_max - c_aff[i])/a[i]
            
            candidates = [delta_lo, delta_hi]

                if a[i] == 0:
                    continue
                for j in range(i+1, len(ms)):
                    if a[i] == a[j]:
                        continue
                    d = (c_aff[j] - c_aff[i]) / (a[i] - a[j])
                    if delta_lo <= d <= delta_hi:
                        candidates.append(d)
                # Kink with unaff_max
                d = (unaff_max - c_aff[i]) / a[i]
                if delta_lo <= d <= delta_hi:
                    candidates.append(d)
            
            # If has_diag, kink between diag and affine:
            if has_diag:
                c_diag = c_aff[diag_idx[0]]
                for i in range(len(ms)):
                    if i == diag_idx[0]:
                        continue
                    # c_aff[i] + delta*a[i] = c_diag - delta^2
                    # delta^2 + delta*a[i] + c_aff[i] - c_diag = 0
                    disc = a[i]**2 - 4*(c_aff[i] - c_diag)
                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        d1 = (-a[i] + sqrt_disc) / 2
                        d2 = (-a[i] - sqrt_disc) / 2
                        if delta_lo <= d1 <= delta_hi:
                            candidates.append(d1)
                        if delta_lo <= d2 <= delta_hi:
                            candidates.append(d2)
                # Vertex of quadratic
                if has_diag:
                    # c_diag - delta^2, vertex at delta=0
                    candidates.append(0.0)
            
            # Evaluate at all candidates
            best_delta = 0.0
            best_val = current_max
            for d in candidates:
                d = max(delta_lo, min(delta_hi, d))
                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx[0]] -= d * d
                val = max(unaff_max, float(vals.max()))
                if val < best_val - 1e-15:
                    best_val = val
                    best_delta = d
            
            if abs(best_delta) > 1e-14:
                h[p] = h0 + best_delta
                improved = True
        
        if not improved:
            break
        h = project_feasible(h, n)
    
    return h, compute_c5(h, n)

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_v(v, n, alpha):
    """Objective in v-space: h_i = 0.5 + v_i for i<n-1, h_{n-1} = 0.5 - sum(v)"""
    h = np.full(n, 0.5)
    h[:-1] += v
    h[-1] = 0.5 - v.sum()
    # Soft penalty for out-of-bounds
    viol = np.maximum(0, -h) + np.maximum(0, h - 1)
    if viol.sum() > 1e-6:
        return 1e6 + viol.sum()
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slbfgs_v(h, n, alphas, maxiter=2000, deadline=None):
    """Optimize in v-space using L-BFGS-B"""
    v0 = h.copy() - 0.5
    bounds = [(-0.5, 0.5)] * (n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline:
            break
        res = minimize(smooth_obj_v, v0, args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        v_opt = res.x
        h_new = np.full(n, 0.5)
        h_new[:-1] += v_opt
        h_new[-1] = 0.5 - v_opt.sum()
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new
            v0 = h_new - 0.5
            # Reset v0 relative to the last element
            v0[:n-1] = h_new[:n-1] - 0.5
    return best_h, best_c
