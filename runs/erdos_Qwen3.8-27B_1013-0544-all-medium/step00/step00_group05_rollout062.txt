import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def soft_max_objective(h, dx, T):
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def soft_max_gradient(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    w = np.exp(T * (vals - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for p in range(n_corr):
        if w[p] < 1e-14:
            continue
        k = p - (n - 1)
        c = dx * w[p]
        if k >= 0:
            grad[k:] += c * (1.0 - h[:n-k])
            grad[:n-k] -= c * h[k:]
        else:
            j = -k
            grad[:n-j] += c * (1.0 - h[j:])
            grad[j:] -= c * h[:n-j]
    return grad

def sum_constraint(h):
    return np.sum(h)

    return np.ones(len(h))

def soft_max_obj(h, dx, T):
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def soft_max_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    w = np.exp(T * (vals - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-14:
            continue
        k = p - (n - 1)
        c = dx * w[p]
        if k >= 0:
            grad[k:] += c * (1.0 - h[:n-k])
            grad[:n-k] -= c * h[k:]
        else:
            j = -k
            grad[:n-j] += c * (1.0 - h[j:])
            grad[j:] -= c * h[:n-j]
    return grad
