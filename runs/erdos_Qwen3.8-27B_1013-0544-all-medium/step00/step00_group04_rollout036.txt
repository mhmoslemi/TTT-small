def compute_c5(h):
    """Compute C5 bound for a step function h on [0,2]"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_h(h, n):
    """Ensure h satisfies 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Adjust: add diff/n to elements that have room
        if diff > 0:
            # Need to increase some elements
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            # Need to decrease some elements
            total = np.sum(h)
            if total > 1e-12:
                h += h * (diff / total)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # Final fix: adjust the largest/smallest element
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):

    n_old = len(h_old)

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)

def simulated_annealing(h0, max_time=60.0, seed=42):
    """Simulated annealing with local search restarts"""
    n = len(h0)
    h = h0.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    
    start = time.time()
    T = 0.05
    eval_count = 0
    
    while (time.time() - start) < max_time:
        # Temperature schedule
        elapsed = time.time() - start
        progress = elapsed / max_time
        T = 0.05 * (0.01) ** progress
        
        if T < 1e-7:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        eps = rng.uniform(0.01, 0.5)
        sign = 1 if rng.random() < 0.5 else -1
        
        h[i] += sign * eps
        h[j] -= sign * eps
        
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= sign * eps
            h[j] += sign * eps
            continue
        
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        eval_count += 1
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] -= sign * eps
            h[j] += sign * eps
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)

def local_search_binary(h0, max_time=60.0, seed=42, use_double=True):
    n = len(h0)
    h = make_binary(h0, n)
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)

    while (time.time() - start) < max_time:
        improved = False
        best_gain = 1e-14
        best_move = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j, None, None)
                h[i] = 1.0
                h[j] = 0.0

        if not improved and best_move is None and use_double and (time.time() - start) < max_time * 0.7:
            ones_list = list(ones)
            zeros_list = list(zeros)
            n_o = len(ones_list)
            n_z = len(zeros_list)
            max_pairs = min(n_o * (n_o - 1) // 2 * n_z * (n_z - 1) // 2, 20000)
            count = 0
            for a in range(n_o):
                for b in range(a + 1, n_o):
                    i1, i2 = ones_list[a], ones_list[b]
                    for c in range(n_z):
                        for d in range(c + 1, n_z):
                            if count >= max_pairs:
                                break
                            j1, j2 = zeros_list[c], zeros_list[d]
                            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
                            new_c5 = compute_c5(h)
                            gain = current_c5 - new_c5
                            if gain > best_gain:
                                best_gain = gain
                                best_move = (i1, i2, j1, j2)
                            h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
                            count += 1
                        if count >= max_pairs:
                            break
                    if count >= max_pairs:
                        break
                if count >= max_pairs:
                    break

        if best_move is not None:
            if best_move[2] is None:
                i, j, _, _ = best_move
                h[i] = 0.0
                h[j] = 1.0
            else:
                i1, i2, j1, j2 = best_move
                h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            current_c5 = compute_c5(h)
            improved = True
        else:
            break

    return h, current_c5

def local_search_continuous(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = normalize_continuous(h0, n)
    current_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    phases = [[0.5, 0.25, 0.1], [0.1, 0.05, 0.02], [0.02, 0.01, 0.005, 0.002]]
    phase = 0
    stall = 0

    while (time.time() - start) < max_time:
        eps_list = phases[phase]
        best_gain = 1e-14
        best_move = None

        if n <= 70:
            pairs = [(i, j) for i in range(n) for j in range(n) if i != j]
        else:
            np_ = min(n * (n - 1) // 3, 4000)
            ii = rng.randint(0, n, np_)
            jj = rng.randint(0, n, np_)
            pairs = [(int(ii[k]), int(jj[k])) for k in range(np_) if int(ii[k]) != int(jj[k])]

        for i, j in pairs:
            for eps in eps_list:
                if h[i] >= eps - 1e-9 and h[j] + eps <= 1.0 + 1e-9:
                    h[i] -= eps; h[j] += eps
                    new_c5 = compute_c5(h)
                    gain = current_c5 - new_c5
                    if gain > best_gain:
                        best_gain = gain; best_move = (i, j, -eps, eps)
                    h[i] += eps; h[j] -= eps
                if h[j] >= eps - 1e-9 and h[i] + eps <= 1.0 + 1e-9:
                    h[j] -= eps; h[i] += eps
                    new_c5 = compute_c5(h)
                    gain = current_c5 - new_c5
                    if gain > best_gain:
                        best_gain = gain; best_move = (j, i, -eps, eps)
                    h[j] += eps; h[i] -= eps

        if best_move is not None:
            i, j, di, dj = best_move
            h[i] = np.clip(h[i] + di, 0, 1)
            h[j] = np.clip(h[j] + dj, 0, 1)
            current_c5 = compute_c5(h)
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                if phase < 2:
                    phase += 1; stall = 0
                else:
                    break

    return h, current_c5

def simulated_annealing(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = h0.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()

    while (time.time() - start) < max_time:
        elapsed = time.time() - start
        T = 0.05 * (0.005) ** (elapsed / max_time)
        if T < 1e-8:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        eps = rng.uniform(0.01, 0.5)
        s = 1.0 if rng.random() < 0.5 else -1.0
        h[i] += s * eps; h[j] -= s * eps
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= s * eps; h[j] += s * eps
            continue
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5; best_h = h.copy()
        else:
            h[i] -= s * eps; h[j] += s * eps

    return best_h, best_c5
