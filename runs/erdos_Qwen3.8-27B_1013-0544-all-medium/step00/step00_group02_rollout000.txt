                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
    h_new = project_feasible(res.x, target_sum)
    c5_new = compute_c5(h_new)
    return h_new, c5_new, res

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target_sum = n / 2.0

    h0 = np.clip(h0, 0.0, 1.0)
    if abs(np.sum(h0) - target_sum) > 1e-10:
        h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with increasing smoothness
    for alpha in [30.0, 80.0, 150.0, 300.0, 600.0]:
        if time_left() < 20:
            break
        h_new, c5_new, _ = try_optimize(best_h, target_sum, alpha, 800, 'L-BFGS-B',
                                         time_left, budget_s)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Nelder-Mead refinement
    if time_left() > 30:
        h_new, c5_new, _ = try_optimize(best_h, target_sum, 0, 500, 'Nelder-Mead',
                                         time_left, budget_s)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Another L-BFGS-B round with high alpha
    for alpha in [400.0, 800.0]:
        if time_left() < 20:
            break
        h_new, c5_new, _ = try_optimize(best_h, target_sum, alpha, 600, 'L-BFGS-B',
                                         time_left, budget_s)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Random restarts with local optimization
    pers = [0.4, 0.25, 0.15, 0.08, 0.04]
    for p in pers:
        if time_left() < 25:
            break
        n_restarts = max(1, int(min(8, (time_left() - 20) / 15)))
        for _ in range(n_restarts):
            if time_left() < 20:
                break
            h_pert = best_h + rng.randn(n) * p
            h_pert = project_feasible(h_pert, target_sum)
            h_new, c5_new, _ = try_optimize(h_pert, target_sum, 200.0, 300, 'L-BFGS-B',
                                             time_left, budget_s)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if time_left() < 20:
                break

    # Phase 5: Final Nelder-Mead polish
    if time_left() > 15:
        h_new, c5_new, _ = try_optimize(best_h, target_sum, 0, 300, 'Nelder-Mead',
                                         time_left, budget_s)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final verification
    best_h = project_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
