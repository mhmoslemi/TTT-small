import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    weights = exp_vals / exp_vals.sum()

        i_max = min(n - 1, n - 1 - s)
        for i in range(i_min, i_max + 1):
            grad[i] += weights[k] * dx * (1.0 - h[i + s])
            grad[i + s] -= weights[k] * dx * h[i]
