def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def fix_sum(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    if s > target:
        diff = s - target
        idx = np.argsort(h)[::-1]
        remaining = diff
        for i in idx:
            red = min(h[i], remaining)
            h[i] -= red
            remaining -= red
            if remaining <= 1e-12:
                break
    else:
        diff = target - s
        idx = np.argsort(h)
        remaining = diff
        for i in idx:
            inc = min(1.0 - h[i], remaining)
            h[i] += inc
            remaining -= inc
            if remaining <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def make_objective(n):
    dx = 2.0 / n
    target = n / 2.0
    def obj(h):
        hc = np.clip(h, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode='full') * dx
        c5 = float(np.max(corr))
        penalty = 50.0 * (np.sum(hc) - target) ** 2
        return c5 + penalty
    return obj
