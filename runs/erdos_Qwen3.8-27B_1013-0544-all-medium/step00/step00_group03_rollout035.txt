def smooth_gradient(h, n, temp):

    c = compute_c(h)
    scaled = c / temp

    weights = np.exp(scaled - m) / np.sum(np.exp(scaled - m))

import numpy as np
import time

def compute_c(h):
    return np.correlate(h, 1 - h, mode="full")

def exact_objective(h, n):
    dx = 2.0 / n
    return float(np.max(compute_c(h)) * dx)

def smooth_gradient(h, n, temp):
    dx = 2.0 / n
    c = compute_c(h)
    scaled = c / temp
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    s = np.sum(exp_scaled)
    weights = exp_scaled / s
    
    n_half = n - 1
    grad = np.zeros(n)
    
    for idx in range(len(c)):
        w = weights[idx]
        if w < 1e-14:
            continue
        lag = idx - n_half
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            m_lag = -lag
            grad[:n-m_lag] += w * (1.0 - h[m_lag:])
            grad[m_lag:] -= w * h[:n-m_lag]
    
    return grad * dx
