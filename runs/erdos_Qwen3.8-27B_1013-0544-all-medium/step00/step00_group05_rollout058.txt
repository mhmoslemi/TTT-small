        for i in range(lo, hi + 1):
            grad[i] += (1.0 - h[i + shift])
            grad[i + shift] -= h[i]
    else:
        s = -shift
        for i in range(s, n):
            grad[i] += (1.0 - h[i - s])
            grad[i - s] -= h[i]
    return c5_val, grad * dx
