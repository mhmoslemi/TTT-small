    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = _project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _compute_c5(h0, dx)
    best_n = n

    # Phase 1: SLSQP with annealing on original n
    epsilon = 35.0
    for rnd in range(40):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 30:
            break
        h_start = best_h.copy()
        if rnd > 0 and rnd % 5 == 0:
            h_start = best_h + np.random.normal(0, 0.06, n)
            h_start = _project_feasible(h_start, target_sum)

        result = _run_slsqp(h_start, dx, target_sum, epsilon, maxiter=350)
        if result is not None and result.x is not None:
            h_new = _project_feasible(result.x, target_sum)
            c5_new = _compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        epsilon = max(4.0, epsilon * 0.83)

    # Phase 2: L-BFGS-B with penalty as alternative
    if budget_s - (time.time() - start_time) > 60:
        penalty_w = 100.0
        eps = 15.0
        for attempt in range(10):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 30:
                break
            h_start = best_h + np.random.normal(0, 0.04, n)
            h_start = _project_feasible(h_start, target_sum)
            result = _run_lbfgsb(h_start, dx, target_sum, eps, penalty_w, maxiter=1500)
            if result is not None and result.x is not None:
                h_new = _project_feasible(result.x, target_sum)
                c5_new = _compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            eps = max(4.0, eps * 0.8)
            penalty_w *= 1.5

    # Phase 3: Local polish on best solution at original n
    if budget_s - (time.time() - start_time) > 30:
        h_pol, c5_pol = _local_polish(best_h, dx, target_sum, n_iter=3000, seed=seed)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 4: Try different n values
    for n_try in [50, 60, 75, 100]:
        elapsed = time.time() - start_time
        h_cand, n_cand, c5_cand = _try_n_value(
            best_h, best_n, n_try, 2.0 / best_n, best_n / 2.0, elapsed, budget_s)
        if c5_cand is not None and c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand
            best_n = n_cand

    # Phase 5: Final polish
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    if budget_s - (time.time() - start_time) > 20:
        h_pol, c5_pol = _local_polish(best_h, dx_final, target_final,
                                      n_iter=2000, seed=seed + 1)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Ensure exact feasibility
    best_h = _project_feasible(best_h, target_final)
    final_c5 = _compute_c5(best_h, dx_final)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Try to use evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h, dx_final, len(best_h))
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
