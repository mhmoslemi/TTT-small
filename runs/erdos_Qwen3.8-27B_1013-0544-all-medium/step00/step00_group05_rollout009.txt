def repair(v, n):
    v = np.array(v, dtype=float, copy=True)
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        for i in np.argsort(v):
            add = min(1.0 - v[i], diff)
            v[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        for i in np.argsort(-v):
            sub = min(v[i], -diff)
            v[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return v

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
