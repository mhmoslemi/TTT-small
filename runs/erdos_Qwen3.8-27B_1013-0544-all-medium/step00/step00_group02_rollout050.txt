from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def max_overlap(h):
    return np.max(compute_c5(h))

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    # Simple water-filling projection
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    # Distribute the difference
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to increase sum: increase values that are < 1
            mask = h < 1.0
            if np.any(mask):
                step = diff / np.sum(mask)
                h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            # Need to decrease sum: decrease values that are > 0
            mask = h > 0.0
            if np.any(mask):
                step = diff / np.sum(mask)
                h[mask] = np.maximum(0.0, h[mask] + step)
    # Final clip
    h = np.clip(h, 0, 1)
    return h

    c = np.correlate(h, 1.0 - h, mode='full')

from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(c))
    s = k - (n - 1)
    grad = np.zeros(n)
    idx = np.arange(n)
    jp = idx + s
    vp = (jp >= 0) & (jp < n)
    grad[vp] += (1.0 - h[jp[vp]])
    jm = idx - s
    vm = (jm >= 0) & (jm < n)
    grad[vm] -= h[jm[vm]]
    grad *= dx
    return float(c[k] * dx), grad

def _project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break

        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.maximum(0.0, h[m] + d / c)
    return h
