import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_overlaps(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def grad_smooth_max(h, dx, eps):
    n = len(h)
    overlaps = get_overlaps(h, dx)
    m_val = overlaps.max()
    w = np.exp((overlaps - m_val) / eps)
    w /= w.sum()
    
    grad = np.zeros(n)
    for m_idx in range(len(overlaps)):
        wm = w[m_idx]
        if wm < 1e-12:
            continue
        j = m_idx - (n - 1)
        # For each k, d(overlap)/d(h[k]) = (1-h[k-j])_valid - h[k+j]_valid
        k = np.arange(n)
        valid1 = (k - j >= 0) & (k - j < n)
        valid2 = (k + j >= 0) & (k + j < n)
        term1 = np.zeros(n)
        term1[valid1] = 1.0 - h[k[valid1] - j]
        term2 = np.zeros(n)
        term2[valid2] = h[k[valid2] + j]
        grad += wm * dx * (term1 - term2)
    
    return grad

def optimize_h(h0, dx, target_sum, budget_s, start_time, rng):
    n = len(h0)
    h = h0.copy()
    best_local = h.copy()
    best_local_c5 = compute_c5(h, dx)
    
    # Phase 1: Gradient descent
    eps = 0.005
    lr = 0.01
    for it in range(3000):
        if time.time() - start_time > budget_s:
            break
        c5 = compute_c5(h, dx)
        if c5 < best_local_c5:
            best_local_c5 = c5
            best_local = h.copy()
        
        g = grad_smooth_max(h, dx, eps)
        h_new = project(h - lr * g, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        if new_c5 <= c5 + 1e-10:
            h = h_new
        else:
            lr *= 0.9
        
        if it % 500 == 0 and it > 0:
            eps = max(0.001, eps * 0.8)
    
    h = best_local.copy()
    
    # Phase 2: Local search (kicked coordinate descent)
    step_size = 0.1
    for it in range(5000):
        if time.time() - start_time > budget_s:
            break
        idx = rng.integers(n)
        delta = rng.uniform(-step_size, step_size)
        h_trial = h.copy()
        old_val = h_trial[idx]
        h_trial[idx] = np.clip(old_val + delta, 0.0, 1.0)
        h_trial = project(h_trial, target_sum)
        
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < best_local_c5:
            best_local_c5 = c5_trial
            best_local = h_trial.copy()
            h = h_trial.copy()

                    h_trial[idx2] = np.clip(h_trial[idx2] + rng.normal(0, 0.05), 0, 1)
                h_trial = project(h_trial, target_sum)
                if compute_c5(h_trial, dx) < best_local_c5:
                    best_local_c5 = compute_c5(h_trial, dx)
                    best_local = h_trial.copy()
                    h = h_trial.copy()
        
        if it % 1000 == 0:
            step_size = max(0.02, step_size * 0.9)
    
    return best_local, best_local_c5

    best_h = None
    best_c5 = float('inf')

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, dx, eps):
    n = len(h)
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m_val = overlaps.max()
    w = np.exp((overlaps - m_val) / eps)
    w /= w.sum()
    grad = np.zeros(n)
    ar = np.arange(n)
    for m_idx in range(len(overlaps)):
        wm = w[m_idx]
        if wm < 1e-12:
            continue
        j = m_idx - (n - 1)
        t1 = np.zeros(n)
        v1 = (ar - j >= 0) & (ar - j < n)

        t2 = np.zeros(n)
        v2 = (ar + j >= 0) & (ar + j < n)
        t2[v2] = h[ar[v2] + j]
        grad += wm * dx * (t1 - t2)
    return grad

        if c5t < best_c5:
            best_c5 = c5t
            best_h = h_t.copy()
            h = h_t.copy()
        elif it < 2000 and rng.random() < 0.08:
            cur = compute_c5(h, dx)
            if c5t - cur < 0.008:
                h = h_t.copy()
        if it % 800 == 799:
            step = max(0.008, step * 0.82)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')

    starts = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
        starts.append((init.copy(), 45))
        for _ in range(12):
            p = init + rng.normal(0, 0.025, len(init))
            starts.append((project(p, len(init) / 2.0), len(init)))

    for n_try in [40, 44, 50, 60]:
        for _ in range(4):
            h_r = rng.random(n_try)
            starts.append((project(h_r, n_try / 2.0), n_try))

    # Binary structured starts
    for n_try in [45, 50]:
        h_b = (rng.random(n_try) > 0.5).astype(float)
        starts.append((project(h_b, n_try / 2.0), n_try))

    # Cosine-based structured
    n = 45
    x = np.linspace(0, 2, n, endpoint=False)
    h_cos = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x / 2)
    starts.append((project(h_cos, n / 2.0), n))

    hard_deadline = start_time + budget_s * 0.92
    total = len(starts)

    for si in range(total):
        if time.time() > hard_deadline:
            break
        h0, n = starts[si]
        dx = 2.0 / n
        ts = n / 2.0
        elapsed = time.time() - start_time
        remaining = max(1.0, (start_time + budget_s * 0.92) - time.time())
        per_cand = remaining / max(1, total - si)
        deadline = time.time() + min(per_cand, budget_s * 0.4)

        h_gd, c5_gd = gradient_descent(h0, dx, ts, deadline, rng)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()

        deadline2 = min(time.time() + per_cand * 0.6, hard_deadline)
        best_h, best_c5 = local_search(h_gd, dx, ts, deadline2, rng, best_c5, best_h)

    # Final polish on best solution
    if best_h is not None:
        n = len(best_h)
        dx = 2.0 / n
        ts = n / 2.0
        deadline_final = start_time + budget_s * 0.97
        best_h, best_c5 = local_search(best_h, dx, ts, deadline_final, rng, best_c5, best_h)

    return best_h, best_c5, len(best_h)
