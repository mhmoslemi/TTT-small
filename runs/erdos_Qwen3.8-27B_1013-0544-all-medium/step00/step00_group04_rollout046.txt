def project(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    h = h.copy()
    # Water filling
    for _ in range(100):
        h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        # Find free variables (not at 0 or 1)
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(free) == 0:
            break
        h[free] += diff / np.sum(free)
    return h

def subgradient_step(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(c)
    s = k_star - (n - 1)  # the shift
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        gs = j + s
        gl = j - s
        val = 0.0
        if 0 <= gs < n:
            val += (1 - h[gs])
        if 0 <= gl < n:
            val -= h[gl]
        grad[j] = dx * val
    
    h_new = h - alpha * grad
    return project(h_new, n / 2.0)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_onto_simplex(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x_i <= 1}"""
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0, 1)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        h[free] += diff / nfree
    return h

def subgradient_descent(h, n_iters=5000, alpha_init=0.05):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    
    for it in range(n_iters):
        alpha = alpha_init / (1 + it * 0.001)
        c = np.correlate(h, 1.0 - h, mode='full')

                val -= h[j - s]

For each coordinate j, given all other h values fixed, find the optimal h[j]:
- c_s as a function of x = h[j] is linear: c_s(x) = a_s * x + b_s
- We want to minimize max_s (a_s * x + b_s) over x ∈ [0,1] (with the constraint that sum(h) = n/2)

        t_lim = rem * 0.35
        h, val = subgradient_descent(h_interp, t_lim)
        if val < best_val:
            t_lim2 = budget_s - (time.time() - t0)
            if t_lim2 > 10:
                h, val = local_search(h, t_lim2 * 0.5)
                if val < best_val:
                    best_h, best_val, best_n = h, val, n1

    # Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        h, val = local_search(best_h, rem * 0.85)
        if val < best_val:
            best_h, best_val, best_n = h, val, len(h)

    best_h = project(best_h, len(best_h) / 2.0)
    best_val = compute_c5(best_h)
    return best_h, best_val, len(best_h)
