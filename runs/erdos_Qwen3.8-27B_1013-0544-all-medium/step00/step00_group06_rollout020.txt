        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def interp_to_res(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / float(n_old)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    return np.interp(x_new, x_old, h, period=1.0)

def compute_grad(h, weights, n):
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = weights[k + n - 1]
        if wk < 1e-12:
            continue
        m_lo = max(0, k)
        m_hi = min(n - 1, n - 1 + k)
        if m_lo <= m_hi:
            grad[m_lo:m_hi + 1] += wk * (1.0 - h[m_lo - k:m_hi + 1 - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] -= wk * h[m_lo2 + k:m_hi2 + 1 + k]
    return grad

def optimize_h(h, n, dx, target_sum, time_budget, rng):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    T = 15.0
    lr = 0.03
    no_improve = 0
    for step in range(500000):
        if time.time() - t0 > time_budget:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = c.max()
        shifted = T * (c - c_max)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            break
        weights /= ws
        grad = compute_grad(h, weights, n)
        h = project_h(h - lr * grad, target_sum)
        cur_c5 = compute_c5(h, dx)
        if cur_c5 < best_c5 - 1e-11:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 0 and no_improve % 120 == 0:
            lr *= 0.92
        if no_improve > 350:
            ns = min(0.12, 0.008 + 0.008 * (no_improve // 350))
            h = project_h(best_h + rng.normal(0, ns, n), target_sum)
            lr = 0.02
            T = max(5.0, T * 0.75)
            no_improve = 0
    return best_h, best_c5

def hill_climb(h, n, dx, target_sum, time_budget, rng):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    scale = 0.008
    for step in range(2000000):
        if time.time() - t0 > time_budget:
            break
        h_new = project_h(h + rng.normal(0, scale, n), target_sum)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            scale = min(0.05, scale * 1.02)
        else:
            if step % 50 == 0:
                scale = max(0.001, scale * 0.97)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h_parent = initial_h_values.copy()
    n_parent = len(h_parent)
    dx_parent = 2.0 / n_parent
    target_parent = n_parent / 2.0
    h_parent = project_h(h_parent, target_parent)

    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent, dx_parent)
    best_n = n_parent

    resolutions = []
    for f in [1.0, 0.6, 0.8, 1.25, 1.5, 2.0, 2.5, 3.0, 0.4]:
        nt = int(round(n_parent * f))
        if 25 <= nt <= 500:
            resolutions.append(nt)
    resolutions = sorted(set(resolutions))

    total_tl = budget_s - 40
    inv_ns = [1.0 / float(n) for n in resolutions]
    tot_inv = sum(inv_ns)
    allocs = [max(5.0, total_tl * iv / tot_inv) for iv in inv_ns]
    sa = sum(allocs)
    if sa > total_tl:
        allocs = [a * total_tl / sa for a in allocs]

    for i, n in enumerate(resolutions):
        remaining = budget_s - (time.time() - t_start)
        if remaining < 20:
            break
        alloc = min(allocs[i], remaining - 15)
        if alloc < 5:
            break

        dx = 2.0 / n
        target_sum = n / 2.0

        if n == best_n:
            h = best_h.copy()
        else:
            h = interp_to_res(best_h, best_n, n)
            h = project_h(h, target_sum)

        h_opt, c5_opt = optimize_h(h, n, dx, target_sum, alloc * 0.75, rng)
        c5_opt2 = compute_c5(h_opt, dx)
        if c5_opt2 < c5_opt:
            c5_opt = c5_opt2
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

    # Final hill climbing on best solution
    remaining = budget_s - (time.time() - t_start)
    if remaining > 10:
        dx_best = 2.0 / best_n
        target_best = best_n / 2.0
        best_h, best_c5 = hill_climb(best_h, best_n, dx_best, target_best, min(remaining - 5, 60), rng)

    # Final gradient polish
    remaining = budget_s - (time.time() - t_start)
    if remaining > 10:
        dx_best = 2.0 / best_n
        target_best = best_n / 2.0
        best_h, best_c5 = optimize_h(best_h, best_n, dx_best, target_best, min(remaining - 5, 30), rng)

    return best_h, best_c5, best_n
