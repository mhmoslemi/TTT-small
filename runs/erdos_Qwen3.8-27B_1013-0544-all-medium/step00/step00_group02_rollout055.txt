    n = len(h)

        if projected.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, c, n, tol_frac=0.02):
    max_c = np.max(c)
    threshold = max_c - tol_frac * max_c
    active = np.where(c >= threshold)[0]
    grad = np.zeros(n)
    for k in active:
        tau = k - (n - 1)
        if 0 <= tau < n:
            grad[tau:] += (1.0 - h[:n - tau])
            grad[:n - tau] -= h[tau:]
        elif -n < tau < 0:
            t = -tau
            grad[:n - t] += (1.0 - h[t:])
            grad[t:] -= h[:n - t]
    grad /= len(active)
    return grad

    n = len(h)
    best_c5, _ = compute_c5_full(h)
    for sweep in range(max_sweeps):

                    improved = True

                    h = project_feasible(h, target_sum)
