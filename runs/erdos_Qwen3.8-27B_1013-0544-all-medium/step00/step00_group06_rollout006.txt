    return corr * dx

def maximin_subgradient(h, tol=0.01):
    n = len(h)
    R = autocorr_vals(h)
    min_val = np.min(R)
    active = np.where(R <= min_val + tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag == 0:
            grad += 2.0 * h
        else:
            i_min = max(0, -lag)
            i_max = min(n - 1, n - 1 - lag)
            for i in range(i_min, i_max + 1):
                j = i + lag
                grad[i] += h[j]
                grad[j] += h[i]
    return grad / len(active)

def cross_grad_c5(h, tol=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    c5 = np.max(vals)
    active = np.where(vals >= c5 - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag == 0:
            grad += (1 - 2 * h)
        else:
            i_min = max(0, -lag)
            i_max = min(n - 1, n - 1 - lag)
            for i in range(i_min, i_max + 1):
                j = i + lag
                grad[j] += (1 - h[i])
                grad[i] -= h[j]
    return grad * dx / len(active)

def pow_mean_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    return float(np.sum(vals ** p) ** (1.0 / p))

def pow_mean_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    sp = np.sum(vals ** p)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = (vals[k] ** (p - 1)) * dx
        if w < 1e-18:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1 - 2 * h)
        else:
            i_min = max(0, -lag)
            i_max = min(n - 1, n - 1 - lag)
            i_vals = np.arange(i_min, i_max + 1)
            j_vals = i_vals + lag
            grad[j_vals] += w * (1 - h[i_vals])
            grad[i_vals] -= w * h[j_vals]
    prefix = sp ** ((1.0 - p) / p)
    return grad * prefix

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(95, 0.5)

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Maximin subgradient (maximize min autocorrelation = minimize C5)
    h_work = best_h.copy()
    lr = 0.008
    momentum = 0.7
    vel = np.zeros(n)
    stall = 0

    for it in range(20000):
        if remaining() < 200:
            break
        grad = maximin_subgradient(h_work, tol=0.008)
        vel = momentum * vel + lr * grad
        h_work = project_feasible(h_work + vel, n)
        c5_val = compute_c5(h_work)
        if c5_val < best_c5 - 1e-10:
            best_h = h_work.copy()
            best_c5 = c5_val
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                lr *= 0.9
                momentum *= 0.98
            if stall > 3000:
                lr = 0.008
                momentum = 0.7
                vel = np.zeros(n)
                stall = 0
        if lr < 1e-6:
            break

    # Phase 2: Cross-gradient (direct C5 minimization) polish
    h_work = best_h.copy()
    lr2 = 0.004
    vel2 = np.zeros(n)
    stall2 = 0
    for it in range(15000):
        if remaining() < 150:
            break
        grad = cross_grad_c5(h_work, tol=0.005)
        vel2 = 0.8 * vel2 - lr2 * grad
        h_work = project_feasible(h_work + vel2, n)
        c5_val = compute_c5(h_work)
        if c5_val < best_c5 - 1e-10:
            best_h = h_work.copy()
            best_c5 = c5_val
            stall2 = 0
        else:
            stall2 += 1
            if stall2 % 150 == 0:
                lr2 *= 0.92
            if stall2 > 2000:
                lr2 = 0.004
                vel2 = np.zeros(n)
                stall2 = 0
        if lr2 < 1e-7:
            break

    # Phase 3: Power mean smooth optimization
    for p in [16, 32, 64]:
        if remaining() < 100:
            break
        h_work = best_h.copy()
        v3 = np.zeros(n)
        lr3 = 0.01
        for it in range(5000):
            if remaining() < 100:
                break
            grad = pow_mean_grad(h_work, p)
            v3 = 0.8 * v3 - lr3 * grad
            h_work = project_feasible(h_work + v3, n)
            c5_val = compute_c5(h_work)
            if c5_val < best_c5 - 1e-10:
                best_h = h_work.copy()
                best_c5 = c5_val
            lr3 = max(lr3 * 0.9997, 0.0005)

    # Phase 4: Random perturbations + re-optimize
    for attempt in range(10):
        if remaining() < 80:
            break
        h_try = best_h + rng.normal(0, 0.05, n)
        h_try = project_feasible(h_try, n)
        # Quick local optimization
        for it in range(3000):
            if remaining() < 80:
                break
            grad = maximin_subgradient(h_try, tol=0.01)
            h_try = project_feasible(h_try + 0.005 * grad, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-10:
                best_h = h_try.copy()
                best_c5 = c5_try

    # Phase 5: Try different n values
    for n_try in [n + 10, n + 20, n - 10]:
        if n_try < 30 or remaining() < 100:
            continue
        old_n = len(best_h)
        old_x = np.linspace(0, 2, old_n, endpoint=False)
        new_x = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(new_x, old_x, np.concatenate([best_h, best_h[:1]]))
        h_interp = project_feasible(h_interp, n_try)
        for it in range(3000):
            if remaining() < 100:
                break
            grad = maximin_subgradient(h_interp, tol=0.01)
            h_interp = project_feasible(h_interp + 0.005 * grad, n_try)
            c5_try = compute_c5(h_interp)
            if c5_try < best_c5 - 1e-10:
                best_h = h_interp.copy()
                best_c5 = c5_try

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
