import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def normalize_sum(h, n):
    """Scale h to have sum = n/2, clipping to [0,1]"""
    s = np.sum(h)
    if s == 0:
        h[:n//2] = 1.0
        return h
    target = n / 2.0
    h = h / s * target
    # Clip and redistribute
    for _ in range(20):
        h = np.clip(h, 0, 1)
        new_sum = np.sum(h)
        if abs(new_sum - target) < 1e-10:
            break
        # Redistribute the difference
        diff = target - new_sum
        if diff > 0:
            mask = h < 1.0
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0, 1)
    return h

def structured_initial(n, seed):
    """Create a structured initial binary solution"""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    # Try different patterns

        h[:n//2] = 1.0
