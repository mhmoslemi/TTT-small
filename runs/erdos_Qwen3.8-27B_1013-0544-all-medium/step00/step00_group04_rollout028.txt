`np.correlate(a, v, mode='full')` computes:

For real arrays: c_k = sum_n a_{n+k} * v_n

So `np.correlate(h, 1-h, mode='full')` gives:
c_k = sum_n h_{n+k} * (1-h)_n = sum_n h_{n+k} * (1-h_n)

This is the cross-correlation of (1-h) with h, or equivalently, the "reverse" of what I defined above. But since we're taking the MAX, it doesn't matter (the max of the correlation and its reverse are the same... wait, no, they're not necessarily the same).

Actually, for real signals, the full cross-correlation R_xy[k] = sum_n x[n+k]*y[n] and R_yx[k] = sum_n y[n+k]*x[n] = R_xy[-k]. So the set of values is the same, just in reverse order. The MAX is the same!

So `max(np.correlate(h, 1-h, 'full'))` = `max(np.correlate(1-h, h, 'full'))`. Good, no issue.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search(h, rng, t_end):
    n = len(h)
    m = n // 2
    ones = np.where(h == 1)[0].astype(np.int64).copy()
    zeros = np.where(h == 0)[0].astype(np.int64).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = 0.008
    no_improve = 0
    while time.time() < t_end:
        a = int(rng.integers(m))
        b = int(rng.integers(m))
        i = ones[a]
        j = zeros[b]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones[a] = j
            zeros[b] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
                T = max(T * 0.98, 1e-7)
            no_improve += 1
        else:
            h[j] = 0.0
            h[i] = 1.0
        T = max(T * 0.99998, 1e-9)
        if no_improve > 15000:
            h = best_h.copy()
            cur_c5 = best_c5
            ones = np.where(h == 1)[0].astype(np.int64).copy()
            zeros = np.where(h == 0)[0].astype(np.int64).copy()
            npert = int(rng.integers(2, 9))
            for _ in range(npert):
                a2 = int(rng.integers(m))
                b2 = int(rng.integers(m))
                i2 = ones[a2]
                j2 = zeros[b2]
                h[i2] = 0.0
                h[j2] = 1.0
                ones[a2] = j2
                zeros[b2] = i2
            cur_c5 = compute_c5(h)
            no_improve = 0
            T = 0.006
    return best_h, best_c5
