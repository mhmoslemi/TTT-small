        if stall > 400:
            break
        best_gain = 0.0
        best_pair = None
        ones_list = list(ones)
        zeros_list = list(zeros)
        # Use all pairs for small n, random subset for large
        if len(ones_list) * len(zeros_list) > 800:
            idx = rng.integers(0, len(ones_list), size=300)
            jdx = rng.integers(0, len(zeros_list), size=300)
            pairs = [(ones_list[i], zeros_list[j]) for i, j in zip(idx, jdx)]
        else:
            pairs = [(i, j) for i in ones_list for j in zeros_list]
        for i, j in pairs:
            h[i] = 0.0
            h[j] = 1.0
            mc = max_corr_val(h)
            h[i] = 1.0
            h[j] = 0.0
            gain = cur - mc
            if gain > best_gain:
                best_gain = gain
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            cur -= best_gain
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            stall += 1
            if stall > 80 and ones and zeros:
                for _ in range(3):
                    i = int(rng.choice(list(ones)))
                    j = int(rng.choice(list(zeros)))
                    h[i] = 0.0
                    h[j] = 1.0
                    ones.discard(i)
                    ones.add(j)
                    zeros.discard(j)
                    zeros.add(i)
                cur = max_corr_val(h)
                stall = 0
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = eval_c5(best_h, 2.0 / n0)
    best_n = n0
    deadline = start_time + budget_s * 0.93

    # Phase 1: Binary local search on promising even n values
    # Allocate budget: 50% to n=44, 20% to n=46/48, 15% to n=50-60, 15% reserve
    phases = [
        (44, 0.45), (46, 0.15), (48, 0.12), (50, 0.08),
        (52, 0.06), (54, 0.05), (56, 0.04), (58, 0.03), (60, 0.02)
    ]
    for n, frac in phases:
        phase_end = start_time + (deadline - start_time) * sum(f for _, f in phases[:phases.index((n, frac))+1])
        phase_end = min(phase_end, deadline)
        if time.time() >= phase_end:
            continue
        n_half = n // 2
        dx = 2.0 / n
        n_starts = max(2, int((phase_end - time.time()) / 15))

        for s in range(n_starts):
            if time.time() >= phase_end or time.time() >= deadline:
                break
            # Initialize
            if s == 0:
                x_o = np.linspace(0, 2, best_n)
                x_n = np.linspace(0, 2, n)
                h_i = np.interp(x_n, x_o, best_h)
                h_bin = (h_i >= 0.5).astype(int)
            else:
                h_bin = np.zeros(n, dtype=int)
                h_bin[rng.choice(n, n_half, replace=False)] = 1
            # Fix count
            cnt = int(h_bin.sum())
            if cnt > n_half:
                h_bin[rng.choice(np.where(h_bin == 1)[0], cnt - n_half, replace=False)] = 0
            elif cnt < n_half:
                h_bin[rng.choice(np.where(h_bin == 0)[0], n_half - cnt, replace=False)] = 1
            h_b, corr_b = local_search_binary(h_bin, phase_end, rng)
            c5_v = corr_b * dx
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h_b.astype(float)
                best_n = n

    # Phase 2: Continuous refinement on best_n
    if time.time() < deadline - 10:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        h = best_h.astype(float).copy()

        def _obj(x):
            hh = project_feasible(x, target)
            return eval_c5(hh, dx)

        try:
            res = minimize(_obj, h, method='Nelder-Mead',
                          options={'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-14})
            h_o = project_feasible(res.x, target)
            c5_o = eval_c5(h_o, dx)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o
        except Exception:
            pass

    # Phase 3: Quick SLSQP polish
    if time.time() < deadline - 5:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        h = best_h.astype(float).copy()
        try:
            res = minimize(lambda x: eval_c5(project_feasible(x, target), dx),
                          h, method='COBYLA',
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                          options={'maxiter': 2000, 'rhof': 1e-12, 'catol': 1e-12})
            h_o = project_feasible(res.x, target)
            c5_o = eval_c5(h_o, dx)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o
        except Exception:
            pass

    # Final verification
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = eval_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
