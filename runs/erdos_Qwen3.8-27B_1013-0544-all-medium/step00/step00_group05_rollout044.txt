def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, n_points, alpha=100):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * corr
    a_max = np.max(a)
    return float(dx * (a_max + np.log(np.sum(np.exp(a - a_max)))) / alpha)

    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

        h[mask] = np.clip(h[mask] + step, 0, 1)
