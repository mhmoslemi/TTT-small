def smooth_objective(h, tau=50, mu=1000):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max
    max_corr = np.max(corr)
    shifted = corr - max_corr
    soft_max = max_corr + np.log(np.sum(np.exp(tau * shifted))) / tau
    # Penalty for sum constraint
    sum_penalty = mu * (np.sum(h) - len(h) / 2) ** 2
    return soft_max + sum_penalty

def smooth_gradient(h, tau=50, mu=1000):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = tau * (corr - max_corr)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    # Gradient of soft max
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        i_start1 = max(0, -lag)
        i_end1 = n - max(0, lag)
        if i_start1 < i_end1:
            grad[i_start1:i_end1] += weights[k] * (1.0 - h[i_start1+lag:i_end1+lag])
        i_start2 = max(0, lag)
        i_end2 = n + min(0, lag)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= weights[k] * h[i_start2-lag:i_end2-lag]
    # Gradient of penalty
    grad += 2 * mu * (np.sum(h) - n / 2)
