    # Use area-based: each new point gets the value of the old bin it's in

    old_edges = np.linspace(0, 2, n + 1)
    new_edges = np.linspace(0, 2, new_n + 1)
    
    # For each new bin, compute the integral of h over that bin

    # Actually, let's use a proper approach
    # Each old bin [old_edges[i], old_edges[i+1]] has value h[i]
    # For each new bin, find the overlap with each old bin

    old_centers = (old_edges[:-1] + old_edges[1:]) / 2
    new_centers = (new_edges[:-1] + new_edges[1:]) / 2
