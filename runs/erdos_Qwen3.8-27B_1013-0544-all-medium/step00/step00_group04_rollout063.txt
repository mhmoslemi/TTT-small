def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(round(float(np.max(corr))))

        j = zeros[rng.randrange(len(zeros))]

        delta = new - current

def make_random(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.float64)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return h

def fix_sum(h, target):
    d = target - int(h.sum())
    if d > 0:
        h[np.where(h == 0)[0][:d]] = 1.0
    elif d < 0:
        h[np.where(h == 1)[0][:(-d)]] = 0.0
    return h

def interpolate_to(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    ox = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    nx = np.linspace(0, 2, n_target, endpoint=False) + 1.0 / n_target
    hi = np.interp(nx, ox, h_src)
    h_out = (hi >= 0.5).astype(np.float64)
    fix_sum(h_out, n_target // 2)
    return h_out
