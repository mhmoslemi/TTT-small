    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def gradient_c_at_k(h, k):
    """Compute gradient of c(k) with respect to h.
    k is the lag (can be negative).
    c(k) = sum_i h[i+k]*(1-h[i]) over valid i.
    """
    n = len(h)

        for m in range(k, n):

        # Second term: -h[m+k] for m = 0, ..., n-k-1
        for m in range(0, n - k):
            grad[m] -= h[m + k]

        # Second term: -h[m-p] for m = p, ..., n-1
        for m in range(p, n):
            grad[m] -= h[m - p]

            # Since others are already in [0,1], we just need 0 <= target_sum - other_sum <= 1

            # For now, just try h[m] = 0, 0.5, 1 and the constrained value
            candidates = [0.0, 1.0, target_sum - other_sum]

import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return np.max(compute_corr(h))

def gradient_c_at_k(h, k, n, dx):
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        p = -k
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] -= h[:n-p]
    return grad * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-13:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
