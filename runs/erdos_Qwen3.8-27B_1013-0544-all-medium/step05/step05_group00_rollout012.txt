def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        sh = alpha * (g - m)

        return x.sum() - n / 2.0

def subgradient_descent(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    step_size = 0.01
    n_corr = 2*n - 1
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1-h, mode='full')
        c_max = corr.max()
        c = c_max * dx
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Subgradient: gradient of the active constraint
        # d/dh[i] of corr[k*] where k* = argmax
        k_star = np.argmax(corr)
        # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # d/dh[i] corr[k] = (1-h[i+k-(n-1)]) - h[i-(k-(n-1))] (if valid)
        # = (1-h[i+k-n+1]) - h[i-k+n-1] (with appropriate bounds)
        
        grad = np.zeros(n)
        offset = k_star - (n-1)  # This is the shift k (can be negative)
        
        for i in range(n):
            # Term 1: h[i] * (1-h[i+offset])
            j1 = i + offset
            if 0 <= j1 < n:
                grad[i] += (1 - h[j1])
            # Term 2: h[i] appears in (1-h) when i = v + offset, i.e., v = i - offset
            j2 = i - offset
            if 0 <= j2 < n:
                grad[i] -= h[j2]
        
        # Negate for minimization
        h = h - step_size * grad
        h = project_feasible(h, n)
        
        # Decay step size
        step_size *= 0.9999
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
