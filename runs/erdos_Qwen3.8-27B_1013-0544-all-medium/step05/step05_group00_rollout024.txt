def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

    def obj(x):

        # Sum of top m
        cs = np.sort(c)[::-1][:m]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

        return h

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / s) * room)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / s) * h)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})

        return h
