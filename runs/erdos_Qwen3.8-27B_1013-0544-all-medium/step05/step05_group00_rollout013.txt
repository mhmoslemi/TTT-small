import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def swap_delta(h, n, i, j, ci):
    lag = ci - (n - 1)
    if lag == 0:
        return 0.0
    delta = 0.0
    for p in (i, j, i - lag, j - lag):
        if p < 0 or p >= n:
            continue
        pl = p + lag
        if pl < 0 or pl >= n:
            continue
        hp = h[p]
        hpl = h[pl]
        old = hp * (1.0 - hpl)
        hp_n = 0.0 if p == i else (1.0 if p == j else hp)
        hpl_n = 0.0 if pl == i else (1.0 if pl == j else hpl)
        new = hp_n * (1.0 - hpl_n)
        delta += new - old
    return delta

def two_swap_delta(h, n, i1, j1, i2, j2, ci):
    lag = ci - (n - 1)
    if lag == 0:
        return 0.0
    delta = 0.0
   removed = {i1, i2}
    added = {j1, j2}
    for p in (i1, i2, j1, j2, i1 - lag, i2 - lag, j1 - lag, j2 - lag):
        if p < 0 or p >= n:
            continue
        pl = p + lag

        hp = h[p]

        hpl_n = 0.0 if pl in removed else (1.0 if pl in added else hpl)
        new = hp_n * (1.0 - hpl_n)
        delta += new - old
    return delta

def fast_local_search(h_init, n, deadline, seed=42, use_two_swap=True):

    h = make_binary(h_init, n)
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = int(round(c.max()))
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)

    while time.time() < deadline:

        thresh = C - 2
        top = np.where(c >= thresh)[0]
        if len(top) > max_top:
            tv = c[top]
            top = top[np.argsort(tv)[::-1][:max_top]]

        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])

        improved = False

            if improved:

                    if c[ci] + d > C - 1.0 + 1e-9:
                        ok = False
                        break
                if ok:
                    h[i] = 1.0
                    h[j] = 0.0

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
