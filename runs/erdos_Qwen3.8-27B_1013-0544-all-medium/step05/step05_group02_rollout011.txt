def smooth_max_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlation values."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Shift c so it's centered (or just work with it as-is)
    # c has 2n-1 entries
    shifted = alpha * (c - c.max())
    w = np.exp(shifted) / np.exp(shifted).sum()
    # Now compute weighted gradient
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        # grad from h[i] * (1-h[i+s]) for valid i
        # For h[p]: contributes (1-h[p+s]) if 0<=p+s<n
        # For h[p]: contributes -h[p-s] if 0<=p-s<n
        if s >= 0:
            # p+s valid when p < n-s
            valid_src = np.arange(0, n - s)
            grad[valid_src] += w[k] * (1 - h[valid_src + s])
            valid_dst = np.arange(s, n)
            grad[valid_dst] += w[k] * (-h[valid_dst - s])
        else:  # s < 0
            # p+s valid when p >= -s
            valid_src = np.arange(-s, n)
            grad[valid_src] += w[k] * (1 - h[valid_src + s])
            valid_dst = np.arange(0, n + s)
            grad[valid_dst] += w[k] * (-h[valid_dst - s])
    grad *= dx
    return grad, c.max()

    return float(np.max(c))

    target = n / 2.0
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0

        x = np.clip(h_new - mid, 0.0, 1.0)

        else:

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    shifted = alpha * (c - c.max())
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()

        w = np.zeros_like(c)
        w[c.argmax()] = 1.0

            n_init = n_try
    # Phase 3: Restarts with noise
    n = len(best_h)
    restart_budget = deadline - time.time() - 50
    if restart_budget > 30:
        n_restarts = max(3, int(restart_budget / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.003 + 0.04 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_box_sum(hp, n)
            alloc = min(70, (deadline - time.time() - 50) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = pg_optimize(hp, n, time.time() + alloc, alpha_lo=200, alpha_hi=100000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 4: Multi-lag polish
    if time.time() < deadline - 20:
        n = len(best_h)
        ml_deadline = min(deadline - 10, time.time() + 30)
        best_h, best_c = multi_lag_polish(best_h, n, ml_deadline, num_lags=3)
    # Phase 5: Final directed polish
    if time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = directed_polish(best_h, n, deadline - 1)
        best_h = project_box_sum(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
