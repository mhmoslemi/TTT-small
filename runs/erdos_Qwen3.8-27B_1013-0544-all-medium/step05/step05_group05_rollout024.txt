def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

            h += diff / s * h

= (1-h[i+k]) [if i+k in range] - (1-h[j+k]) [if j+k in range] - h[i-k] [if i-k in range] + h[j-k] [if j-k in range]

All using OLD h values.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (1-indexed values 0 or 1)."""
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def greedy_improve(h, n, deadline, max_rounds=200):
    """Greedy best-swap improvement for binary h."""
    best_c = compute_c5(h, n)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 0
        best_i, best_j = -1, -1
        # For n=250, 125*125=15625 pairs; compute full corr each time

                c = compute_c5(h, n)

                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
                h[i], h[j] = 0.0, 1.0
        if best_i < 0:
            break
        h[best_i], h[best_j] = 1.0, 0.0
