```python
    """Given current correlation c, compute new correlation after swapping h[i]:0→1 and h[j]:1→0"""
    # delta_l for lag l (l ranges from -(n-1) to n-1)
    # c is indexed as c[k] where k = l + (n-1), so k ranges from 0 to 2n-2
    delta = np.zeros(2*n-1)
    # For each valid lag l:
    # delta_l = (h[j+l] - h[i+l]) + (h[j-l] - h[i-l])
    # where out-of-bounds h[.] = 0
    
    # h[i+l] is nonzero only when 0 <= i+l < n, i.e., -i <= l < n-i
    # h[j+l] is nonzero only when 0 <= j+l < n, i.e., -j <= l < n-j
    # h[i-l] is nonzero only when 0 <= i-l < n, i.e., l < i and l >= i-n+1... 
    # i-l >= 0 → l <= i; i-l < n → l > i-n → l >= i-n+1 (since l is integer)
    # So h[i-l] nonzero for max(-(n-1), i-n+1) <= l <= min(n-1, i)
    # Since 0 <= i < n: i-n+1 <= 0 and i < n, so: max(-(n-1), i-n+1) <= l <= min(n-1, i)
    # i-n+1 <= 0 (since i < n), so lower bound is -(n-1)
    # Upper bound is min(n-1, i) = i (since i < n)
    
    # This is getting complicated. Let me just vectorize it:
    lags = np.arange(-(n-1), n)  # all possible lags
    # h_at_offset: h[pos] if 0 <= pos < n, else 0
    def h_at(pos):
        mask = (pos >= 0) & (pos < n)
        result = np.zeros_like(pos, dtype=float)
        result[mask] = h[pos[mask]]
        return result
    
    delta = (h_at(j + lags) - h_at(i + lags)) + (h_at(j - lags) - h_at(i - lags))
    return c + delta
```

This is O(n) per swap evaluation! Much better than O(n²).

```python
        g[:n+k] += (1.0 - x[-k:])
        g[-k:] -= x[:n+k]
    g *= dx
    g += 2e4 * (x.sum() - n/2.0)
    return g

def lbfgs_phase(h, n, alphas, max_iter_each, deadline):
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h, method='L-BFGS-B', jac=smooth_grad,
                          args=(n, alpha), bounds=bounds,
                          options={'maxiter': max_iter_each, 'ftol': 1e-16, 'gtol': 1e-8})
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new, n) < compute_c5(h, n):
                h = h_new
        except Exception:
            pass
    return h

def binary_sa_fast(h, n, deadline, rng, sa_start=None):
    """Binary SA with O(n) incremental swap evaluation"""
    if sa_start is None:
        sa_start = time.time()
    
    # Ensure binary with exactly n/2 ones
    h = np.rint(h).astype(float)
    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:num_ones-target]] = 0.0
    elif num_ones < target:
        zidx = np.where(h == 0)[0]
        rng.shuffle(zidx)
        h[zidx[:target-num_ones]] = 1.0
    
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = c.max()
    best_c5 = cur_max * dx
    best_h = h.copy()
    
    lags = np.arange(-(n-1), n)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    total_time = max(deadline - sa_start, 1.0)
    
    while time.time() < deadline:
        if not ones_list or not zeros_list:
            break
        elapsed = time.time() - sa_start
        fraction = min(elapsed / total_time, 1.0)
        temp = max(0.0005, 0.05 * (1.0 - fraction)**2)
        
        ii = rng.randint(len(zeros_list))
        jj = rng.randint(len(ones_list))
        i = zeros_list[ii]
        j = ones_list[jj]
        
        h_ip = get_h_at(h, i + lags, n)
        h_jp = get_h_at(h, j + lags, n)
        h_im = get_h_at(h, i - lags, n)
        h_jm = get_h_at(h, j - lags, n)
        
        delta = (h_jp - h_ip) + (h_jm - h_im)
        c_new = c + delta
        new_max = c_new.max()
        new_c5 = new_max * dx
        cur_c5 = cur_max * dx
        
        if new_c5 <= cur_c5:
            h[i], h[j] = 1.0, 0.0
            c = c_new
            cur_max = new_max
            ones_list[jj] = ones_list[-1]; ones_list.pop()
            zeros_list[ii] = zeros_list[-1]; zeros_list.pop()
            ones_list.append(i)
            zeros_list.append(j)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
                h[i], h[j] = 1.0, 0.0
                c = c_new
                cur_max = new_max
                ones_list[jj] = ones_list[-1]; ones_list.pop()
                zeros_list[ii] = zeros_list[-1]; zeros_list.pop()
                ones_list.append(i)
                zeros_list.append(j)
    
    return best_h, best_c5
```
