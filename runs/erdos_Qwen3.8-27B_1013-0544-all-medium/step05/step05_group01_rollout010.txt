    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_lag(h, n, lag):
    """Compute subgradient of f_lag with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        grad[:n+lag] += (1 - h[-lag:])
        grad[-lag:] -= h[:n+lag]
    return grad * dx

def weighted_grad(h, n, alpha=10000):
    """Compute gradient of log-sum-exp smooth approximation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * (c - c.max())
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += weights[k] * grad_lag(h, n, lag)
    return grad

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_lag(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        nl = -lag
        if n + lag > 0:
            grad[:n + lag] += (1.0 - h[nl:])
            grad[nl:] -= h[:n + lag]
    return grad * dx

def guided_search(h, n, deadline, max_iters=60000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 500 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(c)[-5:]
        g = np.zeros(n)
        for k in top_k:
            if c[k] < c.max() - 1e-8:
                continue
            lag = k - (n - 1)
            g += grad_lag(h, n, lag)
        order = np.argsort(g)
        best_gain = 0.0
        best_move = None
        top_n = min(40, n)
        for a in range(top_n):
            i = order[a]
            for b in range(n - top_n, n):
                j = order[b]
                if i == j:
                    continue
                gain = g[j] - g[i]
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
        if best_move is None or best_gain < 1e-9:
            break
        i, j = best_move
        step = min(0.004, h[j] * 0.4, (1.0 - h[i]) * 0.4)
        if step < 1e-10:
            break
        h[i] += step
        h[j] -= step
        h = project_feasible(h, n)
    final_c = compute_c5(h, n)
    if final_c < best_c:
        best_c = final_c
    return h, best_c

def fine_pair_search(h, n, deadline, radius=45):
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        max_outer = 500
        outer = 0
        while improved and time.time() < deadline and outer < max_outer:
            improved = False
            outer += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[j] >= step * 1.01 and h[i] <= 1.0 - step * 1.01:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n = 100

    best_h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    p1_deadline = t0 + (deadline - t0) * 0.30
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    for a in alphas:
        if time.time() > p1_deadline:
            break
        remaining = p1_deadline - time.time()
        if remaining < 15:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=4000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht.copy()

    # Phase 2: Random restarts from best
    if time.time() < deadline - 120:
        p2_deadline = t0 + (deadline - t0) * 0.42
        for restart in range(3):
            if time.time() > p2_deadline:
                break
            scale = [0.01, 0.02, 0.04][restart]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            ht = slsqp_optimize(hp, n, 2000000, maxiter=3000, ftol=1e-16)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht.copy()

    # Phase 3: Guided local search
    if time.time() < deadline - 60:
        p3_deadline = t0 + (deadline - t0) * 0.58
        best_h, c3 = guided_search(best_h, n, p3_deadline, max_iters=80000)
        if c3 < best_c:
            best_c = c3

    # Phase 4: SLSQP polish
    if time.time() < deadline - 50:
        ht = slsqp_optimize(best_h, n, 20000000, maxiter=6000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()

    # Phase 5: Fine pair search (most important for final improvement)
    if time.time() < deadline - 5:
        best_h, c5 = fine_pair_search(best_h, n, deadline - 2, radius=50)
        if c5 < best_c:
            best_c = c5

    # Phase 6: Final guided search with remaining time
    if time.time() < deadline - 2:
        best_h, c6 = guided_search(best_h, n, deadline - 1, max_iters=20000)
        if c6 < best_c:
            best_c = c6

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
