```python
def compute_subgrad(h, n, active_lags):
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in active_lags:
        # Term 1: (1-h[i+lag]) for valid i+lag
        lo = max(0, -lag)
        hi = min(n, n - lag)
        grad[lo:hi] += (1 - h[lo+lag:hi+lag])
        # Term 2: -h[i-lag] for valid i-lag
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        # i ranges from lo2 to hi2-1, and i-lag ranges from lo2-lag to hi2-1-lag
        # h[i-lag] = h[lo2-lag : hi2-lag]
        grad[lo2:hi2] -= h[lo2-lag:hi2-lag]
    grad *= dx / len(active_lags)
    return grad
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    # Bisection on lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute the difference
        if diff > 0:
            room = 1 - h
            s = room.sum()
            if s > 0:
                h += diff / s * room
        else:
            s = h.sum()
            if s > 0:
                h += diff / s * h
    return np.clip(h, 0, 1)
```

Instead of np.correlate(h, 1-h, mode='full'), I can compute it as:
g[k] = sum(h[i] * (1-h[i+k])) for valid i

This is the same as:
g[k] = (sum of h over valid range) - (h · h_shifted[k])

For k=0: g[0] = sum(h) - sum(h²) = n/2 - sum(h²)
For k≠0: g[k] = sum(h over valid range) - sum(h[i]*h[i+|k|])

The sum(h over valid range) depends on k:
- For k > 0 (positive lag): valid i are 0 to n-1-k, so sum = sum(h[0:n-k])
- For k < 0 (negative lag): valid i are |k| to n-1, so sum = sum(h[|k|:n])
- For k = 0: sum = sum(h)

So I can precompute prefix sums of h and compute g[k] quickly.

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-15:
                h += (diff / s) * room
        else:
            s = h.sum()
            if s > 1e-15:
                h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def subgrad_at(h, n, k_star):
    dx = 2.0 / n
    lag = k_star - (n - 1)
    g = np.zeros(n)
    lo, hi = max(0, -lag), min(n, n - lag)
    if lo < hi:
        g[lo:hi] += 1.0 - h[lo + lag:hi + lag]
    lo2, hi2 = max(0, lag), min(n, n + lag)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 - lag:hi2 - lag]
    return g * dx

def slsqp_step(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        s = alpha * (g - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_descent(h, n, deadline, rng, K=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.002
    stall = 0
    while time.time() < deadline:
        g = compute_g(h, n)
        C = g.max()
        top = np.argsort(g)[::-1][:K]
        grad = np.zeros(n)
        for k in top:
            grad += subgrad_at(h, n, k)
        grad /= K
        a = alpha
        found = False
        for _ in range(14):
            hn = project_feasible(h - a * grad, n)
            if compute_c5(hn, n) < C - 1e-13:
                h = hn
                found = True
                break
            a *= 0.4
        if found:
            stall = 0
            alpha = min(a * 1.3, 0.04)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            stall += 1
            alpha = max(a * 0.5, 1e-7)
            if stall > 60:
                s = 0.003 * (1.0 + min(stall / 60.0, 2.0))
                hp = project_feasible(h + s * rng.randn(n), n)
                if compute_c5(hp, n) < best_c:
                    h = hp
                else:
                    h = best_h.copy()
                alpha = 0.002
                stall = 0
    return best_h, best_c

def shake_refine(h, n, deadline, rng, rounds=30):

    for r in range(rounds):

def binary_polish(h, n, deadline):
    if n % 2 != 0:
        return h, compute_c5(h, n)
    target = n // 2
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[::-1][:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1.0)[0].tolist())
    zeros = set(np.where(h_bin == 0.0)[0].tolist())
    for _ in range(20):
        if time.time() > deadline:
            break
        improved = False
        zlist = sorted(zeros)
        olist = sorted(ones)
        for z in zlist:
            if time.time() > deadline:
                break
            for o in olist:
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.discard(o); ones.add(z)
                    zeros.discard(z); zeros.add(o)
                    olist = sorted(ones); zlist = sorted(zeros)
                    improved = True
                else:
                    h_bin[z] = 0.0; h_bin[o] = 1.0
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c
```
