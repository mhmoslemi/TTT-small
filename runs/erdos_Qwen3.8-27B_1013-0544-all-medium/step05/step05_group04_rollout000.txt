                else:
                    accept = rng.random() < np.exp(log_a)
            else:
                accept = False
            if accept:
                cur_c = new_max
                n_accepted += 1
            else:
                cur_h[i] = cur_h[i] - d
                cur_h[j] = cur_h[j] + d
            no_improve += 1

        if no_improve > restart_thresh:
            cur_h = best_h.copy()
            cur_c = best_c
            no_improve = 0

    return best_h, best_c

def local_refine(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    dx = 2.0 / n
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for d in deltas:
        while time.time() < deadline:
            improved = False
            order = rng.permutation(n)
            for i in order:
                if time.time() > deadline:
                    return cur_h, best_c
                for _ in range(5):
                    j = rng.randint(n)
                    if i == j:
                        continue
                    cur_h[i] += d
                    cur_h[j] -= d
                    if 0.0 <= cur_h[i] <= 1.0 and 0.0 <= cur_h[j] <= 1.0:
                        c = compute_c5(cur_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            cur_h[i] -= d
                            cur_h[j] += d
                    else:
                        cur_h[i] -= d
                        cur_h[j] += d
                    cur_h[i] -= d
                    cur_h[j] += d
                    if 0.0 <= cur_h[i] <= 1.0 and 0.0 <= cur_h[j] <= 1.0:
                        c = compute_c5(cur_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            cur_h[i] += d
                            cur_h[j] -= d
                    else:
                        cur_h[i] += d
                        cur_h[j] -= d
            if not improved:
                break
    return cur_h, best_c

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    alphas = [2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 8:
            break
        ht = slsqp(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    if time.time() < deadline - 5:
        h, best_c = local_refine(h, n, deadline - 3, rng)

    if time.time() < deadline - 3:
        ht = slsqp(h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
        if time.time() < deadline - 1:
            h, best_c = local_refine(h, n, deadline - 1, rng)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(150, 0.5)
        n_init = 150

    best_h = None
    best_c = float('inf')

    # Phase 1: Heavy SA on initial n (global exploration)
    alloc1 = min(400, (deadline - t0) * 0.38)
    h_sa, c_sa = simulated_annealing(h_init, n_init, t0 + alloc1, rng,
                                     T0=0.006, T1=1e-8, max_iter=1000000, delta0=0.03)
    best_h, best_c = h_sa, c_sa

    # Phase 2: SLSQP refinement on SA result
    if time.time() < deadline - 120:
        alloc2 = min(150, (deadline - time.time()) * 0.2)
        h_opt, c_opt = full_optimize(best_h, n_init, time.time() + alloc2, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Second SA round at lower temperature (polishing exploration)
    if time.time() < deadline - 100:
        alloc3 = min(180, (deadline - time.time()) * 0.3)
        h_sa2, c_sa2 = simulated_annealing(best_h, n_init, time.time() + alloc3, rng,
                                           T0=0.003, T1=1e-9, max_iter=800000, delta0=0.015)
        if c_sa2 < best_c:
            best_c = c_sa2
            best_h = h_sa2
        # Quick SLQSQP after
        if time.time() < deadline - 30:
            h_opt, c_opt = full_optimize(best_h, n_init, time.time() + min(40, (deadline - time.time()) * 0.3), rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Try different n values (coarser discretization may give lower C)
    if best_h is not None and time.time() < deadline - 80:
        x_init = np.linspace(0, 2, n_init)
        for n_try in [120, 180, 200]:
            if time.time() > deadline - 50:
                break
            if n_try == n_init:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(100, (deadline - time.time()) * 0.3)
            if alloc < 30:
                break
            # SA on new n
            h_new, c_new = simulated_annealing(h_interp, n_try, time.time() + alloc * 0.5, rng,
                                               T0=0.004, T1=1e-8, max_iter=400000, delta0=0.025)
            if time.time() < deadline - 20:
                h_new, c_new = full_optimize(h_new, n_try, time.time() + min(50, (deadline - time.time()) * 0.3), rng)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 5: Final polish on best solution
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        h_final, c_final = full_optimize(best_h, n_best, deadline - 2, rng)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
