  from scipy.optimize import minimize

  def compute_c5(h, n):

      return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

  def thue_morse(n):
      # Generate first n terms of Thue-Morse sequence
      t = np.zeros(n, dtype=float)
      for i in range(1, n):
          t[i] = t[i >> 1] ^ (i & 1)
      return t

      k = n // 2
      idx = np.argsort(h)[::-1]
      out = np.zeros(n)
      out[idx[:k]] = 1.0
      return out

  def project_feasible(h, n):
      target = n / 2.0
      h = np.clip(h, 0.0, 1.0)
      for _ in range(200):

          if abs(diff) < 1e-14:
              break
          if diff > 0:

              s = room.sum()
              if s < 1e-14: break
              h += diff / s * room
          else:
              s = h.sum()
              if s < 1e-14: break
              h += diff / s * h
      return np.clip(h, 0.0, 1.0)

      target = n // 2

      best_c = compute_c5(h, n)
      rng = np.random.RandomState(seed)

              j = ones[rng.randint(len(ones))]
