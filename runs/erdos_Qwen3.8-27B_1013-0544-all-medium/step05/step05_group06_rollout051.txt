    """
    Compute weighted sum of gradients for active lags.
    active_lags: array of lag values k (can be negative)
    weights: array of same length
    Returns: gradient vector of length n
    """

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g_raw(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, K, alpha):
    """One subgradient descent step on C5."""
    g = compute_g_raw(h, n)
    # Find top K values
    top_idx = np.argsort(g)[::-1][:K]
    top_vals = g[top_idx]
    # Softmax weights
    shifted = top_vals - top_vals.max()
    weights = np.exp(10 * shifted) / np.exp(10 * shifted).sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for k_idx, m in enumerate(top_idx):
        w = weights[k_idx]
        if w < 1e-6:
            continue
        lag = m - (n - 1)
        # Term 1: (1-h[i-lag]) for i in [max(0,lag), min(n, n+lag))
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[lo1:hi1] += w * (1.0 - h[idx - lag])
        # Term 2: -h[i+lag] for i in [max(0,-lag), min(n, n-lag))
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[lo2:hi2] -= w * h[idx + lag]
    
    # Projected gradient descent step with backtracking
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = alpha
    for _ in range(10):
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            break
        step *= 0.5
    return best_h, best_c

def slsqp_smooth_max(h, n, alpha, maxiter=200):
    """SLSQP on smooth max approximation."""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        # Smooth max: m + (1/alpha) * log(sum(exp(alpha*(g-m))))
        shifted = alpha * (g - m)
        shifted = np.clip(shifted, -50, 50)  # numerical stability
        sm = np.log(np.exp(shifted).sum())  # log-sum-exp
        return m + sm / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted = np.clip(shifted, -50, 50)
        exp_shifted = np.exp(shifted)
        sum_exp = exp_shifted.sum()
        # d/dx of smooth max = sum_k w_k * dc(k)/dx where w_k = exp(alpha(c(k)-m))/sum
        w = exp_shifted / sum_exp
        # Compute gradient of c(k) for each k, weighted
        grad = np.zeros(n)
        for k_idx in range(len(g)):
            if w[k_idx] < 1e-8:
                continue
            lag = k_idx - (n - 1)
            lo1 = max(0, lag)
            hi1 = min(n, n + lag)
            if lo1 < hi1:
                idx = np.arange(lo1, hi1)
                grad[lo1:hi1] += w[k_idx] * dx * (1.0 - x[idx - lag])
            lo2 = max(0, -lag)
            hi2 = min(n, n - lag)
            if lo2 < hi2:
                idx = np.arange(lo2, hi2)
                grad[lo2:hi2] -= w[k_idx] * dx * x[idx + lag]
        return grad
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_fun_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_fun_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g_raw(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    """Compute F[i, m] = dc(m)/dh[i] for all i, m."""
    n_ = 2 * n - 1
    F = np.zeros((n, n_))
    for m in range(n_):
        lag = m - (n - 1)
        # Term 1: (1-h[i-lag]) for i in [max(0,lag), min(n, n+lag))
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            F[lo1:hi1, m] += (1.0 - h[lo1 - lag:hi1 - lag])
        # Term 2: -h[i+lag] for i in [max(0,-lag), min(n, n-lag))
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            F[lo2:hi2, m] -= h[lo2 + lag:hi2 + lag]
    return F

def F_slsqp(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = np.clip(alpha * (g - m), -50, 50)
        return m + np.log(np.exp(sh).sum()) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = np.clip(alpha * (g - m), -50, 50)
        e = np.exp(sh)
        w = e / e.sum()
        F = compute_F(x, n)
        return (w @ F) * dx
    def cfun(x):
        return x.sum() - n / 2.0

        return np.ones(n)
    try:

def pair_search(h, n, deadline, n_pairs_per_sweep=2000, max_sweeps=5000):

        cur_max = g.max()

        # Find active set

        threshold = g_sorted[min(n_active, len(g)-1)]
        active = np.where(g >= threshold - 1e-10)[0]

            j = rng.randint(n)

            if d_hi - d_lo < 1e-12:
                continue
            s = F[i] - F[j]
            # Find optimal delta: minimize max(g + d*s)
            # For active constraints: g[m] + d*s[m] should be minimized
            # The max is achieved at some m where g[m] + d*s[m] is largest
            # Optimal d is where two active constraints cross
            candidates = [d_lo, d_hi]
            # For each pair of active m1, m2: d = (g[m1]-g[m2])/(s[m2]-s[m1])
            act_s = s[active]
            act_g = g[active]
            for a1 in range(min(len(active), 8)):
                for a2 in range(a1+1, min(len(active), 8)):
                    ds = act_s[a2] - act_s[a1]
                    if abs(ds) > 1e-12:
                        d = (act_g[a1] - act_g[a2]) / ds
                        if d_lo < d < d_hi:
                            candidates.append(d)
            best_d = 0.0
            best_val = cur_max
            for d in candidates:
                d = float(np.clip(d, d_lo, d_hi))
                # Evaluate: max of g + d*s over active set (and check non-active)
                val = float(np.max(g + d * s))

                    best_val = val
                    best_d = d
            if best_d != 0.0:

                h[j] -= best_d
                g += best_d * s
                cur_max = best_val
                improved += 1
