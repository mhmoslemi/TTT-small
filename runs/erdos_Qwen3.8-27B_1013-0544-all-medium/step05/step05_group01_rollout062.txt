    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, 'full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    log_sum = mx + np.log(np.sum(np.exp(shifted - mx)))
    obj = m + log_sum / alpha
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue

            grad[j_lo:j_hi+1] += wk * (1.0 - h[idx1])
        # Second term: -h[idx2] for valid idx2 = j + k_off
        j_lo2 = max(0, -k_off)
        j_hi2 = min(n - 1, n - 1 - k_off)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1) + k_off
            grad[j_lo2:j_hi2+1] -= wk * h[idx2]
    
    return obj, grad

    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    if learning_rate is None:
        learning_rate = 2.0 / (alpha * np.max(np.abs(np.correlate(np.ones(n), np.ones(n), 'full'))) + 1)

        direction = -grad

                h = h_try
                obj_new = c_try
                if c_try < best_c:
                    best_c = c_try
                    best_h = h.copy()
                break
            lr *= 0.5
        
    return best_h, best_c

    return np.sum(x) - n / 2.0

    return np.ones(n)

                        else:

            if not improved:
                break
    return h, best_c

def _lse(s):
    mx = s.max()
    if mx > 500: return mx + np.log(s.size)
    if mx < -500: return mx
