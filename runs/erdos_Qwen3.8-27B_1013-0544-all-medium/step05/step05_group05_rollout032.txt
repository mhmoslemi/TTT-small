                break

            coeff = np.zeros(n_corr)

                k = s + n - 1
                c_val = 0.0
                if 0 <= i + s < n:
                    c_val += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    c_val -= h[i - s]
                coeff[k] = c_val

            # f(x) = max_k (base[k] + x * coeff[k]) = max_k (c_full[k] - hi*coeff[k] + x*coeff[k])
            #      = max_k (c_full[k] + (x-hi)*coeff[k])
            
            # So f(x) = max_k (c_full[k] + (x-hi)*coeff[k])
            # This is a convex PL function of (x-hi)

            # f(delta) = max_k (c_full[k] + delta*coeff[k]) where delta = x - hi
            # Optimal delta minimizes f, with constraint 0 <= hi+delta <= 1

            # - If all coeff[k] at the argmax are > 0, decrease x
            # - If all < 0, increase x
            # - Otherwise, find the crossing point
            
            # Simple approach: binary search on delta
            # f(delta) is convex, so we can use subgradient
            # At delta=0: subgradient is coeff[k*] where k* = argmax c_full
            
            # Even simpler: evaluate at delta = 0, and at the boundaries of feasibility
            # delta_min = -hi, delta_max = 1-hi
            
            delta_min = -hi
            delta_max = 1.0 - hi
            
            # Candidate deltas: 0, delta_min, delta_max, and where slope crosses 0
            # At delta=0, slope is coeff[argmax(c_full)]
            k_star = np.argmax(c_full)
            slope0 = coeff[k_star]
            
            candidates = [0.0, delta_min, delta_max]
            
            # If slope0 > 0, try negative deltas
            if slope0 > 0:
                # f is increasing at 0, so optimum is at negative delta (or at delta_min)
                # Find where slope changes sign
                # Try a few points
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    candidates.append(delta_min * frac)
            elif slope0 < 0:
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    candidates.append(delta_max * frac)
            
            best_delta = 0.0
            new_h_i = hi
            current_max = c_full.max()
            
            for d in candidates:
                d = np.clip(d, delta_min, delta_max)
                # f(d) = max_k (c_full[k] + d * coeff[k])
                val = np.max(c_full + d * coeff)
                if val < current_max - 1e-14:
                    current_max = val
                    best_delta = d
                    new_h_i = hi + d
            
            if best_delta != 0.0:
                h[i] = new_h_i
                c_full += best_delta * coeff
                if c_full.max() * dx < best_c - 1e-14:
                    best_c = c_full.max() * dx
                    improved = True
                else:
                    # Revert
                    c_full -= best_delta * coeff
                    h[i] = hi
        
        if not improved:
            break
    
    return h, best_c

                if abs(best_d) > 1e-14:
                    h_cur[i] = hi_val + best_d
                    c_full += best_d * coeff
        h_new = project_feasible(h_cur, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        cs = h.sum()
        if abs(cs - target) < 1e-8:
            break
        if cs > target:
            lam += max(0.02, abs(lam) * 0.4)
        else:
            lam -= max(0.02, abs(lam) * 0.4)
        if abs(lam) > 50.0:
            lam = 0.0
    return best_h, best_c

def slsqp_smooth(h0, n, alpha, deadline, maxiter=2000):
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = g - m
        return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def binary_search_opt(h0, n, deadline, seed=77):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)
    best_hb = hb.copy()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    st = 0.3
    se = 0.001
    max_it = 800000
    cur_c = best_c
    for it in range(max_it):
        if time.time() > deadline:
            break
        T = st * (se / st) ** (it / max_it)
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        a = zeros[int(rng.randint(len(zeros)))]
        b = ones[int(rng.randint(len(ones)))]
        hb[a] = 1.0
        hb[b] = 0.0
        nc = compute_c5(hb, n)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / max(T, 1e-15)):
            cur_c = nc
            if nc < best_c - 1e-14:
                best_c = nc
                best_hb = hb.copy()
        else:
            hb[a] = 0.0
            hb[b] = 1.0
    return best_hb, best_c

def final_refine(h, n, deadline):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    best_c = compute_c5(h, n)
    best_h = h.copy()
    s_arr = np.arange(-(n - 1), n)
    c_full = np.correlate(h, 1.0 - h, mode='full').copy()
    rng = np.random.RandomState(99)
    for sweep in range(200):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            idx_p = i + s_arr
            idx_m = i - s_arr
            mask_p = (idx_p >= 0) & (idx_p < n)
            mask_m = (idx_m >= 0) & (idx_m < n)
            coeff = np.zeros(n_corr)
            coeff[mask_p] += (1.0 - h[idx_p[mask_p]])
            coeff[mask_m] -= h[idx_m[mask_m]]
            hi_val = h[i]
            d_min = -hi_val
            d_max = 1.0 - hi_val
            if d_min >= d_max - 1e-15:
                continue
            best_d = 0.0
            cur_max = c_full.max()
            for bd in (d_min, d_max):
                fd = (c_full + bd * coeff).max()
                if fd < cur_max - 1e-14:
                    cur_max = fd
                    best_d = bd
            lo_d, hi_d2 = d_min, d_max
            for _ in range(35):
                mid = (lo_d + hi_d2) * 0.5
                tmp = c_full + mid * coeff
                fm = tmp.max()
                if fm < cur_max - 1e-14:
                    cur_max = fm
                    best_d = mid
                idx2 = np.argmax(tmp)
                sl = coeff[idx2]
                if sl > 1e-12:
                    hi_d2 = mid
                elif sl < -1e-12:
                    lo_d = mid
                else:
                    break
            if abs(best_d) > 1e-14:
                h[i] = hi_val + best_d
                c_full += best_d * coeff
                if c_full.max() * dx < best_c - 1e-14:
                    best_c = c_full.max() * dx
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Lagrangian coordinate descent (55% budget)
    p1 = t0 + 0.55 * (budget_s - 5)
    h1, c1 = lagrangian_cd(best_h.copy(), n, p1, seed=seed)
    if c1 < best_c - 1e-14:
        best_c = c1
        best_h = h1.copy()
    # Phase 2: SLSQP polish (15% budget)
    p2 = t0 + 0.70 * (budget_s - 5)
    alphas = [5000, 50000, 500000, 5000000]
    h2 = best_h.copy()
    for a in alphas:
        if time.time() > p2 - 5:
            break
        h2 = slsqp_smooth(h2, n, a, p2 - 5, maxiter=1500)
        c2 = compute_c5(h2, n)
        if c2 < best_c - 1e-14:
            best_c = c2
            best_h = h2.copy()
    # Phase 3: Binary search (15% budget)
    p3 = t0 + 0.85 * (budget_s - 5)
    hb, cb = binary_search_opt(best_h.copy(), n, p3, seed=seed + 1)
    if cb < best_c - 1e-14:
        best_c = cb
        best_h = hb.copy()
    # Phase 4: Final coordinate descent on best (15% budget)
    if time.time() < dl - 2:
        h4, c4 = final_refine(best_h.copy(), n, dl - 1)
        if c4 < best_c - 1e-14:
            best_c = c4
            best_h = h4.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
