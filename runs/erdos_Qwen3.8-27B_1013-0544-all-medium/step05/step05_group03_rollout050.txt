    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

            break

            if not active.any():
                break
            h[active] += diff / active.sum()

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return float(np.max(c))

def project_onto_simplex_box(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Binary search for theta such that sum(clip(h - theta, 0, 1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    # Gradient of C_k w.r.t. h
    # C_k = sum_i h[i] * (1-h[i+lag_k]) where lag_k = k - (n-1)
    # dC_k/dh[i] = (1-h[i+lag_k]) - h[i-lag_k] (with out-of-range = 0)
    # grad = sum_k w[k] * dC_k/dh * dx
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        # Contribution to grad from C_k
        # dC_k/dh[i] = (1-h[i+lag]) * [valid] - h[i-lag] * [valid]
        # This is a shifted version of (1-h) minus a shifted version of h
        g_k = np.zeros(n)
        # First term: (1-h[i+lag]) for valid i+lag
        if lag >= 0:
            g_k[0:n-lag] += (1 - h[lag:n])
            # Second term: -h[i-lag] for valid i-lag
            if lag < n:
                g_k[lag:n] -= h[0:n-lag]
        else:
            lag2 = -lag
            g_k[lag2:n] += (1 - h[0:n-lag2])
            if lag2 < n:
                g_k[0:n-lag2] -= h[lag2:n]
        grad += w[k] * g_k
    return grad * dx

def projected_gradient(h, n, deadline, alpha=1000, max_iter=100000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = softmax_grad(h, n, alpha)
        # Backtracking line search
        c_old = softmax_obj(h, n, alpha)
        step = lr
        found = False
        for _ in range(30):
            h_new = project_onto_simplex_box(h - step * g, n)
            c_new = softmax_obj(h_new, n, alpha)
            if c_new < c_old:
                h = h_new
                found = True
                break
            step *= 0.5
        if not found:
            lr *= 0.8
            if lr < 1e-12:
                break
        else:
            lr = min(lr * 1.1, 10.0)
            c_actual = compute_c5(h, n)
            if c_actual < best_c:
                best_c = c_actual
                best_h = h.copy()
    return best_h, best_c
```
