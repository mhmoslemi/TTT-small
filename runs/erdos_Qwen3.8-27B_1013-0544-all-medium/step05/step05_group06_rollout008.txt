def gradient_descent(h, n, deadline, alpha=10000, lr=0.01, max_iters=5000):
    """Projected gradient descent on smooth max"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    momentum = np.zeros(n)
    beta = 0.9
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        # Smooth max gradient
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        # Gradient of smooth max w.r.t. h
        # g(d) = sum_i h[i]*(1-h[i+d])*dx
        # dg/dh[i] = dx * [(1-h[i+d]) - h[i-d]] (for valid indices)
        grad = np.zeros(n)
        dm = np.arange(2 * n - 1) - (n - 1)
        for i in range(n):
            ip = i + dm
            im = i - dm
            valid_p = (ip >= 0) & (ip < n)
            valid_m = (im >= 0) & (im < n)
            gp = (1.0 - h[ip[valid_p]]) * weights[valid_p]
            gm = h[im[valid_m]] * weights[valid_m]
            grad[i] = dx * (gp.sum() - gm.sum())
        # Update with momentum
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Backtrack
            lr *= 0.8
            h = h_new.copy()  # still accept for now, just reduce lr
        if lr < 1e-8:
            break
    return best_h, best_c

def binary_polish(h, n, deadline):

    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])

            break
        improved = False
        for z in zeros:

                h_bin[o] = 0.0

                    ones.remove(o)
                    ones.append(z)
                    zeros.remove(z)
                    zeros.append(o)
                    improved = True
                    break
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    n_shifts = 2 * n - 1
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        g_top = g[top_idx]
        
        sweep_improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                # Compute s[m] = effect of delta on g[m]
                # s[m] = (1-h[i+dm[m]])*valid - h[i-dm[m]]*valid - (1-h[j+dm[m]])*valid + h[j-dm[m]]*valid
                ip_i = i + dm
                im_i = i - dm
                ip_j = j + dm
                im_j = j - dm
                s = np.zeros(n_shifts)
                vp_i = (ip_i >= 0) & (ip_i < n)
                s[vp_i] += (1.0 - h[ip_i[vp_i]])
                vm_i = (im_i >= 0) & (im_i < n)
                s[vm_i] -= h[im_i[vm_i]]
                vp_j = (ip_j >= 0) & (ip_j < n)
                s[vp_j] -= (1.0 - h[ip_j[vp_j]])
                vm_j = (im_j >= 0) & (im_j < n)
                s[vm_j] += h[im_j[vm_j]]

                d_hi = min(1.0 - h[i], h[j])

                s_top = s[top_idx]
                diff_g = g_top[:, None] - g_top[None, :]
                diff_s = s_top[None, :] - s_top[:, None]

                mask = (np.abs(diff_s) > 1e-15) & (kinks > d_lo) & (kinks < d_hi)

                # Evaluate at candidates (top K only)
                evals = (g_top[None, :] + candidates[:, None] * s_top[None, :]).max(axis=1)
                best_local = np.argmin(evals)
                d_best = float(candidates[best_local])
                new_max = float(evals[best_local])
                
                if new_max < current_max - 1e-13:
                    h[i] += d_best
                    h[j] -= d_best
                    g_top += d_best * s_top

                    g += d_best * s

            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_h, best_c
