from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        v = n - lag
        if v > 0:
            grad[:v] += dx * (1.0 - h[lag:])
            grad[lag:] -= dx * h[:v]
    else:
        v = n + lag
        if v > 0:
            grad[:v] += dx * (1.0 - h[-lag:])
            grad[-lag:] -= dx * h[:v]
    return grad

def subgrad_optimize(h, n, deadline, max_iter=200000):
    h = project_feasible(h.copy(), n)
    best_c, _ = compute_c5(h, n)
    best_h = h.copy()
    c, cvec = compute_c5(h, n)
    
    # Phase 1: larger steps
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Decay
        frac = it / max_iter
        eta = 0.1 * (1 - frac) + 0.001
        
        k_idx = int(np.argmax(cvec))
        lag = k_idx - (n - 1)
        
        if lag == 0:
            grad = (1.0 - 2.0 * h) * (2.0 / n)
        else:
            grad = grad_lag(h, n, lag)
        
        h_new = project_feasible(h - eta * grad, n)
        c_new, cvec_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            c = c_new
            cvec = cvec_new
        elif c_new < c:
            h = h_new
            c = c_new
            cvec = cvec_new
        else:
            # Halve step
            h_new = project_feasible(h - eta * 0.25 * grad, n)
            c_try = compute_c5(h_new, n)[0]
            if c_try < c:
                h = h_new
                c = c_try
                cvec, _ = compute_c5(h, n)
            else:
                # No progress, skip
                pass

def binary_swap_search(b, n, deadline, max_passes=300):

    best_c, _ = compute_c5(b, n)
    ones = list(np.where(b == 1)[0])

def pair_refine(h, n, deadline, max_passes=500):
    """Refine with small pair moves, focusing on active lag"""
    h = h.copy()
    best_c, cvec = compute_c5(h, n)
    best_h = h.copy()
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes // len(deltas)):
            if time.time() > deadline:
                break
            improved = False
            k_idx = int(np.argmax(cvec))
            lag = k_idx - (n - 1)
            
            # Find worst pairs at the active lag
            if lag > 0:
                vals = h[:n-lag] * (1.0 - h[lag:])
            elif lag < 0:
                vals = h[-lag:] * (1.0 - h[:-lag])
            else:
                vals = h * (1.0 - h)
            
            # Sort by worst first
            order = np.argsort(vals)[::-1]
            count = min(50, len(order))  # Only check top 50 pairs per pass
            
            for idx in order[:count]:
                if time.time() > deadline:
                    break
                if lag > 0:
                    i, j = idx, idx + lag
                elif lag < 0:
                    i, j = idx - lag, idx
                else:
                    continue
                
                # Try moving mass from i to j
                hi = h[i] - d
                hj = h[j] + d
                if 0 <= hi and hj <= 1:
                    h[i], h[j] = hi, hj
                    c, cvec_new = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        cvec = cvec_new
                        improved = True
                    else:
                        h[i], h[j] = hi + d, hj - d
            
            if not improved:
                break

    best_c, _ = compute_c5(best_h, n)
    return best_h, best_c

def slsqp_refine(h, n, deadline):
    """Final SLSQP polish"""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        alpha = 50000
        if m > 500:
            return m
        s = alpha * (c - m)
        mx = s.max()
        return m + np.log(np.sum(np.exp(s - mx))) / alpha
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        k_idx = int(np.argmax(c))
        lag = k_idx - (n - 1)
        if lag == 0:
            return (1.0 - 2.0 * x) * dx
        return grad_lag(x, n, lag)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-18})
        h_res = project_feasible(res.x, n)
        c_res, _ = compute_c5(h_res, n)
        return h_res, c_res
    except:
        return h.copy(), compute_c5(h, n)[0]

def optimize_at_n(h_start, n, deadline, seed):

    h = project_feasible(h_start.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        v = n - lag
        if v > 0:
            grad[:v] += dx * (1.0 - h[lag:])
            grad[lag:] -= dx * h[:v]
    else:
        v = n + lag
        if v > 0:
            grad[:v] += dx * (1.0 - h[-lag:])
            grad[-lag:] -= dx * h[:v]
    return grad

def topk_smooth_grad(h, n, alpha, top_k):
    cvec = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    mx = cvec.max()
    lw = alpha * (cvec - mx)
    idx = np.argsort(-lw)[:top_k]
    w = np.exp(lw[idx])
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(idx)); w[0] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for k, wi in zip(idx, w):
        if wi > 1e-10:
            grad += wi * grad_lag(h, n, int(k) - (n - 1))
    return grad, cvec

def nesterov_optimize(h, n, deadline, max_iter=150000, alpha0=5000):
    h = project_feasible(h.copy(), n)
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)[0]
    eta = 0.08
    for t in range(max_iter):
        if time.time() > deadline:
            break
        frac = t / max_iter
        alpha = alpha0 * (1 + frac * 10)
        eta_t = eta / (1 + 0.01 * t)
        if eta_t < 1e-5:
            break
        grad, cvec = topk_smooth_grad(h, n, alpha, 12)
        y_new = h - eta_t * grad
        y_new = project_feasible(y_new, n)
        momentum = (t - 1) / (t + 2) if t > 0 else 0.0
        h_new = y_new + momentum * (y_new - y)
        h_new = project_feasible(h_new, n)
        y = h_new
        c_new = compute_c5(h_new, n)[0]
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def sgsqr_refine(h, n, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        a = 30000.0
        s = a * (c - m)
        mx = s.max()
        return m + np.log(np.sum(np.exp(s - mx))) / a
    def obj_grad(x):
        cvec = np.correlate(x, 1 - x, mode='full') * dx
        k_idx = int(np.argmax(cvec))
        lag = k_idx - (n - 1)
        if lag == 0:
            return (1.0 - 2.0 * x) * dx
        return grad_lag(x, n, lag)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 4000, 'ftol': 1e-18})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res, n)[0]
    except Exception:
        return h.copy(), compute_c5(h, n)[0]

def binary_round(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def binary_local_search(b, n, deadline, max_passes=200):
    b = b.copy()
    best_c = compute_c5(b, n)[0]
    ones = list(np.where(b == 1)[0])
    zeros = list(np.where(b == 0)[0])
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                b[i], b[j] = 0.0, 1.0
                c = compute_c5(b, n)[0]
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    ones = list(np.where(b == 1)[0])
                    zeros = list(np.where(b == 0)[0])
                    break
                else:
                    b[i], b[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return b, best_c

def focused_pair_refine(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c, cvec = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        passes_allowed = 300
        for _pass in range(passes_allowed):
            if time.time() > deadline:
                break
            improved = False
            k_idx = int(np.argmax(cvec))
            lag = k_idx - (n - 1)
            if lag > 0:
                vals = h[:n - lag] * (1.0 - h[lag:])
                order = np.argsort(vals)[::-1][:30]
                pairs = [(int(idx), int(idx) + lag) for idx in order]
            elif lag < 0:
                vals = h[-lag:] * (1.0 - h[:-lag])
                order = np.argsort(vals)[::-1][:30]
                pairs = [(int(idx) - lag, int(idx)) for idx in order]
            else:
                break
            for (i, j) in pairs:
                hi, hj = h[i] - d, h[j] + d
                if 0 <= hi and hj <= 1:
                    h[i], h[j] = hi, hj
                    c, cv = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        best_h = h.copy()
                        cvec = cv
                        improved = True
                    else:
                        h[i], h[j] = hi + d, hj - d
            if not improved:
                break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)[0]
    return best_h, best_c

def optimize_pipeline(h_start, n, deadline, seed):
    np.random.seed(seed)
    h = project_feasible(h_start.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)[0]

    t_sg = min(50, (deadline - time.time()) * 0.35)
    if t_sg > 10:
        h_sg, c_sg = nesterov_optimize(h, n, time.time() + t_sg)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    if time.time() < deadline - 25:
        h_s, c_s = sgsqr_refine(best_h, n, deadline - 25)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    if time.time() < deadline - 20:
        b = binary_round(best_h, n)
        b_opt, c_b = binary_local_search(b, n, deadline - 20)
        if c_b < best_c:
            h_cont = b_opt.astype(float)
            h_c, c_c = nesterov_optimize(h_cont, n, deadline - 10, max_iter=30000)
            if c_c < best_c:
                best_h, best_c = h_c, c_c
            elif c_b < best_c:
                best_h, best_c = b_opt, c_b

    if time.time() < deadline - 5:
        best_h, best_c = focused_pair_refine(best_h, n, deadline - 2)

    if time.time() < deadline - 3:
        h_f, c_f = sgsqr_refine(best_h, n, deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)[0]
    return best_h, best_c
