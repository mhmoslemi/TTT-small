def build_R(h, n):

        mm = (im >= 0) & (im < n)
        R[i0, mp] = 1.0 - h[ip[mp]]
        R[i0, mm] -= h[im[mm]]
    return R

def slsqp_opt(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        # Soft-max approximation
        sm = shifted.max()
        softmax = sm + np.log(np.exp(shifted - sm).sum()) / alpha
        return m + softmax
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h

    # Step size schedule
    alpha_init = 0.1
    alpha_floor = 1e-6
    
    # Project gradient to tangent space
    for t in range(n_iter):
        if time.time() > deadline:
            break
        
        # Find active set
        current_max = g.max()
        threshold = current_max - 1e-6
        active = np.where(g >= threshold)[0]

            active = np.where(g == current_max)[0]

        grad = R[:, active].mean(axis=1)  # shape (n,)
        
        # Project to tangent space of sum constraint

        if alpha < alpha_floor:
            alpha = alpha_floor

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)

            best_c = c_new
            best_h = h_new.copy()
        
        # Update R if h changed significantly
        if np.abs(h_new - h).sum() > 1e-10:
            # Only update rows that changed
            changed = np.where(np.abs(h_new - h) > 1e-10)[0]
            for i in changed:
                update_R_row(R, h_new, i, n, dm)

        g = g_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_R(h, n, dm):
    R = np.zeros((n, 2 * n - 1))
    for i0 in range(n):
        ip = i0 + dm
        mp = (ip >= 0) & (ip < n)
        im = i0 - dm
        mm = (im >= 0) & (im < n)
        R[i0, mp] = 1.0 - h[ip[mp]]
        R[i0, mm] -= h[im[mm]]
    return R

def update_R_row(R, h, i0, n, dm):
    ip = i0 + dm
    mp = (ip >= 0) & (ip < n)
    im = i0 - dm
    mm = (im >= 0) & (im < n)
    R[i0, :] = 0.0
    R[i0, mp] = 1.0 - h[ip[mp]]
    R[i0, mm] -= h[im[mm]]

def subgradient_step(h, n, R, dm, deadline, n_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    alpha = 0.05
    for t in range(n_iter):
        if time.time() > deadline:
            break
        current_max = g.max()
        threshold = current_max - 1e-5
        active = np.where(g >= threshold)[0]
        if len(active) == 0:
            active = np.where(g == current_max)[0]
        grad = R[:, active].mean(axis=1)
        grad -= grad.mean()
        alpha = max(0.01, 0.05 / (1.0 + 0.001 * t))
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        g_new = compute_g(h_new, n)
        c_new = float(g_new.max())
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        changed = np.where(np.abs(h_new - h) > 1e-8)[0]
        for i in changed:
            update_R_row(R, h_new, i, n, dm)
        h = h_new
        g = g_new
    return best_h, best_c

def pair_descent(h, n, deadline, max_sweeps=800):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    R = build_R(h, n, dm)
    g = compute_g(h, n)
    current_max = float(g.max())
    K = 16
    stall = 0
    n_active = min(50, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        top_idx = np.argsort(g)[::-1][:K]
        n_improved = 0
        movable = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(movable) == 0:
            break
        scores = np.sum(np.abs(R[movable, top_idx]), axis=1)
        order = np.argsort(scores)[::-1]
        mv = movable[order[:n_active]]
        for idx in range(len(mv)):
            i = mv[idx]
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for idx2 in range(len(mv)):
                if idx2 == idx:
                    continue
                j = mv[idx2]
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                s = R[i] - R[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                candidates = [(d_lo + d_hi) * 0.5]
                for a in range(K):
                    for b in range(a + 1, K):
                        m1, m2 = top_idx[a], top_idx[b]
                        ds = s[m1] - s[m2]
                        if abs(ds) > 1e-15:
                            dc = (g[m2] - g[m1]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_d = 0.0
                best_v = current_max + 1e-30
                for d in candidates:
                    v = float(np.max(g + d * s))
                    if v < best_v:
                        best_v = v
                        best_d = d
                if best_d != 0.0 and best_v < current_max - 1e-14:
                    h[i] += best_d
                    h[j] -= best_d
                    g += best_d * s
                    update_R_row(R, h, i, n, dm)
                    update_R_row(R, h, j, n, dm)
                    current_max = float(g.max())
                    n_improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            g = compute_g(h, n)
            current_max = float(g.max())
            R = build_R(h, n, dm)
        if n_improved == 0:
            stall += 1
            if stall >= 8:
                break
        else:
            stall = 0
    return best_h, best_c

def slsqp_opt(h, n, alpha, deadline, maxiter=150):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        softmax = sm + np.log(np.exp(shifted - sm).sum()) / alpha
        return m + softmax
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_1swap(h, n, deadline):
    target = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    improved = True
    iters = 0
    while improved and time.time() < deadline and iters < 50:
        improved = False
        iters += 1
        best_gain = 1e-30
        best_pair = None
        for z in zeros:
            for o in ones:
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c_new = compute_c5(h_bin, n)
                if c_new < best_c - best_gain:
                    best_gain = best_c - c_new
                    best_pair = (z, o)
                h_bin[z] = 0.0
                h_bin[o] = 1.0
        if best_pair is not None:
            z, o = best_pair
            h_bin[z] = 1.0
            h_bin[o] = 0.0
            ones.remove(o)
            ones.append(z)
            zeros.remove(z)
            zeros.append(o)
            best_c = compute_c5(h_bin, n)
            improved = True
    return h_bin, best_c
