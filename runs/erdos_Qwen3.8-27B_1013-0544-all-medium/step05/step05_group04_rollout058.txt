    dx = 2.0 / n

            g[j] += -dx * h[i]       # derivative of h[i]*(1-h[j]) w.r.t. h[j]

    w /= w.sum()
    
    g = np.zeros(n)
    # For each i, g[i] = dx * [sum_{valid k} w[k]*(1-h[i+k-n+1]) - sum_{valid k} w[k]*h[i-k+n-1]]
    
    # Vectorize: for the first term, we need for each i:
    # sum over k where 0 <= i+k-n+1 < n of w[k] * (1-h[i+k-n+1])
    # Let j = i+k-n+1, so k = i-j+n-1
    # sum over j where 0<=j<n of w[i-j+n-1] * (1-h[j]) (with bounds on k: 0<=i-j+n-1<=2n-2)
    
    # This is a correlation of w with (1-h), with appropriate windowing
    # First term = correlate(1-h, w, mode='full')[n-1 : n-1+n]  (approximately)
    # Actually, let me just use np.correlate
    
    first = np.correlate(1-h, w, mode='full')  # first[n-1+i] = sum_j (1-h[j]) * w[j+i-n+1]
    # Hmm, this isn't quite right because of the bounds
    
    # Let me just do it with a loop over i but vectorize the inner computation
    for i in range(n):
        # First term: k ranges from max(0, n-1-i) to min(2n-2, 2n-2-i)
        lo = max(0, n - 1 - i)
        hi = min(2 * n - 1, 2 * n - 1 - i)
        if lo < hi:
            j_vals = np.arange(lo, hi) - n + 1 + i  # = i + k - n + 1 for k in [lo, hi)
            g[i] += np.dot(w[lo:hi], 1.0 - h[j_vals])
        
        # Second term: k ranges from max(0, i-n+1) to min(2n-2, i)
        lo2 = max(0, i - n + 1)
        hi2 = min(2 * n - 1, i + 1)
        if lo2 < hi2:
            j_vals2 = i - np.arange(lo2, hi2) + n - 1  # = i - k + n - 1 for k in [lo2, hi2)
            g[i] -= np.dot(w[lo2:hi2], h[j_vals2])
    
    g *= dx
    return g
```

This is O(n²) but with numpy dot products, so it's fast. For n=250, each iteration takes about 250 dot products of length up to 250, which is about 62,500 multiply-adds. Very fast.

OK let me now write the complete solution. I'll combine:
1. Projected subgradient with smooth max (main workhorse)
2. SLSQP polish (for precision)
3. Pair search (for final precision)
4. Multiple restarts

Let me also add a key insight: **the Nesterov averaging trick**. Instead of just tracking the best point, I'll maintain a running average of the iterates, which can give better results for convex problems (and sometimes for non-convex ones too).

Here's my final code:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w /= w.sum()
    val = m + (max_s + np.log((w * w.sum()).sum())) / alpha  # Hmm this is wrong
    # Actually val = m + (max_s + log(sum exp(shifted - max_s))) / alpha
    # But w = exp(shifted - max_s) / sum(exp(shifted - max_s))
    # So sum(exp(shifted - max_s)) = 1/w_max... no
    # Let me just compute val separately
    exps = np.exp(shifted - max_s)
    logsum = max_s + np.log(exps.sum())
    val = m + logsum / alpha
    w = exps / exps.sum()
    
    g = np.zeros(n)
    for i in range(n):
        lo = max(0, n - 1 - i)
        hi = min(2 * n - 1, 2 * n - 1 - i)
        if lo < hi:
            jv = np.arange(lo, hi) - n + 1 + i
            g[i] += np.dot(w[lo:hi], 1.0 - h[jv])
        lo2 = max(0, i - n + 1)
        hi2 = min(2 * n - 1, i + 1)
        if lo2 < hi2:
            jv2 = i - np.arange(lo2, hi2) + n - 1
            g[i] -= np.dot(w[lo2:hi2], h[jv2])
    g *= dx
    return val, g

def subgrad_method(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Subgradient with annealing
    n_steps = 3000
    alphas = np.geomspace(50, 50000, n_steps)
    
    for t in range(n_steps):
        if time.time() > deadline:
            break
        alpha = alphas[t]
        val, g = smooth_max_grad(h, n, alpha)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break
        # Polyak step
        step = min(val / g_norm_sq, 0.005)
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        ms = sh.max()
        if ms > 500:
            return m + (ms + np.log(np.sum(np.exp(sh - ms)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    def sum_c(x):
        return x.sum() - n/2.0
    def sum_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_c, 'jac': sum_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
