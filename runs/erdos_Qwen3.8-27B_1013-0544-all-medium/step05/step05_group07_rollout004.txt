import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lbfgs_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        sm = s.max()
        val = m + (sm + np.log(np.exp(s - sm).sum())) / alpha
        val += 1e5 * (x.sum() - target) ** 2
        return val
    def grad(x):
        c_raw = np.correlate(x, 1.0 - x, mode='full')
        idx = int(np.argmax(c_raw))
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2.0e5 * (x.sum() - target)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_sa(h_init, n, deadline, rng, n_max_iters=300000):
    target = n // 2
    sorted_idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max() * dx)
    best_h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    T = 0.003
    T_min = 1e-9
    cooling = 0.99985
    it = 0
    while time.time() < deadline and it < n_max_iters:
        it += 1
        if not ones or not zeros:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i] = 1.0
        h[j] = 0.0
        c_new = np.correlate(h, 1.0 - h, mode='full')
        c_max_new = float(c_new.max() * dx)
        if c_max_new < best_c:
            best_c = c_max_new
            best_h = h.copy()
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
        else:
            delta = c_max_new - best_c
            if rng.random() > np.exp(-delta / T):
                h[i] = 0.0
                h[j] = 1.0
            else:
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                best_h = h.copy()
                best_c = c_max_new
        T = max(T * cooling, T_min)
    return best_h, best_c

def coord_descent(h, n, deadline, rng, n_passes=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    it = 0
    while time.time() < deadline and it < n_passes:
        it += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = rng.choice([1.0, -1.0])
        d = rng.choice([0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
        hi = h[i] + s * d
        hj = h[j] - s * d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
            else:
                h[i], h[j] = h[i] - s * d, h[j] + s * d
    return best_h, best_c

def make_block_pattern(n, ratio):
    h = np.zeros(n)
    block = int(n * ratio)
    for i in range(n):
        pos = i % (2 * block)
        if pos < block:
            h[i] = 1.0
    return project_feasible(h, n)

def make_sine_pattern(n, freq, rng):
    x = np.arange(n) / n
    h = np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    h = (h > 0).astype(float)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    t_phase1_end = t0 + 0.15 * (budget_s - 5)
    t_phase2_end = t0 + 0.45 * (budget_s - 5)
    t_phase3_end = t0 + 0.75 * (budget_s - 5)

    # Phase 1: L-BFGS smoothing from initial
    for alpha in [200, 1000, 5000, 20000]:
        if time.time() > t_phase1_end - 2:
            break
        h_new = lbfgs_smooth(best_h.copy(), n, alpha, 200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 2: Binary SA from multiple diverse starts
    starts = [best_h.copy()]
    idx_rand = rng.permutation(n)
    hb = np.zeros(n)
    hb[idx_rand[:n // 2]] = 1.0
    starts.append(hb)
    starts.append(make_block_pattern(n, 0.3))
    starts.append(make_block_pattern(n, 0.4))
    starts.append(make_sine_pattern(n, 3, rng))
    starts.append(make_sine_pattern(n, 5, rng))
    starts.append(make_sine_pattern(n, 7, rng))

    time_per_sa = max(5, (t_phase2_end - t_phase1_end) / len(starts))
    for h_start in starts:
        sa_deadline = min(time.time() + time_per_sa, t_phase2_end - 1)
        if time.time() > sa_deadline:
            break
        h_bin, c_bin = binary_sa(h_start, n, sa_deadline, rng)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin.copy()

    # Phase 3: Continuous refinement from best (and other good) solutions
    # Use L-BFGS with increasing alpha
    h_work = best_h.copy()
    for alpha in [5000, 50000, 200000, 1000000, 5000000]:
        if time.time() > t_phase3_end - 5:
            break
        h_new = lbfgs_smooth(h_work.copy(), n, alpha, 300)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_work = h_new.copy()
        else:
            h_work = h_new.copy()

    # Also try from the best binary solution specifically
    h_bin_best = np.zeros(n)
    h_bin_best[np.argsort(best_h)[::-1][:n // 2]] = 1.0
    for alpha in [100000, 1000000, 5000000]:
        if time.time() > t_phase3_end - 2:
            break
        h_new = lbfgs_smooth(h_bin_best.copy(), n, alpha, 200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 4: Coordinate descent polish
    h_new, c_new = coord_descent(best_h.copy(), n, deadline - 10, rng, n_passes=80000)
    if c_new < best_c:
        best_c, best_h = c_new, h_new.copy()

    # Final L-BFGS polish
    if time.time() < deadline - 15:
        h_new = lbfgs_smooth(best_h.copy(), n, 10000000, 500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # One more coordinate descent
    h_new, c_new = coord_descent(best_h.copy(), n, deadline - 2, rng, n_passes=30000)
    if c_new < best_c:
        best_c, best_h = c_new, h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
