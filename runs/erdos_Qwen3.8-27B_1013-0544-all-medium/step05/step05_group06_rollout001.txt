    sm = shifted.max()
    weights = np.exp(shifted - sm)

    # g[m] = sum_i h[i+d_m]*(1-h[i]) where d_m = m-(n-1)
    # dg[m]/dh[i] = (1-h[i-d_m])*1{valid} - h[i+d_m]*1{valid}

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]

        i_hi = min(n, n + d)
        if i_lo < i_hi:

        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F

def update_F(F, i, j, delta, dm):
    n = F.shape[0]
    v1 = (i + dm >= 0) & (i + dm < n)
    cols1 = np.where(v1)[0]
    F[i + dm[v1], cols1] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    cols2 = np.where(v2)[0]
    F[i - dm[v2], cols2] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    cols3 = np.where(v3)[0]
    F[j + dm[v3], cols3] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    cols4 = np.where(v4)[0]
    F[j - dm[v4], cols4] += delta

def slsqp_optimize(h, n, alpha, K, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        top = np.partition(g, -K)[-K:]
        m = top.max()
        shifted = alpha * (top - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        top_idx = np.argsort(g)[::-1][:K]
        top = g[top_idx]
        m = top.max()
        shifted = alpha * (top - m)
        sm = shifted.max()
        w = np.exp(shifted - sm) / np.exp(shifted - sm).sum()
        F = compute_F(x, n)
        grad = (F[:, top_idx] * w).sum(axis=1)
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() < deadline:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def pair_descent(h, n, deadline, max_sweeps=800, K_init=10, K_max=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    K = K_init
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = compute_F(h, n)
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi]
                for a in range(K):
                    for b in range(K):
                        if a == b:
                            continue
                        ds = s[top_idx[b]] - s[top_idx[a]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[a]] - g[top_idx[b]]) / ds
                            if d_lo + 1e-15 < dc < d_hi - 1e-15:
                                candidates.append(dc)
                best_delta = 0.0

                    h[j] -= best_delta
                    g += best_delta * s
                    update_F(F, i, j, best_delta, dm)
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            if improved < 5 and K < K_max:

        else:

    return best_h, best_c

def binary_polish(h, n, deadline, max_rounds=40):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1.0)[0].tolist())
    zeros = set(np.where(h_bin == 0.0)[0].tolist())
    one_list = sorted(ones)
    zero_list = sorted(zeros)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for z in zero_list:
            for o in one_list:
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(o)
                    ones.add(z)
                    zeros.discard(z)
                    zeros.add(o)
                    one_list.remove(o)
                    one_list.append(z)
                    one_list.sort()
                    zero_list.remove(z)
                    zero_list.append(o)
                    zero_list.sort()
                    improved = True
                    break
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0

                break
        if not improved:
