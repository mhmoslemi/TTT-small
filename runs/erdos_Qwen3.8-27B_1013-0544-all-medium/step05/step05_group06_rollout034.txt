def project_feasible(h, n):

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    lo, hi = -2.0, 2.0

def gradient_descent(h, n, deadline, alpha_start=500, alpha_end=1000000, n_iters_per_alpha=50):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = np.geomspace(alpha_start, alpha_end, 20)
    for alpha in alphas:

        beta = 0.9

                best_h = h_new.copy()
            h = h_new
            # Adaptive lr

                lr *= 0.8
            elif c < best_c * (1 - 1e-5):
                lr = min(lr * 1.1, 0.1)
    return best_h, best_c

    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 0.05
        momentum = np.zeros(n)
        beta = 0.9
        for it in range(500):
            if time.time() > deadline:
                break
            f, grad = smooth_grad(h, n, alpha)
            momentum = beta * momentum + (1.0 - beta) * grad
            h_new = h - lr * momentum
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c:

                best_h = h_new.copy()

                lr *= 0.7
