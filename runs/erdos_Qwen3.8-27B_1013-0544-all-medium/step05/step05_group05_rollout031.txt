def binary_sa(h, n, deadline, t_start, t_end, max_iters, seed):
    # h is binary, exactly n//2 ones
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    rng = np.random.RandomState(seed)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end/t_start) ** (it / max_iters)
        # Pick random i from zeros, j from ones
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Try swap
        h[i], h[j] = 1.0, 0.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c = new_c
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
    
    return best_h, best_c

            break

        # Variables: h[0..n-1], t (total n+1 variables)
        # Minimize t: c = [0,...,0, 1]

        #   => sum_i w_lag[i] * h[i] - t <= 0
        # Plus: h[i] <= 1, -h[i] <= 0, sum h[i] = n/2

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def alternating_lp(h0, n, deadline, max_iters=200):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline - 10:
            break
        # Build LP: minimize t
        # s.t. (2/n)*sum_i h_new[i]*(1-h_old[i-lag]) <= t for all lags
        #      0 <= h_new[i] <= 1
        #      sum h_new[i] = n/2
        # Variables: [h_new[0], ..., h_new[n-1], t]
        n_vars = n + 1
        n_lags = 2 * n - 1
        dx = 2.0 / n
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each lag: dx * sum_i coeff[i] * h_new[i] - t <= 0
        # Plus: h_new[i] <= 1 (n constraints)
        n_ineq = n_lags + n
        A_ub = np.zeros((n_ineq, n_vars))
        b_ub = np.zeros(n_ineq)
        
        for m in range(n_lags):
            lag = m - (n - 1)
            for i in range(n):
                j = i - lag
                if 0 <= j < n:
                    A_ub[m, i] = dx * (1.0 - h[j])
            A_ub[m, n] = -1.0
            b_ub[m] = 0.0
        
        for i in range(n):
            A_ub[n_lags + i, i] = 1.0
            b_ub[n_lags + i] = 1.0
        
        A_eq = np.ones((1, n_vars))
        A_eq[0, n] = 0.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:n]
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        else:
            break
    
    return best_h, best_c

def binary_local_search(h0, n, deadline, max_iters=500000, seed=42):

    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

            break
        i = zeros[rng.randint(len(zeros))]

            ones.remove(j)

            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0

    return best_h, best_c

from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h0, n):

    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

            break

        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += dx * (1.0 - best_h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
            if n - to > 0:
                grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:

        grad /= ng
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:

        else:
