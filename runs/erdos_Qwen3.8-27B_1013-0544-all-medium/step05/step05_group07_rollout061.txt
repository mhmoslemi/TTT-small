```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smax_and_grad(h, alpha, lam=1e6):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    cdx = c * dx
    m = cdx.max()
    exp_args = alpha * (cdx - m)
    exp_vals = np.exp(exp_args)
    zp = exp_vals.sum()
    val = m + np.log(zp) / alpha
    val += lam * (h.sum() - n/2.0)**2
    p = exp_vals / zp
    
    grad = np.zeros(n)
    for k_idx in range(len(cdx)):
        if p[k_idx] < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k]) * p[k_idx]
            grad[:n-k] -= h[k:] * p[k_idx]
        else:
            grad[:n+k] += (1.0 - h[-k:]) * p[k_idx]
            grad[-k:] -= h[:n+k] * p[k_idx]
    grad *= dx
    grad += 2*lam*(h.sum() - n/2.0)
    return val, grad

def lbfgs_phase(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        def f(x):
            v, _ = smax_and_grad(np.clip(x, 0, 1), alpha)
            return v
        def g(x):
            _, gr = smax_and_grad(np.clip(x, 0, 1), alpha)
            return gr
        try:
            res = minimize(f, best_h, jac=g, method='L-BFGS-B',
                          bounds=[(0,1)]*n,
                          options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-8, 'maxcor': 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def nesterov_subgrad(h, n, deadline, steps=500000):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    x = h.copy()
    x_prev = h.copy()
    k = 0
    for k in range(1, steps+1):
        if time.time() > deadline:
            break
        beta = (1 - 2.0/(k+2.0)) / (1 + 2.0/(k+2.0))
        y = x + beta * (x - x_prev)
        y = np.clip(y, 0, 1)
        # Compute subgradient at y
        n_y = len(y)
        dx = 2.0 / n_y
        c = np.correlate(y, 1-y, mode='full')
        cdx = c * dx
        k_star = int(np.argmax(cdx))
        ks = k_star - (n_y - 1)
        grad = np.zeros(n_y)
        if ks >= 0:
            grad[ks:] += (1.0 - y[:n_y-ks])
            grad[:n_y-ks] -= y[ks:]
        else:
            grad[:n_y+ks] += (1.0 - y[-ks:])
            grad[-ks:] -= y[:n_y+ks]
        grad *= dx
        # Step
        lr = 0.5 / np.sqrt(k)
        x_new = project_feasible(x - lr * grad, n)
        # Check improvement
        c_check = compute_c5(x_new)
        if c_check < best_c:
            best_c = c_check
            best_h = x_new.copy()
        x_prev = x
        x = x_new
    return best_h, best_c

def pair_descent(h, n, deadline, max_rounds=5):
    """Pairwise coordinate descent on the smoothed objective."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    alphas = [10000, 50000, 200000]
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        for round_num in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            # Subsample pairs for efficiency

            n_pairs = min(5000, n*(n-1)//2)
            for _ in range(n_pairs):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                # Try to improve by adjusting h[i] and h[j]
                # Keep sum constant: h[i] + d, h[j] - d
                for d in [0.01, -0.01, 0.005, -0.005]:
                    hi = best_h[i] + d

                            improved = True
                        else:
                            best_h[i], best_h[j] = hi - d, hj + d
            if not improved:

    return best_h, best_c
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    cdx = c * dx
    m = cdx.max()
    ea = alpha * (cdx - m)
    ev = np.exp(ea)
    zp = ev.sum()
    val = m + np.log(zp) / alpha
    val += 1e6 * (h.sum() - n / 2.0) ** 2
    p = ev / zp
    grad = np.zeros(n)
    for ki in range(len(cdx)):
        if p[ki] < 1e-10:
            continue
        k = ki - (n - 1)
        w = p[ki]
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            grad[:n + k] += w * (1.0 - h[-k:])
            grad[-k:] -= w * h[:n + k]
    grad *= dx
    grad += 2e6 * (h.sum() - n / 2.0)
    return val, grad

    best_c = compute_c5(best_h)
    x = h0.copy()
    x_prev = h0.copy()

        if time.time() > deadline:
            break
        lam = 2.0 / (step + 2.0)
        beta = (1 - lam) / (1 + lam)
        y = np.clip(x + beta * (x - x_prev), 0, 1)

        ks = int(np.argmax(c)) - (n - 1)

        h_rand = np.zeros(n)
        perm = rng.permutation(n)
        h_rand[perm[:n // 2]] = 1.0
        hb2, cb2 = binary_hill_climb(h_rand, n, min(p3, time.time() + 30), rng, max_iter=30000)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()

    # Phase 4: More L-BFGS from best (continuous) with high alpha (10%)
    p4 = t0 + 0.95 * (budget_s - 5)
    h_pert = project_feasible(best_h + 0.02 * rng.randn(n), n)
    h4, c4 = lbfgs_phase(h_pert, n, p4, [500000, 2000000, 10000000], maxiter_per=200)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: Pair descent polish (5%)
    h5, c5v = pair_descent(best_h, n, deadline, rng, n_rounds=6)
    if c5v < best_c:
        best_c, best_h = c5v, h5.copy()

    # Final: one more L-BFGS
    if time.time() < deadline - 2:
        h6, c6 = lbfgs_phase(best_h, n, deadline, [50000000, 100000000], maxiter_per=100)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
```
