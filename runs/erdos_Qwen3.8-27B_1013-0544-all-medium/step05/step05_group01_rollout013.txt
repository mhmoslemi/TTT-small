import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lses_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    w /= w_sum
    obj = (mx + np.log(w_sum)) / alpha
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk < 1e-10:
            continue
        if k >= 0:
            nk = n - k
            grad[:nk] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:nk]
        else:
            nk = -k
            grad[nk:] += wk * (1.0 - h[:n-nk])
            grad[:n-nk] -= wk * h[nk:]
    grad *= dx / alpha
    return obj, grad

def penalty_optimize(h, n, alpha, deadline, lam=100.0):
    """L-BFGS-B optimization with penalty for sum constraint"""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        l, g = lses_and_grad(x, n, alpha)
        penalty = lam * (np.sum(x) - n/2.0)**2
        pgrad = g + 2*lam*(np.sum(x) - n/2.0) * np.ones(n)
        return l + penalty, pgrad
    
    try:
        res = minimize(obj, h, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 5000, 'gtol': 1e-10, 'maxfun': 5000})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fine_pair_search(h, n, deadline, deltas=None, max_passes=500):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:

            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):

                    hj = h[j] - d

                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = h[i] - d, h[j] + d
            if not improved:
                break
    return h, best_c

def gradient_descent(h, n, alpha, deadline, lr=0.01, maxiter=10000):
    """Projected gradient descent with backtracking line search"""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        obj, grad = lses_and_grad(h, n, alpha)
        # Add sum constraint gradient
        s = np.sum(h) - n/2.0
        grad = grad + 2*100.0*s*np.ones(n)
        
        # Project gradient to feasible direction (maintain sum)
        # Simple approach: just use the gradient, project after step
        for _ls in range(20):
            step = lr * np.linalg.norm(grad) / (np.linalg.norm(grad)**2 + 1e-10)
            step = min(step, 0.01)
            new_h = project_feasible(h - step * grad, n)
            c_new = compute_c5(new_h, n)
            if c_new < best_c or c_new < compute_c5(h, n):
                h = new_h
                break
            lr *= 0.5
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def bottleneck_local_search(h, n, deadline, max_passes=10000):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1-h, mode='full') * dx
        k_max = np.argmax(c)  # index in [0, 2n-2]
        k = k_max - (n - 1)  # actual lag
        
        # Compute score for each position
        score = np.zeros(n)
        if k >= 0:
            # h[i+k] valid for i in [0, n-k)
            score[:n-k] += h[k:]
            # h[i-k] valid for i in [k, n)
            score[k:] += h[:n-k]
        else:
            kk = -k
            # h[i+kk] valid for i in [0, n-kk)
            score[:n-kk] += h[kk:]
            # h[i-kk] valid for i in [kk, n)
            score[kk:] += h[:n-kk]
        
        # Find best i (max score) and j (min score)
        # With constraint h[i]+d <= 1 and h[j]-d >= 0
        i_best = np.argmax(score)
        j_best = np.argmin(score)
        
        if i_best == j_best:
            break

        improved = False
        for d in [0.01, 0.005, 0.002, 0.001, 0.0005]:

                continue

            h[j_best] -= d

            if c_new < best_c - 1e-15:
                best_c = c_new
                improved = True
                break
            else:
                h[i_best] -= d
                h[j_best] += d
        
        if not improved:
            # Try second-best pair
            # Or try swapping the direction
            for d in [0.005, 0.002, 0.001]:
                if h[j_best] + d > 1.0 or h[i_best] - d < 0.0:
                    continue
                h[i_best] -= d
                h[j_best] += d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                    break
                else:
                    h[i_best] += d
                    h[j_best] -= d
        
        if not improved:
            # Try all pairs with top/bottom scores
            sorted_idx = np.argsort(score)
            found = False
            for idx_i in range(n-1, max(n-20, -1), -1):
                for idx_j in range(min(20, n)):

                            found = True
                            break
                        h[i] -= d
                        h[j] += d
                if found:
                    break
            if not found:
                break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lses_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    w /= w_sum
    obj = (mx + np.log(w_sum)) / alpha
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk < 1e-10:
            continue
        if k >= 0:
            nk = n - k
            grad[:nk] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:nk]
        else:
            nk = -k
            grad[nk:] += wk * (1.0 - h[:n - nk])
            grad[:n - nk] -= wk * h[nk:]
    grad *= dx / alpha
    return obj, grad

def lbfgsb_optimize(h, n, alpha, deadline, lam=500.0):
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        l, g = lses_grad(x, n, alpha)
        s = np.sum(x) - n / 2.0
        penalty = lam * s * s
        pgrad = g + 2.0 * lam * s * np.ones(n)
        return l + penalty, pgrad

    try:
        res = minimize(obj_and_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 8000, 'gtol': 1e-12, 'maxfun': 8000})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def full_pair_search(h, n, deadline, deltas, max_passes=300):

    for d in deltas:
        if time.time() > deadline:
            break

            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):

                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj

                            improved = True
                        else:

                break
    return h, best_c
