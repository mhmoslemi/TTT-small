import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            grad_neg = -k
            g[:n - grad_neg] += (1.0 - x[grad_neg:])
            g[grad_neg:] += -x[:n - grad_neg]
        g *= dx
        g += 2e5 * (x.sum() - n / 2.0)
        return g
    res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
    return project_feasible(res.x, n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - n / 2.0) ** 2
        return val

        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            s = -k
            g[:n - s] += (1.0 - x[s:])
            g[s:] += -x[:n - s]
        g *= dx
        g += 2e5 * (x.sum() - n / 2.0)
        return g
    res = minimize(obj, h, method='L-BFGS-B', jac=grad,

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - n / 2.0) ** 2
        return float(val)
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            s = -k
            g[:n - s] += (1.0 - x[s:])
            g[s:] += -x[:n - s]
        g *= dx
        g += 2e5 * (x.sum() - n / 2.0)
        return g
    res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project_feasible(res.x, n)

def binary_hill_climb(h_init, n, deadline):
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:target]] = 1.0
    best_c, _ = compute_c5(h, n)
    while time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 1e-15
        best_move = None
        for i in zeros:
            for j in ones:
                h[i], h[j] = 1.0, 0.0
                c, _ = compute_c5(h, n)
                h[i], h[j] = 0.0, 1.0
                d = best_c - c
                if d > best_delta:
                    best_delta = d
                    best_move = (i, j)
        if best_move:
            i, j = best_move
            h[i], h[j] = 1.0, 0.0
            best_c -= best_delta
            improved = True
        if not improved:
            break
    return h, best_c

def sa_binary(h_init, n, deadline, rng, t_start=0.008, t_min=0.0004):
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:target]] = 1.0
    best_c, _ = compute_c5(h, n)
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    temp = t_start
    stall = 0
    while time.time() < deadline:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        c_new, _ = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            stall = 0
        elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-12)):
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            stall += 1
        else:
            h[i], h[j] = 0.0, 1.0
            stall += 1
        temp = max(temp * 0.9997, t_min)
        if stall > 15000:
            h = best_h.copy()
            for _ in range(max(1, n // 6)):
                o = list(np.where(h == 1)[0])
                z = list(np.where(h == 0)[0])
                if not o or not z:
                    break
                a = z[rng.randint(len(z))]
                b = o[rng.randint(len(o))]
                h[a], h[b] = 1.0, 0.0
            best_c, _ = compute_c5(h, n)
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            stall = 0
            temp = t_start * 0.4
    return best_h, best_c

def continuous_refine(h, n, deadline, rng):
    best_c, _ = compute_c5(h, n)
    best_h = h.copy()
    scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for scale in scales:
        if time.time() > deadline:
            break
        for _ in range(3000):
            if time.time() > deadline:
                break
            i, j = rng.randint(n, 2)
            d = rng.choice([-1.0, 1.0]) * scale
            hi, hj = h[i] + d, h[j] - d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h[i], h[j] = hi, hj
                c, _ = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = hi - d, hj + d
    return best_h, best_c

def informed_binary(h_init, n, deadline, rng):
    """SA with informed moves targeting worst shift."""
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:target]] = 1.0
    best_c, _ = compute_c5(h, n)
    best_h = h.copy()
    temp = 0.006
    stall = 0
    while time.time() < deadline:
        _, c_vals = compute_c5(h, n)
        k_star = int(np.argmax(c_vals))
        shift = k_star - (n - 1)
        if shift > 0:
            bad = [i for i in range(n - shift) if h[i] == 1 and h[i + shift] == 0]
        elif shift < 0:
            s = -shift
            bad = [i for i in range(s, n) if h[i] == 1 and h[i - s] == 0]
        else:
            bad = []
        if bad and rng.random() < 0.6:
            i = bad[rng.randint(len(bad))]
            j = i + shift
            if 0 <= j < n:
                h[i], h[j] = 0.0, 1.0
                c_new, _ = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    stall = 0
                    temp = max(temp * 0.999, 0.0003)
                elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-12)):
                    stall += 1
                else:
                    h[i], h[j] = 1.0, 0.0
                    stall += 1
                temp = max(temp * 0.9997, 0.0003)
        else:
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            c_new, _ = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                stall = 0
                temp = max(temp * 0.999, 0.0003)
            elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-12)):
                stall += 1
            else:
                h[i], h[j] = 0.0, 1.0
                stall += 1
            temp = max(temp * 0.9997, 0.0003)
        if stall > 20000:
            h = best_h.copy()
            for _ in range(n // 5):
                o = list(np.where(h == 1)[0])
                z = list(np.where(h == 0)[0])
                if not o or not z:
                    break
                a = z[rng.randint(len(z))]
                b = o[rng.randint(len(o))]
                h[a], h[b] = 1.0, 0.0
            best_c, _ = compute_c5(h, n)
            stall = 0
            temp = 0.005
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)[0]

    # Phase 1: Quick L-BFGS-B (8%)
    p1 = t0 + 0.08 * (budget_s - 3)
    for alpha in [5000, 30000, 100000]:
        if time.time() > p1 - 2:
            break
        h_new = lbfgs_smooth(best_h.copy(), n, alpha, 600)
        c_new = compute_c5(h_new, n)[0]
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 2: Binary informed SA + hill climb (42%)
    p2 = t0 + 0.50 * (budget_s - 3)
    time_seg = (p2 - t0) / 4.0
    # 2a: Informed binary SA
    p2a = t0 + time_seg
    hb, cb = informed_binary(best_h.copy(), n, p2a, rng)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # 2b: Hill climb
    p2b = t0 + time_seg * 2
    hc, cc = binary_hill_climb(best_h.copy(), n, p2b)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()
    # 2c: SA from hill-climbed
    p2c = t0 + time_seg * 3
    hb2, cb2 = sa_binary(best_h.copy(), n, p2c, rng)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()
    # 2d: Final hill climb
    p2d = p2
    hc2, cc2 = binary_hill_climb(best_h.copy(), n, p2d)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()

    # Phase 3: Continuous refinement (22%)
    p3 = t0 + 0.72 * (budget_s - 3)
    h_new, c_new = continuous_refine(best_h.copy(), n, p3, rng)
    if c_new < best_c:
        best_c, best_h = c_new, h_new.copy()

    # Phase 4: L-BFGS-B with increasing alpha (28%)
    alphas = [50000, 100000, 300000, 1000000, 3000000, 10000000]
    time_per_alpha = (deadline - t0) / (len(alphas) + 2)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        h_new = lbfgs_smooth(best_h.copy(), n, alpha, 400)
        c_new = compute_c5(h_new, n)[0]
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Final polish: micro continuous refine
    if time.time() < deadline - 5:
        h_new, c_new = continuous_refine(best_h.copy(), n, deadline, rng)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)[0]
    return best_h, best_c, n
