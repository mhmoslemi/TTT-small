import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_search(hb, n, deadline, max_sweeps=300):
    best_c = compute_c5(hb, n)
    ones = set(np.where(hb == 1)[0].tolist())
    zeros = set(np.where(hb == 0)[0].tolist())
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zlist = list(zeros)
        olist = list(ones)
        for i in zlist:
            if time.time() > deadline:
                return hb, best_c
            for j in olist:
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    break
                else:
                    hb[i], hb[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return hb, best_c

def greedy_2swap(hb, n, deadline, max_iters=20000):
    best_c = compute_c5(hb, n)
    rng = np.random.RandomState(42)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        i1, i2 = rng.choice(zeros, 2, replace=False)
        j1, j2 = rng.choice(ones, 2, replace=False)
        hb[i1], hb[i2] = 1.0, 1.0
        hb[j1], hb[j2] = 0.0, 0.0
        c = compute_c5(hb, n)
        if c < best_c - 1e-15:
            best_c = c
        else:
            hb[i1], hb[i2] = 0.0, 0.0
            hb[j1], hb[j2] = 1.0, 1.0
    return hb, best_c

def simulated_annealing(hb, n, deadline, t_init=0.008, t_final=0.00005):
    cur_hb = hb.copy()
    cur_c = compute_c5(cur_hb, n)
    best_hb = cur_hb.copy()
    best_c = cur_c
    ones = list(np.where(cur_hb == 1)[0])
    zeros = list(np.where(cur_hb == 0)[0])
    rng = np.random.RandomState(999)
    count = 0
    while time.time() < deadline:
        count += 1
        progress = min(1.0, count / 80000.0)
        temp = t_init * (t_final / t_init) ** progress
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        cur_hb[i], cur_hb[j] = 1.0, 0.0
        new_c = compute_c5(cur_hb, n)
        if new_c <= cur_c + 1e-16:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_hb = cur_hb.copy()
            ones[jj] = i
            zeros[ii] = j
        else:
            delta = new_c - cur_c
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_c = new_c
                ones[jj] = i
                zeros[ii] = j
            else:
                cur_hb[i], cur_hb[j] = 0.0, 1.0
    return best_hb, best_c

def ils(hb, n, deadline, max_rounds=200, perturb_k=3):
    best_hb = hb.copy()
    best_c = compute_c5(hb, n)
    cur_hb = hb.copy()
    rng = np.random.RandomState(555)

        if time.time() > deadline - 5:
            break
        local_dl = min(time.time() + 8, deadline - 2)
        cur_hb, cur_c = greedy_search(cur_hb.copy(), n, local_dl, max_sweeps=150)
        if cur_c < best_c:
            best_c = cur_c
            best_hb = cur_hb.copy()
        # Perturb

        zeros = list(np.where(cur_hb == 0)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)
        k = min(perturb_k, len(ones), len(zeros))
        for i in range(k):
            cur_hb[ones[i]], cur_hb[zeros[i]] = 0.0, 1.0
    return best_hb, best_c

def make_initial_binary(n, rng):
    target = n // 2
    best_hb = None
    best_c = float('inf')
    for p in range(2, n + 1):
        pattern = np.zeros(n)
        for i in range(n):
            if (i * p) % n < target:
                pattern[i] = 1.0
        s = int(np.sum(pattern))
        if s > target:
            pos = np.where(pattern == 1)[0]
            rng.shuffle(pos)
            pattern[pos[:s - target]] = 0
        elif s < target:
            pos = np.where(pattern == 0)[0]
            rng.shuffle(pos)
            pattern[pos[:target - s]] = 1
        c = compute_c5(pattern, n)
        if c < best_c:
            best_c = c
            best_hb = pattern.copy()
    return best_hb if best_hb is not None else np.zeros(n)
