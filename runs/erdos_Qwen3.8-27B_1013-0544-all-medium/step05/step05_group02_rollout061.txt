    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.04
    mu = 0.82
    eps = 0.006
    ni = 0
    for it in range(8000):
        if time.time() > deadline:
            break
        g, f = active_grad(h, n, eps)
        g_eff = g - g.mean()
        v = mu * v - lr * g_eff
        cand = project(h + v, n)
        cf = compute_c5(cand, n)
        if cf < best_c - 1e-14:
            best_c = cf
            best_h = cand.copy()
            ni = 0
        else:
            ni += 1
        h = cand
        if ni > 35:
            lr *= 0.7
            v *= 0.4
            if lr < 1e-7:
                break
        if it % 80 == 0:
            eps = min(0.05, eps * 1.04)
    return best_h, best_c

def full_opt(h0, n, deadline):
    h = project(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for a in [500, 2000, 8000, 40000]:
        if time.time() > deadline - 35:
            break
        ht = slsqp_run(h, n, a)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c, best_h, h = c, ht.copy(), ht
    if time.time() < deadline - 50:
        hm, cm = mom_subgrad(h, n, min(deadline - 30, time.time() + 45))
        if cm < best_c:
            best_c, best_h, h = cm, hm, hm
    if time.time() < deadline - 12:
        hp, cp = greedy_polish(h, n, deadline - 8)
        if cp < best_c:
            best_c, best_h = cp, hp
    if time.time() < deadline - 5:
        ht = slsqp_run(best_h, n, 150000, 4000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht
        hp, cp = greedy_polish(best_h, n, deadline - 1)
        if cp < best_c:
            best_c, best_h = cp, hp
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(250, 0.5)
        n0 = 250
    alloc1 = min(380, (deadline - t0) * 0.38)
    best_h, best_c = full_opt(h0, n0, t0 + alloc1)
    cur_n = len(best_h)
    for n_try in [200, 300, 350]:
        if n_try == cur_n or time.time() > deadline - 140:
            continue
        rem = deadline - time.time()
        if rem < 140:
            break
        x1 = np.linspace(0, 2, cur_n)
        x2 = np.linspace(0, 2, n_try)
        hi = project(np.interp(x2, x1, best_h), n_try)
        alloc = min(120, rem * 0.25)
        ho, co = full_opt(hi, n_try, time.time() + alloc)
        if co < best_c:
            best_c, best_h, cur_n = co, ho, n_try
    if time.time() < deadline - 70:
        n = len(best_h)
        rem = deadline - time.time() - 15
        n_restarts = max(3, int(rem / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            sc = 0.006 + 0.05 * (r / max(1, n_restarts - 1))
            hp = project(best_h + rng.normal(0, sc, n), n)
            alloc = min(55, (deadline - time.time() - 20) * 0.35)
            if alloc < 30:
                break
            ho, co = full_opt(hp, n, time.time() + alloc)
            if co < best_c:
                best_c, best_h = co, ho
    if time.time() < deadline - 5:
        n = len(best_h)
        hp, cp = greedy_polish(best_h, n, deadline - 1)
        if cp < best_c:
            best_c, best_h = cp, hp
        best_h = project(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
