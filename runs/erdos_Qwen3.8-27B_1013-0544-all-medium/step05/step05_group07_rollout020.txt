import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_grad(h, i, n):
    """Compute alpha_k = dc_k/dh[i] for all k (indexed 0..2n-2)."""
    # k_index = L + (n-1) where L is the actual lag
    # c_L = sum_{j in valid} h[j](1-h[j-L])
    # dc_L/dh[i] = (1-h[i-L])*I[i in valid(L), i-L in [0,n-1]] - h[i+L]*I[i+L in valid(L), i in [0,n-1]]
    
    L = np.arange(-(n-1), n)  # all lags, length 2n-1
    
    # valid range for c_L:
    # if L >= 0: j in [L, n-1]
    # if L < 0:  j in [0, n-1+L]
    j_lo = np.maximum(L, 0)
    j_hi = np.minimum(n-1, n-1+L)
    
    # Term 1: i in [j_lo, j_hi] AND i-L in [0, n-1]
    t1_valid = (i >= j_lo) & (i <= j_hi) & (i - L >= 0) & (i - L <= n-1)
    
    # Term 2: i+L in [j_lo, j_hi] AND i in [0, n-1] (always true)
    t2_valid = (i + L >= j_lo) & (i + L <= j_hi)
    
    alpha = np.zeros(2*n-1)
    alpha[t1_valid] = 1.0 - h[i - L[t1_valid]]
    alpha[t2_valid] += -h[i + L[t2_valid]]
    
    return alpha, L

            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_coord_grad(h, i, n):
    L = np.arange(-(n-1), n, dtype=np.float64)
    j_lo = np.maximum(L, 0)
    j_hi = np.minimum(n-1, n-1+L)
    t1 = (i >= j_lo) & (i <= j_hi) & (i - L >= 0) & (i - L <= n-1)
    t2 = (i + L >= j_lo) & (i + L <= j_hi)
    alpha = np.zeros(2*n-1)
    if t1.any():

    if t2.any():
        alpha[t2] -= h[np.clip(i + L[t2], 0, n-1)]
    return alpha

def solve_1d(alpha, beta):
    lo, hi = 0.0, 1.0
    for _ in range(50):

        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(alpha * m1 + beta)
        f2 = np.max(alpha * m2 + beta)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

            break

            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_coord_grad(h, i, n):
    L = np.arange(-(n-1), n)
    j_lo = np.maximum(L, 0)
    j_hi = np.minimum(n - 1, n - 1 + L)
    t1 = (i >= j_lo) & (i <= j_hi) & (i - L >= 0) & (i - L <= n - 1)
    t2 = (i + L >= j_lo) & (i + L <= j_hi)
    alpha = np.zeros(2 * n - 1)
    if np.any(t1):
        alpha[t1] = 1.0 - h[i - L[t1]]
    if np.any(t2):
        alpha[t2] -= h[i + L[t2]]
    return alpha

def solve_1d(alpha, beta):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(alpha * m1 + beta) < np.max(alpha * m2 + beta):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coord_descent(h, n, deadline, max_sweeps=10000):
    dx = 2.0 / n
    h = np.clip(h.copy(), 0.0, 1.0)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1 - h, mode='full')
    stale = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        max_delta = 0.0
        for i in range(n):
            alpha = make_coord_grad(h, i, n)
            beta = c - alpha * h[i]
            t_new = solve_1d(alpha, beta)
            d = t_new - h[i]
            if abs(d) > 1e-12:
                c += d * alpha
                h[i] = t_new
                if abs(d) > max_delta:
                    max_delta = abs(d)
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full')
        c_new = float(np.max(c) * dx)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if max_delta < 1e-11:
            break
        if stale > 100:
            break
    return best_h, best_c

def smooth_obj_grad(h, n, alpha_param):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha_param * (c - m)
    sm = shifted.max()
    exp_vals = np.exp(shifted - sm)
    p = exp_vals / exp_vals.sum()
    obj = m + (sm + np.log(exp_vals.sum())) / alpha_param
    return obj, p

def lbfgs_smooth(h, n, alpha_param, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        val, _ = smooth_obj_grad(x, n, alpha_param)
        val += 1e6 * (x.sum() - target) ** 2
        return val
    
    def grad(x):
        c_raw = np.correlate(x, 1 - x, mode='full')
        _, p = smooth_obj_grad(x, n, alpha_param)
        g = np.zeros(n)
        for i in range(n):
            alpha_i = make_coord_grad(x, i, n)
            g[i] = np.dot(p, alpha_i) * dx
        g += 2e6 * (x.sum() - target)
        return g
    
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h
