        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Power mean L-BFGS (45%)
    p1_end = t0 + 0.45 * (budget_s - 5)
    p_schedule = [2, 4, 8, 16, 32, 64, 128]
    for p in p_schedule:
        if time.time() > p1_end - 30:
            break
        def pobj(x, pp=p):
            v, _ = pmean_val_grad(x, n, pp)
            return v + 1e5 * (x.sum() - n / 2.0) ** 2
        def pgrad(x, pp=p):
            _, g = pmean_val_grad(x, n, pp)
            g = g + 2e5 * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(pobj, best_h.copy(), method='L-BFGS-B', jac=pgrad,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 2: Top-k smooth L-BFGS (20%)
    p2_end = t0 + 0.65 * (budget_s - 5)
    for k_top in [3, 5, 8]:
        for alpha in [200, 1000, 5000]:
            if time.time() > p2_end - 20:
                break
            def tkobj(x, kt=k_top, al=alpha):
                v, _ = topk_smooth_val_grad(x, n, kt, al)
                return v + 1e5 * (x.sum() - n / 2.0) ** 2
            def tkgrad(x, kt=k_top, al=alpha):
                _, g = topk_smooth_val_grad(x, n, kt, al)
                g = g + 2e5 * (x.sum() - n / 2.0)
                return g
            try:
                res = minimize(tkobj, best_h.copy(), method='L-BFGS-B', jac=tkgrad,
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 3: Subgradient momentum (15%)
    p3_end = t0 + 0.80 * (budget_s - 5)
    h_sg, c_sg = subgrad_momentum(best_h.copy(), n, p3_end, rng)
    if c_sg < best_c:
        best_c = c_sg
        best_h = h_sg.copy()

    # Phase 4: BB-step optimization (10%)
    p4_end = t0 + 0.90 * (budget_s - 5)
    h_bb, c_bb = bb_step_optimize(best_h.copy(), n, p4_end, rng)
    if c_bb < best_c:
        best_c = c_bb
        best_h = h_bb.copy()

    # Phase 5: Local search + binary relaxation (10%)
    p5_end = deadline
    # Binary relaxation from rounded solution
    idx_top = np.argsort(best_h)[::-1][:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx_top] = 1.0
    h_br, c_br = binary_relax(h_bin, n, p5_end - 30, rng)
    if c_br < best_c:
        best_c = c_br
        best_h = h_br.copy()

    # Final local search
    h_ls, c_ls = local_search(best_h.copy(), n, deadline - 3, rng)
    if c_ls < best_c:
        best_c = c_ls
        best_h = h_ls.copy()

    # Final polish with high-alpha top-k
    if time.time() < deadline - 10:
        def fpobj(x):
            v, _ = topk_smooth_val_grad(x, n, 2, 10000)
            return v + 1e6 * (x.sum() - n / 2.0) ** 2
        def fpgrad(x):
            _, g = topk_smooth_val_grad(x, n, 2, 10000)
            g = g + 2e6 * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(fpobj, best_h.copy(), method='L-BFGS-B', jac=fpgrad,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 200, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
