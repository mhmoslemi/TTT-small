def simulated_annealing(h0, n, deadline, t_start=0.5, t_end=0.001, max_iters=500000, seed=777):

    target = n // 2
    idx = np.argsort(h0)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    cur_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    best_c = cur_c
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max_iters)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i_idx = zeros[rng.randint(len(zeros))]
        j_idx = ones[rng.randint(len(ones))]
        h_bin[i_idx] = 1.0
        h_bin[j_idx] = 0.0
        new_c = compute_c5(h_bin, n)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-15)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[i_idx] = 0.0
            h_bin[j_idx] = 1.0
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_c5(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = g - m
    return m + np.logaddexp.reduce(sh) / alpha

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)})
    def obj(x):
        return smooth_max_c5(x, n, alpha)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0

        to = -lag
        L = n - to
        if L > 0:
            g[:L] += dx * (1.0 - h[to:])
        if to < n:
            g[to:] -= dx * h[:n - to]
    return g

def weighted_subgrad_bt(h0, n, deadline, max_iter=300000, alpha_w=2000, top_k=5, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.08
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g_all = compute_all_g(best_h, n)
        m = g_all.max()
        order = np.argsort(g_all)[-top_k:]
        w = np.exp(alpha_w * (g_all[order] - m))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, k in zip(w, order):
            lag = k - (n - 1)
            grad += wi * grad_single(best_h, n, lag)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        curr_lr = lr
        found = False
        for _ in range(18):
            ht = project_feasible(best_h - curr_lr * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-12:
                best_h = ht
                best_c = ct
                lr = min(curr_lr * 1.08, 0.4)
                found = True
                break
            curr_lr *= 0.65
        if not found:
            lr *= 0.93
            if lr < 1e-9:
                break
    return best_h, best_c

def simulated_annealing(h0, n, deadline, t_start=0.5, t_end=0.0005, max_iters=500000, seed=777):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    cur_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    best_c = cur_c
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max_iters)
        j_idx = ones[rng.randint(len(ones))]
        i_idx = zeros[rng.randint(len(zeros))]
        h_bin[i_idx] = 1.0
        h_bin[j_idx] = 0.0
        new_c = compute_c5(h_bin, n)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-15)):
            cur_c = new_c
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[i_idx] = 0.0
            h_bin[j_idx] = 1.0
    return best_h, best_c

def binary_local_search(h0, n, deadline, max_sweeps=50000, seed=123):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 2
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    # Phase 1: SLSQP at n0 (25%)
    p1 = t0 + 0.25 * (budget_s - 2)
    h = best_h.copy()
    for alpha in [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]:
        if time.time() > p1 - 3:
            break
        h = slsqp_opt(h, n0, alpha, 2000, p1 - 3)
        c = compute_c5(h, n0)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    # Phase 2: Weighted subgradient (15%)
    p2 = t0 + 0.40 * (budget_s - 2)
    h, c = weighted_subgrad_bt(best_h.copy(), n0, p2, max_iter=200000, alpha_w=2000, top_k=5, seed=42)
    if c < best_c - 1e-14:
        best_c = c
        best_h = h.copy()
    # Phase 3: SLSQP polish (10%)
    p3 = t0 + 0.50 * (budget_s - 2)
    if time.time() < p3 - 3:
        h = slsqp_opt(best_h.copy(), best_n, 2000000, 2000, p3 - 3)
        c = compute_c5(h, best_n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    # Phase 4: Try n=300 (15%)
    p4 = t0 + 0.65 * (budget_s - 2)
    if time.time() < p4 - 20:
        n300 = 300
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n300)
        h300 = project_feasible(np.interp(x_new, x_old, best_h), n300)
        for alpha in [500, 5000, 50000, 500000, 5000000]:
            if time.time() > p4 - 5:
                break
            h300 = slsqp_opt(h300, n300, alpha, 1500, p4 - 5)
            c = compute_c5(h300, n300)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h300.copy()
                best_n = n300
        if time.time() < p4 - 3:
            src = best_h.copy() if best_n == n300 else h300.copy()
            h300, c = weighted_subgrad_bt(src, n300, p4 - 3, max_iter=80000, alpha_w=2000, top_k=5, seed=99)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h300.copy()
                best_n = n300
    # Phase 5: Binary SA + local search (20%)
    p5 = t0 + 0.85 * (budget_s - 2)
    p5a = t0 + 0.75 * (budget_s - 2)
    if time.time() < p5a - 5:
        ha, ca = simulated_annealing(best_h.copy(), best_n, p5a - 5,
                                      t_start=0.3, t_end=0.0003, max_iters=600000, seed=999)
        if ca < best_c - 1e-14:
            best_c = ca
            best_h = ha.copy()
    if time.time() < p5 - 3:
        hb, cb = binary_local_search(best_h.copy(), best_n, p5 - 3, max_sweeps=40000, seed=456)
        if cb < best_c - 1e-14:
            best_c = cb
            best_h = hb.copy()
    # Phase 6: Final continuous polish (15%)
    if time.time() < dl - 10:
        h = slsqp_opt(best_h.copy(), best_n, 50000000, 2000, dl - 10)
        c = compute_c5(h, best_n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    if time.time() < dl - 3:
        h, c = weighted_subgrad_bt(best_h.copy(), best_n, dl - 3, max_iter=80000,
                                    alpha_w=5000, top_k=10, seed=77)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
