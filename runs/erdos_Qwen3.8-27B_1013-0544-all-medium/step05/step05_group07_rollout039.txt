```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    # Project onto {0 <= h <= 1, sum(h) = n/2}
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-12: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def binary_local_search(hb, n, deadline, rng, max_iter=200000):
    """Intensive local search on binary sequences with single swaps and block moves."""
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    ones = sorted(np.where(hb == 1)[0].tolist())
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # With some probability, do a block move
        if rng.random() < 0.3 and len(ones) > 3:
            # Pick a block of consecutive ones
            block_size = min(rng.randint(2, 6), len(ones))
            start_idx = rng.randint(0, len(ones) - block_size + 1)
            block = ones[start_idx:start_idx + block_size]
            
            # Check if they're consecutive
            if all(block[j+1] - block[j] == 1 for j in range(block_size - 1)):
                # Try to move this block
                # Find a new position where block_size consecutive zeros exist
                zeros_positions = np.where(hb == 0)[0]
                if len(zeros_positions) >= block_size:
                    # Try random new positions
                    for _ in range(20):
                        new_start = rng.randint(0, n - block_size + 1)
                        # Check if new position is all zeros
                        if all(hb[new_start + j] == 0 for j in range(block_size)):
                            # Do the move
                            for j in range(block_size):
                                hb[new_start + j] = 1.0
                                hb[block[j]] = 0.0
                            c_new = compute_c5(hb, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = hb.copy()
                                ones = sorted(np.where(hb == 1)[0].tolist())
                            else:
                                # Undo
                                for j in range(block_size):
                                    hb[block[j]] = 1.0
                                    hb[new_start + j] = 0.0
            continue
        
        # Single swap
        if not ones:
            break
        i = rng.choice(ones)
        # Find a zero to swap with
        zeros = np.where(hb == 0)[0]
        if len(zeros) == 0:
            break
        j = rng.choice(zeros)
        
        hb[i], hb[j] = 0.0, 1.0
        c_new = compute_c5(hb, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = hb.copy()
            ones.remove(i)
            ones.append(j)
            ones.sort()
        else:
            hb[i], hb[j] = 1.0, 0.0
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def corr_values(hb, n):
    return np.correlate(hb, 1.0 - hb, mode='full')

def binary_search(hb, n, deadline, rng):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    ones = list(np.where(hb == 1)[0])
    zeros = list(np.where(hb == 0)[0])
    stall = 0
    while time.time() < deadline:
        best_delta = 1e-10
        best_move = None
        for i in ones:
            for j in zeros:
                hb[i], hb[j] = 0.0, 1.0
                c = compute_c5(hb, n)
                d = c - best_c
                if d < -best_delta:
                    best_delta = -d
                    best_move = (i, j)
                hb[i], hb[j] = 1.0, 0.0
        if best_move is not None:
            i, j = best_move
            hb[i], hb[j] = 0.0, 1.0
            best_c = compute_c5(hb, n)
            best_h = hb.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            stall = 0
        else:
            stall += 1
            if stall >= 15:
                k = rng.randint(4, 12)
                for _ in range(k):
                    if not ones or not zeros:
                        break
                    ii = ones[rng.randint(len(ones))]
                    jj = zeros[rng.randint(len(zeros))]
                    hb[ii], hb[jj] = 0.0, 1.0
                    ones.remove(ii)
                    ones.append(jj)
                    zeros.remove(jj)
                    zeros.append(ii)
                best_c = compute_c5(hb, n)
                best_h = hb.copy()
                stall = 0
    return best_h, best_c

def smooth_obj(x, n, alpha, target):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    val += 1e3 * (x.sum() - target) ** 2
    return val

def smooth_grad(x, n, alpha, target):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-8:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += w[idx] * (1.0 - x[:n - k])
            g[:n - k] += -w[idx] * x[k:]
        else:
            g[:n + k] += w[idx] * (1.0 - x[-k:])
            g[-k:] += -w[idx] * x[:n + k]
    g *= dx
    g += 2e3 * (x.sum() - target)
    return g

def continuous_optimize(h_init, n, deadline):
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [200, 800, 3000, 10000, 40000, 150000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        try:
            res = minimize(smooth_obj, h, args=(n, alpha, target),
                         method='L-BFGS-B', jac=smooth_grad,
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c
```
