import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, n, alpha):
    """Returns (objective_value, gradient) for smooth max objective."""
    g = compute_all_g(h, n)
    m = g.max()
    sh = (g - m) * alpha
    sh_max = sh.max()
    exp_sh = np.exp(sh - sh_max)
    softmax = exp_sh / exp_sh.sum()
    obj = m + (sh_max + np.log(exp_sh.sum())) / alpha
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(g)):
        w = softmax[k] * dx
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w * (1.0 - h[:n-lag])
                grad[:n-lag] -= w * h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n-to] += w * (1.0 - h[to:])
            grad[to:] -= w * h[:n-to]
    return obj, grad

def adam_refine(h0, n, deadline, alpha_start=100, alpha_end=1000000, n_alphas=10, lr=0.005):
    """Adam optimizer with smooth max, increasing alpha."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    for alpha in alphas:
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        t = 0
        while time.time() < deadline:
            t += 1
            obj, grad = smooth_max_obj_grad(h, n, alpha)
            m_v = 0.9 * m_v + 0.1 * grad
            v_v = 0.999 * v_v + 0.001 * grad**2
            m_hat = m_v / (1 - 0.9**t)
            v_hat = v_v / (1 - 0.999**t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = project_feasible(h - step, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            if t > 2000:
                break
    
    return best_h, best_c

def subgrad_line_search(h0, n, deadline, max_iter=50000):
    """Subgradient method with exact line search on the max function."""
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += dx * (1.0 - best_h[:n-lag])
                grad[:n-lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n-to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n-to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        # Binary search for best step
        lo, hi = 0.0, lr * 2
        best_step = lo
        best_val = m
        for _ in range(20):
            mid = (lo + hi) / 2
            ht = project_feasible(best_h - mid * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_val:
                best_val = ct
                best_step = mid
                lo = mid
            else:
                hi = mid
        if best_val < best_c - 1e-14:
            best_h = project_feasible(best_h - best_step * grad, n)
            best_c = best_val
            lr = min(lr * 1.1, 1.0)
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
    return best_h, best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = (g - m) * alpha
    exp_sh = np.exp(sh - sh.max())
    sw = exp_sh / exp_sh.sum()
    obj = m + (sh.max() + np.log(exp_sh.sum())) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(g)):
        w = sw[k] * dx
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += w * (1.0 - h[to:])
            grad[to:] -= w * h[:n - to]
    return obj, grad

def adam_refine(h0, n, deadline, alphas, lr=0.003, max_it=4000):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

            break
        mv = np.zeros(n)
        vv = np.zeros(n)

            if time.time() > deadline:
                break
            _, grad = smooth_obj_grad(h, n, alpha)
            mv = 0.9 * mv + 0.1 * grad
            vv = 0.999 * vv + 0.001 * grad ** 2
            mh = mv / (1.0 - 0.9 ** t)
            vh = vv / (1.0 - 0.999 ** t)
            h_new = project_feasible(h - lr * mh / (np.sqrt(vh) + 1e-8), n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def subgrad_polish(h0, n, deadline, max_iter=100000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    lr = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += dx * (1.0 - best_h[:n - lag])
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        best_step = 0.0
        best_val = m
        step = lr
        for _ in range(12):
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_val:

                best_step = step
                step *= 1.4
            else:

        if best_val < best_c - 1e-14:

            best_c = best_val
            lr = min(max(lr, best_step * 0.8), 1.0)

            if lr < 1e-10:

    return best_h, best_c
