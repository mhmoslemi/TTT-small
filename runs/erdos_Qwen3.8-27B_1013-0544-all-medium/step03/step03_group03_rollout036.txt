    x_max = np.max(x)
    if x_max <= 0:
        return 0.0
    xn = x / x_max
    return x_max * np.sum(xn**p) ** (1.0/p)

So the first loop should be for d = 0 to n-1, and the second for d = 1 to n-1. Let me verify the ranges:

Now, for the paired coordinate descent:
- Given pair (i, j), compute g_i = grad_coord(h, i, n) and g_j = grad_coord(h, j, n)
- delta_g = g_i - g_j
- Current corr = np.correlate(h, 1-h, mode='full')
- Find δ* that minimizes max_m (corr[m] + δ * delta_g[m])
  subject to: max(0, h[i]-1) ≤ -δ ≤ min(1, h[i]) and max(0, h[j]-1) ≤ δ ≤ min(1, h[j])
  i.e., lo = max(h[i]-1, h[j]-1) ≤ δ ≤ hi = min(1-h[i], 1-h[j])
  
  Wait: h[i] + δ ∈ [0,1] means -h[i] ≤ δ ≤ 1-h[i]
  h[j] - δ ∈ [0,1] means h[j]-1 ≤ δ ≤ h[j]
  So: lo = max(-h[i], h[j]-1), hi = min(1-h[i], h[j])

- Binary search for δ where subgradient changes sign:
  subgrad(δ) = delta_g[argmax_m (corr[m] + δ * delta_g[m])]
  if subgrad < 0: lo = δ (need to increase δ)
  if subgrad > 0: hi = δ (need to decrease δ)

Let me also reconsider the L-BFGS-B phase. The key advantage of L-BFGS-B over SLSQP is speed (it can do many more iterations in the same time). For a 210-variable problem, L-BFGS-B should be very fast (each iteration is O(n) for the gradient, and the quasi-Newton update is O(n²) but only for the most recent ~10 gradients, so effectively O(n)).

The objective for L-BFGS-B:
obj(h) = smooth_max(corr(h)) * dx + penalty * (sum(h) - n/2)²

The gradient:
∂obj/∂h[i] = (∂ smooth_max / ∂corr[m*]) * (∂corr[m*]/∂h[i]) * dx + 2*penalty*(sum(h)-n/2)

where m* is the argmax lag (or a weighted combination for the smooth max).

For the log-sum-exp smooth max:
∂ smooth_max / ∂corr[m] = exp(β*corr[m]) / Σ exp(β*corr[m'])
= softmax(β*corr)

So the gradient is:
∂obj/∂h[i] = dx * Σ_m softmax(β*corr)[m] * ∂corr[m]/∂h[i] + 2*penalty*(sum(h)-n/2)

This can be computed efficiently if I precompute the softmax weights and the gradient vectors.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_lse(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -700, 700)
    vals = np.exp(sh)
    return m + np.log(vals.sum()) / beta, vals / vals.sum()

def grad_corr_all(h, i, n):
    """Compute d(corr[m])/d(h[i]) for all m."""
    g = np.zeros(2 * n - 1)
    for d in range(n):
        m = n - 1 + d
        val = 0.0
        if i + d < n:
            val += (1.0 - h[i + d])
        if i >= d:
            val -= h[i - d]
        g[m] = val
    for d in range(1, n):
        m = n - 1 - d
        val = 0.0
        if i >= d:
            val += (1.0 - h[i - d])
        if i + d < n:
            val -= h[i + d]
        g[m] = val
    return g

def obj_grad_lse(h, dx, beta, pen_lam, target_sum):
    """Objective and gradient for L-BFGS-B."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm, weights = smooth_max_lse(corr, beta)
    obj = sm * dx + pen_lam * (h.sum() - target_sum) ** 2
    
    # Gradient
    grad = np.zeros(n)
    for i in range(n):
        gi = grad_corr_all(h, i, n)
        grad[i] = dx * np.dot(weights, gi) + 2 * pen_lam * (h.sum() - target_sum)
    return obj, grad

def paired_cd_step(h, i, j, n, dx):
    """One step of paired coordinate descent on (i, j)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    g_i = grad_corr_all(h, i, n)
    g_j = grad_corr_all(h, j, n)
    delta_g = g_i - g_j
    
    # Feasible range for delta
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return 0.0
    
    # Binary search for optimal delta
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = corr + mid * delta_g
        sm = np.max(vals)
        # Subgradient: delta_g at argmax
        sg = delta_g[np.argmax(vals)]
        if abs(sg) < 1e-15:
            break
        if sg < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0
