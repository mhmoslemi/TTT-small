```python
        # corr at lag s = h^T Q_s h where Q_s is a specific matrix
        # Actually, corr_s = sum_j h[j](1-h[j+s]) = sum_j h[j] - sum_j h[j]h[j+s]
        # This is linear + quadratic in h

        # Or: minimize the maximum using epigraph form
        # min t

        #      0 <= h <= 1
        #      sum(h) = target
        
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        # Build correlation expressions
        # corr_s = sum_j h[j] - h[j]*h[j+s]
        # In cvxpy, h[j]*h[j+s] is non-convex (bilinear)
        # So we can't directly use cvxpy for this
        
        # Alternative: linearize around current solution
        # corr_s ≈ corr_s(h_cur) + grad_s • (h - h_cur)
        # This gives a linear program!

        n1 = n - 1

        for lag_i in top_lags:

            i_s = idx + s
            i_ms = idx - s
            v1 = (i_s >= 0) & (i_s < n)
            v2 = (i_ms >= 0) & (i_ms < n)
            g = np.zeros(n)
            g[v1] += (1.0 - h_cur[i_s[v1]])
            g[v2] -= h_cur[i_ms[v2]]

        # s.t. 0 <= h <= 1, sum(h) = target
```

Let me combine ideas:
1. **Active set LP**: At each iteration, linearize the top-k correlations and solve the resulting LP to find the best h in the linearized model. This gives a full step (like in active set methods for programming).
2. **Update active set**: After solving, recompute correlations and update which lags are active.
3. **Repeat** until convergence.

This is the "active set method" for the min-max problem, and it should converge much faster than gradient descent.

The LP at each step is:
minimize: (sum of active gradients) • h
subject to: 0 ≤ h ≤ 1, sum(h) = target

This is a standard LP that can be solved very fast (n=280, 2n+1 constraints).

```python
def active_set_method(h, time_limit=300, start_k=5, end_k=50):
    """Active set method for min-max correlation."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project_h(h, target)
    t0 = time.time()
    n1 = n - 1
    idx = np.arange(n)
    
    cur_c5 = np.max(compute_corr(h)) * dx
    h_prev = h.copy()
    
    for it in range(1000):
        if time.time() - t0 > time_limit:
            break
        
        corr = compute_corr(h)
        max_val = np.max(corr)
        
        # Active set: lags with corr close to max
        threshold = max_val - 0.5  # within 0.5 of max
        active = np.where(corr >= threshold)[0]
        if len(active) > end_k:
            active = np.argsort(corr)[::-1][:end_k]
        
        # Compute combined gradient
        grad = np.zeros(n)
        for lag_i in active:
            s = int(lag_i) - n1
            i_s = idx + s
            i_ms = idx - s
            v1 = (i_s >= 0) & (i_s < n)
            v2 = (i_ms >= 0) & (i_ms < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[i_s[v1]])
            if np.any(v2):
                grad[v2] -= h[i_ms[v2]]
        
        # Solve LP: min grad • h subject to 0<=h<=1, sum(h)=target
        # Use scipy's linear programming
        from scipy.optimize import linprog
        c_vec = grad  # objective coefficients
        A_eq = np.ones((1, n))
        b_eq = np.array([target])
        bounds_lp = [(0, 1)] * n
        
        res = linprog(c_vec, A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
        
        if res.success and res.x is not None:
            h_new = project_h(res.x, target)
            c5_new = np.max(compute_corr(h_new)) * dx
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                h_prev = h_new
            else:
                # Try with a step: h = (1-t)*h_old + t*h_lp
                found = False
                for t_val in [0.5, 0.25, 0.1]:
                    h_blend = project_h((1-t_val)*h + t_val*res.x, target)
                    c5_blend = np.max(compute_corr(h_blend)) * dx
                    if c5_blend < cur_c5 - 1e-15:
                        h = h_blend
                        cur_c5 = c5_blend
                        found = True
                        break
                if not found:
                    # No improvement, try perturbing
                    break
        else:
            break
    
    return h, cur_c5
```

import numpy as np
from scipy.optimize import linprog, minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def active_set_optimize(h, time_limit=120, max_iter=500):
    """Active set linear programming for min-max correlation."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project_h(h, target)
    t0 = time.time()
    n1 = n - 1
    idx = np.arange(n)
    A_eq = np.ones((1, n))
    b_eq = np.array([target])
    bounds_lp = [(0.0, 1.0)] * n
    
    cur_c5 = np.max(compute_corr(h)) * dx
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        corr = compute_corr(h)
        max_val = np.max(corr)
        
        # Active set: top lags within tolerance of max
        tol = max(0.1, 0.02 * max_val)
        active = np.where(corr >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 100:
            active = np.argsort(corr)[::-1][:100]
        
        # Combined gradient
        grad = np.zeros(n)
        for lag_i in active:
            s = int(lag_i) - n1
            i_s = idx + s
            i_ms = idx - s
            v1 = (i_s >= 0) & (i_s < n)
            v2 = (i_ms >= 0) & (i_ms < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[i_s[v1]])
            if np.any(v2):
                grad[v2] -= h[i_ms[v2]]

        try:
            res = linprog(grad, A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs',

            break
        
        if not res.success or res.x is None:
            break
