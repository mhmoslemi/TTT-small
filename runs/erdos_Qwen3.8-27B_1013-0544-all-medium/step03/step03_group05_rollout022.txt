def slsqp_polish(h, n, target, dx, time_left, best_state):
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in [300, 800, 2000]:
        if time_left() < 3:
            break
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
        def obj_wrapper(x):
            return smooth_max_obj(x, dx, beta)
        def con_wrapper(x):
            return sum_constraint(x, target)
        def jac_wrapper(x):
            return sum_jac(x, target)
        cons = [{'type': 'eq', 'fun': con_wrapper, 'jac': jac_wrapper}]
        try:
            res = minimize(obj_wrapper, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_state[0]:
                best_state[0] = c5_new
                best_state[1] = h_new.copy()
                best_state[2] = n
        except Exception:
            pass
    return best_state[1]


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_c5 = compute_c5(h_parent)
    best_state = [best_c5, h_parent.copy(), n_parent]

    # Primary: work at parent resolution and nearby
    n_list = sorted(set([n_parent, 300, 350]))
    n_list = [n for n in n_list if 80 <= n <= 500]

    for n in n_list:
        if time_left() < 35:
            break
        target = n / 2.0
        dx = 2.0 / n

        # Build initial candidates
        inits = []
        if n == n_parent:
            inits.append(h_parent.copy())
        else:
            inits.append(project_h(resample(h_parent, n), target))
            if best_state[2] != n:
                inits.append(project_h(resample(best_state[1], n), target))

        x = np.linspace(0, 2, n)
        # Structured inits
        h_c1 = 0.5 + 0.25 * np.cos(np.pi * x) + 0.06 * np.sin(2 * np.pi * x)
        inits.append(project_h(h_c1, target))
        h_c2 = 0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.08 * np.sin(3 * np.pi * x)
        inits.append(project_h(h_c2, target))
        # Bump structures
        h_b1 = 0.5 + 0.3 * np.exp(-((x - 0.5) / 0.35) ** 2)
        inits.append(project_h(h_b1, target))
        h_b2 = 0.5 + 0.25 * (np.exp(-((x - 0.35) / 0.25) ** 2) + np.exp(-((x - 1.65) / 0.25) ** 2))
        inits.append(project_h(h_b2, target))

        for h_init in inits:
            if time_left() < 20:
                break
            cur = project_h(h_init.copy(), target)
            c5_cur = compute_c5(cur)
            if c5_cur < best_state[0]:
                best_state[0] = c5_cur
                best_state[1] = cur.copy()
                best_state[2] = n

            # Phase 1: simple momentum descent
            cur, c5_cur, _ = simple_descent(
                cur, n, target, dx, 4000, time_left, 15, best_state)

            # Phase 2: Nesterov descent
            cur, c5_cur, _ = nesterov_descent(
                cur, n, target, dx, 3000, time_left, 12, best_state)

        # SLSQP polish
        if time_left() > 8:
            if best_state[2] == n:
                h_pol = best_state[1].copy()
            else:
                h_pol = project_h(resample(best_state[1], n), target)
            slsqp_polish(h_pol, n, target, dx, time_left, best_state)

    # Final ultra-fine polish at best resolution
    if time_left() > 4:
        best_h = best_state[1].copy()
        best_n = best_state[2]
        target_f = best_n / 2.0
        dx_f = 2.0 / best_n

        for it in range(30000):
            if time_left() < 1.5:
                break
            corr_raw = np.correlate(best_h, 1.0 - best_h, mode="full")
            corr_val = corr_raw * dx_f
            center = best_n - 1
            # Single worst lag for final precision
            top_idx = np.argsort(corr_val)[::-1][:2]
            vals = corr_val[top_idx]
            w = np.exp((vals - vals[0]) * 600.0)
            w /= w.sum()
            grad = np.zeros(best_n)
            for t_idx, wi in zip(top_idx, w):
                lag = int(t_idx) - center
                grad += wi * grad_lag(best_h, lag, best_n, dx_f)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            step = 0.003
            improved = False
            for _ in range(30):
                h_trial = project_h(best_h - step * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_state[0] - 1e-16:
                    best_h = h_trial
                    best_state[0] = c5_trial
                    best_state[1] = h_trial.copy()
                    improved = True
                    break
                step *= 0.5
            if not improved and it > 100:
                # Tiny perturbation to escape
                if it % 500 == 0:
                    noise = np.random.normal(0, 0.0005, best_n)
                    h_trial = project_h(best_h + noise, target_f)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_state[0]:
                        best_h = h_trial
                        best_state[0] = c5_trial
                        best_state[1] = h_trial.copy()

    return best_state[1], best_state[0], best_state[2]
