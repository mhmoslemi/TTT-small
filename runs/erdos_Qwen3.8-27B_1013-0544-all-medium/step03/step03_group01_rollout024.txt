```python
def grad_logsumexp(h, n, alpha):
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    
    # Only compute gradients for significantly weighted correlations
    significant = w > 1e-10
    if not np.any(significant):
        return np.zeros(n)
    
    grad = np.zeros(n)
    for idx in np.where(significant)[0]:
        s = idx - (n-1)
        t = abs(s)
        if t == 0:
            grad += w[idx] * (1 - 2*h)
        else:
            grad[t:] += w[idx] * (1 - h[:n-t])
            grad[:n-t] -= w[idx] * h[t:]
    
    return grad
```

```python
    t = abs(s)

    if t == 0:

    else:

        max_idx = np.argmax(c)
        return grad_single(h, n, max_idx)
    w /= wsum

        else:

            grad[:n-t] -= w[idx] * h[t:]
```

```python
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.3

def local_refine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        break
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= s * d
                                h[j] += s * d
    return h, best_c
```
