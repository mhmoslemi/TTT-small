```python
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)

        return smooth_loss(x, n, alpha, mu)
    def gfun(x):
        return smooth_grad(x, n, alpha, mu)
    try:
        res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                      bounds=[(0,1)]*n,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except:
        return h

def subgrad_step(h, n, step_size):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_val = c.max()
    tol = max(1e-10, 1e-6 * max_val)
    active = np.where(c >= max_val - tol)[0]
    
    # Compute subgradient (average of gradients of active lags)
    grad = np.zeros(n)
    for k in active:
        offset = k - (n-1)
        if offset == 0:

        elif offset > 0:
            g = np.zeros(n)
            g[:n-offset] += (1 - h[offset:])
            g[offset:] += -h[:n-offset]

            off = -offset
            g = np.zeros(n)
            g[off:] += (1 - h[:n-off])
            g[:n-off] += -h[off:]
        grad += g
```

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-13:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_loss(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 0)
    e = np.exp(shifted)
    return m + np.log(e.sum()) / alpha + mu * (h.sum() - n / 2.0) ** 2

def smooth_grad(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 0)
    e = np.exp(shifted)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        offset = k - (n - 1)
        if offset == 0:
            grad += w[k] * (1 - 2 * h)
        elif offset > 0:
            grad[:n - offset] += w[k] * (1 - h[offset:])
            grad[offset:] += -w[k] * h[:n - offset]
        else:
            off = -offset
            grad[off:] += w[k] * (1 - h[:n - off])
            grad[:n - off] += -w[k] * h[off:]
    grad *= dx
    grad += 2 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgsb_step(h, n, alpha, mu, maxiter, ftol):
    def fun(x):
        return smooth_loss(x, n, alpha, mu)
    def gfun(x):
        return smooth_grad(x, n, alpha, mu)
    try:
        res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_step(h, n, step_size):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_val = c.max()
    tol = max(1e-10, 1e-5 * max_val)
    active = np.where(c >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        offset = k - (n - 1)
        if offset == 0:
            grad += (1 - 2 * h)
        elif offset > 0:
            grad[:n - offset] += (1 - h[offset:])
            grad[offset:] += -h[:n - offset]
        else:
            off = -offset
            grad[off:] += (1 - h[:n - off])
            grad[:n - off] += -h[off:]
    grad *= dx / len(active)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new, float(max_val)

def coord_search(h, n, i, other_sum, dx):
    """Golden section search for coordinate i maintaining sum"""
    lo, hi = 0.0, 1.0
    h_save = h[i]
    h[i] = 0.5 * (lo + hi)
    other_target = other_sum - 0.5 * (lo + hi)
    for _ in range(40):
        m1 = lo + 0.382 * (hi - lo)
        m2 = hi - 0.382 * (hi - lo)
        h[i] = 0.5 * (m1 + m2)
        # Scale others to maintain sum
        others = np.arange(n) != i
        curr_other = h[others].sum()
        if curr_other > 1e-14:
            h[others] *= (other_target / curr_other)
            h = np.clip(h, 0, 1)
        c1 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        h[i] = 0.5 * (hi + lo)
        curr_other = h[others].sum()
        if curr_other > 1e-14:
            h[others] *= (other_target / curr_other)
            h = np.clip(h, 0, 1)
        c2 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c1 < c2:
            hi = 0.5 * (m1 + m2)
        else:
            lo = 0.5 * (m1 + m2)
    h[i] = 0.5 * (lo + hi)
    curr_other = h[others].sum()
    if curr_other > 1e-14:
        h[others] *= (other_target / curr_other)
        h = np.clip(h, 0, 1)
    h[i] = h_save  # revert, caller will set
    return 0.5 * (lo + hi)

def fourier_subspace(h, n, K, deadline, rng):
    x = np.arange(n)
    A_init = np.array([2.0 / n * np.sum(h * np.cos(2 * np.pi * (k + 1) * x / n)) for k in range(K)])
    B_init = np.array([2.0 / n * np.sum(h * np.sin(2 * np.pi * (k + 1) * x / n)) for k in range(K)])

    def coeffs_to_h(coeffs):
        A = coeffs[:K]
        B = coeffs[K:]
        hh = np.full(n, 0.5)
        for k in range(K):
            hh += A[k] * np.cos(2 * np.pi * (k + 1) * x / n) + B[k] * np.sin(2 * np.pi * (k + 1) * x / n)
        return hh

    bounds = [(-0.48, 0.48)] * (2 * K)

    def de_obj(coeffs):
        hh = coeffs_to_h(coeffs)
        if hh.min() < -0.05 or hh.max() > 1.05:
            return 5.0
        hh = project_feasible(np.clip(hh, 0, 1), n)
        return compute_c5(hh, n)

    x0 = np.concatenate([A_init, B_init])
    if time.time() < deadline - 15:
        try:
            de_res = differential_evolution(de_obj, bounds, maxiter=100,
                                            popsize=12, tol=1e-7, seed=rng,
                                            polish=False, updating='deferred')
            best_coeffs = de_res.x
        except Exception:
            best_coeffs = x0
    else:
        best_coeffs = x0
    hh = coeffs_to_h(best_coeffs)
    hh = project_feasible(np.clip(hh, 0, 1), n)
    return hh

    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:

        if time.time() > p1_end:

            best_c = c_new
            best_h = h_new

        if time.time() > p2_end - 30:

        h_f = fourier_subspace(best_h, n, K, p2_end - 10, rng)
        c_f = compute_c5(h_f, n)

        # Polish

            c_new = compute_c5(h_new, n)
```
