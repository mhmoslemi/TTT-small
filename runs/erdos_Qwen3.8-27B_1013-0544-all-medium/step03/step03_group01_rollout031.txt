import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_array(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sla_gradients(h, n):
    """Compute gradient matrix g[m, j] = dc[m]/dh[j] where c[m] is the m-th correlate value * dx"""
    dx = 2.0 / n
    g = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for m in range(2 * n - 1):
        k = m - (n - 1)
        # For each j: g[m,j] = (1-h[j-k])*1[0≤j-k<n] + (-h[j+k])*1[0≤j+k<n]
        # First term: j-k in [0, n-1] → j in [k, k+n-1] ∩ [0, n-1]
        # Second term: j+k in [0, n-1] → j in [-k, n-1-k] ∩ [0, n-1]
        if k >= 0:
            # First term: j in [k, n-1]
            g[m, k:n] += dx * omh[:n - k]  # 1-h[0..n-1-k] = omh[0..n-1-k]
            # Second term: j in [0, n-1-k]
            g[m, 0:n-k] += -dx * h[k:n]  # -h[k..n-1]
        else:
            kn = -k
            # First term: j in [0, n-1-kn]
            g[m, 0:n-kn] += dx * omh[kn:n]  # 1-h[kn..n-1]
            # Second term: j in [kn, n-1]
            g[m, kn:n] += -dx * h[:n-kn]  # -h[0..n-1-kn]
    return g

def sla_step(h, n, damp=1.0):
    """One SLA iteration"""
    c_arr = compute_c5_array(h, n)
    g = sla_gradients(h, n)
    
    # LP: min t s.t. g[m] @ h_new - t <= c_arr[m] - g[m] @ h
    #     0 <= h_new <= 1, sum(h_new) = n/2
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.hstack([g, -np.ones((2 * n - 1, 1))])
    b_ub = c_arr - g @ h
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return project_feasible(damp * h_new + (1.0 - damp) * h, n)
    return h.copy()

def sla_optimize(h0, n, deadline, max_iters=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    damp = 1.0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = sla_step(h, n, damp)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new < best_c + 1e-10:  # making progress
            h = h_new
            # Gradually increase damping
            damp = min(1.0, damp * 1.01)
        else:
            # No progress, reduce damping
            damp = max(0.1, damp * 0.8)
            if damp <= 0.1:
                # Try a small perturbation
                h = project_feasible(h + 0.01 * np.random.randn(n), n)
                damp = 0.5
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        shifted = alpha * (c - c.max())
        mx = shifted.max()
        val = c.max() + np.log(np.sum(np.exp(shifted - mx)) + 1e-300) / alpha
        return float(val)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, min(i + 50, n)):  # limit to nearby pairs for speed

                        if c < best_c:

                        else:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_array(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14 * n:
        return h
    lo, hi = (0.0, 1.0) if s > target else (-1.0, 0.0)
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sla_gradients(h, n):
    dx = 2.0 / n
    g = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            g[m, k:n] += dx * omh[:n - k]
            g[m, 0:n - k] += -dx * h[k:n]
        else:
            kn = -k
            g[m, 0:n - kn] += dx * omh[kn:n]
            g[m, kn:n] += -dx * h[:n - kn]
    return g

def sla_step(h, n, damp=1.0):
    c_arr = compute_c5_array(h, n)
    g = sla_gradients(h, n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([g, -np.ones((2 * n - 1, 1))])
    b_ub = c_arr - g @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return project_feasible(damp * res.x[:n] + (1.0 - damp) * h, n)
    except:
        pass
    return h.copy()

def sla_optimize(h0, n, deadline, max_iters=20000):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    damp = 1.0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = sla_step(h, n, damp)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            h = h_new.copy()
            stall += 1
            if stall > 30:
                damp = max(0.3, damp * 0.85)
                if stall > 150:
                    break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        shifted = alpha * (c - c.max())
        mx = shifted.max()
        return float(c.max() + np.log(np.sum(np.exp(shifted - mx)) + 1e-300) / alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_local(h, n, deadline, deltas, window=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, min(i + window, n)):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i], h[j] = hi - s * d, hj + s * d
    return h, best_c

def gradient_descent_step(h, n, alpha, lr):
    c_arr = compute_c5_array(h, n)
    shifted = alpha * c_arr
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    omh = 1.0 - h
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[k:n] += w[m] * dx * omh[:n - k]
            g[0:n - k] += -w[m] * dx * h[k:n]
        else:
            kn = -k
            g[0:n - kn] += w[m] * dx * omh[kn:n]
            g[kn:n] += -w[m] * dx * h[:n - kn]
    return project_feasible(h - lr * g, n)
