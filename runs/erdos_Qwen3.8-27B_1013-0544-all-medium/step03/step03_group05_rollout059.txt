- For each term where h[i] appears as the (1-h) factor: -h[i-ℓ] * (v - old)

So c_ℓ(v) = c_ℓ(old) + (v - old) * [(1 - h[i+ℓ]) - h[i-ℓ]] (with appropriate bounds)

This is LINEAR in v! So for each lag, c_ℓ is linear in v.

Therefore max_ℓ c_ℓ(v) is the max of linear functions = convex piecewise linear.

To find the minimum of a convex piecewise linear function on [0,1], we can:
1. Find all the "breakpoints" where the argmax changes
2. Evaluate at each breakpoint and at the endpoints
3. Or use a bisection/ternary search approach

For the bisection approach on the subgradient:
- The subgradient of f(v) = max_ℓ c_ℓ(v) at v is the set of a_ℓ for all ℓ that achieve the max
- f is minimized where 0 is in the subgradient, i.e., where there exist both positive and negative a_ℓ among the active constraints (or a_ℓ = 0)

This is clean but might be overkill. For a quick improvement, let me just do a ternary search or evaluate at a few points.

Actually, the simplest effective approach: for each coordinate i, evaluate f(v) at v = 0, v = 1, and v = the point where the two largest a_ℓ * v + b_ℓ cross. This gives the optimum in O(n) per coordinate.

```python
def coord_optimize(h, i, target, dx):
    """Find optimal h[i] given all other h fixed."""
    n = len(h)
    old = h[i]
    
    # Compute base correlations (with h[i] = 0, i.e., remove contribution of h[i])
    # c_ℓ = sum of all terms not involving h[i] + h[i] * a_ℓ
    # So: c_ℓ(h[i]=v) = b_ℓ + v * a_ℓ
    
    # Compute a_ℓ = dc_ℓ/dh[i]
    # and b_ℓ = c_ℓ - old * a_ℓ
    
    corr_current = np.correlate(h, 1.0 - h, mode="full") * dx
    center = n - 1
    
    a = np.zeros(2*n-1)
    for t in range(2*n-1):
        lag = t - center
        if lag == 0:
            a[t] = (1 - 2*h[i]) * dx
        elif lag > 0:
            val = 0
            if i < n - lag:  # i appears as first factor
                val += (1 - h[i+lag]) * dx
            if i >= lag:  # i appears as (1-h) factor
                val += -h[i-lag] * dx
            a[t] = val
        else:
            l = -lag
            val = 0
            if i >= l:  # i appears as first factor
                val += (1 - h[i-l]) * dx
            if i < n - l:  # i appears as (1-h) factor
                val += -h[i+l] * dx
            a[t] = val
    
    b = corr_current - old * a
    
    # Now minimize max_t (b[t] + a[t]*v) for v in [0,1]
    # This is a 1D convex problem
    # Use the fact that at optimum, either:
    # - v = 0 or v = 1 (boundary)
    # - Two (or more) constraints are active and their slopes straddle 0
    
    # Simple: evaluate at v=0, v=1, and at intersection points of top constraints
    candidates = [0.0, 1.0, old]
    
    # Find the top few lags at current point and their intersection points
    f_current = b + a * old
    top_idx = np.argsort(f_current)[::-1][:10]
    
    for i1 in range(len(top_idx)):
        for i2 in range(i1+1, len(top_idx)):
            t1, t2 = top_idx[i1], top_idx[i2]
            if abs(a[t1] - a[t2]) > 1e-15:
                v_cross = (b[t2] - b[t1]) / (a[t1] - a[t2])
                if 0 <= v_cross <= 1:
                    candidates.append(v_cross)
    
    best_v = old
    best_val = np.inf
    for v in candidates:
        v = np.clip(v, 0, 1)
        val = np.max(b + a * v)
        if val < best_val:
            best_val = val
            best_v = v
    
    return best_v
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj_grad(x, beta, dx, target, pen):
    """Smooth max objective + penalty, and gradient."""
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    val = corr * dx
    m = val.max()
    sh = np.clip(beta * (val - m), -100, 100)
    w = np.exp(sh)
    ws = w.sum()
    obj = m + np.log(ws) / beta + pen * (x.sum() - target) ** 2
    
    g = np.zeros(n)
    center = n - 1
    for t in range(2 * n - 1):
        wt = w[t] / ws
        if wt < 1e-12:
            continue
        lag = t - center

            g += wt * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n-lag] += wt * dx * (1.0 - x[lag:n])
            g[lag:] += -wt * dx * x[:n-lag]

            g[l:] += wt * dx * (1.0 - x[:n-l])
            g[:n-l] += -wt * dx * x[l:]
    g += 2.0 * pen * (x.sum() - target)
    return obj, g

        o, _ = sm_obj_grad(x, beta, dx, target, pen)
        return o

        _, gr = sm_obj_grad(x, beta, dx, target, pen)
        return gr

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(res.x, target)

        return h0
```

```python
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def sm_obj_grad(x, beta, dx, target, pen):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    val = corr * dx
    m = val.max()
    sh = np.clip(beta * (val - m), -100, 100)
    w = np.exp(sh)
    ws = w.sum()
    obj = m + np.log(ws) / beta + pen * (x.sum() - target) ** 2
    g = np.zeros(n)
    center = n - 1
    for t in range(2 * n - 1):
        wt = w[t] / ws
        if wt < 1e-12:
            continue
        lag = t - center
        if lag == 0:
            g += wt * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n - lag] += wt * dx * (1.0 - x[lag:n])
            g[lag:] += -wt * dx * x[:n - lag]
        else:
            l = -lag
            g[l:] += wt * dx * (1.0 - x[:n - l])
            g[:n - l] += -wt * dx * x[l:]
    g += 2.0 * pen * (x.sum() - target)
    return obj, g


def lgbfb(h0, target, dx, beta, pen, maxiter=200):

                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_h(res.x, target)

        return h0.copy()
```
