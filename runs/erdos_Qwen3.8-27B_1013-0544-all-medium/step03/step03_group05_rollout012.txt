import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad


def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def obj_smooth_max(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx


def nesterov_subgradient(h, n, target, dx, max_iters, start_time, budget_s,
                         best_state, mu=0.85, alpha0=0.08, decay=0.5e-5,
                         explore_every=150, explore_noise=0.015):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    v = np.zeros(n)
    lag_hist = []
    stall = 0
    best_local = cur_c5
    best_local_h = cur.copy()

    for it in range(max_iters):
        if time.time() - start_time > budget_s - 15:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = min(3, n // 50 + 1)
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        w = np.exp((vals - vals[0]) * 80.0)
        w /= w.sum()

        grad = np.zeros(n)
        center = n - 1
        for t_idx, wi in zip(top_idx, w):
            lag = int(t_idx) - center
            grad += wi * grad_for_lag(cur, lag, n, dx)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        alpha = alpha0 / (1.0 + decay * it)
        v = mu * v - alpha * grad

        improved = False
        for trial in range(15):
            h_trial = project_h(cur + v, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_local:
                    best_local = cur_c5
                    best_local_h = cur.copy()
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            v *= 0.5
        if not improved:

        else:
            stall = 0

        if stalled_recover(stall, cur, n, target, explore_noise):

            cur_c5 = compute_c5(cur)
            v = np.zeros(n)
            stall = 0

    return best_local_h, best_local


def stalled_recover(stall, cur, n, target, noise):

def slsqp_phase(h, n, target, dx, betas, start_time, budget_s, best_state):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for beta in betas:
        if time.time() - start_time > budget_s - 20:
            break
        obj = partial(obj_smooth_max, dx=dx, beta=beta)
        con_fun = lambda x, t=target: np.sum(x) - t
        con_jac = lambda x: np.ones(x.shape)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj, cur, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-12})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                cur = h_new
                cur_c5 = c5_new
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n
        except Exception:
            pass
    return cur, cur_c5

        grad = grad_for_lag(cur, lag, n, dx)
