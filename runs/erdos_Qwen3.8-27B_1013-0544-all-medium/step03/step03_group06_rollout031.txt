    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(c)
    shift = np.clip(beta * (c - m), -500, 500)
    exp_shift = np.exp(shift)
    sum_exp = np.sum(exp_shift)
    sm = m + np.log(sum_exp) / beta
    weights = exp_shift / sum_exp
    return sm, weights

    corr = np.correlate(h, 1.0 - h, mode="full")

            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n-d]
    
    return sm * dx, grad * dx

    return best_h, best_c5, n_final
