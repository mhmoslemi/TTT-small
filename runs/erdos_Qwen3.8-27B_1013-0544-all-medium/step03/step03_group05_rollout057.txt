import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift))) / beta


def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_smooth_max(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    weights = np.exp(x_shift)
    wsum = weights.sum()
    if wsum < 1e-300:
        idx = np.argmax(corr)
        wsum = 1.0
        weights = np.zeros(len(corr))
        weights[idx] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    center = n - 1
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k]
        lag = k - center
        al = abs(lag)
        if al == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            grad[:n - al] += w * (1.0 - h[al:])
            grad[al:] -= w * h[:n - al]
    return grad * dx


def obj_smooth_max(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * dx


def sum_con(h, target):
    return np.sum(h) - target


def sum_con_jac(h, target):
    return np.ones(h.shape)


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def gradient_polish(cur, n, target, dx, max_iters, state, t0, budget):
    for it in range(max_iters):
        elapsed = time.time() - t0
        if budget - elapsed < 10:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full") * dx
        top_k = 2 if it < 2000 else 1
        top_idx = np.argsort(corr)[::-1][:top_k]
        vals = corr[top_idx]
        temp = 2000.0
        wts = np.exp((vals - vals[0]) * temp)
        wts /= wts.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, wt in zip(top_idx, wts):
            lag = int(t_idx) - center
            al = abs(lag)
            if al == 0:
                grad[:] += wt * (1.0 - 2.0 * cur)
            else:
                grad[:n - al] += wt * (1.0 - cur[al:])
                grad[al:] -= wt * cur[:n - al]
        grad *= dx
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        grad /= gn
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            h_trial = project_h(cur - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < state[0] - 1e-16:
                cur = h_trial
                state[0] = c5_trial
                state[1] = cur.copy()
                state[2] = n
                improved = True
                break
        if not improved and it % 50 == 0:
            T = 0.003 * max(0.05, 1.0 - it / max_iters)
            noise = np.random.normal(0, T, n)
            h_trial = project_h(cur + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < state[0] + 2 * T:
                cur = h_trial
    return cur
