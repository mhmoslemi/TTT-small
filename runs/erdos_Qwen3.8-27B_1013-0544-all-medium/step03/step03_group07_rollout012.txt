def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx

    for k in range(len(corr)):

        s = k - (n - 1)

            grad[ms] += wk * (1.0 - h[ms - s])

def slsqp_refine(h, maxiter, beta):

    bounds = Bounds(np.zeros(n), np.ones(n))
    one_vec = np.ones(n)

        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    def con_func(x):
        return x.sum() - target
    
    def con_jac(x):
        return one_vec
    
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]

def block_coordinate_descent(h, n_blocks, max_rounds, beta):
    """Optimize blocks of consecutive points one at a time."""
    n = len(h)
    block_size = n // n_blocks
    target = n / 2.0
    h = h.copy()
    best_v = compute_c5(h)
    
    for round_idx in range(max_rounds):
        improved = False
        for b in range(n_blocks):
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            blk = end - start
            
            # Optimize h[start:end] while fixing the rest
            # Use SLSQP on the block with the constraint that total sum = target
            fixed_sum = h.sum() - h[start:end].sum()
            block_target = target - fixed_sum
            
            if block_target < 0 or block_target > blk:
                continue
            
            h_blk = h[start:end].copy()
            
            def blk_obj(x):
                h_temp = h.copy()
                h_temp[start:end] = x
                v, _ = smoothmax_val_grad(h_temp, beta)
                return v
            
            def blk_grad(x):
                h_temp = h.copy()
                h_temp[start:end] = x
                _, g = smoothmax_val_grad(h_temp, beta)
                return g[start:end]
            
            bounds = Bounds(np.zeros(blk), np.ones(blk))
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - block_target,
                     'jac': lambda x: np.ones(blk)}]

                              bounds=bounds, constraints=cons,

                # Adjust to meet sum constraint

                    else:
                        h_new_blk += diff / blk

                    h[start:end] = h_new_blk
                    best_v = v
                    improved = True
            except Exception:
                pass

    n = len(h_init)
    
    # Create piecewise-constant representation
    seg_width = n / n_segments
    # Heights for each segment
    heights = np.array([h_init[int(i * seg_width + seg_width/2)] for i in range(n_segments)])
    heights = np.clip(heights, 0, 1)

        h = np.zeros(n)
        for i in range(n_segments):
            start = int(i * seg_width)
            end = int((i + 1) * seg_width)
            h[start:end] = heights[i]
        return h
    
    def obj(x):
        h = heights_to_h(x)
        v, _ = smoothmax_val_grad(h, beta)
        return v
    
    def grad(x):
        h = heights_to_h(x)
        _, g = smoothmax_val_grad(h, beta)
        # Aggregate gradient for each segment
        gx = np.zeros(n_segments)
        for i in range(n_segments):
            start = int(i * seg_width)
            end = int((i + 1) * seg_width)
            gx[i] = g[start:end].mean()
        return gx
    
    bounds = Bounds(np.zeros(n_segments), np.ones(n_segments))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_height_sum,
             'jac': lambda x: np.ones(n_segments)}]
    
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = heights_to_h(res.x)
        return project_feasible(h_out, n / 2.0)
    except Exception:
        return project_feasible(heights_to_h(heights), n / 2.0)

def interp_to_n(h, n_new):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()

            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
