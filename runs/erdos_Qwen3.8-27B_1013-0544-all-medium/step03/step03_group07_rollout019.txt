    return project_feasible(h_new, n_new / 2.0)

def block_descent(h, n_blocks, max_rounds, lr):
    """Cyclic block coordinate descent."""
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    block_size = n // n_blocks
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for rnd in range(max_rounds):
        for b in range(n_blocks):
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            # Compute gradient for the block
            val, grad = smoothmax_topk_val_grad(x, 200.0, 15)
            # Update block
            x[start:end] = np.clip(x[start:end] - lr * grad[start:end], 0, 1)
            # Re-project
            x = project_feasible(x, target)
            v = compute_c5(x)
            if v < best_v:
                best_v = v
                best_x = x.copy()
    
    return best_x, best_v

from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_topk_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_corr = corr[top_idx]
    m = np.max(top_corr)
    shifted = top_corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    if sw < 1e-300:
        w = np.zeros(len(top_corr))
        w[np.argmax(top_corr)] = 1.0
    else:
        w = exps / sw
    val = (m + np.log(max(sw, 1e-300)) / beta) * dx
    grad = np.zeros(n)
    for idx in range(len(w)):
        wk = w[idx]
        if wk < 1e-12:
            continue
        s = int(top_idx[idx]) - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        else:
            t = -s
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] += wk * (-h[t:])

    return val, grad

        return x.sum() - n / 2.0

        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h.copy(), n / 2.0)

def lbbfgs_fast(h, beta, top_k, lam, maxiter):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def fun(x):
        v, g = smoothmax_topk_val_grad(x, beta, top_k)
        d = x.sum() - n / 2.0
        return v + lam * d * d
