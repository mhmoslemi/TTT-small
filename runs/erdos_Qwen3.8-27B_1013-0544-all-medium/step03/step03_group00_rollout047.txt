def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

        return smoothmax_obj_grad(x, beta, penalty, top_k)[1]
    
    res = minimize(f, h, jac=fprime, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return project_feasible(res.x, target)

    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_obj_grad(h, beta, penalty, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    val += penalty * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        lo_a = max(0, s)
        hi_a = min(n, n + s)
        if hi_a > lo_a:
            grad[lo_a:hi_a] += wk * (1.0 - h[lo_a - s:hi_a - s])
        lo_b = max(0, -s)
        hi_b = min(n, n - s)
        if hi_b > lo_b:
            grad[lo_b:hi_b] += wk * (-h[lo_b + s:hi_b + s])
    grad *= dx
    grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return val, grad


def lbfgs_optimize(h, beta, penalty, top_k, maxiter):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smoothmax_obj_grad(x, beta, penalty, top_k)[0]

    def fprime(x):
        return smoothmax_obj_grad(x, beta, penalty, top_k)[1]

    try:
        res = minimize(f, h, jac=fprime, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def make_binary_near(h_template, rng):
    n = len(h_template)
    h_noisy = h_template + rng.randn(n) * 0.15
    target = n // 2
    sorted_vals = np.sort(h_noisy)
    thresh = sorted_vals[target]
    h_bin = (h_noisy >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        to_flip = rng.choice(ones, n_ones - target, replace=False)
        h_bin[to_flip] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        to_flip = rng.choice(zeros, target - n_ones, replace=False)
        h_bin[to_flip] = 1
    return h_bin


def refine_chain(h, betas, top_k, maxiter_per_beta, penalty=1e8):
    for beta in betas:
        h = lbfgs_optimize(h, beta, penalty, top_k, maxiter_per_beta)
    return h
