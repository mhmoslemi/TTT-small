import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(np.clip(beta*(c-m),-500,500)))/len(c))/beta) * dx

def sum_cos(target):
    def fun(x): return x.sum() - target
    def jac(x): return np.ones_like(x)
    return {'type': 'eq', 'fun': fun, 'jac': jac}

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0,2,n_new), np.linspace(0,2,len(h)), h)

def slsqp_opt(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [sum_cos(target)]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except:
        return h0

def binary_search(h, t0, budget, max_rounds=5000):
    """Local search: try random bit swaps, keep improvements."""
    n = len(h)
    target = n / 2
    b = (h >= 0.5).astype(float)
    # Fix count
    n_ones = int(b.sum())
    if n_ones != int(target):
        if n_ones > int(target):
            ones = np.where(b==1)[0]
            order = np.argsort(h[ones])[:n_ones - int(target)]
            b[ones[order]] = 0
        else:
            zeros = np.where(b==0)[0]
            order = np.argsort(-h[zeros])[:int(target) - n_ones]
            b[zeros[order]] = 1
    best = compute_c5(b)
    ones = np.where(b==1)[0]
    zeros = np.where(b==0)[0]
    stalls = 0
    for rnd in range(max_rounds):
        if time.time() - t0 > budget - 3:
            break
        # Try random swaps
        found = False
        n_try = min(200, len(ones)*len(zeros))
        for _ in range(n_try):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            b[i], b[j] = 0.0, 1.0
            c = compute_c5(b)
            if c < best - 1e-14:
                best = c
                ones = np.where(b==1)[0]
                zeros = np.where(b==0)[0]
                found = True
            else:
                b[i], b[j] = 1.0, 0.0
        if found:
            stalls = 0
        else:
            stalls += 1
            if stalls > 50:
                # Perturb: flip a few random bits
                k = np.random.randint(3, 10)
                for _ in range(k):
                    i = ones[np.random.randint(len(ones))]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def cons_fun(x, target):
    return x.sum() - target

def cons_jac(x, target):
    return np.ones(len(x))

def smooth_max_obj(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(np.clip(beta * (c - m), -500, 500))) / len(c)) / beta) * dx

def do_slsqp(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def binary_ls(h, t0, budget, rnds=8000, batch=150):

        else:

            pick = np.argsort(-h[zi])[:int(target) - n1]
            b[zi[pick]] = 1
    best = compute_c5(b)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    stall = 0
    for r in range(rnds):
        if time.time() - t0 > budget - 3:
            break
        found = False
        for _ in range(batch):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            b[i], b[j] = 0.0, 1.0
            c = compute_c5(b)
            if c < best - 1e-14:
                best = c
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                found = True
            else:
                b[i], b[j] = 1.0, 0.0
        if found:
            stall = 0
        else:

            if stall > 40:

                cb = compute_c5(b)
                if cb < best + 0.05:
                    best = cb
                    stall = 0
                else:
                    break
    return b, best

def coord_polish(h, t0, budget, sweeps=25):

    best = compute_c5(h)

                c = compute_c5(t)
                if c < best - 1e-14:
                    h, best, imp = t, c, True
                    break
        if not imp:
            break
    return h, best
