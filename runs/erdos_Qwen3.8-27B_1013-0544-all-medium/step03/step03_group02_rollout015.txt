def subgrad_desc(cur, target, n, n_iter, step, t0, budget):
    best_local = compute_c5(cur)
    stall = 0
    best_trial = cur.copy()
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        g = compute_sml_grad(cur, 5000.0, 2.0 / n, threshold=1e-6)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        trial = project_h(cur - step * g, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_trial = cur.copy()
            best_local = c_try
            stall = 0
            step = min(step * 1.05, 0.05)

            stall += 1
            step *= 0.7

    return best_trial

def coord_descent(cur, target, n, max_sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    cur = cur.copy()
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        deltas = [0.04, -0.04, 0.02, -0.02, 0.008, -0.008]
        for j in range(n):
            if budget - (time.time() - t0) < 1:

                trial[j] += d

                if c_try < best_c5 - 1e-15:

                    best_c5 = c_try

                    break

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(200, 0.5)

    n = len(h_parent)
    target = n / 2.0
    best_h = project_h(h_parent, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuation (fast convergence)
    for beta in [15, 40, 100, 300]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 25:
            break
        penalty = 50.0 / beta
        cur = do_lgbf(best_h, n, target, beta, penalty, maxiter=800, t0=t0, budget=budget_s)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 2: SLSQP with increasing smoothness
    for beta in [200, 500, 1500]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 25:
            break
        cur = do_slsqp(best_h, n, target, beta, maxiter=3000, t0=t0, budget=budget_s)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 3: Subgradient with adaptive step
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        cur = subgrad_desc(best_h, target, n, n_iter=2000, step=0.015, t0=t0, budget=budget_s)
        c = compute_c5(cur)

    if remaining > 30:

    # Phase 5: Restart with perturbations
    for att in range(8):

        if remaining < 40:
            break
        scale = 0.04 / (att + 1)
        hp = project_h(best_h + np.random.randn(n) * scale, target)
        penalty = 30.0
        hp = do_lgbf(hp, n, target, 300, penalty, maxiter=400, t0=t0, budget=budget_s)
        hp = do_slsqp(hp, n, target, 500, maxiter=2000, t0=t0, budget=budget_s)
        hp = subgrad_desc(hp, target, n, n_iter=800, step=0.01, t0=t0, budget=budget_s)
        hp, c = coord_descent(hp, target, n, max_sweeps=20, t0=t0, budget=budget_s)
        if c < best_c5:
            best_h, best_c5 = hp.copy(), c

    remaining = budget_s - (time.time() - t0)
    if remaining > 20:

        best_h = do_slsqp(best_h, n, target, 3000, maxiter=5000, t0=t0, budget=budget_s)

    # Ensure exact constraint
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
