        if abs(delta) > 1e-15:
            h[i] = t
            corr += delta * slopes
        if (idx + 1) % interval == 0:
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
    h = project_feasible(h, target_sum)
    return h

def obj_fn(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(x, target_sum):
    return x.sum() - target_sum

def con_jac(x, target_sum):
    return np.ones(len(x))

def slsqp_phase(h, n, target_sum, dx, rng, beta_list, max_iter=3000, time_limit=60):
    t_start = time.time()
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in beta_list:
        if time.time() - t_start > time_limit:
            break
        for trial in range(3):
            if time.time() - t_start > time_limit:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 = project_feasible(x0 + rng.randn(n) * (0.004 / trial), target_sum)
            try:
                res = minimize(obj_fn, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(210, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Coordinate descent (bulk of time)
    stall_count = 0
    while budget_s - (time.time() - t0) > 280:
        prev_c5 = best_c5
        h = coord_sweep(h, target_sum, rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                # Perturb to escape
                h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
                for _ in range(5):
                    h = coord_sweep(h, target_sum, rng)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                stall_count = 0

    # Phase 2: SLSQP polish
    if budget_s - (time.time() - t0) > 120:
        best_h, best_c5 = slsqp_phase(best_h, n, target_sum, dx, rng,
                                      [100, 500, 2000, 10000], max_iter=3000, time_limit=100)

    # Phase 3: More coordinate descent with intermediate projection
    if budget_s - (time.time() - t0) > 60:
        h = best_h.copy()
        while budget_s - (time.time() - t0) > 40:
            h = coord_sweep_inter(h, target_sum, rng, interval=5)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Grid exploration
    if budget_s - (time.time() - t0) > 40:
        x_old = np.linspace(0, 2, n, endpoint=False)
        candidates = sorted(set([n-20, n-10, n+10, n+20, n+40, 2*n]))
        for n_try in candidates:
            if budget_s - (time.time() - t0) < 15 or n_try < 50 or n_try > 600:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            for _ in range(15):
                h_t = coord_sweep(h_t, ts_t, rng)
            c5 = compute_c5(h_t)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_t.copy()
                n = n_try
                dx = dx_t
                target_sum = ts_t

    # Phase 5: Final high-beta SLSQP
    if budget_s - (time.time() - t0) > 15:
        best_h, best_c5 = slsqp_phase(best_h, n, target_sum, dx, rng,
                                      [50000, 200000], max_iter=5000, time_limit=15)

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
