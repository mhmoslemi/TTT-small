    # the LP relaxation (relaxing z >= 0) gives a lower bound.

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def local_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline - 3:
            break
        for i in range(n):
            if time.time() > deadline - 3:
                break
            for j in range(i + 1, n):
                for direction in (1, -1):
                    hi = h[i] + direction * d
                    hj = h[j] - direction * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i] -= direction * d
                            h[j] += direction * d
    return h, best_c

def anneal_step(h, n, rng, T):
    i = rng.randint(n)
    j = rng.randint(n)
    if i == j:
        return h, compute_c5(h, n), False
    d = rng.uniform(-0.03, 0.03)
    hi = h[i] + d
    hj = h[j] - d
    if 0 <= hi <= 1 and 0 <= hj <= 1:
        h[i] = hi
        h[j] = hj
        c_new = compute_c5(h, n)
        return h, c_new, True
    return h, compute_c5(h, n), False

def simulated_anneal(h, n, deadline, rng, T0=0.015, T1=0.0003):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    n_total = 3000
    for step in range(n_total):
        if time.time() > deadline - 5:
            break
        T = T0 * (T1 / T0) ** (step / max(n_total - 1, 1))
        h, c_new, moved = anneal_step(h, n, rng, T)
        if not moved:
            continue
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h, _, _ = anneal_step(h, n, rng, -T)
    return best_h, best_c

def optimize_n(h0, n, deadline, rng):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    def rem():
        return deadline - time.time()

    for alpha in [500, 2000, 10000, 50000, 200000, 1000000, 5000000]:
        if rem() < 12:
            break
        ht = slsqp_step(best_h, n, alpha, 2500, 1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if rem() > 80:
        sa_dl = time.time() + min(rem() * 0.35, 120)
        h_sa, c_sa = simulated_anneal(best_h, n, sa_dl, rng)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
            for alpha in [200000, 1000000, 5000000]:
                if rem() < 10:
                    break
                ht = slsqp_step(best_h, n, alpha, 2000, 1e-15)
                c = compute_c5(ht, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = ht

    if rem() > 30:
        best_h, best_c = local_search(best_h, n, deadline)

    for alpha in [5000000, 20000000, 100000000]:
        if rem() < 8:
            break
        ht = slsqp_step(best_h, n, alpha, 3000, 1e-17)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if rem() > 10:
        best_h, best_c = local_search(best_h, n, deadline, [0.002, 0.001, 0.0005, 0.0002, 0.0001])

    if rem() > 5:
        ht = slsqp_step(best_h, n, 50000000, 2000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def make_fourier_start(n, rng):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    n_modes = rng.randint(2, 7)
    for _ in range(n_modes):
        freq = rng.randint(1, 8)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 100

    best_h_overall = None
    best_c_overall = float('inf')

    if h_init is not None:
        dl1 = t0 + (deadline - t0) * 0.40
        h_res, c_res = optimize_n(h_init, n_init, dl1, rng)
        best_h_overall = (h_res, n_init)
        best_c_overall = c_res

    if h_init is not None:
        n_restarts = 6
        restart_deadline = min(deadline - 120, dl1 + (deadline - t0) * 0.30)
        time_per_restart = (restart_deadline - time.time()) / max(n_restarts, 1)
        for r in range(n_restarts):
            if time.time() > restart_deadline - 5:
                break
            if time_per_restart < 40:
                break
            scale = 0.03 + 0.08 * (r / n_restarts)
            hp = best_h_overall[0] + rng.normal(0, scale, n_init)
            hp = project_feasible(hp, n_init)
            rd = time.time() + time_per_restart
            h_res, c_res = optimize_n(hp, n_init, rd, rng)
            if c_res < best_c_overall:
                best_c_overall = c_res
                best_h_overall = (h_res, n_init)

    for n_try in [80, 120, 60]:
        if time.time() > deadline - 40:
            break
        phase_dl = min(time.time() + 50, deadline - 25)
        if phase_dl < time.time() + 15:
            break
        h_start = make_fourier_start(n_try, rng)
        h_res, c_res = optimize_n(h_start, n_try, phase_dl, rng)
        if c_res < best_c_overall:
            best_c_overall = c_res
            best_h_overall = (h_res, n_try)

    if best_h_overall is not None:
        best_h = best_h_overall[0].copy()
        n = best_h_overall[1]
    else:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c_overall = compute_c5(best_h, n)

    if deadline - time.time() > 15:
        best_h, c = local_search(best_h, n, deadline, [0.005, 0.002, 0.001, 0.0005, 0.0002])
        if c < best_c_overall:
            best_c_overall = c

    if deadline - time.time() > 8:
        ht = slsqp_step(best_h, n, 100000000, 2000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c_overall:
            best_h = ht
            best_c_overall = c

    if deadline - time.time() > 8:
        best_h, c = local_search(best_h, n, deadline, [0.001, 0.0005, 0.0002, 0.0001])
        if c < best_c_overall:
            best_c_overall = c

    best_h = project_feasible(best_h, n)
    best_c_overall = compute_c5(best_h, n)
    return best_h, best_c_overall, n
