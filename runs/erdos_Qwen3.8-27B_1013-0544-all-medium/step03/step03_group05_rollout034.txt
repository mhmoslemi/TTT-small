        corr_raw = np.correlate(h, 1.0 - h, mode="full")

        x = cp.Variable(n)
        t = cp.Variable()

                    t_bound = max(t_bound * 0.95, c5)

The problem is: minimize max_k C_k(h) subject to constraints.
Where C_k(h) = sum_i h[i] * (1-h[i+k]) * dx (bilinear).

LP relaxation at point h0:
minimize t
subject to:
  For each active k: linearization of C_k at h0 <= t
  0 <= h <= 1
  sum(h) = target

The linearization of C_k at h0 is:
C_k(h) ≈ C_k(h0) + grad_k(h0) . (h - h0) = grad_k(h0) . h + (C_k(h0) - grad_k(h0) . h0)

So the constraint is: grad_k(h0) . h + (C_k(h0) - grad_k(h0) . h0) <= t

This should be correct. The LP will give t >= min max C_k (since we're relaxing), so t is a lower bound on the true optimum.

Wait, no. We're minimizing t, and the constraints say C_k <= t. So t is an upper bound on each C_k. The LP minimizes t, so it finds the smallest t such that all linearized C_k <= t. Since the linearization underestimates C_k (for concave parts) or overestimates (for convex parts), the relationship between LP's t and the true optimal max C_k is not straightforward.

Actually, for bilinear functions, the linearization at h0 gives:
C_k(h) = C_k(h0) + grad_k(h0).(h-h0) + (h-h0)^T H_k (h-h0)

where H_k is the Hessian (which for bilinear functions is constant and indefinite). So the linearization can be above or below the true value depending on the direction of (h-h0).

This means the LP is NOT a valid relaxation in the usual sense. It's just a local approximation. So the LP solution might give t < true optimum (if the linearization underestimates) or t > true optimum (if it overestimates).

This is still useful as an optimization heuristic, but I can't use it as a certificate.

import numpy as np
from scipy.optimize import minimize, Bounds
import cvxpy as cp
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, lag, n, dx):
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag]) * dx
        g[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        g[:n - l] += (1.0 - h[l:]) * dx
        g[l:] += -h[:n - l] * dx
    return g


def resample(h_old, n_new):
    if len(h_old) == n_new:
        return h_old.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h_old)), h_old)


def smooth_max_obj(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    cs = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(cs)) / len(c)) / beta) * dx


def sum_constraint(x, target):
    return x.sum() - target


def sum_jac(x, target):
    return np.ones(x.shape)


def active_grad(h, n, dx, top_k, sharpness):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    center = n - 1
    top_idx = np.argsort(corr)[::-1][:top_k]
    vals = corr[top_idx]
    w = np.exp(sharpness * (vals - vals[0]))
    w /= w.sum()
    g = np.zeros(n)
    for idx, wt in zip(top_idx, w):
        lag = int(idx) - center
        g += wt * grad_lag(h, lag, n, dx)
    return g, corr, top_idx


def subgradient_opt(h, n, target, dx, max_iters, t0, budget, min_time, best_state):
    cur = h.copy()
    c5 = compute_c5(cur)
    if c5 < best_state[0]:
        best_state[0] = c5
        best_state[1] = cur.copy()
        best_state[2] = n
    v = np.zeros(n)
    mom = 0.8
    for it in range(max_iters):
        if time.time() - t0 > budget - min_time:
            break
        g, corr, _ = active_grad(cur, n, dx, 5, 300.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        step = 0.025 * (0.97 ** (it // 500))
        trial = project_h(cur - step * g + mom * v, target)
        c5t = compute_c5(trial)
        if c5t < c5:
            v = mom * v + (cur - trial)
            cur = trial
            c5 = c5t
            if c5 < best_state[0]:
                best_state[0] = c5
                best_state[1] = cur.copy()
                best_state[2] = n
        else:
            trial = project_h(cur - 0.5 * step * g + mom * v, target)
            c5t = compute_c5(trial)
            if c5t < c5:
                v = mom * v + (cur - trial)
                cur = trial
                c5 = c5t
                if c5 < best_state[0]:
                    best_state[0] = c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
            else:
                v *= 0.5
                if it % 300 == 0 and it > 0:
                    T = 0.002 * np.exp(-it / 5000.0)
                    trial = project_h(cur + np.random.normal(0, T, n), target)
                    if compute_c5(trial) < c5:
                        cur = trial
                        c5 = compute_c5(cur)
                        if c5 < best_state[0]:
                            best_state[0] = c5
                            best_state[1] = cur.copy()
                            best_state[2] = n
    return cur, c5


def lp_polish(h, n, target, dx, max_iters, t0, budget, min_time, best_state):
    cur = h.copy()
    c5 = compute_c5(cur)
    if c5 < best_state[0]:
        best_state[0] = c5
        best_state[1] = cur.copy()
        best_state[2] = n
    t_ub = c5 + 0.001
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > budget - min_time:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full") * dx
        center = n - 1
        top_k = 8
        top_idx = np.argsort(corr)[::-1][:top_k]
        x = cp.Variable(n)
        t = cp.Variable()
        cons = [x >= 0, x <= 1, cp.sum(x) == target, t <= t_ub]
        for idx in top_idx:
            lag = int(idx) - center
            g = grad_lag(cur, lag, n, dx)
            f0 = corr[idx]
            const = f0 - np.dot(g, cur)
            cons.append(cp.sum(g * x) + const <= t)
        prob = cp.Problem(cp.Minimize(t), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-10)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=10000, abs_tol=1e-12)
            except Exception:
                break
        if x.value is None:
            break
        h_lp = project_h(x.value, target)
        c5_lp = compute_c5(h_lp)

                cur = h_trial
                c5 = c5_trial
                t_ub = c5 + 0.0005
                stall = 0
                improved = True
                if c5 < best_state[0]:
                    best_state[0] = c5
                    best_state[1] = cur.copy()
                    best_state[2] = n

            t_ub = max(t_ub * 0.85, c5)

    return cur, c5
