        else:
            x_old = np.linspace(0, 2, n_cur)
            x_new = np.linspace(0, 2, n_pts)
            h_work = project_h(np.interp(x_new, x_old, best_h), target)
        c_work = compute_c5(h_work)
        if c_work < best_c5:
            best_h = h_work.copy()
            best_c5 = c_work
            n_cur = n_pts

        # Gradient descent with smooth max
        for beta in [25, 50, 100, 200, 400, 800, 1600, 3200, 6400]:
            if tl() < 6:
                break
            _target_store[0] = target
            _target_store[1] = beta
            bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
            for restart in range(2):
                if tl() < 4:
                    break
                try:
                    res = minimize(obj_sm_pen, h_work, method='L-BFGS-B',
                                   jac=grad_sm_pen, bounds=bounds,
                                   options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-10})
                    h_new = project_h(res.x, target)
                    c_new = compute_c5(h_new)
                    if c_new < c_work:
                        h_work = h_new
                        c_work = c_new
                except Exception:
                    pass
            # Projected gradient with backtracking
            for it in range(3000):
                if tl() < 3:
                    break
                _, grad = smooth_max_h_grad(h_work, beta)
                grad -= np.mean(grad)
                gn = np.linalg.norm(grad)
                if gn < 1e-13:
                    break
                grad /= gn
                step = 0.04
                found = False
                for _ in range(15):
                    h_t = project_h(h_work - step * grad, target)
                    c_t = compute_c5(h_t)
                    if c_t < c_work - 1e-16:
                        h_work = h_t
                        c_work = c_t
                        found = True
                        break
                    step *= 0.5
                if not found and it > 100:
                    break
                if not found and it % 400 == 399:
                    p = np.random.randn(n_pts) * 0.01
                    p -= np.mean(p)
                    h_t = project_h(h_work + p, target)
                    if compute_c5(h_t) < c_work:
                        h_work = h_t
                        c_work = compute_c5(h_work)
        if c_work < best_c5:
            best_h = h_work.copy()
            best_c5 = c_work
            n_cur = n_pts

    # Phase 2: Subgradient + leveling on active lags
    n_f = len(best_h)
    target_f = n_f / 2.0
    for it in range(40000):
        if tl() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_top = min(5, max(2, n_f // 25))
        top_idx = np.argsort(corr)[::-1][:n_top]
        lags = [int(i) - (n_f - 1) for i in top_idx]

        # Try leveling direction first
        d = leveling_direction(best_h, lags, n_top)
        if d is not None:
            for s in [0.03, 0.015, 0.008]:
                h_t = project_h(best_h - s * d, target_f)
                c_t = compute_c5(h_t)
                if c_t < best_c5:
                    best_h = h_t
                    best_c5 = c_t
                    break

        # Subgradient on top lags
        grad = subgrad_top(best_h, lags)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn > 1e-12:
            grad /= gn
            for s in [0.025, 0.012, 0.006, 0.003, 0.001]:
                h_t = project_h(best_h - s * grad, target_f)
                c_t = compute_c5(h_t)
                if c_t < best_c5:
                    best_h = h_t
                    best_c5 = c_t
                    break

        if it % 500 == 499:
            p = np.random.randn(n_f) * 0.005
            p -= np.mean(p)
            h_t = project_h(best_h + p, target_f)
            if compute_c5(h_t) < best_c5:
                best_h = h_t
                best_c5 = compute_c5(best_h)

    # Phase 3: Pairwise coordinate moves
    for it in range(100000):
        if tl() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top = np.argsort(corr)[::-1][:2]
        imp = False
        for idx in top:
            lag = int(idx) - (n_f - 1)
            if lag == 0:
                j = np.argmin(np.minimum(best_h, 1.0 - best_h))
                for d in [0.012, -0.012, 0.006, -0.006]:
                    h_t = best_h.copy()
                    h_t[j] += d
                    if 0 <= h_t[j] <= 1:
                        h_t = project_h(h_t, target_f)
                        if compute_c5(h_t) < best_c5:
                            best_h = h_t
                            best_c5 = compute_c5(best_h)
                            imp = True
                            break
            else:
                al = abs(lag)
                if al >= n_f:
                    continue
                nv = n_f - al
                for _ in range(6):
                    i = np.random.randint(nv)
                    j = i + al
                    for d in [0.018, -0.018, 0.009, -0.009, 0.025, -0.025]:
                        h_t = best_h.copy()
                        h_t[i] += d
                        h_t[j] -= d
                        if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                            if abs(h_t.sum() - target_f) > 1e-10:
                                h_t = project_h(h_t, target_f)
                            ct = compute_c5(h_t)
                            if ct < best_c5:
                                best_h = h_t
                                best_c5 = ct
                                imp = True
                                break
                    if imp:
                        break
            if imp:
                break
        if not imp and it % 800 == 799:
            p = np.random.randn(n_f) * 0.003
            p -= np.mean(p)
            h_t = project_h(best_h + p, target_f)
            if compute_c5(h_t) < best_c5:
                best_h = h_t
                best_c5 = compute_c5(best_h)

    # Phase 4: Final SLSQP polish
    if tl() > 12:
        n_p = len(best_h)
        _target_store[0] = n_p / 2.0
        _target_store[1] = 5000
        bounds_p = Bounds(np.zeros(n_p), np.ones(n_p))
        try:
            res = minimize(obj_sm, best_h, method='SLSQP', bounds=bounds_p,
                           constraints=[{'type': 'eq', 'fun': con_eq, 'jac': jcon_eq}],
                           options={'maxiter': 4000, 'ftol': 1e-18})
            h_n = project_h(res.x, n_p / 2.0)
            cn = compute_c5(h_n)
            if cn < best_c5:
                best_h = h_n
                best_c5 = cn
        except Exception:
            pass

    # One more subgradient sweep
    for it in range(3000):
        if tl() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top = np.argsort(corr)[::-1][:4]
        lags = [int(i) - (n_f - 1) for i in top]
        grad = subgrad_top(best_h, lags)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        grad /= gn
        for s in [0.015, 0.008, 0.004, 0.002]:
            h_t = project_h(best_h - s * grad, target_f)
            if compute_c5(h_t) < best_c5:
                best_h = h_t
                best_c5 = compute_c5(best_h)
                break

    return best_h, best_c5, len(best_h)
