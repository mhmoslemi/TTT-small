    deltas = [0.004, -0.004, 0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]
    for sw in range(n_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 0.5:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best - 1e-15:
                    cur = trial
                    best = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)
    n = best_n
    target = n / 2.0

    # Phase 1: L-BFGS-B warmup with moderate beta
    for beta in [30, 80]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 30:
            break
        cur = lbfgsb_opt(best_h, n, target, beta, 5.0, 8000, t0, budget_s)
        cc = compute_c5(cur)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 2: Block coordinate optimization
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        cur, cc = block_opt(best_h, n, target, 8, 12, t0, budget_s, beta=400, mu=15.0)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 3: L-BFGS-B with higher beta
    for beta in [200, 500]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 25:
            break
        cur = lbfgsb_opt(best_h, n, target, beta, 15.0, 10000, t0, budget_s)
        cc = compute_c5(cur)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 4: Projected gradient descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        cur_g, c5_g = proj_grad_desc(best_h, n, target, 50000, 0.003, t0, budget_s, beta=300)
        if c5_g < best_c5:
            best_h, best_c5 = cur_g, c5_g

    # Phase 5: More block optimization
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        cur, cc = block_opt(best_h, n, target, 4, 20, t0, budget_s, beta=600, mu=25.0)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 6: L-BFGS-B high beta refinement
    for beta in [800, 1500]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        cur = lbfgsb_opt(best_h, n, target, beta, 30.0, 8000, t0, budget_s)
        cc = compute_c5(cur)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 7: Random restarts
    remaining = budget_s - (time.time() - t0)
    for att in range(10):
        if budget_s - (time.time() - t0) < 12:
            break
        scale = 0.008 / (att + 1)
        hp = project_h(best_h + np.random.randn(n) * scale, target)
        hp = lbfgsb_opt(hp, n, target, 400, 20.0, 3000, t0, budget_s)
        circ = compute_c5(hp)
        if circ < best_c5:
            best_h, best_c5 = hp, circ

    # Phase 8: Coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = coord_polish(best_h, n, target, 40, t0, budget_s)

    # Phase 9: Final ultra-high-beta L-BFGS-B
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        cur = lbfgsb_opt(best_h, n, target, 3000, 50.0, 5000, t0, budget_s)
        cc = compute_c5(cur)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Final coordinate polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        best_h, best_c5 = coord_polish(best_h, n, target, 20, t0, budget_s)

    return best_h, best_c5, best_n
