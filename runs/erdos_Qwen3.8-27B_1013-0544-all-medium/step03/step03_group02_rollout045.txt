import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sml_obj(x, beta, dx, target, pen):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sml = m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -60, 60)))) / beta
    sv = np.sum(x) - target
    return sml * dx + pen * sv * sv

def sum_eq(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def do_lbfgsb(h0, n, target, beta, bounds, maxiter, pen):
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n, target, pen),
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': maxiter*n})
        return project_h(r.x, target)
    except Exception:
        return h0

def do_slsqp(h0, n, target, beta, bounds, maxiter, pen=0.0):
    if pen > 0:
        def obj(x):
            return sml_obj(x, beta, 2.0/n, target, pen)
        cons = []
    else:
        def obj(x):
            return sml_obj(x, beta, 2.0/n, target, 0.0)
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_line_search(cur, target, df, t0, budget, max_sweeps):
    n = len(cur)
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            lo, hi = max(0.0, cur[j] - df*(1+sw)), min(1.0, cur[j] + df*(1+sw))
            if lo >= hi:
                continue
            best_val = cur[j]
            best_c = best_c5
            for _ in range(20):
                if hi - lo < 1e-10:
                    break
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                t1 = cur.copy(); t1[j] = m1; t1 = project_h(t1, target)
                t2 = cur.copy(); t2[j] = m2; t2 = project_h(t2, target)
                c1 = compute_c5(t1)
                c2 = compute_c5(t2)
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
                if c1 < best_c - 1e-14:
                    best_c = c1; best_val = m1
                if c2 < best_c - 1e-14:
                    best_c = c2; best_val = m2
            if best_c < best_c5 - 1e-14:
                cur[j] = best_val
                cur = project_h(cur, target)
                best_c5 = best_c
                improved = True
        if not improved:
            df *= 0.5
            if df < 1e-6:
                break
    return cur, best_c5

def swap_descent(cur, target, n, t0, budget, max_sweeps=5):
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 3:
            break
        idx_high = np.where(cur > 0.3)[0]
        idx_low = np.where(cur < 0.7)[0]
        np.random.shuffle(idx_high)
        np.random.shuffle(idx_low)
        for i in idx_high[:80]:
            if budget - (time.time() - t0) < 1:
                break
            for j in idx_low[:80]:
                if i == j:
                    continue
                avail = min(cur[i], 1.0 - cur[j])
                if avail < 1e-6:
                    continue
                best_d = 0.0
                best_c = best_c5
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    d = avail * frac
                    t = cur.copy()
                    t[i] -= d
                    t[j] += d
                    c = compute_c5(t)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_d = d
                if best_d > 0:
                    cur[i] -= best_d
                    cur[j] += best_d
                    best_c5 = best_c
    return cur, best_c5

def block_perturb(cur, n, strength):
    h = cur.copy()
    nblocks = np.random.randint(2, 6)
    for _ in range(nblocks):
        start = np.random.randint(0, n - 10)
        length = np.random.randint(5, 20)
        shift = np.random.choice([-1, 1]) * (np.random.randint(1, 5) * strength)
        seg = h[start:start+length].copy()
        h[start:start+length] = seg + shift
    return project_h(h, n / 2.0)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)
