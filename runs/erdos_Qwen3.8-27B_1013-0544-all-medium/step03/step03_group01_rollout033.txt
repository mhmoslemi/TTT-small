```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14 * n:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h

    # Δc_k = δ * D_k where D_k = h[j+k] - h[i+k] - h[i-k] + h[j-k]

        val_jmk = h[j-k] if 0 <= j-k < n else 0.0
        val_imk = h[i-k] if 0 <= i-k < n else 0.0

    lo = max(-h[i], h[j] - 1)
    hi = min(1 - h[i], h[j])
    
    if lo >= hi:

    # Minimize max_idx (corr[idx] + δ * D[idx]) over δ ∈ [lo, hi]

    a, b = lo, hi

        f1 = np.max(corr + c1 * D)
        f2 = np.max(corr + c2 * D)
        if f1 < f2:
            b = c2
        else:
            a = c1
    opt_d = (a + b) / 2
```

```python
def precompute_V(h, n):
    """V[i][s] = h[i-s] + h[i+s] for shift s, with 0 padding. Shape: (n, 2n-1)"""
    V = np.zeros((n, 2*n-1))
    for i in range(n):
        for s in range(-(n-1), n):
            m = n - 1 + s
            val = 0.0
            if 0 <= i-s < n: val += h[i-s]
            if 0 <= i+s < n: val += h[i+s]

    return V
```

```python
def compute_V(h, n):
    # Create extended h with zeros: length n + 2*(n-1) = 3n-2
    # ext[p] = h[p-(n-1)] if 0 <= p-(n-1) < n, else 0
    ext = np.zeros(3*n - 2)
    ext[n-1:2*n-1] = h  # positions n-1 to 2n-2 correspond to h[0] to h[n-1]
    
    V = np.zeros((n, 2*n-1))
    for i in range(n):
        # h[i-s] for s = m-(n-1): position in ext = (i + n - 1) - (m - (n-1)) ... 
        # actually: position in original = i - s = i - (m - (n-1)) = i + n - 1 - m
        # position in ext = (i + n - 1 - m) + (n-1) = i + 2n - 2 - m
        # h[i+s]: position in original = i + s = i + m - (n-1) = i + m - n + 1
        # position in ext = (i + m - n + 1) + (n-1) = i + m
        
        p1 = i + 2*n - 2 - np.arange(2*n-1)  # for h[i-s]
        p2 = i + np.arange(2*n-1)              # for h[i+s]
        
        V[i, :] = ext[p1] + ext[p2]
    
    return V
```
