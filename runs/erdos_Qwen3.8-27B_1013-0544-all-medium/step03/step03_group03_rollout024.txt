import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    """Compute smooth approximation of max(x)"""
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag, n):
    """Gradient of correlation value at given lag (corrected)."""
    grad = np.zeros(n)
    k = abs(lag)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        # j in [0, k-1]
        if k > 0:
            grad[0:k] = -h[k:2*k]
        # j in [k, n-1-k]
        if k < n - k:
            j = np.arange(k, n-k)
            grad[j] = 1.0 - h[j-k] - h[j+k]
        # j in [n-k, n-1]
        j = np.arange(n-k, n)
        grad[j] = 1.0 - h[j-k]
    else:
        # j in [0, k-1]
        j = np.arange(k)
        grad[j] = 1.0 - h[j+k]
        # j in [k, n-1-k]
        if k < n - k:
            j = np.arange(k, n-k)
            grad[j] = 1.0 - h[j-k] - h[j+k]
        # j in [n-k, n-1]
        j = np.arange(n-k, n)
        grad[j] = -h[j-k]
    return grad

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h):
    return np.ones(len(h))

def nesterov_opt(h0, n, k_top, beta_w, lr0, decay, mom, max_it, budget_fn, min_time=30):
    """Nesterov accelerated projected subgradient."""
    target_sum = n / 2.0
    x = h0.copy()
    x_prev = h0.copy()
    best = h0.copy()
    best_c = compute_c5(best)
    no_improve = 0
    
    for it in range(max_it):
        if budget_fn() < min_time:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        top = np.argsort(corr)[-k_top:]
        w = np.exp(beta_w * (corr[top] - corr[top].max()))
        w /= w.sum()
        
        g = np.zeros(n)
        for j, idx in enumerate(top):
            lag = idx - (n - 1)
            g += w[j] * grad_lag(x, lag, n)
        
        lr = lr0 / (1.0 + decay * it)
        y = x + mom * (x - x_prev)
        x_new = project_feasible(y - lr * g, target_sum)
        x_prev = x
        x = x_new
        
        if (it + 1) % 5 == 0:
            c = compute_c5(x)
            if c < best_c - 1e-7:
                best_c = c
                best = x.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 200:
                break
    
    return best, best_c

def slsqp_refine(h0, n, betas, nudges, budget_fn, min_time=20):
    """SLSQP refinement with smooth max."""
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'args': (target_sum,)}]
    # Add jac
    cons[0]['jac'] = lambda h: np.ones(n)
    
    best = h0.copy()
    best_c = compute_c5(best)
    
    for beta in betas:
        for nu in nudges:
            if budget_fn() < min_time:
                break
            x0 = best.copy() + np.random.randn(n) * nu if nu > 0 else best.copy()
            x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(lambda h: obj_smooth(h, dx, beta), x0,
                              method='SLSQP', bounds=bounds,
                              constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c = compute_c5(h_new)
                if c < best_c:
                    best_c = c
                    best = h_new.copy()
            except Exception:
                pass
    
    return best, best_c
