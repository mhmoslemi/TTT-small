def fast_corr_full(h):
    """Compute correlation of h with 1-h in full mode using FFT."""
    n = len(h)
    g = 1.0 - h
    N = 2 ** int(np.ceil(np.log2(2 * n - 1)))
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    corr = np.fft.irfft(H * np.conj(G), N)[:2*n-1]
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_corr_full(h)
    return float(np.max(corr) * dx)

def smoothmax_and_grad(h, beta, top_k=20):
    n = len(h)
    dx = 2.0 / n
    corr = fast_corr_full(h)
    m = len(corr)
    
    if top_k is not None and top_k < m:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(m)
    
    c_sel = corr[top_idx]
    cmax = np.max(c_sel)
    exps = np.exp(beta * (c_sel - cmax))
    sw = exps.sum()
    w = exps / sw
    val = (cmax + np.log(sw) / beta) * dx
    
    # Gradient: for each j, grad[j] = sum_i w_i * dcorr_i/dh_j * dx
    # dcorr_i/dh_j = (1-h[j+i-(n-1)]) - h[j-i+(n-1)] (when indices valid)
    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        wi = w[idx_i]
        if wi < 1e-15:
            continue
        i = top_idx[idx_i]
        s = i - (n - 1)  # shift
        # First term: sum over j where j+s is valid: (1-h[j+s])
        # j+s in [0, n-1] means j in [-s, n-1-s]
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wi * (1.0 - h[j_lo+s:j_hi+s])
        # Second term: sum over j where j-s is valid: -h[j-s]
        # j-s in [0, n-1] means j in [s, s+n-1]
        j_lo2 = max(0, s)
        j_hi2 = min(n, s + n)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] += wi * (-h[j_lo2-s:j_hi2-s])
    grad *= dx
    return val, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_optimize(h_init, beta, top_k, penalty_mu, maxiter):
    n = len(h_init)
    target = n / 2.0
    
    def obj(x):
        f, _ = smoothmax_and_grad(x, beta, top_k)
        penalty = penalty_mu * (x.sum() - target) ** 2
        return f + penalty

        g = g + 2 * penalty_mu * (x.sum() - target) * np.ones(n)
        return g
    
    res = minimize(obj, h_init, jac=objg, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:

        if free.sum() == 0:

import numpy as np
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def cd_optimize(h, n_sweeps, mu, dx, seed=0, tol=1e-12):
    n = len(h)
    target = n / 2.0
    rng = np.random.RandomState(seed)
    h = project_feasible(h, target)
    best_c = compute_c5(h)
    best_h = h.copy()
    gr = (np.sqrt(5) - 1) / 2
    m = 2 * n - 1
    s_vals = np.arange(m) - (n - 1)
    for sweep in range(n_sweeps):
        h_old = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        for j in order:
            va = (j + s_vals >= 0) & (j + s_vals < n)
            vb = (j - s_vals >= 0) & (j - s_vals < n)
            b = np.zeros(m)
            b[va] = 1.0 - h[j + s_vals[va]]
            b[vb] -= h[j - s_vals[vb]]
            A = corr - h[j] * b
            S = h.sum() - h[j]
            def f_obj(t):
                return np.max(A + b * t) * dx + mu * (S + t - target) ** 2
            lo, hi = 0.0, 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            for _ in range(50):
                if f_obj(c1) < f_obj(c2):
                    hi = c2
                else:
                    lo = c1
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
            new_val = (lo + hi) / 2.0
            delta = new_val - h[j]
            if abs(delta) > 1e-15:
                corr += delta * b
                h[j] = new_val
        h = project_feasible(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if np.max(np.abs(h - h_old)) < tol:
            break
    return best_h, best_c

def basin_hop_cd(h_best, best_c5, n_hops, noise_base, noise_decay, mu, dx, seed, t0, budget):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget - (time.time() - t0) < 60:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_local, c_local = cd_optimize(h_pert, 40, mu, dx, seed=seed + hop)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()

    # Phase 2: CD with larger penalty (tighter constraint enforcement)

        h_interp = interp_to_n(h_best, n_try)

        h_best, best_c5 = basin_hop_cd(h_best, best_c5, 10, 0.008, 0.85,
