import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def coord_optimize(h, n, max_sweeps=200, time_limit=None, t0=None):
    """Coordinate descent: optimize each coordinate to minimize max correlation."""
    dx = 2.0 / n
    target_sum = n / 2.0
    num_lags = 2 * n - 1
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    h_rev = h[::-1]
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        improved = False
        for j in range(n):
            # Compute a vector
            a = np.zeros(num_lags)
            a[n-1-j : 2*n-1-j] += (1.0 - h)
            a[j : j+n] -= h_rev
            
            # c = corr - a * h[j]

            # Minimize f(x) = max(c + a*x) over x in [0,1]

            for _ in range(50):

        if c5 < best_c5:
            best_c5 = c5
        else:
            # If no improvement after projection, maybe we're stuck

                break
    
    return h, best_c5

3. Find the worst lag L* (argmax corr)
4. Try all 2-flips and find the one that most reduces corr_L* (or the max)
5. Apply the best flip
6. Repeat
