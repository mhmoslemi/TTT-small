import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_corr(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    cs = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(cs)) / len(corr)) / beta) * dx

def sml_obj(h_in, beta, dx):
    return smooth_max_corr(h_in, beta, dx)

def sum_eq(h_in, target):
    return np.sum(h_in) - target

def sum_eq_jac(h_in, target):
    return np.ones(h_in.shape)

def correct_subgradient(h, top_k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        la = abs(lag)
        if la == 0:
            grad[:] += (1.0 - 2.0 * h)
        else:
            grad[:la] += -h[la:2*la]
            if n - 2 * la >= 1:
                grad[la:n-la] += (1.0 - h[:n-2*la]) - h[2*la:n]
            if la > 0 and n - 2 * la >= 0:
                grad[n-la:] += (1.0 - h[n-2*la:n-la])
    return grad / len(top_idx), corr

def nesterov_desc(cur, target, n, t0, budget, max_iter=8000, init_step=0.03):
    best = compute_c5(cur)
    y = cur.copy()
    m = np.zeros(n)
    beta_mom = 0.9
    step = init_step
    stal = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        t_iter = (it + 1) / (it + 2)
        m = beta_mom * m
        grad, _ = correct_subgradient(y, top_k=5)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        m -= grad / gn
        trial = project_h(y - step * m, target)
        c_try = compute_c5(trial)
        if c_try < best - 1e-15:
            cur = trial
            best = c_try
            stal = 0
            step = min(step * 1.05, 0.1)
        else:
            stal += 1
            step *= 0.7
            if stal > 100:
                step = init_step
                m = np.zeros(n)
                stal = 0
        y = trial + t_iter * (trial - cur)
        y = project_h(y, target)
        if step < 1e-6:
            break
    return cur, best

def block_coord_descent(cur, target, n, t0, budget, block_size=4, n_sweeps=80):
    best_c5 = compute_c5(cur)
    deltas = (0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.01, -0.01)
    for sw in range(n_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for start in range(0, n, block_size):
            if budget - (time.time() - t0) < 1:
                break
            end = min(start + block_size, n)
            for d in deltas:
                trial = cur.copy()
                trial[start:end] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 10:
            break
    return cur, best_c5

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.success or (hasattr(r, 'x') and r.x is not None):
            return project_h(r.x, target)
    except Exception:
        pass
    return h0

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new),
                     np.linspace(0.0, 2.0, len(h)), h)

def structured_perturb(h, target, n, rng):
    """Generate diverse perturbation starting points"""
    choice = rng.randint(4)
    if choice == 0:
        # Power transformation
        p = rng.choice([0.5, 0.7, 1.5, 2.0, 3.0])
        h_p = h.copy()
        h_p = np.clip(np.sign(h_p - 0.5) * np.abs(h_p - 0.5 + 0.5) ** p, 0, 1)
        h_p = project_h(h_p, target)
    elif choice == 1:
        # Low-frequency sine perturbation
        k = rng.randint(1, 5)
        amp = rng.uniform(0.05, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n)
        h_p = h + amp * np.sin(2 * np.pi * k * x / 2.0 + phase)
        h_p = project_h(h_p, target)
    elif choice == 2:

        t = rng.uniform(0.1, 0.4)
        h_p = (1 - t) * h + t * np.full(n, 1.0 / n * target)
        h_p = project_h(h_p, target)
    else:
        # Shift perturbation
        shift = rng.randint(1, max(2, n // 20))
        h_p = np.roll(h, shift)
        h_p = project_h(h_p, target)
    return h_p
