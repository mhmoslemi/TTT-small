```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    c_shifted = c - m
    exp_c = np.exp(c_shifted)
    sum_exp = exp_c.sum()
    obj = m + np.log(sum_exp) / alpha
    weights = exp_c / sum_exp
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        w = weights[k_idx] * dx
        if w < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad[:] += w * (1.0 - 2.0*h)
            else:
                grad[lag:] += w * (1.0 - h[:n-lag])
                grad[:n-lag] -= w * h[lag:]
        else:
            j = -lag
            if j == 0:
                grad[:] += w * (1.0 - 2.0*h)
            else:
                grad[:n-j] += w * (1.0 - h[j:])
                grad[j:] -= w * h[:n-j]
    return float(obj), grad

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    def obj_wrapper(x):
        val, _ = smooth_obj_and_grad(x, n, alpha)
        return val
    def grad_wrapper(x):
        _, g = smooth_obj_and_grad(x, n, alpha)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_wrapper, h, jac=grad_wrapper, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgs_optimize(h, n, alpha, maxiter, ftol):
    def obj_wrapper(x):
        val, _ = smooth_obj_and_grad(x, n, alpha)
        return val
    def grad_wrapper(x):
        _, g = smooth_obj_and_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_wrapper, h, jac=grad_wrapper, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local_opt(h, n, deadline, deltas):
    """Local optimization using pair moves."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s*d
                    hj = h[j] - s*d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i], h[j] = hi - s*d, hj + s*d
    return h, best_c

def subgrad_refine(h, n, deadline):
    """Subgradient descent targeting the active shift."""
    best_c = compute_c5(h, n)
    alpha = 0.1
    for _ in range(5000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        # Compute subgradient
        grad = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                grad[lag:] += (1.0 - h[:n-lag])
                grad[:n-lag] -= h[lag:]
        else:
            j = -lag
            if j == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                grad[:n-j] += (1.0 - h[j:])
                grad[j:] -= h[:n-j]
        # Step
        step = alpha / (np.linalg.norm(grad) + 1e-10)
        h_new = project_feasible(h - step * grad * (2.0/n), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha *= 0.8
            if alpha < 1e-6:
                break
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def sum_con_fun(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(len(x))

def smooth_obj_wrap(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    cs = c - m
    return float(m + np.log(np.sum(np.exp(cs))) / alpha)

def smooth_grad_wrap(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    cs = c - m
    exp_c = np.exp(cs)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - x[:n - lag])
            grad[:n - lag] -= wk * x[lag:]
        else:
            j = -lag
            grad[:n - j] += wk * (1.0 - x[j:])
            grad[j:] -= wk * x[:n - j]
    return grad

def do_lbfgs(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_wrap, h, args=(n, alpha),
                      jac=lambda x: smooth_grad_wrap(x, n, alpha),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def do_slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj_wrap, h, args=(n, alpha),
                      jac=lambda x: smooth_grad_wrap(x, n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    alpha = 0.05
    for _ in range(10000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            j = -lag
            grad[:n - j] += (1.0 - h[j:])
            grad[j:] -= h[:n - j]
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        step = alpha / (gn * gn)  # Polyak-like step
        h_new = project_feasible(h - step * grad * (2.0 / n), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
