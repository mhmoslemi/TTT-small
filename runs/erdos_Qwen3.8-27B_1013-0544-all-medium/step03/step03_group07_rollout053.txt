                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += wk * (1.0 - h[ms - s])
            mp = np.arange(0, n - s)
            grad[mp] += wk * (-h[mp + s])
        elif s < 0:
            t = -s
            ms = np.arange(0, n - t)
            grad[ms] += wk * (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += wk * (-h[mp - t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

    _n_global = len(h)
    _beta_global = beta
    target = _n_global / 2.0
    bounds = Bounds(np.zeros(_n_global), np.ones(_n_global))
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]

        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_topk(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            if s >= 0:
                ms = np.arange(s, n)
                grad[ms] += (1.0 - y[ms - s])
                mp = np.arange(0, n - s)
                grad[mp] += (-y[mp + s])
            elif s < 0:
                tt = -s
                ms = np.arange(0, n - tt)
                grad[ms] += (1.0 - y[ms + tt])
                mp = np.arange(tt, n)
                grad[mp] += (-y[mp - tt])
            else:
                grad[:] += (1.0 - 2.0 * y)
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v
