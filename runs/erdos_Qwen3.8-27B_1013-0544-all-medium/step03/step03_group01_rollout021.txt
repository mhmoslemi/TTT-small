def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (diff / t) * h
    h = np.clip(h, 0.0, 1.0)
    # Fix exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(h)
        if diff > 0:
            for i in reversed(idx):
                h[i] = min(1.0, h[i] + diff)
                diff -= (min(1.0, h[i]) - h[idx[idx > i].min() if (idx > i).any() else i])

def compute_a(h, n, j):
    a = np.zeros(2 * n - 1)
    lo = max(0, j - n + 1)
    hi = min(n - 1, j)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        a[ks + n - 1] += 1.0 - h[j - ks]
    lo2 = max(-(n - 1), -j)
    hi2 = min(n - 1, n - 1 - j)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        a[ks + n - 1] -= h[j + ks]
    return a

def opt_v_1d(c, a, v):

    if u_lo >= u_hi - 1e-15:
        return v
    f_lo = c + u_lo * a
    f_hi = c + u_hi * a
    k_lo = np.argmax(f_lo)
    k_hi = np.argmax(f_hi)
    if k_lo == k_hi:
        if a[k_lo] > 1e-14:
            return 0.0
        elif a[k_lo] < -1e-14:
            return 1.0
        return v
    denom = a[k_lo] - a[k_hi]
    if abs(denom) < 1e-14:
        return 0.0 if f_lo.max() <= f_hi.max() else 1.0
    u_star = (c[k_hi] - c[k_lo]) / denom
    v_star = v + u_star
    return float(np.clip(v_star, 0.0, 1.0))

def coord_sweep(h, n, c, order):
    for j in order:
        v = h[j]
        a = compute_a(h, n, j)
        v_new = opt_v_1d(c, a, v)
        if abs(v_new - v) > 1e-14:
            c += (v_new - v) * a
            h[j] = v_new
    return h, c

def c5_from_correlation(c, n):
    return float(np.max(c) * 2.0 / n)

def coord_descent(h, n, deadline, rng, max_sweeps=5000):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    best_max = c.max()
    stall = 0

            break

            order = np.arange(n)
        elif sweep % 7 == 1:
            order = np.arange(n - 1, -1, -1)
        else:
            order = rng.permutation(n)
        h, c = coord_sweep(h, n, c, order)
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full')
        cur = c.max()
        if cur < best_max - 1e-13:
            best_max = cur
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
    return h, c, best_max

def pair_refine(h, n, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    best = c.max()
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

                        best = c.max()
