        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr(h, weights, n):
    grad = np.zeros(n)
    center = n - 1
    for t in range(2 * n - 1):
        w = weights[t]
        if w < 1e-10:
            continue
        lag = t - center
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            l = abs(lag)
            if l < n - l:
                grad[:l] += w * (1.0 - h[l:])
                grad[l:n-l] += w * (1.0 - h[2*l:] - h[:n-2*l])
                grad[n-l:] += w * (-h[:l])
            else:
                grad[:n-l] += w * (1.0 - h[l:])
                grad[n-l:] += w * (-h[:l])
    return grad

def grad_descent(h, target, n, max_iters, time_fn, min_time, beta):
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur_c5
    best_h = cur.copy()
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        weights = compute_weights(corr, beta)
        grad = grad_corr(cur, weights, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        step = 0.05 / (1.0 + 0.004 * it)
        improved = False
        for _ in range(20):
            trial = project_h(cur - step * grad, target)
            c5_t = compute_c5(trial)
            if c5_t < cur_c5 - 1e-15:
                cur = trial
                cur_c5 = c5_t
                improved = True
                if cur_c5 < best:
                    best = cur_c5
                    best_h = cur.copy()
                break
            step *= 0.5
            if step < 1e-10:
                break
        if not improved:
            T = 0.004 * np.exp(-it / 800.0)
            noise = np.random.normal(0, T, n)
            trial = project_h(cur + noise, target)
            c5_t = compute_c5(trial)
            if c5_t < cur_c5 + 3.0 * T:
                cur = trial
                cur_c5 = c5_t
                if cur_c5 < best:
                    best = c5_t
                    best_h = cur.copy()
    return best_h, best

def single_lag_refine(h, target, max_iters, time_fn, min_time):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        worst_t = int(np.argmax(corr))
        weights = np.zeros(2 * n - 1)
        weights[worst_t] = 1.0
        grad = grad_corr(cur, weights, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for step in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            trial = project_h(cur - step * grad, target)
            c5_t = compute_c5(trial)
            if c5_t < cur_c5 - 1e-14:
                cur = trial
                cur_c5 = c5_t
                improved = True
                break
        if not improved:
            break
    return cur, cur_c5

def pair_swap_refine(h, target, time_fn, min_time):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for _ in range(500):
        if time_fn() < min_time:
            break
        improved = False
        for _ in range(300):
            i, j = np.random.choice(n, 2, replace=False)
            for delta in [0.03, 0.07, 0.12]:
                if 0 <= cur[i] - delta and cur[j] + delta <= 1:
                    trial = cur.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    c5_t = compute_c5(trial)
                    if c5_t < cur_c5 - 1e-13:
                        cur = trial
                        cur_c5 = c5_t
                        improved = True
                        break
                elif 0 <= cur[j] - delta and cur[i] + delta <= 1:
                    trial = cur.copy()
                    trial[j] -= delta
                    trial[i] += delta
                    c5_t = compute_c5(trial)
                    if c5_t < cur_c5 - 1e-13:
                        cur = trial
                        cur_c5 = c5_t
                        improved = True
                        break
            if improved:
                break
        if not improved:
            break
    return cur, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 200
        h_parent = thue_morse(n_parent)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    n_list = sorted(set([n_parent, 120, 160, 200, 250, 320]))
    n_list = [n for n in n_list if 80 <= n <= 350]

    for n in n_list:
        if time_left() < 40:
            break
        target = n / 2.0

        inits = []
        # Resample parent
        if n_parent != n:
            x_old = np.linspace(0, 2, n_parent)
            x_new = np.linspace(0, 2, n)
            h1 = np.interp(x_new, x_old, h_parent)
        else:
            h1 = h_parent.copy()
        inits.append(project_h(h1, target))
        # Thue-Morse
        inits.append(thue_morse(n))
        # Best so far resampled
        if n != len(best_h):
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n)
            h3 = np.interp(x_new, x_old, best_h)
        else:
            h3 = best_h.copy()
        inits.append(project_h(h3, target))

        for h_init in inits:
            if time_left() < 20:
                break
            cur = project_h(h_init.copy(), target)
            cur_c5 = compute_c5(cur)
            # Phase 1: equalize with low beta
            cur, cur_c5 = grad_descent(cur, target, n, 3000, time_left, 15, beta=30)
            # Phase 2: focus with medium beta
            cur, cur_c5 = grad_descent(cur, target, n, 4000, time_left, 10, beta=150)
            # Phase 3: single lag refine
            cur, cur_c5 = single_lag_refine(cur, target, 5000, time_left, 5)
            # Phase 4: pair swaps
            cur, cur_c5 = pair_swap_refine(cur, target, time_left, 3)
            # Phase 5: final single lag
            cur, cur_c5 = single_lag_refine(cur, target, 3000, time_left, 3)

            if cur_c5 < best_c5:
                best_h = cur.copy()
                best_c5 = cur_c5

    # Final polish on best solution
    n_f = len(best_h)
    target_f = n_f / 2.0
    if time_left() > 5:
        best_h, best_c5 = grad_descent(best_h, target_f, n_f, 3000, time_left, 4, beta=300)
    if time_left() > 3:
        best_h, best_c5 = single_lag_refine(best_h, target_f, 20000, time_left, 2)
    if time_left() > 2:
        best_h, best_c5 = pair_swap_refine(best_h, target_f, time_left, 1)
    if time_left() > 1:
        best_h, best_c5 = single_lag_refine(best_h, target_f, 20000, time_left, 0.5)

    return best_h, best_c5, len(best_h)
