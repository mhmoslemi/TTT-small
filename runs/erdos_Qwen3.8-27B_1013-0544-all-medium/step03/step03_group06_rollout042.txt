        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

def subgrad_c5(h, lags):

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(c, beta):
    m = float(np.max(c))
    shifted = np.clip(beta * (c.astype(float) - m), -400, 400)
    return m + float(np.log(np.sum(np.exp(shifted)) / len(c))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_c5(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for k in lags:
        if k == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < k < n:
            grad[k:] += (1.0 - h[:-k])
            grad[:n-k] -= h[k:]
        elif -n < k < 0:
            m = -k
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    for beta in [200, 500, 1000, 2000, 5000, 10000, 20000, 50000]:
        if time_left() < 15:
            break

            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, float(beta)) * dx
        def con_eq(h_in):
            return np.sum(h_in) - target_sum
        def con_jac(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_b, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                           options={'maxiter': 3000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    for n_sa in [50, 70, 90]:
        if time_left() < 60:
            break
        target_sa = n_sa / 2.0
        h_sa = project_h(np.clip(np.random.rand(n_sa), 0.05, 0.95), target_sa)
        c5_cur = compute_c5(h_sa)
        best_sa = h_sa.copy()
        best_sa_c5 = c5_cur
