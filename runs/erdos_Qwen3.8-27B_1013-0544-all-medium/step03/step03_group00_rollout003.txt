    h = project_feasible(h0.copy(), target_sum)
    h_prev = h.copy()
    for k in range(n_iter):
        val, grad = smoothmax_val_grad(h, beta, top_k)
        # Backtracking line search
        if lr is None:
            lr_k = 1.0 / (beta * 4.0 / n + 1e-8)
        else:

def slsqp_solve(h, beta, maxiter, top_k=None, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

def interp_to_n(h, n_new):

    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    lo, hi = -h.max(), 1.0 - h.min()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            if n - s > 0:
                grad[0:n-s] += wk * (1.0 - h[s:n])
                grad[s:n] -= wk * h[0:n-s]
        else:
            t = -s
            if t < n:
                grad[t:n] += wk * (1.0 - h[0:n-t])
                grad[0:n-t] -= wk * h[t:n]
    grad *= dx
    return val, grad

def proj_grad(h0, target_sum, beta, n_iter, top_k=15, momentum=0.8, lr_init=None):
    n = len(h0)
    h = project_feasible(h0.copy(), target_sum)
    h_prev = h.copy()
    if lr_init is None:
        lr = 2.0 / (beta * 4.0 / n)
    else:
        lr = lr_init
    for k in range(n_iter):
        val, grad = smoothmax_val_grad(h, beta, top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        h_try = project_feasible(h - lr * grad, target_sum)
        val_try, _ = smoothmax_val_grad(h_try, beta, top_k)
        attempts = 0
        while val_try > val - 1e-15 and attempts < 15:
            lr *= 0.6
            h_try = project_feasible(h - lr * grad, target_sum)
            val_try, _ = smoothmax_val_grad(h_try, beta, top_k)
            attempts += 1
            if lr < 1e-10:
                break
        h_new = project_feasible(h_try + momentum * (h_try - h_prev), target_sum)
        val_new, _ = smoothmax_val_grad(h_new, beta, top_k)
        if val_new > val:
            h_new = h_try
        h_prev = h
        h = h_new
    return h

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_new / 2.0)

def make_perturbation(h, target_sum, rng, style):
    n = len(h)
    h_p = h.copy()

        h_p += rng.randn(n) * 0.015

    else:
        h_p = h * np.exp(rng.randn(n) * 0.02)
    return project_feasible(h_p, target_sum)
