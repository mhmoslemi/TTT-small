    return project_feasible(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)

    best_c5 = compute_c5(h_best)

    # Phase 1: SLSQP with increasing beta at starting resolution
    n = len(h_best)
    for beta in [100, 250, 500, 1000]:
        if rem() < 25:
            break
        h_new = slsqp_opt(h_best, beta, 500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Binarization + re-optimization
    for tau in [0.3, 0.15, 0.08, 0.04]:
        if rem() < 25:
            break
        h_bin = binarize_smooth(h_best, tau)
        for beta in [300, 700]:
            if rem() < 15:
                break
            h_new = slsqp_opt(h_bin, beta, 400)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_bin = h_best.copy()

    # Phase 3: Structured sinusoidal perturbations
    n = len(h_best)
    for freq in range(2, 7):
        if rem() < 20:
            break
        for ph in [0.0, 0.7854, 1.5708, 2.356]:
            if rem() < 15:
                break
            pert = 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
            h_pert = project_feasible(h_best + pert, n / 2.0)
            h_new = slsqp_opt(h_pert, 500, 400)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 4: Higher resolution
    for n_try in [600, 800, 1000]:
        if rem() < 100:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [100, 300, 600, 1200]:
            if rem() < 35:
                break
            h_new = slsqp_opt(h_interp, beta, 400)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()

    # Phase 5: Binarization at higher resolution
    n = len(h_best)
    for tau in [0.2, 0.1, 0.05, 0.02]:
        if rem() < 35:
            break
        h_bin = binarize_smooth(h_best, tau)
        h_new = slsqp_opt(h_bin, 600, 500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 6: Nesterov smoothmax (broadcast gradient)
    if rem() > 80:
        h_nest = slsqp_opt(h_best, 200, 300)
        c0 = compute_c5(h_nest)
        h_ng, c_ng = nesterov_smoothmax(h_nest, 8000, 0.03, 0.99985, 200)
        if c_ng < best_c5:
            h_best, best_c5 = h_ng, c_ng
        if rem() > 30:
            h_new = slsqp_opt(h_best, 800, 500)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 7: m-sequence initialization + optimization
    if rem() > 60:
        n_cur = len(h_best)
        h_mseq = gen_mseq_init(n_cur, seed + 1)
        h_new = slsqp_opt(h_mseq, 300, 500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
            h_new = slsqp_opt(h_best, 800, 500)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 8: Interpolate to even higher resolution if time allows
    if rem() > 120:
        n_try = len(h_best) + 400
        h_interp = interp_to_n(h_best, n_try)
        for beta in [200, 500, 1000]:
            if rem() < 40:
                break
            h_new = slsqp_opt(h_interp, beta, 400)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()

    # Phase 9: Final high-beta polish
    if rem() > 25:
        h_new = slsqp_opt(h_best, 1500, 600)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
