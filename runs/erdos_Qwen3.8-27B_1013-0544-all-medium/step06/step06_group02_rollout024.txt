            if tot < 1e-14:

    """Local search on binary patterns. h_bin is 0/1 array with sum n/2."""

    h = h_bin.copy().astype(float)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:

        improved = False
        # Random order of positions
        order = np.random.permutation(n)
        for i in order:
            if h[i] != 1:  # only consider flipping 1s

            for j in order:

                    improved = True
                    break
                else:
                    h[i] = 1; h[j] = 0
            if improved:
                break
        if not improved:
            break
    return h, best_c
