where ∂c_k/∂h[p] = (1-h[p+d_k])I[0≤p+d_k<n] - h[p-d_k]I[0≤p-d_k<n]

So m_k = (1-h[p+d_k])I[0≤p+d_k<n] - h[p-d_k]I[0≤p-d_k<n] - (1-h[q+d_k])I[0≤q+d_k<n] + h[q-d_k]I[0≤q-d_k<n]

And we want to minimize max_k (c_k(0) + m_k * δ) over δ in the feasible range.

This is a 1D convex optimization problem. The minimum of max of linear functions occurs at a breakpoint, which is where two lines intersect. We can find it by:
- Computing all breakpoints: δ* = (c_j(0) - c_i(0)) / (m_i - m_j) for all pairs (i,j) with m_i ≠ m_j
- Evaluating f at each breakpoint and at the endpoints
- Taking the minimum

But there are O(K²) breakpoints which is O(n²) for K=2n-1. That's manageable for n=300 (about 90000 evaluations per pair step).

Actually, a simpler approach: the minimum of max of linear functions can be found by binary search on the value t:
- Is there a δ such that c_k(0) + m_k * δ ≤ t for all k?
- This is: c_k(0) - t ≤ -m_k * δ for all k
- If m_k > 0: δ ≤ (t - c_k(0)) / m_k
- If m_k < 0: δ ≥ (t - c_k(0)) / m_k
- If m_k = 0: need c_k(0) ≤ t

So for a given t, the feasible δ range is:
[max over k with m_k<0 of (t-c_k(0))/m_k, lower_bound] ∩ [upper_bound, min over k with m_k>0 of (t-c_k(0))/m_k]

We can binary search on t to find the minimum. Each check is O(n). Total: O(n log p) where p is the precision.

But actually, there's an even simpler approach for 1D:
The function f(δ) = max_k (m_k * δ + c_k(0)) is convex and piecewise linear. Its minimum is unique (unless it's flat). We can find it by:
1. The leftmost breakpoint where the slope changes from negative to positive
2. The slope of f at δ is m_{k*(δ)} where k*(δ) = argmax_k (m_k * δ + c_k(0))
3. As δ increases, the slope is non-decreasing (since f is convex)

So we can do a ternary search or binary search on the slope.

Actually, the simplest efficient approach:
1. Compute the upper envelope of the lines y = m_k * δ + c_k(0)
2. The minimum of the upper envelope is at a vertex of the upper envelope
3. We can find the upper envelope in O(K log K) using a convex hull trick

But for n=300, K=599, so O(K²) = ~360000 is fine for a single evaluation. We can just compute all pairwise intersections and check which ones are on the upper envelope.

Actually, even simpler: since we just need the minimum value, we can do a golden section search or ternary search on the convex function f(δ). Each evaluation is O(K) = O(n). So the total cost is O(n * log(precision/range)) ≈ O(n * 50) = O(15000) per pair step.

            # dc_k/dh[p] = (1-h[p+off])*I[0<=p+off<n] - h[p-off]*I[0<=p-off<n]
            p_plus = p + offsets
            p_minus = p - offsets
            q_plus = q + offsets
            q_minus = q - offsets

            pm_valid = (p_minus >= 0) & (p_minus < n)
            qp_valid = (q_plus >= 0) & (q_plus < n)
            qm_valid = (q_minus >= 0) & (q_minus < n)

            pp_vals = np.zeros(2*n-1)
            pm_vals = np.zeros(2*n-1)
            qp_vals = np.zeros(2*n-1)
            qm_vals = np.zeros(2*n-1)

            pm_vals[pm_valid] = h[p_minus[pm_valid]]
            qp_vals[qp_valid] = h[q_plus[qp_valid]]
            qm_vals[qm_valid] = h[q_minus[qm_valid]]

                c += m_k * d_opt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _logsumexp(a):
    m = a.max()
    if m > 500:
        return m + np.log(np.sum(np.exp(a - m)))
    return np.log(np.sum(np.exp(a)))

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(_logsumexp(alpha * c) / alpha * dx)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    scaled = alpha * c
    w = np.exp(scaled - scaled.max())
    w /= w.sum()
    # w has length 2n-1, w[k] corresponds to c_k
    # d_k = k - (n-1)
    grad = np.zeros(n)
    for p in range(n):
        lo = max(0, p - n + 1)
        hi = min(n - 1, p + n - 1)
        js = np.arange(lo, hi + 1)
        # Part 1: sum w[k] for valid k (where 0 <= p+k-(n-1) < n)
        k_lo = max(0, n - 1 - p)
        k_hi = min(2 * n - 2, n - 1 + p)
        s1 = w[k_lo:k_hi + 1].sum()
        # Part 2: sum w[k] * x[p+k-(n-1)] for valid k → x[j] * w[j-p+n-1]
        ks2 = js - p + (n - 1)
        s2 = (w[ks2] * x[js]).sum()
        # Part 3: sum w[k] * x[p-k+(n-1)] for valid k → x[j] * w[p-j+n-1]
        ks3 = p - js + (n - 1)
        s3 = (w[ks3] * x[js]).sum()
        grad[p] = dx * (s1 - s2 - s3)
    return grad

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, jac=smooth_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def pair_cd(h, n, deadline, rng, n_trials=300, n_gold=50):
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    offsets = np.arange(2 * n - 1) - (n - 1)
    gr_c = (np.sqrt(5.0) - 1.0) / 2.0
    
    for _pass in range(100000):
        if time.time() > deadline:
            break
        improved = False
        for _t in range(n_trials):
            if time.time() > deadline:
                break
            p = rng.randint(n)
            q = rng.randint(n)
            if p == q:
                continue
            
            p_plus = p + offsets
            p_minus = p - offsets
            q_plus = q + offsets
            q_minus = q - offsets
            
            pp_v = (p_plus >= 0) & (p_plus < n)
            pm_v = (p_minus >= 0) & (p_minus < n)
            qp_v = (q_plus >= 0) & (q_plus < n)
            qm_v = (q_minus >= 0) & (q_minus < n)
            
            pp_val = np.where(pp_v, h[np.clip(p_plus, 0, n-1)], 0.0)
            pm_val = np.where(pm_v, h[np.clip(p_minus, 0, n-1)], 0.0)
            qp_val = np.where(qp_v, h[np.clip(q_plus, 0, n-1)], 0.0)
            qm_val = np.where(qm_v, h[np.clip(q_minus, 0, n-1)], 0.0)
            
            gp = (1.0 - pp_val) * pp_v.astype(np.float64) - pm_val * pm_v.astype(np.float64)
            gq = (1.0 - qp_val) * qp_v.astype(np.float64) - qm_val * qm_v.astype(np.float64)
            
            mk = gp - gq
            if np.max(np.abs(mk)) < 1e-15:
                continue
            
            d_lo = max(-h[p], h[q] - 1.0)
            d_hi = min(1.0 - h[p], h[q])
            if d_hi - d_lo < 1e-12:
                continue
            
            a, b = d_lo, d_hi
            c1 = a + (1.0 - gr_c) * (b - a)
            c2 = a + gr_c * (b - a)
            for _g in range(n_gold):
                f1 = np.max(c + mk * c1)
                f2 = np.max(c + mk * c2)
                if f1 < f2:
                    b = c2; c2 = c1; c1 = a + (1.0 - gr_c) * (b - a)
                else:
                    a = c1; c1 = c2; c2 = a + gr_c * (b - a)
            
            d_opt = (a + b) * 0.5
            f_opt = np.max(c + mk * d_opt)
            
            if f_opt < current_max - 1e-13:
                h[p] += d_opt
                h[q] -= d_opt
                c += mk * d_opt
                current_max = f_opt
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = a.max()
    if m > 500:
