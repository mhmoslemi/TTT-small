def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    target=n/2.0
    h=np.clip(h,0,1)
    for _ in range(100):
       diff=target-h.sum()
       if abs(diff)<1e-12: break
       if diff>0:
          room=1-h; s=room.sum()
          if s<1e-14: break
          h += (diff/s)*room
       else:
          s=h.sum()
          if s<1e-14: break
          h += (diff/s)*h
    return np.clip(h,0,1)

def to_binary(h,n):

import time

_G_N = 0
_G_ACTIVE = None
_G_ALPHA = 1.0


def _compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def _project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def _interp_h(h, n_old, n_new):
    if n_new == n_old:
        return np.asarray(h, dtype=float).copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return _project_feasible(np.interp(x_new, x_old, np.asarray(h, dtype=float)), n_new)


def _select_active(c, size):
    order = np.argsort(c)[::-1]
    cmax = float(c[order[0]])
    limit = min(size, len(order))
    sel = []
    for idx in order:
        if len(sel) >= limit:
            break
        if c[idx] >= cmax - 0.25:
            sel.append(int(idx))
    if len(sel) < min(4, limit):
        sel = [int(x) for x in order[:min(4, limit)]]
    return sel


def _count_obj(x):
    return float(x[-1])


def _count_obj_jac(x):
    j = np.zeros(len(x))
    j[-1] = 1.0
    return j


def _active_eq_fun(x):

def _active_ineq_fun(x):

    t = float(x[-1])
    c = np.correlate(h, 1.0 - h, mode='full')
    return t - c[_G_ACTIVE]


def _active_ineq_jac(x):
    h = x[:-1]
    n = _G_N
    m = len(_G_ACTIVE)
    J = np.zeros((m, n + 1))
    for r, idx in enumerate(_G_ACTIVE):
        d = int(idx) - (n - 1)
        row = J[r]
        if d >= 0:
            row[:n - d] -= (1.0 - h[d:])
            row[d:] += h[:n - d]
        else:
            e = -d
            row[e:] -= (1.0 - h[:n - e])
            row[:n - e] += h[e:]
        row[n] = 1.0
    return J


def _active_slsqp(h, n, active, T0, maxiter, ftol):
    global _G_N, _G_ACTIVE
    if n <= 0 or len(active) == 0:
        return None
    _G_N = int(n)
    _G_ACTIVE = list(active)
    x0 = np.empty(n + 1)
    x0[:-1] = h
    x0[-1] = float(T0)
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'eq', 'fun': _active_eq_fun, 'jac': _active_eq_jac},
        {'type': 'ineq', 'fun': _active_ineq_fun, 'jac': _active_ineq_jac}
    ]
    try:
        res = minimize(
            _count_obj, x0, jac=_count_obj_jac, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': ftol, 'disp': False}
        )
    except Exception:
        return None
    if res is None or res.x is None or not np.all(np.isfinite(res.x)):
        return None
    nh = _project_feasible(res.x[:-1], n)
    c = np.correlate(nh, 1.0 - nh, mode='full')
    return nh, float(c.max())


def _active_polish(h, n, deadline, sizes, rounds, maxiter):
    h = np.asarray(h, dtype=float).copy()
    best_h = h.copy()
    c = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_m = float(c.max())
    for act in sizes:
        cur = best_h.copy()
        for rr in range(int(rounds)):
            if time.time() > deadline:
                break
            mi = max(30, int(maxiter * (1.0 - 0.15 * rr)))
            c = np.correlate(cur, 1.0 - cur, mode='full')
            m = float(c.max())
            active = _select_active(c, act)
            if not active:
                break
            sol = _active_slsqp(cur, n, active, m + 0.1, mi, 1e-12)
            if sol is None:
                break
            nh, nm = sol
            if nm < best_m - 1e-9:
                best_h = nh.copy()
                best_m = nm
                cur = nh.copy()
            else:
                break
    return best_h, best_m


def _smooth_obj(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    a = _G_ALPHA
    m = float(c.max())
    ex = np.exp(a * (c - m))
    s = float(ex.sum())
    if s <= 0.0:
        return m
    return m + math.log(s) / a


def _smooth_jac(x):
    h = x
    n = _G_N
    c = np.correlate(h, 1.0 - h, mode='full')
    a = _G_ALPHA
    m = float(c.max())
    w = np.exp(a * (c - m))
    ws = float(w.sum())
    if ws <= 0.0:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    center = n - 1
    for idx, wt in enumerate(w):
        if wt < 1e-15:
            continue
        d = int(idx) - center
        if d >= 0:
            g[:n - d] += wt * (1.0 - h[d:])
            g[d:] -= wt * h[:n - d]
        else:
            e = -d
            g[e:] += wt * (1.0 - h[:n - e])
            g[:n - e] -= wt * h[e:]
    return g


def _smooth_slsqp(h, n, alpha, maxiter, ftol):
    global _G_ALPHA, _G_N
    _G_ALPHA = float(alpha)
    _G_N = int(n)
    h0 = np.asarray(h, dtype=float).copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _active_eq_fun, 'jac': _active_eq_jac}]
    try:
        res = minimize(
            _smooth_obj, h0, jac=_smooth_jac, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': ftol, 'disp': False}
        )
    except Exception:
        return None
    if res is None or res.x is None or not np.all(np.isfinite(res.x)):
        return None
    nh = _project_feasible(res.x, n)
    c = np.correlate(nh, 1.0 - nh, mode='full')
    return nh, float(c.max())


def _emit_product(rows, cols, vals, b, p, i, j):
    r = len(b)
    b.append(0.0)
    rows.extend([r, r])
    cols.extend([p, i])
    vals.extend([1.0, -1.0])

    r = len(b)
    b.append(1.0)
    rows.extend([r, r])
    cols.extend([p, j])
    vals.extend([1.0, 1.0])

    r = len(b)
    b.append(0.0)
    rows.extend([r, r, r])
    cols.extend([p, i, j])
    vals.extend([-1.0, 1.0, -1.0])


def _milp_candidate(n, time_limit):
    if not _HAS_MILP or n % 2 != 0 or n < 16:
        return None
    try:
        total_p = n * (n - 1)
        nv = n + total_p + 1
        mvar = nv - 1

        c_obj = np.zeros(nv)
        c_obj[mvar] = 1.0

        rows, cols, vals, b = [], [], [], []
        poff = n

        for d in range(1, n):
            group = []
            for i in range(n - d):
                p = poff
                poff += 1
                group.append(p)
                _emit_product(rows, cols, vals, b, p, i, i + d)

            r = len(b)
            b.append(0.0)
            rows.append(r); cols.append(mvar); vals.append(-1.0)
            for p in group:
                rows.append(r); cols.append(p); vals.append(1.0)

            group = []
            for i in range(d, n):
                p = poff
                poff += 1
                group.append(p)
                _emit_product(rows, cols, vals, b, p, i, i - d)

            r = len(b)
            b.append(0.0)
            rows.append(r); cols.append(mvar); vals.append(-1.0)
            for p in group:
                rows.append(r); cols.append(p); vals.append(1.0)

        r = len(b)
        b.append(0.5 * n)
        for i in range(n):
            rows.append(r)
            cols.append(i)
            vals.append(1.0)

        A = sparse.lcoo_matrix((vals, (rows, cols)), shape=(len(b), nv)).tocsr()
        ub = np.asarray(b, dtype=float)
        lb = np.full(len(b), -np.inf)
        lb[-1] = 0.5 * n
        ub[-1] = 0.5 * n
        cons = LinearConstraint(A, lb, ub)

        lbv = np.zeros(nv)
        ubv = np.ones(nv)
        ubv[mvar] = float(n)

        intg = np.zeros(nv, dtype=int)
        intg[:n] = 1

        res = milp(
            c_obj, constraints=cons, integrality=intg,
            bounds=Bounds(lbv, ubv),
            options={'time_limit': int(max(1, time_limit))}
        )
        if res is None or res.x is None:
            return None

        hx = np.asarray(res.x[:n], dtype=float)
        br = np.rint(hx)
        if not np.all(np.abs(hx - br) < 1e-5):
            return None
        br = br.astype(float)

        target = n // 2
        s = int(round(br.sum()))
        if abs(s - target) <= 1:
            if s > target:
                ones = np.where(br == 1.0)[0]
                if len(ones):
                    br[ones[0]] = 0.0
            elif s < target:
                zeros = np.where(br == 0.0)[0]
                if len(zeros):
                    br[zeros[0]] = 1.0

        if abs(int(round(br.sum())) - target) != 0:
            return None
        if br.min() < -1e-9 or br.max() > 1.0 + 1e-9:
            return None

        cc = _compute_c5(br, n)
        if math.isfinite(cc):
            return br, cc, n
    except Exception:
        return None
    return None
