import numpy as np
import time

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def compute_autocorr(h, n):
    # R_k = sum_i h[i]*h[i+k]*dx for the "same" part
    # Actually R_k = 1 - f_k where f_k = corr[k]
    # We want to maximize min_k R_k = min_k (1 - f_k) = 1 - max_k f_k
    # So it's equivalent to minimizing max_k f_k
    return 1.0 - compute_corr(h, n)

def project_feasible(h, n):
    target = n / 2.0
    # Bisection for lambda: find lambda s.t. sum(clip(h - lambda, 0, 1)) = target
    lo = h.min() - 0.1
    hi = h.max() + 0.1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h_p = np.clip(h - lam, 0, 1)
    # Fine-tune
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-10) & (h_p < 1 - 1e-10)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
            h_p = np.clip(h_p, 0, 1)
    return h_p

def soft_min_autocorr(h, n, alpha):
    # Returns (-objective, gradient) for minimizing the soft-min of autocorrelation
    # Which is equivalent to maximizing the soft-min = minimizing C5
    # Objective: -(1/alpha) * log(sum(exp(-alpha * R_k)))
    # where R_k = 1 - f_k
    # Equivalent to: (1/alpha) * log(sum(exp(alpha * f_k))) - 1
    # (the -1 doesn't affect gradient or argmin)
    corr = compute_corr(h, n)  # f_k values
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    log_sum = sm + np.log(np.exp(shifted - sm).sum())
    obj = m + log_sum / alpha  # This is (1/alpha)*log(sum(exp(alpha*corr)))
    # Weights for gradient
    weights = np.exp(shifted - sm)
    weights /= weights.sum()
    return obj, weights, corr

def compute_gradient(h, n, weights):
    # grad of obj = sum_k w_k * f_k where f_k = corr_k
    # ∂f_k/∂h[j] = (1-h[j-k])*dx - h[j+k]*dx (derived earlier)
    # Wait, let me re-derive.
    # f_k = sum_i h[i+k]*h_i_complement... 
    # Actually f_k = <h, shift_k(1-h)> = sum_i h[i] * (1-hshift[i]) * dx
    # where hshift[i] = h[i-k] (shift h by k)
    # f_k = sum_i h[i] * (1 - h[i-k]) * dx
    # ∂f_k/∂h[j] = (1-h[j-k])*dx (from h[i] term, i=j)
    #              - h[j+k]*dx (from (1-h[i-k]) term, where i-k=j => i=j+k)
    # So ∂f_k/∂h[j] = (1-h[j-k])*dx - h[j+k]*dx
    # Total grad[j] = sum_k w_k * ((1-h[j-k])*dx - h[j+k]*dx)
    #              = sum_k w_k * dx - sum_k w_k * h[j-k]*dx - sum_k w_k * h[j+k]*dx
    #              = dx - dx * S1[j] - dx * S2[j]
    # where S1[j] = sum_k w_k * h[j-k], S2[j] = sum_k w_k * h[j+k]
    dx = 2.0 / n
    m = n - 1
    S1 = np.zeros(n)
    S2 = np.zeros(n)
    k_arr = np.arange(-m, m+1)
    for ki in range(len(weights)):
        if weights[ki] < 1e-14:
            continue
        k = ki - m
        wk = weights[ki]
        if k >= 0:
            S1[k:] += wk * h[:n-k]
            S2[:n-k] += wk * h[k:]
        else:
            kn = -k
            S1[:n-kn] += wk * h[kn:]
            S2[kn:] += wk * h[:n-kn]
    grad = dx - dx * S1 - dx * S2
    return grad

def pgd_solve(h_init, n, alpha, n_iter, lr, deadline):
    # Projected gradient descent
    h = h_init.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    # Precompute
    m = n - 1
    step = lr
    for it in range(n_iter):
        if time.time() > deadline:
            break

def project_feasible(h, n):

    lo = h.min() - 0.5
    hi = h.max() + 0.5

    corr = compute_corr(h, n)
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exp_shifted = np.exp(shifted - sm)
    log_sum = sm + np.log(exp_shifted.sum())

            grad[k:] += wk * (1.0 - h[:n-k]) * dx

            grad[:n-kn] += wk * (1.0 - h[kn:]) * dx

import numpy as np

from scipy.optimize import minimize

_slq = {'n': 0, 'alpha': 0}

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 0.5
    hi = h.max() + 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def obj_grad(h, n, alpha):
    corr = compute_corr(h, n)
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    es = np.exp(sh - sm)
    ew = es.sum()
    obj = m + (sm + np.log(ew)) / alpha
    w = es / ew
    dx = 2.0 / n
    mi = n - 1
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:
            continue
        k = ki - mi
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n - k]) * dx
            if k < n:
                grad[:n - k] -= wk * h[k:] * dx
        else:
            kn = -k
            grad[:n - kn] += wk * (1.0 - h[kn:]) * dx
            if kn < n:
                grad[kn:] -= wk * h[:n - kn] * dx
    return obj, grad

            stall = 0

    m = c.max()
    s = a * (c - m)
    sm = s.max()
    return m + (sm + np.log(np.exp(s - sm).sum())) / a

def _slq_grad(x):
    return obj_grad(x, _slq['n'], _slq['alpha'])[1]

def _slq_eq(x):
    return x.sum() - _slq['n'] / 2.0

def slsqp_optimize(h_init, n, alpha, maxiter, ftol):

    _slq['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n

    try:

                       bounds=bounds, constraints=cons,

def binary_greedy(h_init, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = set(int(x) for x in np.where(h == 1)[0])
    zeros = set(int(x) for x in np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break

        for i in list(zeros):
            if time.time() > deadline:

            for j in ones:

                # h[j] changes 1->0: effect on corr

                jmn = j - m_range

                    cur_c = nc
                    ones.discard(j)
                    zeros.discard(i)
                    ones.add(i)
                    zeros.add(j)

                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize

_slq = {'n': 0, 'alpha': 0}

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 0.5
    hi = h.max() + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def obj_grad(h, n, alpha):
    corr = compute_corr(h, n)
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    es = np.exp(sh - sm)
    ew = es.sum()
    obj = m + (sm + np.log(ew)) / alpha
    w = es / ew
    dx = 2.0 / n
    S_plus = np.zeros(n)
    S_minus = np.zeros(n)
    mi = n - 1
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:
            continue
        k = ki - mi
        if k >= 0:
            S_plus[:n - k] += wk * h[k:]
            S_minus[k:] += wk * h[:n - k]
        else:
            kn = -k
            S_plus[kn:] += wk * h[:n - kn]
            S_minus[:n - kn] += wk * h[kn:]
    grad = dx * (1.0 - S_plus - S_minus)
    return obj, grad

def pgd_optimize(h_init, n, alpha, n_iter, lr, deadline):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = lr
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = obj_grad(h, n, alpha)
        h_new = project_feasible(h - step * grad, n)
        obj_new, _ = obj_grad(h_new, n, alpha)
        if obj_new < obj - 1e-15:
            step = min(step * 1.03, 4.0 / n)
            stall = 0
        else:
            step *= 0.92
            stall += 1
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            stall = 0
        if stall > 800 or step < 1e-12:
            break
    return best_h, best_c

def _slq_obj(x):
    n = _slq['n']
    a = _slq['alpha']
    c = compute_corr(x, n)
    m = c.max()
    s = a * (c - m)
    sm = s.max()
    return m + (sm + np.log(np.exp(s - sm).sum())) / a

def _slq_grad(x):
    return obj_grad(x, _slq['n'], _slq['alpha'])[1]

def _slq_eq(x):
    return x.sum() - _slq['n'] / 2.0

def _slq_eq_jac(x):
    return np.ones(_slq['n'])

def slsqp_optimize(h_init, n, alpha, maxiter, ftol):
    _slq['n'] = n
    _slq['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_eq, 'jac': _slq_eq_jac}]
    try:
        res = minimize(_slq_obj, h_init, method='SLSQP', jac=_slq_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h_init.copy()

def binary_local_search(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = set(int(x) for x in np.where(h == 1)[0])
    zeros = set(int(x) for x in np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.discard(j)
                    zeros.discard(i)
                    ones.add(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
