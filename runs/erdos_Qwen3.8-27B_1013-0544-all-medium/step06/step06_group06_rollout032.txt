    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - mid, 0, 1)
    # Fine adjustment
    diff = target - h_proj.sum()

        free = (h_proj > 0) & (h_proj < 1)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        free = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if free.sum() > 0:
            hp[free] += diff / free.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def sm_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mc = corr.max()
    ec = np.exp(alpha * (corr - mc))
    w = ec / ec.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)

            if s < n:
                g[:n-s] += wk * (1.0 - h[s:])
            if s > 0:

        else:
            s = -s
            g[s:] += wk * (1.0 - h[:n-s])
            if s < n:
                g[:n-s] -= wk * h[s:]
    return g * dx

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        free = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if free.sum() > 0:
            hp[free] += diff / free.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def sm_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mc = corr.max()
    ec = np.exp(alpha * (corr - mc))
    w = ec / ec.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            s = -s
            g[s:] += wk * (1.0 - h[:n-s])
            g[:n-s] -= wk * h[s:]
    return g * dx

def optimize(h, n, a0, a1, ns, lr0, dl, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h.copy(), n)
    bc = compute_c5(h, n)
    bh = h.copy()
    lr = lr0
    st = 0
    for i in range(ns):
        if time.time() > dl:
            break
        fr = i / max(ns - 1, 1)
        al = a0 * (a1 / a0) ** fr
        g = sm_grad(h, n, al)
        ok = False
        for _ in range(12):
            hn = project_feasible(h - lr * g, n)
            cn = compute_c5(hn, n)
            if cn < bc - 1e-14:
                h, bc, bh, st = hn, cn, hn.copy(), 0
                ok = True
                break
            lr *= 0.5
        if not ok:
            st += 1
            if st > 35:
                h = project_feasible(h + 0.01 * rng.randn(n), n)
                lr, st = lr0 * 0.3, 0
        lr = min(lr * 1.03, lr0)
    return bh, bc

def tm_init(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def bin_polish(h, n, dl, seed=42):
    rng = np.random.RandomState(seed)
    t = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:t]] = 1.0
    cb = compute_c5(hb, n)
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    corr = np.correlate(hb, 1 - hb, mode='full').astype(float)
    dx = 2.0 / n
    for _ in range(80000):
        if time.time() > dl:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        hb2 = hb.copy()
        hb2[i], hb2[j] = 1.0, 0.0
        c2 = compute_c5(hb2, n)
        if c2 < cb - 1e-14:
            hb = hb2
            cb = c2
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
    return hb, cb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5

    if c1 < bc:

    # Phase 3: High alpha

    if c3 < bc:
        bc, bh = c3, h3

    # Phase 4: Thue-Morse restart
    p4 = t0 + 0.70 * (budget_s - 5)

        if c4 < bc:

    h5, c5 = bin_polish(bh, n, p5, seed + 200)
    if c5 < bc:
        bc, bh = c5, h5

    # Phase 6: Final continuous polish with very high alpha
    if time.time() < dl - 5:
        h6, c6 = optimize(bh, n, 10000, 100000, 350, 0.1, dl, seed + 300)
        if c6 < bc:
            bc, bh = c6, h6

    bh = project_feasible(bh, n)
    bc = compute_c5(bh, n)
    return bh, bc, n
