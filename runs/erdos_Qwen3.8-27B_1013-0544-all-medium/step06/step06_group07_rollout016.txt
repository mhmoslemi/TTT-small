            for _ in range(nk):
                ii = zeros_list[rng.randint(lz)]
                jj = ones_list[rng.randint(lo)]
                h[ii] = 1.0
                h[jj] = 0.0
                zi = zeros_list.index(ii)
                zeros_list[zi] = jj
                oj = ones_list.index(jj)
                ones_list[oj] = ii
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_improve = 0
            T = min(t_start * 0.9, T * 4.0)
    return best_h, best_c

def greedy_best(h, n, deadline, max_sweeps=100, seed=42):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_dc = 0
        best_i = -1
        best_j = -1
        for i in zeros:
            di = i + m_range
            msk_p = (di >= 0) & (di < n)
            msk_m = ((i - m_range) >= 0) & ((i - m_range) < n)
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                dc = nc - cur_c
                if dc < best_dc:
                    best_dc = dc
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                ii = zeros[rng.randint(len(zeros))]
                jj = ones[rng.randint(len(ones))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    h_bin = round_to_binary(best_h, n)

    # Multi-start SA phase: try different initial configurations and parameters
    starts = []
    starts.append(h_bin)
    starts.append(round_to_binary(best_h + 0.15 * rng.randn(n), n))
    starts.append(round_to_binary(best_h + 0.25 * rng.randn(n), n))
    # Alternating pattern
    alt = np.zeros(n)
    alt[::2] = 1.0
    if alt.sum() > n / 2:
        alt = alt.copy()
        alt[alt.sum() - n // 2:] = 0.0
        idx2 = np.where(alt == 1)[0]
        if len(idx2) > n // 2:
            alt[idx2[n // 2:]] = 0.0
    elif alt.sum() < n / 2:
        alt = alt.copy()
        idx0 = np.where(alt == 0)[0]
        need = n // 2 - int(alt.sum())
        alt[idx0[:need]] = 1.0
    starts.append(alt)

    # Allocate time: ~80% SA, ~12% greedy, ~8% final polish
    sa_end = t0 + 0.80 * (budget_s - 5)
    greedy_end = t0 + 0.92 * (budget_s - 5)

    n_runs = 6
    per_run_time = (sa_end - t0) / n_runs

    configs = [
        (0.05, 0.000005, 50000),
        (0.03, 0.000003, 60000),
        (0.06, 0.00001, 35000),
        (0.025, 0.000002, 70000),
        (0.04, 0.000004, 45000),
        (0.035, 0.000003, 55000),
    ]

    for run_idx in range(n_runs):
        phase_dl = t0 + (run_idx + 1) * per_run_time
        if time.time() > phase_dl - 2:
            continue
        s = seed + run_idx * 17
        ts, te, pat = configs[run_idx % len(configs)]
        start_h = starts[run_idx % len(starts)]
        hb, cb = binary_sa(start_h, n, phase_dl, seed=s, t_start=ts, t_end=te, patience=pat)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            h_bin = hb.copy()

    # Greedy local search phase
    if time.time() < greedy_end - 5:
        hg, cg = greedy_best(best_h, n, greedy_end, max_sweeps=80, seed=seed + 500)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    # Final SA polish from best
    if time.time() < dl - 15:
        hb, cb = binary_sa(best_h, n, dl - 10, seed=seed + 999, t_start=0.015, t_end=0.000002, patience=80000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Final greedy
    if time.time() < dl - 3:
        hg, cg = greedy_best(best_h, n, dl - 2, max_sweeps=40, seed=seed + 1000)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
