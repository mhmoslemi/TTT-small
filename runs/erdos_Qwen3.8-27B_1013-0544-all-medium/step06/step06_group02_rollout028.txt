def grad_sm(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    z = alpha * (c - m)
    z_max = z.max()
    w = np.exp(z - z_max)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # w is indexed 0 to 2n-2, corresponding to lag k = index - (n-1)
    # For each lag k, weight w[k+n-1]:
    #   grad[i] += w[k+n-1]*dx * [(1-h[i-k]) if 0<=i-k<n else 0] - w[k+n-1]*dx * [h[i+k] if 0<=i+k<n else 0]
    for idx in range(2*n-1):
        if w[idx] < 1e-15:
            continue
        k = idx - (n-1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            # (1-h[i-k]) for i=k,...,n-1: h[0:nv] gives h[i-k] for i=k..n-1
            grad[k:n] += wm * (1.0 - h[0:nv])
            # h[i+k] for i=0,...,n-1-k: h[k:n] gives h[i+k] for i=0..nv-1
            grad[0:nv] -= wm * h[k:n]

            nv = n - kk
            # (1-h[i-k]) = (1-h[i+kk]) for i=0,...,n-1-kk: h[kk:n] gives h[i+kk]
            grad[0:nv] += wm * (1.0 - h[kk:n])
            # h[i+k] = h[i-kk] for i=kk,...,n-1: h[0:nv] gives h[i-kk]

from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

    return h, best_c

def slsqp_optimize(h, n, deadline, alpha):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def nesterov_opt(h, n, deadline, alpha, lr, momentum):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    x = h.copy()
    for it in range(100000):
        if time.time() > deadline:
            break
        y = project(x + momentum * v, n)
        g = grad_sm(y, n, alpha)
        x_new = project(y - lr * g, n)
        v = x_new - x
        x = x_new
        if it % 5 == 0:
            c = compute_c5(x, n)
            if c < best_c:
                best_c = c
                best_h = x.copy()
            if c > best_c + 0.005:
                lr *= 0.5
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)

    h = project(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B
    t1 = t0 + budget_s * 0.22
    h1, c1 = lbfgs_optimize(h, n, t1, [100, 500, 2000, 10000, 50000])
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: SLSQP
    if time.time() < deadline - budget_s * 0.45:
        for a in [100000, 500000]:
            if time.time() > deadline - budget_s * 0.45:
                break
            h2, c2 = slsqp_optimize(best_h, n, min(time.time() + 50, deadline - budget_s * 0.44), alpha=a)
            if c2 < best_c:
                best_c, best_h = c2, h2

    # Phase 3: Nesterov
    if time.time() < deadline - budget_s * 0.35:
        t3 = min(time.time() + budget_s * 0.08, deadline - budget_s * 0.34)
        h3, c3 = nesterov_opt(best_h, n, t3, alpha=50000, lr=0.0003, momentum=0.85)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Systematic pair search (main refinement)
    if time.time() < deadline - budget_s * 0.15:
        t4 = min(time.time() + budget_s * 0.30, deadline - budget_s * 0.14)
        h4, c4 = systematic_pair_search(best_h, n, t4,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                         radius=100)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Random pair search (finer steps)
    if time.time() < deadline - budget_s * 0.06:
        t5 = min(time.time() + budget_s * 0.08, deadline - budget_s * 0.05)
        h5, c5 = fast_pair_search(best_h, n, t5, rng,
                                   [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Final L-BFGS + pair search
    if time.time() < deadline - 15:
        t6 = min(time.time() + 30, deadline - 15)
        h6, c6 = lbfgs_optimize(best_h, n, t6, [200000, 500000, 2000000])
        if c6 < best_c:
            best_c, best_h = c6, h6
        if time.time() < deadline - 5:
            h7, c7 = systematic_pair_search(best_h, n, deadline - 3,
                                             [0.0002, 0.0001, 0.00005], radius=50)
            if c7 < best_c:
                best_c, best_h = c7, h7

    best_h = project(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
