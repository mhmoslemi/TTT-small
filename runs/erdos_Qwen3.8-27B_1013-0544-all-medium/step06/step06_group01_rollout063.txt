```python
def compute_c5(h, n):

def project_onto_simplex_box(h, n):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Use bisection on lambda
    # We want sum(clip(h - lam, 0, 1)) = target
    lo, hi = -1.0, 2.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target - projected.sum()
    # Distribute the difference to elements that are not at boundary
    mask = (projected > 0) & (projected < 1)
    if mask.any() and abs(diff) > 1e-12:
        projected[mask] += diff / mask.sum()
        projected = np.clip(projected, 0.0, 1.0)
    return projected

    w = e / e_sum  # weights, sum to 1
    grad = np.zeros(n)
    # For each p: grad[p] = dx * sum_k w[k] * [(1-h[p+t_k])*I - h[p-t_k]*I]
    # where t_k = k - n + 1
    for p in range(n):
        val = 0.0
        for k in range(2 * n - 1):
            t = k - n + 1
            wt = w[k]
            if wt < 1e-15:
                continue
            if 0 <= p + t < n:
                val += wt * (1.0 - h[p + t])
            if 0 <= p - t < n:
                val -= wt * h[p - t]
        grad[p] = dx * val
        # Add penalty gradient
        grad[p] += penalty * (s - n / 2.0)
    
    return obj, grad
```

```python
import numpy as np

import time

def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, n):
    """Project onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for the right shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2, 0, 1)
    # Fixup
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

def smooth_obj_grad(h, n, alpha, mu):
    """Smooth approximation of C5 + mu/2 * (sum(h)-n/2)^2"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    e = np.exp(alpha * (corr - cmax))
    esum = e.sum()
    obj = (cmax + np.log(esum) / alpha) * dx
    
    s = h.sum()
    obj += 0.5 * mu * (s - n/2.0)**2
    
    w = e / esum  # weights
    # Gradient via correlations
    B = np.correlate(w, h, mode='full')[0:n]
    C_full = np.correlate(w, h[::-1], mode='full')
    C = C_full[n-1-np.arange(n)]
    
    # A via prefix sum
    prefix = np.concatenate(([0.0], np.cumsum(w)))
    A = np.zeros(n)
    for p in range(n):
        lo = max(0, n-1-p)
        hi = min(2*n-2, n-1+p) + 1
        A[p] = prefix[hi] - prefix[lo]
    
    grad = dx * (A - B - C) + mu * (s - n/2.0) * np.ones(n)
    return obj, grad
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5*(lo+hi), 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        m = (h > 1e-10) & (h < 1.0-1e-10)
        if m.any():
            h[m] += diff/m.sum()
    return np.clip(h, 0, 1)

def _obj(x, n, alpha, mu):
    dx = 2.0/n
    corr = np.correlate(x, 1.0-x, mode='full')
    cm = corr.max()
    e = np.exp(alpha*(corr-cm))
    es = e.sum()
    obj = (cm + np.log(es)/alpha)*dx
    s = x.sum()
    obj += 0.5*mu*(s-n/2.0)**2
    return obj

def _grad(x, n, alpha, mu):
    dx = 2.0/n
    corr = np.correlate(x, 1.0-x, mode='full')
    cm = corr.max()
    e = np.exp(alpha*(corr-cm))
    w = e/e.sum()
    B = np.correlate(w, x, mode='full')[:n]
    Cf = np.correlate(w, x[::-1], mode='full')
    C = Cf[n-1-np.arange(n)]
    pr = np.concatenate(([0.0], np.cumsum(w)))
    A = np.zeros(n)
    for p in range(n):
        A[p] = pr[min(2*n-2, n-1+p)+1] - pr[max(0, n-1-p)]
    s = x.sum()
    return dx*(A-B-C) + mu*(s-n/2.0)

def _lbfgs(h, n, alpha, mu, maxiter):

                       method='L-BFGS-B', bounds=[(0,1)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxls': 40})
        return np.clip(res.x, 0, 1)
    except:
        return h

def _cd(h, n, deadline, grid=200):
    dx = 2.0/n
    h = h.copy()
    corr = np.correlate(h, 1.0-h, mode='full')
    nc = 2*n-1
    cm = corr.max()
    t = np.arange(-(n-1), n)
    ns = np.ones(nc, dtype=bool); ns[n-1] = False
    corr_ns = corr.copy()
    
    while time.time() < deadline:
        imp = False
        corr_ns = corr.copy()  # keep non-self part for reference
        for p in range(n):
            if time.time() > deadline:
                return h, cm*dx
            hp = h[p]
            pt = p+t; ptm = p-t
            vf = (pt>=0)&(pt<n); vb = (ptm>=0)&(ptm<n)
            s = np.zeros(nc)
            s[vf] += 1.0-h[pt[vf]]
            s[vb] -= h[ptm[vb]]
            
            d = np.linspace(-hp, 1.0-hp, grid)
            prod = d[:,None]*s[ns][None,:]
            mns = (corr[ns][None,:]+prod).max(axis=1)
            sv = corr[n-1] + d*s[n-1] - d*d
            nm = np.maximum(mns, sv)
            bi = np.argmin(nm)
            bd = d[bi]
            if nm[bi] < cm - 1e-14 and abs(bd) > 1e-10:

                corr += bd*s
                corr[n-1] -= bd*bd
                cm = corr.max()
                imp = True
        if not imp:
            break
    return h, cm*dx
```
