import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def lag_gradient(h, n, lag):
    """Gradient of correlation at given lag w.r.t. h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # grad[i] = (1-h[i+lag]) for i < n-lag, minus h[i-lag] for i >= lag
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        # lag < 0, m = -lag
        # grad[i] = (1-h[i-m]) for i >= m, minus h[i+m] for i < n-m
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return grad

        c = c[top_idx]
    else:

    val = smooth_max_val(c, alpha)
    w = smooth_max_grad_weights(c, alpha)

    if top_idx is not None:

    for _ in range(500):

        if abs(d) < 1e-13:

        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:

        else:
            s = h.sum()
            if s < 1e-13:

            h += (d / s) * h

    m = c.max()

    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sm_w(c, alpha):

    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)

        g[:n-m] -= h[m:]
    return g
