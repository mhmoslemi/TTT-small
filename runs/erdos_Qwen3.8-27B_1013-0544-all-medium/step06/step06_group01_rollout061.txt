import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h, n, max_passes=50, deadline=None):
    """Local search on binary solutions using pair flips."""
    h = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n // 2:
        # Remove excess ones
        excess = len(ones) - n // 2
        for idx in ones[:excess]:
            h[idx] = 0
    elif len(ones) < n // 2:
        need = n // 2 - len(ones)
        for idx in zeros[:need]:
            h[idx] = 1
    
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(c))
    dx = 2.0 / n
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for i in ones:
            for j in zeros:
                # Try flipping i: 1→0 and j: 0→1
                h[i] = 0
                h[j] = 1
                new_c = np.correlate(h, 1.0 - h, mode='full')
                new_max = float(np.max(new_c))
                if new_max < best_c - 1e-14:
                    best_c = new_max
                    c = new_c
                    improved = True
                    # Update lists
                    ones.remove(i)
                    zeros.remove(j)
                    ones.append(j)
                    zeros.append(i)
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
    
    return h, best_c * dx
