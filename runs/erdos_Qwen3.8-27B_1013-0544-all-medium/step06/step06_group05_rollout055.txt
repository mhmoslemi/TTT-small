def fast_binary_ls(h_init, n, deadline, max_sweeps=10000):
    dx = 2.0 / n
    h = h_init.copy()
    npts = 2 * n - 1
    
    def compute_role_matrices(h):
        # RoleA[p] = (1-h)[::-1][0:n-p-1] at positions [p+1, n-1+p]
        # RoleB[p] = h[0:p+1] at positions [n-1-p, 2n-2-p] (if p < n-1)
        RoleA = np.zeros((n, npts))
        RoleB = np.zeros((n, npts))
        rev_1mh = (1.0 - h)[::-1]
        for p in range(n):
            # Role A: length n-p-1, at positions p+1 to p+n-1
            la = n - p - 1
            if la > 0:
                RoleA[p, p+1:p+1+la] = rev_1mh[:la]
            # Role B: length p+1, at positions n-1-p to n-1-p+p = n-1
            if p < n - 1:  # only if p+1 < n... actually p can be up to n-2
                lb = p + 1
                RoleB[p, n-1-p:n-1-p+lb] = h[:lb]
        return RoleA, RoleB
    
    RoleA, RoleB = compute_role_matrices(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = c.max()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in zeros:
            if time.time() > deadline:
                break
            base = c + RoleA[i] + RoleB[i]
            # For all j in ones:
            new_c_all = base[None, :] - RoleA[ones, :] - RoleB[ones, :]
            new_max_all = new_c_all.max(axis=1)
            best_idx = new_max_all.argmin()
            best_new_max = new_max_all[best_idx]
            
            if best_new_max < cur_max:
                j = ones[best_idx]
                h[i] = 1.0
                h[j] = 0.0
                # Update RoleA and RoleB for positions i and j
                # Actually, Roles depend on h, so after changing h, all Roles change!
                # This is a problem...
                
                # Recompute Roles
                RoleA, RoleB = compute_role_matrices(h)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur_max = c.max()
                improved = True
        
        if not improved:
            break
    
    return h, float(cur_max * dx)
