from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    val = c.max() + (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha
    return val

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

            nv = n - kk
            grad[kk:n] += (dx / k) * (1.0 - h[0:nv])
            grad[0:nv] += (dx / k) * (-h[kk:n])
    return grad

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

            return topk_mean_obj(x, n, topk)
        def grad(x):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _corr_grad_single(h, n, m):
    dx = 2.0 / n
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] += dx * (1.0 - h[k:n])
        g[k:n] += -dx * h[0:nv]
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += dx * (1.0 - h[0:nv])
        g[0:nv] += -dx * h[kk:n]
    return g

def topk_mean_obj(x, n, k):
    c = compute_c5_vec(x, n)
    idx = np.argpartition(-c, min(k, len(c)-1))[:k]
    return float(c[idx].mean())

def topk_mean_grad(x, n, k):
    c = compute_c5_vec(x, n)
    idx = np.argpartition(-c, min(k, len(c)-1))[:k]
    g = np.zeros(len(x))
    for m in idx:
        g += _corr_grad_single(x, n, m)
    return g / k

def smooth_max_obj_wrap(x, n, alpha):
    c = compute_c5_vec(x, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    return float(c.max() + (mx + np.log(np.sum(np.exp(cs - mx)))) / alpha)

def smooth_max_grad_wrap(x, n, alpha):
    c = compute_c5_vec(x, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    g = np.zeros(len(x))
    for m in range(len(c)):
        if w[m] < 1e-12:
            continue
        g += w[m] * _corr_grad_single(x, n, m)
    return g

def lbfgs_smooth(h, n, deadline, alphas=(200, 500, 2000, 10000), max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return smooth_max_obj_wrap(x, n, alpha)
        def grad(x):
            return smooth_max_grad_wrap(x, n, alpha)
        t_alloc = min(40, deadline - time.time() - 5)
        if t_alloc < 5:
            break
        iters = max(50, min(max_iters, int(t_alloc * 3)))
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12,
                                   'maxfun': iters * 15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def lbfgs_topk(h, n, deadline, k=3, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return topk_mean_obj(x, n, k)
    def grad(x):
        return topk_mean_grad(x, n, k)
    try:

                               'maxfun': max_iters * 15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def momentum_gd(h, n, deadline, topk=3, lr=0.002, beta=0.85, max_iter=8000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    momentum = np.zeros(n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = topk_mean_grad(h, n, topk)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = project_feasible(h - lr * momentum, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            h = h_new
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.5
            if lr < 1e-11:
                break
    return best_h, best_c

def pair_search(h, n, deadline, radius=80):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def upsample_and_optimize(h, n_from, n_to, deadline):
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_to)

    if time.time() < deadline - 5:
        h_new, c_new = lbfgs_topk(h_new, n_to, deadline, k=3, max_iters=100)
    return h_new, c_new
