def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:
        g[:n-m] -= h[m:]
        g[m:] += 1.0 - h[:n-m]
    else:
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

def solve_lp(h_curr, n, trust, top_k=80):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    idx = np.argsort(c)[-top_k:]
    h_v = cp.Variable(n)
    t_v = cp.Variable()
    cons = [cp.sum(h_v) == n / 2.0, h_v >= 0, h_v <= 1,
            cp.norm(cp.abs(h_v - h_curr), np.inf) <= trust]
    for k in idx:
        lag = k - (n - 1)
        g = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (g @ (h_v - h_curr)) <= t_v)
    prob = cp.Problem(cp.Minimize(t_v), cons)
    for solver in [cp.SCS, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except:
            pass
    if prob.status not in ['optimal', 'optimal_inaccurate'] or h_v.value is None:
        return h_curr.copy(), c.max()
    h_new = project_feasible(h_v.value, n)
    return h_new, compute_c5(h_new, n)

def outer_loop(h, n, deadline, max_iters=300):
    h_curr = h.copy()
    best_h, best_c = h_curr.copy(), compute_c5(h_curr, n)
    trust = 0.08
    stagnant = 0
    for _ in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new, c_new = solve_lp(h_curr, n, trust)
        if c_new < best_c - 1e-11:
            best_c, best_h, stagnant = c_new, h_new.copy(), 0
        else:
            stagnant += 1
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            trust = max(0.002, trust * 0.6)
        elif step < trust * 0.05:
            trust = max(0.002, trust * 0.75)
        else:
            trust = min(0.25, trust * 1.15)
        h_curr = h_new
        if stagnant > 30:
            break
    return best_h, best_c

def _smooth(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cc = c[top_idx]
    m = cc.max()
    s = alpha * (cc - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def slsqp_refine(h, n, alpha, top_m=8, maxiter=4000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: _smooth(x, n, top_idx, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def local_search(h, n, deadline, n_top=6):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            top = np.argsort(c)[-n_top:]
            for k_s in top:
                if time.time() > deadline:
                    break
                lag = k_s - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                stride = max(1, (n - m) // 300)
                for i in range(0, n - m, stride):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if abs(h[i] - h[j]) < 1e-6:
                        continue
                    if h[i] > h[j]:
                        delta = min(d, (h[i]-h[j])*0.35)
                    else:
                        delta = min(d, (h[j]-h[i])*0.35)
                    if delta < 1e-9:
                        continue
                    if h[i] > h[j]:
                        h[i] -= delta; h[j] += delta
                    else:
                        h[i] += delta; h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new; improved = True
                    else:
                        h[i] += delta if h[i] <= h[j] else -delta
                        h[j] -= delta if h[i] <= h[j] else -delta
                        # Restore
                        if h[i] > h[j]:
                            h[i] += delta; h[j] -= delta
                        else:
                            h[i] -= delta; h[j] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    if time.time() < deadline - 25:
        e = time.time() + min(50, (deadline - time.time()) * 0.4)
        h2, c2 = outer_loop(best_h, n, e)
        if c2 < best_c: best_h, best_c = h2, c2
    if time.time() < deadline - 15:
        for a in [3000, 30000, 150000]:
            if time.time() > deadline - 15: break
            h2 = slsqp_refine(best_h, n, a)
            c2 = compute_c5(h2, n)
            if c2 < best_c: best_h, best_c = h2, c2
    if time.time() < deadline - 20:
        e = time.time() + min(35, (deadline - time.time()) * 0.3)
        h2, c2 = outer_loop(best_h, n, e)
        if c2 < best_c: best_h, best_c = h2, c2
    if time.time() < deadline - 12:
        h2 = slsqp_refine(best_h, n, 500000, top_m=5, maxiter=5000)
        c2 = compute_c5(h2, n)
        if c2 < best_c: best_h, best_c = h2, c2
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, min(deadline - 1, time.time() + 25))
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except:
        h_init = None
        n_init = 200
    best_h, best_c = None, float('inf')
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.35)
        best_h, best_c = optimize_at_n(h_init, n_init, t0 + alloc)
    for n_try in [100, 150, 200, 250, 400, 500, 600]:
        if time.time() > deadline - 90: break
        if best_h is not None and n_try == len(best_h): continue
        rem = deadline - time.time() - 50
        if rem < 45: break
        alloc = min(90, rem * 0.28)
        if best_h is not None:
            h_start = np.interp(np.linspace(0,2,n_try), np.linspace(0,2,len(best_h)), best_h)
        elif h_init is not None:
            h_start = np.interp(np.linspace(0,2,n_try), np.linspace(0,2,n_init), h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h2, c2 = optimize_at_n(project_feasible(h_start, n_try), n_try, time.time()+alloc)
        if c2 < best_c: best_h, best_c = h2, c2
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        rem = deadline - time.time() - 35
        for r in range(max(1, int(rem // 65))):
            if time.time() > deadline - 35: break
            hp = project_feasible(best_h + rng.normal(0, 0.012*(1+0.3*r), n), n)
            alloc = min(55, (deadline - time.time() - 35) * 0.4)
            if alloc < 25: break
            h2, c2 = optimize_at_n(hp, n, time.time()+alloc)
            if c2 < best_c: best_h, best_c = h2, c2
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        h2 = slsqp_refine(best_h, n, 1000000, top_m=5, maxiter=6000)
        c2 = compute_c5(h2, n)
        if c2 < best_c: best_h, best_c = h2, c2
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(200, 0.5), 200)
        best_c = compute_c5(best_h, 200)
    return best_h, best_c, len(best_h)
