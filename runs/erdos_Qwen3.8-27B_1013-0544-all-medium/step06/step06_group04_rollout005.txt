```python
import numpy as np
import time
import math
from scipy.optimize import minimize

_MASKS = {}


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def shifted_values(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    omh = 1.0 - h
    vals = np.empty(2 * n - 1, dtype=float)
    vals[n - 1] = float(np.dot(h, omh)) * dx
    for s in range(1, n):
        m = n - s
        vals[n - 1 + s] = float(np.dot(h[:m], omh[s:])) * dx
        vals[n - 1 - s] = float(np.dot(h[s:], omh[:m])) * dx
    return vals


def sum_constraint(x, n):
    return float(x.sum() - 0.5 * n)


def gradient_from_weights(h, n, w):
    h = np.asarray(h, dtype=float)
    omh = 1.0 - h
    g = np.zeros(n, dtype=float)
    w0 = float(w[n - 1])
    if w0 != 0.0:
        g += w0 * (omh - h)
    dx = 2.0 / float(n)
    for s in range(1, n):
        m = n - s
        wp = float(w[n - 1 + s])
        if wp != 0.0:
            g[:m] += wp * omh[s:]
            g[s:] -= wp * h[:m]
        wn = float(w[n - 1 - s])
        if wn != 0.0:
            g[s:] += wn * omh[:m]
            g[:m] -= wn * h[s:]
    return g * dx


def smooth_obj(x, n, alpha):
    h = np.asarray(x, dtype=float)
    vals = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
    m = float(vals.max())
    if alpha <= 0.0:
        return m
    z = alpha * (vals - m)
    return m + math.log(float(np.exp(z).sum())) / float(alpha)


def smooth_jac(x, n, alpha):
    h = np.asarray(x, dtype=float)
    vals = shifted_values(h, n)
    m = float(vals.max())
    z = alpha * (vals - m)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 0.0:
        w = np.ones_like(vals) / float(vals.size)
    else:
        w /= s
    return gradient_from_weights(h, n, w)


def smooth_optimize(h0, n, deadline, seed=42, alphas=(1500.0, 8000.0)):
    h0 = np.asarray(h0, dtype=float)
    if len(h0) != n:
        x0 = np.linspace(0.0, 2.0, len(h0), endpoint=False)
        x1 = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x1, x0, h0, left=h0[0], right=h0[-1])
    h = project_feasible(h0, n)
    best = h.copy()
    bestc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": sum_constraint, "args": (n,)}]
    maxit = 80 if n > 320 else 150
    for alpha in alphas:
        if time.time() > deadline - 4.0:
            break
        try:
            res = minimize(
                smooth_obj,
                h,
                args=(n, alpha),
                method="SLSQP",
                jac=smooth_jac,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxit, "ftol": 1e-11},
            )
            cand = project_feasible(res.x, n)
            c = compute_c5(cand, n)
            if c < bestc:
                bestc = c
                best = cand.copy()
                h = cand.copy()
        except Exception:
            pass
    return best, bestc

    last_imp = 0
    it = 0

        it += 1
        vals = shifted_values(h, n)
        mv = float(vals.max())

        z = (vals[idx] - mv) / 0.0025
        wl = np.exp(z)

        else:
            wl /= s
        w = np.zeros(2 * n - 1, dtype=float)
        w[idx] = wl
        grad = gradient_from_weights(h, n, w)

                break

        if not accepted:
            a = int(rng.randint(n))
            b = int(rng.randint(n))
            if a != b:
                h2 = h.copy()
                h2[a], h2[b] = h2[b], h2[a]
                v2 = compute_c5(h2, n)
                if v2 < cur or (v2 - cur <= 0.0008 and rng.random() < math.exp(-(v2 - cur) / 0.0008)):
                    h = h2
                    cur = v2
                    accepted = True
            if (not accepted) and it % 70 == 0:
                amp = 0.015 + 0.03 * rng.random()
                h2 = project_feasible(h + amp * rng.randn(n), n)
                v2 = compute_c5(h2, n)
                if v2 < cur + 0.0015 and rng.random() < math.exp(-(v2 - cur) / max(0.0005, 0.0015)):
                    h = h2
                    cur = v2
                    accepted = True

        if cur < bestc - 1e-13:
            bestc = cur
            best = h.copy()
            last_imp = it

        if it - last_imp > 12000:
            h = project_feasible(best + 0.04 * rng.randn(n), n)
            cur = compute_c5(h, n)

    return best, bestc


def get_masks(n):
    if n not in _MASKS:
        arr = [0] * (n + 1)
        m = 0
        for k in range(1, n + 1):
            m = (m << 1) | 1
            arr[k] = m
        _MASKS[n] = arr
    return _MASKS[n]


def bits_from_h(h, n):
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        x0 = np.linspace(0.0, 2.0, len(h), endpoint=False)
        x1 = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x1, x0, h, left=h[0], right=h[-1])
    target = n // 2
    idx = np.argpartition(h, -target)[-target:]
    bits = 0
    for i in idx:
        bits |= 1 << int(i)
    return bits


def array_from_bits(bits, n):
    arr = np.zeros(n, dtype=float)
    for i in range(n):
        if (bits >> i) & 1:
            arr[i] = 1.0
    return arr


def random_balanced_bits(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    bits = 0
    for i in idx:
        bits |= 1 << int(i)
    return bits


def bits_max_leq(bits, n, limit, order=None):
    pref = get_masks(n)
    if order is None:
        rng = np.random.RandomState(0)
        order = list(range(1, n))
        order.reverse()
    mx = 0
    for s in order:
        sh = bits >> s
        left = (bits & pref[n - s]).bit_count()
        right = sh.bit_count()
        inter = (bits & sh).bit_count()
        total = left + right - 2 * inter
        diff = left - right
        md = (total + abs(diff)) >> 1
        if md > limit:
            return limit + 1
        if md > mx:
            mx = md
    return mx


def bits_lag_counts(bits, n):
    pref = get_masks(n)
    pos = [0] * n
    neg = [0] * n
    for s in range(1, n):
        sh = bits >> s
        left = (bits & pref[n - s]).bit_count()
        right = sh.bit_count()
        inter = (bits & sh).bit_count()
        total = left + right - 2 * inter
        diff = left - right
        pos[s] = (total + diff) // 2
        neg[s] = (total - diff) // 2
    return pos, neg


def targeted_binary_improve(bits, n, fullmask):
    pos, neg = bits_lag_counts(bits, n)
    cur = 0
    for s in range(1, n):
        if pos[s] > cur:
            cur = pos[s]
        if neg[s] > cur:
            cur = neg[s]
    if cur <= 1:
        return (bits, cur)
    seen = set()
    cands = []
    for s in range(1, n):
        if pos[s] >= cur - 1 and pos[s] > 0:
            cnt = 0
            for a in range(0, n - s):
                if ((bits >> a) & 1) and (((bits >> (a + s)) & 1) == 0):
                    z, o = a + s, a
                    key = (z, o)
                    if key not in seen:
                        seen.add(key)
                        cands.append(key)
                        cnt += 1
                        if cnt >= 70:
                            break
        if neg[s] >= cur - 1 and neg[s] > 0:
            cnt = 0
            for a in range(0, n - s):
                if (((bits >> a) & 1) == 0) and (((bits >> (a + s)) & 1) == 1):
                    z, o = a, a + s
                    key = (z, o)
                    if key not in seen:
                        seen.add(key)
                        cands.append(key)
                        cnt += 1
                        if cnt >= 70:
                            break
        if len(cands) >= 320:
            break
    for z, o in cands:
        nb = (bits ^ (1 << z) ^ (1 << o)) & fullmask
        m = bits_max_leq(nb, n, cur - 1)
        if m < cur:
            return nb, m
    return None


def bitset_search(n, h0, deadline, seed=42):
    rng = np.random.RandomState(seed)
    fullmask = (1 << n) - 1
    if h0 is not None and len(h0) == n:
        bits = bits_from_h(h0, n) & fullmask
    else:
        bits = random_balanced_bits(n, rng) & fullmask
    target = n // 2
    while bits.bit_count() > target:
        i = int(rng.randint(n))
        if (bits >> i) & 1:
            bits ^= 1 << i
    while bits.bit_count() < target:
        i = int(rng.randint(n))
        if ((bits >> i) & 1) == 0:
            bits ^= 1 << i
    bits &= fullmask

    cur = bits_max_leq(bits, n, n)
    best = cur
    bestbits = bits
    T = 0.55
    it = 0
    last = 0

    while time.time() < deadline:
        it += 1
        if it % 2500 == 0:
            T *= 0.985
        if cur > best + 3 and it % 700 == 0:
            bits = bestbits
            for _ in range(5):
                while True:
                    a = int(rng.randint(n))
                    if ((bits >> a) & 1) == 0:
                        break
                while True:
                    b = int(rng.randint(n))
                    if (bits >> b) & 1:
                        break
                bits ^= (1 << a) | (1 << b)
                bits &= fullmask
            cur = bits_max_leq(bits, n, n)

        while True:
            z = int(rng.randint(n))
            if ((bits >> z) & 1) == 0:
                break
        while True:
            o = int(rng.randint(n))
            if (bits >> o) & 1:
                break

        nb = (bits ^ (1 << z) ^ (1 << o)) & fullmask
        limit = cur + 1
        nm = bits_max_leq(nb, n, limit)

        if nm <= cur:
            bits = nb
            cur = nm
            last = it
            if cur < best:
                best = cur
                bestbits = bits
        elif nm <= limit and rng.random() < math.exp(-(nm - cur) / max(T, 1e-3)):
            bits = nb
            cur = nm

        if it - last > 140:
            res = targeted_binary_improve(bits, n, fullmask)
            if res is not None:
                bits, cur = res
                last = it
                if cur < best:
                    best = cur
                    bestbits = bits

    return array_from_bits(bestbits, n), float(2.0 * best / n)


def resample_h(hold, n_new):
    hold = np.asarray(hold, dtype=float)
    if len(hold) == n_new:
        return project_feasible(hold, n_new)
    x0 = np.linspace(0.0, 2.0, len(hold), endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_new, endpoint=False)
    hnew = np.interp(x1, x0, hold, left=hold[0], right=hold[-1])
    return project_feasible(hnew, n_new)


def random_smooth_h(n, rng, level=0.32):
    r = rng.randn(n)
    r = np.cumsum(np.cumsum(r))
    r -= r.mean()
    sd = float(r.std())
    if sd > 1e-12:
        r /= sd
    return project_feasible(0.5 + level * r, n)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    budget = max(30.0, float(budget_s))
    deadline = t_start + budget - 3.0
    rng = np.random.RandomState(seed)

    hparent = None
    try:
        hparent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        hparent = None

    if hparent is None or len(hparent) < 24:
        hparent = random_smooth_h(250, rng, 0.33)
    if len(hparent) % 2 == 1:
        hparent = resample_h(hparent, len(hparent) + 1)

    n0 = len(hparent)
    hparent = project_feasible(hparent, n0)

    best_h = hparent.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    raw_ns = [n0, n0 - 50, n0 - 100, n0 + 50, n0 + 100, 128, 160, 200, 250, 320]
    ns = []
    for nn in raw_ns:
        ee = int((nn // 2) * 2)
        if 80 <= ee <= 520 and ee not in ns:
            ns.append(ee)

    phase1 = t_start + 0.36 * (budget - 10.0)
    hb, cb = smooth_optimize(best_h, best_n, phase1, seed=seed, alphas=(1200.0, 6000.0))
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        best_n = len(hb)

    hb, cb = minimax_refine(best_h, best_n, deadline=phase1, seed=seed + 17)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        best_n = len(hb)

    if best_n % 2 == 0:
        bit_deadline = t_start + 0.58 * (budget - 10.0)
        if time.time() < bit_deadline:
            hb, cb = bitset_search(best_n, best_h, bit_deadline, seed=seed + 31)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = len(hb)

    phase2 = t_start + 0.86 * (budget - 10.0)
    others = [nn for nn in ns if nn != best_n]
    handled = 0
    for nn in others[:5]:
        if time.time() > phase2 - 25.0:
            break
        handled += 1
        rem = max(12.0, phase2 - time.time())
        sl = min(75.0, max(12.0, rem / max(1, len(others) - handled + 1)))
        d_n = time.time() + sl
        hstart = resample_h(hparent, nn)

        if sl > 45.0 and time.time() < d_n - 18.0:
            hb, cb = smooth_optimize(
                hstart,
                nn,
                time.time() + sl * 0.32,
                seed=seed + 1000 + nn,
                alphas=(5000.0, 20000.0),
            )
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = nn
                hstart = hb.copy()

        if nn % 2 == 0 and time.time() < d_n - 6.0:
            hb, cb = bitset_search(nn, hstart, d_n, seed=seed + 7000 + nn)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = nn

    while time.time() < deadline - 8.0:
        hb, cb = smooth_optimize(
            best_h,
            best_n,
            time.time() + min(45.0, deadline - time.time() - 8.0),
            seed=seed + 90000,
            alphas=(15000.0, 80000.0),
        )
        if cb < best_c - 1e-12:
            best_c = cb
            best_h = hb.copy()
            best_n = nn = len(hb)
        else:
            break

    if time.time() < deadline - 3.0:
        hb, cb = minimax_refine(best_h, best_n, deadline=deadline - 1.0, seed=seed + 55555)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = len(hb)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
