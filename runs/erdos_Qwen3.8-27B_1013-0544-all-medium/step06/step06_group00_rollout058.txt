def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    """Compute gradient of log-sum-exp approximation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx  # length 2n-1
    # log-sum-exp
    max_g = g.max()
    exp_g = np.exp(alpha * (g - max_g))
    p = exp_g / exp_g.sum()  # probability weights
    # p has length 2n-1, indexed k=0..2n-2, shift s_k = k-(n-1)
    grad = np.zeros(n)
    for m in range(n):
        # First term: sum p[k] * (1-h[m+s_k]) where 0 <= m+s_k < n
        # s_k = k-(n-1), so 0 <= m+k-(n-1) < n => n-1-m <= k <= 2n-2-m
        lo = max(0, n - 1 - m)
        hi = min(2 * n - 2, 2 * n - 2 - m)
        if lo <= hi:
            k_vals = np.arange(lo, hi + 1)
            s_vals = k_vals - (n - 1)
            grad[m] += p[lo:hi+1].sum() - np.dot(p[lo:hi+1], h[m + s_vals])
        # Second term: -sum p[k] * h[m-s_k] where 0 <= m-s_k < n

        lo2 = max(0, m)
        hi2 = min(2 * n - 2, m + n - 1)
        if lo2 <= hi2:
            k_vals = np.arange(lo2, hi2 + 1)
            s_vals = k_vals - (n - 1)
            grad[m] -= np.dot(p[lo2:hi2+1], h[m - s_vals])
    return dx * grad

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_lse(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    mg = g.max()
    eg = np.exp(alpha * (g - mg))
    p = eg / eg.sum()
    grad = np.zeros(n)
    for m in range(n):
        lo = max(0, n - 1 - m)
        hi = min(2 * n - 2, 2 * n - 2 - m)
        if lo <= hi:
            hs = m + lo - (n - 1)
            he = m + hi - (n - 1)
            grad[m] += p[lo:hi + 1].sum() - np.dot(p[lo:hi + 1], h[hs:he + 1])
        lo2 = max(0, m + 1 - n)
        hi2 = min(2 * n - 2, m + n - 1)
        if lo2 <= hi2:
            hs2 = m - hi2 + (n - 1)
            he2 = m - lo2 + (n - 1)
            grad[m] -= np.dot(p[lo2:hi2 + 1], h[hs2:he2 + 1][::-1])
    return dx * grad

def adam_opt(h, n, deadline, alpha_start=50, alpha_end=5000000, n_iter=20000, lr=0.005):
    """ADAM optimization with increasing alpha."""
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m1 = np.zeros(n)
    v1 = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    n_alpha_steps = min(n_iter, max(1, n_iter // 8))
    alphas = np.geomspace(alpha_start, alpha_end, n_alpha_steps)
    iters_per_alpha = max(1, n_iter // len(alphas))
    
    for ai in range(len(alphas)):
        alpha = alphas[ai]
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            t += 1
            g = grad_lse(h, n, alpha)
            m1 = b1 * m1 + (1 - b1) * g
            v1 = b2 * v1 + (1 - b2) * g * g
            m1_hat = m1 / (1 - b1 ** t)
            v1_hat = v1 / (1 - b2 ** t)
            h_new = h - lr * m1_hat / (np.sqrt(v1_hat) + eps)
            h = project_feasible(h_new, n)
            if it % 100 == 0:
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def grad_lse(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    mg = g.max()
    eg = np.exp(alpha * (g - mg))
    p = eg / eg.sum()
    grad = np.zeros(n)
    for m in range(n):
        # T1
        lo = max(0, n - 1 - m)
        hi = min(2 * n - 2, 2 * n - 2 - m)
        if lo <= hi:
            hs = m + lo - (n - 1)
            he = m + hi - (n - 1)
            grad[m] += p[lo:hi + 1].sum() - np.dot(p[lo:hi + 1], h[hs:he + 1])
        # T2
        lo2 = max(0, m)
        hi2 = min(2 * n - 2, m + n - 1)
        if lo2 <= hi2:
            hs2 = m - hi2 + (n - 1)
            he2 = m - lo2 + (n - 1)
            grad[m] -= np.dot(p[lo2:hi2 + 1], h[hs2:he2 + 1][::-1])
    return dx * grad

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_lse(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    mg = g.max()
    eg = np.exp(alpha * (g - mg))
    p = eg / eg.sum()
    grad = np.zeros(n)
    for m in range(n):
        lo = max(0, n - 1 - m)
        hi = min(2 * n - 2, 2 * n - 2 - m)
        if lo <= hi:
            hs = m + lo - (n - 1)
            he = m + hi - (n - 1)
            grad[m] += p[lo:hi + 1].sum() - np.dot(p[lo:hi + 1], h[hs:he + 1])
        lo2 = m
        hi2 = min(2 * n - 2, m + n - 1)
        if lo2 <= hi2:
            hs2 = m - hi2 + (n - 1)
            he2 = m - lo2 + (n - 1)
            grad[m] -= np.dot(p[lo2:hi2 + 1], h[hs2:he2 + 1][::-1])
    return dx * grad

def adam_opt(h, n, deadline, alpha_seq, iters_per_alpha, lr=0.01, b1=0.9, b2=0.999):
    h = project_feasible(h.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    m1, v1, t = np.zeros(n), np.zeros(n), 0
    eps = 1e-8
    for alpha in alpha_seq:
        for _ in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            t += 1
            g = grad_lse(h, n, alpha)
            m1 = b1 * m1 + (1 - b1) * g
            v1 = b2 * v1 + (1 - b2) * g * g
            mh = m1 / (1 - b1 ** t)
            vh = v1 / (1 - b2 ** t)
            h = project_feasible(h - lr * mh / (np.sqrt(vh) + eps), n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range; msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range; msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range; msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range; msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1: delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1: delta[k2] += 1.0

                    corr += delta
                    cur_c = nc
                    ones.remove(j); zeros.remove(i)
                    ones.append(i); zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:

        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c
