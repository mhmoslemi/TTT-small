def greedy_local_search(h, n, deadline, max_sweeps=10000):
    """Greedy local search for binary h. Returns (best_h, best_c)."""
    dx = 2.0 / n
    h = binarize(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        # Try all pairs, find best improvement
        best_delta = 0
        best_i, best_j = -1, -1
        for i in zeros:
            for j in ones:
                h[i], h[j] = 1.0, 0.0
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = new_corr.max()
                if new_max < cur_max:
                    best_delta = new_max - cur_max
                    best_i, best_j = i, j
                h[i], h[j] = 0.0, 1.0
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if best_i >= 0:
            h[best_i], h[best_j] = 1.0, 0.0
            cur_max = cur_max + best_delta
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            improved = True
        if not improved:
            break
    
    return best_h, best_max * dx

def ils(h, n, deadline, rng):
    dx = 2.0 / n
    h = binarize(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    
    while time.time() < deadline:
        # Local search phase: first-improvement with random ordering
        n_iter = 0
        max_iter = 50 * n
        while n_iter < max_iter:
            n_iter += 1
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            i = int(rng.choice(zeros))
            j = int(rng.choice(ones))
            h[i], h[j] = 1.0, 0.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(new_corr.max())
            if new_max <= cur_max:
                corr = new_corr
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            else:
                h[i], h[j] = 0.0, 1.0
                # Revert
                # We didn't update corr, so it's still valid
        # Perturbation phase
        n_pert = rng.randint(5, 15)

            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            i = int(rng.choice(zeros))
            j = int(rng.choice(ones))
            h[i], h[j] = 1.0, 0.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(corr.max())
