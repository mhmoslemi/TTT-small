        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        new_corr = corr[:, None, :] + C_full[zeros_idx, None, :] - C_full[ones_idx, None, :]
        maxes = new_corr.max(axis=2)
        flat_best = maxes.argmin()
        ri = flat_best // len(ones_idx)
        rj = flat_best % len(ones_idx)
        if maxes[ri, rj] >= cur_max - 1e-9:
            break
        i = zeros_idx[ri]
        j = ones_idx[rj]
        h[i] = 1.0
        h[j] = 0.0
        corr = corr + C_full[i] - C_full[j]
        cur_max = corr.max()
        stale.add(i)
        stale.add(j)
        if j in zeros_idx:
            pass
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    if stale:
        for i in stale:
            C_full[i] = (1.0 - h[i]) * mask[i]
    return best_h, float(best_max * dx)

def sa_binary_search(h, n, deadline, max_iters=50000):
    m = 2 * n - 1
    lags = np.arange(m) - (n - 1)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    mask = (lags[None, :] >= 0) & (lags[None, :] < n)
    C_full = (1.0 - h[None, :]) * mask
    stale = set()
    rng = np.random.RandomState(42)
    T0 = 2.0
    T_end = 0.05
    T = T0
    cooling = (T_end / T0) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        for i in range(n):
            if i in stale:
                C_full[i] = (1.0 - h[i]) * mask[i]
                stale.discard(i)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        d = C_full[i] - C_full[j]
        new_corr = corr + d
        new_max = new_corr.max()
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_max = new_max
            stale.add(i)
            stale.add(j)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        T *= cooling
    if stale:
        for i in stale:
            C_full[i] = (1.0 - h[i]) * mask[i]
    return best_h, float(best_max * dx)

def slsqp_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [500, 2000, 10000]:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1.0 - x, mode='full')
            gm = g.max()
            sh = alpha * (g - gm)
            sm = sh.max()
            return (gm + sm + np.log(np.exp(sh - sm).sum())) / alpha * (2.0 / n)
        try:
            res = minimize(obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                          options={'maxiter': 300, 'ftol': 1e-15})
            hc = project_feasible(res.x, n)
            cc = compute_c5(hc, n)
            if cc < best_c:
                best_c = cc
                best_h = hc
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h_bin = binarize(h0, n)
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    p1 = t0 + 0.45 * (budget_s - 5)
    hb, cb = fast_binary_search(best_h.copy(), n, p1)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    p2 = t0 + 0.60 * (budget_s - 5)
    if time.time() < p2:
        ha, ca = sa_binary_search(best_h.copy(), n, p2)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()
    p3 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p3:
        hc, cc = slsqp_optimize(best_h.copy(), n, p3)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
    p4 = dl - 3
    if time.time() < p4:
        h_bin2 = binarize(best_h, n)
        hb2, cb2 = fast_binary_search(h_bin2, n, p4)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    if time.time() < dl - 2:
        hf, cf = slsqp_optimize(best_h.copy(), n, dl - 2)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
