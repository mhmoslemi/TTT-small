def subgradient_step(h, n, step_size, deadline):
    """
    Subgradient method for the max objective.
    For the active lag, compute subgradient and take a step.
    """
    c = compute_all_lags(h, n)
    c_max = c.max()
    # Get all active lags (within epsilon of max)
    active = np.where(c >= c_max - 1e-6)[0]
    
    # Compute average subgradient over active lags
    grad = np.zeros(n)
    dx = 2.0 / n
    for k_idx in active:
        lag = int(k_idx - (n - 1))
        if lag == 0:
            g = (1 - 2 * h) * dx
        else:
            m = abs(lag)
            g = np.zeros(n)
            if lag > 0:
                g[:n-m] -= h[m:] * dx
                g[m:] += (1 - h[:n-m]) * dx
            else:
                g[:n-m] += (1 - h[m:]) * dx
                g[m:] -= h[:n-m] * dx
        grad += g / len(active)
    
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                h_new = h.copy()
                if lag > 0:
                    h_new[pos] -= delta
                    h_new[pos + m] += delta
                else:
                    h_new[pos + m] -= delta
                    h_new[pos] += delta
                if min(h_new.min(), 1 - h_new.max()) < -1e-12:
                    continue
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    h = h_new
                    best_c = c_new
                    best_h = h.copy()
                    moved = True
                    break
            if not moved:
                stuck += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Aggressive targeted local search
    p1_end = t0 + min(350, (budget_s - 5) * 0.35)
    best_h, best_c = targeted_local_search(best_h, n, p1_end)

    # Phase 2: SLSQP polish
    if time.time() < deadline - 120:
        for alpha in [5000, 20000, 100000]:
            if time.time() > deadline - 120:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 3: Multi-lag balanced search
    if time.time() < deadline - 60:
        h, c_h = multi_lag_balanced_search(best_h, n, deadline - 60)
        if c_h < best_c:
            best_c = c_h
            best_h = h.copy()

    # Phase 4: More local search
    if time.time() < deadline - 30:
        best_h, best_c = targeted_local_search(best_h, n, deadline - 30)

    # Phase 5: Subgradient polish
    if time.time() < deadline - 15:
        h, c_h = subgradient_polish(best_h, n, deadline - 15)
        if c_h < best_c:
            best_c = c_h
            best_h = h.copy()

    # Phase 6: Final SLSQP + local search
    if time.time() < deadline - 5:
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        best_h, best_c = targeted_local_search(best_h, n, deadline - 1)

    # Phase 7: Try perturbed restart if time remains
    if time.time() < deadline - 20:
        for scale in [0.01, 0.005]:
            if time.time() > deadline - 10:
                break
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            h_opt, c_opt = targeted_local_search(hp, n, deadline - 5)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
