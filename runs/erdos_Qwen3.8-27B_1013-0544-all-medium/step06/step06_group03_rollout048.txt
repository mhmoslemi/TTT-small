def compute_all_lags(h, n):

            if tot < 1e-14: break

def _lag_gradient(h_curr, n, lag):
    """Gradient of the lag correlation w.r.t. h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad

        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(x, n, alpha, lags):
    """Gradient of smooth max of selected lags."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[lags]
    m = active.max()
    shifted = alpha * (active - m)

    else:

        lag = k - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (d / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (d / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m >= n: return g
    if m == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    if lag > 0:
        g[:n-m] -= h[m:]
        g[m:] += 1.0 - h[:n-m]
    else:
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

def lp_cutting_plane(h, n, deadline, max_iters=200, init_top=15):
    """Cutting plane LP with adaptive active set."""
    dx = 2.0 / n
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)
    active = set()
    trust = 0.4
    for it in range(max_iters):
        if time.time() > deadline: break
        c = compute_all_lags(h_curr, n)
        top_m = min(init_top + it // 5, 2*n-1)
        top_idx = np.argsort(c)[-top_m:]
        for k in top_idx:
            active.add(k)
        if len(active) > 40:
            active = set(np.argsort(c)[-40:])
        hv = cp.Variable(n)
        tv = cp.Variable()
        cons = [cp.sum(hv) == n/2.0, hv >= 0, hv <= 1,
                cp.norm(hv - h_curr, 1) <= trust]
        for k in active:
            lag = k - (n-1)
            ck = c[k]
            gr = _lag_grad(h_curr, n, lag)
            cons.append(ck + dx * (gr @ (hv - h_curr)) <= tv)
        prob = cp.Problem(cp.Minimize(tv), cons)
        solved = False
        for solver in [cp.GLOP, cp.ECOS, cp.CBC]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        if not solved or hv.value is None:
            break
        h_new = project_feasible(hv.value, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10: break
        h_curr = h_new
        trust = max(0.01, trust * 0.82)
    return best_h, best_c

def _smooth_obj(x, n, alpha, lags):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    a = c[lags]
    m = a.max()
    s = alpha * (a - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _smooth_grad(x, n, alpha, lags):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    a = c[lags]
    m = a.max()
    s = alpha * (a - m)
    mx = s.max()
    if mx > 400:
        e = np.exp(s - mx)
    else:
        e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for i, k in enumerate(lags):
        if w[i] < 1e-12: continue
        lag = k - (n-1)
        g += w[i] * dx * _lag_grad(x, n, lag)
    return g

def _sc(x, n): return float(x.sum() - n/2.0)
def _scj(x, n): return np.ones(n)

def slsqp_solve(h, n, alpha, lags, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type':'eq', 'fun': _sc, 'jac': _scj, 'args':(n,)}]
    try:
        res = minimize(_smooth_obj, h, jac=lambda x: _smooth_grad(x, n, alpha, lags),
                       args=(), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                    if cn < best_c - 1e-14:

    # Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        c = compute_all_lags(best_h, n)
        top = np.argsort(c)[-10:]
        h_new = slsqp_solve(best_h, n, 5000000, top, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
