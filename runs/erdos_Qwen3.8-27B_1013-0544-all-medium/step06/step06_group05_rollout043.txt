import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    sorted_h = np.sort(h)[::-1]
    thresh = sorted_h[n // 2]
    h_bin = (h >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        excess = np.where(h_bin == 1)[0]
        h_bin[excess[n_ones - n // 2:]] = 0.0
    elif n_ones < n // 2:
        deficit = np.where(h_bin == 0)[0]
        h_bin[deficit[:n // 2 - n_ones]] = 1.0
    return h_bin

    zeros = np.where(h == 0)[0].copy()

            corr = new_corr
            zeros[ii] = j
            ones[jj] = i
            if cur_max < best_max:

                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    return best_h, float(best_max * 2.0 / n)

def slsqp_polish(h, n, deadline):
    dx = 2.0 / n
    alpha = 5000.0
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,

    except Exception:
        return h
