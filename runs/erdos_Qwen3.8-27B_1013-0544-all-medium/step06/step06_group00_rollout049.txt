```python
def subgrad_descent_multi(h, n, deadline, lr_init=0.3, momentum=0.9, top_k=3):
    dx = 2.0 / n
    h = h.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    no_improve = 0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1-h, mode='full') * dx
        # Use top-k lags with exponential weighting
        sorted_idx = np.argsort(corr)[::-1]
        weights = np.exp(np.arange(top_k) * -1.0)  # 1, e^-1, e^-2, ...
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, idx in zip(weights, sorted_idx[:top_k]):
            j = idx - (n - 1)
            g += w * grad_g_j(h, j, n, dx)
        
        vel = momentum * vel + (1.0 - momentum) * g
        h = project_feasible(h - lr * vel, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                lr *= 0.9
                vel *= 0.3
                no_improve = 0
                if lr < 0.001:
                    break
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_N = 400
_ALPHA = 1000.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_g_j(h, j, n, dx):
    grad = np.zeros(n)
    if j == 0:
        return dx * (1.0 - 2.0 * h)
    elif j > 0:
        grad[:n-j] += dx * (1.0 - h[j:])
        grad[j:] += -dx * h[:n-j]
    else:
        jj = -j
        grad[jj:] += dx * (1.0 - h[:n-jj])
        grad[:n-jj] += -dx * h[jj:]
    return grad

def subgrad_descent(h, n, deadline, lr_init=0.4, momentum=0.9, top_k=1):
    dx = 2.0 / n
    h = h.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    no_improve = 0
    wlog = np.exp(-np.arange(min(top_k, 10)).astype(float))
    wlog /= wlog.sum()
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        si = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for w, idx in zip(wlog, si):
            j = int(idx) - (n - 1)
            g += w * grad_g_j(h, j, n, dx)
        vel = momentum * vel + (1.0 - momentum) * g
        h = project_feasible(h - lr * vel, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1500:
                lr *= 0.92
                vel *= 0.4
                no_improve = 0
                if lr < 0.0005:
                    break
    return best_h, best_c

def _smooth_obj(x):
    dx = 2.0 / _N
    alpha = _ALPHA
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    if sm > 700:
        lse = m + sm / alpha
    else:
        lse = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    penalty = 200.0 * (x.sum() - _N / 2.0) ** 2
    return lse + penalty

def lbfgsb_smooth(h, n, alpha, maxiter):
    global _N, _ALPHA
    _N = n
    _ALPHA = alpha
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    _N = n
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B smooth for interior solution
    p1 = t0 + 0.10 * total
    h = best_h.copy()
    for alpha in [100, 500, 2000, 10000]:
        if time.time() > p1 - 5:
            break
        h = lbfgsb_smooth(h, n, alpha, 1500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 2: Binary local search
    p2 = t0 + 0.18 * total
    hb, cb = binary_local_search(best_h, n, p2, max_sweeps=250)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 3: Subgradient with top-1 (fast, long)
    p3 = t0 + 0.45 * total
    h = best_h.copy()
    hg, cg = subgrad_descent(h, n, p3, lr_init=0.5, momentum=0.9, top_k=1)
    if cg < best_c:
        best_c, best_h = cg, hg.copy()
    # Phase 4: Subgradient with top-3 (smoother)
    p4 = t0 + 0.60 * total
    h = best_h.copy()
    hg, cg = subgrad_descent(h, n, p4, lr_init=0.3, momentum=0.92, top_k=3)
    if cg < best_c:
        best_c, best_h = cg, hg.copy()
    # Phase 5: L-BFGS-B polish
    p5 = t0 + 0.75 * total
    h = best_h.copy()
    for alpha in [5000, 50000, 500000, 5000000]:
        if time.time() > p5 - 5:
            break
        h = lbfgsb_smooth(h, n, alpha, 3000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 6: Multi-restart
    p6 = t0 + 0.92 * total
    for trial in range(12):
        if time.time() > p6 - 10:
            break
        noise_scale = 0.02 + 0.01 * trial
        h_trial = best_h + noise_scale * rng.randn(n)
        h_trial = project_feasible(h_trial, n)
        dl_t = min(t0 + total - 5, time.time() + 35)
        ht, ct = subgrad_descent(h_trial, n, dl_t, lr_init=0.4, momentum=0.88, top_k=2)
        if ct < best_c:
            ht2 = lbfgsb_smooth(ht, n, 200000, 1500)
            ct2 = compute_c5(ht2, n)
            if ct2 < best_c:
                best_c, best_h = ct2, ht2.copy()
            else:
                best_c, best_h = ct, ht.copy()
    # Phase 7: Final high-alpha polish
    h = best_h.copy()
    for alpha in [5000000, 50000000]:
        if time.time() > dl - 3:
            break
        h = lbfgsb_smooth(h, n, alpha, 5000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
