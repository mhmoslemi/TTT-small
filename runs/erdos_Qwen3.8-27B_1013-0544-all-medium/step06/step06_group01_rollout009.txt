def subgradient_step(h, n, lr):
    """Take a subgradient step on f(h) = max_k c_k * dx"""

    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    m = k_star  # correlation index
    k = m - (n - 1)  # lag

    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
