```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n, rng):
    """Convert to binary while maintaining sum = n/2"""
    target = n // 2
    h_sorted = np.sort(h)[::-1]
    # Take top n/2 as 1s
    indices = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[indices] = 1.0
    return b

def binary_local_search(b, n, deadline, max_sweeps=20):
    """Local search on binary sequences"""
    b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    best_c = compute_c5(b, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Random order for diversity
        rng_local = np.random.RandomState(sweep * 1000 + 42)
        one_order = ones[rng_local.permutation(len(ones))]
        zero_order = zeros[rng_local.permutation(len(zeros))]
        
        for i in one_order:
            if time.time() > deadline:
                break
            for j in zero_order:
                if i == j:

                # Try swap: b[i] = 0, b[j] = 1
                b[i] = 0
                b[j] = 1
                c_new = compute_c5(b, n)

                    improved = True

                    zeros = np.where(b == 0)[0]

                    b[i] = 1
                    b[j] = 0

    best_b = b.copy()
    best_c = compute_c5(b, n)
    T_start = 0.01
    T_end = 0.0001
    total_iters = int((deadline - time.time()) * iters_per_sec)

        T = T_start * (T_end / T_start) ** progress
```

```python
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr

    if lag > 0:

        grad[m:] -= h_curr[:n - m]
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    indices = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[indices] = 1.0
    return b

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def binary_local_search(b, n, deadline, rng, max_sweeps=50):
    b = b.copy()
    best_c = compute_c5(b, n)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        perm1 = rng.permutation(len(ones))
        perm0 = rng.permutation(len(zeros))
        
        for idx_i in perm1:
            if time.time() > deadline:
                break
            i = ones[idx_i]
            for idx_j in perm0:
                j = zeros[idx_j]
                b[i] = 0
                b[j] = 1
                c_new = compute_c5(b, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    improved = True
                    ones = np.where(b == 1)[0]
                    zeros = np.where(b == 0)[0]
                    perm1 = rng.permutation(len(ones))
                    perm0 = rng.permutation(len(zeros))
                    break  # restart inner loop with new sets
                else:
                    b[i] = 1
                    b[j] = 0
        if not improved:
            break
    
    return b, best_c

def subgradient_descent(h, n, deadline, max_iters=5000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_size = 0.01
    decay = 0.999
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_idx = int(np.argmax(c))
        lag = k_idx - (n - 1)
        g = lag_gradient(h, n, lag)
        h_new = h - step_size * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        step_size *= decay
    
    return best_h, best_c

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    if top_idx is not None:
        c = np.correlate(x, 1 - x, mode='full') * dx
        c = c[top_idx]
    else:
        c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_refine(h, n, deadline, top_m=8):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for alpha in [5000, 50000, 200000]:
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        obj = lambda x: smooth_max_obj(x, n, alpha, top_idx)
        try:
            res = minimize(obj, h_curr, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
                h_curr = h_new
        except Exception:
            pass
    return best_h, best_c

def fine_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(h, n)
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    m = abs(lag)
    if m == 0 or m >= n:
        return h, best_c
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            if lag > 0:
                pairs = [(i, i + m) for i in range(n - m)]
            else:
                pairs = [(i + m, i) for i in range(n - m)]
            for (a, b) in pairs:
                if time.time() > deadline:
                    break
                if h[a] > h[b]:
                    delta = min(d, (h[a] - h[b]) * 0.5, h[a], 1.0 - h[b])
                    if delta < 1e-10:
                        continue
                    h[a] -= delta
                    h[b] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[a] += delta
                        h[b] -= delta
                elif h[b] > h[a]:
                    delta = min(d, (h[b] - h[a]) * 0.5, h[b], 1.0 - h[a])
                    if delta < 1e-10:
                        continue
                    h[a] += delta
                    h[b] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[a] -= delta
                        h[b] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Binarize and do binary local search
    if time.time() < deadline - 20:
        b = binarize(h, n)
        b_end = min(deadline - 15, time.time() + 30)
        b, c_b = binary_local_search(b, n, b_end, rng)
        if c_b < best_c:
            best_c = c_b
            best_h = b.copy()
    
    # Phase 2: Subgradient descent (continuous)
    if time.time() < deadline - 15:
        h_end = min(deadline - 10, time.time() + 20)
        h_sg, c_sg = subgradient_descent(best_h, n, h_end)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    
    # Phase 3: SLSQP refinement
    if time.time() < deadline - 10:
        h_new = slsqp_refine(best_h, n, alpha=50000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: Active set refinement
    if time.time() < deadline - 8:
        h_as, c_as = active_set_refine(best_h, n, deadline - 5)

    if time.time() < deadline - 2:

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
```

```python
def binary_swap_delta(b, n, i, j, lags):
    """Compute change in c(L) for each L in lags when swapping b[i]:1→0, b[j]:0→1"""
    deltas = []
    for L in lags:
        if L == 0:
            # c(0) = sum h[i](1-h[i]) = 0 for binary, always 0
            deltas.append(0.0)
            continue
        L = abs(L)
        delta = 0.0
        # i changes from 1 to 0
        if i + L <= n - 1:
            delta += b[i + L]  # was contributing 0, now contributes b[i+L]
        if i >= L:
            delta -= (1 - b[i - L])  # was contributing (1-b[i-L]), now 0
        # j changes from 0 to 1
        if j + L <= n - 1:
            delta -= b[j + L]  # was contributing b[j+L], now 0
        if j >= L:
            delta += (1 - b[j - L])  # was contributing 0, now (1-b[j-L])
        deltas.append(delta)
    return deltas
```

- i as t: h[i]*(1-h[i+M]) → 0*(1-h[i+M]) = 0. Change: -(1-h[i+M]) if i+M <= n-1
- i as t+M (t=i-M): h[i-M]*(1-h[i]) → h[i-M]*(1-0) = h[i-M]. Change: +h[i-M] if i >= M

Position j (0→1):
- j as t: h[j]*(1-h[j+M]) → 1*(1-h[j+M]) = 1-h[j+M]. Change: +(1-h[j+M]) if j+M <= n-1
- j as t+M (t=j-M): h[j-M]*(1-h[j]) → h[j-M]*(1-1) = 0. Change: -h[j-M] if j >= M

So for L < 0, M = -L:
Δc(L) = -(1-b[i+M])*1[i+M<=n-1] + b[i-M]*1[i>=M] + (1-b[j+M])*1[j+M<=n-1] - b[j-M]*1[j>=M]

If L > 0: Δc = b[i+M]*1[i+M<n] - (1-b[i-M])*1[i>=M] - b[j+M]*1[j+M<n] + (1-b[j-M])*1[j>=M]

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    indices = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[indices] = 1.0
    return b

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def binary_local_search(b, n, deadline, rng, max_sweeps=30):
    b = b.copy()
    best_c = compute_c5(b, n)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        perm1 = rng.permutation(len(ones))
        perm0 = rng.permutation(len(zeros))
        for idx_i in perm1:
            if time.time() > deadline:
                break
            i = ones[idx_i]
            for idx_j in perm0:
                j = zeros[idx_j]
                b[i] = 0
                b[j] = 1
                c_new = compute_c5(b, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    improved = True
                    ones = np.where(b == 1)[0]
                    zeros = np.where(b == 0)[0]
                    perm1 = rng.permutation(len(ones))
                    perm0 = rng.permutation(len(zeros))
                    break
                else:
                    b[i] = 1
                    b[j] = 0
        if not improved:
            break
    return b, best_c

def multi_subgradient(h, n, deadline, top_m=5, max_iters=20000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_size = 0.008
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            g += lag_gradient(h, n, lag)
        g /= len(top_idx)
        h_new = h - step_size * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 0 and it > 0:
            step_size *= 0.97
    return best_h, best_c

def obj_func(x, n, alpha, top_idx):
    dx = 2.0 / n
    if top_idx is not None:
        c = np.correlate(x, 1 - x, mode='full') * dx
        c = c[top_idx]
    else:
        c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=3000, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_func, h, args=(n, alpha, top_idx), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_refine(h, n, deadline):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for alpha, top_m in [(5000, 5), (20000, 8), (100000, 12)]:
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        h_new = slsqp_refine(h_curr, n, alpha, maxiter=3000, top_idx=top_idx)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
    return best_h, best_c

def fine_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-3:]
            for k_idx in top_idx:
                if time.time() > deadline:
                    break
                lag = k_idx - (n - 1)
                m = abs(lag)
                if m == 0 or m >= n:
                    continue
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if lag > 0:
                        hi, hj = h[i + m], h[i]
                    else:
                        hi, hj = h[i], h[i + m]
                    if hi > hj:
                        delta = min(d, (hi - hj) * 0.5, hi, 1.0 - hj)
                        if delta < 1e-10:
                            continue
                        if lag > 0:
                            h[i + m] -= delta
                            h[i] += delta
                        else:
                            h[i] -= delta
                            h[i + m] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            if lag > 0:
                                h[i + m] += delta
                                h[i] -= delta
                            else:
                                h[i] += delta
                                h[i + m] -= delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 25:
        b = binarize(h, n)
        b_end = min(deadline - 20, time.time() + 25)
        b, c_b = binary_local_search(b, n, b_end, rng)
        if c_b < best_c:
            best_c = c_b
            best_h = b.copy()

    if time.time() < deadline - 15:
        sg_end = min(deadline - 12, time.time() + 18)
        h_sg, c_sg = multi_subgradient(best_h, n, sg_end, top_m=5)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    if time.time() < deadline - 10:
        h_new = slsqp_refine(best_h, n, alpha=30000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 8:
        h_as, c_as = active_set_refine(best_h, n, deadline - 5)
        if c_as < best_c:
            best_c = c_as
            best_h = h_as

    if time.time() < deadline - 3:
        best_h, best_c = fine_local_search(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
```
