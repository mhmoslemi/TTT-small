def compute_c_all(h, n):
    dx=2.0/float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full') * (2.0/float(n))))

def smooth_value_from_c(c, alpha):
    if alpha <= 1.0:
       return float(np.max(c))
    m = float(np.max(c))
    sh = alpha * (c - m)
    # sh max = 0
    s = float(np.exp(sh).sum())
    return m + float(np.log(s) / alpha)

def smooth_grad(h, n, alpha):
    c = compute_c_all(h,n)
    val = smooth_value_from_c(c, alpha)
    m = float(np.max(c))
    if alpha <= 1.0:
        idx = int(np.argmax(c)); w = np.zeros(c.size); w[idx]=1.0
    else:
        sh = alpha*(c-m)
        ew = np.exp(sh)
        s = ew.sum()
        w = ew / s
    N=n
    P = np.zeros(N); Nw=np.zeros(N); S=np.zeros(N)
    P[1:] = w[N:]
    Nw[1:] = w[:N-1][::-1]
    S[1:] = P[1:] + Nw[1:]
    prefP = np.zeros(N)
    if N>1: prefP[1:] = np.cumsum(P[1:])
    cumN = np.zeros(N)
    if N>1: cumN[1:] = np.cumsum(Nw[1:])
    prefN = cumN[::-1]
    Psum = np.convolve(S, h, mode='full')[:N]

def slsqp_smooth(h,n,alpha,maxiter=200,ftol=1e-12):
    h0=project_feasible(h,n)
    try:
       bounds=[(0.0,1.0)]*n
       cons=[{'type':'eq','fun':_sum_con,'jac':_sum_con_jac,'args':(n,)}]
       res=minimize(_smooth_fun,h0,jac=_smooth_jac,args=(n,alpha),method='SLSQP',bounds=bounds,constraints=cons,options={'maxiter':int(maxiter),'ftol':float(ftol)})
       if res.x is not None:
          hn=project_feasible(res.x,n)
          if np.all(np.isfinite(hn)):
             return hn
    except Exception:
       pass
    return h0

import numpy as np
import time
from scipy.optimize import minimize


def c_all(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))


def c5(h, n):
    return float(np.max(c_all(h, n)))


def smooth_value(c, alpha):
    m = float(c.max())
    if alpha <= 1.0:
        return m
    s = float(np.exp(alpha * (c - m)).sum())
    return m + float(np.log(s) / alpha)


def smooth_grad(h, n, alpha):
    cc = c_all(h, n)
    val = smooth_value(cc, alpha)
    N = n
    m = float(cc.max())
    if alpha <= 1.0:
        w = np.zeros(cc.shape, dtype=float)
        w[int(cc.argmax())] = 1.0
    else:
        ew = np.exp(alpha * (cc - m))
        w = ew / ew.sum()
    P = np.zeros(N)
    Nm = np.zeros(N)
    S = np.zeros(N)
    P[1:] = w[N:]
    Nm[1:] = w[:N - 1][::-1]
    S[1:] = P[1:] + Nm[1:]
    pp = np.zeros(N)
    pp[1:] = np.cumsum(P[1:])
    cn = np.zeros(N)
    cn[1:] = np.cumsum(Nm[1:])
    pn = cn[::-1]
    psum = np.convolve(S, h, mode='full')[:N]
    fsum = np.correlate(h, S, mode='full')[N - 1:2 * N - 1]
    g = (w[N - 1] * (1.0 - 2.0 * h) + pp + pn - psum - fsum) * (2.0 / float(N))
    return val, g


def project(h, n, iters=60):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    if abs(float(h.sum()) - target) > 1e-13:
        lo = -float(h.max()) - 1.0
        hi = 1.0 - float(h.min()) + 1.0
        for _ in range(iters):
            mid = 0.5 * (lo + hi)
            s = float(np.clip(h + mid, 0.0, 1.0).sum())
            if s < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            break
        add = diff / idx.size
        if diff > 0 and add > (1.0 - h[idx]).min():
            add = (1.0 - h[idx]).min()
        if diff < 0 and add < -h[idx].min():
            add = -h[idx].min()
        h[idx] += add
    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        sel = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
        idx = int(np.argmax(sel))
        v = h[idx] + diff
        if 0.0 <= v <= 1.0:
            h[idx] = v
    return np.clip(h, 0.0, 1.0)

    return float(np.sum(x) - 0.5 * n)


def sum_con_jac(x, n):
    return np.ones(n, dtype=float)


def smooth_fun(x, n, alpha):
    return smooth_value(c_all(x, n), alpha)


def smooth_jac(x, n, alpha):
    return smooth_grad(x, n, alpha)[1]


def slsqp_s(h, n, alpha, maxiter=100, ftol=1e-12):
    h0 = project(h, n)
    try:
        res = minimize(
            smooth_fun, h0, jac=smooth_jac, args=(n, alpha), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}],
            options={'maxiter': int(maxiter), 'ftol': float(ftol)}
        )
        if res.x is not None:
            hn = project(res.x, n)
            if np.all(np.isfinite(hn)):

    except Exception:
        pass
    return h0


def lag_val(h, k, n):
    dx = 2.0 / float(n)
    idx = int(k) - (n - 1)
    if idx == 0:
        return float(np.dot(h, 1.0 - h) * dx)
    if idx > 0:
        return float(np.dot(h[idx:], 1.0 - h[:-idx]) * dx)
    m = -idx
    return float(np.dot(h[:n - m], 1.0 - h[m:]) * dx)


def lag_grad(h, k, n):
    idx = int(k) - (n - 1)
    g = np.zeros(n, dtype=float)
    if idx == 0:
        return 1.0 - 2.0 * h
    if idx > 0:
        m = idx
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    else:
        m = -idx
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g


def t_fun(x, n):
    return float(x[n])


def t_jac(x, n):
    g = np.zeros(n + 1, dtype=float)
    g[n] = 1.0
    return g


def sum_con_x(x, n):
    return float(np.sum(x[:n]) - 0.5 * n)


def sum_con_x_jac(x, n):
    g = np.zeros(n + 1, dtype=float)
    g[:n] = 1.0
    return g


def act_fun(x, k, n):
    return float(x[n] - lag_val(x[:n], k, n))


def act_jac(x, k, n):
    g = lag_grad(x[:n], k, n) * (2.0 / float(n))
    out = np.empty(n + 1, dtype=float)
    out[:n] = -g
    out[n] = 1.0
    return out


def slsqp_sact(h, n, cap=5, maxiter=80, ftol=1e-11, tol=1e-5):
    h0 = project(h, n)
    cc = c_all(h0, n)
    C = float(cc.max())
    near = np.where(cc >= C - tol)[0]
    if near.size > cap:
        order = np.argsort(cc[near])[::-1]
        idx = near[order[:cap]]
    else:
        idx = near
    if idx.size == 0:
        idx = np.argsort(cc)[-max(1, cap):]
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h0
    x0[n] = C + 1e-5
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': sum_con_x, 'jac': sum_con_x_jac, 'args': (n,)}]
    for k in idx:
        cons.append({'type': 'ineq', 'fun': act_fun, 'jac': act_jac, 'args': (int(k), n)})
    try:
        res = minimize(
            t_fun, x0, jac=t_jac, args=(n,), method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': float(ftol)}
        )
        if res.x is not None:
            hn = project(res.x[:n], n)
            if np.all(np.isfinite(hn)):
                return hn
    except Exception:
        pass
    return h0


def proj_grad(h, n, deadline, alphas=(50000, 150000), step0=0.03):
    h = project(h, n)
    besth = h.copy()
    bestc = c5(h, n)
    for alpha in alphas:
        step = step0
        for _ in range(400):
            if time.time() > deadline:
                return besth, bestc
            val, g = smooth_grad(h, n, alpha)
            ng = float(np.linalg.norm(g))
            if ng < 1e-12:
                break
            d = -g / ng
            r = step
            acc = False
            for __ in range(35):
                if time.time() > deadline:
                    break
                hp = project(h + r * d, n, 40)
                cc = c_all(hp, n)
                cp = float(cc.max())
                valn = smooth_value(cc, alpha)
                if cp < bestc - 1e-14 or (valn < val - 1e-10 and cp < bestc + 1e-8):
                    h = hp
                    val = valn
                    if cp < bestc:
                        bestc = cp
                        besth = hp.copy()
                    acc = True
                    step = min(0.20, max(0.001, r * 1.5))
                    break
                r *= 0.70
                if r < 1e-6:
                    break
            if not acc:
                step *= 0.70
                if step < 1e-5:
                    break
    return besth, bestc


def interp_to(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, m)
    x1 = np.linspace(0.0, 2.0, n)
    return project(np.interp(x1, x0, h), n)


def gen_start(n, rng, typ):
    x = np.linspace(0.0, 2.0, n)
    if typ == 0:
        h = np.full(n, 0.5)
    elif typ == 1:
        h = 0.5 + rng.uniform(0.15, 0.45) * np.tanh(
            (x - rng.uniform(0.4, 1.6)) / rng.uniform(0.2, 0.9)
        )
    elif typ == 2:
        a = rng.uniform(0.8, 2.2)
        s = rng.uniform(0.15, 0.6)
        h = a * np.exp(-((x - 1.0) / s) ** 2)
    elif typ == 3:
        h = np.full(n, 0.5)
        for _ in range(4):
            h += rng.uniform(0.04, 0.25) * np.sin(
                np.pi * rng.randint(1, 6) * x + rng.uniform(0.0, 2.0 * np.pi)
            )
    else:
        r = rng.uniform(0.25, 0.8)
        a = rng.uniform(1.0, 3.0)
        cc = rng.uniform(0.6, 1.4)
        h = a * np.clip(1.0 - np.abs(x - cc) / r, 0.0, 1.0)
    return project(h, n)


def optimize_at_n(hst, n, deadline, rng):
    h = project(hst, n)
    besth = h.copy()
    bestc = c5(h, n)

    if n <= 200:
        alphas = ((500, 50), (2000, 70), (8000, 90), (30000, 90))
    else:
        alphas = ((500, 40), (2000, 60), (8000, 70), (30000, 70))

    for a, mi in alphas:
        if time.time() > deadline - 8:
            break
        mi_use = iter_cap(mi, deadline)
        hn = slsqp_s(besth, n, a, mi_use, 1e-12)
        cn = c5(hn, n)
        if cn < bestc:
            bestc = cn
            besth = hn.copy()
        if time.time() < deadline - 15 and a in (2000, 30000):
            hp, cp = proj_grad(besth, n, min(deadline - 6, time.time() + 12), (a * 2,), 0.02)
            if cp < bestc:
                bestc = cp
                besth = hp.copy()

    if n <= 250:
        acts = ((3, 60), (5, 70), (8, 80))
    else:
        acts = ((3, 40), (5, 50))

    for cap, mi in acts:
        if time.time() > deadline - 12:
            break
        mi_use = iter_cap(mi, deadline)
        hn = slsqp_sact(besth, n, cap, mi_use, 1e-11, 1e-5)
        cn = c5(hn, n)
        if cn < bestc:
            bestc = cn
            besth = hn.copy()
        if time.time() < deadline - 8:
            hp, cp = proj_grad(besth, n, min(deadline - 4, time.time() + 8), (20000, 80000), 0.015)
            if cp < bestc:
                bestc = cp
                besth = hp.copy()

    if time.time() < deadline - 5:
        hn = slsqp_s(besth, n, 100000, iter_cap(80, deadline), 1e-13)
        cn = c5(hn, n)
        if cn < bestc:
            bestc = cn
            besth = hn.copy()

    if time.time() < deadline - 2:
        hp, cp = proj_grad(besth, n, deadline - 1, (80000, 250000), 0.012)
        if cp < bestc:
            bestc = cp
            besth = hp.copy()

    besth = project(besth, n)
    bestc = c5(besth, n)
    return besth, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 6.0
    rng = np.random.RandomState(seed)

    try:
        init = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if init.size < 10:
            raise ValueError
    except Exception:
        init = None

    besth = None
    bestc = float('inf')

    nlist = []
    if init is not None:
        nlist.append(len(init))
    else:
        nlist.append(250)
    for vv in (200, 150, 250, 100, 120, 300, 220, 320):
        if vv not in nlist and 40 <= vv <= 320:
            nlist.append(vv)

    for i, n in enumerate(nlist):
        if time.time() > deadline - 25:
            break
        rem = float(deadline - time.time())
        if i == 0 and init is not None:
            alloc = min(280.0, max(50.0, rem * 0.28))
            starts = [project(init, n)]
        else:
            alloc = min(110.0, max(30.0, rem * 0.16))
            if besth is not None:
                starts = [interp_to(besth, n)]
            elif init is not None:
                starts = [interp_to(init, n)]
            else:
                starts = [np.full(n, 0.5)]
            if n <= 220 and alloc > 75.0:
                starts.append(gen_start(n, rng, 3))
            if n <= 160 and alloc > 115.0:
                starts.append(gen_start(n, rng, 1))
        if time.time() + alloc > deadline - 8:
            alloc = max(20.0, float(deadline - time.time()) - 15.0)
        st0 = time.time()
        end = st0 + alloc
        ns = max(1, len(starts))
        for j, st in enumerate(starts):
            subend = st0 + alloc * (j + 1) / float(ns)
            if subend > deadline - 5:
                subend = deadline - 5
            if time.time() > subend - 6:
                break
            hp, c = optimize_at_n(st, n, subend, rng)
            if c < bestc:
                bestc = c
                besth = hp.copy()
            if time.time() > deadline - 30:
                break

    if besth is not None and time.time() < deadline - 70:
        n = len(besth)
        sigs = (0.005, 0.01, 0.02, 0.035, 0.06)
        nr = max(1, int((float(deadline - time.time()) - 30.0) / 50.0))
        for r in range(nr):
            if time.time() > deadline - 35:
                break
            sig = sigs[min(r, len(sigs) - 1)]
            hp = project(besth + rng.normal(0.0, sig, n), n)
            alloc = min(55.0, max(0.0, float(deadline - time.time()) - 25.0) * 0.45)
            if alloc < 20.0:
                break
            hopt, c = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c < bestc:
                bestc = c
                besth = hopt.copy()

    if besth is not None:
        n = len(besth)
        if time.time() < deadline - 20:
            hn = slsqp_sact(besth, n, 10, iter_cap(70, deadline), 1e-12, 1e-4)
            cn = c5(hn, n)
            if cn < bestc:
                bestc = cn
                besth = hn.copy()
        if time.time() < deadline - 12:
            hn = slsqp_s(besth, n, 150000, iter_cap(100, deadline), 1e-14)
            cn = c5(hn, n)
            if cn < bestc:
                bestc = cn
                besth = hn.copy()
        if time.time() < deadline - 4:
            hp, cp = proj_grad(besth, n, deadline - 1, (150000, 400000), 0.01)
            if cp < bestc:
                bestc = cp
                besth = hp.copy()
        besth = project(besth, n)
        bestc = c5(besth, n)

    if besth is None:
        n = 150
        besth = project(np.full(n, 0.5), n)
        bestc = c5(besth, n)

    return besth, float(bestc), len(besth)
