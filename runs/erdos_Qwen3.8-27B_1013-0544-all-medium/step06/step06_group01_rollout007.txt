                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    total_time = deadline - time.time()
    # Phase A: Aggressive CD
    t_a = 0.5 * total_time
    if t_a > 10:
        h_cd, c_cd = fast_cd_v2(h, n, max_passes=5000, grid=250,
                                 deadline=time.time() + t_a, rng=rng, penalty=3.0)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    # Phase B: SLSQP
    if time.time() < deadline - 25:
        h_s = slsqp_opt(h, n, alpha=100000, maxiter=2000)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h, best_c = h_s, c_s
    if time.time() < deadline - 20:
        h_s = slsqp_opt(h, n, alpha=500000, maxiter=2000)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h, best_c = h_s, c_s
    # Phase C: Fine CD
    if time.time() < deadline - 15:
        t_c = min(80, deadline - time.time() - 10)
        if t_c > 5:
            h_c, c_c = fast_cd_v2(h, n, max_passes=5000, grid=500,
                                   deadline=time.time() + t_c, rng=rng, penalty=0.05)
            h_c = project_feasible(h_c, n)
            c_c = compute_c5(h_c, n)
            if c_c < best_c:
                h, best_c = h_c, c_c
    # Phase D: Pair search
    if time.time() < deadline - 10:
        h, best_c = pair_search(h, n, deadline - 5,
                                [0.005, 0.002, 0.001, 0.0005], radius=50)
    # Phase E: Final SLSQP
    if time.time() < deadline - 8:
        h_f = slsqp_opt(h, n, alpha=1000000, maxiter=3000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            h, best_c = h_f, c_f
    # Phase F: Final pair
    if time.time() < deadline - 4:
        h, best_c = pair_search(h, n, deadline - 2,
                                [0.001, 0.0005, 0.0002, 0.0001], radius=30)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Main optimization
    alloc1 = min(550, (deadline - t0) * 0.50)
    h_opt, c_opt = optimize_candidate(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try n=250
    if time.time() < deadline - 280:
        n_try = 250
        x_init = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_init, best_h), n_try)
        alloc = min(220, (deadline - time.time()) * 0.25)
        if alloc > 60:
            h_o2, c_o2 = optimize_candidate(h_interp, n_try, time.time() + alloc, rng)
            if c_o2 < best_c:
                best_c, best_h = c_o2, h_o2
    # Multi-restart
    if time.time() < deadline - 120:
        n = len(best_h)
        remaining = deadline - time.time() - 60
        n_restart = max(2, int(remaining / 120))
        for r in range(n_restart):
            if time.time() > deadline - 70:
                break
            scale = 0.01 + 0.03 * (r / max(1, n_restart - 1))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(90, (deadline - time.time() - 40) * 0.4)
            if alloc < 30:
                break
            h_r, c_r = optimize_candidate(hp, n, time.time() + alloc, rng)
            if c_r < best_c:
                best_c, best_h = c_r, h_r
    # Final polish
    if time.time() < deadline - 15:
        n = len(best_h)
        h_f = slsqp_opt(best_h, n, alpha=2000000, maxiter=5000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, deadline - 2,
                                      [0.0005, 0.0002, 0.0001, 0.00005], radius=50)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
