def smooth_obj_and_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    logsum = np.log(exps.sum())
    f = m + logsum / alpha
    p = exps / exps.sum()
    grad = np.zeros(n)
    sig = np.where(p > 1e-12)[0]
    for k in sig:
        s = k - (n - 1)
        pk = p[k]
        # term1: pk * (1 - x[i-s]) for i where 0 <= i-s < n
        # i ranges from max(0, s) to min(n, n+s) - 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += pk * (1.0 - x[i_lo - s:i_hi - s])
        # term2: -pk * x[i+s] for i where 0 <= i+s < n
        # i ranges from max(0, -s) to min(n, n-s) - 1

        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= pk * x[i_lo2 + s:i_hi2 + s]
    grad *= dx
    return f, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, S):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    logsum = np.log(exps.sum())
    f = m + logsum / alpha
    p = exps / exps.sum()
    grad = np.zeros(n)
    sig = np.where(p > 1e-14)[0]
    for k in sig:
        s = k - (n - 1)
        pk = p[k]
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += pk * (1.0 - x[i_lo - s:i_hi - s])

        i_hi2 = min(n, n - s)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= pk * x[i_lo2 + s:i_hi2 + s]
    grad *= dx
    return f, grad

def slsqp_opt(h, n, alpha, maxiter, ftol):
    def obj(x):
        return smooth_obj_and_grad(x, alpha, n)[0]
    def grad(x):
        return smooth_obj_and_grad(x, alpha, n)[1]
    def cfun(x):
        return x.sum() - n / 2.0
    def cgrad(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cgrad}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun <= obj(h) + 1e-15:
            return project_box_sum(res.x, n / 2.0)
    except Exception:
        pass
    return h

            bnc = cur_c
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jm < 0)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < bnc:
                    bnc = nc
                    bj = j
            if bj >= 0 and bnc < cur_c - 1e-15:
                j = bj
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = bnc
                ones.remove(j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary greedy
    p1 = t0 + 0.08 * total
    hb, cb = greed_binary(h0, n, p1, max_sweeps=80)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: SLSQP with analytical gradients, geometric alpha schedule
    p2 = t0 + 0.50 * total
    h = best_h.copy()
    alphas = [50, 200, 1000, 5000, 20000, 100000, 5e5, 2e6, 1e7, 5e7, 2e8, 1e9, 5e9]
    for alpha in alphas:
        if time.time() > p2 - 5:
            break
        h = slsqp_opt(h, n, alpha, 800, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # Add a PG step between SLSQP calls
        h = pg_step(h, n, alpha, lr=2.0, nsteps=50)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: High-alpha refinement with multiple passes
    p3 = t0 + 0.68 * total
    h = best_h.copy()
    for _ in range(4):
        if time.time() > p3 - 3:
            break
        for alpha in [5e9, 2e10]:
            h = slsqp_opt(h, n, alpha, 1500, 1e-20)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        h = pg_step(h, n, 1e10, lr=1.0, nsteps=200)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Multi-resolution
    p4 = t0 + 0.84 * total
    for nh in [200, 300, 500, 600]:
        if time.time() > p4 - 12:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_box_sum(np.interp(xn, xo, best_h), nh / 2.0)
        for alpha in [5000, 50000, 500000, 5e6, 5e7]:
            if time.time() > p4 - 5:
                break
            hh = slsqp_opt(hh, nh, alpha, 400, 1e-18)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Phase 5: Final polish
    p5 = dl - 2
    if time.time() < p5:
        h = best_h.copy()
        for alpha in [2e10, 1e11, 5e11]:
            if time.time() > p5 - 1:
                break
            h = slsqp_opt(h, n, alpha, 3000, 1e-20)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        # Small noise restart
        for trial in range(2):
            if time.time() > p5 - 0.5:
                break
            h = best_h + 0.003 * rng.randn(n)
            h = project_box_sum(h, n / 2.0)
            h = slsqp_opt(h, n, 1e11, 2000, 1e-20)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_box_sum(best_h, n / 2.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
