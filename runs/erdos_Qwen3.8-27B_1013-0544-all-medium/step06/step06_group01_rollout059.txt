def subgradient_descent(h, n, max_iters=5000, deadline=None):

    best_c = compute_c5(h, n)

        if deadline is not None and time.time() > deadline:
            break

        curr_max = c.max()
        # Active set: terms within 1% of max
        active = c >= curr_max * 0.99
        n_active = active.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for p in range(n):
            h0 = h[p]
            # Gradient of c[m] wrt h[p]:
            # = (1-h[jf]) if jf valid - h[jb] if jb valid
            # where jf = p+m-(n-1), jb = p-m+(n-1)
            # For m = n-1: (1-2*h[p]) - ... actually (1-h[p]) - h[p] = 1-2h[p]
            # and we need to handle the -d^2 term separately (but for subgradient, 
            # the quadratic term's gradient is -2*d which is 0 at d=0, so we just use the linear part)
            
            # Sum over active m's
            g = 0.0
            # Vectorized:
            # m values where forward is valid: m in [n-1-p, 2n-2-p]
            # m values where backward is valid: m in [p, p+n-1]
            m_f = np.arange(max(0, n-1-p), min(2*n-1, 2*n-1-p))
            m_b = np.arange(max(0, p), min(2*n-1, p+n))
            
            # For forward: contributes (1-h[jf]) where jf = p+m-(n-1)
            if len(m_f) > 0:
                jf = p + m_f - (n-1)
                mask_f = active[m_f]
                if np.any(mask_f):
                    g += (1.0 - h[jf[mask_f]]).sum()
            
            # For backward: contributes -h[jb] where jb = p-m+(n-1)
            if len(m_b) > 0:
                jb = p - m_b + (n-1)
                mask_b = active[m_b]
                if np.any(mask_b):
                    g -= h[jb[mask_b]].sum()
            
            grad[p] = g / n_active
        
        # Step
        norm = np.linalg.norm(grad)
        if norm < 1e-15:
            break
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            # Decrease step slightly
            step *= 0.98
        else:
            step *= 0.9
        
        h = h_new
        if step < 1e-8:
            break
    
    return best_h, best_c

Let me think about this differently. What's the fundamental limitation?

The objective is non-convex (bilinear). SLSQP finds a local minimum of the smooth approximation. The quality depends on:
1. How good the starting point is
2. How well the smooth approximation matches the true max
3. Whether SLSQP converges to the true optimum of the smooth problem

            h_o, c_o = slsqp_run(hp, n, [5000, 50000], 600)
            h_o, c_o = coord_cd(h_o, n, 80, deadline, grid_size=40)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: Final high-alpha SLSQP + coord descent
    if time.time() < deadline - 20:
        n = len(best_h)
        hf, cf = slsqp_run(best_h, n, [500000, 1000000, 2000000], 2000)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd(best_h, n, 200, deadline, grid_size=80)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 6: Very fine coord descent
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = coord_cd(best_h, n, 100, deadline, grid_size=100)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
