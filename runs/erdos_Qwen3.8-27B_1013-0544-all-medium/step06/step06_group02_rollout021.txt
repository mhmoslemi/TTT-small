    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13: continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])

            grad[kk:n] += wm * (1.0 - h[0:nv])

Δc[m] = δ * (∂c[m]/∂h[i] - ∂c[m]/∂h[j])

The new max is max_m (c[m] + δ * d[m]) where d[m] = ∂c[m]/∂h[i] - ∂c[m]/∂h[j].

To minimize the new max, we want to find δ that minimizes max_m (c[m] + δ * d[m]).

This is a 1D piecewise-linear minimization. The optimum is at a "breakpoint" where two terms cross.

For practical purposes, we can just check a few candidate δ values:
- δ = min over m where d[m] > 0 of (c_max - c[m]) / d[m] (the step that just brings the worst term up to c_max)
- δ = the value that equalizes the top 2 terms

Actually, the simplest approach: just try δ = 0.001, 0.005, 0.01, 0.05 and see which gives the best max. This is fast and works well in practice.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(m, h, n):
    g = np.zeros(n)
    k = m - (n - 1)
    if k >= 0:
        nv = n - k
        g[0:nv] += (1.0 - h[k:n])
        g[k:n] += -h[0:nv]
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += (1.0 - h[0:nv])
        g[0:nv] += -h[kk:n]
    return g

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13: continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def grad_exact_max(h, n, tol=1e-8):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    active = c >= c_max - tol
    if active.sum() == 0: active[np.argmax(c)] = True
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        grad += lag_grad(m, h, n) * dx
    grad /= active.sum()
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh)))) / alpha

def smooth_obj_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def lbfgs_phase(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [200, 1000, 5000, 50000, 200000, 1000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3: break
        t_rem = deadline - time.time() - 2
        maxit = min(500, max(20, int(t_rem * 30)))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_obj_grad,
                          args=(n, alpha), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgrad_momentum(h, n, deadline, lr_init=0.02, momentum=0.85):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = lr_init
    stall = 0
    for _ in range(50000):
        if time.time() > deadline: break
        g = grad_exact_max(h, n)
        v = momentum * v + (1 - momentum) * g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15: break
        v_dir = v / v_norm
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            step = lr * factor
            h_new = project_feasible(h - step * v_dir, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            lr *= 0.7
            stall += 1
            if stall > 50:
                h = project_feasible(h + 0.01 * np.random.randn(n) * (1 - h) * h, n)
                lr = lr_init * 0.5
                stall = 0
        if lr < 1e-10: break
    return best_h, best_c

def targeted_pair_search(h, n, deadline, max_pairs_per_round=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        round_improved = 0
        while time.time() < deadline:
            c = compute_c5_vec(best_h, n)
            c_max = c.max()
            active = np.argsort(c)[-5:]
            g = np.zeros(n)
            for m in active:
                g += lag_grad(m, best_h, n)
            g /= len(active)
            # Rank pairs by expected benefit: h[i] high, g[i] low; h[j] low, g[j] high
            benefit = g.copy()
            candidates = []
            for i in range(n):
                if best_h[i] < d: continue
                for j in range(n):
                    if i == j: continue
                    if best_h[j] > 1.0 - d: continue
                    score = benefit[i] - benefit[j]
                    if score < -0.01:
                        candidates.append((score, i, j))
            candidates.sort()
            tried = 0
            improved_any = False
            for score, i, j in candidates[:max_pairs_per_round]:
                if tried >= max_pairs_per_round: break
                tried += 1
                best_h[i] -= d
                best_h[j] += d
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved_any = True
                    round_improved += 1
                else:
                    best_h[i] += d
                    best_h[j] -= d
                if time.time() > deadline:
                    return best_h, best_c
            if not improved_any:
                break
    return best_h, best_c

def fine_pair_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_c5_vec(best_h, n)
    c_max = c.max()
    active_idx = np.where(c >= c_max - 1e-4)[0]
    for d in [0.001, 0.0005, 0.0002, 0.0001]:
        improved = False
        while time.time() < deadline and improved:
            improved = False
            h = best_h.copy()
            for i in range(n):
                if time.time() > deadline: break
                if h[i] < d: continue
                for j in np.sort(np.argsort(h)[::-1])[:100]:
                    if i == j or h[j] > 1.0 - d: continue
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
        best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def try_n_value(n_try, best_h, best_c, n_cur, deadline, rng):
    x_old = np.linspace(0, 2, n_cur)
    x_new = np.linspace(0, 2, n_try)
    h_interp = np.interp(x_new, x_old, best_h)
    h_interp = project_feasible(h_interp, n_try)

                                   lr_init=0.01, momentum=0.8)

    return None, best_c, n_try
