  def grad_fast(h, n, alpha):
      """Fast gradient using FFT and vectorized operations."""
      c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
      c_shift = alpha * (c - c.max())
      w = np.exp(c_shift)
      w /= w.sum()
      dx = 2.0 / n
      m = 2 * n - 1  # length of correlation
      
      # S[i] = sum of w over window [max(0, n-1-i), min(m-1, n-1+i)]
      cumw = np.cumsum(w)
      lo = np.maximum(0, n - 1 - np.arange(n))
      hi = np.minimum(m - 1, n - 1 + np.arange(n))
      S = np.zeros(n)
      S[lo]  # placeholder
      # S[i] = cumw[hi[i]] - cumw[lo[i]-1] (with cumw[-1] = 0)
      S = cumw[hi]
      S -= np.where(lo > 0, cumw[lo - 1], 0.0)
      
      # A[i] = (h * w)[i+n-1] (linear convolution)
      # B[i] = (h * w_rev)[i+n-1] (linear convolution with reversed w)
      N_fft = 1

          N_fft *= 2

      W = np.fft.rfft(w, N_fft)
      Wr = np.fft.rfft(w[::-1], N_fft)
      
      conv1 = np.fft.irfft(H * W, N_fft)[:n + m - 1]  # length 3n-2
      conv2 = np.fft.irfft(H * Wr, N_fft)[:n + m - 1]
      
      A = conv1[n-1:2*n-1]
      B = conv2[n-1:2*n-1]
      
      return dx * (S - A - B)

      val = smooth_max_obj(x, n, alpha)
