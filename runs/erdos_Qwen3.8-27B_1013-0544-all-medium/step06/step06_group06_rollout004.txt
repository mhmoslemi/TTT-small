import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def thue_morse_bin(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2

        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

    m = np.arange(2 * n - 1)
    p = np.arange(n)
    q1 = p[:, None] - m[None, :] + n - 1

    q2 = p[:, None] + m[None, :] - n + 1

    tq2 = np.where(valid2, q2, 0)
    term2 = np.where(valid2, -h[tq2], 0.0)
    return term1 + term2

def local_search(h_init, n, deadline, rng, top_k=20, max_kicks=200):
    dx = 2.0 / n
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    kick_count = 0
    stall = 0
    no_improve = 0
    temp = 0.002

    while time.time() < deadline:
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        top_idx = np.argsort(corr)[-top_k:]
        effect = compute_effect(h, n)
        ez_top = effect[zeros][:, top_idx]
        eo_top = effect[ones][:, top_idx]
        ct = corr[top_idx][None, None, :]
        new_max = (ct + ez_top[:, None, :] - eo_top[None, :, :]).max(axis=2)
        flat = int(new_max.argmin())
        bi, bj = flat // no, flat % no
        new_c = new_max[bi, bj] * dx
        if new_c < cur_c - 1e-14:
            i, j = zeros[bi], ones[bj]
            h[i], h[j] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros = np.delete(zeros, bi)
            ones = np.delete(ones, bj)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            stall = 0
            no_improve = 0

            no_improve += 1

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
