```python
        # Log-sum-exp for smooth max
        alpha = 500.0
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log((np.exp(sh - sm)).sum())) / alpha

        return x.sum() - n / 2.0

                         constraints=cons,
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_slq_n = [250]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def _slq_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log((np.exp(sh - sm)).sum())) / alpha

def _slq_con_fun(x):
    return x.sum() - _slq_n[0] / 2.0

def _slq_con_jac(x):
    return np.ones(_slq_n[0])

def slsqp_optimize(h, n, deadline, maxiter=3000, alpha=1000.0):
    _slq_n[0] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    try:
        result = minimize(_slq_obj, h, args=(n, alpha), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        return project_feasible(result.x, n)
    except Exception:
        return h

def ills_binary_search(h, n, deadline, rng, max_iters=100000, perturb_size=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        crit = np.where(corr >= cur_max - 1.0)[0]
        improved = False

        for ck in crit[:5]:
            if time.time() > deadline or improved:
                break
            lag = ck - (n - 1)
            if lag == 0:
                continue
            ones = np.where(h == 1)[0]
            swaps = []
            for m in ones:
                im = m - lag
                if 0 <= im < n and h[im] == 0:
                    swaps.append((im, m))
            if len(swaps) > 40:
                idx = rng.choice(len(swaps), size=40, replace=False)
                swaps = [swaps[i] for i in idx]
            best_nm = cur_max
            best_p = None
            best_nc = None
            for (i, j) in swaps:
                h[i] = 1.0
                h[j] = 0.0
                nc = np.correlate(h, 1.0 - h, mode='full')
                nm = nc.max()
                if nm < best_nm:
                    best_nm = nm
                    best_p = (i, j)
                    best_nc = nc
                h[i] = 0.0
                h[j] = 1.0
            if best_p is not None:
                i, j = best_p
                h[i] = 1.0
                h[j] = 0.0
                corr = best_nc
                cur_max = best_nm
                improved = True
                stall = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()

        if not improved:
            stall += 1
            if stall > 30:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                npert = rng.randint(2, perturb_size + 1)
                for _ in range(npert):
                    oi = rng.choice(ones)
                    zi = rng.choice(zeros)
                    h[zi] = 1.0
                    h[oi] = 0.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                cur_max = corr.max()
                stall = 0

    return best_h, float(best_max * dx)

def binarize(h, n):
    srt = np.sort(h)[::-1]
    thresh = srt[n // 2]
    h_bin = (h >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        excess = np.where(h_bin == 1)[0]
        for idx in excess[n_ones - n // 2:]:
            h_bin[idx] = 0.0
    elif n_ones < n // 2:
        deficit = np.where(h_bin == 0)[0]
        for idx in deficit[:n // 2 - n_ones]:
            h_bin[idx] = 1.0
    return h_bin

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    except Exception:
        n = 250
        best_h = random_binary(n, rng)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: ILS binary search from parent (45% budget)
    p1 = t0 + 0.45 * (budget_s - 12)
    h1, c1 = ills_binary_search(best_h.copy(), n, p1, rng, max_iters=300000, perturb_size=5)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: SLSQP continuous polish (15% budget)
    p2 = t0 + 0.60 * (budget_s - 12)
    if time.time() < p2:
        for alpha in [800.0, 3000.0, 15000.0]:
            if time.time() > p2:
                break
            h2 = slsqp_optimize(best_h.copy(), n, p2, maxiter=2500, alpha=alpha)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    # Phase 3: Re-binarize + ILS (25% budget)
    p3 = t0 + 0.85 * (budget_s - 12)
    if time.time() < p3:
        h_bin = binarize(best_h, n)
        h3, c3 = ills_binary_search(h_bin, n, p3, rng, max_iters=200000, perturb_size=6)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: Try different n values (15% budget)
    p4 = deadline - 8
    if time.time() < p4:
        for n_try in [200, 240, 300]:
            if time.time() > p4 - 60:
                break
            h_r = random_binary(n_try, rng)
            alloc_end = min(time.time() + 80, p4)
            h4, c4 = ills_binary_search(h_r, n_try, alloc_end, rng, max_iters=80000, perturb_size=4)
            if c4 < best_c:
                best_c = c4
                best_h = h4
                n = n_try

    # Final polish
    if time.time() < deadline - 3:
        hf = slsqp_optimize(best_h.copy(), n, deadline - 2, maxiter=1500, alpha=5000.0)
        cf = compute_c5(hf, n)
        if cf < best_c:
            best_c = cf
            best_h = hf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
