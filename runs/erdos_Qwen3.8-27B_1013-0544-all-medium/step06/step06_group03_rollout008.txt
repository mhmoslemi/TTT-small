def binary_derivs(b,n):
    L=2*n-1
    k=np.arange(L, dtype=int) - (n-1)
    bd=b.astype(np.int8)
    Ddown=np.zeros((n,L), dtype=np.int8)
    Dup=np.zeros((n,L), dtype=np.int8)
    for r in range(n):
        A=np.zeros(L, dtype=np.int8)
        rq = r - k
        m1 = (rq >= 0) & (rq < n)
        if m1.any(): A[m1] += (1 - bd[rq[m1]])
        rp = r + k
        m2 = (rp >= 0) & (rp < n)
        if m2.any(): A[m2] -= bd[rp[m2]]
        Ddown[r] = -A
        Dup[r] = A
        Ddown[r, n-1] -= 1
        Dup[r, n-1] -= 1
    return Ddown,Dup
```

def active_lse_fun(x,n,idxs,alpha):
    vals, grads = lag_values_grads(x,n,idxs)
    mx = float(np.max(vals))
    sh = alpha*(vals-mx)
    e = np.exp(sh)
    s = float(np.sum(e))
    obj = mx + math.log(s)/alpha
    g=np.zeros(n)
    for w,gr in zip(e/s, grads):
       if w > 1e-14: g += w*gr
    return obj,g

def project_box_sum(x,n):
    x=np.asarray(x,float).copy()
    S=n/2.0
    lo=float(np.min(x))-1.0; hi=float(np.max(x))+1.0
    for _ in range(60):
       mid=(lo+hi)*0.5
       y=np.clip(x-mid,0.0,1.0)
       s=float(y.sum())
       if s > S: lo=mid
       else: hi=mid
    y=np.clip(x-hi,0.0,1.0)
    diff=S-float(y.sum())
    interior=(y>1e-12)&(y<1-1e-12)
    if diff !=0.0:
       if interior.any():
          y[interior] += diff/float(interior.sum())
       else:
          if diff>0:
             room=1.0-y; idx=np.flatnonzero(room>1e-12)
          else:
             room=y; idx=np.flatnonzero(room>1e-12)
          if idx.size:
             tot=room[idx].sum()
             if tot>0:
                y[idx] += (diff/tot)*room[idx]
    y=np.clip(y,0,1)
    # exact adjust
    diff=S-float(y.sum())
    if abs(diff)>1e-14:
       if diff>0: idx=int(np.argmax(1.0-y))
       else: idx=int(np.argmax(y))
       if 0<=idx<n:
          y[idx]=np.clip(y[idx]+diff,0,1)
    return y

def compute_c5_vec(h, n):
    return np.asarray(np.correlate(h, 1.0 - h, mode="full"), dtype=float) * (2.0 / float(n))


def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def project_box_sum(x, n):
    x = np.asarray(x, dtype=float).copy()
    S = 0.5 * float(n)
    if n <= 0:
        return x
    if (float(x.sum()) == S or abs(float(x.sum()) - S) < 1e-9) and np.all((x <= 1.0 + 1e-8) & (x >= -1e-8)):
        if np.all((x < 1e-8) | (x > 1.0 - 1e-8)):
            return np.round(np.clip(x, 0.0, 1.0))
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - hi, 0.0, 1.0)
    diff = S - float(y.sum())
    if abs(diff) > 1e-13:
        interior = (y > 1e-10) & (y < 1.0 - 1e-10)
        if interior.any():
            y[interior] += diff / float(interior.sum())
        else:
            if diff > 0.0:
                idx = np.flatnonzero(y < 1.0 - 1e-12)
            else:
                idx = np.flatnonzero(y > 1e-12)
            if idx.size:
                room = 1.0 - y[idx] if diff > 0.0 else y[idx]
                pos = idx[(room > 1e-12)]
                if pos.size:
                    y[pos] += (diff / float(room[pos].sum())) * room[pos]
    y = np.clip(y, 0.0, 1.0)
    diff = S - float(y.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            cand = np.flatnonzero(y < 1.0 - 1e-12)
            if cand.size:

                y[j] = np.clip(y[j] + diff, 0.0, 1.0)

            cand = np.flatnonzero(y > 1e-12)
            if cand.size:
                j = cand[int(np.argmax(y[cand]))]
                y[j] = np.clip(y[j] + diff, 0.0, 1.0)
    return y


def bits_from_vals(vals, n):
    vals = np.asarray(vals, dtype=float)
    if vals.size != n:
        xp = np.linspace(0.0, 1.0, vals.size)
        xq = np.linspace(0.0, 1.0, n)
        vals = np.interp(xq, xp, vals)
    b = np.zeros(n, dtype=np.int8)
    if n > 1:
        idx = np.argsort(vals, kind="stable")[-(n // 2):]
        b[idx] = 1
    return b


def interp_float(h, nold, nnew):
    if nold == nnew:
        return np.asarray(h, dtype=float).copy()
    xold = np.linspace(0.0, 2.0, nold, endpoint=False)
    xnew = np.linspace(0.0, 2.0, nnew, endpoint=False)
    return np.interp(xnew, xold, np.asarray(h, dtype=float))


def pattern_vals(n, kind):
    i = np.arange(n, dtype=float)
    if kind == 0:
        vals = np.zeros(n, dtype=float)
        for j in range(n):
            vals[j] = int(bin(j).count("1") & 1)
        return vals
    if kind == 1:
        return np.remainder(i * 0.38196601125, 1.0)
    if kind == 2:
        return np.remainder(i * 0.31803398875, 1.0)
    if kind == 3:
        return np.remainder(i * 0.5857864376, 1.0)
    return np.sin(2.0 * np.pi * i / float(n)) + 0.6 * np.sin(4.0 * np.pi * i / float(n) + 0.9)


def best_shift_cont(h, n, step=1):
    h0 = np.asarray(h, dtype=float).copy()
    best_h = h0
    best_c = compute_c5(h0, n)
    if n > 700:
        return best_h, best_c
    if step < 1:
        step = 1
    for s in range(step, n, step):
        h2 = np.roll(h0, s)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    h2 = h0[::-1].copy()
    c2 = compute_c5(h2, n)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    return best_h, best_c


def binary_corr(b, n):
    b32 = np.asarray(b, dtype=np.int32)
    o32 = np.ones_like(b32) - b32
    return np.asarray(np.correlate(b32, o32, mode="full"), dtype=np.int32)


def binary_derivs(b, n):
    L = 2 * n - 1
    b8 = np.asarray(b, dtype=np.int8)
    rev = (1 - b8[::-1]).astype(np.int8)
    Dd = np.zeros((n, L), dtype=np.int8)
    Du = np.zeros((n, L), dtype=np.int8)
    for r in range(n):
        A = np.zeros(L, dtype=np.int8)
        A[r:r + n] += rev
        A[n - 1 - r:n - 1 - r + n] -= b8
        Dd[r] = -A
        Du[r] = A
        Dd[r, n - 1] -= 1
        Du[r, n - 1] -= 1
    return Dd, Du


def target_max_for_n(n, target):
    return int(math.floor(target * float(n) * 0.5 + 1e-12))


def binary_hill(n, bits, deadline, tmax, rng):
    S = n // 2
    bits = np.asarray(bits, dtype=np.int8).copy()
    if int(bits.sum()) != S:
        bits = bits_from_vals(bits.astype(float), n)
    c = binary_corr(bits, n)
    best_b = bits.copy()
    best_m = int(c.max())
    if best_m <= tmax:
        return best_b, float(2.0 * best_m) / float(n), best_m

    BIG = 10000000
    kicks = 0
    max_kicks = 60

    while time.time() < deadline - 0.5:
        m = int(c.max())
        if m <= tmax:
            break
        Dd, Du = binary_derivs(bits, n)
        ones = np.flatnonzero(bits == 1)
        zeros = np.flatnonzero(bits == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        cur_sq = int(c @ c)
        cur_score = m * BIG + cur_sq
        best_score = cur_score
        best_adj = None

        for p in ones:
            base = c + Dd[p]
            for q in zeros:
                tmp = base + Du[q]
                i1 = n - 1 + int(p) - int(q)
                i2 = n - 1 + int(q) - int(p)
                tmp[i1] += 1
                tmp[i2] += 1
                nm = int(tmp.max())
                if nm > m:
                    continue
                lower = nm * BIG
                if lower >= best_score:
                    continue
                sq2 = int(tmp @ tmp)
                score = lower + sq2
                if score < best_score:
                    best_score = score
                    best_adj = (int(p), int(q))

        if best_adj is not None:
            p, q = best_adj
            bits[p] = 0
            bits[q] = 1
            c = binary_corr(bits, n)
            m = int(c.max())
            if m < best_m:
                best_m = m
                best_b = bits.copy()
            continue

        if kicks < max_kicks:
            kallow = 1
            if kicks > 15:
                kallow = 2
            ok = False
            for _ in range(24):
                p = int(ones[rng.randint(int(ones.size))])
                q = int(zeros[rng.randint(int(zeros.size))])
                tmp = c + Dd[p] + Du[q]
                i1 = n - 1 + p - q
                i2 = n - 1 + q - p
                tmp[i1] += 1
                tmp[i2] += 1
                nm = int(tmp.max())
                if nm <= m + kallow:
                    if nm <= m or rng.rand() < 0.45:
                        bits[p] = 0
                        bits[q] = 1
                        c = tmp.astype(np.int32, copy=False).copy()
                        ok = True
                        break
            if ok:
                kicks += 1
                m = int(c.max())
                if m < best_m:
                    best_m = m
                    best_b = bits.copy()
                continue
        break

    if int(binary_corr(bits, n).max()) < best_m:
        best_m = int(binary_corr(bits, n).max())
        best_b = bits.copy()
    return best_b, float(2.0 * best_m) / float(n), best_m


def lag_values_grads(h, n, idxs):
    idxs = np.asarray(idxs, dtype=int)
    center = n - 1
    vals = np.zeros(len(idxs), dtype=float)
    grads = []
    for a in range(len(idxs)):
        k = int(idxs[a]) - center
        g = np.zeros(n, dtype=float)
        if k == 0:
            vals[a] = float(np.sum(h * (1.0 - h)))
            g = 1.0 - 2.0 * h
        elif k > 0:
            l = n - k
            seg = h[l:]
            vals[a] = float(np.sum(seg) - float(np.dot(seg, h[:l])))
            g[l:] += 1.0 - h[:l]
            g[:l] -= seg
        else:
            l = n + k
            seg = h[:l]
            vals[a] = float(np.sum(seg) - float(np.dot(seg, h[l:])))
            g[:l] += 1.0 - h[l:]
            g[l:] -= seg
        grads.append(g)
    return vals, grads


def active_lse_fun(x, n, idxs, alpha):
    vals, grads = lag_values_grads(x, n, idxs)
    mx = float(np.max(vals))
    sh = alpha * (vals - mx)
    e = np.exp(sh)
    s = float(np.sum(e))
    obj = mx + math.log(s) / alpha
    g = np.zeros(n, dtype=float)
    w = e / s
    for a in range(len(idxs)):
        if w[a] > 1e-14:
            g += w[a] * grads[a]
    return obj, g


def sum_con(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_con_jac(x, n):
    return np.ones(n, dtype=float)


def slsqp_refine(h, n, idxs, alpha, maxiter):
    x0 = project_box_sum(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    try:
        res = minimize(active_lse_fun, x0, args=(n, idxs, alpha),
                       method="SLSQP", bounds=bounds, constraints=cons,
                       options={"maxiter": int(maxiter), "ftol": 1e-14, "disp": False})
        x = np.asarray(res.x, dtype=float)
    except Exception:
        x = x0
    return project_box_sum(x, n)


def continuous_refine(h, n, deadline):
    h = project_box_sum(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mit1 = 420 if n > 250 else 550
    mit2 = 260 if n > 250 else 360
    for alpha in (400.0, 2000.0, 10000.0, 40000.0):
        if time.time() > deadline - 12.0:
            break
        cv = compute_c5_vec(h, n)
        order = np.argsort(cv)[::-1]
        idxs = np.unique(np.concatenate((order[:30], np.array([n - 1], dtype=int))))
        h2 = slsqp_refine(h, n, idxs, alpha, mit1)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2
        if time.time() > deadline - 12.0:
            break
        cv = compute_c5_vec(h, n)
        order = np.argsort(cv)[::-1]
        idxs = np.unique(np.concatenate((order[:8], np.array([n - 1], dtype=int))))
        h2 = slsqp_refine(h, n, idxs, min(120000.0, alpha * 2.0), mit2)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2
    return best_h, compute_c5(best_h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    target = 0.38080

    try:
        hpar = np.asarray(initial_h_values, dtype=float).reshape(-1).copy()
        npar = int(len(hpar))
    except Exception:
        hpar = None
        npar = 0

    best_h = None
    best_c = float("inf")

    if hpar is not None and npar > 0:
        hpar = project_box_sum(hpar, npar)
        if npar <= 650 and time.time() < deadline - 8.0:
            step = 1 if npar <= 360 else 2
            hpar, cpar = best_shift_cont(hpar, npar, step)
        else:
            cpar = compute_c5(hpar, npar)
        best_h = hpar.copy()
        best_c = cpar

        if npar % 2 == 0 and time.time() < deadline - 35.0:
            b = bits_from_vals(best_h, npar)
            end1 = time.time() + min(220.0, max(30.0, (deadline - time.time()) * 0.28))
            b, bc, bm = binary_hill(npar, b, end1, target_max_for_n(npar, target), rng)
            if bc < best_c:
                best_c = bc
                best_h = b.astype(float)
            if best_c <= target:
                best_h = project_box_sum(best_h, npar)
                return best_h, compute_c5(best_h, npar), npar

        if time.time() < deadline - 60.0:
            end2 = time.time() + min(220.0, max(25.0, (deadline - time.time()) * 0.30))
            hc, cc = continuous_refine(hpar, npar, end2)
            if cc < best_c:
                best_c = cc
                best_h = hc
            if best_c <= target:
                best_h = project_box_sum(best_h, npar)
                return best_h, compute_c5(best_h, npar), npar

        if time.time() < deadline - 45.0:
            end3 = time.time() + min(160.0, max(20.0, (deadline - time.time()) * 0.28))
            hb, cb = continuous_refine(best_h, len(best_h), end3)
            if cb < best_c:
                best_c = cb
                best_h = hb
            if best_c <= target:
                nfin = len(best_h)
                best_h = project_box_sum(best_h, nfin)
                return best_h, compute_c5(best_h, nfin), nfin

    nlist = []
    if npar > 0 and npar % 2 == 0:
        nlist.append(npar)
    for nn in (64, 80, 100, 128, 150, 168, 200, 224, 240, 256, 288, 300, 360, 450):
        if nn not in nlist:
            nlist.append(nn)

    for ntry in nlist:
        if best_c <= target:
            break
        if time.time() > deadline - 75.0:
            break
        starts = []
        if best_h is not None and len(best_h) > 0:
            starts.append(bits_from_vals(interp_float(best_h, len(best_h), ntry), ntry))
        pmaxk = 2 if ntry > 350 else 4
        for kk in range(pmaxk):
            starts.append(bits_from_vals(pattern_vals(ntry, kk), ntry))
        if not starts:
            continue

        remaining = max(0.0, deadline - time.time() - 65.0)
        total = min(160.0, max(35.0, remaining * 0.18))
        if ntry == npar and hpar is not None:
            total *= 0.45
        if ntry > 350:
            total *= 0.65
        max_starts = max(1, int(total // 12.0))
        starts = starts[:max_starts]
        per = total / float(max(1, len(starts)))

        for sb in starts:
            if best_c <= target or time.time() > deadline - 60.0:
                break
            end = time.time() + max(10.0, min(80.0, per))
            if end < time.time() + 8.0:
                break
            b, bc, bm = binary_hill(ntry, sb, end, target_max_for_n(ntry, target), rng)
            if bc < best_c:
                best_c = bc
                best_h = b.astype(float)

    if best_h is None:
        nfb = 128
        endfb = min(deadline - 1.0, time.time() + 60.0)
        b, bc, bm = binary_hill(nfb, bits_from_vals(pattern_vals(nfb, 1), nfb),
                                endfb, target_max_for_n(nfb, target), rng)
        best_h = b.astype(float)
        best_c = bc

    if best_h is not None and best_c > target and time.time() < deadline - 18.0:
        nbest = len(best_h)
        if nbest <= 350:
            hf, cf = continuous_refine(best_h, nbest, deadline - 1.0)
            if cf < best_c:
                best_c = cf
                best_h = hf

    nfin = int(len(best_h))
    best_h = project_box_sum(best_h, nfin)
    best_c = compute_c5(best_h, nfin)
    return best_h, best_c, nfin
