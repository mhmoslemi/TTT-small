import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _grad_from_w(w, h, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def topk_obj(x, n, k, alpha):
    c = compute_c5_vec(x, n)
    if k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -k)[-k:]
    topk = c[idx]
    m = topk.max()
    sh = alpha * (topk - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def topk_grad(x, n, k, alpha):
    c = compute_c5_vec(x, n)
    nc = len(c)
    if k >= nc:
        idx = np.arange(nc)
    else:
        idx = np.argpartition(c, -kc)[-kc:] if (kc := k) else np.arange(nc)
        idx = np.argpartition(c, -k)[-k:]
    topk = c[idx]
    m = topk.max()
    sh = alpha * (topk - m)
    mx = sh.max()
    w_top = np.exp(sh - mx)
    w_top /= w_top.sum()
    w = np.zeros(nc)
    for pos, m_idx in enumerate(idx):
        w[m_idx] = w_top[pos]
    return _grad_from_w(w, x, n)

def lbfgs_opt(h, n, k, alpha, max_iters=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(topk_obj, h, args=(n, k, alpha), jac=lambda x: topk_grad(x, n, k, alpha),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-14,
                               'maxfun': max_iters * 5, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h
