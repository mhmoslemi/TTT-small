def project_feasible(h, n):

        b = d_hi

        mx = float(np.max(sh))

        return x.sum() - n / 2.0

    b = d_hi

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _gs_eval(c_seg, s, um, c_mid, center_in, mid_idx, d, sumh, n, penalty, hp):
    vals = c_seg + d * s
    if center_in:
        vals[mid_idx] = c_mid + d * (1.0 - 2.0 * hp) - d * d
    val = max(um, float(np.max(vals)))
    if penalty > 0:
        val += penalty * (sumh + d - 0.5 * n) ** 2
    return val

def _golden_cd(c_seg, s, um, c_mid, center_in, mid_idx, d_lo, d_hi, sumh, n, penalty, hp):
    gr = 0.6180339887498949
    a = d_lo
    b = d_hi
    if b - a < 1e-14:
        return 0.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    fc1 = _gs_eval(c_seg, s, um, c_mid, center_in, mid_idx, c1, sumh, n, penalty, hp)
    fc2 = _gs_eval(c_seg, s, um, c_mid, center_in, mid_idx, c2, sumh, n, penalty, hp)
    for _ in range(55):
        if b - a < 1e-13:
            break
        if fc1 < fc2:
            b = c2; c2 = c1; fc2 = fc1
            c1 = b - gr * (b - a)
            fc1 = _gs_eval(c_seg, s, um, c_mid, center_in, mid_idx, c1, sumh, n, penalty, hp)
        else:
            a = c1; c1 = c2; fc1 = fc2
            c2 = a + gr * (b - a)
            fc2 = _gs_eval(c_seg, s, um, c_mid, center_in, mid_idx, c2, sumh, n, penalty, hp)
    return (a + b) * 0.5

def exact_cd_pass(h, n, c, sumh, order, penalty):
    n_full = 2 * n - 1
    mid = n - 1
    improved = False
    for p in order:
        d_lo = -h[p]
        d_hi = 1.0 - h[p]
        if d_hi - d_lo < 1e-14:
            continue
        k_lo = min(n - 1 - p, p)
        k_hi = n - 1 + p
        um = 0.0
        if k_lo > 0:
            um = float(np.max(c[:k_lo]))
        if k_hi < n_full:
            um = max(um, float(np.max(c[k_hi + 1:])))
        ks = np.arange(k_lo, k_hi + 1)
        s = np.zeros(len(ks))
        idx_f = p + ks - (n - 1)
        vf = (idx_f >= 0) & (idx_f < n)
        if np.any(vf):
            s[vf] += (1.0 - h[idx_f[vf]])
        idx_b = p - ks + (n - 1)
        vb = (idx_b >= 0) & (idx_b < n)
        if np.any(vb):
            s[vb] -= h[idx_b[vb]]
        center_in = (k_lo <= mid <= k_hi)
        c_seg = c[ks].copy()
        c_mid_val = float(c[mid])
        mid_idx = mid - k_lo if center_in else -1
        hp = h[p]
        d_opt = _golden_cd(c_seg, s, um, c_mid_val, center_in, mid_idx, d_lo, d_hi, sumh, n, penalty, hp)
        if abs(d_opt) > 1e-13:
            c[ks] += d_opt * s
            if center_in:
                c[mid] -= d_opt * d_opt
            h[p] += d_opt
            sumh += d_opt
            improved = True
    return h, c, sumh, improved

def fast_cd(h, n, max_passes, deadline, rng, penalty):
    h = h.copy()
    c = compute_corr(h, n)
    sumh = float(h.sum())
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.permute(order)
        else:
            order = np.roll(order, _pass % n)
        h, c, sumh, improved = exact_cd_pass(h, n, c, sumh, order, penalty)
        if not improved:
            break
    return h, float(np.max(c)) * (2.0 / n)

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    if mx > 500:
        return (m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha) * dx
    return (m + float(np.log(np.sum(np.exp(sh)))) / alpha) * dx

def _slsqp_con(x, n):
    return x.sum() - n / 2.0

def _slsqp_jac(x):
    return np.ones_like(x)

def slsqp_smooth(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    t1 = min(deadline - time.time(), 250)
    if t1 > 30:
        h_cd, c_cd = fast_cd(h, n, 3000, time.time() + t1, rng, 2.0)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    if time.time() < deadline - 80:
        for a in [50000, 200000]:
            if time.time() > deadline - 40:
                break
            h_s = slsqp_smooth(h, n, a, 2000)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                h, best_c = h_s, c_s
    if time.time() < deadline - 40:
        t3 = min(150, deadline - time.time() - 15)
        h_cd2, c_cd2 = fast_cd(h, n, 3000, time.time() + t3, rng, 0.0)
        h_cd2 = project_feasible(h_cd2, n)

        if c_cd2 < best_c:

    if time.time() < deadline - 15:

        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            h, best_c = h_f, c_f
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def sa_with_cd(h, n, deadline, rng, n_iter=200):
    h = h.copy()
    c_val = compute_c5(h, n)
    T = 0.02
    cooling = T / (1.0 + 0.05 * n_iter)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        hp = h.copy()
        mask = rng.random(n) < 0.15
        hp[mask] += rng.normal(0, T, mask.sum())

            c_new = compute_c5(hp2, n)
        else:
            c_new = compute_c5(hp, n)
            hp2 = hp
        delta = c_new - c_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h, c_val = hp2, c_new
        T = max(T * 0.97, 0.001)
    return h, c_val
