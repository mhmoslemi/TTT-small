import numpy as np
import time
from scipy.optimize import minimize

FULL_CONFIGS = (
    (200.0, 1.0e3, 80),
    (2000.0, 1.0e4, 100),
    (20000.0, 1.0e5, 120),
    (200000.0, 1.0e6, 120),
    (800000.0, 1.0e6, 80),
)
SHORT_CONFIGS = (
    (1000.0, 1.0e4, 80),
    (20000.0, 1.0e5, 100),
)
MED_BASES = (0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002)
FINE_BASES = (0.002, 0.001, 0.0005, 0.0002, 0.0001, 5.0e-5)
FACTORS = (1.0, 0.5, 0.25, 0.125)


def c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def c5(h, n):
    return float(np.max(c_vec(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(h, 0.0, 1.0)
    target = 0.5 * n

    lo, hi = -1.0, 1.0
    for _ in range(75):
        lam = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - lam, 0.0, 1.0)))
        if s > target:
            lo = lam
        else:
            hi = lam

    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(80):
        diff = target - float(np.sum(x))
        if abs(diff) < 1.0e-14:
            break
        if diff > 0.0:
            room = 1.0 - x
            tot = float(np.sum(room))
            if tot < 1.0e-15:
                break
            x += (diff / tot) * room
        else:
            tot = float(np.sum(x))
            if tot < 1.0e-15:
                break
            x += (diff / tot) * x
    return np.clip(x, 0.0, 1.0)


def grad_matrix(h, n):
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((n, L), dtype=float)
    if n <= 0:
        return G
    G[:, n - 1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        pos = n - 1 + d
        neg = n - 1 - d
        m = n - d
        G[:m, pos] += dx * (1.0 - h[d:])
        G[d:, pos] += -dx * h[:m]
        G[d:, neg] += dx * (1.0 - h[:m])
        G[:m, neg] += -dx * h[d:]
    return G


def active_weights(c, top=16, scale=1.0e-5):
    L = len(c)
    k = min(top, L)
    if k <= 0:
        return np.arange(0, dtype=int), np.zeros(0)
    if k == L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -k)[-k:]
    m = float(np.max(c))
    s = (c[idx] - m) / scale
    w = np.exp(s)
    ssum = float(np.sum(w))
    if ssum <= 0.0:
        w = np.zeros_like(c[idx])
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ssum
    return idx, w


def update_Gw(G, Gw, wlag, n, p, delta, dx):
    val = -dx * delta
    G[p, n - 1] += 2.0 * val
    Gw[p] += 2.0 * val * wlag[n - 1]
    if p > 0:
        rows = np.arange(p)
        d = p - rows
        c1 = n - 1 + d
        c2 = n - 1 - d
        G[rows, c1] += val
        G[rows, c2] += val
        Gw[rows] += val * (wlag[c1] + wlag[c2])
    if p + 1 < n:
        rows = np.arange(p + 1, n)
        d = rows - p
        c1 = n - 1 + d
        c2 = n - 1 - d
        G[rows, c1] += val
        G[rows, c2] += val
        Gw[rows] += val * (wlag[c1] + wlag[c2])


def pair_descent(h, n, deadline, base_steps):
    if n < 2:
        return h, c5(h, n)

    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    L = 2 * n - 1
    best = c5(h, n)

    for base in base_steps:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            c = c_vec(h, n)
            G = grad_matrix(h, n)
            bc = float(np.max(c))
            if bc < best:
                best = bc
            else:
                best = bc

            idx, w = active_weights(c, top=16, scale=1.0e-5)
            wlag = np.zeros(L, dtype=float)
            wlag[idx] = w
            Gw = G.dot(wlag)

            improved = False
            accepts = 0
            accept_limit = 150

            for i in range(n - 1):
                if time.time() > deadline:
                    break
                hi = h[i]
                gwi = Gw[i]
                for j in range(i + 1, n):
                    slo = gwi - Gw[j]
                    if abs(slo) * base < 1.0e-11:
                        continue

                    if slo > 0.0:
                        t = -base
                        lim = hi if hi < (1.0 - h[j]) else (1.0 - h[j])
                    else:
                        t = base
                        lim = (1.0 - hi) if (1.0 - hi) < h[j] else h[j]

                    if lim <= 1.0e-13:
                        continue
                    if abs(slo) * lim < 1.0e-11:
                        continue

                    if t > 0.0:
                        if t > lim:
                            t = 0.999999 * lim
                    else:
                        if -t > lim:
                            t = -0.999999 * lim

                    tin = abs(t)
                    if tin < 1.0e-15:
                        continue

                    diff = j - i
                    idx1 = n - 1 + diff
                    idx2 = n - 1 - diff
                    dv = G[i] - G[j]

                    bt = 0.0
                    bv = best
                    bb = None
                    for f in FACTORS:
                        tt = t * f
                        if tt == 0.0:
                            continue
                        bvec = c + tt * dv
                        vv = float(np.max(bvec))
                        t2 = tt * tt * dx
                        v1 = bvec[idx1] + t2
                        v2 = bvec[idx2] + t2
                        if v1 > vv:
                            vv = v1
                        if v2 > vv:
                            vv = v2
                        if vv < bv:
                            bv = vv
                            bt = tt
                            bb = bvec

                    if bv < best - 1.0e-13:
                        c = bb
                        t2 = bt * bt * dx
                        c[idx1] += t2
                        c[idx2] += t2
                        best = bv

                        h[i] += bt
                        h[j] -= bt
                        update_Gw(G, Gw, wlag, n, i, bt, dx)
                        update_Gw(G, Gw, wlag, n, j, -bt, dx)

                        improved = True
                        accepts += 1
                        hi = h[i]
                        gwi = Gw[i]
                        if accepts >= accept_limit:
                            break
                if accepts >= accept_limit:
                    break

            if not improved:
                break

    h = project_feasible(h, n)
    best = c5(h, n)
    return h, best


def smooth_value(x, n, alpha, mu):
    c = c_vec(x, n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    ma = float(np.max(sh))
    e = np.exp(sh - ma)
    s = float(np.sum(e))
    logsum = ma if s <= 0.0 else ma + float(np.log(s))
    val = m + logsum / alpha
    if mu > 0.0:
        s2 = float(np.sum(x) - 0.5 * n)
        val += mu * s2 * s2
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = c_vec(x, n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    ma = float(np.max(sh))
    w = np.exp(sh - ma)
    ssum = float(np.sum(w))
    if ssum <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ssum

    dx = 2.0 / n
    L = 2 * n - 1
    grad = np.zeros(n, dtype=float)
    for idx in range(L):
        wm = float(w[idx])
        if wm < 1.0e-15:
            continue
        k = idx - (n - 1)
        a = dx * wm
        if k >= 0:
            nv = n - k
            grad[:nv] += a * (1.0 - x[k:])
            grad[k:] += -a * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += a * (1.0 - x[:nv])
            grad[:nv] += -a * x[kk:]
    if mu > 0.0:
        grad += 2.0 * mu * (float(np.sum(x)) - 0.5 * n)
    return grad


def lbfgs_phase(h, n, deadline, configs=FULL_CONFIGS):
    h = project_feasible(h, n).copy()
    best = c5(h, n)
    bounds = [(0.0, 1.0)] * n

    for alpha, mu, it in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(
                smooth_value,
                h,
                jac=smooth_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": it, "ftol": 1.0e-15, "gtol": 1.0e-12,
                         "maxfun": it * 8, "maxls": 25},
            )
            x = res.x
            if x is None or not np.all(np.isfinite(x)):
                continue
            x = project_feasible(x, n)
            cx = c5(x, n)
            if cx < best - 1.0e-14:
                best = cx
                h = x
        except Exception:
            pass
    return h, best


def slsqp_obj(x, n, alpha):
    return smooth_value(x, n, alpha, 0.0)


def sum_con(x, n):
    return float(np.sum(x) - 0.5 * n)


def sum_jac(x, n):
    return np.ones(n, dtype=float)


def slsqp_phase(h, n, deadline, alphas=(5.0e4, 2.0e5, 6.0e5)):
    h = project_feasible(h, n).copy()
    best = c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_jac, "args": (n,)}]

    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(
                slsqp_obj,
                h,
                args=(n, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 120, "ftol": 1.0e-13},
            )
            x = res.x
            if x is None or not np.all(np.isfinite(x)):
                continue
            x = project_feasible(x, n)
            cx = c5(x, n)
            if cx < best - 1.0e-14:
                best = cx
                h = x
        except Exception:
            pass
    return h, best


def make_perturb(n, rng, strength):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    p = np.zeros(n, dtype=float)
    maxm = min(10, n // 2)
    for m in range(1, maxm + 1):
        amp = rng.rand() / float(m * m + 1)
        ph = rng.rand() * 6.283185307179586
        p += amp * np.cos(np.pi * m * x + ph)
    p -= float(np.mean(p))
    st = float(np.std(p))
    if st > 1.0e-12:
        p *= strength / st
    return p
