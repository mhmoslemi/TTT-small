import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_topk_obj_grad(h, n, alpha, topk):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    k = min(topk, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_c = c[top_idx]
    m = top_c.max()
    shifted = alpha * (top_c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    Z = exp_s.sum()
    obj = m + (sm + np.log(Z)) / alpha
    w = exp_s / Z
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        wi = w[ii]
        if wi < 1e-12:
            continue
        lag = int(top_idx[ii]) - (n - 1)
        ml = abs(lag)
        if ml == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif ml < n:
            if lag > 0:
                grad[:n - ml] -= wi * dx * h[ml:]
                grad[ml:] += wi * dx * (1.0 - h[:n - ml])
            else:
                grad[:n - ml] += wi * dx * (1.0 - h[ml:])
                grad[ml:] -= wi * dx * h[:n - ml]
    return obj, grad

def pgd_momentum(h0, n, deadline, topk=8):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [50, 200, 1000, 5000, 20000]
    for alpha in alphas:
        seg_end = min(deadline, time.time() + max(20, (deadline - time.time()) * 0.18))
        lr = 2.0 / (alpha * n * 0.01)
        lr = min(lr, 0.5)
        mu = 0.8
        v = np.zeros(n)
        stall = 0
        while time.time() < seg_end:
            obj, grad = smooth_topk_obj_grad(h, n, alpha, topk)
            v = mu * v - lr * grad
            h_new = project_feasible(h + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    lr *= 0.7
                    stall = 0
            h = h_new
            mu = min(mu * 0.9997, 0.95)
    return best_h, best_c

def simulated_annealing(h0, n, deadline, rng):

    best_h = h.copy()

    T0 = 0.003
    T_end = 0.00005

    total_steps = max(total_steps, 1000)

            break
        i = rng.randint(n)
        j = rng.randint(n)

            continue

            if d < 1e-7:

            h[i] += d

            continue

            cur_c = c_new
            if c_new < best_c:

                best_h = h.copy()
        else:

                h[i] += d
                h[j] -= d
            else:
                h[i] -= d
                h[j] += d

        if step % 5000 == 0 and step > 0:

def slsqp_smooth(h, n, alpha, topk, maxiter=2000):
    dx = 2.0 / n

    def obj_func(x):
        o, _ = smooth_topk_obj_grad(x, n, alpha, topk)
        return o

    def jac_func(x):
        _, g = smooth_topk_obj_grad(x, n, alpha, topk)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, jac=jac_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
