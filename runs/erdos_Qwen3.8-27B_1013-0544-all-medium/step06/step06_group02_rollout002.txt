Hmm, this "min of constraints" approach isn't quite right for SLSQP. SLSQP expects each constraint to be non-negative (for 'ineq' type). If I use min, it's not differentiable at the min.

Let me instead use separate constraints. The issue is that SLSQP calls each constraint function independently, so I need each to be a separate function. But I can't define them at runtime without closures.

Alternative: Use a single inequality constraint that is the maximum of all the violations:
max_i (a_i @ x - b_i) ≤ 0

This is equivalent to all a_i @ x ≤ b_i. The gradient is the gradient of the max, which is the a_i for the active one.

    idx = np.argmax(violations)
    return -_G['a_list'][idx].copy()

    m = c.max()

def lbfgs_round(h, n, alpha, mu, maxiter, deadline):

        return h, compute_c5(h, n)

        res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,

def slsqp_round(h, n, alpha, gap, maxiter, deadline):
    if time.time() > deadline - 10:
        return h, compute_c5(h, n)
    build_active_data(h, n, gap)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}]
    try:
        res = minimize(smobj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new

        return h, compute_c5(h, n)

def local_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        while time.time() < deadline:
            improved = False
            g = smgrad(h, n, 200000)
            order = np.argsort(-np.abs(g))
            for i in order[:min(80, n)]:
                if time.time() > deadline:
                    return h, best_c
                partners = [(i + 1) % n, (i - 1) % n, order[0], order[len(order)//2]]
                for j in partners:
                    if j == i:
                        continue
                    for d in [step, -step, step*0.5]:
                        if h[i] + d < 0 or h[i] + d > 1:
                            continue
                        if h[j] - d < 0 or h[j] - d > 1:
                            continue
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                if improved:
                    break
            if not improved:
                break
    return h, best_c
