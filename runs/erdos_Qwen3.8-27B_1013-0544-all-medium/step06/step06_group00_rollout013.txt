    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # Compute softmax weights
    shifted = alpha * (c - c.max())
    exps = np.exp(shifted)
    p = exps / exps.sum()  # shape (2n-1,)
    
    grad = np.zeros(n)
    m_range = np.arange(-(n-1), n)  # lags
    for k in range(2*n-1):
        l = k - (n-1)  # lag
        # Contribution from first factor: (1-h[x+l]) for x+l in [0,n)
        # For shift l: h[x] * (1-h[x+l])
        # d/dh[x] of c[k] = dx * (1-h[x+l]) - dx * h[x-l]

def greedy_binary(h_init, n, deadline, max_sweeps=200):

    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    m_range = np.arange(-(n-1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:

                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0

                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)

    exps = np.exp(shifted)
    val = c.max() + np.log(exps.sum()) / alpha
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        l = k - (n - 1)
        if l >= 0:
            v1 = np.arange(0, n - l)
            grad[v1] += pk * dx * (1.0 - x[v1 + l])
            v2 = np.arange(l, n)
            grad[v2] -= pk * dx * x[v2 - l]
        else:
            ll = -l
            v1 = np.arange(ll, n)
            grad[v1] += pk * dx * (1.0 - x[v1 - ll])
            v2 = np.arange(0, n - ll)
            grad[v2] -= pk * dx * x[v2 + ll]
    return val, grad

def con_fun(x):
    return x.sum() - len(x) / 2.0

def con_jac(x):
    return np.ones(len(x))

def slsqp_solve(h, n, alpha, maxiter=3000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(smooth_obj_fn, h, args=(alpha,), method='SLSQP',
                       jac=True, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=8000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    zeros_arr = np.where(h == 0)[0]
    ones_arr = np.where(h == 1)[0]
    T = 0.025
    cool = 0.99997
    scale = 1500.0
    m_range = np.arange(-(n - 1), n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i, j = zeros_arr[ai], ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        delta[n - 1 + (j - i)] += 1.0
        delta[n - 1 + (i - j)] += 1.0
        nc = (corr + delta).max() * dx
        accept = nc <= cur_c or (T > 1e-14 and rng.random() < np.exp(-scale * (nc - cur_c) / T))
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        T *= cool
    return best_h, best_c

def alpha_schedule(start=20, end=200000000, n_levels=25):
    return [start * (end / start) ** (i / (n_levels - 1)) for i in range(n_levels)]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: binary greedy (8%)
    p1 = t0 + 0.08 * total
    hb, cb = greedy_binary(h0, n, p1, max_sweeps=80)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: smooth optimization with progressive alpha (45%)
    p2 = t0 + 0.53 * total
    alphas = alpha_schedule(20, 200000000, n_levels=25)
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 8:
            break
        h = slsqp_solve(h, n, alpha, maxiter=5000, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        h = slsqp_solve(h, n, alpha * 2, maxiter=2000, ftol=1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 3: binary annealing (12%)
    p3 = t0 + 0.65 * total
    ha, ca = anneal_binary(best_h, n, p3, seed=seed, n_iters=8000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    # Phase 4: re-smooth from best (10%)
    p4 = t0 + 0.75 * total
    h = best_h.copy()
    for alpha in [50000, 500000, 5000000, 50000000, 200000000]:
        if time.time() > p4 - 3:
            break
        h = slsqp_solve(h, n, alpha, maxiter=3000, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 5: multi-resolution (10%)
    p5 = t0 + 0.85 * total
    for nh in [200, 350, 500]:
        if time.time() > p5 - 10:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [50000, 500000, 5000000]:
            if time.time() > p5 - 5:
                break
            hh = slsqp_solve(hh, nh, alpha, maxiter=2000, ftol=1e-15)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h = ch, hh.copy()
    # Phase 6: diverse restarts (remaining)
    p6 = dl - 3
    for trial in range(8):
        if time.time() > p6 - 5:
            break
        if trial == 0:
            h = best_h + 0.002 * rng.randn(n)
        elif trial == 1:
            h = np.clip(1.0 - best_h, 0, 1)
            h = project_feasible(h, n)
        elif trial == 2:
            h = best_h.copy()
            block = n // 4
            h[:block] = 1.0 - h[:block]
            h = project_feasible(h, n)
        else:
            h = best_h + 0.005 * (trial % 3 + 1) * rng.randn(n)
            h = project_feasible(h, n)
        for alpha in [100000, 5000000, 100000000]:
            if time.time() > p6 - 3:
                break
            h = slsqp_solve(h, n, alpha, maxiter=2000, ftol=1e-17)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
