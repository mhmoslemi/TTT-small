import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_lp_matrix(y, n):
    """Build the (2n-1) x n matrix A where A[m,i] gives coeff of h[i] in c_m."""
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            # c_m = sum_{i=0}^{n-1-k} h[i] * y[i+k]
            for i in range(n-k):
                A[m, i] = y[i+k]
        else:
            # k < 0, kk = -k > 0
            kk = -k
            # c_m = sum_{i=kk}^{n-1} h[i] * y[i-kk]
            for i in range(kk, n):
                A[m, i] = y[i-kk]
    return A

def solve_lp(y, n):
    """Solve the LP: min t s.t. dx*A*h <= t, sum(h)=n/2, 0<=h<=1"""
    dx = 2.0 / n
    A = build_lp_matrix(y, n)
    # Objective: minimize t (last variable)
    c_obj = np.concatenate([np.zeros(n), [1.0]])
    # Inequality: dx * A @ h - t <= 0
    # [dx*A | -1] @ [h; t] <= 0
    A_ub = np.hstack([dx * A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    # Equality: sum(h) = n/2
    A_eq = np.concatenate([np.ones(1, n), [0.0]])
    b_eq = np.array([n / 2.0])
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def alternating_lp(h, n, max_iters=100, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        y = 1.0 - h_cur
        h_new = solve_lp(y, n)
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        
        # Check convergence
        if np.max(np.abs(h_new - h_cur)) < 1e-10:
            break
        h_cur = h_new
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(max(0,i-80), min(n, i+81)):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def lbfags_refine(h, n, deadline):
    """Use L-BFGS-B with smooth max for refinement."""
    from scipy.special import logsumexp
    
    def obj_smooth(x, alpha):
        c = np.correlate(x, 1-x, mode='full') * (2.0/n)
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        return m + np.exp(logsumexp(s - mx) - mx) / alpha
    
    def obj(x):
        return obj_smooth(x, 500.0)
    
    bounds = [(0.0, 1.0)] * n
    # Use penalty for constraint
    mu = 100.0
    def obj_pen(x):
        c = np.correlate(x, 1-x, mode='full') * (2.0/n)
        m = c.max()
        s = 500.0 * (c - m)
        mx = s.max()
        val = m + np.exp(logsumexp(s - mx) - mx) / 500.0
        val += mu * (x.sum() - n/2.0)**2
        return val
    
    def grad(x):
        c = np.correlate(x, 1-x, mode='full') * (2.0/n)
        # Simplified gradient
        g = np.zeros(n)
        w = np.exp(c - c.max())
        w /= w.sum()
        dx = 2.0/n
        for m in range(2*n-1):
            if w[m] < 1e-12: continue
            k = m - (n-1)
            wm = w[m] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1-x[k:n])
                g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1-x[0:nv])
                g[0:nv] += wm * (-x[kk:n])
        g += 2.0 * mu * (x.sum() - n/2.0)
        return g
    
    best_c = compute_c5(h, n)
    best_h = h.copy()
    try:
        res = minimize(obj_pen, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 200, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    except:
        pass
    return best_h, best_c

For m = n-1 (zero lag): max(0, n-1-(n-1)) = 0, min(n, n) = n. So n ranges 0 to n-1.
result[n-1] = sum_{n=0}^{n-1} h[n] * y[n] ✓

For m = n (lag 1): max(0, n-(n-1)) = 1, min(n, n+1) = n. So n ranges 1 to n-1.
result[n] = sum_{n=1}^{n-1} h[n] * y[n+1] 

Hmm wait, that doesn't seem right. Let me check with numpy directly.

np.correlate([a,b,c], [x,y,z], mode='full') = 
  [a*z, a*y+b*z, a*x+b*y+c*z, b*x+c*y, c*x]

So for n=3:
m=0: h[0]*y[2]
m=1: h[0]*y[1] + h[1]*y[2]

m=3: h[1]*y[0] + h[2]*y[1]

So in general, for m (0-indexed):
k = m - (n-1)
If k >= 0: result[m] = sum_{i=0}^{n-1-k} h[i] * y[i+k]
If k < 0: result[m] = sum_{i=-k}^{n-1} h[i] * y[i+k] = sum_{i=-k}^{n-1} h[i] * y[i-|k|]

This matches what I had before. Good.

For each m:

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_lp_matrix(y, n):
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            A[m, :n - k] = y[k:n]
        else:
            kk = -k
            A[m, kk:] = y[:n - kk]
    return A

def solve_lp(y, n):
    dx = 2.0 / n
    A = build_lp_matrix(y, n)
    c_obj = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([dx * A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.concatenate([np.ones(1, n), [0.0]])
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def alternating_lp(h, n, max_iters=100, deadline=None, omega=1.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    stall = 0
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        y = 1.0 - h_cur
        h_new = solve_lp(y, n)
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
        h_cur = (1.0 - omega) * h_cur + omega * h_new
        if np.max(np.abs(h_new - h_cur)) < 1e-12:
            break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=80):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def lbfgs_refine(h, n, deadline, alpha=500.0, mu=50.0):
    dx = 2.0 / n

    def obj_pen(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        val = m + np.exp(logsumexp(s - mx) - mx) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)

    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        s = alpha * (c - c.max())
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-12:
                continue
            k = m - (n - 1)
            wm = w[m] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - x[k:n])
                g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - x[0:nv])
                g[0:nv] += wm * (-x[kk:n])
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 150, 'ftol': 1e-16, 'maxfun': 500})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    except Exception:
        pass
    return best_h, best_c
