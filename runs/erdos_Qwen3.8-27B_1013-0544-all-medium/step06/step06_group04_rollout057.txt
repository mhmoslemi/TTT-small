```python
def subgradient_step(h, n, alpha=100, eta=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    max_c = corr.max()
    # Active set: lags where corr is close to max
    S = np.where(corr >= max_c - alpha * dx)[0]
    if len(S) == 0:
        S = np.where(corr == max_c)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in S:
        shift = int(k) - (n-1)  # k - (n-1) is the actual shift
        # ∂f_k/∂h[s] = (1-h[s-shift]) - h[s+shift] (with validity)
        # s-shift valid when 0 <= s-shift < n
        # s+shift valid when 0 <= s+shift < n
        for s in range(n):
            g = 0.0
            if 0 <= s - shift < n:
                g += 1 - h[s - shift]
            if 0 <= s + shift < n:
                g -= h[s + shift]
            grad[s] += g
    grad *= dx / len(S)
    
    if eta is None:
        # Polak-Ribiere step size or fixed
        eta = 0.01
    
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new
```
