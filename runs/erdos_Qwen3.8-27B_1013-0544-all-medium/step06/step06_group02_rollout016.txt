```python
    c = compute_c5_vec(h, n)
    # Find top k indices
    idx = np.argsort(c)[-n_active:]
    dx = 2.0 / n
    gradients = np.zeros((len(idx), n))
    for m, k_idx in enumerate(idx):

            g[0:nv] += (1.0 - h[k:n])

            g[kk:n] += (1.0 - h[0:nv])
            g[0:nv] += (-h[kk:n])

    return c, idx, gradients
```

```python
import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

    return val

    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx] * alpha
        if wm < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * dx * (1.0 - h[k:n])
            grad[k:n] += wm * dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * dx * (1.0 - h[0:nv])
            grad[0:nv] += wm * dx * (-h[kk:n])
    return grad

def p_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n/2.0)**2

def p_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

def run_lbfgs(h, n, deadline, maxit=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    mus = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 5000.0]
    for i in range(len(alphas)):
        if time.time() > deadline - 5:
            break
        alpha, mu = alphas[i], mus[i]
        try:
            res = minimize(p_obj, h, jac=p_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': maxit*20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except:
            pass
    return h, best_c

def grad_pair_search(h, n, deadline):

    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

            cmax = c.max()
            thresh = cmax - 0.02 * cmax

                    wg[0:nv] += w * (-h[kk:n])

def kicked_local_search(h, n, deadline, rng):
```

```python
    # For coordinate i:
    # f(h_i) = h_i - h_i * (sum_k p_k * (h_{i+k} + h_{i-k})) + ... 
    # Wait, let me be more careful.
    # sum_k p_k * sum_i h_i * h_{i+k} = sum_i sum_k p_k h_i h_{i+k}
    # Terms with h_i: sum_k p_k h_i h_{i+k} (first index is i) + sum_k p_k h_{i-k} h_i (second index is i)
    # = h_i * (sum_k p_k h_{i+k} + sum_k p_k h_{i-k}) [summing over valid indices]
    # 
    # So f(h) = sum_j h_j - sum_i h_i * (sum_k p_k h_{i+k} + sum_k p_k h_{i-k})
    # = sum_i h_i * [1 - sum_k p_k h_{i+k} - sum_k p_k h_{i-k}]
    #   [with h_{*} = 0 for * out of range]
    # 
    # For coordinate i, fixing others:
    # f = const + h_i * [1 - sum_k p_k h_{i+k} - sum_k p_k h_{i-k}]
    #   where the sum excludes k=0 for the "h_i" part... 
    #   Actually for k=0: p_0 * h_i * h_i appears.
    # 
    # Let me redo: the quadratic term involving h_i is:
    # sum_k p_k h_i h_{i+k} + sum_k p_k h_{i-k} h_i
    # = h_i * sum_k p_k (h_{i+k} + h_{i-k})
    # For k=0: h_i * 2*p_0*h_i = 2*p_0*h_i²
    # For k≠0: h_i * p_k * (h_{i+k} + h_{i-k})
    # 
    # So: f = const + h_i - 2*p_0*h_i² - h_i * sum_{k≠0} p_k(h_{i+k} + h_{i-k})
    # = const + h_i*(1 - sum_{k≠0} p_k(h_{i+k} + h_{i-k})) - 2*p_0*h_i²
    # 
    # This is a concave quadratic in h_i (coefficient of h_i² is -2*p_0 ≤ 0)
    # Minimum over [0,1] is at a boundary or at the vertex.
    pass
```

This is getting very involved. Let me just write a practical implementation that combines the best ideas:

1. L-BFGS-B with smooth max (proven to work)
2. Gradient-guided pair search (faster than blind search)
3. A simple coordinate descent step (try h_i = 0 and h_i = 1 for each i)
4. Random kicks to escape local optima
5. Multiple n values

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx] * alpha
        if wm < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * dx * (1.0 - h[k:n])
            grad[k:n] += wm * dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * dx * (1.0 - h[0:nv])
            grad[0:nv] += wm * dx * (-h[kk:n])
    return grad

def _pobj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _pgrad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, maxit=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 5.0), (200, 10.0), (1000, 20.0), (5000, 50.0),
               (20000, 100.0), (100000, 200.0), (500000, 500.0),
               (2000000, 2000.0), (10000000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_pobj, h, jac=_pgrad, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14,
                                    'maxfun': maxit * 20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_gradient(h, n, tol_frac=0.02):
    c = compute_c5_vec(h, n)
    cmax = c.max()
    thresh = cmax - tol_frac * cmax
    active = np.where(c >= thresh)[0]
    dx = 2.0 / n
    wg = np.zeros(n)
    for m_idx in active:
        w = c[m_idx]
        k = m_idx - (n - 1)
        if k >= 0:
            nv = n - k
            wg[0:nv] += w * (1.0 - h[k:n])
            wg[k:n] += w * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            wg[kk:n] += w * (1.0 - h[0:nv])
            wg[0:nv] += w * (-h[kk:n])
    return wg

def fast_pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        while time.time() < deadline:
            wg = active_gradient(h, n)
            order = np.argsort(wg)
            improved = False
            count = 0
            for ai in range(n):
                if time.time() > deadline:
                    return h, best_c
                i = order[ai]
                for bj in range(n - 1, ai, -1):
                    j = order[bj]
                    if wg[j] - wg[i] < 1e-8:
                        break
                    if h[j] - d >= 0 and h[i] + d <= 1:
                        h[i] += d; h[j] -= d
                        c_new = compute_c5(h, n)
                        count += 1
                        if c_new < best_c - 1e-14:
                            best_c = c_new; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0 and h[j] + d <= 1:
                        h[i] -= d; h[j] += d
                        c_new = compute_c5(h, n)
                        count += 1
                        if c_new < best_c - 1e-14:
                            best_c = c_new; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def coord_bangbang(h, n, deadline):
    best_c = compute_c5(h, n)
    target = n / 2.0
    for sweep in range(500):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old = h[i]
            for new_val in (0.0, 1.0):
                if abs(new_val - old) < 1e-10:
                    continue
                h[i] = new_val
                h_proj = project_feasible(h, n)
                if abs(h_proj[i] - h[i]) < 1e-10:
                    c_new = compute_c5(h_proj, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        h = h_proj
                        improved = True
                        break
                h[i] = old
        if not improved:
            break
    return h, best_c

def kicked_search(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    kick_schedule = [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01]
    for kick_size in kick_schedule:
        while time.time() < deadline - 20:
            alloc = min(15, deadline - time.time() - 5)
            if alloc < 5:
                break
            h, c = run_lbfgs(best_h, n, time.time() + alloc * 0.4, maxit=100)
            if time.time() < deadline - 10:
                h, c = fast_pair_search(h, n, min(time.time() + 8, deadline - 5))
            if c < best_c - 1e-13:
                best_c = c
                best_h = h.copy()
            else:
                idx = rng.choice(n, size=rng.randint(3, 15), replace=False)
                h_new = best_h.copy()
                h_new[idx] += rng.uniform(-kick_size, kick_size, size=len(idx))
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c + 0.005:
                    best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Main L-BFGS optimization
    t1 = t0 + 350
    best_h, best_c = run_lbfgs(best_h, n, t1, maxit=400)

    # Phase 2: Gradient-guided pair search
    if time.time() < deadline - 400:
        t2 = min(t0 + 550, deadline - 400)
        best_h, best_c = fast_pair_search(best_h, n, t2)

    # Phase 3: More L-BFGS with refinement
    if time.time() < deadline - 300:
        t3 = min(t0 + 700, deadline - 300)
        best_h, best_c = run_lbfgs(best_h, n, t3, maxit=500)

    # Phase 4: Coordinate bang-bang + pair search
    if time.time() < deadline - 200:
        t4 = min(t0 + 800, deadline - 200)
        best_h, best_c = coord_bangbang(best_h, n, t4)
    if time.time() < deadline - 150:
        t5 = min(t0 + 870, deadline - 150)
        best_h, best_c = fast_pair_search(best_h, n, t5)

    # Phase 5: Kicked local search
    if time.time() < deadline - 100:
        t6 = min(t0 + 1020, deadline - 100)
        best_h, best_c = kicked_search(best_h, n, t6, rng)

    # Phase 6: Try different n values
    if time.time() < deadline - 50:
        for n_try in [200, 400, 500]:
            if time.time() > deadline - 30:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 20)
            if alloc < 20:
                break
            h_t, c_t = run_lbfgs(h_interp, n_try, time.time() + alloc * 0.5, maxit=200)
            if time.time() < deadline - 10:
                h_t, c_t = fast_pair_search(h_t, n_try, min(time.time() + 20, deadline - 5))
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 7: Final polish
    nf = len(best_h)
    if time.time() < deadline - 3:
        best_h, best_c = run_lbfgs(best_h, nf, deadline - 1, maxit=100)
    if time.time() < deadline - 1:
        best_h, best_c = fast_pair_search(best_h, nf, deadline - 0.5)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
