        # Need to flip some 1s to 0s
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        for i in idx[:current_ones - target]:
            h[i] = 0
    else:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        for i in idx[:target - current_ones]:
            h[i] = 1

def gs_flatten(n, n_iter=2000, rng=None):
    """Gerchberg-Saxton style spectral flattening for binary sequences."""
    if rng is None:
        rng = np.random.RandomState(42)
    # Initialize with Thue-Morse
    b = thue_morse(n).astype(float) * 2 - 1  # ±1
    # Adjust to have equal +1 and -1
    target = n // 2
    pos = np.where(b == 1)[0]
    neg = np.where(b == -1)[0]
    if len(pos) > target:
        idx = rng.choice(pos, len(pos) - target, replace=False)
        b[idx] = -1
    elif len(neg) > target:
        idx = rng.choice(neg, len(neg) - target, replace=False)
        b[idx] = 1
    
    best_b = b.copy()
    best_min_ac = -np.inf
    
    for it in range(n_iter):
        # Compute periodic autocorrelation
        B = np.fft.fft(b)
        P = np.abs(B)**2
        # Flatten the power spectrum (keep DC)
        # P[0] should be 0 (since sum b = 0)
        # Compress dynamic range
        alpha = 0.3 + 0.4 * (1 - it/n_iter)  # start aggressive, become gentle
        P_new = P ** alpha
        # Preserve the total power
        P_new = P_new * P.sum() / P_new.sum()
        # Update: new B has same phase, new magnitude
        phase = B / (np.abs(B) + 1e-30)
        B_new = phase * np.sqrt(P_new)
        b_new = np.fft.ifft(B_new).real
        # Threshold to binary
        b_bin = np.sign(b_new)
        b_bin[b_bin == 0] = 1
        # Rebalance
        pos = np.where(b_bin == 1)[0]
        neg = np.where(b_bin == -1)[0]
        if len(pos) > target:
            # Flip the weakest +1s
            order = np.argsort(b_new[pos])
            flip = order[:len(pos) - target]
            b_bin[pos[flip]] = -1
        elif len(neg) > target:
            order = np.argsort(-b_new[neg])
            flip = order[:len(neg) - target]
            b_bin[neg[flip]] = 1
        b = b_bin
        # Evaluate
        h = (b + 1) / 2
        c5 = compute_c5(h, n)
        if c5 < best_min_ac or best_min_ac == -np.inf:
            best_min_ac = c5
            best_b = b.copy()
    return (best_b + 1) / 2, best_min_ac

def swap_delta_fast(h, i, j, n):
    """Compute delta in correlation when swapping h[i] and h[j]."""
    # We swap: h[i] becomes 1-h[i], h[j] becomes 1-h[j]
    # (assuming binary: h[i] goes 0->1 or 1->0)
    # delta_c[k] for the full correlation
    delta = np.zeros(2 * n - 1)
    bi = h[i]
    bj = h[j]
    # Effect of changing h[i] from bi to 1-bi:
    # c[k] = sum_j h[j+k](1-h[j])
    # Changing h[i]: affects terms where j=i (the (1-h[j]) part) and j+k=i (the h[j+k] part)
    # (1-h[i]) changes from (1-bi) to bi: delta = 2*bi - 1
    # h[i] changes from bi to 1-bi: delta = 2*bi - 1 (in the h[j+k] position)
    
    d_i = 2.0 * bi - 1.0  # change in h[i]
    d_j = 2.0 * bj - 1.0
    
    # For the (1-h[j]) part: c[k] includes (1-h[j]) * h[j+k]
    # When h[i] changes by d_i, (1-h[i]) changes by -d_i
    # This affects c[k] for all k where j=i, i.e., c[n-1+k] for all valid k
    # c[n-1+k] = sum_j h[j+k](1-h[j]), the term with j=i is h[i+k](1-h[i])
    # delta from (1-h[i]) change: -d_i * h[i+k] for each k
    for k in range(-(n-1), n):
        idx = n - 1 + k
        if 0 <= idx < 2*n - 1:
            j_pos = i + k
            if 0 <= j_pos < n:
                # (1-h[i]) changes by -d_i, multiplied by h[i+k]
                delta[idx] += -d_i * h[j_pos]
            j_pos2 = i + k  # same
            # h[i+k] changes by d_i (when i+k is the "h[j+k]" position, j is arbitrary)
            # Actually, h[i] appears in h[j+k] when j+k=i, i.e., j=i-k
            if 0 <= i - k < n:
                # h[i] as the "h[j+k]" term, with j = i-k
                # (1-h[i-k]) is the other factor
                delta[idx] += d_i * (1 - h[i - k])
    
    # Similarly for h[j]
    for k in range(-(n-1), n):
        idx = n - 1 + k
        j_pos = j + k
        if 0 <= j_pos < n:
            delta[idx] += -d_j * h[j_pos]
        if 0 <= j - k < n:
            delta[idx] += d_j * (1 - h[j - k])
    
    return delta

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def evaluate_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full').copy(), dx

def project_to_binary(h, n):
    """Project a real vector to a binary vector with exactly n//2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def thue_morse_seq(n):
    """Thue-Morse sequence as 0/1 array."""
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

    # Initialize with Thue-Morse
    b = thue_morse_seq(n).copy()

    ones = np.where(b == 1)[0]

def swap_delta_vec(h, i, j, n):
    """Vectorized computation of correlation delta when swapping positions i and j."""
    # h is binary. Swapping means h[i] goes to 1-h[i], h[j] goes to 1-h[j]
    # corr[k] = sum_m h[m+k] * (1-h[m]) for the full correlation (index n-1+k)
    # Let's compute the change.
    # After swap: h'[i] = 1-h[i], h'[j] = 1-h[j]
    # delta[k] = corr_new[k] - corr_old[k]
    
    # corr[n-1+k] = sum_m h[m+k](1-h[m])
    # Terms affected:
    # - m = i: h[i+k](1-h[i]) -> h[i+k](1-h'[i]) = h[i+k]*h[i] (since h is binary, 1-h'[i] = h[i])
    #   delta = h[i+k]*(h[i] - (1-h[i])) = h[i+k]*(2*h[i]-1)
    # - m+k = i, i.e., m = i-k: h[i](1-h[i-k]) -> h'[i](1-h[i-k]) = (1-h[i])(1-h[i-k])
    #   delta = (1-h[i])(1-h[i-k]) - h[i](1-h[i-k]) = (1-h[i] - h[i])(1-h[i-k]) = (1-2h[i])(1-h[i-k])
    #   = -(2h[i]-1)(1-h[i-k])
    
    di = 2.0 * h[i] - 1.0  # = 1 if h[i]=1, = -1 if h[i]=0
    dj = 2.0 * h[j] - 1.0
    
    delta = np.zeros(2*n-1)
    
    # Contribution from position i as "m" (the (1-h[m]) factor):
    # For each k: m=i, need 0 <= i+k < n
    # delta_c[n-1+k] += h[i+k] * di
    k_range_i = np.arange(max(0, -(i)), min(n-1, n-1-i))  # k such that 0 <= i+k < n
    # Actually: 0 <= i+k < n => -i <= k < n-i
    k_lo = max(0, -i)
    k_hi = min(n-1, n-1-i)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi+1)
        delta[n-1+ks] += h[i+ks] * di
    
    # Contribution from position i as "m+k" (the h[m+k] factor):
    # For each k: m+k=i, m=i-k, need 0 <= i-k < n
    # delta_c[n-1+k] += -di * (1-h[i-k])
    k_lo2 = max(0, i-(n-1))
    k_hi2 = min(n-1, i)
    if k_lo2 <= k_hi2:
        ks = np.arange(k_lo2, k_hi2+1)
        delta[n-1+ks] += -di * (1-h[i-ks])
    
    # Same for j
    k_lo = max(0, -j)
    k_hi = min(n-1, n-1-j)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi+1)
        delta[n-1+ks] += h[j+ks] * dj
    
    k_lo2 = max(0, j-(n-1))
    k_hi2 = min(n-1, j)
    if k_lo2 <= k_hi2:
        ks = np.arange(k_lo2, k_hi2+1)
        delta[n-1+ks] += -dj * (1-h[j-ks])
    
    return delta

                break

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def thue_morse_seq(n):
    tm = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm.astype(float)

def rebalance(h, n, rng):
    target = n // 2
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        flip = rng.choice(ones, len(ones) - target, replace=False)
        h = h.copy()
        h[flip] = 0
    elif len(ones) < target:
        zeros = np.where(h == 0)[0]
        flip = rng.choice(zeros, target - len(ones), replace=False)
        h = h.copy()
