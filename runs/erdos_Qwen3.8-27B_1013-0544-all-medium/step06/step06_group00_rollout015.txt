    corr = np.correlate(h, 1-h, mode='full')
    v = corr * dx
    # Find top M

            if s < 1e-14:

            h += diff / s * h

    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    v = corr * dx
    v_max = v.max()
    log_sum = np.log(np.exp(alpha * (v - v_max)).sum()) / alpha
    return float(v_max + log_sum)

def smooth_max_grad(h, n, alpha, M=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    v = corr * dx
    top_idx = np.argsort(v)[-M:]
    top_v = v[top_idx]
    w = np.exp(alpha * (top_v - top_v.max()))
    w /= w.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    for wm, ki in zip(w, top_idx):
        if wm < 1e-12:
            continue
        shift = int(ki) - (n - 1)
        jp = idx + shift
        jm = idx - shift
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        grad[mask_p] += wm * dx * (1.0 - h[jp[mask_p]])
        grad[mask_m] -= wm * dx * h[jm[mask_m]]
    return grad

def _obj(x):
    return smooth_max_obj(x, _opt_state['n'], _opt_state['alpha'])

def _grad(x):
    return smooth_max_grad(x, _opt_state['n'], _opt_state['alpha'])

def _cons_fun(x):
    return x.sum() - _opt_state['n'] / 2.0

def _cons_jac(x):
    return np.ones(_opt_state['n'])

def slsqp_opt(h0, n, alpha, maxiter, ftol):
    _opt_state['n'] = n
    _opt_state['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h0, method='SLSQP', jac=_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def proj_grad_descent(h, n, alpha, n_iter, lr_init, lr_decay):
    h_cur = h.copy()
    for i in range(n_iter):
        grad = smooth_max_grad(h_cur, n, alpha, M=50)
        lr = lr_init * (lr_decay ** i)
        h_cur = project_feasible(h_cur - lr * grad, n)
    return h_cur

import numpy as np
from scipy.optimize import minimize
import time

_opt_state = {}

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    v = corr * dx
    v_max = v.max()
    return float(v_max + np.log(np.exp(alpha * (v - v_max)).sum()) / alpha)

def smooth_max_grad(h, n, alpha, M=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    v = corr * dx
    top_idx = np.argsort(v)[-M:]
    top_v = v[top_idx]
    w = np.exp(alpha * (top_v - top_v.max()))
    w /= w.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    for wm, ki in zip(w, top_idx):
        if wm < 1e-12:
            continue
        k = int(ki) - (n - 1)
        jm = idx - k
        jp = idx + k
        mask_m = (jm >= 0) & (jm < n)
        mask_p = (jp >= 0) & (jp < n)
        grad[mask_m] += wm * dx * (1.0 - h[jm[mask_m]])
        grad[mask_p] -= wm * dx * h[jp[mask_p]]
    return grad

def _obj(x):
    return smooth_max_obj(x, _opt_state['n'], _opt_state['alpha'])

def _grad(x):
    return smooth_max_grad(x, _opt_state['n'], _opt_state['alpha'], _opt_state['M'])

def _cons_fun(x):
    return x.sum() - _opt_state['n'] / 2.0

def _cons_jac(x):
    return np.ones(_opt_state['n'])

def slsqp_opt(h0, n, alpha, maxiter, ftol, M=50):
    _opt_state['n'] = n
    _opt_state['alpha'] = alpha
    _opt_state['M'] = M
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h0, method='SLSQP', jac=_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def proj_grad(h, n, alpha, n_iter, lr_init, lr_decay, M=50):
    h_cur = h.copy()
    for i in range(n_iter):
        grad = smooth_max_grad(h_cur, n, alpha, M)
        lr = lr_init * (lr_decay ** i)
        h_cur = project_feasible(h_cur - lr * grad, n)
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:

        n = len(h)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha (main workhorse with correct gradient)
    p1_end = t0 + 0.55 * total

        (100, 400, 1e-13, 30),

        (1000, 350, 1e-14, 40),

        (20000, 250, 1e-15, 50),
