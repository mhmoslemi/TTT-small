def smooth_c5_obj_x(x, n, alpha, lam):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    sh = alpha*(c-m)
    se = np.exp(sh).sum()
    obj = m + log(se)/alpha
    if lam: obj += lam*np.sum(x*(1-x))
    return obj
def smooth_c5_jac(x, n, alpha, lam):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    sh = alpha*(c-m)
    expsh = np.exp(sh)
    se = expsh.sum()
    W = expsh / se

    L=n-1
    grad = np.convolve(1-x, W[::-1], mode='full')[L:L+n] - np.convolve(x, W, mode='full')[L:L+n]
    if lam: grad += lam*(1-2*x)

def project_simplex(z, target):
    n = len(z)
    # if target <=0 etc
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    # sum clip(z - t) is decreasing in t
    for _ in range(80):
       mid = (lo+hi)/2
       s = np.clip(z-mid,0,1).sum()
       if s > target: lo = mid
       else: hi = mid
    t=(lo+hi)/2

import numpy as np
import time
from scipy.optimize import minimize

_OPT = {'alpha': 100.0, 'lam': 1e-8, 'target': 125.0}
_YOPT = {'alpha': 300.0, 'mu': 2.0, 'target': 125.0}


def clean_array(a):
    a = np.asarray(a, dtype=float)
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.5)
    return a


def fix_sum(x, target):
    x = clean_array(x)
    x = np.clip(x, 0.0, 1.0)
    for _ in range(30):
        diff = float(target - np.sum(x))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - x
            s = float(np.sum(room))
            if s <= 1e-15:
                break
            x = x + (diff / s) * room
        else:
            s = float(np.sum(x))
            if s <= 1e-15:
                break
            x = x + (diff / s) * x
        np.clip(x, 0.0, 1.0, out=x)
    return x


def fast_project(z, target):
    z = clean_array(z)
    return fix_sum(np.clip(z, 0.0, 1.0), target)


def project_simplex(z, target):
    z = clean_array(z)
    n = int(z.size)
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    if (np.all(z >= -1e-12) and np.all(z <= 1.0 + 1e-12)
            and abs(float(np.sum(z)) - float(target)) < 1e-10):
        return np.clip(z, 0.0, 1.0)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return fix_sum(np.clip(z - 0.5 * (lo + hi), 0.0, 1.0), target)


def final_adjust(h, n):
    h = clean_array(h)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    h = fix_sum(h, target)
    for _ in range(8):
        diff = float(target - np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            i = int(np.argmax(room))
            add = min(diff, float(room[i]))
            if add > 0.0:
                h[i] += add
        else:
            i = int(np.argmax(h))
            sub = min(-diff, float(h[i]))
            if sub > 0.0:
                h[i] -= sub
    return np.clip(h, 0.0, 1.0)


def count_vector(h):
    h = clean_array(h)
    return np.correlate(h, 1.0 - h, mode='full')


def max_count(h):
    return float(np.max(count_vector(h)))


def c5_bound(h, n):
    return max_count(h) * (2.0 / float(n))


def softmax_c(c, alpha):
    m = float(np.max(c))
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = m
    else:
        w = e / s
        val = m + float(np.log(s)) / alpha
    return val, m, w


def gradient_from_weights(x, w):
    n = int(x.size)
    L = n - 1
    g1 = np.convolve(1.0 - x, w, mode='full')
    g2 = np.convolve(x, w[::-1], mode='full')
    return g1[L:L + n] - g2[L:L + n]


def smooth_jac_info(x, alpha):
    x = clean_array(x)
    c = count_vector(x)
    val, m, w = softmax_c(c, alpha)
    g = gradient_from_weights(x, w)
    g -= float(np.mean(g))
    return g, val, m


def smooth_obj(x, alpha, lam=0.0):
    c = count_vector(x)
    val, _, _ = softmax_c(c, alpha)
    if lam != 0.0:
        val += lam * float(np.sum(x * (1.0 - x)))
    return float(val)


def smooth_jac(x, alpha, lam=0.0):
    g, _, _ = smooth_jac_info(x, alpha)
    if lam != 0.0:
        g = g + lam * (1.0 - 2.0 * x)
    g = g - float(np.mean(g))
    return g


def _slsqp_obj(x):
    return smooth_obj(x, _OPT['alpha'], _OPT['lam'])


def _slsqp_jac(x):
    return smooth_jac(x, _OPT['alpha'], _OPT['lam'])


def _slsqp_eq(x):
    return float(np.sum(x) - _OPT['target'])


def slsqp_step(x, n, alpha, maxiter, deadline, lam=1e-8):
    if time.time() >= deadline:
        return x
    target = 0.5 * float(n)
    _OPT['alpha'] = float(alpha)
    _OPT['lam'] = float(lam)
    _OPT['target'] = target
    x0 = project_simplex(x, target)
    bounds = [(0.0, 1.0)] * int(n)
    cons = {'type': 'eq', 'fun': _slsqp_eq}
    try:
        res = minimize(_slsqp_obj, x0, jac=_slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': 1e-12})
        x1 = res.x
    except Exception:
        return x
    x1 = clean_array(x1)
    return project_simplex(x1, target)


def sigmoid(y):
    y = np.clip(clean_array(y), -12.0, 12.0)
    return np.clip(1.0 / (1.0 + np.exp(-y)), 1e-12, 1.0 - 1e-12)


def _logit_obj(y):
    h = sigmoid(y)
    c = count_vector(h)
    val, _, _ = softmax_c(c, _YOPT['alpha'])
    diff = float(np.sum(h) - _YOPT['target'])
    val += _YOPT['mu'] * diff * diff
    return float(val)


def _logit_jac(y):
    h = sigmoid(y)
    c = count_vector(h)
    _, _, w = softmax_c(c, _YOPT['alpha'])
    g = gradient_from_weights(h, w)
    diff = float(np.sum(h) - _YOPT['target'])
    g = g + 2.0 * _YOPT['mu'] * diff
    return g * (h * (1.0 - h))


def slsqp_logit(h, n, alpha, mu, maxiter, deadline):
    if time.time() >= deadline:
        return h
    target = 0.5 * float(n)
    _YOPT['alpha'] = float(alpha)
    _YOPT['mu'] = float(mu)
    _YOPT['target'] = target
    hp = np.clip(clean_array(h), 1e-8, 1.0 - 1e-8)
    y0 = np.log(hp / (1.0 - hp))
    y0 = np.clip(y0, -8.0, 8.0)
    try:
        res = minimize(_logit_obj, y0, jac=_logit_jac, method='L-BFGS-B',
                       options={'maxiter': int(maxiter), 'ftol': 1e-12,
                                'gtol': 1e-8,
                                'maxfun': int(max(1000, maxiter * 20))})
        h1 = sigmoid(res.x)
    except Exception:
        return h
    h1 = clean_array(h1)
    return project_simplex(h1, target)

def round_balanced(h, n):
    target = n // 2
    order = np.argsort(-h)
    hb = np.zeros(n, dtype=float)
    hb[order[:target]] = 1.0
    return hb


def binary_polish(h, n, deadline, topk=70, seed=42):
    if n % 2 != 0 or n < 6:
        hb = round_balanced(h, n)
        return hb, float(max_count(hb))
    target = n // 2
    if target <= 0 or target >= n:
        hb = round_balanced(h, n)
        return hb, float(max_count(hb))
    rng = np.random.RandomState(seed)
    hb = round_balanced(h, n)
    corr = count_vector(hb).astype(float)
    cur = float(np.max(corr))
    best = hb.copy()
    bestc = cur
    L = n - 1
    stalls = 0
    kicks = 0
    while time.time() < deadline:
        rev = (1.0 - 2.0 * hb)[::-1]
        B = np.zeros((n, 2 * n - 1), dtype=float)
        for p in range(n):
            B[p, p:L + p + 1] = rev
            B[p, L] = 0.0
        zeros = np.where(hb < 0.5)[0]
        ones = np.where(hb >= 0.5)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        nt = min(int(topk), 2 * n - 1)
        top = np.argsort(corr)[-nt:]
        D = corr[top][None, None, :] + B[zeros, top][:, None, :] - B[ones, top][None, :, :]
        vals = D.max(axis=2)
        flat = int(np.argmin(vals))
        ii = flat // int(ones.size)
        jj = flat % int(ones.size)
        i = int(zeros[ii])
        j = int(ones[jj])
        newc = float(vals[ii, jj])
        accepted = False
        if newc < cur - 1e-12:
            hb[i] = 1.0
            hb[j] = 0.0
            corr = count_vector(hb).astype(float)
            cur = float(np.max(corr))
            stalls = 0
            accepted = True
        elif kicks < 40:
            k = int(np.random.RandomState(seed + int(kicks)).choice([1, 1, 1, 2, 3]))
            k = max(1, min(k, int(zeros.size), int(ones.size)))
            zi = rng.choice(zeros, k, replace=False)
            oj = rng.choice(ones, k, replace=False)
            delta = np.zeros(2 * n - 1, dtype=float)
            for a in range(k):
                delta += B[int(zi[a])] - B[int(oj[a])]
            newcorr = corr + delta
            c2 = float(np.max(newcorr))
            if c2 <= cur + 2.0 and c2 < bestc + 4.0:
                for a in range(k):
                    hb[int(zi[a])] = 1.0
                    hb[int(oj[a])] = 0.0
                corr = newcorr
                cur = c2
                kicks += 1
                accepted = True
        if accepted:
            if cur < bestc - 1e-12:
                bestc = cur
                best = hb.copy()
            continue
        stalls += 1
        if stalls > 5:
            break
    return best, float(bestc)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget_s = max(20.0, float(budget_s))
    except Exception:
        budget_s = 1100.0
    deadline = t0 + budget_s - 2.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).reshape(-1).copy()
        if h0.size == 0:
            raise ValueError
        n = int(h0.size)
    except Exception:
        n = 250
        xg = np.arange(n)
        h0 = (xg * 0.618033988749895) % 1.0

    target = 0.5 * float(n)
    h0 = project_simplex(h0, target)
    best_h = h0.copy()
    best_m = float(max_count(best_h))
    best_c = best_m * (2.0 / float(n))

    def update(hc):
        nonlocal best_h, best_m, best_c
        hh = final_adjust(hc, n)
        mm = float(max_count(hh))
        if mm < best_m - 1e-12:
            best_m = mm
            best_h = hh.copy()
            best_c = best_m * (2.0 / float(n))
        return best_h

    t1 = t0 + 0.30 * budget_s
    start = best_h.copy()
    for alpha in (20.0, 80.0, 300.0):
        if time.time() >= min(t1, deadline) - 2.0:
            break
        seg = max(10.0, 0.06 * budget_s)
        end = min(t1, deadline - 2.0, time.time() + seg)
        if end <= time.time():
            continue
        hp, mb, hcur, mcur = polish_grad(start, n, end, alpha, max_iter=2000, eta0=0.5)
        if mb < best_m - 1e-12:
            best_m = mb
            best_h = hp.copy()
            best_c = best_m * (2.0 / float(n))
        mstart = float(max_count(start))
        if mcur <= mstart + 1e-8:
            start = hcur.copy()
        else:
            start = hp.copy()

    t2 = t0 + 0.62 * budget_s
    maxiter_sl = 45 if budget_s > 600.0 else (25 if budget_s > 150.0 else 10)

    if time.time() < t2 - 3.0:
        update(slsqp_step(best_h, n, 60.0, maxiter_sl, t2, 1e-9))
    if time.time() < t2 - 3.0:
        update(slsqp_step(best_h, n, 200.0, maxiter_sl, t2, 1e-9))

    perturb_specs = ((0.005, 100.0), (0.015, 150.0), (0.03, 250.0))
    for sig, alpha in perturb_specs:
        if time.time() >= t2 - 3.0:
            break
        hpert = project_simplex(best_h + sig * rng.randn(n), target)
        update(slsqp_step(hpert, n, alpha, maxiter_sl, t2, 1e-9))

    if n % 2 == 0 and time.time() < t2 - 3.0:
        hbin = round_balanced(best_h, n)
        hbin = project_simplex(hbin + 0.015 * rng.randn(n), target)
        update(slsqp_step(hbin, n, 250.0, maxiter_sl, t2, 1e-9))

    if time.time() < t2 - 3.0:
        xw = 2.0 * np.arange(n) / float(n)
        hwave = project_simplex(0.5 + 0.5 * np.sin(2.0 * np.pi * 3.0 * xw + 0.3), target)
        update(slsqp_step(hwave, n, 100.0, max(8, maxiter_sl // 2), t2, 1e-9))

    tlog = t0 + 0.78 * budget_s
    maxlogit = 40 if budget_s > 600.0 else 18
    if time.time() < tlog - 2.0:
        update(slsqp_logit(best_h, n, 300.0, 2.0, maxlogit, tlog))
        if time.time() < tlog - 2.0:
            hpert = project_simplex(best_h + 0.02 * rng.randn(n), target)
            update(slsqp_logit(hpert, n, 600.0, 4.0, max(8, maxlogit - 6), tlog))

    t3 = t0 + 0.88 * budget_s
    if n % 2 == 0 and time.time() < t3 - 12.0:
        bdeadline = min(t3 - 8.0, time.time() + 60.0)
        hb, mb = binary_polish(best_h, n, bdeadline, topk=70, seed=int(seed) + 7)
        hfrom = project_simplex(hb + 0.02 * rng.randn(n), target)
        end = min(t3 - 4.0, deadline - 2.0, time.time() + 80.0)
        if end > time.time() + 5.0:
            hp, m_new, hcur, _ = polish_grad(hfrom, n, end, 500.0, max_iter=800, eta0=0.4)
            if m_new < best_m - 1e-12:
                best_m = m_new
                best_h = hp.copy()
                best_c = best_m * (2.0 / float(n))
        hfrom2 = project_simplex(hb + 0.005 * rng.randn(n), target)
        end = min(t3 - 1.0, deadline - 2.0, time.time() + 45.0)
        if end > time.time() + 3.0:
            hp, m_new, hcur, _ = polish_grad(hfrom2, n, end, 800.0, max_iter=600, eta0=0.3)
            if m_new < best_m - 1e-12:
                best_m = m_new
                best_h = hp.copy()
                best_c = best_m * (2.0 / float(n))

    if time.time() < deadline - 4.0:
        hp, m_new, hcur, _ = polish_grad(best_h, n, deadline - 2.0, 1000.0,
                                         max_iter=2000, eta0=0.25)
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_h = hp.copy()
            best_c = best_m * (2.0 / float(n))

    if time.time() < deadline - 6.0:
        update(slsqp_logit(best_h, n, 1200.0, 6.0, 35, deadline - 2.0))
    if time.time() < deadline - 4.0:
        update(slsqp_step(best_h, n, 800.0, 25, deadline - 2.0, 1e-10))

    best_h = final_adjust(best_h, n)
    best_c = float(max_count(best_h) * (2.0 / float(n)))
    return best_h, best_c, int(n)
