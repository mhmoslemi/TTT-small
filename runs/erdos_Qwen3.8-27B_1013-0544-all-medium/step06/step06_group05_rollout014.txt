    # First term: (1-h[p-l]) for valid p-l
    # For lag l (ranging -(n-1) to n-1), the first term at position p is (1-h[p-l]) if 0<=p-l<n
    # This is equivalent to: roll (1-h) and mask
    term1 = np.zeros((n, nfull))
    term2 = np.zeros((n, nfull))
    for l in range(-(n-1), n):
        m = l + (n-1)
        # (1-h[p-l]) valid for 0 <= p-l < n → l <= p < l+n
        # Since p in [0, n): p in [max(0,l), min(n, l+n))
        p_start = max(0, l)
        p_end = min(n, l + n)
        if p_start < p_end:
            term1[p_start:p_end, m] = 1.0 - h[p_start - l : p_end - l]
        # -h[p+l] valid for 0 <= p+l < n → -l <= p < n-l
        # Since p in [0, n): p in [max(0,-l), min(n, n-l))
        p_start2 = max(0, -l)
        p_end2 = min(n, n - l)
        if p_start2 < p_end2:
            term2[p_start2:p_end2, m] = -h[p_start2 + l : p_end2 + l]
    return term1 + term2

            if s < 1e-14:

        # Try all (p, q) pairs in order of "promisingness"
        # Score: how much the swap reduces the max
        # For speed, just try all pairs and pick the best
        best_new_max = cur_max
        best_pq = None
        
        for p in zeros:
            for q in ones:
                new_max = (corr + deltas[p] + deltas[q]).max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_pq = (p, q)
        
        if best_pq is not None and best_new_max < cur_max:
            p, q = best_pq
            h[p] = 1.0
            h[q] = 0.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_max = corr.max()
            improved = True
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c
```

The inner loop is O(|zeros| * |ones| * nfull) = O(125 * 125 * 499) ≈ 7.8M per sweep. With the max() being a vectorized operation, this should be fast in numpy. ```python
def fast_binary_search(h, n, deadline, max_sweeps=5000):
    dx = 2.0 / n
    nfull = 2*n - 1
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    
    while time.time() < deadline:
        # Precompute deltas for all positions
        deltas = np.zeros((n, nfull))
        for lag in range(-(n-1), n):
            m = lag + (n-1)
            a = abs(lag)
            if lag >= 0:
                p_start = lag
                p_end = n
                if p_start < p_end:
                    deltas[p_start:p_end, m] += 1.0 - h[p_start-lag : p_end-lag]
                p_start2 = 0
                p_end2 = n - lag
                if p_start2 < p_end2:
                    deltas[p_start2:p_end2, m] -= h[p_start2+lag : p_end2+lag]
            else:
                p_start = 0
                p_end = n - a
                if p_start < p_end:
                    deltas[p_start:p_end, m] += 1.0 - h[p_start+a : p_end+a]
                p_start2 = a
                p_end2 = n
                if p_start2 < p_end2:
                    deltas[p_start2:p_end2, m] -= h[p_start2-a : p_end2-a]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Find best swap
        best_new_max = cur_max
        best_pq = None
        for p in zeros:
            base = corr + deltas[p]
            all_new = (base[:, None] + deltas[ones]).max(axis=0)
            q_best = all_new.argmin()
            if all_new[q_best] < best_new_max:
                best_new_max = all_new[q_best]
                best_pq = (p, ones[q_best])
        
        if best_pq is not None and best_new_max < cur_max:
            p, q = best_pq
            h[p] = 1.0
            h[q] = 0.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_max = corr.max()
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c
```
