def obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    softmax = exp_s / exp_s.sum()
    obj = m + (sm + np.log(exp_s.sum())) / alpha
    
    k_range = np.arange(-(n-1), n)
    weights = (1.0 / alpha) * softmax
    weights[np.argmax(c)] += 1.0
    
    grad = np.zeros(n)
    nonzero_k = np.where(weights > 1e-12)[0]
    
    for idx in nonzero_k:
        k = int(k_range[idx])
        w = weights[idx]
        # grad[j] += dx * w * [(1-h[j+k]) if 0≤j+k<n] - dx * w * [h[j-k] if 0≤j-k<n]
        if k >= 0:
            # j+k in [0,n): j in [0, n-k)
            # contribution: (1-h[j+k]) for j in [0, n-k)
            grad[:n-k] += w * (1.0 - h[k:])
            # j-k in [0,n): j in [k, n)
            # contribution: -h[j-k] for j in [k, n)
            grad[k:] += w * (-h[:n-k])
        else:
            kk = -k
            # j+k = j-kk in [0,n): j in [kk, n)
            # contribution: (1-h[j-kk]) for j in [kk, n)
            grad[kk:] += w * (1.0 - h[:n-kk])
            # j-k = j+kk in [0,n): j in [0, n-kk)
            # contribution: -h[j+kk] for j in [0, n-kk)
            grad[:n-kk] += w * (-h[kk:])
    
    grad *= dx
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    ac = alpha * c
    sm = ac.max()
    exp_s = np.exp(ac - sm)
    obj = (sm + np.log(exp_s.sum())) / alpha
    softmax = exp_s / exp_s.sum()
    k_range = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = softmax[idx]
        if w < 1e-11:
            continue
        k = int(k_range[idx])
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] += w * (-h[kk:])
    grad *= dx
    return obj, grad

def lbfqs_opt(h, n, alpha, maxiter=5000, penalty=1e4):
    dx = 2.0 / n
    def obj(x):
        o, g = obj_grad(x, n, alpha)
        return o + penalty * (x.sum() - n / 2.0) ** 2
    def grad(x):
        o, g = obj_grad(x, n, alpha)
        g += 2.0 * penalty * (x.sum() - n / 2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: binary local search
    p1 = t0 + 0.04 * total
    hb, cb = greed_binary(h0, n, p1, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: multi-stage L-BFGS-B
    p2 = t0 + 0.45 * total
    alphas = [20, 40, 80, 160, 320, 640, 1280, 2560, 5120, 10240, 20480, 40960, 81920, 163840]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 3:
            break
        h = lbfqs_opt(h, n, alpha, maxiter=5000, penalty=1e4)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 3: perturbation restarts
    p3 = t0 + 0.82 * total
    for trial in range(20):
        if time.time() > p3 - 3:
            break
        scale = 0.003 + 0.004 * (trial % 6)
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [5000, 20000, 80000, 200000]:
            if time.time() > p3 - 2:
                break
            h = lbfqs_opt(h, n, alpha, maxiter=3000, penalty=1e4)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    # Phase 4: other resolutions
    if time.time() < dl - 35:
        for nh in [300, 200, 500]:
            if time.time() > dl - 30:
                break
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            hh = np.interp(xn, xo, best_h)
            hh = project_feasible(hh, nh)
            for alpha in [2000, 10000, 50000]:
                if time.time() > dl - 25:
                    break
                hh = lbfqs_opt(hh, nh, alpha, maxiter=3000, penalty=1e4)
                c = compute_c5(hh, nh)
                if c < best_c:
                    best_c, n, best_h = c, nh, hh.copy()
    # Phase 5: final high-alpha polish
    if time.time() < dl - 5:
        h = best_h
        for alpha in [100000, 300000, 800000]:
            if time.time() > dl - 3:
                break
            h = lbfqs_opt(h, n, alpha, maxiter=5000, penalty=1e4)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
