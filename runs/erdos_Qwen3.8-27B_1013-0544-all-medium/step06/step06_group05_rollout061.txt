def check_swap_vectorized(h, i, j, c, n, crit):
    """Check if swap (i:0→1, j:1→0) keeps max ≤ C. Returns True if feasible."""
    L = crit - (n - 1)  # shape (n_crit,)
    # delta for each critical lag
    delta = np.zeros(len(L))
    ip = i + L  # shape (n_crit,)
    ip2 = i - L
    jp = j + L
    jp2 = j - L
    
    valid_ip = (ip >= 0) & (ip < n)

    delta[valid_jp] -= 1 - h[jp[valid_jp]]
    delta[valid_jp2] += h[jp2[valid_jp2]]
    
    C = c.max()

    c = np.correlate(h, 1.0 - h, mode='full')

            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:

    return np.clip(h, 0.0, 1.0)

    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

    """Compute gradient of (1/alpha)*log(sum exp(alpha*c_k)) w.r.t. h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = c - c.max()
    w = np.exp(alpha * shifted)
    w = w / w.sum()
    # grad[i] = dx * [corr(w, 1-h)[i] - corr(w_flip, h)[i+n-1]]
    t1 = np.correlate(w, 1.0 - h, mode='full')[:n]
    t2 = np.correlate(w[::-1], h, mode='full')[n-1:2*n-1]
    grad = dx * (t1 - t2)
    return grad, c

    """Projected gradient descent with increasing alpha."""
    n_log = max_steps
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    for step in range(max_steps):
        if time.time() > deadline:
            break
        # Interpolate alpha
        frac = step / n_log
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        for _ in range(20):

                if c_new < best_c:
                    best_c = c_new

                break

        # Occasionally reset lr
        if step % 100 == 0:
            lr = 0.01
    return best_h, best_c

        i_batch = rng.choice(zeros, size=n_batch)
        j_batch = rng.choice(ones, size=n_batch)

        for idx in range(n_batch):

            L = crit - (n - 1)
            delta = np.zeros(len(L))
            ip = i + L
            ip2 = i - L
            jp = j + L
            jp2 = j - L
            v1 = (ip >= 0) & (ip < n)
            v2 = (ip2 >= 0) & (ip2 < n)
            v3 = (jp >= 0) & (jp < n)
            v4 = (jp2 >= 0) & (jp2 < n)
            delta[v1] += 1.0 - h[ip[v1]]
            delta[v2] -= h[ip2[v2]]
            delta[v3] -= 1.0 - h[jp[v3]]
            delta[v4] += h[jp2[v4]]

                continue

            h[i] = 1.0
            h[j] = 0.0
            c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            new_C = c.max()
            
            if new_C < C:
                C = new_C
                if C < best_c:
                    best_c = C
                    best_h = h.copy()
                found_in_batch = True
                stuck_count = 0
                break
            else:
                h[i] = 0.0
                h[j] = 1.0

            stuck_count += 1
            if stuck_count > 30:

                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                for _ in range(n_pert):

                        h[rng.choice(zi)] = 1.0
                        oi = np.where(h == 1)[0]
                        zi = np.where(h == 0)[0]
                c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                C = c.max()
                stuck_count = 0

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = c - c.max()
    w = np.exp(alpha * shifted)
    w = w / w.sum()
    t1 = np.correlate(w, 1.0 - h, mode='full')[:n]
    t2 = np.convolve(w, h, mode='full')[n-1:2*n-1]
    grad = dx * (t1 - t2)
    return grad, c

def grad_descent_smooth(h_init, n, deadline, alpha_start=50.0, alpha_end=50000.0, max_steps=5000):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    n_steps = max_steps
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / n_steps
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        grad, c = smooth_max_grad(h, n, alpha)
        cur_c = c.max()
        improved = False
        for _ in range(15):
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                improved = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            lr *= 0.5
        if not improved:
            lr = 0.05
        if step % 200 == 0 and step > 0:
            lr = 0.05
    return best_h, best_c

            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            crit = np.where(c >= C - tol)[0]
            if len(crit) == 0:
                crit = np.array([np.argmax(c)])
            n_batch = min(batch_size, len(zeros) * len(ones))
            i_batch = rng.choice(zeros, size=n_batch)
            j_batch = rng.choice(ones, size=n_batch)
            found = False
            L_arr = crit - (n - 1)
            c_crit = c[crit]
            for idx in range(n_batch):
                i, j = i_batch[idx], j_batch[idx]
                delta = np.zeros(len(L_arr))
                ip = i + L_arr
                ip2 = i - L_arr
                jp = j + L_arr
                jp2 = j - L_arr
                v1 = (ip >= 0) & (ip < n)
                v2 = (ip2 >= 0) & (ip2 < n)
                v3 = (jp >= 0) & (jp < n)
                v4 = (jp2 >= 0) & (jp2 < n)
                delta[v1] += 1.0 - cur[ip[v1]]
                delta[v2] -= cur[ip2[v2]]
                delta[v3] -= 1.0 - cur[jp[v3]]
                delta[v4] += cur[jp2[v4]]
                if np.any(c_crit + delta > C):
                    continue
                cur[i] = 1.0
                cur[j] = 0.0
                c = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
                new_C = c.max()
                if new_C < C:
                    C = new_C
                    found = True
                    stuck = 0
                    if C < best_c:
                        best_c = C
                        best_cval = C * dx
                        best_h = cur.copy()
                    break
                else:
                    cur[i] = 0.0
                    cur[j] = 1.0
            if not found:
                stuck += 1
                if stuck > 50:
                    np.random.shuffle(i_batch)
                    stuck = 0
                    if stuck > 200:
                        break
    return best_h, float(best_cval)

def perimeter_fill(n, phase=0):
    h = np.zeros(n)
    target = n // 2
    idx = np.array([(i + phase) % n for i in range(target)])
    h[idx] = 1.0
    return h

def interleaved_fill(n, period, count, offset=0):
    h = np.zeros(n)
    target = n // 2
    pos = np.arange(offset, n, period)
    if len(pos) >= target:
        h[pos[:target]] = 1.0
    else:
        h[pos] = 1.0
        remaining = target - h.sum()
        avail = np.where(h == 0)[0]
        if len(avail) >= remaining:
            h[avail[:int(remaining)]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary search from initial (30% budget)
    p1 = t0 + 0.30 * (budget_s - 10)
    h_bin = make_binary(best_h, n)
    hb, cb = fast_binary_search(h_bin, n, p1, max_iters=500000, seed=seed, n_restarts=3)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Continuous gradient polish (20%)
    p2 = t0 + 0.50 * (budget_s - 10)
    if time.time() < p2:
        h_cont = best_h.copy()
        h_cont = project_feasible(h_cont, n)
        hc, cc = grad_descent_smooth(h_cont, n, p2, alpha_start=50.0, alpha_end=50000.0, max_steps=4000)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()

    # Phase 3: Re-binarize and search (25%)
    p3 = t0 + 0.75 * (budget_s - 10)
    if time.time() < p3:
        h_bin2 = make_binary(best_h, n)
        hb2, cb2 = fast_binary_search(h_bin2, n, p3, max_iters=400000, seed=seed + 500, n_restarts=4)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 4: Try alternate starting structures (10%)
    p4 = t0 + 0.85 * (budget_s - 10)
    if time.time() < p4:
        for start_fn in [lambda: perimeter_fill(n, 0),
                         lambda: perimeter_fill(n, n // 4),
                         lambda: interleaved_fill(n, 4, 2, 0)]:
            if time.time() > p4:
                break
            h_alt = start_fn()
            h_alt_bin, c_alt = fast_binary_search(h_alt, n, p4, max_iters=100000, seed=seed + 999, n_restarts=1)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt_bin.copy()

    # Phase 5: Final continuous polish (10%)
    p5 = dl - 3
    if time.time() < p5:
        h_cont = best_h.copy()
        h_cont = project_feasible(h_cont, n)
        hc2, cc2 = grad_descent_smooth(h_cont, n, p5, alpha_start=500.0, alpha_end=200000.0, max_steps=3000)
        if cc2 < best_c:
            best_c = cc2
            best_h = hc2.copy()

    # Final check: compare continuous vs binary version
    best_h = project_feasible(best_h, n)
    c_cont = compute_c5(best_h, n)
    h_bin_final = make_binary(best_h, n)
    c_bin = compute_c5(h_bin_final, n)
    if c_bin < c_cont:
        best_c = c_bin
        best_h = h_bin_final
    else:
        best_c = c_cont

    return best_h, best_c, n
