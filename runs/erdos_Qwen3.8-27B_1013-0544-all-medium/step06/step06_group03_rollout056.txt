import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def gradient_smooth_max(h, n, alpha, top_m=10):
    """Compute gradient of smooth max of top_m lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Use top_m lags for the gradient
    top_idx = np.argsort(c)[-top_m:]
    # Weights proportional to softmax
    active_c = c[top_idx]
    m = active_c.max()
    shifted = alpha * (active_c - m)
    mx = shifted.max()
    if mx > 500:
        weights = np.exp(shifted - mx)
    else:
        weights = np.exp(shifted)
    weights /= weights.sum()
    # Compute gradient as weighted sum of lag gradients
    grad = np.zeros(n)
    for w, k_idx in zip(weights, top_idx):
        lag = k_idx - (n - 1)
        m_abs = abs(lag)
        if m_abs == 0:
            grad[:] += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - m_abs] += w * dx * (1.0 - h[m_abs:])
            grad[m_abs:] += w * dx * (-h[:n - m_abs])
        else:
            grad[:n - m_abs] += w * dx * (-h[m_abs:])
            grad[m_abs:] += w * dx * (1.0 - h[:n - m_abs])
    return grad, c

def projected_gradient_descent(h, n, alpha, deadline, momentum=0.9, top_m=10):
    h_curr = h.copy()
    v = np.zeros(n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.1  # initial learning rate
    for it in range(100000):
        if time.time() > deadline:
            break
        grad, c = gradient_smooth_max(h_curr, n, alpha, top_m)
        v = momentum * v - lr * grad
        h_new = project_feasible(h_curr + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # Backtracking line search
        step = h_new - h_curr
        if c_new > best_c:
            # Reduce step
            lr = max(lr * 0.7, 1e-8)
            h_new = h_curr + 0.5 * step
            h_new = project_feasible(h_new, n)
        else:
            lr = min(lr * 1.05, 1.0)
            # Occasionally increase momentum effect
        h_curr = h_new
    return best_h, best_c

                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return project_feasible(res.x, n)

def _outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m_abs = abs(lag)
        if m_abs == 0:
            grad = dx * (1.0 - 2.0 * h_curr)
        elif lag > 0:
            grad = np.zeros(n)
            grad[:n - m_abs] += dx * (1.0 - h_curr[m_abs:])
            grad[m_abs:] += dx * (-h_curr[:n - m_abs])
        else:
            grad = np.zeros(n)
            grad[:n - m_abs] += dx * (-h_curr[m_abs:])
            grad[m_abs:] += dx * (1.0 - h_curr[:n - m_abs])
        cons.append(c[k] + (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def _outer_approx(h, n, deadline, max_iters=40):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new = _outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def _balance_lags(h, n, top_m=5, n_iter=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(n_iter):
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        top_vals = c[top_idx]
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-10:
            break
        k_max = top_idx[np.argmax(top_vals)]
        k_min = top_idx[np.argmin(top_vals)]
        lag_max = k_max - (n - 1)
        lag_min = k_min - (n - 1)
        m_max, m_min = abs(lag_max), abs(lag_min)
        if m_max == 0:
            g_max = dx * (1.0 - 2.0 * h)
        elif lag_max > 0:
            g_max = np.zeros(n)
            g_max[:n - m_max] += dx * (1.0 - h[m_max:])
            g_max[m_max:] += dx * (-h[:n - m_max])
        else:
            g_max = np.zeros(n)
            g_max[:n - m_max] += dx * (-h[m_max:])
            g_max[m_max:] += dx * (1.0 - h[:n - m_max])
        if m_min == 0:
            g_min = dx * (1.0 - 2.0 * h)
        elif lag_min > 0:
            g_min = np.zeros(n)
            g_min[:n - m_min] += dx * (1.0 - h[m_min:])
            g_min[m_min:] += dx * (-h[:n - m_min])
        else:
            g_min = np.zeros(n)
            g_min[:n - m_min] += dx * (-h[m_min:])
            g_min[m_min:] += dx * (1.0 - h[:n - m_min])
        step_dir = -g_max + g_min
        step_dir -= np.mean(step_dir)
        for lr in [0.01, 0.005, 0.001, 0.0005]:
            h_new = project_feasible(h + lr * step_dir, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                break
    return h, best_c

def _fine_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for delta in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001, 5e-5]:
        if time.time() > deadline:
            break
        for _ in range(300):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            found = False
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-8:
                    amt = min(delta, (h[i] - h[j]) * 0.3, h[i] * 0.5, (1 - h[j]) * 0.5)
                    if amt < 1e-10:
                        continue
                    h[i] -= amt; h[j] += amt
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new; found = True; break
                    else:
                        h[i] += amt; h[j] -= amt
                elif h[j] > h[i] + 1e-8:
                    amt = min(delta, (h[j] - h[i]) * 0.3, h[j] * 0.5, (1 - h[i]) * 0.5)
                    if amt < 1e-10:
                        continue
                    h[i] += amt; h[j] -= amt
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new; found = True; break
                    else:
                        h[i] -= amt; h[j] += amt
            if not found:
                break
    return h, best_c

def optimize_h(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 50:
        oo_end = time.time() + min(50, (deadline - time.time()) * 0.3)
        h_oo, c_oo = _outer_approx(h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo; best_h = h_oo
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > deadline - 25:
            break
        h_new = _lbfgs_step(best_h, n, alpha, 100.0, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new; best_h = h_new
    if time.time() < deadline - 20:
        for alpha in [50000, 200000]:
            if time.time() > deadline - 15:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-5:]
            h_new = _lbfgs_step_active(best_h, n, alpha, 200.0, top_idx, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new; best_h = h_new
    if time.time() < deadline - 15:
        h_bal, c_bal = _balance_lags(best_h, n, top_m=5, n_iter=30)
        if c_bal < best_c:
            best_c = c_bal; best_h = h_bal
    if time.time() < deadline - 5:
        h_fs, c_fs = _fine_local_search(best_h, n, deadline - 1)
        if c_fs < best_c:
            best_c = c_fs; best_h = h_fs
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    n_list = []
    if h_init is not None:
        n_list.append(n_init)
    for n in [250, 350, 200, 400]:
        if n not in n_list:
            n_list.append(n)
    for n in n_list:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 60
        if remaining < 40:
            break
        if h_init is not None and n == n_init:
            h_start = h_init.copy()
        elif best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            h_start = np.interp(np.linspace(0, 2, n), x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            h_start = np.interp(np.linspace(0, 2, n), x_init, h_init)
        else:
            h_start = np.full(n, 0.5)
        h_start = project_feasible(h_start, n)
        alloc = min(130, remaining * 0.4)
        h_opt, c_opt = optimize_h(h_start, n, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt; best_h = h_opt
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        for r in range(4):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(45, (deadline - time.time() - 20) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_h(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt; best_h = h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = _lbfgs_step(best_h, n, 500000, 500.0, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new; best_h = h_new
        best_h, best_c = _fine_local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
