    dx = 2.0 / n
    M = n / 2  # max possible correlation before scaling
    # Actually the correlation values range from 0 to min(n/2, n/2) = n/2
    
    corr = np.correlate(h, 1-h, mode='full').astype(np.float64)
    cur_max = corr.max()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    T = t_start
    cooling = (t_end / t_start) ** (1.0 / n_iterations)
    
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        # Pick random pair
        i = random.choice(ones)  # position to flip from 1 to 0
        j = random.choice(zeros)  # position to flip from 0 to 1
        
        # Compute Δr(k) for all k
        # Δr(k) = -(1-h[i+k]) + (1-h[j+k]) + h[i-k] - h[j-k]
        # with boundary checks
        
        # This is O(n) per swap
        # Vectorized:
        delta = np.zeros(2*n-1)
        # For each k_idx (0 to 2n-2), actual lag k = k_idx - (n-1)
        lags = np.arange(2*n-1) - (n-1)

        valid1 = (i + lags >= 0) & (i + lags < n)
        delta[valid1] -= (1 - h[i + lags[valid1]])
        
        # Term 2: +(1-h[j+k]) for valid j+k
        valid2 = (j + lags >= 0) & (j + lags < n)
        delta[valid2] += (1 - h[j + lags[valid2]])
        
        # Term 3: +h[i-k] for valid i-k
        valid3 = (i - lags >= 0) & (i - lags < n)
        delta[valid3] += h[i - lags[valid3]]
        
        # Term 4: -h[j-k] for valid j-k
        valid4 = (j - lags >= 0) & (j - lags < n)
        delta[valid4] -= h[j - lags[valid4]]
        
        new_corr = corr + delta
        new_max = new_corr.max()
        
        # Accept or reject
        if new_max <= cur_max or random.random() < np.exp(-(new_max - cur_max) / T):
            # Accept
            h[i] = 0
            h[j] = 1
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            corr = new_corr
            cur_max = new_max
        
        T *= cooling

import numpy as np

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-11:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def slsqp_polish(h, n, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        alpha = 5000.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    hb = np.zeros(n)

    return hb
