import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    """Compute gradient of C_k w.r.t. h, vectorized."""
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] -= h[k:]
        g[k:] += 1.0 - h[0:nv]
    else:
        kk = -k
        nv = n - kk
        g[0:nv] += 1.0 - h[kk:]
        g[kk:] -= h[0:nv]
    return g

def smooth_max_grad(h, n, alpha):
    """Compute gradient of smooth max of all C_k using log-sum-exp."""
    c = compute_corr(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        g += w[m] * dx * lag_gradient(h, n, k)
    return g

def active_lag_grad(h, n, T=5):
    """Gradient targeting only the top T active lags."""
    c = compute_corr(h, n)
    top_idx = np.argsort(c)[-T:]
    dx = 2.0 / n
    g = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        g += dx * lag_gradient(h, n, k)
    return g, c

def focused_gradient_descent(h, n, deadline, top_T=5, lr_init=0.05):
    """GD targeting active lags with backtracking line search."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stagnant = 0
    
    for it in range(50000):
        if time.time() > deadline:
            break
        g, c = active_lag_grad(h, n, top_T)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Backtracking

        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                stagnant = 0
                break
            step *= 0.5
        if not improved:

            stagnant += 1

                h = project_feasible(h + 0.01 * np.random.randn(n), n)
                stagnant = 0
            if lr < 1e-14:

    return best_h, best_c

def binary_push(h, n):
    """Push h towards binary {0,1} values."""
    # Threshold at various levels

    best_c = compute_c5(h, n)

    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha):

    mx = sh.max()

    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _grad(x, n, alpha):

    mx = sh.max()
