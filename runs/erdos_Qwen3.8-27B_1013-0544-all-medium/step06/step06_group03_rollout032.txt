def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            scale = target_sum / tot
            h = np.clip(h * scale, 0, 1)
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

def lag_gradient(h, n, lag):
    """Gradient of the overlap at a specific lag."""
    m = abs(lag)
    grad = np.zeros(n)
    if m >= n:
        return grad
    if lag >= 0:
        # c = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])  [times dx]
        # dc/dh[i] for i in [0, n-1-m]: (1-h[i+m])
        # dc/dh[i] for i in [m, n-1]: -h[i-m]
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        # lag < 0, m = -lag
        # c = sum_{i=m}^{n-1} h[i]*(1-h[i-m])

        # dc/dh[i] for i in [0, n-1-m]: -h[i+m]
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad

def cvxpy_outer(h_curr, n, trust=0.1):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2*n - 1):
        lag = k - (n-1)
        g = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (g @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)

            break

    elif m < n:
        if lag > 0:
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
        else:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
    return g

def cvx_outer(h, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1, cp.abs(hv - h) <= trust]
    for k in range(2 * n - 1):
        g = lag_grad(h, n, k - (n - 1))
        cons.append(c[k] + dx * (g @ (hv - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.ECOS, cp.CBC]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return h.copy()
    v = hv.value
    return project_feasible(v, n) if v is not None else h.copy()

def cvx_loop(h, n, deadline, max_it=40):
    best_h, best_c = h.copy(), compute_c5(h, n)
    trust = 0.15
    for _ in range(max_it):
        if time.time() > deadline:
            break
        h_new = cvx_outer(h, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c, best_h = c_new, h_new.copy()
        if np.max(np.abs(h_new - h)) < 1e-9:
            break
        h, trust = h_new, max(0.008, trust * 0.85)
    return best_h, best_c

def sa_search(h, n, deadline):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c5(h, n)
    c_curr = best_c
    T, Tmin = 0.004, 1e-9
    nl = 2 * n - 1
    stale = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        if it % 2000 == 0:
            T = max(T * (0.90 if stale > 4 else 0.94), Tmin)
        stale = 0
        c = compute_all_lags(h, n)
        probs = np.maximum(c, 1e-10)
        probs /= probs.sum()
        k = np.random.choice(nl, p=probs)
        lag = k - (n - 1)
        if lag == 0:
            continue
        m = abs(lag)
        if m >= n:
            continue
        i = np.random.randint(0, n - m)
        j = i + m
        delta = np.random.randn() * 0.006
        delta = np.clip(delta, -0.03, 0.03)
        if abs(delta) < 1e-7:
            continue
        h[i] = np.clip(h[i] - delta, 0, 1)
        h[j] = np.clip(h[j] + delta, 0, 1)
        c_new = compute_c5(h, n)
        dc = c_new - c_curr
        if dc < -1e-14:
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        elif dc < 0 or np.random.random() < np.exp(-dc / max(T, 1e-15)):
            c_curr = c_new
        else:
            h[i] = np.clip(h[i] + delta, 0, 1)
            h[j] = np.clip(h[j] - delta, 0, 1)
            stale += 1
    return best_h, best_c

def local_refine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(400):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k = int(np.argmax(c))
        lag = k - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        for i in range(n - m):
            if time.time() > deadline:
                break
            j = i + m
            if h[i] > h[j] + 1e-6:
                d = min(0.004, (h[i] - h[j]) * 0.4, h[i])
                h[i] -= d
                h[j] += d
                cn = compute_c5(h, n)
                if cn < best_c - 1e-14:
                    best_c = cn
                    improved = True
                else:
                    h[i] += d
                    h[j] -= d
            elif h[j] > h[i] + 1e-6:
                d = min(0.004, (h[j] - h[i]) * 0.4, h[j])
                h[i] += d
                h[j] -= d
                cn = compute_c5(h, n)
                if cn < best_c - 1e-14:
                    best_c = cn
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
        if not improved:
            break
    return h, best_c

def optimize(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    if time.time() < deadline - 50:
        h1, c1 = cvx_loop(best_h, n, time.time() + min(70, (deadline - time.time()) * 0.3))
        if c1 < best_c:
            best_c, best_h = c1, h1
    for alpha in [8000, 40000, 150000]:
        if time.time() > deadline - 25:
            break
        c = compute_all_lags(best_h, n)
        top = np.argsort(c)[-6:]
        h2 = slsqp(best_h, n, alpha, idx=top, maxiter=1500)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 15:
        h2 = slsqp(best_h, n, 400000, maxiter=1500)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 25:
        h3, c3 = sa_search(best_h, n, deadline - 12)
        if c3 < best_c:
            best_c, best_h = c3, h3
    if time.time() < deadline - 10:
        h2 = slsqp(best_h, n, 800000, maxiter=2000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 2:
        best_h, best_c = local_refine(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 3
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0, n0 = np.full(300, 0.5), 300
    best_h, best_c = None, float('inf')
    if h0 is not None:
        alloc = min(350, (deadline - t0) * 0.40)
        best_h, best_c = optimize(h0, n0, t0 + alloc)
    for nt in [200, 400, 250]:
        if time.time() > deadline - 80:
            break
        if best_h is not None and nt == len(best_h):
            continue
        rem = deadline - time.time() - 35
        if rem < 50:
            break
        alloc = min(100, rem * 0.3)
        if best_h is not None:
            hs = np.interp(np.linspace(0, 2, nt), np.linspace(0, 2, len(best_h)), best_h)
        else:
            hs = np.interp(np.linspace(0, 2, nt), np.linspace(0, 2, n0), h0)
        hs = project_feasible(hs, nt)
        ho, co = optimize(hs, nt, time.time() + alloc)
        if co < best_c:
            best_c, best_h = co, ho
    if best_h is not None and time.time() < deadline - 55:
        n = len(best_h)
        rem = deadline - time.time() - 30
        nr = max(1, int(rem / 60))
        for r in range(nr):
            if time.time() > deadline - 35:
                break
            hp = project_feasible(best_h + np.random.randn(n) * 0.01, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.35)
            if alloc < 20:
                break
            ho, co = optimize(hp, n, time.time() + alloc)
            if co < best_c:
                best_c, best_h = co, ho
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h2 = slsqp(best_h, n, 1500000, maxiter=4000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2
        best_h, best_c = local_refine(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n0, 0.5), n0)
        best_c = compute_c5(best_h, n0)
    return best_h, best_c, len(best_h)
