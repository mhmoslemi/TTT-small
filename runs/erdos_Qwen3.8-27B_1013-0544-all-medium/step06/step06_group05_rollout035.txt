import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14: break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    ac -= ac.max()
    lse = (ac.max() + np.log(np.exp(ac).sum())) / alpha
    pen = lam * (x.sum() - n / 2.0) ** 2
    return lse + pen

def lse_gradient(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    ac -= ac.max()
    w = np.exp(ac)
    w /= w.sum()
    one_minus_x = 1.0 - x
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if lo < hi:
            T1[lo:hi] += w[k] * one_minus_x[lo - lag:hi - lag]
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            T2[lo2:hi2] += w[k] * x[lo2 - lag:hi2 - lag]
    grad_lse = (T1 - T2) * dx
    grad_pen = 2 * lam * (x.sum() - n / 2.0) * np.ones(n)
    return grad_lse + grad_pen

def continuous_opt(h_init, n, deadline):
    """Run L-BFGS-B with increasing alpha and penalty."""
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    
    alphas = [100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        for lam in [10, 100, 1000]:
            if time.time() > deadline:
                break
            x0 = best_h.copy()
            bounds = [(0.0, 1.0)] * n
            
            def obj(x):
                return lse_objective(x, n, alpha, lam)
            
            def grad(x):
                return lse_gradient(x, n, alpha, lam)
            
            try:
                res = fmin_l_bfgs_b(obj, x0, fprime=grad, 
                                     bounds=bounds, pgtol=1e-12, 
                                     maxiter=500, iprint=0)
                h_new = project_feasible(res, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    lse = (ac_max + np.log(exp_ac.sum())) / alpha
    w = exp_ac / exp_ac.sum()
    one_minus_x = 1.0 - x
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            T1[lo1:hi1] += w[k] * one_minus_x[lo1 - lag:hi1 - lag]
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            T2[lo2:hi2] += w[k] * x[lo2 - lag:hi2 - lag]
    obj = lse + lam * (x.sum() - n / 2.0) ** 2
    grad = (T1 - T2) * dx + 2 * lam * (x.sum() - n / 2.0) * np.ones(n)
    return obj, grad

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    alphas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]

        if time.time() > deadline:
            break
        for lam in [10.0, 100.0, 1000.0]:
            if time.time() > deadline:
                break
            x0 = best_h.copy()
            bounds = [(0.0, 1.0)] * n
            
            def obj(x):
                return lse_obj_grad(x, n, alpha, lam)[0]
            
            def grad(x):
                return lse_obj_grad(x, n, alpha, lam)[1]
            
            try:

                pass
