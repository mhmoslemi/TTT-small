def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()

            s = room.sum()

            if s < 1e-14:

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0

        new_corr = corr + delta
        nc = new_corr.max()
        delta_c = nc - cur_c

        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        step += 1
        if step < total_expected * hold_frac:
            T = t_start
        else:
            T *= cool_rate

        if no_improve > patience:
            nk = 5 + rng.randint(15)
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            for _ in range(nk):
                if len(zi) > 0 and len(oi) > 0:
                    ii = zi[rng.randint(len(zi))]
                    jj = oi[rng.randint(len(oi))]
                    h[ii] = 1.0
                    h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max()
            no_improve = 0
            T = t_start
            step = 0

    return best_h, best_c * dx

def fast_local_search(h, n, deadline, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max()

    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_nc = cur_c
        best_i, best_j = -1, -1
        best_new_corr = None
        n_try = min(2000, len(zeros) * len(ones))
        for _ in range(n_try):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            if i == j:
                continue
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            delta[k2] += 1.0
            nc = (corr + delta).max()
            if nc < best_nc - 1e-16:
                best_nc = nc
                best_i, best_j = i, j
                best_new_corr = corr + delta
        if best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = best_new_corr
            cur_c = best_nc
        else:
            break
    return h, cur_c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = round_to_binary(h0, n).astype(float)
    best_c = compute_c5(best_h, n)

    # Phase 1: Multiple SA runs with diverse parameters
    p1_end = t0 + 0.68 * (budget_s - 5)
    n_runs = 8
    run_dur = (p1_end - t0) / n_runs

    # Diverse SA parameter configurations
    configs = [
        (0.03, 0.00008, 25000, 0.10),
        (0.05, 0.00004, 35000, 0.05),
        (0.08, 0.00002, 45000, 0.00),
        (0.02, 0.00006, 20000, 0.15),
        (0.12, 0.00003, 55000, 0.00),
        (0.04, 0.00005, 30000, 0.08),
        (0.15, 0.00001, 60000, 0.00),
        (0.06, 0.00004, 40000, 0.05),
    ]

    for r in range(n_runs):
        if time.time() > p1_end - 5:
            break
        seg_end = min(t0 + (r + 1) * run_dur, p1_end - 3)
        ts, te, pat, hf = configs[r]
        s = seed + r * 137

        if r % 3 == 0:
            start_h = best_h.copy()
        elif r % 3 == 1:
            rng_r = np.random.RandomState(s)
            start_h = round_to_binary(project_feasible(best_h + 0.2 * rng_r.randn(n), n), n)
        else:
            rng_r = np.random.RandomState(s)
            start_h = np.zeros(n)
            idx = rng_r.choice(n, size=n // 2, replace=False)
            start_h[idx] = 1.0

        hb, cb = sa_run(start_h, n, seg_end, s, ts, te, pat, hf)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 2: Local search polish
    p2_end = t0 + 0.85 * (budget_s - 5)
    if time.time() < p2_end - 3:
        hl, cl = fast_local_search(best_h, n, p2_end, seed=seed + 999)
        if cl < best_c:
            best_c = cl
            best_h = hl.copy()

    # Phase 3: Another SA from the polished solution
    p3_end = dl - 3
    if time.time() < p3_end - 5:
        hb, cb = sa_run(best_h, n, p3_end, seed + 1500, 0.015, 0.00001, 12000, 0.0)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

        if time.time() < dl - 2:
            hl2, cl2 = fast_local_search(best_h, n, dl - 1, seed=seed + 2000)
            if cl2 < best_c:
                best_c = cl2
                best_h = hl2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
