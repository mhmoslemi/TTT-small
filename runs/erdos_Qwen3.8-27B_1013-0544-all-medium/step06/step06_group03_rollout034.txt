def compute_gradient(h, n, weights):
    """
    Compute gradient of weighted sum of lags.
    weights[k] is the weight for lag index k (0 to 2n-2).
    Returns gradient array of length n.
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            m = -lag
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
    return dx * grad

        else:

    exp_shifted = np.exp(shifted - sm)
    weights = exp_shifted / exp_shifted.sum()

    grad = np.zeros(n)

        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:

            grad[lag:] -= w * h[:n - lag]
        else:
            m2 = -lag
            grad[m2:] += w * (1.0 - h[:n - m2])
            grad[:n - m2] -= w * h[m2:]
    return val, dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_shifted = a - a.max()
    exp_a = np.exp(a_shifted)
    w = exp_a / exp_a.sum()
    val = m + (a.max() + np.log(exp_a.sum())) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            ml = -lag
            grad[ml:] += wk * (1.0 - h[:n - ml])
            grad[:n - ml] -= wk * h[ml:]
    return val, dx * grad

def slsqp_pass(h, n, alpha, maxiter=3000):
    def obj(x):
        c = compute_all_lags(x, n)
        m = c.max()
        a = alpha * (c - m)
        a_s = a - a.max()
        return m + (a_s.max() + np.log(np.sum(np.exp(a_s)))) / alpha
    def obj_grad(x):
        _, g = smooth_max_and_grad(x, n, alpha)
        return g
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_descent(h, n, deadline, alpha=2000, n_iters=300):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = np.zeros(n)
    lr = 0.05
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, grad = smooth_max_and_grad(h, n, alpha)
        momentum = 0.9 * momentum + 0.1 * grad
        h_trial = project_feasible(h - lr * momentum, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-15:

            best_h = h_trial.copy()

        else:
            lr *= 0.8
            if lr < 0.001:

        if it % 20 == 0:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_s = a - a.max()
    exp_a = np.exp(a_s)
    w = exp_a / exp_a.sum()
    val = m + (a_s.max() + np.log(exp_a.sum())) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            ml = -lag
            grad[ml:] += wk * (1.0 - h[:n - ml])
            grad[:n - ml] -= wk * h[ml:]
    return val, dx * grad

def _slsqp_obj(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    a = alpha * (c - m)
    a_s = a - a.max()
    return m + (a_s.max() + np.log(np.sum(np.exp(a_s)))) / alpha

def _slsqp_jac(x, n, alpha):
    _, g = smooth_max_and_grad(x, n, alpha)
    return g

def _sum_eq(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_pass(h, n, alpha, maxiter=2500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_jac, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    best_c = compute_c5(h, n)
    momentum = np.zeros(n)
    lr = 0.03
    alpha = float(alpha_init)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, grad = smooth_max_and_grad(h, n, alpha)
        momentum = 0.85 * momentum + 0.15 * grad
        h_trial = project_feasible(h - lr * momentum, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-15:
            best_c = c_trial
            best_h = h_trial.copy()
            h = h_trial
            lr = min(lr * 1.05, 0.3)
        else:
            lr *= 0.85
            if lr < 0.002:
                lr = 0.03
        if it % 30 == 0:
            alpha *= 1.15
    return best_h, best_c
