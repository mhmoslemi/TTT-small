def run_fw(h_start, n, deadline, top_k, alpha, perturb_scale, rng, best_h, best_c, stochastic):
    target = 0.5*n
    h = project_feasible(h_start, target)
    fp, fn, c = eval_parts(h, n)
    if c < best_c:
        best_c = c; best_h = h.copy()
    no_improve=0; pert_scale=perturb_scale
    while time.time() < deadline:
        if c < best_c:
            best_c=c; best_h=h.copy()
        if c > best_c + 1e-3:
            h = project_feasible(best_h + 0.02*rng.standard_normal(n), target)
            fp,fn,c=eval_parts(h,n); no_improve=0; continue
        w = make_weights_from_parts(fp,fn,alpha,top_k,1e-12,rng,stochastic)
        g=weighted_grad(h,n,w)
        g_t = g - g.mean()
        gnorm = float(np.linalg.norm(g_t))
        d = None
        if gnorm > 1e-10:
            z = project_feasible(-g_t, target)
            d = z - h
            dnorm = float(np.linalg.norm(d))
            if dnorm < 1e-9:
                d = None
        if d is None:
            no_improve += 1
            if no_improve > 50:
                if pert_scale > 1e-5:
                    h = project_feasible(h + pert_scale*rng.standard_normal(n), target)
                    pert_scale *= 0.5
                    fp,fn,c=eval_parts(h,n)
                    no_improve=0
                else:
                    break
            continue
        # line search
        base_c = c
        accepted=False
        # optionally step cap: fractions full etc.
        for a in (1.0, 0.5, 0.25, 0.125):
            cand = h + a*d
            cand = np.clip(cand, 0.0, 1.0)
            # restore sum
            diff = target - float(cand.sum())
            if abs(diff) > 1e-12:
                if diff > 0:
                    k = int(np.argmax(1.0 - cand))
                    cand[k] = min(1.0, cand[k] + diff)
                else:
                    k = int(np.argmin(cand))

def eval_parts(h, n):
    s = float(np.sum(h))
    nfft = 1 << (2*n - 1).bit_length()
    fr = np.fft.rfft(h, nfft)
    ac = np.fft.irfft(fr * np.conjugate(fr), nfft)[:n]
    ac = np.where(ac > 0.0, ac, 0.0)
    pref = np.cumsum(h)
    fp = np.empty(n); fn = np.empty(n)
    fp[0] = s - ac[0]; fn[0] = fp[0]
    if n > 1:
        Lpos = np.empty(n)
        Lpos[0] = s
        Lpos[1:] = pref[n-2::-1]
        Lneg = np.empty(n)
        Lneg[0] = s
        Lneg[1:] = s - pref[:-1]
        fp[1:] = Lpos[1:] - ac[1:]
        fn[1:] = Lneg[1:] - ac[1:]
    fp = np.where(fp > 0.0, fp, 0.0)
    fn = np.where(fn > 0.0, fn, 0.0)
    c = max(float(fp.max()), float(fn.max()))
    return fp, fn, c

def weighted_grad(h,n,w):
    grad=np.zeros(n)
    w0=float(w[0])
    if w0 != 0.0:
        grad += w0 * (1.0 - 2.0*h)
    if n > 1:
       wp = w[1:n]
       wn = w[n:]
       for d in range(1,n):
           A = float(wp[d-1]); B = float(wn[d-1])
           if A == 0.0 and B == 0.0:
               continue
           m = n-d
           C = A+B
           grad[:m] += A
           grad[d:] += B
           if C != 0.0:
               grad[:m] -= C * h[d:]
               grad[d:] -= C * h[:-d]
    return grad

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


def eval_parts(h, n):
    s = float(np.sum(h))
    nfft = 1 << (2 * n - 1).bit_length()
    fr = np.fft.rfft(h, nfft)
    ac = np.fft.irfft(fr * np.conj(fr), nfft)[:n]
    ac = np.maximum(ac, 0.0)

    fp = np.empty(n)
    fn = np.empty(n)
    fp[0] = s - ac[0]
    fn[0] = fp[0]

    if n > 1:
        pref = np.cumsum(h)
        lpos = pref[n - 2::-1]
        lneg = s - pref[:-1]
        fp[1:] = lpos - ac[1:]
        fn[1:] = lneg - ac[1:]

    fp = np.maximum(fp, 0.0)
    fn = np.maximum(fn, 0.0)
    return fp, fn, max(float(fp.max()), float(fn.max()))


def compute_c5_direct(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(v, target):
    v = np.asarray(v, dtype=float).copy()
    n = v.size
    if n == 0:
        return v

    lo = float(v.min()) - 2.0 - target / n
    hi = float(v.max()) + 2.0 + target / n

    for _ in range(45):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(y.sum())

    if diff > 1e-10:
        order = np.argsort(-(1.0 - y))
        added = 0.0
        for idx in order:
            if added >= diff:
                break
            add = min(1.0 - y[idx], diff - added)
            if add > 0.0:
                y[idx] += add
                added += add
    elif diff < -1e-10:
        rem = -diff
        order = np.argsort(y)
        taken = 0.0
        for idx in order:
            if taken >= rem:
                break
            take = min(y[idx], rem - taken)
            if take > 0.0:
                y[idx] -= take
                taken += take

    return y


def smooth_value_from_parts(fp, fn, alpha):
    n = len(fp)
    vals = np.empty(2 * n - 1)
    vals[0] = fp[0]
    if n > 1:
        vals[1:n] = fp[1:]
        vals[n:] = fn[1:]
    m = float(vals.max())
    if alpha <= 0.0:
        return m
    z = np.exp(alpha * (vals - m))
    return m + math.log(float(z.sum())) / alpha


def make_weights_from_parts(fp, fn, alpha, top_k, eps, rng, stochastic):
    n = len(fp)
    vals = np.empty(2 * n - 1)
    vals[0] = fp[0]
    if n > 1:
        vals[1:n] = fp[1:]
        vals[n:] = fn[1:]

    out = np.zeros(vals.shape, dtype=float)
    m = vals.size

    if top_k and top_k > 0:
        if stochastic and rng is not None:
            L = min(m, max(10, 3 * top_k, top_k + 10))
        else:
            L = min(m, int(top_k))
        if L >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(vals, m - L)[-L:]
            if stochastic and rng is not None and L > top_k:
                idx = idx[rng.choice(L, top_k, replace=False)]
    else:
        idx = np.arange(m)

    if alpha > 0.0:
        zv = vals[idx]
        ez = np.exp(alpha * (zv - float(zv.max())))
        tot = float(ez.sum())
        if tot > 0.0:
            out[idx] = ez / tot
    else:
        out[idx] = 1.0 / float(len(idx))

    if eps > 0.0:
        out[out < eps] = 0.0
        tot = float(out.sum())
        if tot > 0.0:
            out = out / tot
    return out


def weighted_grad(h, n, w):
    grad = np.zeros(n, dtype=float)
    w0 = float(w[0])
    if w0 != 0.0:
        grad += w0 * (1.0 - 2.0 * h)

    if n > 1:
        wp = w[1:n]
        wn = w[n:]
        for d in range(1, n):
            a = float(wp[d - 1])
            b = float(wn[d - 1])
            if a == 0.0 and b == 0.0:
                continue
            m = n - d
            c = a + b
            grad[:m] += a
            grad[d:] += b
            if c != 0.0:
                grad[:m] -= c * h[d:]
                grad[d:] -= c * h[:-d]
    return grad


def run_fw(h_start, n, deadline, top_k, alpha, perturb_scale, rng,
           best_h, best_c, stochastic):
    target = 0.5 * n
    h = project_feasible(h_start, target)
    fp, fn, c = eval_parts(h, n)

    if c < best_c:
        best_c = c
        best_h = h.copy()

    no_improve = 0
    pert_scale = float(perturb_scale)

    while time.time() < deadline:
        if c < best_c:
            best_c = c
            best_h = h.copy()

        if c > best_c + 1e-3:
            h = project_feasible(best_h + 0.02 * rng.standard_normal(n), target)
            fp, fn, c = eval_parts(h, n)
            no_improve = 0
            continue

        w = make_weights_from_parts(fp, fn, alpha, top_k, 1e-12, rng, stochastic)
        g = weighted_grad(h, n, w)
        gt = g - float(np.mean(g))
        gnorm = float(np.linalg.norm(gt))

        d = None
        if gnorm > 1e-10:
            z = project_feasible(-gt, target)
            d = z - h
            if float(np.linalg.norm(d)) < 1e-9:
                d = None

        accepted = False
        if d is not None:
            base_c = c
            climb = 0.0
            if no_improve > 20:
                climb = 1e-8
            if no_improve > 45:
                climb = 1e-7
            if no_improve > 80:
                climb = 3e-6

            for a in (1.0, 0.5, 0.25, 0.125):
                cand = h + a * d
                cand = np.clip(cand, 0.0, 1.0)
                diff = target - float(cand.sum())
                if abs(diff) > 1e-8:
                    cand = project_feasible(cand, target)
                elif diff > 1e-12:
                    k = int(np.argmax(1.0 - cand))
                    cand[k] = min(1.0, cand[k] + diff)
                elif diff < -1e-12:
                    k = int(np.argmin(cand))
                    cand[k] = max(0.0, cand[k] + diff)

                cpf, cfn, cc = eval_parts(cand, n)
                if cc < base_c - 1e-13 or cc < base_c + climb:
                    h = cand
                    fp, fn, c = cpf, cfn, cc
                    accepted = True
                    if cc < best_c:
                        best_c = cc
                        best_h = h.copy()
                    no_improve = 0
                    break

        if not accepted:
            no_improve += 1

        if no_improve > 70:
            if pert_scale > 1e-5:
                h = project_feasible(h + pert_scale * rng.standard_normal(n), target)
                pert_scale *= 0.55
                fp, fn, c = eval_parts(h, n)
                no_improve = 0
                if c > best_c + 2e-3:
                    h = project_feasible(best_h + 0.02 * rng.standard_normal(n), target)
                    fp, fn, c = eval_parts(h, n)
            else:
                break

    return best_h, best_c


def random_binary_start(n, rng):
    h = np.zeros(n, dtype=float)
    k = int(round(0.5 * n))
    if k > 0 and k < n:
        h[rng.choice(n, k, replace=False)] = 1.0
    return project_feasible(h, 0.5 * n)


def thue_morse_start(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = int(round(0.5 * n))
    cur = int(t.sum())
    if cur > target:
        idx = np.where(t == 1)[0]
        rng.shuffle(idx)
        if cur - target > 0:
            t[idx[:cur - target]] = 0
    elif cur < target:
        idx = np.where(t == 0)[0]
        rng.shuffle(idx)
        if target - cur > 0:
            t[idx[:target - cur]] = 1
    return project_feasible(t.astype(float), 0.5 * n)


def parent_binary_start(h, n, rng):
    h2 = np.zeros(n, dtype=float)
    k = int(round(0.5 * n))
    if k > 0:
        idx = np.argsort(h)[::-1][:k]
        h2[idx] = 1.0
    return project_feasible(h2, 0.5 * n)


def sine_start(n):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * x / max(1, n))
    return project_feasible(h, 0.5 * n)


def lbfgs_fun(x, n, alpha, pen):
    target = 0.5 * n
    x = np.clip(x, 0.0, 1.0)
    fp, fn, _ = eval_parts(x, n)
    val = smooth_value_from_parts(fp, fn, alpha)
    ds = float(np.sum(x)) - target
    return val + pen * ds * ds


def lbfgs_jac(x, n, alpha, pen):
    target = 0.5 * n
    x = np.clip(x, 0.0, 1.0)
    fp, fn, _ = eval_parts(x, n)
    w = make_weights_from_parts(fp, fn, alpha, 0, 0.0, None, False)
    g = weighted_grad(x, n, w)
    ds = float(np.sum(x)) - target
    g = g + 2.0 * pen * ds
    return g


def lbfgs_polish(h, n, alpha, pen, maxiter):
    if minimize is None:
        return h
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            lbfgs_fun, h,
            jac=lbfgs_jac,
            args=(n, alpha, pen),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "maxfun": int(max(60, 8 * maxiter)),
                "ftol": 1e-14,
                "gtol": 1e-12,
            },
        )
        x = res.x
    except Exception:
        return h
    return project_feasible(x, 0.5 * n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_end = t0 + max(20.0, float(budget_s) - 12.0)
    work_end = hard_end - 1.5
    total = max(1.0, work_end - t0)

    try:
        n = int(len(initial_h_values))
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 250
        h0 = 0.5 * np.ones(n, dtype=float)

    target = 0.5 * n
    h0 = project_feasible(h0, target)

    true_best_h = h0.copy()
    true_best_c = compute_c5_direct(h0, n)
    cursor_h = h0.copy()
    cursor_c = true_best_c

    rng = np.random.RandomState(seed)

    main_end = t0 + 0.56 * total
    main_modes = [
        (1, 0), (3, 0), (5, 0), (10, 0),
        (20, 0), (50, 0), (10, 100), (20, 300),
    ]
    for i, (tk, al) in enumerate(main_modes):
        now = time.time()
        if now >= main_end:
            break
        rem = main_end - now
        if i < 4:
            alloc = min(rem, max(20.0, 0.09 * total + 5.0))
        else:
            alloc = min(rem, max(14.0, 0.045 * total + 4.0))
        noise = 0.0 if i == 0 else 0.003 + 0.002 * (i % 4)
        if noise > 0.0:
            start = project_feasible(cursor_h + noise * rng.standard_normal(n), target)
        else:
            start = cursor_h.copy()
        child_h, child_c = run_fw(
            start, n, time.time() + alloc, tk, al, 0.025, rng,
            cursor_h, cursor_c, stochastic=(i >= 4),
        )
        cdir = compute_c5_direct(child_h, n)
        if cdir < true_best_c:
            true_best_c = cdir
            true_best_h = child_h.copy()
        if child_c < cursor_c:
            cursor_h = child_h.copy()
            cursor_c = child_c

    div_end = t0 + 0.80 * total
    div_modes = [
        (0, 0), (0, 300), (0, 800), (0, 1500), (10, 500), (30, 1000),
    ]
    for i, (tk, al) in enumerate(div_modes):
        now = time.time()
        if now >= div_end:
            break
        rem = div_end - now
        alloc = min(rem, max(13.0, 0.034 * total + 4.0))
        scale = min(0.12, 0.008 + 0.011 * i)
        start = project_feasible(true_best_h + scale * rng.standard_normal(n), target)
        child_h, child_c = run_fw(
            start, n, time.time() + alloc, tk, al, 0.35 * scale, rng,
            true_best_h, true_best_c, stochastic=True,
        )
        cdir = compute_c5_direct(child_h, n)
        if cdir < true_best_c:
            true_best_c = cdir
            true_best_h = child_h.copy()
        if child_c < cursor_c:
            cursor_h = child_h.copy()
            cursor_c = child_c

    rand_end = t0 + 0.90 * total
    starts = [
        parent_binary_start(true_best_h, n, rng),
        thue_morse_start(n, rng),
        random_binary_start(n, rng),
        sine_start(n),
    ]
    for st in starts:
        now = time.time()
        if now >= rand_end:
            break
        rem = rand_end - now
        alloc = min(rem, max(10.0, 0.016 * total + 3.0))
        child_h, child_c = run_fw(
            st, n, time.time() + alloc, 10, 400, 0.05, rng,
            true_best_h, true_best_c, stochastic=True,
        )
        cdir = compute_c5_direct(child_h, n)
        if cdir < true_best_c:
            true_best_c = cdir
            true_best_h = child_h.copy()
        if child_c < cursor_c:
            cursor_h = child_h.copy()
            cursor_c = child_c

    if minimize is not None and time.time() < hard_end - 45.0:
        for al in (1000.0, 2200.0):
            now = time.time()
            if now > hard_end - 30.0:
                break
            maxiter = int(min(80, max(20, (hard_end - now) // 8)))
            hp = lbfgs_polish(project_feasible(true_best_h, target), n, al, 1e5, maxiter)
            cp = compute_c5_direct(hp, n)
            if cp < true_best_c:
                true_best_c = cp
                true_best_h = hp.copy()

    if time.time() < hard_end - 12.0:
        start = project_feasible(true_best_h + 0.004 * rng.standard_normal(n), target)
        child_h, child_c = run_fw(
            start, n, hard_end - 3.0, 5, 600, 0.012, rng,
            true_best_h, true_best_c, stochastic=False,
        )
        cdir = compute_c5_direct(child_h, n)
        if cdir < true_best_c:
            true_best_c = cdir
            true_best_h = child_h.copy()

    true_best_h = project_feasible(true_best_h, target)
    final_c = compute_c5_direct(true_best_h, n)
    if final_c < true_best_c:
        true_best_c = final_c
    return true_best_h, true_best_c, n
