def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 300:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    grad = np.zeros(n)
    idx = np.where(w > 1e-15)[0]
    for i in idx:
        k = i - (n - 1)
        wm = w[i] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def run_lbfgs(h, n, deadline, alpha, mu, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj(x):
        return smooth_max_val(x, n, alpha) + mu * (x.sum() - target) ** 2

    def grad(x):
        return smooth_max_grad(x, n, alpha) + 2.0 * mu * (x.sum() - target)

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def nesterov_pd(h, n, deadline, alpha=200000, lr0=0.05):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    y = h.copy()
    beta = 0.5
    lr = lr0
    mu = 10.0
    for t in range(1, 20000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(y, n, alpha) + 2.0 * mu * (y.sum() - target)
        x_new = project_feasible(y - lr * g, n)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = x_new + (beta / beta_new - 1.0) * (x_new - h)
        y_new = project_feasible(y_new, n)
        beta = beta_new
        h = x_new
        lr = lr0 / np.sqrt(t + 1)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    return best_h, best_c

def targeted_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in step_sizes:
        if time.time() > deadline:
            break
        improved = False
        g = smooth_max_grad(h, n, 100000)
        order = np.argsort(-g)
        max_i = min(60, n)
        max_j = min(60, n)
        for it in range(max_i):
            if time.time() > deadline:
                return h, best_c
            i = order[it]
            best_j = -1
            best_delta = 1e-15
            for jj in range(n - max_j, n):
                j = order[jj]
                if i == j:
                    continue
                if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    h[i] += step
                    h[j] -= step
                    delta = best_c - c
                    if delta > best_delta:
                        best_delta = delta
                        best_j = j
            if best_j >= 0:
                h[i] -= step
                h[best_j] += step
                h = project_feasible(h, n)
                best_c = compute_c5(h, n)
                improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS with increasing alpha
    alphas = [2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    phase1_end = min(t0 + 320, deadline - 350)
    for a in alphas:
        if time.time() > phase1_end:
            break
        alloc = min(45, phase1_end - time.time() - 3)
        if alloc < 12:
            break
        mu = max(1.0, a * 0.02)
        h_new, c_new = run_lbfgs(best_h, n, time.time() + alloc, alpha=a, mu=mu, maxiter=600)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: Nesterov projected gradient
    if time.time() < deadline - 200:
        t2 = min(time.time() + 100, deadline - 200)
        h_new, c_new = nesterov_pd(best_h, n, t2, alpha=500000, lr0=0.03)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 3: Very high alpha L-BFGS
    if time.time() < deadline - 120:
        for a in [5000000, 20000000, 100000000]:
            if time.time() > deadline - 80:
                break
            alloc = min(50, deadline - time.time() - 50)
            if alloc < 15:
                break
            mu = max(10.0, a * 0.005)
            h_new, c_new = run_lbfgs(best_h, n, time.time() + alloc, alpha=a, mu=mu, maxiter=2000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 4: Targeted local search
    if time.time() < deadline - 50:
        t4 = min(time.time() + 80, deadline - 50)
        h_new, c_new = targeted_local_search(best_h, n, t4)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Try nearby n values
    if time.time() < deadline - 60:
        for n_try in [n - 10, n + 10, n + 20]:
            if n_try < 100 or n_try > 800:
                continue
            if time.time() > deadline - 25:
                break
            x_old = np.linspace(0, 2, n)
            x_new_pts = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new_pts, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(35, deadline - time.time() - 12)
            if alloc < 12:
                break
            h_t, c_t = run_lbfgs(h_interp, n_try, time.time() + alloc,
                                 alpha=5000000, mu=10000.0, maxiter=800)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try

    # Phase 6: Final polish with maximum alpha
    if time.time() < deadline - 3:
        nf = len(best_h)
        h_new, c_new = run_lbfgs(best_h, nf, deadline - 2,
                                  alpha=200000000, mu=100000.0, maxiter=5000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
