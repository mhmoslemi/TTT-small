    """Find optimal h[i] given all other coordinates fixed."""
    dx = 2.0 / n
    # For each lag k, compute a_k and b_k such that c_k = a_k * h[i] + b_k
    # c_k = sum_m h[m]*(1-h[m+k])*dx
    # Terms involving h[i]:
    #   m=i: h[i]*(1-h[i+k])*dx (if i+k < n)
    #   m=i-k: h[i-k]*(1-h[i])*dx (if i-k >= 0)
    
    a = np.zeros(2*n - 1)  # coefficient of h[i] for each lag
    b = np.zeros(2*n - 1)  # constant for each lag
    
    # c_k for lag index k in [-n+1, n-1], stored at position k+n-1
    for k in range(-(n-1), n):
        pos = k + n - 1
        # c_k = sum over valid m of h[m]*(1-h[m+k])*dx

        m_lo = max(0, -k)
        m_hi = min(n, n-k)
        # (exclusive upper bound for m)

        total = 0.0
        coeff = 0.0  # coefficient of h[i]

def solve_coord(h, i, n):
    """Find optimal value for h[i] given all others fixed."""
    dx = 2.0 / n
    c_curr = np.correlate(h, 1-h, mode='full') * dx  # (2n-1,)
    
    a = np.zeros(2*n - 1)
    for k in range(-(n-1), n):
        pos = k + n - 1
        coeff = 0.0
        # Term where m = i: h[i]*(1-h[i+k])*dx, valid if 0 <= i+k < n
        if 0 <= i+k < n:

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

        else:
            tot = h.sum()
            if tot < 1e-14:

            h = np.clip(h * (target / tot), 0, 1)
    return h

    """Generate Thue-Morse sequence of length n (values 0 or 1)."""
    t = np.zeros(n, dtype=np.uint8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

    t = thue_morse(n)
    h = t.copy()

    # For other n, we may need to adjust

    return h

        if amin > 1e-12:
            hi = mid  # all slopes positive, f increasing, go left
        elif amax < -1e-12:
            lo = mid  # all slopes negative, f decreasing, go right
        else:
            break  # 0 is in subgradient

    dx = 2.0 / n
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)

            if time.time() > deadline:
                return h, best_c

            c_curr = np.correlate(h, 1 - h, mode='full') * dx

            # For lag k (stored at position k + n - 1):
            # a_k = dx*(1-h[i+k])*1{0<=i+k<n} - dx*h[i-k]*1{0<=i-k<n}
            a = np.zeros(2 * n - 1)

                pos = k + n - 1
                a[pos] += dx * (1.0 - h[i + k])
            # m=i-k term: valid for k in [i-(n-1), i]
            k_start2 = max(-(n-1), i - (n-1))
            k_end2 = min(n-1, i)
            for k in range(k_start2, k_end2 + 1):
                pos = k + n - 1
                a[pos] -= dx * h[i - k]
            
            b = c_curr - a * h[i]
            
            t_star = solve_coord_1d(a, b, h[i])
            
            if abs(t_star - h[i]) > 1e-9:
                h[i] = t_star
                improved_any = True
        
        # Project to satisfy sum constraint
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
        elif not improved_any:
            break
    
    return h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (target / tot), 0, 1)
    return h

def thue_morse_init(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return project_feasible(t, n)

def solve_1d_convex(a, b, n_lags):
    """Solve min_{t in [0,1]} max_k(a[k]*t + b[k])."""
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        fmid = vals.max()
        active = a[vals >= fmid - 1e-13]
        if active.size == 0:
            break
        if active.min() > 1e-12:
            hi = mid
        elif active.max() < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, deadline, max_rounds=200):
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    m = 2 * n - 1
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        any_change = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            # Compute current correlations
            c_curr = np.correlate(h, 1 - h, mode='full') * dx
            # Compute a (coefficient of h[i] in each c_k)
            a = np.zeros(m)
            # m=i term: a[n-1-i : 2n-1-i] += dx*(1-h)
            a[n-1-i : 2*n-1-i] += dx * (1.0 - h)
            # m=i-k term: a[i : i+n] -= dx*h[::-1]
            a[i : i+n] -= dx * h[::-1]
            
            b = c_curr - a * h[i]
            t_star = solve_1d_convex(a, b, m)
            
            if abs(t_star - h[i]) > 1e-10:
                h[i] = t_star
                any_change = True
        
        if not any_change:
            break
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    
    return h, best_c

def sa_pair_search(h, n, deadline, t_start=0.05, t_end=0.001, radius=None):
    """Simulated annealing on pair moves."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(42)
    
    n_iter = 0
    t = t_start
    cooling = (t_end / t_start) ** (1.0 / 5000.0)
    
    for _ in range(100000):
        if time.time() > deadline:
            break
        n_iter += 1
        t *= cooling
        if t < t_end:
            t = t_end
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.005, 0.05) * rng.choice([1, -1])
        
        hi_h, hj_h = h[i], h[j]
        if hi_h + d < 0 or hi_h + d > 1 or hj_h - d < 0 or hj_h - d > 1:
            continue
        
        h[i] += d
        h[j] -= d
        c_new = compute_c5(h, n)
        
        if c_new <= best_c:
            best_h = h.copy()
            best_c = c_new
        elif rng.random() > np.exp(-(c_new - best_c) / max(t, 1e-10)):
            h[i] = hi_h
            h[j] = hj_h
    
    return best_h, best_c

def smooth_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_opt(h, n, deadline, alpha=5000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        return smooth_obj(x, n, alpha) + 10.0 * (x.sum() - target) ** 2
    
    def grad(x):
        return smooth_grad(x, n, alpha) + 20.0 * (x.sum() - target) * np.ones(n)
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-15, 'maxfun': 5000})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def fine_pair_search(h, n, deadline, deltas=None, radius=50):

- For each m, reduce h[m]·(1-h[m+k])
- This means: either decrease h[m] (if h[m+k] < 1) or increase h[m+k] (if h[m] > 0)

1. Identify the top few lags at or near the maximum

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (target / tot), 0, 1)
    return h

def thue_morse_init(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return project_feasible(t, n)

def solve_1d_convex(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        fmid = vals.max()
        active = a[vals >= fmid - 1e-13]
        if active.size == 0:
            break
        if active.min() > 1e-12:
            hi = mid
        elif active.max() < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, deadline, max_rounds=200):
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    m = 2 * n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            return h, best_c
        any_change = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            c_curr = np.correlate(h, 1 - h, mode='full') * dx
            a = np.zeros(m)
            a[n-1-i : 2*n-1-i] += dx * (1.0 - h)
            a[i : i+n] -= dx * h[::-1]
            b = c_curr - a * h[i]
            t_star = solve_1d_convex(a, b)
            if abs(t_star - h[i]) > 1e-10:
                h[i] = t_star
                any_change = True
        if not any_change:
            break
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, best_c

def sa_pair_search(h, n, deadline, t_start=0.05, t_end=0.0005, seed=42):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    t = t_start
    n_steps = 20000
    cooling = (t_end / t_start) ** (1.0 / n_steps)
    for _ in range(n_steps * 5):
        if time.time() > deadline:
            break
        t = max(t * cooling, t_end)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.002, 0.04) * (1 if rng.random() < 0.5 else -1)
        hi, hj = h[i], h[j]
        if hi + d < 0 or hi + d > 1 or hj - d < 0 or hj - d > 1:
            continue
        h[i] += d; h[j] -= d
        c_new = compute_c5(h, n)
        if c_new <= best_c:
            best_h = h.copy()
            best_c = c_new
        elif rng.random() > np.exp(-(c_new - best_c) / max(t, 1e-12)):
            h[i] = hi; h[j] = hj
    return best_h, best_c

def smooth_max_obj(h, n, alpha, mu):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(len(c)):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_opt(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [2000, 10000, 50000]
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        mu = 5.0
        def obj(x):
            return smooth_max_obj(x, n, alpha, mu)
        def grad(x):
            return smooth_max_grad(x, n, alpha, mu)
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas=None, radius=40):
