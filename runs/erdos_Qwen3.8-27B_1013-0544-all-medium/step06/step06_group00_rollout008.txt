import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_objective(n, alpha):
    dx = 2.0 / n
    def obj_and_grad(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        shifted_c = alpha * (c - m)
        sm = shifted_c.max()
        exp_s = np.exp(shifted_c - sm)
        logsum = sm + np.log(exp_s.sum())
        val = m + logsum / alpha
        # Gradient
        p = exp_s / exp_s.sum()  # probabilities over shifts
        grad = np.zeros(n)
        nc = 2 * n - 1
        for k in range(nc):
            if p[k] < 1e-16:
                continue
            ws = p[k] * dx
            m_shift = k - (n - 1)
            # (1-h[j+m]) for valid j
            js = max(0, -m_shift)
            je = min(n, n - m_shift)
            if js < je:
                grad[js:je] += ws * (1.0 - h[js + m_shift:je + m_shift])
            # -h[j-m] for valid j
            js2 = max(0, m_shift)
            je2 = min(n, n + m_shift)
            if js2 < je2:
                grad[js2:je2] -= ws * h[js2 - m_shift:je2 - m_shift]
        return val, grad
    return obj_and_grad

def l_bfgs_optimize(h0, n, alpha, maxiter=2000, deadline=None):
    n_c = 2 * n - 1
    obj_g = make_objective(n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}
    options = {'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15}

                      method='L-BFGS-B', bounds=bounds, options=options)

        return h0
    h_new = project_feasible(res.x, n)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def eval_and_grad(h, n, alpha):
    """Compute objective and gradient of log-sum-exp approximation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    p = exp_s / exp_s.sum()
    val = m + (sm + np.log(exp_s.sum())) / alpha
    
    # Gradient
    nc = 2 * n - 1
    grad = np.zeros(n)
    for k in range(nc):
        if p[k] < 1e-14:
            continue
        w = p[k] * dx
        ms = k - (n - 1)
        # (1-h[j+ms]) for valid j: j+ms in [0,n-1] => j in [max(0,-ms), min(n-1, n-1-ms)]
        jlo = max(0, -ms)
        jhi = min(n, n - ms)
        if jlo < jhi:
            grad[jlo:jhi] += w * (1.0 - h[jlo + ms:jhi + ms])
        # -h[j-ms] for valid j: j-ms in [0,n-1] => j in [max(0,ms), min(n-1, n-1+ms)]
        jlo2 = max(0, ms)
        jhi2 = min(n, n + ms)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= w * h[jlo2 - ms:jhi2 - ms]
    
    return val, grad

def lbfgs_step(h, n, alpha, maxiter, deadline, penalty=0.0, mu=0.0):
    """One L-BFGS-B optimization step with optional penalty for sum constraint."""
    target = n / 2.0
    
    def obj(x):
        v, _ = eval_and_grad(x, n, alpha)
        if penalty > 0:
            s = x.sum() - target
            v += mu * s + 0.5 * penalty * s * s
        return v
    
    def grad(x):
        _, g = eval_and_grad(x, n, alpha)
        if penalty > 0:
            s = x.sum() - target
            g = g + 2.0 * (mu + penalty * s)
        return g
    
    bounds = [(0.0, 1.0)] * n
    opts = {'maxiter': maxiter, 'gtol': 1e-12}
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds, options=opts)
        return res.x
    except Exception:
        return h

def subgradient_polish(h, n, n_steps, step_scale, deadline):
    """Direct subgradient step on the max."""
    h = h.copy()
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = c.argmax()
        ms = k_star - (n - 1)
        # Gradient of c[k_star]
        dx = 2.0 / n
        g = np.zeros(n)
        jlo = max(0, -ms)
        jhi = min(n, n - ms)
        if jlo < jhi:
            g[jlo:jhi] += (1.0 - h[jlo + ms:jhi + ms])
        jlo2 = max(0, ms)
        jhi2 = min(n, n + ms)
        if jlo2 < jhi2:
            g[jlo2:jhi2] -= h[jlo2 - ms:jhi2 - ms]
        g *= dx
        # Step
        h = h - step_scale * g
        h = project_feasible(h, n)
    return h
