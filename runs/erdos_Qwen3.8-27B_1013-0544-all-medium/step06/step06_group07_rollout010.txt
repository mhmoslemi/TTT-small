def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)

def swap_delta(h, n, i, j, mr):
    d = np.zeros(2 * n - 1)
    im = i + mr
    m = (im >= 0) & (im < n)
    d[m] += 1.0 - h[im[m]]
    imn = i - mr
    m = (imn >= 0) & (imn < n)
    d[m] -= h[imn[m]]
    jm = j + mr
    m = (jm >= 0) & (jm < n)
    d[m] -= 1.0 - h[jm[m]]
    jmn = j - mr
    m = (jmn >= 0) & (jmn < n)
    d[m] += h[jmn[m]]
    d[n - 1] -= 2.0
    d[n - 1 + (j - i)] += 1.0
    d[n - 1 + (i - j)] += 1.0
    return d
