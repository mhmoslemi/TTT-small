def exact_line_search_pair(h, n, p, q, C):
    """Find optimal t for pair move (p: +t, q: -t) using exact evaluation."""
    # d_k for each lag
    dx = 2.0 / n
    m = 2 * n - 1  # number of correlation values
    d = np.zeros(m)
    for k_idx in range(m):
        k = k_idx - (n - 1)
        dk = 0.0
        pk = p + k
        if 0 <= pk < n:
            dk += (1.0 - h[pk])
        qk = q + k
        if 0 <= qk < n:
            dk -= (1.0 - h[qk])
        pm = p - k
        if 0 <= pm < n:
            dk -= h[pm]
        qm = q - k
        if 0 <= qm < n:
            dk += h[qm]
        d[k_idx] = dk
    
    # C_new(t) = C + t*d, want min_t max_k (C[k] + t*d[k])
    # subject to 0 <= h[p]+t <= 1 and 0 <= h[q]-t <= 1
    t_lo = max(0, -h[p])  # h[p] + t >= 0
    t_hi = min(1 - h[p], h[q])  # h[p]+t <= 1 and h[q]-t >= 0
    
    if t_lo >= t_hi:
        return 0.0
    
    # Find breakpoints where C[i] + t*d[i] = C[j] + t*d[j]
    # t = (C[j] - C[i]) / (d[i] - d[j])
    # Only need to check breakpoints in [t_lo, t_hi]
    
    # For efficiency, only consider top lags (near current max)
    C_sorted = np.argsort(C)[::-1]
    top_lags = C_sorted[:20]  # top 20 lags

    best_val = np.max(C)
    for t in candidates:
        val = np.max(C + t * d)
        if val < best_val:
            best_val = val
            best_t = t

def exact_line_search(C, d, t_lo, t_hi, top_M=20):
    """Find t in [t_lo, t_hi] minimizing max(C + t*d)."""
    if t_lo >= t_hi:
        return 0.0
    order = np.argsort(C)[::-1][:top_M]
    C_top = C[order]
    d_top = d[order]
    M = len(order)
    cands = [t_lo, t_hi]
    for i in range(M):
        for j in range(i + 1, M):
            dd = d_top[i] - d_top[j]
            if abs(dd) > 1e-14:
                tb = (C_top[j] - C_top[i]) / dd
                if t_lo < tb < t_hi:
                    cands.append(tb)
    best_t = 0.0
    best_v = np.max(C)
    for t in cands:
        v = np.max(C + t * d)
        if v < best_v:
            best_v = v
            best_t = t
    return best_t

    best_h = h.copy()
