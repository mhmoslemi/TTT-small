def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return val

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

from scipy.optimize import minimize, line_search
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_vals(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return val, sh, mx, c

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

def nesterov_optimize(h, n, deadline, alpha_start=100, alpha_end=5000, max_iter=5000):
    """Nesterov accelerated gradient with adaptive alpha."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    x = h.copy()  # current point
    y = h.copy()  # Nesterov extrapolated point
    
    beta = 0.9
    lr = 0.01
    no_improve = 0
    total_iter = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        total_iter += 1
        
        # Adaptive alpha
        progress = min(1.0, total_iter / max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute gradient at y
        g = smooth_max_grad(y, n, alpha)
        
        # Nesterov update
        beta_new = (1 + np.cbrt(1 + 3 * beta ** 2)) / 2
        x_new = y - lr * g
        x_new = project_feasible(x_new, n)
        
        c_new = compute_c5(x_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update y for next iteration
        y = x_new + (beta_new / beta - 1) * (x_new - x)
        y = np.clip(y, 0, 1)
        beta = beta_new
        
        # Adaptive learning rate
        if no_improve >= 20:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break
        elif it > 50 and it % 50 == 0:
            lr *= 1.05  # slowly increase if making progress
    
    return best_h, best_c

def lbgfgs_optimize(h, n, deadline, alpha_schedule=None, max_iters=200):

    bounds = [(0.0, 1.0)] * n

            break

def targeted_pair_search(h, n, deadline, step_sizes=None):

                    return h, best_c

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

    c = compute_c5_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()

    dx = 2.0 / n
