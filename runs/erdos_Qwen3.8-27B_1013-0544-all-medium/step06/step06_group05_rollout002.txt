```
def project_feasible(a, n):
    a = np.asarray(a, dtype=np.float64)
```

`gradient_from_p` optimize: Use range.
```
def gradient_from_p(h, p):
    n = len(h)
    c = n - 1
    G = np.zeros(n, dtype=np.float64)
    G += p[c] * (1.0 - 2.0 * h)
    for d in range(1, n):
        pp = p[c + d]
        pn = p[c - d]
        if pp == 0.0 and pn == 0.0:
            continue
        v = n - d
        if pp != 0.0:
            G[:v] -= pp * h[d:]
            G[d:] += pp * (1.0 - h[:v])
        if pn != 0.0:
            G[:v] += pn * (1.0 - h[d:])
            G[d:] -= pn * h[:v]
    return G
```

```
def lbfgs_polish(h, n, alpha, rho, maxiter):
    global _OPT_ALPHA, _OPT_RHO, _OPT_TARGET
    h0 = project_feasible(h.copy(), n)
    _OPT_ALPHA = float(alpha)
    _OPT_RHO = float(rho)
    _OPT_TARGET = 0.5 * n
    try:

                       bounds=[(0.0, 1.0)] * n,

        out = project_feasible(res.x, n)
    except Exception:
        out = h0
    return out
```

```
def pair_soft_value(h, alpha, i, j, t):
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_OPT_ALPHA = 1.0
_OPT_RHO = 0.0
_OPT_TARGET = 0.0

def project_feasible(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1.0, neginf=0.0)
    a = np.clip(a, 0.0, 1.0)
    target = 0.5 * n
    diff = float(a.sum() - target)
    b = a - diff
    if b.min() >= -1e-12 and b.max() <= 1.0 + 1e-12:
        return np.clip(b, 0.0, 1.0)
    lo = -float(a.max()) - 1.0
    hi = 1.0 - float(a.min()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(a + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(a + 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, n):
    f = np.correlate(h, 1.0 - h, mode='full')
    return float(f.max()) * (2.0 / float(n))

def soft_value(h, alpha):
    f = np.correlate(h, 1.0 - h, mode='full')
    m = f.max()
    s = alpha * (f - m)
    sm = s.max()
    e = np.exp(s - sm)
    return float(m) + (float(sm) + float(np.log(e.sum()))) / float(alpha)

def soft_info(h, alpha):
    f = np.correlate(h, 1.0 - h, mode='full')
    m = f.max()
    s = alpha * (f - m)
    sm = s.max()
    e = np.exp(s - sm)
    J = float(m) + (float(sm) + float(np.log(e.sum()))) / float(alpha)
    p = e / e.sum()
    return J, p, float(m)

def gradient_from_p(h, p):
    n = len(h)
    c = n - 1
    G = np.zeros(n, dtype=np.float64)
    G += p[c] * (1.0 - 2.0 * h)
    for d in range(1, n):
        pp = p[c + d]
        pn = p[c - d]
        if pp == 0.0 and pn == 0.0:
            continue
        v = n - d
        if pp != 0.0:
            G[:v] -= pp * h[d:]
            G[d:] += pp * (1.0 - h[:v])
        if pn != 0.0:
            G[:v] += pn * (1.0 - h[d:])
            G[d:] -= pn * h[:v]
    return G

def _pe_obj(x):
    h = np.clip(x, 0.0, 1.0)
    val = soft_value(h, _OPT_ALPHA)
    rho = _OPT_RHO
    if rho:
        v = float(h.sum() - _OPT_TARGET)
        val += 0.5 * rho * v * v
    return float(val)

def _pe_jac(x):
    h = np.clip(x, 0.0, 1.0)
    _, p, _ = soft_info(h, _OPT_ALPHA)
    G = gradient_from_p(h, p)
    rho = _OPT_RHO
    if rho:
        G += rho * float(h.sum() - _OPT_TARGET)
    return G

def lbfgs_polish(h, n, alpha, rho, maxiter):
    global _OPT_ALPHA, _OPT_RHO, _OPT_TARGET
    h0 = project_feasible(h.copy(), n)
    _OPT_ALPHA = float(alpha)
    _OPT_RHO = float(rho)
    _OPT_TARGET = 0.5 * n
    try:
        res = minimize(_pe_obj, h0, jac=_pe_jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': int(maxiter), 'ftol': 1e-12,
                                'gtol': 1e-8, 'maxls': 30, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def _sl_cons(x):
    return float(np.clip(x, 0.0, 1.0).sum() - _OPT_TARGET)

def _sl_cons_grad(x):
    return np.ones(len(x), dtype=np.float64)

def slsqp_polish(h, n, alpha, maxiter):
    global _OPT_ALPHA, _OPT_RHO, _OPT_TARGET
    h0 = project_feasible(h.copy(), n)
    _OPT_ALPHA = float(alpha)
    _OPT_RHO = 0.0
    _OPT_TARGET = 0.5 * n
    try:
        cons = [{'type': 'eq', 'fun': _sl_cons, 'jac': _sl_cons_grad}]
        res = minimize(_pe_obj, h0, jac=_pe_jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': 1e-9, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def binarize(h, n):
    order = np.argsort(-np.asarray(h, dtype=float))
    out = np.zeros(n, dtype=float)
    out[order[:n // 2]] = 1.0
    return out

def best_shift(h, n):
    h = np.asarray(h, dtype=float)
    best = compute_c5(h, n)
    out = h.copy()
    if n <= 1:
        return out
    stride = 1 if n <= 600 else max(1, n // 120)
    for s in range(1, n, stride):
        hh = np.roll(h, s)
        c = compute_c5(hh, n)
        if c < best - 1e-13:
            best = c
            out = hh
    return out

def random_perturb(h, n, scale, rng):
    g = rng.normal(0.0, scale, n)
    g -= g.mean()
    return project_feasible(h + g, n)

def pair_soft_value(h, alpha, i, j, t):
    ai = float(h[i] - t)
    bj = float(h[j] + t)
    if ai < -1e-14 or ai > 1.0 + 1e-14 or bj < -1e-14 or bj > 1.0 + 1e-14:
        return float('inf')
    hh = h.copy()
    hh[i] = min(max(ai, 0.0), 1.0)
    hh[j] = min(max(bj, 0.0), 1.0)
    return soft_value(hh, alpha)

    no_imp = 0

                if v2 < best_J:
                    best_J, best_t = v2, t2
                if v1 < v2:
                    b_t = t2
                else:
                    a_t = t1
            h_new = h.copy()
            h_new[best_i] = max(0.0, min(1.0, float(h[best_i]) - best_t))
            h_new[best_j] = max(0.0, min(1.0, float(h[best_j]) + best_t))
            J_new = soft_value(h_new, alpha)
            c_new = compute_c5(h_new, n)
            if (c_new < c_cur - 1e-12) or (J_new < J - 1e-10 and c_new <= c_cur + 1e-9):
                h = h_new
                c_cur = c_new
                J, p, _ = soft_info(h, alpha)
                G = gradient_from_p(h, p)
                no_imp = 0
                if c_cur < c_best - 1e-12 or (c_cur <= c_best + 1e-11 and J < J_best - 1e-7):
                    c_best, h_best, J_best = c_cur, h.copy(), J
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp >= 12:
            base = h_best.copy()
            g = rng.normal(0.0, 0.04, n)
            g -= g.mean()
            h = project_feasible(base + 0.7 * g, n)
            c_cur = compute_c5(h, n)
            if c_cur > c_best + 0.01:
                h = base.copy()
                c_cur = c_best
            J, p, _ = soft_info(h, alpha)
            G = gradient_from_p(h, p)
            no_imp = 0
    h_best = project_feasible(h_best, n)
    return h_best, compute_c5(h_best, n)

def projected_soft(h, n, alpha, max_iters, deadline, rng=None):
    if rng is None:
        rng = np.random
    h = project_feasible(h.copy(), n)
    cbest = compute_c5(h, n)
    hbest = h.copy()
    J, p, _ = soft_info(h, alpha)
    G = gradient_from_p(h, p)
    step = 0.05
    for _ in range(int(max_iters)):
        if time.time() > deadline:
            break
        accepted = False
        s = step
        for _ in range(15):
            cand = project_feasible(h - s * G, n)
            Jc = soft_value(cand, alpha)
            if Jc < J - 1e-12:
                h = cand
                J = Jc
                c = compute_c5(h, n)
                if c < cbest:
                    cbest, hbest = c, h.copy()
                J, p, _ = soft_info(h, alpha)
                G = gradient_from_p(h, p)
                step = min(2.0, s * 1.2)
                accepted = True
                break
            s *= 0.5
        if not accepted:
            if step < 1e-6:
                break
            g = rng.normal(0.0, 0.02, n)
            g -= g.mean()
            h = project_feasible(h + g, n)
            J, p, _ = soft_info(h, alpha)
            G = gradient_from_p(h, p)
            c = compute_c5(h, n)
            if c < cbest:
                cbest, hbest = c, h.copy()
            step = 0.05
    return project_feasible(hbest, n), cbest

def upsample(h, m):
    old = len(h)
    new = old * m
    xo = (np.arange(old, dtype=float) + 0.5) / float(old)
    xn = (np.arange(new, dtype=float) + 0.5) / float(new)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 250
        h = np.full(n, 0.5)
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    best_n = n

    raw = [h, best_shift(h, n), binarize(h, n),
           best_shift(binarize(h, n), n), best_shift(h[::-1].copy(), n),
           random_perturb(h, n, 0.06, rng), random_perturb(h, n, 0.15, rng)]
    starts = []
    for hs in raw:
        hs = project_feasible(hs, n)
        starts.append((compute_c5(hs, n), len(starts), hs))
    starts.sort()
    scale = 250.0 / float(max(n, 250))

    for idx in range(min(6, len(starts))):
        if time.time() > dl - 40:
            break
        c_start, _, h0 = starts[idx]
        if idx > 0 and c_start > best_c + 0.02:
            continue
        hcur, ccur = h0.copy(), c_start
        alphas = (20.0, 100.0, 400.0) if idx == 0 else (60.0, 300.0)
        rhos = (5.0, 50.0) if idx == 0 else (25.0,)
        lb_base = int(650 * scale) if idx == 0 else int(220 * scale)
        if dl - time.time() < 120:
            lb_base = min(lb_base, 120)
        if dl - time.time() < 60:
            lb_base = min(lb_base, 60)
        pair_base = 120 if idx == 0 else 40
        for alpha in alphas:
            if time.time() > dl - 45:
                break
            for rho in rhos:
                if time.time() > dl - 35:
                    break
                h2 = lbfgs_polish(hcur, n, alpha, rho, lb_base)
                c2 = compute_c5(h2, n)
                if c2 < ccur - 1e-12:
                    hcur, ccur = h2, c2
                elif c2 <= ccur + 1e-8:
                    hcur = h2
                if ccur < best_c - 1e-12:
                    best_c, best_h, best_n = ccur, hcur.copy(), n
            if time.time() > dl - 30:
                break
            h3, c3 = pair_descent(hcur, n, alpha, min(pair_base, int(300 * scale)),
                                  min(dl - 5.0, time.time() + (45.0 if idx == 0 else 18.0)),
                                  rng, 10 if idx == 0 else 7)
            if c3 < ccur - 1e-12:
                hcur, ccur = h3, c3
            if ccur < best_c - 1e-12:
                best_c, best_h, best_n = ccur, hcur.copy(), n

    if time.time() < dl - 160 and best_n <= 320:
        s = 250.0 / float(max(best_n, 250))
        hr = best_shift(best_h[::-1].copy(), best_n)
        cr = compute_c5(hr, best_n)
        if cr < best_c:
            best_c, best_h, best_n = cr, hr.copy(), best_n
        if time.time() < dl - 110:
            h2 = lbfgs_polish(hr, best_n, 300.0, 50.0, int(140 * s))
            c2 = compute_c5(h2, best_n)
            if c2 < best_c:
                best_c, best_h, best_n = c2, h2.copy(), best_n
            h3, c3 = pair_descent(best_h, best_n, 800.0, int(70 * s),
                                  min(dl - 25.0, time.time() + 35.0), rng, 8)
            if c3 < best_c:
                best_c, best_h, best_n = c3, h3.copy(), best_n

    if time.time() < dl - 100 and best_n <= 600:
        s = 250.0 / float(max(best_n, 250))
        for alpha, rho, it in ((1000.0, 100.0, int(450 * s)), (2000.0, 300.0, int(240 * s))):
            if time.time() > dl - 50:
                break
            h2 = lbfgs_polish(best_h, best_n, alpha, rho, max(30, it))
            c2 = compute_c5(h2, best_n)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()
            if time.time() < dl - 40:
                h3, c3 = pair_descent(best_h, best_n, 1000.0, int(80 * s),
                                      min(dl - 15.0, time.time() + 35.0), rng, 8)
                if c3 < best_c:
                    best_c, best_h = c3, h3.copy()

    if time.time() < dl - 80 and best_n <= 600:
        s = 250.0 / float(max(best_n, 250))
        h2, c2 = projected_soft(best_h, best_n, 1200.0, int(450 * s), dl - 15.0, rng)
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2.copy(), best_n

    if time.time() < dl - 90 and best_n <= 600:
        s = 250.0 / float(max(best_n, 250))
        h2 = slsqp_polish(best_h, best_n, 800.0, int(160 * s))
        c2 = compute_c5(h2, best_n)
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2.copy(), best_n

    if time.time() < dl - 220 and best_n <= 400:
        n2 = int(best_n * 2)
        s2 = 250.0 / float(max(n2, 250))
        for h0 in (np.repeat(best_h, 2), upsample(best_h, 2)):
            if time.time() > dl - 140:
                break
            hcur = project_feasible(h0, n2)
            ccur = compute_c5(hcur, n2)
            for alpha in (50.0, 300.0):
                if time.time() > dl - 80:
                    break
                h2 = lbfgs_polish(hcur, n2, alpha, 25.0, int(170 * s2))
                c2 = compute_c5(h2, n2)
                if c2 < ccur - 1e-12:
                    hcur, ccur = h2, c2
                if ccur < best_c:
                    best_c, best_h, best_n = ccur, hcur.copy(), n2
            if time.time() < dl - 80:
                h3, c3 = pair_descent(hcur, n2, 500.0, int(70 * s2),
                                      min(dl - 30.0, time.time() + 40.0), rng, 8)
                if c3 < best_c:
                    best_c, best_h, best_n = c3, h3.copy(), n2

    if time.time() < dl - 25 and best_n <= 800:
        s = 250.0 / float(max(best_n, 250))
        h2 = lbfgs_polish(best_h, best_n, 1200.0, 250.0, int(140 * s))
        c2 = compute_c5(h2, best_n)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()
        if time.time() < dl - 10:
            h3, c3 = pair_descent(best_h, best_n, 1500.0, int(45 * s), dl - 3.0, rng, 6)
            if c3 < best_c:
                best_c, best_h = c3, h3.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
