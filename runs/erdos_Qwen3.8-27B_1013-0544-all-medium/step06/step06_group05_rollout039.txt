    #   p + k == i: h[i] changed from 0 to 1, so (1-h[i]) changed from 1 to 0 → delta[p,m] -= 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_deltas(h, n):
    lags = np.arange(-(n-1), n)
    n_corrs = 2 * n - 1
    positions = np.arange(n)[:, None] + lags[None, :]
    valid_plus = (positions >= 0) & (positions < n)
    h_plus = np.zeros((n, n_corrs))
    h_plus[valid_plus] = 1.0 - h[positions[valid_plus]]
    positions_minus = np.arange(n)[:, None] - lags[None, :]
    valid_minus = (positions_minus >= 0) & (positions_minus < n)
    h_minus = np.zeros((n, n_corrs))
    h_minus[valid_minus] = h[positions_minus[valid_minus]]
    return h_plus - h_minus

def update_deltas(deltas, n, i, j):
    """Update deltas after h[i] goes 0->1 and h[j] goes 1->0."""
    lags = np.arange(-(n-1), n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        p = i - k
        if 0 <= p < n:
            deltas[p, m] -= 1.0
        p = j - k
        if 0 <= p < n:
            deltas[p, m] += 1.0
        p = i + k
        if 0 <= p < n:
            deltas[p, m] -= 1.0
        p = j + k
        if 0 <= p < n:
            deltas[p, m] += 1.0

def fast_binary_anneal(h, n, deadline, max_iters=20000, seed=42):
    dx = 2.0 / n
    cur = h.copy().astype(float)
    best_h = cur.copy()
    best_corr = np.max(np.correlate(cur, 1-cur, mode='full'))
    cur_corr = np.correlate(cur, 1-cur, mode='full').astype(np.float64)
    cur_max = cur_corr.max()
    deltas = compute_deltas(cur, n)
    n_corrs = 2 * n - 1
    rng = np.random.RandomState(seed)
    T = 2.0
    T_min = 0.05
    cooling = 0.9997
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        zeros_idx = np.where(cur == 0)[0]
        ones_idx = np.where(cur == 1)[0]
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        
        # Find best swap
        # new_max[i,j] = max_m(cur_corr[m] + deltas[i,m] - deltas[j,m])
        diff = deltas[zeros_idx][:, None, :] - deltas[ones_idx][None, :, :] + cur_corr[None, None, :]
        new_maxes = diff.max(axis=2)
        flat_idx = np.unravel_index(np.argmin(new_maxes), new_maxes.shape)
        best_i = zeros_idx[flat_idx[0]]
        best_j = ones_idx[flat_idx[1]]
        best_new_max = new_maxes[flat_idx]
        
        # Accept/reject
        accept = True
        if best_new_max > cur_max + 1e-10:
            delta = best_new_max - cur_max
            if T > T_min:
                if rng.rand() > np.exp(-delta / T):
                    accept = False
        
        if accept:
            cur[best_i] = 1.0
            cur[best_j] = 0.0
            update_deltas(deltas, n, int(best_i), int(best_j))
            cur_corr = np.correlate(cur, 1-cur, mode='full').astype(np.float64)
            cur_max = cur_corr.max()
            if cur_max < best_corr:
                best_corr = cur_max
                best_h = cur.copy()
        T = max(T * cooling, T_min)
    
    return best_h, float(best_corr * dx)

def slsqp_optimize(h, n, deadline, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        alpha = 1000.0
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log((np.exp(sh - sm)).sum())) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        alpha = 1000.0
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        # grad of sum_k w[k] * g[k]
        lags = np.arange(-(n-1), n)
        positions = np.arange(n)[:, None] + lags[None, :]
        valid_plus = (positions >= 0) & (positions < n)
        h_plus = np.zeros((n, 2*n-1))
        h_plus[valid_plus] = 1.0 - x[positions[valid_plus]]
        positions_minus = np.arange(n)[:, None] - lags[None, :]
        valid_minus = (positions_minus >= 0) & (positions_minus < n)
        h_minus = np.zeros((n, 2*n-1))
        h_minus[valid_minus] = x[positions_minus[valid_minus]]
        return (h_plus - h_minus) @ w
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h

def coordinate_descent(h, n, max_rounds=100):
    """Try changing individual coordinates to discrete levels."""
    dx = 2.0 / n
    h = h.copy()
    levels = [0.0, 0.25, 0.5, 0.75, 1.0]
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            old_val = h[i]
            best_val = old_val
            best_c = compute_c5(project_feasible(h, n), n)
            for lv in levels:
                if lv == old_val:
                    continue
                h[i] = lv
                h_p = project_feasible(h, n)
                c = compute_c5(h_p, n)
                if c < best_c - 1e-12:
                    best_c = c
                    best_val = lv
            h[i] = best_val
            if best_val != old_val:
                improved = True
        if not improved:
            break
    return project_feasible(h, n)

def make_golden_pattern(n):
    """Create a low-discrepancy binary pattern using golden ratio."""
    h = np.zeros(n)
    phi = (1 + 5**0.5) / 2
    positions = sorted((int((i * phi) % 1.0 * n) for i in range(n // 2)))
    for p in positions:
        h[p] = 1.0
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_deltas(h, n):
    lags = np.arange(-(n - 1), n)
    n_corrs = 2 * n - 1
    positions_plus = np.arange(n)[:, None] + lags[None, :]
    valid_plus = (positions_plus >= 0) & (positions_plus < n)
    h_plus = np.zeros((n, n_corrs))
    h_plus[valid_plus] = 1.0 - h[positions_plus[valid_plus]]
    positions_minus = np.arange(n)[:, None] - lags[None, :]
    valid_minus = (positions_minus >= 0) & (positions_minus < n)
    h_minus = np.zeros((n, n_corrs))
    h_minus[valid_minus] = h[positions_minus[valid_minus]]
    return h_plus - h_minus

def update_deltas_fast(deltas, n, i, j):
    lags = np.arange(-(n - 1), n)
    n_corrs = 2 * n - 1
    m_idx = np.arange(n_corrs)
    p1 = i - lags
    v1 = (p1 >= 0) & (p1 < n)
    p2 = i + lags
    v2 = (p2 >= 0) & (p2 < n)
    p3 = j - lags
    v3 = (p3 >= 0) & (p3 < n)
    p4 = j + lags
    v4 = (p4 >= 0) & (p4 < n)
    deltas[p1[v1], m_idx[v1]] -= 1.0
    deltas[p2[v2], m_idx[v2]] -= 1.0
    deltas[p3[v3], m_idx[v3]] += 1.0
    deltas[p4[v4], m_idx[v4]] += 1.0

def fast_binary_anneal(h, n, deadline, max_iters=20000, seed=42):
    dx = 2.0 / n
    cur = h.copy().astype(float)
    best_h = cur.copy()
    cur_corr = np.correlate(cur, 1 - cur, mode='full').astype(np.float64)
    cur_max = cur_corr.max()
    best_corr = cur_max
    deltas = compute_deltas(cur, n)
    rng = np.random.RandomState(seed)
    T = 3.0
    T_min = 0.02
    cooling = 0.9998
    n_corrs = 2 * n - 1

    for it in range(max_iters):
        if time.time() > deadline:
            break
        zeros_idx = np.where(cur == 0)[0]
        ones_idx = np.where(cur == 1)[0]
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        diff = (deltas[zeros_idx][:, None, :] - deltas[ones_idx][None, :, :]
                + cur_corr[None, None, :])
        new_maxes = diff.max(axis=2)
        flat_idx = np.unravel_index(np.argmin(new_maxes), new_maxes.shape)
        best_i = zeros_idx[flat_idx[0]]
        best_j = ones_idx[flat_idx[1]]
        best_new_max = new_maxes[flat_idx]
        accept = best_new_max <= cur_max + 1e-10
        if not accept and T > T_min:
            if rng.rand() < np.exp(-(best_new_max - cur_max) / T):
                accept = True
        if accept:
            cur[best_i] = 1.0
            cur[best_j] = 0.0
            update_deltas_fast(deltas, n, int(best_i), int(best_j))
            cur_corr = np.correlate(cur, 1 - cur, mode='full').astype(np.float64)
            cur_max = cur_corr.max()
            if cur_max < best_corr - 1e-12:
                best_corr = cur_max
                best_h = cur.copy()
        elif T > T_min and rng.rand() < 0.05:
            ri = rng.choice(zeros_idx)
            rj = rng.choice(ones_idx)
            cur[ri] = 1.0
            cur[rj] = 0.0
            update_deltas_fast(deltas, n, int(ri), int(rj))
            cur_corr = np.correlate(cur, 1 - cur, mode='full').astype(np.float64)
            cur_max = cur_corr.max()
        T = max(T * cooling, T_min)
    return best_h, float(best_corr * dx)

def slsqp_optimize(h, n, deadline, alpha=1000.0, maxiter=5000):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        lags = np.arange(-(n - 1), n)
        pos_p = np.arange(n)[:, None] + lags[None, :]
        vp = (pos_p >= 0) & (pos_p < n)
        hp = np.zeros((n, 2 * n - 1))
        hp[vp] = 1.0 - x[pos_p[vp]]
        pos_m = np.arange(n)[:, None] - lags[None, :]
        vm = (pos_m >= 0) & (pos_m < n)
        hm = np.zeros((n, 2 * n - 1))
        hm[vm] = x[pos_m[vm]]
        return (hp - hm) @ w

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_from_continuous(h, n):
    sorted_h = np.sort(h)[::-1]
    thresh = sorted_h[n // 2]
    h_bin = (h >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        excess = np.where(h_bin == 1)[0]
        h_bin[excess[n_ones - n // 2:]] = 0.0
    elif n_ones < n // 2:
        deficit = np.where(h_bin == 0)[0]
        h_bin[deficit[:n // 2 - n_ones]] = 1.0
    return h_bin

def make_golden_pattern(n):
    h = np.zeros(n)
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    positions = sorted(int(((i + 1) * phi) % 1.0 * n) for i in range(n // 2))
    for p in positions:
        h[p] = 1.0
    return h
