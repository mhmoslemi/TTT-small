        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        # Soft-max for smoothness
        alpha = 5000
        shifted = (corr - m) * alpha
        shifted = np.clip(shifted, -500, 0)
        log_sum = np.log(np.exp(shifted).sum())
        return m + log_sum / alpha
    
    def objective_grad(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        alpha = 5000
        shifted = (corr - m) * alpha
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted) / np.exp(shifted).sum()

        # d/corr[k]/dx[j] = (1-x[j+k-(n-1)])*dx - x[j-k-(n-1)]*dx (for valid indices)

            m_lo = max(0, -s)
