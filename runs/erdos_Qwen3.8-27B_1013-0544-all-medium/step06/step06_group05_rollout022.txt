def compute_delta(i, j, h, n):
    shifts = np.arange(-(n-1), n)
    # h_i_s = h[i+s] for valid s, 0 otherwise
    idx_i_s = (i + shifts).astype(int)
    valid_i_s = (idx_i_s >= 0) & (idx_i_s < n)
    h_i_s = np.where(valid_i_s, h[np.clip(idx_i_s, 0, n-1)], 0.0)
    
    idx_i_neg_s = (i - shifts).astype(int)
    valid_i_neg_s = (idx_i_neg_s >= 0) & (idx_i_neg_s < n)
    h_i_neg_s = np.where(valid_i_neg_s, h[np.clip(idx_i_neg_s, 0, n-1)], 0.0)
    
    idx_j_s = (j + shifts).astype(int)
    valid_j_s = (idx_j_s >= 0) & (idx_j_s < n)
    h_j_s = np.where(valid_j_s, h[np.clip(idx_j_s, 0, n-1)], 0.0)
    
    idx_j_neg_s = (j - shifts).astype(int)
    valid_j_neg_s = (idx_j_neg_s >= 0) & (idx_j_neg_s < n)
    h_j_neg_s = np.where(valid_j_neg_s, h[np.clip(idx_j_neg_s, 0, n-1)], 0.0)
    
    return (h_i_s + h_i_neg_s) - (h_j_s + h_j_neg_s)

    # Affected: (p=i-s, s_idx) and (p=i+s, s_idx) for each s

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def build_profiles(h, n):
    shifts = np.arange(-(n - 1), n)
    profiles = np.zeros((n, 2 * n - 1))
    p = np.arange(n)
    for k in range(2 * n - 1):
        s = shifts[k]
        col = np.zeros(n)
        ps = p + s
        v1 = (ps >= 0) & (ps < n)
        col[v1] += h[ps[v1]]
        pn = p - s
        v2 = (pn >= 0) & (pn < n)
        col[v2] += h[pn[v2]]
        profiles[:, k] = col
    return profiles

def update_profiles(profiles, shifts, n, i, di, j, dj):
    pm = i - shifts
    pp = i + shifts
    vm = (pm >= 0) & (pm < n)
    vp = (pp >= 0) & (pp < n)
    profiles[pm[vm], np.flatnonzero(vm)] += di
    profiles[pp[vp], np.flatnonzero(vp)] += di
    pm = j - shifts
    pp = j + shifts
    vm = (pm >= 0) & (pm < n)
    vp = (pp >= 0) & (pp < n)
    profiles[pm[vm], np.flatnonzero(vm)] += dj
    profiles[pp[vp], np.flatnonzero(vp)] += dj

def slsqp_polish(h, n, deadline, maxiter=3000):
    dx = 2.0 / n
    target = n / 2.0
    alpha = 3000.0

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        x = np.clip(res.x, 0.0, 1.0)
        d = target - x.sum()
        if abs(d) > 1e-10:
            if d > 0:
                room = 1.0 - x
                s = room.sum()
                if s > 1e-14:

            else:
                s = x.sum()
                if s > 1e-14:
                    x = np.clip(x + d / s * x, 0, 1)

        return h
