import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def sa_optimize(h_init, n, deadline, seed=42, T_start=0.4, T_end=0.005):
    """Simulated annealing on binary sequences."""
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    
    rng = np.random.RandomState(seed)
    
    # Precompute
    ones_list = np.where(h == 1)[0]
    zeros_list = np.where(h == 0)[0]
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    it = 0
    n_iter_target = 5000000  # will be limited by deadline
    
    while time.time() < deadline:
        it += 1
        t_frac = min(it / n_iter_target, 1.0)
        T = T_start * (T_end / T_start) ** t_frac
        
        # Random swap: pick a 1 and a 0, swap them
        i = rng.randint(n)
        j = rng.randint(n)
        
        # Ensure i is a 1 and j is a 0 (or vice versa)
        if h[i] == h[j]:
            continue
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        
        # Compute new energy
        new_c = compute_c5(h, n)
        
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c
