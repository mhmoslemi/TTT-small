import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha, rc=0.0):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if rc > 0:
        grad += 2.0 * rc * (h - 0.5)
    return grad

def _obj(x, n, alpha, mu, rc):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    if rc > 0:
        val += rc * np.sum((x - 0.5) ** 2)
    return float(val)

def _grad(x, n, alpha, mu, rc):
    g = grad_smooth_max(x, n, alpha, rc)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (50, 5.0, 0.0), (200, 10.0, 0.0), (1000, 20.0, 0.0),
        (5000, 50.0, 0.01), (20000, 100.0, 0.05),
        (100000, 500.0, 0.1), (500000, 2000.0, 0.0),
        (2000000, 5000.0, 0.0), (10000000, 20000.0, 0.0),
        (50000000, 100000.0, 0.0)
    ]
    for alpha, mu, rc in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_obj, h, jac=_grad,
                          args=(n, alpha, mu, rc), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 8,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_phase(h, n, deadline, alpha_init=100.0, lr_init=0.02):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

        v = 0.9 * v - lr * g
        h_new = project_feasible(h + v, n)

        if c_new < best_c - 1e-15:

            best_h = h_new.copy()
            h = h_new
            failed = 0
        else:

            if failed % 20 == 0:

                v *= 0.3
                alpha *= 1.5

    return best_h, best_c

def simulated_annealing(h, n, deadline, t_init=0.001, t_final=1e-10, cooling=0.9998):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    t = t_init
    rng = np.random.RandomState(777)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    while time.time() < deadline and t > t_final:
        d = steps[rng.randint(len(steps))]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        if h[i] >= d and h[j] + d <= 1.0:
            h[i] -= d; h[j] += d
            c = compute_c5(h, n)
            delta = c - best_c
            if delta < -1e-15:
                best_c = c
                best_h = h.copy()
            elif delta <= 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
                pass  # accept
            else:
                h[i] += d; h[j] -= d
        elif h[j] >= d and h[i] + d <= 1.0:
            h[j] -= d; h[i] += d
            c = compute_c5(h, n)
            delta = c - best_c
            if delta < -1e-15:
                best_c = c
                best_h = h.copy()
            elif delta <= 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
                pass
            else:
                h[j] += d; h[i] -= d
        
        t *= cooling
    
    return best_h, best_c
