import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(obj, alpha):
    # Log-sum-exp smoothing of max
    m = obj.max()
    shifted = alpha * (obj - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, maxiter):
    dx = 2.0 / n
    def objective(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(c, alpha)
    
    def grad(x):
        # Gradient of smoothed max
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        # d/dx of c[k] for each k
        # c[k] = sum_i x[i+k](1-x[i]) * dx for appropriate i
        # This is complex; let SLSQP compute numerically
        return None
    
    def constraint_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    
    res = minimize(objective, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, n)

The cross-term occurs when both positions i and j appear in the same product term h[s+k]*(1-h[s]). This happens when:
- s+k = i and s = j (i.e., k = i-j, m = N-1+i-j)
- s+k = j and s = i (i.e., k = j-i, m = N-1+j-i)

The true change at this m (just from this term) is:
(1-h[i])*h[j] - h[i]*(1-h[j]) = h[j] - h[i] (for binary, with h[i]=0, h[j]=1: change = 1)

The individual contributions I computed:
- "i as s+k": 1-h[s] = 1-h[j] (for binary h[j]=1: change = 0)
- "j as s": h[s+k] = h[i] (for binary h[i]=0: change = 0)
Sum of individual: 0

So the cross correction at m = N-1+i-j is: true_change - sum_individual = 1 - 0 = 1 (for binary).

For the case s+k=j, s=i (m = N-1+j-i):
True change: h[i]*(1-h[j]) - h[j]*(1-h[i]) (for binary h[i]=0, h[j]=1: 0*0 - 1*1 = -1)

Sum individual: -1+1 = 0
Cross correction: -1-0 = -1.

- At m = N-1+(i-j): +1
- At m = N-1+(j-i): -1
