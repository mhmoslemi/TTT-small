    shifted = alpha * (corr - cmax)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    m_alpha = cmax + np.log(sum_exp) / alpha
    
    # softmax weights
    probs = exp_shifted / sum_exp
    
    n1 = n - 1
    one_mh = 1.0 - h
    first = np.zeros(n)
    second = np.zeros(n)
    for j in range(n):
        first[j] = np.dot(one_mh, probs[n1-j:n1-j+n])
        second[j] = np.dot(h, probs[j:j+n][::-1])
    
    grad = dx * (first - second)
    
    s = h.sum()
    penalty = lam * (s - n/2.0) ** 2
    grad = grad + 2.0 * lam * (s - n/2.0)
    
    obj = m_alpha + penalty
    return obj, grad

def optimize_pnorm(h_init, n, p_list, lam, deadline):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for p in p_list:
        if time.time() > deadline - 5:
            break
        
        def fun(x):
            return pnorm_grad(x, n, p, lam)[0]
        
        def fprime(x):
            return pnorm_grad(x, n, p, lam)[1]
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=fprime,
                          bounds=bounds, options={'maxiter': 500, 'gtol': 1e-8, 'maxls': 40})
            h = project_feasible(res.x, n)
        except Exception:
            pass
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                delta[k1] += 1.0
                k2 = n-1 + (i-j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                j = best_j
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
