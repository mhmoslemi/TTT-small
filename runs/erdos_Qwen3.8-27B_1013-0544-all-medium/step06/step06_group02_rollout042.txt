    return float(np.max(compute_correlation(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_correlation(x, n)
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    if smax > 500:
        val = m + (smax + np.log(np.sum(np.exp(s - smax)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s - smax)) + np.exp(-smax)) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_correlation(x, n)
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    w = np.exp(s - smax)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[k:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[k:n])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[kk:n])
                grad[kk:n] += wm * (-x[0:nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_stage(h, n, deadline, alpha_list, max_iters=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        t_sec = min(60, deadline - time.time())
        mu = 1.0 + 0.1 * alpha  # penalty weight scales with alpha
        try:
            res = minimize(softmax_obj, best_h, jac=softmax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20,
                                   'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def nesterov_pd(h, n, deadline, alpha=1000, lr_init=0.005):
    """Nesterov momentum projected gradient descent."""
    h = project_feasible(h.copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    momentum = 0.9
    stuck_count = 0
    
    for it in range(50000):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        y = h + momentum * v
        y = project_feasible(y, n)
        g = softmax_grad(y, n, alpha, 0.0)  # no penalty, use projection
        
        # Project gradient onto the constraint (sum(h) = n/2)
        # The tangent space is {d : sum(d) = 0, d_i ≠ 0 only where 0 < h_i < 1}
        g_proj = g - np.sum(g * (h > 1e-10) * (h < 1 - 1e-10)) / max(1, ((h > 1e-10) * (h < 1 - 1e-10)).sum()) * ((h > 1e-10) * (h < 1 - 1e-10))
        # Actually, simpler: just subtract the mean of the free components
        free = (h > 1e-8) & (h < 1 - 1e-8)
        if free.sum() > 0:
            g[free] -= g[free].mean()
        g[~free] = 0
        
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count % 100 == 0:
                lr *= 0.8
        
        h = h_new
        momentum = min(0.99, momentum * 1.0001)
    
    return best_h, best_c

def pair_search_gradient(h, n, deadline, step_sizes=None):
    """Guided pair search using gradient differences."""
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    alpha = 1000
    
    for step in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            g = softmax_grad(h, n, alpha, 0.0)
            # Try pairs with large gradient difference
            # For efficiency, only try a subset
            n_try = min(n, 200)
            indices = np.random.choice(n, n_try, replace=False) if n > n_try else np.arange(n)
            for ii in range(len(indices)):
                for jj in range(ii+1, len(indices)):
                    i, j = indices[ii], indices[jj]
                    gd = g[i] - g[j]
                    if abs(gd) < 1e-10:
                        continue
                    # Try moving from j to i (if g[i] < g[j], decreasing h[i] and increasing h[j] helps)
                    # Actually, we want to decrease the objective, so move in -gradient direction
                    # If g[i] > g[j], we want to increase h[j] and decrease h[i]
                    direction = -np.sign(gd)  # move h[i] by -direction*step, h[j] by +direction*step
                    delta = step * direction
                    if h[i] + delta >= 0 and h[i] + delta <= 1 and h[j] - delta >= 0 and h[j] - delta <= 1:
                        h[i] += delta
                        h[j] -= delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
    return h, best_c

def pair_search_exhaustive(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:

                lo = max(0, i - 1) if radius is None else max(0, i - radius)

        h_t, c_t = lbfgs_stage(h_interp, n_try, time.time() + alloc * 0.5,
                               [100, 1000, 10000, 50000, 200000], max_iters=150)
        if time.time() < deadline - 25:
            h_t, c_t = pair_search_local(h_t, n_try, min(time.time() + 25, deadline - 10),
                                         [0.005, 0.002, 0.001, 0.0005], radius=50)
        if c_t < c_best:
            c_best = c_t
            h_best = h_t
            n_best = n_try
    return h_best, n_best, c_best

def perturb_restart(h, n, rng, scale=0.02):
    h_p = h + rng.randn(n) * scale
    h_p = project_feasible(h_p, n)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n = 250

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B progressive alpha
    t1 = min(t0 + 180, deadline - 350)
    best_h, best_c = lbfgs_stage(best_h, n, t1,
                                  [30, 100, 500, 2000, 10000, 50000], max_iters=250)

    # Phase 2: Nesterov momentum
    if time.time() < deadline - 250:
        t2 = min(time.time() + 80, deadline - 250)
        h2, c2 = nesterov_pd(best_h, n, t2, alpha=5000, lr_init=0.004)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: High-alpha L-BFGS-B
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        best_h, best_c = lbfgs_stage(best_h, n, t3,
                                      [50000, 200000, 500000, 2000000], max_iters=400)

    # Phase 4: Perturbation restarts to escape local minima
    if time.time() < deadline - 180:
        for restart in range(3):
            if time.time() > deadline - 100:
                break
            h_p = perturb_restart(best_h, n, rng, scale=0.015)
            alloc = min(40, deadline - time.time() - 60)
            if alloc < 20:
                break
            h_r, c_r = lbfgs_stage(h_p, n, time.time() + alloc * 0.6,
                                    [1000, 10000, 100000], max_iters=200)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 5: Nesterov again on best
    if time.time() < deadline - 120:
        t5 = min(time.time() + 50, deadline - 120)
        h5, c5 = nesterov_pd(best_h, n, t5, alpha=10000, lr_init=0.002)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Guided pair search
    if time.time() < deadline - 80:
        t6 = min(time.time() + 50, deadline - 80)
        best_h, best_c = pair_search_guided(best_h, n, t6, rng)

    # Phase 7: Local exhaustive pair search
    if time.time() < deadline - 50:
        t7 = min(time.time() + 60, deadline - 50)
        best_h, best_c = pair_search_local(best_h, n, t7,
                                            [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                            radius=60)

    # Phase 8: Try different n
    if time.time() < deadline - 40:
        best_h, n, best_c = try_other_n(best_h, n, best_c, deadline - 15, rng)

    # Phase 9: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = lbfgs_stage(best_h, nf, deadline - 1,
                                      [200000, 1000000], max_iters=100)
        best_h, best_c = pair_search_local(best_h, nf, deadline - 1,
                                            [0.001, 0.0005, 0.0002, 0.0001], radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
