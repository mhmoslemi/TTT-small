def resample_step(h,n_target):
    n=len(h)
    if n==n_target: return h.copy()
    if n_target == 0: return h.copy()
    # if n_target multiple or integer factor, use repeat/aggregate
    if n_target % n == 0:
        fac = n_target//n
        return np.repeat(h, fac)
    if n % n_target == 0:
        fac = n//n_target
        return h.reshape(n_target, fac).mean(axis=1)
    # nearest
    old_idx = np.linspace(0,n-1,n_target)
    new_idx = np.linspace(0,n-1,n_target)
    # if downsample use average of interval? Simpler nearest.

def compute_c5(h,n):
    dx=2.0/float(n)

def project_feasible(z,n):
    target=0.5*float(n)
    z=np.asarray(z,dtype=float)
    h=np.clip(z,0.0,1.0)
    s=float(h.sum())
    if abs(target-s) <= 1e-12:
        return h
    lo=float(np.min(z))-1.0
    hi=float(np.max(z))+1.0
    for _ in range(60):
        tau=0.5*(lo+hi)
        h2=np.clip(z-tau,0.0,1.0)
        if h2.sum() > target:
            lo=tau
        else:
            hi=tau
    return np.clip(z-0.5*(lo+hi),0.0,1.0)

def smooth_obj_from_corr(corr,alpha):
    m=float(np.max(corr))
    a=alpha*(corr-m)
    a-=a.max()
    return m + float((a.max() + np.log(np.exp(a).sum()))/alpha)

def grad_from_weights(h,n,e):
    g=np.zeros(n,dtype=float)
    mid=n-1
    w=e[mid]
    if w != 0.0:
        g += w*(1.0-2.0*h)
    for k in range(1,n):
        wp=e[mid+k]; wm=e[mid-k]
        if wp == 0.0 and wm == 0.0:
            continue
        l=n-k
        # slices

        g[k:] += wm*(1.0 - h[:l]) - wp*h[:l]
    return g

import numpy as np
import time
from scipy.optimize import minimize

_SL_N = 0
_SL_ALPHA = 1.0
_SL_ORIENT = 1


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(z, n):
    target = 0.5 * float(n)
    z = np.asarray(z, dtype=float)
    h = np.clip(z, 0.0, 1.0)
    if abs(float(h.sum()) - target) <= 1e-12:
        return h
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        h2 = np.clip(z - tau, 0.0, 1.0)
        if h2.sum() > target:
            lo = tau
        else:
            hi = tau
    return np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_obj_from_corr(corr, alpha):
    m = float(np.max(corr))
    a = alpha * (corr - m)
    a -= a.max()
    return m + float((a.max() + np.log(np.exp(a).sum())) / alpha)


def smooth_obj_fast(h, n, alpha):
    return smooth_obj_from_corr(np.correlate(h, 1.0 - h, mode='full'), alpha)


def compute_overlap_orient(h, n, orient):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if orient > 0:
        return corr
    return corr[::-1]


def grad_from_weights(h, n, e):
    g = np.zeros(n, dtype=float)
    mid = n - 1
    w = e[mid]
    if w != 0.0:
        g += w * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = e[mid + k]
        wm = e[mid - k]
        if wp == 0.0 and wm == 0.0:
            continue
        l = n - k
        g[:l] += wp * (1.0 - h[k:]) - wm * h[k:]
        g[k:] += wm * (1.0 - h[:l]) - wp * h[:l]
    return g


def detect_orientation(n):
    i = np.arange(n, dtype=float)
    h = (np.sin(1.23 * i + 0.45) > -0.43).astype(float)
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    mid = n - 1
    best = 1
    best_err = 1e30
    for orient in (1, -1):
        O = corr if orient > 0 else corr[::-1]
        err = 0.0
        for d in (1, 2, 5, 10):
            if d < n:
                op = np.dot(h[:n - d], b[d:])
                on = np.dot(h[d:], b[:n - d])
                err += abs(float(O[mid + d] - op)) + abs(float(O[mid - d] - on))
        if err < best_err:
            best_err = err
            best = orient
    return best


def continuous_polish(h, n, deadline, orient, max_iter=250, polish_seed=1):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(polish_seed)
    if n > 350:
        alphas = (30.0, 300.0, 3000.0)
        max_iter = min(max_iter, 80)
    else:
        alphas = (30.0, 150.0, 700.0, 3500.0, 15000.0)
    step = 0.8
    for alpha in alphas:
        if time.time() >= deadline:
            break
        step = 0.8
        no_accept = 0
        since_best = 0
        for _ in range(max_iter):
            if time.time() >= deadline:
                break
            O = compute_overlap_orient(h, n, orient)
            m = float(O.max())
            a = alpha * (O - m)
            a -= a.max()
            S = m + float((a.max() + np.log(np.exp(a).sum())) / alpha)
            e = np.exp(a)
            e /= e.sum()
            g = grad_from_weights(h, n, e)
            c_cur = compute_c5(h, n)
            accepted = False
            for t in (1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.025, 0.0125):
                trial = project_feasible(h - step * t * g, n)
                S2 = smooth_obj_fast(trial, n, alpha)
                if S2 <= S + 1e-8:
                    c2 = compute_c5(trial, n)
                    if c2 < c_cur - 1e-13 or (c2 <= c_cur + 1e-12 and S2 < S - 1e-12):
                        h = trial
                        if c2 < best_c - 1e-12:
                            best_c = c2
                            best_h = h.copy()
                            since_best = 0
                        else:
                            since_best += 1
                        accepted = True
                        break
            if not accepted:
                step *= 0.5
                no_accept += 1
                if step < 1e-5 or no_accept > 12:
                    break
            if since_best > 80:
                h = project_feasible(h + 0.02 * rng.normal(size=n), n)
                c2 = compute_c5(h, n)
                if c2 < best_c:
                    best_c = c2
                    best_h = h.copy()
                    since_best = 0
                else:
                    since_best = 0
                step = 0.8
                no_accept = 0
    return best_h, best_c


def slsqp_obj(x):
    return smooth_obj_from_corr(np.correlate(x, 1.0 - x, mode='full'), _SL_ALPHA)


def slsqp_jac(x):
    O = compute_overlap_orient(x, _SL_N, _SL_ORIENT)
    m = float(O.max())
    a = _SL_ALPHA * (O - m)
    a -= a.max()
    e = np.exp(a)
    e /= e.sum()
    return grad_from_weights(x, _SL_N, e)


def slsqp_eq(x):
    return float(x.sum()) - 0.5 * _SL_N


def slsqp_improve(h, n, deadline, orient, best_h, best_c):
    global _SL_N, _SL_ALPHA, _SL_ORIENT
    _SL_ORIENT = orient
    cur = project_feasible(h, n)
    alphas = (300.0, 1500.0, 8000.0) if n <= 350 else (200.0, 1000.0)
    maxit = 220 if n <= 350 else 100
    for alpha in alphas:
        if time.time() >= deadline:
            break
        _SL_N = n
        _SL_ALPHA = alpha
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_eq}]
        try:
            res = minimize(slsqp_obj, cur, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=slsqp_jac,
                           options={'maxiter': maxit, 'ftol': 1e-12, 'disp': False})
            cand = project_feasible(res.x, n)
        except Exception:
            cand = cur
        c_cur = compute_c5(cur, n)
        c_cand = compute_c5(cand, n)
        if c_cand < best_c:
            best_c = c_cand
            best_h = cand.copy()
        if c_cand <= c_cur + 1e-12:
            cur = cand
        if time.time() >= deadline:
            break
    return best_h, best_c


def binarize(h, n):
    n = int(n)
    out = np.zeros(n, dtype=float)
    if n <= 0:
        return out
    order = np.argsort(-h)
    out[order[:n // 2]] = 1.0
    return out


def random_starts(n, seed):
    r = np.random.RandomState(seed)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    out = []
    out.append(binarize(np.sin(np.pi * x) ** 2, n))
    out.append(binarize(0.55 * np.cos(2.0 * np.pi * x + r.randn()) +
                        0.45 * np.cos(4.0 * np.pi * x + r.randn()), n))
    y = np.cumsum(r.randn(n))
    y = y - np.linspace(0.0, 1.0, n) * (y[-1] - y[0])
    out.append(binarize(y + 0.03 * r.randn(n), n))
    out.append(binarize(0.40 * np.cos(6.0 * np.pi * x + r.randn()) +
                        0.04 * r.randn(n), n))
    return out


def perturb_binary(h, n, rng, k):
    h = np.array(h, dtype=float).copy()
    ones = np.flatnonzero(h == 1.0)
    zeros = np.flatnonzero(h == 0.0)
    for _ in range(k):
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[rng.randint(ones.size)]
        j = zeros[rng.randint(zeros.size)]
        h[i] = 0.0
        h[j] = 1.0
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
    return h


def binary_steepest(h, n, deadline, orient, max_sweeps=200):
    h = np.array(h, dtype=float).copy()
    n = int(n)
    if (h == 1.0).sum() != n // 2:
        h = binarize(h, n)
    m = 2 * n - 1
    mid = n - 1
    c = compute_overlap_orient(h, n, orient).copy()
    cur = float(c.max())
    best_c = cur
    best_h = h.copy()
    dx = 2.0 / float(n)
    sweeps = 0
    while time.time() < deadline and sweeps < max_sweeps:
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        b = 1.0 - h
        hr = h[::-1]
        J = zeros.size
        ZB = np.zeros((J, m), dtype=float)
        for idx, j in enumerate(zeros):
            l1 = n - j
            if l1 > 0:
                ZB[idx, mid - j:mid - j + l1] += b[:l1]
            ZB[idx, j:j + n] -= hr
        best_val = cur
        best_i = -1
        best_j = -1
        row_idx = np.arange(J)
        for ii in range(ones.size):
            if time.time() > deadline:
                break
            i = ones[ii]
            base = c.copy()
            l2 = n - i
            if l2 > 0:
                base[mid - i:mid - i + l2] -= b[:l2]
            base[i:i + n] += hr
            V = ZB + base
            if mid > 0:
                left = V[:, :mid].max(axis=1)
            else:
                left = np.full(J, -1e30)
            if mid + 1 < m:
                right = V[:, mid + 1:].max(axis=1)
            else:
                right = np.full(J, -1e30)
            no_aff = np.maximum(left, right)
            v0 = V[:, mid]
            p1 = mid + (zeros - i)
            p2 = mid - (zeros - i)
            exact = np.maximum(no_aff, v0 - 2.0)
            exact = np.maximum(exact, V[row_idx, p1] + 1.0)
            exact = np.maximum(exact, V[row_idx, p2] + 1.0)
            loc = int(np.argmin(exact))
            val = float(exact[loc])
            if val < best_val - 1e-9:
                best_val = val
                best_i = int(i)
                best_j = int(zeros[loc])
        if best_i >= 0:
            i = best_i
            j = best_j
            l1 = n - j
            if l1 > 0:
                c[mid - j:mid - j + l1] += b[:l1]
            l2 = n - i
            if l2 > 0:
                c[mid - i:mid - i + l2] -= b[:l2]
            c[i:i + n] += hr
            c[j:j + n] -= hr
            c[mid] -= 2.0
            c[mid + (j - i)] += 1.0
            c[mid + (i - j)] += 1.0
            h[i] = 0.0
            h[j] = 1.0
            b[i] = 1.0
            b[j] = 0.0
            c = np.rint(c)
            cur = float(c.max())
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            sweeps += 1
        else:
            break
    return best_h, float(best_c * dx)


def resample_step(h, n_target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == n_target:
        return h.copy()
    if n == 0:
        return np.zeros(n_target)
    if n == 1:
        return np.full(n_target, h[0])
    xold = np.linspace(0.0, 1.0, n, endpoint=False)
    xnew = np.linspace(0.0, 1.0, n_target, endpoint=False)
    return np.interp(xnew, xold, h)
