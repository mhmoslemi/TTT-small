import time

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def obj_smooth(x, n, alpha, k):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    if k >= len(c):
        ck = c
    else:
        part = np.partition(c, len(c) - k)
        thresh = part[-k]
        ck = c[c >= thresh]
    m = ck.max()
    s = m + np.log(np.sum(np.exp(ck - m))) / alpha
    return float(s * dx)

def make_cons(n):
    fun = lambda x: x.sum() - n / 2.0
    jac = lambda x: np.ones(n)
    return [{'type': 'eq', 'fun': fun, 'jac': jac}]

    bounds = [(0.0, 1.0)] * n

        res = minimize(obj_smooth, h, args=(n, alpha, k), method='SLSQP',
                       bounds=bounds, constraints=make_cons(n),

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_topk(x, n, alpha, k):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    if k >= len(c):
        ck = c
    else:
        part = np.partition(c, len(c) - k)
        thresh = part[-k]
        ck = c[c >= thresh]
    m = ck.max()
    s = m + np.log(np.sum(np.exp(ck - m))) / alpha
    return float(s * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, k, maxiter=2000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_topk, h, args=(n, alpha, k), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def cd_analytical(h, n, deadline, rng, max_passes=300, grid=30):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    N = n
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(N)
        rng.shuffle(indices)
        for p in indices:
            if h[p] <= 1e-10 or h[p] >= 1.0 - 1e-10:
                continue
            m_lo = N - 1 - p
            m_hi = max(N - 1 + p, 2 * N - 2 - p)
            m_hi = min(m_hi, 2 * N - 2)
            s = np.zeros(m_hi - m_lo + 1)
            ms = np.arange(m_lo, m_hi + 1)
            idx1 = ms + p - N + 1
            valid1 = (idx1 >= 0) & (idx1 < N)
            c1_range = (ms >= N - 1 - p) & (ms <= N - 1 + p)
            if np.any(c1_range & valid1):
                s[c1_range & valid1] += 1.0 - h[idx1[c1_range & valid1]]
            idx2 = ms + p - N + 1
            c2_range = (ms >= N - 1 - p) & (ms <= 2 * N - 2 - p)
            if np.any(c2_range & valid1):
                s[c2_range & valid1] -= h[idx2[c2_range & valid1]]
            um = 0.0
            if m_lo > 0:
                um = float(np.max(c[:m_lo]))
            if m_hi < 2 * N - 2:
                um = max(um, float(np.max(c[m_hi + 1:])))
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[m_lo:m_hi + 1].copy() + d * s
                ci = (ms == (N - 1))
                if np.any(ci):
                    new_c[ci] = c[N - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-11:
                d = best_d
                affected = c[m_lo:m_hi + 1].copy() + d * s
                ci = (ms == (N - 1))
                if np.any(ci):
                    affected[ci] = c[N - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                c[m_lo:m_hi + 1] = affected
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[m_lo:m_hi + 1])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / N)

def refine(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    t_remain = lambda: deadline - time.time()
    for k in [8, 15, 30, 60]:
        if t_remain() < 30:
            break
        for alpha in [5000, 50000, 200000]:
            if t_remain() < 20:
                break
            h_s = slsqp_opt(h, n, alpha, k, maxiter=1200)
            c_s = compute_c5(h_s, n)
            if c_s < best_c - 1e-12:
                h, best_c = h_s, c_s
    if t_remain() > 25:
        t_cd = min(60, t_remain() - 15)
        if t_cd > 15:
            h_cd, c_cd = cd_analytical(h, n, time.time() + t_cd, rng,
                                       max_passes=250, grid=25)
            h_cd = project_feasible(h_cd, n)
            c_cd = compute_c5(h_cd, n)
            if c_cd < best_c:
                h, best_c = h_cd, c_cd
    if t_remain() > 20:
        h_f = slsqp_opt(h, n, alpha=500000, k=8, maxiter=3000, ftol=1e-18)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            h, best_c = h_f, c_f
    if t_remain() > 15:
        h_cd2, c_cd2 = cd_analytical(h, n, time.time() + min(30, t_remain() - 5),
                                     rng, max_passes=150, grid=35)
        h_cd2 = project_feasible(h_cd2, n)
        c_cd2 = compute_c5(h_cd2, n)
        if c_cd2 < best_c:
            h, best_c = h_cd2, c_cd2
    return h, best_c

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
