def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    """Compute the change in correlate vector when swapping h[i] 0->1 and h[j] 1->0"""
    delta = np.zeros(2 * n - 1)

def sa_binary(h_bin, n, deadline, seed=42, t_start=0.02, t_end=0.00001, patience=25000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    no_improve = 0
    step = 0
    T = t_start
    # Cooling: reduce T by factor each step
    # We'll use a simple exponential cooldown with adaptive rate
    cool_per_step = 0.99997  # Very slow cooling
    
    while time.time() < deadline:
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = float(new_corr.max()) * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        step += 1
        T *= cool_per_step
        if no_improve > patience:
            nk = 3 + rng.randint(5)
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]

            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = float(corr.max()) * dx
            no_improve = 0
            T = max(T * 3.0, t_start * 0.5)

def upsample(h, n_old, n_new):

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)

def make_structured(n, rng):
    """Create structured initial binary patterns"""
    h = np.zeros(n)
    target = n // 2
    method = rng.randint(4)
    if method == 0:
        # Arithmetic progression
        step = n / target
        positions = np.round(np.arange(target) * step).astype(int)
        positions = np.mod(positions, n)
        h[np.unique(positions)][:min(target, len(np.unique(positions)))] = 1.0
        if h.sum() != target:
            idx = np.where(h == 0)[0]
            need = target - int(h.sum())
            if need > 0 and len(idx) >= need:
                h[idx[:need]] = 1.0
    elif method == 1:
        # Beatty sequence (related to golden ratio)
        alpha = 1.6180339887498949
        positions = np.floor(np.arange(1, target + 1) * alpha).astype(int)
        positions = np.mod(positions, n)
        unique_pos = np.unique(positions)
        h[unique_pos[:target]] = 1.0
        if h.sum() != target:
            zero_idx = np.where(h == 0)[0]
            one_idx = np.where(h == 1)[0]
            need = target - int(h.sum())
            if need > 0:
                h[zero_idx[:need]] = 1.0
            elif need < 0:
                h[one_idx[:abs(need)]] = 0.0
    elif method == 2:
        # Alternating blocks
        block = max(1, n // (2 * target // 3 + 1))
        pattern = np.tile([1, 0], max(1, n // block))[:n]
        h = pattern.astype(float)
        diff = target - int(h.sum())
        if diff > 0:
            idx = np.where(h == 0)[0]
            h[idx[:diff]] = 1.0
        elif diff < 0:
            idx = np.where(h == 1)[0]
            h[idx[:abs(diff)]] = 0.0
    else:
        # Random
        h[rng.choice(n, target, replace=False)] = 1.0
    return h

def continuous_refine(h, n, deadline, seed=42):
    """Refine using a smoothed min-max objective with L-BFGS-B"""
    dx = 2.0 / n
    target = n / 2.0
    
    def objective(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1 - x, mode='full')
        m = c.max()
        # Smooth max: log-sum-exp
        alpha = 500
        shifted = (c - m) / alpha
        smax = m + np.log(np.exp(shifted).sum()) / alpha
        # Penalty for constraint violation
        s = x.sum()
        penalty = 100.0 * (s - target) ** 2
        return smax * dx + penalty
    
    def objective_wrapper(x):
        return objective(x)
    
    bounds = [(0.0, 1.0)] * n
    x0 = np.clip(h, 0, 1)
    # Scale to approximately satisfy constraint
    if x0.sum() > 0:
        x0 = x0 * (target / x0.sum())
    x0 = np.clip(x0, 0, 1)
    
    res = minimize(objective_wrapper, x0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 500, 'ftol': 1e-12})
    return project_feasible(res.x, n)
