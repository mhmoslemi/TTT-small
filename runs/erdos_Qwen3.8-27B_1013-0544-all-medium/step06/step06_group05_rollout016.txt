import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n):
    """Project to binary with exactly n/2 ones (top n/2 values)."""
    thresh = np.partition(h, n - n//2)[n - n//2]
    h_bin = (h >= thresh).astype(float)
    # Fix count if needed
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        excess = np.where(h_bin == 1)[0]
        # Remove excess ones with smallest h values
        order = np.argsort(h[excess])[::-1]
        for idx in order[:n_ones - n//2]:
            h_bin[excess[idx]] = 0.0
    elif n_ones < n // 2:
        deficit = np.where(h_bin == 0)[0]
        order = np.argsort(h[deficit])[::-1]
        for idx in order[:n//2 - n_ones]:
            h_bin[deficit[idx]] = 1.0
    return h_bin

def correlation_vector(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def local_search_binary(h, n, deadline, tabu_len=50, sa_temp=0.5, max_iters=500000):
    """Simulated annealing with tabu search on binary solutions."""
    dx = 2.0 / n
    cur = h.copy()
    corr = correlation_vector(cur, n)
    cur_max = corr.max()
    best_h = cur.copy()
    best_max = cur_max
    
    tabu = {}  # position -> iteration when it will expire
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and sa_temp > 0.01:
            sa_temp *= 0.95
        
        # Choose a zero to turn to one
        zeros = np.where(cur == 0)[0]
        ones = np.where(cur == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        if it - tabu.get(i, -1000) < tabu_len or it - tabu.get(j, -1000) < tabu_len:
            continue
        
        # Try the move
        cur[i] = 1.0
        cur[j] = 0.0
        new_corr = correlation_vector(cur, n)
        new_max = new_corr.max()
        
        if new_max < cur_max or rng.rand() < np.exp(-(new_max - cur_max) / max(sa_temp, 1e-10)):
            # Accept
            corr = new_corr
            cur_max = new_max
            tabu[i] = it
            tabu[j] = it
            if new_max < best_max:
                best_max = new_max
                best_h = cur.copy()
        else:
            # Revert
            cur[i] = 0.0
            cur[j] = 1.0
    
    return best_h, float(best_max * dx)
