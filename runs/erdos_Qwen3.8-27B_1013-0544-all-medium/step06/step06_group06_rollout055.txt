The key is to generate GOOD candidates efficiently.

**Candidate generation heuristic**:
- Find the top T lags (those with highest correlation)
- For each top lag k, find the positions that contribute most to the overlap at that lag
- A position p contributes to lag k if h[p]=1 and h[p±k]=0 (or vice versa)
- Generate candidates by swapping "high contribution" ones with "low impact" zeros

But with numpy, computing one correlation is O(n²) ≈ 62500 ops (very fast, maybe 10μs). So checking, say, 1000 candidate swaps would take about 10ms. That's fine.

Let me design a cleaner algorithm:

```
Algorithm:
1. Start with initial h (from parent or Thue-Morse)
2. Repeat until time expires:
   a. Compute corr = correlate(h, 1-h, 'full')
   b. cur_max = max(corr)
   c. Identify top T lags (T = ~20-30)
   d. Generate candidate swaps:
      - For each "hot" one position (a 1 that appears in many top-lag overlaps):
        - Try swapping with each "cold" zero position (a 0 that appears in few top-lag overlaps)
      - Or: just try random swaps and evaluate
   e. Evaluate top K candidates (compute new corr for each)
   f. If best candidate improves, accept it
   g. If stuck for S iterations, do a random kick (accept a worse move)
3. Return best h found
```

For step (d), the "hot" positions are those that appear in the overlap for the top lags. Specifically, for lag k (relative), the overlap is:
sum_{i=0}^{n-1-|k|} h[i+|k|] * (1-h[i])

A position p is "hot" for lag k if h[p]=1 and h[p-k]=0 (p is the "1" part) OR h[p]=0 and h[p+k]=1 (p is the "0" part... wait, we can only swap 1s with 0s).

So: position p with h[p]=1 is "hot" for lag k if h[p-k]=0 (it's a "1-0" pair at lag k).
Position q with h[q]=0 is "cold" if for most top lags k, h[q+k]≠1 (swapping it to 1 won't create new overlaps).

This gives a natural candidate set: swap hot ones with cold zeros.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def gen_thue_morse(n):
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def to_binary(t, n, rng):
    h = t.astype(float)
    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:num_ones - target]]] = 0
    elif num_ones < target:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:target - num_ones]]] = 1
    return h

def local_search(h, n, deadline, rng, top_k=15):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    n_lags = len(corr)
    center = n - 1
    stall = 0
    kick_count = 0

    while time.time() < deadline:
        top_lags = np.argsort(corr)[-top_k:]
        one_hot = np.zeros(len(ones), dtype=np.int32)
        zero_col = np.zeros(len(zeros), dtype=np.int32)

        for li in top_lags:
            lag = li - center
            if lag >= 0:
                valid = range(n - lag)
                for i in valid:
                    if h[i + lag] > 0.5 and h[i] < 0.5:
                        oi = int(np.searchsorted(ones, i + lag))
                        if oi < len(ones) and ones[oi] == i + lag:
                            one_hot[oi] += 1
                        zi = int(np.searchsorted(zeros, i))
                        if zi < len(zeros) and zeros[zi] == i:
                            zero_col[zi] += 1
            else:
                lag2 = -lag
                for i in range(lag2, n):
                    if h[i] > 0.5 and h[i - lag2] < 0.5:
                        oi = int(np.searchsorted(ones, i))
                        if oi < len(ones) and ones[oi] == i:
                            one_hot[oi] += 1
                        zi = int(np.searchsorted(zeros, i - lag2))
                        if zi < len(zeros) and zeros[zi] == i - lag2:
                            zero_col[zi] += 1

        improved = False
        if len(ones) > 0 and len(zeros) > 0:
            n_try = min(6, len(ones))
            n_tryz = min(6, len(zeros))
            top_o = np.argsort(one_hot)[::-1][:n_try]
            bot_z = np.argsort(zero_col)[:n_tryz]
            best_gain = cur_c
            best_pair = None
            for oi in top_o:
                for zi in bot_z:
                    ip, jp = zeros[zi], ones[oi]
                    h[jp] = 0.0
                    h[ip] = 1.0
                    nc = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                    h[jp] = 1.0
                    h[ip] = 0.0
                    if nc < best_gain - 1e-15:
                        best_gain = nc
                        best_pair = (ip, jp)
            if best_pair is not None:
                ip, jp = best_pair
                h[jp] = 0.0
                h[ip] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c = float(corr.max() * dx)
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            stall += 1
            if stall > 25 and kick_count < 50:
                kick_count += 1
                ks = rng.randint(1, 4)

                ko = rng.choice(len(ones), ks, replace=False)

    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=200000.0, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        cs = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.exp(cs).sum()) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_h = h_start.copy()

    # Phase 1: Local search from parent (35%)
    t1 = t0 + 0.35 * (budget_s - 5)
    hb, cb = local_search(best_h.copy(), n, t1, rng, top_k=15)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Thue-Morse start (15%)
    t2 = t0 + 0.50 * (budget_s - 5)
    if time.time() < t2 - 10:
        tm = to_binary(gen_thue_morse(n), n, rng)
        hb, cb = local_search(tm, n, t2 - 5, rng, top_k=15)

    t3 = t0 + 0.65 * (budget_s - 5)

        for al in [50000.0, 300000.0, 1000000.0]:
            if time.time() > t3 - 5:
                break
            hc = slsqp_polish(hc, n, t3 - 3, al)
            hb = to_binary((hc >= 0.5).astype(np.int8), n, rng)
            c = compute_c5(hb, n)
            if c < best_c:
                best_c, best_h = c, hb.copy()

                    hp[zrs[rng.randint(len(zrs))]] = 1.0
                    ons = np.where(hp == 1)[0]
                    zrs = np.where(hp == 0)[0]
            ts = min(time.time() + tp, t4 - 10)
            hb, cb = local_search(hp, n, ts, rng, top_k=15)
            if cb < best_c:
                best_c, best_h = cb, hb.copy()

    # Phase 5: Final polish (10%)

        hb, cb = local_search(best_h.copy(), n, dl - 2, rng, top_k=10)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    return best_h, best_c, n
