```python
    # (where terms are 0 if index is out of range)
    
    # Let me compute d for each lag
    d = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        # Contribution from h[j]: (1-h[j+k]) if j+k valid, minus h[j-k] if j-k valid
        dj = 0.0
        if k >= 0:
            if j + k < n:
                dj += (1.0 - h[j+k])
            if j - k >= 0:
                dj -= h[j-k]
        else:
            kk = -k
            if j - kk >= 0:  # j + k >= 0
                dj += (1.0 - h[j+k])
            if j + kk < n:  # j - k < n
                dj -= h[j-k]
        
        di = 0.0
        if k >= 0:
            if i + k < n:
                di += (1.0 - h[i+k])
            if i - k >= 0:
                di -= h[i-k]
        else:
            kk = -k
            if i - kk >= 0:
                di += (1.0 - h[i+k])
            if i + kk < n:
                di -= h[i-k]
        
        d[m] = dj - di
    
    # Now f(delta) = max_m (c[m] + delta * d[m])
    # We want to minimize f(delta) over feasible delta range
    # Feasibility: h[i] + delta >= 0, h[i] + delta <= 1, h[j] - delta >= 0, h[j] - delta <= 1
    # So delta in [max(-h[i], h[j]-1), min(1-h[i], h[j])]
    
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    
    if delta_max <= delta_min:
        return 0.0, c.max()
    
    # f(delta) = max_m (c[m] + delta * d[m])
    # This is convex. Find minimum.
    # At the minimum, either delta = delta_min, delta = delta_max,
    # or two of the lines cross at the minimum.
    
    # Find the minimum by checking breakpoints
    # A breakpoint occurs where c[a] + delta*d[a] = c[b] + delta*d[b]
    # delta = (c[b] - c[a]) / (d[a] - d[b])
    
    # For efficiency, just check:
    # 1. delta_min, delta_max
    # 2. For each pair of lines (a,b) with d[a] != d[b], the intersection point
    #    delta = (c[b] - c[a]) / (d[a] - d[b])
    #    if delta_min <= delta <= delta_max, evaluate f(delta)
    
    # But this is O((2n-1)²) which is too slow for n=300.
    # Instead, use the fact that we only need the minimum of a convex piecewise-linear function.
    # We can use a ternary search or bisection on the derivative.
    
    # The derivative of f(delta) is d[argmax_m (c[m] + delta*d[m])]
    # which is a step function (non-decreasing as delta increases, since we always pick the max)
    # Wait, that's not right. The derivative is one of the d[m] values, and it's non-decreasing
    # only if the d values are ordered... which they're not necessarily.
    
    # Actually, for f(delta) = max_m (c[m] + delta*d[m]):
    # f'(delta) = d[m*(delta)] where m*(delta) = argmax_m (c[m] + delta*d[m])
    # As delta increases, the argmax can change. The derivative is a step function
    # that jumps between d values.
    
    # For minimization: we want f'(delta) = 0, i.e., d[m*(delta)] = 0
    # If d[m*(delta)] > 0, we want to decrease delta
    # If d[m*(delta)] < 0, we want to increase delta
    
    # So the optimal delta is where the argmax switches from a negative d to a positive d.
    # Equivalently, it's at the intersection of the "upper envelope" lines.
    
    # Simple approach: ternary search on f(delta)
    # f is convex (max of linear functions), so ternary search works.
    
    def f_of_delta(dlt):
        return np.max(c + dlt * d)
    
    # Ternary search
    a, b = delta_min, delta_max
    for _ in range(60):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        if f_of_delta(m1) < f_of_delta(m2):
            b = m2
        else:
            a = m1
    
    delta_opt = (a + b) / 2
    return delta_opt, f_of_delta(delta_opt)
```

Actually, even simpler: just use the gradient direction. The steepest descent direction is -grad. Project this onto the feasible set (sum constraint + box constraint). This is equivalent to finding the best pair swap in one go.

Let me think about this differently. The problem is:
minimize f(h) = max_k c_k
subject to: h in [0,1]^n, sum(h) = n/2

The subdifferential of f at h is conv{dc/dh : c_k = max}. So a subgradient is:
g = sum_{k: c_k = max} lambda_k * dc_k/dh, where lambda_k >= 0, sum lambda_k = 1

For the smooth-max, we use all k with weights w_k.

The projected gradient step would be:
h_new = project(h - lr * g, [0,1]^n, sum = n/2)

The projection onto {x in [0,1]^n : sum(x) = n/2} is:
- Start with x = h - lr * g
- Solve: find alpha such that sum(clip(x + alpha, 0, 1)) = n/2
- Return clip(x + alpha, 0, 1)

This is a 1D root-finding problem, solvable by bisection.

The key advantage of this approach: it naturally handles the sum constraint and box constraints, and it's O(n) per step (plus O(n²) for the gradient).

```python
def project_sum_box(x, target_sum):
    """Project x onto {y in [0,1]^n : sum(y) = target_sum}."""
    # Binary search for alpha: sum(clip(x + alpha, 0, 1)) = target_sum
    lo = target_sum - x.sum() - x.max()  # lower bound for alpha... actually
    
    # Simpler: binary search alpha
    f = lambda a: np.clip(x + a, 0, 1).sum() - target_sum
    # f is non-decreasing in a
    lo = min(0, target_sum - x.sum()) - 1  # ensure f(lo) < 0
    hi = max(0, target_sum - x.sum()) + 1  # ensure f(hi) > 0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
    
    return np.clip(x + (lo + hi) / 2, 0, 1)
```

Wait, this projection isn't quite right. The projection onto the intersection of [0,1]^n and {sum = target} is not simply a uniform shift. It's the Euclidean projection, which requires:
- If x is already in [0,1] and sum(x) = target, return x
- Otherwise, find alpha such that sum(clip(x + alpha, 0, 1)) = target

Actually, for the constraint sum(y) = target with 0 <= y_i <= 1, the Euclidean projection IS a uniform shift followed by clipping, because the constraint matrix is all ones (the null space is the set of vectors summing to 0, and the projection onto the affine set {sum = target} is a uniform shift).

Wait, more carefully: the projection of x onto {y : sum(y) = target} (without box constraints) is x + (target - sum(x))/n * 1 (uniform shift). With box constraints, it's:
- Find alpha such that sum(clip(x + alpha, 0, 1)) = target
- Return clip(x + alpha, 0, 1)

This is correct because the feasible set is the intersection of an affine space (sum = target) and a box ([0,1]^n), and the projection onto the affine space is a uniform shift. The projected gradient method applies the shift and then clips, which is the correct handling of the box constraint in the projected gradient method.

**Parametrization idea**: Instead of optimizing h directly, optimize the "transition points" of a step function. If h has k distinct blocks, we need 2k parameters (k block values + k-1 transition points). This dramatically reduces the dimensionality.

For example, if we assume h is a step function with k blocks:
- Parameters: (t_1, ..., t_{k-1}, v_1, ..., v_k) where t_i are transition points and v_i are block values
- h(x) = v_i for t_{i-1} <= x < t_i

The constraint sum(h)*dx = 1 becomes: sum(v_i * (t_i - t_{i-1})) = 1

This is a lower-dimensional problem (2k-1 parameters instead of n), and the objective is smooth (the correlation is a piecewise smooth function of the parameters).

However, this has the disadvantage that we need to know k in advance, and the correlations have discontinuities when a transition point crosses a sample point.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return m + (mx + np.log(np.exp(s - mx).sum())) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_smooth(h, n, alpha, mu, maxiter=200):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        val = m + (mx + np.log(np.exp(s - mx).sum())) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    def jac(x):
        g = smooth_max_grad(x, n, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13,
                               'maxfun': maxiter * 12, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_gradient(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 50000, 100000]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 1.0 / (alpha * n / 2)
        for it in range(500):
            if time.time() > deadline:
                break
            g = smooth_max_grad(h, n, alpha)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                break
            step = lr * h
            improved = False
            for _ in range(30):
                h_new = project_feasible(h - step * g / g_norm * g_norm, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    h = h_new
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                step *= 0.5
            if not improved:
                lr *= 0.3
                if lr < 1e-14:
                    break
            else:
                lr = min(lr * 1.2, 1.0 / (alpha * n / 2) * 4)
    return best_h, best_c

def pair_line_search(h, i, j, n):
    c = compute_c5_vec(h, n)
    d = np.zeros(2 * n - 1)
    for m_idx in range(2 * n - 1):
        k = m_idx - (n - 1)
        dj = 0.0
        if j + k >= 0 and j + k < n:
            dj += (1.0 - h[j + k])
        if j - k >= 0 and j - k < n:
            dj -= h[j - k]
        di = 0.0
        if i + k >= 0 and i + k < n:
            di += (1.0 - h[i + k])
        if i - k >= 0 and i - k < n:
            di -= h[i - k]
        d[m_idx] = dj - di
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    if delta_max <= delta_min + 1e-15:
        return 0.0, c.max()
    def f_of_delta(dlt):
        return np.max(c + dlt * d)
    a, b = delta_min, delta_max
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if f_of_delta(m1) < f_of_delta(m2):
            b = m2
        else:
            a = m1
    delta_best = 0.0
    c_best = compute_c5(h, n)
    for dlt in [delta_min, (a + b) / 2.0, delta_max, 0.0]:
        if dlt < delta_min or dlt > delta_max:
            continue
        c_test = f_of_delta(dlt)
        if c_test < c_best - 1e-14:
            c_best = c_test
            delta_best = dlt
    return delta_best, c_best

def smart_pair_search(h, n, deadline, top_k=200):
    best_c = compute_c5(h, n)
    for outer in range(2000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, 500)
        diffs = np.zeros((n, n))
        for i in range(n):
            diffs[i, :] = g[i] - g
        np.fill_diagonal(diffs, -np.inf)
        flat = diffs.flatten()
        top_idx = np.argpartition(flat, -top_k)[-top_k:]
        best_delta = 0.0
        best_new_c = best_c
        best_pair = None
        for fi in top_idx:
            i = fi // n
            j = fi % n
            delta, c_new = pair_line_search(h, i, j, n)
            if c_new < best_new_c - 1e-15:
                best_new_c = c_new
                best_delta = delta
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] += best_delta
            h[j] -= best_delta
            h = project_feasible(h, n)
            best_c = compute_c5(h, n)
        else:
            break
    return h, best_c

def try_n_value(h_best, n_best, n_try, deadline):
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_try)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_try)
    t_local = min(time.time() + 30, deadline - 5)

        h_new = lbfgs_smooth(h_interp, n_try, alpha, 10.0, maxiter=150)

    if time.time() < deadline - 10:
        h_interp, _ = smart_pair_search(h_interp, n_try, deadline - 5, top_k=100)
    c_try = compute_c5(h_interp, n_try)
    return h_interp, c_try, n_try
```
