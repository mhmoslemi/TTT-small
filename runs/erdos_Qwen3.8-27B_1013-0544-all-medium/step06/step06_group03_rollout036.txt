    best_h, best_c = h, c
    other_ns = [150, 200, 250, 350, 400, 500]
    for n_try in other_ns:
        if time.time() > deadline - 90:
            break
        if n_try == n_init:
            continue
        x_b = np.linspace(0.0, 2.0, len(best_h))
        x_n = np.linspace(0.0, 2.0, n_try)
        h_start = np.interp(x_n, x_b, best_h)
        h_start = project_feasible(h_start, n_try)
        alloc = min(200, (deadline - time.time()) * 0.3)
        if alloc < 50:
            break
        h_try, c_try = optimize_at_n(h_start, n_try, time.time() + alloc, seed=seed + n_try)
        if c_try < best_c:
            best_c = c_try; best_h = h_try
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        h_new, c_new = pair_cd(best_h, n, min(deadline - 5, time.time() + 80),
                                max_sweeps=5, n_steps=12, seed=seed + 777)
        if c_new < best_c:
            best_c = c_new; best_h = h_new
        h_new, c_new = fine_polish(best_h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new; best_h = h_new
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
    if best_h is None:
        best_h = project_feasible(np.full(200, 0.5), 200)
        best_c = compute_c5(best_h, 200)
    return best_h, best_c, len(best_h)
