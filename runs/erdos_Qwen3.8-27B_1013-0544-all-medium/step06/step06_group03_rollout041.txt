def project_sum_bounds(h, n):
    """Project h onto {x : 0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Bisection for lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2, 0, 1)
    # Final correction for numerical precision
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - result
            mask = room > 1e-12
            result[mask] += diff / mask.sum() * (room[mask] / room[mask].sum())
        else:
            mask = result > 1e-12
            result[mask] += diff / mask.sum() * (result[mask] / result[mask].sum())
    return np.clip(result, 0, 1)

            hp = project_sum_bounds(hp, n)
            alloc = min(55, (deadline - time.time() - 25) * 0.45)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_polish(best_h, n, deadline - 5)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_fin, c_fin = local_search(best_h, n, deadline - 1)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
        best_h = project_sum_bounds(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_sum_bounds(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
