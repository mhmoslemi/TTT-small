def efficient_sa(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.00002, patience=50000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    # Maintain sets of zero and one indices
    zeros = set(i for i in range(n) if h[i] == 0)
    ones = set(i for i in range(n) if h[i] == 1)
    zeros_list = list(zeros)
    ones_list = list(ones)
    
    no_improve = 0
    T = t_start
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    
    while time.time() < deadline:
        # Random selection
        i = zeros_list[rng.randint(len(zeros_list))]
        j = ones_list[rng.randint(len(ones_list))]
        
        # Compute delta efficiently
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c

            # Accept
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            
            # Update index sets
            zeros_list.remove(i)
            ones_list.remove(j)
            ones_list.append(i)
            zeros_list.append(j)
            
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cool_rate
        if no_improve > patience:
            # Restart with perturbation
            for _ in range(rng.randint(2, 8)):
                ii = zeros_list[rng.randint(len(zeros_list))]
                jj = ones_list[rng.randint(len(ones_list))]
                h[ii] = 1.0

    return best_h, best_c

    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    # Phase 1: SLSQP continuation (12%)
    p1 = t0 + 0.12 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > p1 - 2:
            break
        h = slsqp_run(h, n, alpha, 800)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Binary SA with multiple starts (53%)
    p2 = t0 + 0.65 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    starts = [
        (h_bin, seed, 0.45),
        (round_to_binary(best_h + 0.08 * rng.randn(n), n), seed + 100, 0.30),
        (round_to_binary(best_h + 0.2 * rng.randn(n), n), seed + 200, 0.25),
    ]
    sa_total = p2 - t0
    cumulative = 0
    for k, (sh, ss, frac) in enumerate(starts):
        if time.time() > p2 - 2:
            break
        cumulative += sa_total * frac
        end_t = min(t0 + cumulative, p2 - 1)
        if time.time() > end_t - 1:
            end_t = p2 - 1
        hb, cb = sa_binary(sh, n, end_t, ss, 0.03, 0.00001, 45000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 3: SLSQP polish from best binary (15%)
    p3 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p3 - 5:
        hp = project_feasible(best_h + 0.015 * rng.randn(n), n)
        for alpha in [5000, 30000, 150000, 800000]:
            if time.time() > p3 - 2:
                break
            hp = slsqp_run(hp, n, alpha, 2000)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    # Phase 4: Final SA round (10%)
    p4 = dl - 25
    if time.time() < p4 - 5:
        h_bin2 = round_to_binary(best_h, n)
        hb2, cb2 = sa_binary(h_bin2, n, p4, seed + 500, 0.015, 0.00003, 30000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    # Phase 5: Greedy polish
    if time.time() < dl - 4:
        hg, cg = greedy_polish(best_h, n, dl - 3, 30)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()
    # Final SLSQP micro-polish
    if time.time() < dl - 2:
        hp = project_feasible(best_h.copy(), n)
        hp = slsqp_run(hp, n, 500000, 500)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
