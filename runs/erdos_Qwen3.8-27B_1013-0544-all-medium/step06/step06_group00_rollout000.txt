def compute_grad_subgrad(h, n):
    """Compute subgradient of max_k c_k(h) with respect to h."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = int(np.argmax(c))
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        # For shift s >= 0:
        # c_s = sum_{n=0}^{n-1-s} h[n+s] * (1-h[n])
        # d(c_s)/d(h[i]) = (1-h[i-s]) * [i >= s] - h[i+s] * [i <= n-1-s]
        if s == 0:
            grad[:] = 1.0 - h - h  # = 1 - 2h
        else:
            # First term: (1-h[i-s]) for i in [s, n-1]
            grad[s:] += (1 - h[:n-s])
            # Second term: -h[i+s] for i in [0, n-1-s]
            grad[:n-s] -= h[s:]
    else:
        m = -s
        # c_{-m} = sum_{n=m}^{n-1} h[n-m] * (1-h[n])
        # d(c_{-m})/d(h[i]) = (1-h[i+m]) * [i <= n-1-m] - h[i-m] * [i >= m]
        if m == 0:
            grad[:] = 1.0 - h - h
        else:
            # First term: (1-h[i+m]) for i in [0, n-1-m]
            grad[:n-m] += (1 - h[m:])
            # Second term: -h[i-m] for i in [m, n-1]
            grad[m:] -= h[:n-m]
    return grad * dx, s, c[k_star] * dx

    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        if alpha < 1:
            return m
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        result = project_feasible(res.x, n)
    except Exception:
        result = h
    return result
