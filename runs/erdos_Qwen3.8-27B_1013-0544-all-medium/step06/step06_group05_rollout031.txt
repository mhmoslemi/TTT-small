```python
def compute_c5(h, n):

def binarize(h, n):
    """Convert continuous h to binary with exactly n/2 ones"""
    sorted_h = np.sort(h)[::-1]
    thresh = sorted_h[n // 2]
    h_bin = (h >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        # Break ties by keeping the first n/2
        candidates = np.where(h_bin == 1)[0]
        # Remove excess
        for idx in candidates[n_ones - n//2:]:
            h_bin[idx] = 0.0
    elif n_ones < n // 2:
        candidates = np.where(h_bin == 0)[0]
        for idx in candidates[:n//2 - n_ones]:
            h_bin[idx] = 1.0
    return h_bin

def swap_delta(h, i, j, n):

    Returns array of deltas for affected k values and their indices"""

    def safe_h(x):
        if 0 <= x < n:
            return h[x]
        return 0.0

    # h[i-d] nonzero when i-d in [0,n-1]: d in [i-(n-1), i], k in [i, i+n-1]
    # Similarly for j
    
    k_min = max(0, min(n-1-i, i, n-1-j, j))
    k_max = min(2*n-2, max(2*n-2-i, i+n-1, 2*n-2-j, j+n-1))
    
    if k_min > k_max:
        return np.array([]), np.array([], dtype=int)
    
    ks = np.arange(k_min, k_max + 1)

    h_i_plus = np.where((i_plus >= 0) & (i_plus < n), h[i_plus[(i_plus >= 0) & (i_plus < n)]], 0.0)

    deltas = -(1.0 - h_i_plus) + h_i_minus + (1.0 - h_j_plus) - h_j_minus
    return ks, deltas
```

```python
def simulated_annealing(h_init, n, T0, Tf, cooling, n_iters, deadline, seed):
    """Simulated annealing for binary sequence optimization"""
    rng = np.random.RandomState(seed)
    cur = h_init.copy()
    cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = cur_corr.max()
    
    best = cur.copy()
    best_max = cur_max
    
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (Tf / T0) ** (it / max(n_iters, 1))
        
        # Pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # Perform swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_corr = np.correlate(cur, 1.0 - cur, mode='full')
        new_max = new_corr.max()
        
        # Accept?
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / (T + 1e-12)):
            cur_max = new_max
            cur_corr = new_corr
            # Update sets
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
        else:
            # Revert
            cur[i] = 1.0
            cur[j] = 0.0
    
    return best, best_max
```

```python
def gradient_descent_c5(h, n, deadline, alpha=0.01, n_iters=100000):
    """Minimize max_k c_k via projected gradient descent"""
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = np.correlate(h, 1-h, mode='full').max() * dx
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1-h, mode='full')
        cur_max = c.max()
        
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
        
        # Get all argmax indices (subgradient)
        k_stars = np.where(c == cur_max)[0]
        d_stars = k_stars - (n - 1)
        
        # Compute average subgradient
        grad = np.zeros(n)
        for d in d_stars:
            # ∂c/∂h[i] = (1-h[i+d]) - h[i-d]
            i_plus = np.arange(n) + d
            i_minus = np.arange(n) - d
            val_plus = np.where((i_plus >= 0) & (i_plus < n), 0.0, 0.0)
            val_minus = np.where((i_minus >= 0) & (i_minus < n), 0.0, 0.0)
            
            # For i where i+d is in range: (1-h[i+d])
            valid_plus = (i_plus >= 0) & (i_plus < n)
            grad[valid_plus] += (1.0 - h[i_plus[valid_plus]])
            
            # For i where i-d is in range: -h[i-d]
            valid_minus = (i_minus >= 0) & (i_minus < n)
            grad[valid_minus] -= h[i_minus[valid_minus]]
        
        grad /= len(d_stars)
        
        # Gradient descent step
        h = h - alpha * grad
        # Project onto [0,1]^n with sum = n/2
        h = project_feasible(h, n)
        
        # Adaptive step
        c_new = np.correlate(h, 1-h, mode='full').max()
        if c_new > cur_max * 1.01:
            alpha *= 0.9
            h = project_feasible(best_h + alpha * grad, n)  # undo and retry smaller
        elif it % 1000 == 0:
            alpha *= 1.01
    
    return best_h, best_c
```

```python
def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        # Bisection-style projection
        if diff > 0:
            # Need to increase sum: add to elements with room
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

    order = np.argsort(h)[::-1]

    rng = np.random.RandomState(seed)
    cur = h_init.copy()

    cur_max = cur_corr.max()

    zeros = np.where(cur == 0)[0]

            break
        
        T = T0 * (T_final / T0) ** (it / n_total)

        j = zeros[rng.randint(len(zeros))]

        new_max = new_corr.max()

            cur_max = new_max
            cur_corr = new_corr
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
```

    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

            break

            s = room.sum()
            if s < 1e-14:

            if s < 1e-14:

def legendre_seq(n):
    """Generate a sequence based on quadratic residues mod (n+1) (prime)"""
    # Find next prime >= n+1
    p = n + 1
    while not all(p % i != 0 for i in range(2, int(p**0.5)+1)):
        p += 1
    h = np.zeros(n)
    for x in range(1, p):
        x2 = (x*x) % p
        for y in range(1, x2):
            if (y*y) % p == x2:
                if x <= n:
                    h[x-1] = 1.0
                break
