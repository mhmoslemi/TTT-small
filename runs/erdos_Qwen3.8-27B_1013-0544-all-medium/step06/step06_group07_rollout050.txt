            break

            s = h.sum()
            if s < 1e-14:

            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            for _ in range(nk):
                ii = zi[rng.randint(len(zi))]
                jj = oi[rng.randint(len(oi))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_improve = 0
            T = t_start * (0.4 + 0.4 * rng.random())
    return best_h, best_c

def steepest_descent(h, n, deadline, max_sweeps=30):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        best_dc = 0.0
        best_move = None
        zi = np.where(h == 0)[0]
        oi = np.where(h == 1)[0]
        for i in zi:
            if time.time() > deadline:
                break
            for j in oi:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                dc = nc - cur_c
                if dc < best_dc - 1e-15:
                    best_dc = dc
                    best_move = (i, j, delta)
        if best_move is not None:
            i, j, delta = best_move
            h[i] = 1.0
            h[j] = 0.0
            corr += delta
            cur_c += best_dc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Quick SLSQP for good smooth start
    t1 = t0 + 45
    hs = h0.copy()
    for alpha in [3000, 30000, 150000]:
        if time.time() > t1:
            break
        hs = slsqp_refine(hs, n0, alpha, maxiter=300)
        c = compute_c5(hs, n0)
        if c < best_c:
            best_c = c
            best_h = hs.copy()
            best_n = n0
    h_bin = round_to_binary(hs, n0)

    # Phase 2: Many SA episodes at n0
    t2 = t0 + 0.55 * budget_s
    n_eps = 12
    ep_dur = max(15, (t2 - time.time()) / n_eps - 1)
    for ep in range(n_eps):
        if time.time() > t2:
            break
        ep_dl = min(time.time() + ep_dur, t2)
        s = seed + ep * 7
        if ep == 0:
            start = h_bin
        elif ep == 1:
            start = round_to_binary(hs + 0.12 * rng.randn(n0), n0)
        elif ep == 2:
            start = round_to_binary(hs + 0.25 * rng.randn(n0), n0)
        elif ep == 3:
            start = make_random_binary(n0, s)
        elif ep == 4:
            start = round_to_binary(np.sort(hs)[::-1], n0)
        elif ep == 5:
            start = np.tile([1.0, 1.0, 0.0], n0 // 3 + 1)[:n0]
            start = round_to_binary(start, n0)
        else:
            start = round_to_binary(best_h + 0.15 * rng.randn(best_n) if best_n == n0 else hs, n0)
            if start.shape[0] != n0:
                start = h_bin
        ts = 0.008 + 0.008 * rng.random()
        pt = 6000 + int(6000 * rng.random())
        hb, cb = sa_binary(start, n0, ep_dl, seed=s, t_start=ts, patience=pt)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = n0
            if cb <= (n0 // 2) * (2.0 / n0) - (2.0 / n0):
                pass  # Found very good binary

    # Phase 3: Steepest descent polish
    t3 = t0 + 0.65 * budget_s
    if time.time() < t3:
        hd, cd = steepest_descent(round_to_binary(best_h, n0), n0, t3, max_sweeps=20)
        if cd < best_c:
            best_c = cd
            best_h = hd.copy()
            best_n = n0

    # Phase 4: Try other n values
    t4 = t0 + 0.82 * budget_s
    for n_try in [200, 240, 280, 300]:
        if time.time() > t4:
            break
        if best_n == n_try:
            ht = best_h.copy()
        else:
            xo = np.linspace(0, 2, best_n)
            xn = np.linspace(0, 2, n_try)
            ht = project_feasible(np.interp(xn, xo, best_h), n_try)
        hb_t = round_to_binary(ht, n_try)
        ep_dl = min(time.time() + 50, t4)
        hb, cb = sa_binary(hb_t, n_try, ep_dl, seed=seed + 500 + n_try, t_start=0.012, patience=7000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = n_try

    # Phase 5: Final SLSQP refinement
    if time.time() < dl - 25:
        hf = best_h.copy()
        for alpha in [30000, 100000, 400000]:
            if time.time() > dl - 8:
                break
            hf = slsqp_refine(hf, best_n, alpha, maxiter=300)
            c = compute_c5(hf, best_n)
            if c < best_c:
                best_c = c
                best_h = hf.copy()

    # Phase 6: Final SA
    if time.time() < dl - 5:
        hbf = round_to_binary(best_h, best_n)
        hb, cb = sa_binary(hbf, best_n, dl - 2, seed=seed + 999, t_start=0.008, patience=5000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
