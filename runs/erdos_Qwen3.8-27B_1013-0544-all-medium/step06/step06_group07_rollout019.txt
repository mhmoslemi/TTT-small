        step_done = True
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(8)
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            for _ in range(nk):
                if len(zi) == 0 or len(oi) == 0:
                    break
                a = zi[rng.randint(len(zi))]
                b = oi[rng.randint(len(oi))]
                h[a], h[b] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_improve = 0
            T = min(t_start, T * 5.0)
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
    return best_h, best_c

def targeted_local_search(h_bin, n, max_iter=500000, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    for _ in range(max_iter):
        improved = False
        critical = np.where(corr * dx > best_c - dx * 2)[0]
        order = rng.permutation(len(zeros_idx) * len(ones_idx))
        tried = 0
        limit = min(len(order), 20000)
        for idx in order[:limit]:
            ii = idx % len(zeros_idx)
            jj = idx // len(zeros_idx)
            i = zeros_idx[ii]
            j = ones_idx[jj]
            delta = make_swap_delta(h, n, i, j, m_range)
            new_corr = corr + delta
            nc = new_corr.max() * dx
            if nc < best_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                best_c = nc
                best_h = h.copy()
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                improved = True
                break
        if not improved:
            if tried > 5000:
                break
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=2000, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Quick SLSQP on continuous
    p1 = t0 + 0.08 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [1000, 10000, 100000]:
        if time.time() > p1:
            break
        h = slsqp_refine(h, n, alpha, 1500, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Enhanced SA at current n
    p2 = t0 + 0.45 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    for attempt in range(4):
        if time.time() > p2:
            break
        if attempt == 0:
            start = h_bin
            s = seed
        elif attempt == 1:
            start = round_to_binary(best_h + 0.15 * rng.randn(n), n)
            s = seed + 100
        elif attempt == 2:
            start = round_to_binary(best_h + 0.3 * rng.randn(n), n)
            s = seed + 200
        else:
            start = round_to_binary(np.full(n, 0.5) + 0.5 * rng.randn(n), n)
            s = seed + 300
        hb, cb = enhanced_sa(start, n, p2, seed=s, t_start=0.05, t_end=0.00005, patience=40000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 3: Targeted local search
    p3 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p3:
        hb, cb = targeted_local_search(best_h, n, max_iter=200000, seed=seed+10)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 4: Try at n*2 resolution
    p4 = t0 + 0.92 * (budget_s - 5)
    n2 = n * 2
    if time.time() < p4 - 30:
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n2)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, n2)
        hbin2 = round_to_binary(hh, n2)
        hb2, cb2 = enhanced_sa(hbin2, n2, p4, seed=seed+400, t_start=0.04, t_end=0.00005, patience=35000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = n2
    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 10:
        hp = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [50000, 200000]:
            if time.time() > deadline - 5:
                break
            hp = slsqp_refine(hp, n, alpha, 1500, 1e-15)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
