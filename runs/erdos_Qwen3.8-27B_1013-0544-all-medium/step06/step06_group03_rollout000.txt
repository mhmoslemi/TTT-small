            best_c = c_ls
            best_h = h_ls.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    # Main optimization on initial n
    if h_init is not None:
        alloc1 = min(250, (deadline - t0) * 0.35)
        h_opt, c_opt = refine(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

    # Try a few other n values
    n_candidates = [150, 200, 250, 400]
    if h_init is not None and n_init in n_candidates:
        n_candidates.remove(n_init)
    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        alloc = min(100, remaining * 0.25)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = refine(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Random restarts on best n
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        n_restarts = 0
        while time.time() < deadline - 30:
            n_restarts += 1
            scale = 0.008 + 0.008 * (n_restarts % 5)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 25) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = refine(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final ultra-fine polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
