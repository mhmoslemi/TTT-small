def project_feasible(y, target):
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    n = y.size
    lo = float(np.min(y) - 1.0)
    hi = float(np.max(y) + 1.0)
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(y-mid,0.0,1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y-mid,0.0,1.0)
    # final correction
    for _ in range(8):
        diff = target - z.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - z))
            add = min(diff, 1.0 - z[idx] + 1e-15)
            z[idx] += add
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx] + 1e-15)
            z[idx] -= sub
    return z

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(y, target):
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    target = float(target)
    lo = float(np.min(y) - 1.0)
    hi = float(np.max(y) + 1.0)
    mid = 0.5 * (lo + hi)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - mid, 0.0, 1.0)
    for _ in range(8):
        diff = target - float(z.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - z))
            z[idx] += min(diff, 1.0 - z[idx] + 1e-15)
        else:
            idx = int(np.argmax(z))
            z[idx] -= min(-diff, z[idx] + 1e-15)
    return z

def erdos_bound(h, n):
    if n <= 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))

def overlap_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")[::-1]

def softmax_weights(O, alpha):
    if alpha <= 0:
        return np.full(O.size, 1.0 / float(O.size))
    m = float(O.max())
    z = alpha * (O - m)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        e = np.zeros(O.size)
        e[int(np.argmax(O))] = 1.0
        s = 1.0
    return e / s

def weighted_gradient_direct(h, p):
    n = h.size
    g = np.zeros(n, dtype=float)
    center = n - 1
    for ci in range(2 * n - 1):
        wk = p[ci]
        if wk == 0.0:
            continue
        k = ci - center
        if k >= 0:
            m = n - k
            if m <= 0:
                continue
            g[:m] += wk * (1.0 - h[k:n])
            g[k:n] -= wk * h[:m]
        else:
            q = -k
            m = n - q
            g[:m] -= wk * h[q:n]
            g[q:n] += wk * (1.0 - h[:m])
    return g

def value_grad(x, n, alpha, mass_penalty=0.0):
    O = overlap_counts(x)
    m = float(O.max())
    if alpha > 0:
        z = np.clip(alpha * (O - m), -745.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            e = np.zeros(O.size)
            e[int(np.argmax(O))] = 1.0
            s = 1.0
        p = e / s
        val = m + np.log(s) / alpha
    else:
        p = np.full(O.size, 1.0 / float(O.size))
        val = m
    g = weighted_gradient_direct(x, p)
    if mass_penalty > 0.0:
        mass = float(x.sum() - 0.5 * n)
        val += 0.5 * mass_penalty * mass * mass
        g = g + (mass_penalty * mass) * np.ones(n, dtype=float)
    return float(val), g

def fun_lbfgs(x, n, alpha, mu):
    return value_grad(x, n, alpha, mu)

def fun_slsqp(x, n, alpha):
    return value_grad(x, n, alpha, 0.0)

def mass_fun(x, n):
    return float(np.sum(x) - 0.5 * n)

def lbfgs_refine(h, n, alpha, maxiter, deadline, mu=100.0):
    x0 = project_feasible(h, 0.5 * n)
    rem = deadline - time.time()
    if rem <= 0:
        return x0
    if rem < 15:
        maxiter = min(maxiter, 8)
    elif rem < 45:
        maxiter = min(maxiter, 25)
    if maxiter < 1:
        return x0
    try:
        res = minimize(
            fun_lbfgs, x0, jac=True, args=(n, alpha, mu),
            method="L-BFGS-B", bounds=[(0.0, 1.0)] * n,
            options={"maxiter": int(maxiter), "ftol": 1e-13, "maxls": 30}
        )
        if np.all(np.isfinite(res.x)):
            return project_feasible(res.x, 0.5 * n)
    except Exception:
        pass
    return x0

def slsqp_refine(h, n, alpha, maxiter, deadline):
    x0 = project_feasible(h, 0.5 * n)
    if n > 350:
        return x0
    rem = deadline - time.time()
    if rem <= 5:
        return x0
    if rem < 25:
        maxiter = min(maxiter, 8)
    elif rem < 90:
        maxiter = min(maxiter, 30)
    if maxiter < 1:
        return x0
    try:
        res = minimize(
            fun_slsqp, x0, jac=True, args=(n, alpha),
            method="SLSQP", bounds=[(0.0, 1.0)] * n,
            constraints=[{"type": "eq", "fun": mass_fun, "args": (n,)}],
            options={"maxiter": int(maxiter), "ftol": 1e-12}
        )
        if np.all(np.isfinite(res.x)):
            return project_feasible(res.x, 0.5 * n)
    except Exception:
        pass
    return x0

def pg_refine(h, n, alpha, deadline, maxiters, seed, step=0.35, trials=6):
    h = project_feasible(h, 0.5 * n)
    dx = 2.0 / float(n)
    O = overlap_counts(h)
    cur = float(O.max()) * dx
    best = cur
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    stall = 0
    for it in range(maxiters):
        if time.time() >= deadline:
            break
        p = softmax_weights(O, alpha)
        g = weighted_gradient_direct(h, p)
        g -= g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-11:
            break
        eta = step / gn
        accepted = False
        for _ in range(trials):
            if time.time() >= deadline:
                break
            h2 = project_feasible(h - eta * g, 0.5 * n)
            O2 = overlap_counts(h2)
            c2 = float(O2.max()) * dx
            if c2 < cur - 1e-14:
                h, O, cur = h2, O2, c2
                accepted = True
                if cur < best:
                    best = cur
                    best_h = h.copy()
                break
            eta *= 0.5
            if eta < 1e-13:
                break
        if accepted:
            stall = 0
            if it % 500 == 499:
                nz = rng.normal(0.0, 0.002, n)
                nz -= nz.mean()
                h3 = project_feasible(h + nz, 0.5 * n)
                O3 = overlap_counts(h3)
                c3 = float(O3.max()) * dx
                if c3 < cur + 1e-4 and c3 < best + 2e-4:
                    h, O, cur = h3, O3, c3
        else:
            stall += 1
            if stall > 20:
                nz = rng.normal(0.0, 0.005, n)
                nz -= nz.mean()
                h3 = project_feasible(h + nz, 0.5 * n)
                O3 = overlap_counts(h3)
                c3 = float(O3.max()) * dx
                if c3 < cur + 5e-5 and c3 < best + 1e-4:
                    h, O, cur = h3, O3, c3
                    stall = 0
                    continue
                break
    return best_h, best

def binarize(h, n):
    target = int(0.5 * n)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=float)
    b[idx[:target]] = 1.0
    return b

def resample_h(h, n_old, n_new):
    if n_new <= 0 or n_old <= 0:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return project_feasible(
        np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1])),
        0.5 * n_new
    )

def snap_extremes(h, n, tol):
    h2 = h.copy()
    h2[h2 < tol] = 0.0
    h2[h2 > 1.0 - tol] = 1.0
    return project_feasible(h2, 0.5 * n)

def make_fallback(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)
    t = int(0.5 * n)
    h[idx[:t]] = 1.0
    h = h + 0.05 * rng.normal(size=n)
    return project_feasible(h, 0.5 * n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + total - 3.0
    seed_int = int(seed) % 1000000007
    rng = np.random.RandomState(seed_int)

    try:
        h = np.array(initial_h_values, dtype=float).copy().ravel()
        if h.size < 2 or not np.all(np.isfinite(h)):
            raise ValueError
        n = int(h.size)
    except Exception:
        n = 250
        h = make_fallback(n, seed_int)

    if n > 600:
        h = resample_h(h, n, 500)
        n = 500

    best_h = project_feasible(h, 0.5 * n).copy()
    best_n = n
    best_c = erdos_bound(best_h, best_n)
    work = max(1.0, total - 5.0)

    p1 = t0 + 0.10 * work
    for alpha in (50.0, 200.0, 1000.0, 8000.0):
        if time.time() > p1 or time.time() > deadline - 20:
            break
        cand = lbfgs_refine(best_h, best_n, alpha, 50, deadline, 100.0)
        if np.all(np.isfinite(cand)):
            c = erdos_bound(cand, best_n)
            if c < best_c:
                best_c, best_h = c, cand

    p2 = p1 + 0.18 * work
    if best_n <= 350:
        for alpha, iters in ((1000.0, 60), (10000.0, 75), (100000.0, 75), (1000000.0, 70)):
            if time.time() > p2 or time.time() > deadline - 35:
                break
            cand = slsqp_refine(best_h, best_n, alpha, iters, deadline)
            c = erdos_bound(cand, best_n)
            if c < best_c and np.all(np.isfinite(cand)):
                best_c, best_h = c, cand

    p3 = p2 + 0.12 * work
    for pi, (alpha, iters) in enumerate(((5000.0, 250), (50000.0, 180))):
        if time.time() > p3 or time.time() > deadline - 20:
            break
        cand, c = pg_refine(best_h, best_n, alpha, deadline, iters, seed_int + 1000 + pi * 131)
        if c < best_c:
            best_c, best_h = c, cand

    p4 = p3 + 0.30 * work
    r = 0
    while r < 25 and time.time() < p4 - 5 and time.time() < deadline - 50:
        sig = 0.005 + 0.08 * rng.random()
        noise = rng.normal(0.0, sig, best_n)
        noise -= noise.mean()
        cand = project_feasible(best_h + noise, 0.5 * best_n)
        if r % 7 == 0:
            cand = project_feasible(binarize(cand, best_n), 0.5 * best_n)
        if not np.all(np.isfinite(cand)):
            cand = best_h.copy()
        for alpha in (500.0, 5000.0, 50000.0):
            if time.time() > p4 - 10 or time.time() > deadline - 30:
                break
            cand = lbfgs_refine(cand, best_n, alpha, 35, deadline, 100.0)
            c = erdos_bound(cand, best_n)
            if c < best_c and np.all(np.isfinite(cand)):
                best_c, best_h = c, cand
            elif c > best_c + 1e-4:
                cand = best_h.copy()
        r += 1

    if best_n < 500 and time.time() < deadline - 180:
        n2 = 500
        cand = resample_h(best_h, best_n, n2)
        p5 = min(deadline - 100, time.time() + 180)
        for alpha in (1000.0, 10000.0, 100000.0):
            if time.time() > p5 - 30:
                break
            cand = lbfgs_refine(cand, n2, alpha, 45, deadline, 100.0)
            c = erdos_bound(cand, n2)
            if c < best_c and np.all(np.isfinite(cand)):
                best_c, best_h, best_n = c, cand, n2
        if best_n == n2 and time.time() < deadline - 60:
            cand, c = pg_refine(best_h, best_n, 50000.0, min(deadline - 40, time.time() + 60), 120, seed_int + 99999)
            if c < best_c:
                best_c, best_h = c, cand

    if time.time() < deadline - 12:
        if best_n <= 350:
            for alpha, iters in ((100000.0, 55), (1000000.0, 70), (10000000.0, 50)):
                if time.time() > deadline - 6:
                    break
                cand = slsqp_refine(best_h, best_n, alpha, iters, deadline)
                c = erdos_bound(cand, best_n)
                if c < best_c and np.all(np.isfinite(cand)):
                    best_c, best_h = c, cand
        else:
            for alpha in (1000.0, 10000.0, 100000.0):
                if time.time() > deadline - 6:
                    break
                cand = lbfgs_refine(best_h, best_n, alpha, 40, deadline, 100.0)
                c = erdos_bound(cand, best_n)
                if c < best_c and np.all(np.isfinite(cand)):
                    best_c, best_h = c, cand
        for tol in (1e-7, 1e-5):
            if time.time() > deadline - 3:
                break
            cand = snap_extremes(best_h, best_n, tol)
            c = erdos_bound(cand, best_n)
            if c < best_c and np.all(np.isfinite(cand)):
                best_c, best_h = c, cand

    best_h = project_feasible(best_h, 0.5 * best_n)
    best_c = erdos_bound(best_h, best_n)
    return best_h, float(best_c), int(best_n)
