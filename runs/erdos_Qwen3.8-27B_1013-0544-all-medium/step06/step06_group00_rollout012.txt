def lp_optimize_step(h, n, eps, solver='GLOP'):
    """One step of LP-based optimization."""
    import cvxpy as cp
    
    C = np.correlate(h, 1-h, mode='full').astype(float)
    # C has length 2n-1, index m corresponds to lag k = m-(n-1)
    
    # Compute gradient matrix G: G[j, m] = dC[m]/dh[j]
    # G[j, m] = (1-h[j-k]) * [0<=j-k<=n-1] - h[j+k] * [0<=j+k<=n-1]
    # where k = m - (n-1)
    
    # For efficiency, compute G as a matrix
    G = np.zeros((n, 2*n-1))
    for m in range(2*n-1):
        k = m - (n-1)
        # Term 1: (1-h[j-k]) for j where 0<=j-k<=n-1, i.e., k<=j<=n-1+k
        # Term 2: -h[j+k] for j where 0<=j+k<=n-1, i.e., -k<=j<=n-1-k
        # For term 1: j ranges from max(0,k) to min(n-1, n-1+k)
        # For term 2: j ranges from max(0,-k) to min(n-1, n-1-k)
        j1_start = max(0, k)
        j1_end = min(n, n+k)  # exclusive
        if j1_end > j1_start:
            G[j1_start:j1_end, m] = 1 - h[j1_start-k:j1_end-k]
        
        j2_start = max(0, -k)
        j2_end = min(n, n-k)  # exclusive
        if j2_end > j2_start:
            G[j2_start:j2_end, m] -= h[j2_start+k:j2_end+k]
    
    # Solve LP
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        C + G.T @ delta <= t,  # shape: (2n-1)
        delta >= -eps,
        delta <= eps,
        delta >= -h,  # h + delta >= 0
        delta <= 1 - h,  # h + delta <= 1
        cp.sum(delta) == 0
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=solver)
    
    if delta.value is None:
        return h, C.max()
    
    d = delta.value
    # Line search
    best_h_local = h.copy()
    best_t = C.max()
    for lam in [1.0, 0.5, 0.25, 0.125]:
        h_try = h + lam * d
        h_try = np.clip(h_try, 0, 1)
        # Project to maintain sum
        h_try = project_feasible(h_try, n)
        c_try = np.correlate(h_try, 1-h_try, mode='full').max()
        if c_try < best_t:
            best_t = c_try
            best_h_local = h_try
    return best_h_local, best_t

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            room = 1.0 - h
            s = room.sum()

                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps, solver='GLOP'):

    mc = float(C.max())
    G = compute_gradients(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()

        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
