def smooth_max_val(x, alpha):

def slsqp_top(h, n, alpha, top_k, maxiter):

    ti = np.argsort(c)[-top_k:].copy()

    def obj(x):
        cc = all_lags(x, n)
        return smooth_max_val(cc[ti], alpha)

def local_search(h, n, deadline, max_passes=4):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.003, 0.001, 0.0003, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            c = all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            p = abs(lag)
            if p >= n:
                break
            for i in range(n - p):
                if time.time() > deadline:
                    break
                j = i + p
                if h[i] > h[j] + 1e-10:
                    delta = min(d, (h[i] - h[j]) * 0.3, h[i], 1.0 - h[j])
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-10:
                    delta = min(d, (h[j] - h[i]) * 0.3, h[j], 1.0 - h[i])

                    c_new = compute_c5(h, n)

                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
            if not improved:
                break
    return h, best_c

def proj_subgrad(h, n, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    v = np.zeros(n)
    step = 0.05
    mom = 0.85
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = all_lags(h_curr, n)
        c_max = c.max()
        thr = c_max - (c_max - c.min()) * 0.1
        active = np.where(c > thr)[0]
        w = (c[active] - thr)
        w = w / (w.sum() + 1e-30)
        grad = np.zeros(n)
        for idx, wi in zip(active, w):
            grad += wi * lag_grad(h_curr, n, int(idx) - (n - 1))
        v = mom * v - step * grad
        h_new = project(h_curr + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            step = min(step * 1.02, 0.3)
        else:
            step *= 0.93
            v *= 0.5
        h_curr = h_new
    return best_h, best_c

def binary_start(h, n):
    b = (h > 0.5).astype(float)
    ones = int(b.sum())
    target = n // 2
    if ones > target:
        idx = np.where(b == 1)[0]
        order = np.argsort(h[idx])[:ones - target]
        b[idx[order]] = 0
    elif ones < target:
        idx = np.where(b == 0)[0]
        order = np.argsort(-h[idx])[:target - ones]
        b[idx[order]] = 1
    return b

def optimize_at_n(h_start, n, deadline, rng):
