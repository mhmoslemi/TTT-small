from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0, 1)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0, 1)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, tol_frac=0.01, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    lag0_idx = n - 1
    all_idx = np.arange(2 * n - 1)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_raw = np.correlate(h, 1 - h, mode='full')
        m = c_raw.max()
        threshold = m - tol_frac * (m + 1e-10) * (2.0 / n) * n / 2.0
        active_mask = (c_raw >= threshold) & (all_idx != lag0_idx)
        active = np.where(active_mask)[0]
        if len(active) < 2:
            active = np.argsort(c_raw)[-6:]
            active = active[all_idx[active] != lag0_idx]
        if len(active) == 0:
            break
        a_lags = (all_idx[active] - (n - 1)).astype(int)
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            a_vals = np.zeros(len(active))
            for idx, lag in enumerate(a_lags):
                a = 0.0
                if lag > 0:
                    if i + lag < n:
                        a += (1.0 - h[i + lag])
                    if i - lag >= 0:
                        a -= h[i - lag]
                elif lag < 0:
                    ll = -lag
                    if i >= ll:
                        a += (1.0 - h[i - ll])
                    if i + ll < n:
                        a -= h[i + ll]
                a_vals[idx] = a
            b_vals = c_raw[active] - h[i] * a_vals
            candidates = [0.0, 1.0]

                            candidates.append(x_int)
            best_x = h[i]
            best_val = float(np.max(a_vals * best_x + b_vals))
            for x in candidates:
                xc = min(max(x, 0.0), 1.0)
                val = float(np.max(a_vals * xc + b_vals))
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = xc

                    h[mask] = np.clip(h[mask] + diff / mask.sum(), 0, 1)
