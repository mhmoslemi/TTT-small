        B_p = W[hi_b + 1] - W[lo_b]
        
        grad[p] = (1.0 - h[p]) * A_p - h[p] * B_p
    
    return grad * (2.0 / n)  # include dx factor

def nesterov_optimize(h, n, deadline, alpha=5000, lr_init=0.01, momentum=0.9):
    """Nesterov-accelerated gradient descent with projection."""
    h_curr = h.copy()
    h_prev = h.copy()
    momentum_val = momentum
    lr = lr_init
    
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(10000):
        if time.time() > deadline:
            break
        
        # Nesterov lookahead
        h_look = h_curr + momentum_val * (h_curr - h_prev)
        h_look = project_feasible(h_look, n)
        
        # Compute gradient
        g = grad_smooth_max(h_look, n, alpha)
        
        # Backtracking line search
        t = lr
        f_look = smooth_max_val(np.correlate(h_look, 1.0 - h_look, mode='full'), alpha) * (2.0/n)
        while t > 1e-8:
            h_trial = h_look - t * g
            h_trial = project_feasible(h_trial, n)
            f_trial = smooth_max_val(np.correlate(h_trial, 1.0 - h_trial, mode='full'), alpha) * (2.0/n)
            if f_trial <= f_look - 0.5 * t * np.dot(g, g):  # This is wrong for projected gradient
                break
            t *= 0.5
        
        h_new = h_look - t * g
        h_new = project_feasible(h_new, n)
        
        # Update
        h_prev = h_curr
        h_curr = h_new
        momentum_val = min(0.999, 0.9 + 0.1 * (it / 1000))
        
        # Track best
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.95
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    if mx > 500:
        return m + (mx + float(np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + float(np.log(np.sum(np.exp(sh)))) / alpha

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    weights = np.exp(sh - mx)
    weights /= weights.sum()
    W = np.zeros(2 * n)
    W[1:] = np.cumsum(weights)
    grad = np.zeros(n)
    for p in range(n):
        lo_a = max(0, n - 1 - p)
        hi_a = min(2 * n - 2, n - 1 + p)
        A_p = W[hi_a + 1] - W[lo_a]
        lo_b = max(0, p)
        hi_b = min(2 * n - 2, p + n - 1)
        B_p = W[hi_b + 1] - W[lo_b]
        grad[p] = ((1.0 - h[p]) * A_p - h[p] * B_p) * dx
    return grad

def obj_and_grad(x, n, alpha):
    val = smooth_max_val(np.correlate(x, 1.0 - x, mode='full'), alpha) * (2.0 / n)
    g = grad_smooth_max(x, n, alpha)
    return val, g

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_and_grad, h, args=(n, alpha), jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        try:
            res = minimize(lambda x: smooth_max_val(np.correlate(x, 1.0-x, mode='full'), alpha)*(2.0/n),
                           h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            return project_feasible(res.x, n)
        except Exception:
            return project_feasible(h, n)

def nesterov_opt(h, n, deadline, alpha=5000, lr_init=0.005):
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    mu = 0.9
    lr = lr_init
    for it in range(20000):
        if time.time() > deadline:
            break
        h_look = h_curr + mu * (h_curr - h_prev)
        h_look = project_feasible(h_look, n)
        g = grad_smooth_max(h_look, n, alpha)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        t = lr
        f_look = smooth_max_val(np.correlate(h_look, 1.0 - h_look, mode='full'), alpha) * (2.0 / n)
        for _ in range(30):
            h_trial = h_look - t * g
            h_trial = project_feasible(h_trial, n)
            f_trial = smooth_max_val(np.correlate(h_trial, 1.0 - h_trial, mode='full'), alpha) * (2.0 / n)
            if f_trial <= f_look - 0.4 * t * gnorm * gnorm:
                break
            t *= 0.5
        h_new = h_look - t * g
        h_new = project_feasible(h_new, n)
        h_prev = h_curr
        h_curr = h_new
        mu = min(0.999, 0.9 + 0.09 * min(1.0, it / 2000.0))
        if it % 200 == 50:
            lr *= 0.97
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c - 1e-12:
            best_c = c_curr
            best_h = h_curr.copy()
    return best_h, best_c

def fast_cd(h, n, max_passes, grid, deadline, rng, penalty=2.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n - 1)]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - n / 2.0) ** 2
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c = new_c.copy()
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - n / 2.0) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= d * h[p + ks[ids_b] - (n - 1)]
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        oh = h[i].copy()
                        h[i] = min(1.0, h[i] + d)
                        h[j] = max(0.0, h[j] - d)
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-15:
                            best_c = cval
                            improved = True
                        else:
                            h[i], h[j] = oh, h[j]
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        oh = h[i].copy()
                        h[i] = max(0.0, h[i] - d)
                        h[j] = min(1.0, h[j] + d)
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-15:
                            best_c = cval
                            improved = True
                        else:
                            h[i], h[j] = oh, h[i]
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    rem = deadline - time.time()

    # Phase A: Nesterov

    # Phase B: CD
    if time.time() < deadline - 40:

        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    # Phase C: SLSQP with gradient
    if time.time() < deadline - 30:
        for a in [20000, 100000, 400000]:
            if time.time() > deadline - 25:

            h_s = slsqp_optimize(h, n, alpha=a, maxiter=2000)

    # Phase D: Fine CD

    if time.time() < deadline - 3:
        h, best_c = pair_search(h, n, deadline - 2,
