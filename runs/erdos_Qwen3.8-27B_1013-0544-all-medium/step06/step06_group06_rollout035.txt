def continuous_optimize(h, n, deadline, alpha=1000.0):

    h = h.astype(float).copy()
    h = project_fast(h, n/2)
    
    c = np.correlate(h, 1 - h, mode='full')
    cur_c = c.max() * (2.0/n)

    prev_h = h.copy()
    prev_grad = None
    lr = 0.01
    
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full')
        cm = c.max()
        shifted = alpha * (c - cm)
        smax = shifted.max()
        exp_shifted = np.exp(shifted - smax)
        weights = exp_shifted / exp_shifted.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-10:

                grad[:n-k] += w * (1 - h[k:])
                grad[k:] -= w * h[:n-k]
            else:
                kk = -k
                grad[kk:] += w * (1 - h[:n-kk])
                grad[:n-kk] -= w * h[kk:]

        if prev_grad is not None:
            s = h - prev_h
            y = grad - prev_grad
            sy = s.dot(y)
            yy = y.dot(y)
            if yy > 1e-10 and sy > 1e-10:
                lr = s.dot(s) / sy
            elif yy > 1e-10:
                lr = s.dot(s) / yy
        prev_h = h.copy()
        prev_grad = grad.copy()
        
        # Gradient step with damping
        h_new = h - lr * grad
        h_new = project_fast(h_new, n / 2)

        if c_new_max < best_c:
            best_c = c_new_max
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

        cm = c.max()
        shifted = alpha * (c - cm)
        smax = shifted.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_fast(h_raw, target):
    n = len(h_raw)
    h_raw = np.asarray(h_raw, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h_raw - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h_p = np.clip(h_raw - t, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-10:
        free_mask = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        nf = int(free_mask.sum())
        if nf > 0:
            h_p[free_mask] += diff / nf
    return np.clip(h_p, 0.0, 1.0)

def sa_binary(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    t0_val = 4.0
    tf_val = 0.003
    it = 0
    n_est = 400000
    no_improve = 0
    kick_cd = 0

        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i] = 1.0
        h[j] = 0.0
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c = new_corr.max()
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c = new_c
            corr = new_corr
            zi = np.where(zeros == i)[0]
            oj = np.where(ones == j)[0]
            if len(zi) > 0 and len(oj) > 0:
                zeros = np.delete(zeros, zi[0])
                ones = np.delete(ones, oj[0])
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 0.0
            h[j] = 1.0
            no_improve += 1

            nk = int(rng.randint(4, 9))
            nk = min(nk, len(zeros), len(ones))
            ri = rng.choice(len(zeros), nk, replace=False)
            rj = rng.choice(len(ones), nk, replace=False)
            for a in range(nk):
                h[zeros[ri[a]]] = 1.0
                h[ones[rj[a]]] = 0.0
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            corr = np.correlate(h, 1 - h, mode='full')
            cur_c = corr.max()
            no_improve = 0

def smooth_max_grad(h, alpha, n):
    c = np.correlate(h, 1 - h, mode='full')
    cm = c.max()
    shifted = alpha * (c - cm)
    smax = shifted.max()
    exp_s = np.exp(shifted - smax)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        k = idx - n + 1
        if k >= 0:
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wi * (1.0 - h[:n - kk])
            grad[:n - kk] -= wi * h[kk:]
    return grad

def cont_optimize(h, n, deadline, alpha=2000.0):
    h = h.astype(float).copy()
    h = project_fast(h, n / 2.0)
    target = n / 2.0
    c = np.correlate(h, 1 - h, mode='full')
    best_c = float(c.max() * 2.0 / n)
    best_h = h.copy()
    prev_h = h.copy()
    prev_grad = smooth_max_grad(h, alpha, n)
    lr = 0.1
    it = 0
    while time.time() < deadline:
        grad = smooth_max_grad(h, alpha, n)
        s = h - prev_h
        y = grad - prev_grad
        sy = float(s.dot(y))
        yy = float(y.dot(y))
        ss = float(s.dot(s))
        if yy > 1e-12 and 0.1 < sy < 10.0 * yy:
            lr_new = ss / sy
        elif yy > 1e-12:
            lr_new = 0.05 * ss / (yy + 1e-12)
        else:
            lr_new = lr
        lr = min(max(lr_new, 1e-5), 0.5)
        prev_h = h.copy()
        prev_grad = grad.copy()
        h_new = h - lr * grad
        h_new = project_fast(h_new, target)
        c_new = np.correlate(h_new, 1 - h_new, mode='full')
        cnm = float(c_new.max() * 2.0 / n)
        if cnm < best_c:
            best_c = cnm
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c

def lbfgsb_cont(h, n, deadline, alpha=5000.0):
    n_pen = 1e8
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        cm = c.max()
        shifted = alpha * (c - cm)
        smax = shifted.max()
        es = np.exp(shifted - smax)
        val = cm + np.log(es.sum()) / alpha
        val += n_pen * (x.sum() - n / 2.0) ** 2
        return val
    def grad_fn(x):
        c = np.correlate(x, 1 - x, mode='full')
        cm = c.max()
        shifted = alpha * (c - cm)
        smax = shifted.max()
        es = np.exp(shifted - smax)
        w = es / es.sum()
        g = np.zeros(n)
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-10:
                continue
            k = idx - n + 1
            if k >= 0:
                g[:n - k] += wi * (1.0 - x[k:])
                g[k:] -= wi * x[:n - k]
            else:
                kk = -k
                g[kk:] += wi * (1.0 - x[:n - kk])
                g[:n - kk] -= wi * x[kk:]
        g += 2.0 * n_pen * (x.sum() - n / 2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad_fn,
                       bounds=bounds, options={'maxiter': 800, 'gtol': 1e-10,
                                               'maxfun': 20000})
        return project_fast(res.x, n / 2.0)
    except Exception:
        return h
