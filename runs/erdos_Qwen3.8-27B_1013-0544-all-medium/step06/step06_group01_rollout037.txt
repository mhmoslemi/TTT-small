```
def grad_corr_weights(h,w,n):
    g = np.empty(n, dtype=float)
    wc = float(w[n-1])
    g.fill(wc)
    if n > 1:
        pos = w[n:2*n-1]
        neg = w[0:n-1][::-1]
        cp = np.zeros(n); cp[1:] = np.cumsum(pos)
        cn = np.zeros(n); cn[1:] = np.cumsum(neg)
        idx = np.arange(n)
        g += cp[idx] + cn[n-1-idx]
        if wc:
            g -= 2.0*wc*h
        combo = pos + neg
        nonzero = np.nonzero(combo)[0]
        for k in nonzero:
            d = k + 1
            wd = float(combo[k])
            m = n - d
            g[:m] -= wd*h[d:]
            g[d:] -= wd*h[:m]
    elif n == 1:
        # only center, c = h(1-h), grad wc*(1-2h)
        g[0] = wc*(1.0 - 2.0*h[0])
    return g
```

```
def correlate_comp(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_eval(h):
    n=len(h)
    return float(np.max(correlate_comp(h)) * (2.0/n))

def project_feasible(y,n):
    target=0.5*n
    y=np.asarray(y,dtype=float).copy()
    y=np.clip(y,0.0,1.0)
    s=y.sum()
    if abs(s-target)<1e-13:
       return y
    lo=y.min()-1.0
    hi=y.max()+1.0
    for _ in range(80):
       lam=0.5*(lo+hi)
       val=np.clip(y-lam,0.0,1.0).sum()
       if val>target: lo=lam
       else: hi=lam
    x=np.clip(y-0.5*(lo+hi),0.0,1.0)
    # adjust residual
    for _ in range(20):
       diff=target-x.sum()
       if abs(diff)<1e-14: break
       if diff>0:
           idx=np.argmin(x)
           add=min(diff,1.0-x[idx])
           x[idx]+=add
       else:
           idx=np.argmax(x)
           sub=min(-diff,x[idx])
           x[idx]-=sub
    return np.clip(x,0.0,1.0)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time


def correlate_comp(h):
    return np.correlate(h, 1.0 - h, "full")


def c5_eval(h):
    return float(np.max(correlate_comp(h)) * (2.0 / len(h)))


def project_feasible(y, n):
    y = np.asarray(y, dtype=float)
    y = np.where(np.isfinite(y), y, 0.5)
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * n
    if abs(float(y.sum()) - target) < 1e-13:
        return y
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        if float(np.clip(y - lam, 0.0, 1.0).sum()) > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    for _ in range(max(1, n)):
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            room = 1.0 - x[idx]
            if room <= 1e-15:
                continue
            add = min(diff, room)
            x[idx] += add
            diff -= add
        else:
            idx = int(np.argmax(x))
            room = x[idx]
            if room <= 1e-15:
                continue
            sub = min(-diff, room)
            x[idx] -= sub
            diff += sub
    return np.clip(x, 0.0, 1.0)


def sum_eq(x, n):
    return float(np.sum(x) - 0.5 * n)


def sum_jac(x, n):
    return np.ones((1, n), dtype=float)


def grad_corr_weights(h, w, n):
    g = np.empty(n, dtype=float)
    wc = float(w[n - 1])
    g.fill(wc)
    if n <= 1:
        if n == 1:
            g[0] = wc * (1.0 - 2.0 * h[0])
        return g
    pos = w[n:]
    neg = w[n - 2::-1]
    cp = np.zeros(n, dtype=float)
    cn = np.zeros(n, dtype=float)
    cp[1:] = np.cumsum(pos)
    cn[1:] = np.cumsum(neg)
    idx = np.arange(n)
    g += cp[idx] + cn[n - 1 - idx]
    if wc != 0.0:
        g -= 2.0 * wc * h
    combo = pos + neg
    nz = np.nonzero(combo)[0]
    for kk in nz:
        d = int(kk) + 1
        wd = float(combo[kk])
        m = n - d
        g[:m] -= wd * h[d:]
        g[d:] -= wd * h[:m]
    return g


def smooth_obj(x, n, alpha):
    c = correlate_comp(x)
    m = float(np.max(c))
    z = alpha * (c - m)
    return float((m + np.log(np.sum(np.exp(z))) / alpha) * (2.0 / n))


def smooth_grad(x, n, alpha):
    c = correlate_comp(x)
    m = float(np.max(c))
    p = np.exp(alpha * (c - m))
    s = float(np.sum(p))
    if s <= 0.0:
        w = np.full(c.size, 1.0 / c.size, dtype=float)
    else:
        w = p / s
    return (2.0 / n) * grad_corr_weights(x, w, n)


def penalty_obj(x, n, alpha, pen):
    c = correlate_comp(x)
    m = float(np.max(c))
    z = alpha * (c - m)
    base = (m + np.log(np.sum(np.exp(z))) / alpha) * (2.0 / n)
    d = float(np.sum(x) - 0.5 * n)
    return float(base + 0.5 * pen * d * d)


def penalty_grad(x, n, alpha, pen):
    c = correlate_comp(x)
    m = float(np.max(c))
    p = np.exp(alpha * (c - m))
    s = float(np.sum(p))
    if s <= 0.0:
        w = np.full(c.size, 1.0 / c.size, dtype=float)
    else:
        w = p / s
    g = (2.0 / n) * grad_corr_weights(x, w, n)
    d = float(np.sum(x) - 0.5 * n)
    g += pen * d
    return g


def lbfgs_phase(h, n, alpha, pen, maxiter):
    h = project_feasible(h, n)
    try:
        res = minimize(penalty_obj, h, args=(n, alpha, pen), method="L-BFGS-B",
                       jac=penalty_grad, bounds=[(0.0, 1.0)] * n,
                       options={"maxiter": maxiter, "ftol": 1e-13})
        if res is not None and np.all(np.isfinite(res.x)):
            h2 = res.x
        else:
            h2 = h
    except Exception:
        h2 = h
    return project_feasible(h2, n)


def slsqp_phase(h, n, alpha, maxiter):
    h = project_feasible(h, n)
    cons = [{"type": "eq", "fun": sum_eq, "jac": sum_jac, "args": (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method="SLSQP",
                       jac=smooth_grad, constraints=cons,
                       bounds=[(0.0, 1.0)] * n,
                       options={"maxiter": maxiter, "ftol": 1e-13})
        if res is not None and np.all(np.isfinite(res.x)):
            h2 = res.x
        else:
            h2 = h
    except Exception:
        h2 = h
    return project_feasible(h2, n)


def build_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n), dtype=float)
    G[n - 1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        rp = n - 1 + d
        rn = n - 1 - d
        m = n - d
        G[rp, d:] += 1.0
        G[rn, :m] += 1.0
        hp = h[d:]
        hm = h[:m]
        G[rp, :m] -= hp
        G[rn, :m] -= hp
        G[rp, d:] -= hm
        G[rn, d:] -= hm
    return G

        if time.time() > deadline:

        cap = caps[min(cap_i, len(caps) - 1)]
        c = correlate_comp(h)
        G = build_grad_matrix(h, n)
        m = 2 * n - 1
        A = np.empty((m, n + 1), dtype=float)
        A[:, :n] = G
        A[:, -1] = -1.0
        obj = np.zeros(n + 1, dtype=float)
        obj[-1] = 1.0
        Aeq = np.zeros((1, n + 1), dtype=float)
        Aeq[0, :n] = 1.0
        bnd = []
        for v in h:
            vf = float(v)
            bnd.append((max(-vf, -cap), min(1.0 - vf, cap)))
        bnd.append((0.0, float(np.max(c))))
        delta = None

    if deadline - time.time() > 60:
        alloc = min(260.0, (deadline - time.time()) * 0.30)
        h2, c2 = polish_candidate(h, n, time.time() + alloc, rng, True)
        if c2 < bc:
            bh, bc = h2, c2
    x0 = np.linspace(0.0, 2.0, bn)
    for nt in (240, 384, 512, 768):
        if nt == bn:
            continue
        rem = deadline - time.time()
        if rem < 160:
            break
        hy = project_feasible(np.interp(np.linspace(0.0, 2.0, nt), x0, bh), nt)
        cy = c5_eval(hy)
        if cy < bc + 0.01:
            alloc = min(170.0, rem * 0.25)
            if alloc < 70:
                continue
            h2, c2 = polish_candidate(hy, nt, time.time() + alloc, rng,
                                      cy <= bc + 0.001)
            if c2 < bc:
                bh, bc, bn = h2, c2, nt
                x0 = np.linspace(0.0, 2.0, bn)
    while time.time() < deadline - 80:
        scale = 0.02 + 0.03 * float(rng.rand())
        hp = project_feasible(bh + rng.normal(0.0, scale, bn), bn)
        alloc = min(90.0, (deadline - time.time()) * 0.30)
        if alloc < 45:
            break
        h2, c2 = polish_candidate(hp, bn, time.time() + alloc, rng, False)
        if c2 < bc:
            bh, bc = h2, c2
    rem = deadline - time.time()
    if rem > 60:
        alloc = min(130.0, rem)
        h2, c2 = polish_candidate(bh, bn, min(deadline - 5, time.time() + alloc), rng, True)
        if c2 < bc:
            bh, bc = h2, c2
    bh = project_feasible(bh, bn)
    bc = c5_eval(bh)
    return bh, bc, bn
```
