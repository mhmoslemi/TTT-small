import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        exp_vals = np.exp(shifted - sm)
        logsum = sm + np.log(exp_vals.sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    dx = 2.0 / n
    h = to_binary(h_init, n).copy()

    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] = np.clip(k1, 0, 2*n-2)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] = np.clip(k2, 0, 2*n-2)

    g = np.correlate(x, 1.0 - x, mode='full') * dx  # length 2n-1
    M = g.max()

    sm = shifted.max()
    exp_vals = np.exp(shifted - sm)
    p = exp_vals / exp_vals.sum()  # softmax weights, length 2n-1
    val = M + np.log(exp_vals.sum()) / alpha

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 3:
            break
        h = slsqp_smooth(h, n, alpha, 8000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 3: binary annealing
    p3 = t0 + 0.53 * total
    ha, ca = anneal_binary(best_h, n, p3, seed=seed, n_iters=3000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    # Phase 4: SLSQP from annealed
    p4 = t0 + 0.72 * total
    h = best_h.copy()
    for alpha in [1e5, 1e6, 1e7, 1e8, 5e8]:
        if time.time() > p4 - 3:
            break
        h = slsqp_smooth(h, n, alpha, 8000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 5: try other resolutions
    p5 = t0 + 0.88 * total
    for nh in [300, 500, 600]:
        if time.time() > p5 - 8:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hq, cq = greed_binary(hh, nh, p5 - 6, max_sweeps=80)
        if cq < compute_c5(hh, nh):
            hh = hq
        for alpha in [1e5, 1e6, 1e7]:
            if time.time() > p5 - 4:
                break
            hh = slsqp_smooth(hh, nh, alpha, 5000, 1e-18)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
    # Phase 6: final polish with perturbation restarts
    p6 = dl - 2
    if time.time() < p6:
        for trial in range(8):
            if time.time() > p6 - 1:
                break
            h = best_h + 0.002 * rng.standard_normal(n)
            h = project_feasible(h, n)
            for alpha in [5e7, 5e8]:
                if time.time() > p6 - 0.5:
                    break
                h = slsqp_smooth(h, n, alpha, 10000, 1e-18)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
