def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n/2.0
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask): break
            h[mask] += d / float(np.count_nonzero(mask))
        else:
            mask = h > 1e-12
            if not np.any(mask): break

def build_alpha(h, dx):
    n = len(h); m=2*n-1; z=n-1
    A = np.zeros((n,m))
    for idx in range(m):
        s = idx - z
        if s < 0:
            lo = -s; hi = n-1
        else:
            lo = 0; hi = n-1-s
        A[lo:hi+1, idx] += 1.0
        A[lo:hi+1, idx] -= h[lo+s:hi+s+1]
        pl = lo+s; ph = hi+s
        A[pl:ph+1, idx] -= h[lo:hi+1]
    return A * dx

import time
import numpy as np
from scipy.optimize import linprog

def compute_g(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, float(c.max())

def compute_c5(h, n):
    return compute_g(h, n)[1]

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += d / float(np.count_nonzero(mask))
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] -= (-d) / float(np.count_nonzero(mask))
    out = np.clip(h, 0.0, 1.0)
    d = target - out.sum()
    if abs(d) > 1e-11:
        idx = np.argsort(out)[:max(1, int(0.02 * n))] if d > 0 else np.argsort(out)[-max(1, int(0.02 * n)):][::-1]
        out[idx] += d / float(len(idx))
        out = np.clip(out, 0.0, 1.0)
    return out

def build_alpha(h, dx):
    n = len(h)
    m = 2 * n - 1
    z = n - 1
    A = np.zeros((n, m))
    for idx in range(m):
        s = idx - z
        lo = -s if s < 0 else 0
        hi = n - 1 if s < 0 else n - 1 - s
        A[lo:hi + 1, idx] += 1.0
        A[lo:hi + 1, idx] -= h[lo + s:hi + s + 1]
        pl = lo + s
        ph = hi + s
        A[pl:ph + 1, idx] -= h[lo:hi + 1]
    return A * dx

def lp_direction(g, alpha, h, C_current):
    try:
        n = len(h)
        m = alpha.shape[1]
        A_ub = np.column_stack((alpha.T, -np.ones(m)))
        b_ub = -g
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        lo = -h
        hi = 1.0 - h
        bounds = list(zip(lo, hi))
        bounds.append((0.0, max(1.0, C_current)))
        res = linprog(A_ub, b_ub, A_eq=A_eq, b_eq=b_eq, c=c, bounds=bounds, method='highs')
        if res is None or res.status != 0:
            return None, None
        delta = res.x[:n].astype(float).copy()
        z = float(res.x[n])
        if z > C_current - 1e-11:
            return None, None
        delta -= delta.mean()
        delta = np.clip(delta, -h - 1e-9, 1.0 - h + 1e-9)
        delta -= delta.mean()
        if np.max(np.abs(delta)) < 1e-14:
            return None, None
        return delta, z
    except Exception:
        return None, None

def line_search_delta(h, delta, target, C_current, deadline):
    n = len(h)
    best_c = C_current
    best_lam = 0.0
    best_h = None
    base_lams = (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1, 3e-1, 1.0)
    for lam in base_lams:
        if time.time() > deadline:
            break
        hnew = np.clip(h + lam * delta, 0.0, 1.0)
        if abs(target - hnew.sum()) > 1e-8:
            hnew = project_feasible(hnew, n)
        c = compute_c5(hnew, n)
        if c < best_c - 1e-12:
            best_c = c
            best_lam = float(lam)
            best_h = hnew.copy()
    if best_h is None:
        return None, C_current
    for _ in range(2):
        if time.time() > deadline:
            break
        if best_lam <= 0.01:
            lo = 0.0
            hi = min(1.0, max(0.05, best_lam * 3.0 + 0.01))
        elif best_lam >= 0.9:
            lo = max(0.0, best_lam - 0.2)
            hi = 1.0
        else:
            lo = max(0.0, best_lam - 0.2)
            hi = min(1.0, best_lam + 0.2)
        if hi <= lo:
            break
        lams = np.linspace(lo, hi, 25)
        for lam in lams:
            hnew = np.clip(h + lam * delta, 0.0, 1.0)
            if abs(target - hnew.sum()) > 1e-8:
                hnew = project_feasible(hnew, n)
            c = compute_c5(hnew, n)
            if c < best_c - 1e-12:
                best_c = c
                best_lam = float(lam)
                best_h = hnew.copy()
        if hi - lo < 1e-7:
            break
    return best_h, best_c

def pair_sweep(h, g, alpha, dx, C, deadline, rng, K=18, chunk=6000):

    m = 2 * n - 1
    z = n - 1
    if n < 2:
        return h, C
    pi, pj = np.triu_indices(n, 1)
    total = len(pi)
    if total > 60000:
        order = rng.choice(total, 20000, replace=False)
    else:
        order = np.arange(total)
        rng.shuffle(order)
    active = np.argsort(g)[-K:]
    ga = g[active].copy()
    Aa = alpha[:, active]
    zero_pos = -1
    special = []
    for pos, idx in enumerate(active):
        if idx == z:
            zero_pos = pos
        else:
            d = abs(idx - z)
            if d < n:
                special.append((pos, d))
    best_h = h.copy()
    best_c = C
    small = (1e-4, 2e-4, 5e-4, 1e-3, 2e-3, 5e-3, 1e-2, 2e-2, 5e-2, 1e-1, 2e-1, 5e-1)
    for start in range(0, len(order), chunk):
        if time.time() > deadline - 2:
            break
        idxs = order[start:start + chunk]
        i = pi[idxs]
        j = pj[idxs]
        a = np.maximum(-h[i], h[j] - 1.0)
        b = np.minimum(1.0 - h[i], h[j])
        valid = (b - a) > 1e-12
        if not np.any(valid):
            continue
        c1 = Aa[i] - Aa[j]
        P = i.size

            c2[:, zero_pos] = -2.0 * dx
        dvec = np.abs(i - j)

            c2[:, pos] += (dvec == d).astype(float) * dx
        best_sur = np.full(P, C, dtype=float)
        best_t = np.zeros(P, dtype=float)

        for s in small:
            tvecs.append(np.clip(-s, a, b))
            tvecs.append(np.clip(s, a, b))
        for tv in tvecs:
            val = ga + c1 * tv[:, None] + c2 * tv[:, None] * tv[:, None]
            mv = val.max(axis=1)
            better = valid & (mv + 1e-12 < best_sur)
            if np.any(better):
                best_sur[better] = mv[better]
                best_t[better] = tv[better]
        cand = np.where(valid & (best_sur < C - 1e-9))[0]

            ref = cand[np.argpartition(best_sur[cand], 25)[:25]]
        else:
            ref = cand
        for idx in ref:
            ai = a[idx]
            bi = b[idx]
            cent = best_t[idx]
            span = max(1e-3, 0.1 * abs(bi - ai))
            lo = max(ai, cent - span)
            hi = min(bi, cent + span)
            if hi > lo:
                tt = np.linspace(lo, hi, 17)
                mv = (ga + c1[idx][None, :] * tt[:, None] + c2[idx][None, :] * tt[:, None] * tt[:, None]).max(axis=1)
                bidx = int(np.argmin(mv))
                if mv[bidx] < best_sur[idx] - 1e-12:
                    best_sur[idx] = mv[bidx]
                    best_t[idx] = float(tt[bidx])
        local_best_c = best_c
        li = None
        lj = None
        lt = None
        for idx in cand:
            t = best_t[idx]
            if abs(t) < 1e-14:
                continue
            ii = int(i[idx])
            jj = int(j[idx])
            d = abs(ii - jj)
            c1f = alpha[ii] - alpha[jj]
            tmp = g + c1f * t
            q = t * t
            tmp[z] += -2.0 * dx * q
            ip = z + d
            if ip < m:
                tmp[ip] += dx * q
            im = z - d
            if im >= 0:
                tmp[im] += dx * q
            full = float(tmp.max())
            if full < local_best_c - 1e-12:
                local_best_c = full
                li = ii
                lj = jj
                lt = float(t)
        if li is not None:
            hnew = h.copy()
            hnew[li] += lt
            hnew[lj] -= lt
            hnew = np.clip(hnew, 0.0, 1.0)
            diff = n / 2.0 - hnew.sum()
            if abs(diff) > 1e-11:
                hnew = project_feasible(hnew, n)
            cfull = compute_c5(hnew, n)
            if cfull < best_c - 1e-12:
                best_c = cfull
                best_h = hnew
                return best_h, best_c
    return h, C

def random_perturb(h, n, rng, strength=0.05):
    h2 = h.copy()
    h2 += strength * rng.randn(n)
    for _ in range(max(4, int(0.1 * n))):
        ii = rng.randint(n)
        jj = rng.randint(n)
        if ii != jj:
            h2[ii], h2[jj] = h2[jj], h2[ii]
    return project_feasible(h2, n)

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return project_feasible(t.astype(float), n)

def resample_h(h, new_n):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    return project_feasible(np.interp(x_new, x_old, h), new_n)

def improve_local(h0, n, deadline, seed, max_perturb=0):
    h = project_feasible(h0, n)
    g, C = compute_g(h, n)
    best_h = h.copy()
    best_c = C
    dx = 2.0 / n
    target = n / 2.0
    rng = np.random.RandomState(seed)
    pert = 0
    fail = 0
    while time.time() < deadline:
        if C < best_c - 1e-13:
            best_c = C
            best_h = h.copy()
        alpha = build_alpha(h, dx)
        improved = False
        if time.time() < deadline - 3:
            delta, z = lp_direction(g, alpha, h, C)
            if delta is not None:
                hnew, cnew = line_search_delta(h, delta, target, C, deadline)
                if hnew is not None and cnew < C - 1e-11:
                    h = hnew
                    g, C = compute_g(h, n)
                    improved = True
                    fail = 0
        if not improved and time.time() < deadline - 4:
            hnew, Cn = pair_sweep(h, g, alpha, dx, C, deadline, rng, K=18, chunk=6000)
            if Cn < C - 1e-11:
                h = hnew
                g, C = compute_g(h, n)
                improved = True
                fail = 0
        if improved:
            continue
        if C > best_c + 0.002 and fail > 2:
            break
        if pert < max_perturb:
            h = random_perturb(h, n, rng, 0.04)
            g, C = compute_g(h, n)
            pert += 1
            fail = 0
        else:
            break
    best_h = project_feasible(best_h, n)
    bc = compute_c5(best_h, n)
    if bc < best_c:
        best_c = bc
        best_h = h.copy()
    return best_h, float(best_c)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    usable = max(1, int(budget_s) - 5)
    deadline = t0 + usable
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = 1e9
    best_n = 0
    have_parent = False
    try:
        ph = np.array(initial_h_values, dtype=float).copy()
        if ph.size:
            have_parent = True
            best_n = len(ph)
            best_h = project_feasible(ph, best_n)
            best_c = compute_c5(best_h, best_n)
    except Exception:
        pass
    if have_parent:
        tlim = t0 + 0.50 * usable
        h1, c1 = improve_local(best_h, best_n, min(deadline, tlim), seed, 0)
        if c1 < best_c - 1e-13:
            best_c = float(c1)
            best_h = h1
    else:
        best_n = 250
        best_h = thue_morse_init(best_n, rng)
        best_c = compute_c5(best_h, best_n)
        tlim = t0 + 0.50 * usable
        h1, c1 = improve_local(best_h, best_n, min(deadline, tlim), seed, 0)
        if c1 < best_c - 1e-13:
            best_c = float(c1)
            best_h = h1
    if time.time() < deadline - 80 and best_h is not None:
        ns = []
        for nn in (best_n - 4, best_n - 8, best_n + 4, best_n + 8, 240, 200, 252, 260, 300):
            if 20 <= nn <= 600 and nn != best_n and nn not in ns:
                ns.append(nn)
        max_cand = 3 if usable > 900 else 2
        k = 0
        for nn in ns:
            if time.time() > deadline - 70 or k >= max_cand:
                break
            rem = deadline - time.time() - 10
            if rem < 40:
                break
            tlim = time.time() + min(rem * 0.45, 180.0)
            if abs(nn - best_n) <= 16:
                hstart = resample_h(best_h, nn)
            else:
                hstart = thue_morse_init(nn, rng)
            h1, c1 = improve_local(hstart, nn, min(deadline, tlim), seed + 100 + k, 0)
            if c1 < best_c - 1e-13:
                best_c = float(c1)
                best_h = h1
                best_n = nn
            k += 1
    if best_h is not None and time.time() < deadline - 60:
        ncur = len(best_h)
        for r in range(2):
            rem = deadline - time.time() - 10
            if rem < 70:
                break
            hstart = random_perturb(best_h, ncur, rng, 0.05)
            tlim = time.time() + min(rem * 0.40, 150.0)
            h1, c1 = improve_local(hstart, ncur, min(deadline, tlim), seed + 1000 + r, 0)
            if c1 < best_c - 1e-13:
                best_c = float(c1)
                best_h = h1
                best_n = ncur
    if best_h is None:
        best_n = 250
        best_h = thue_morse_init(best_n, rng)
        best_c = compute_c5(best_h, best_n)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
