```python
def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

```python
def optimize_coordinate(h, n, i):
    """Find optimal h[i] given all other coordinates fixed."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    ks = np.arange(-(n-1), n)
    ik = i + ks
    ikn = i - ks
    valid_k = (ik >= 0) & (ik < n)
    valid_kn = (ikn >= 0) & (ikn < n)
    
    v = h[i]
    h_ik = np.where(valid_k, h[ik[valid_k]], 0.0)
    h_ikn = np.where(valid_kn, h[ikn[valid_kn]], 0.0)
    
    c_k = (1.0 - h_ik) - h_ikn  # coefficient of v for each k
    A_k = corr[n-1+ks] - v*(1.0-h_ik) - h_ikn*(1.0-v)  # constant term
    
    B_0 = corr[n-1] - v*(1.0-v)
    
    def f(v_test):
        vals = A_k + c_k * v_test
        vals[n-1] = v_test*(1.0-v_test) + B_0  # k=0 is at index n-1
        return vals.max()
    
    # Find minimum: try grid + refine
    grid = np.linspace(0, 1, 51)
    fvals = np.array([f(v) for v in grid])
    i_best = fvals.argmin()
    
    # Refine with golden section
    lo, hi = grid[max(0,i_best-1)], grid[min(50,i_best+1)]
    for _ in range(50):
        m1 = lo + 0.382*(hi-lo)
        m2 = lo + 0.618*(hi-lo)
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    best_v = (lo + hi) / 2
    return best_v
```
