import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def compute_profiles(h, n):
    """Compute profile[p, k] for all p, k.
    profile[p, k] = h[p - d_k] + h[p + d_k] where d_k = k - (n-1), out-of-bounds = 0."""
    # d ranges from -(n-1) to (n-1), mapped to k = 0 to 2n-2
    d = np.arange(-(n-1), n)  # length 2n-1
    # For each p, profile[p, :] = h[p-d] (with bounds) + h[p+d] (with bounds)
    profiles = np.zeros((n, 2*n-1))
    for p in range(n):
        # h[p - d] for valid indices
        idx1 = p - d  # indices into h
        valid1 = (idx1 >= 0) & (idx1 < n)
        profiles[p, valid1] += h[idx1[valid1]]
        # h[p + d] for valid indices
        idx2 = p + d
        valid2 = (idx2 >= 0) & (idx2 < n)
        profiles[p, valid2] += h[idx2[valid2]]
    return profiles
