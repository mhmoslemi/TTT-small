import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def slsqp_opt(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = (g - m) / alpha
        return m + np.logaddexp.reduce(sh)
    bounds = [(0.0, 1.0)] * n
    cons = ({"type": "eq", "fun": lambda x: x.sum() - n / 2.0},)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def grad_fd(h, n, w_full):
    """Compute gradient using vectorized loop over lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(len(w_full)):
        wv = w_full[k_idx]
        if wv < 1e-10:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[:n-k] += wv * (1.0 - h[k:])
            grad[k:] -= wv * h[:n-k]
        elif k < 0:
            kk = -k
            grad[kk:] += wv * (1.0 - h[:n-kk])
            grad[:n-kk] -= wv * h[kk:]
        else:
            grad += wv * (1.0 - 2.0 * h)
    return grad * dx

def proj_grad_desc(h, n, deadline, lr_init=0.005, tau_init=0.05):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    tau = tau_init
    stall = 0
    
    while time.time() < deadline:
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        sh = (g - m) / tau
        w_full = np.exp(sh)
        w_full /= w_full.sum()
        
        grad = grad_fd(h, n, w_full)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.05, 0.05)
        else:
            h = h_new  # still accept to explore
            stall += 1
            if stall % 50 == 0:
                lr = max(lr * 0.95, 1e-4)
                tau = max(tau * 0.98, 0.001)
    
    return best_h, best_c

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.02, t_end=1e-5, patience=25000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    T = t_start
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 500000.0)
    
    while time.time() < deadline:
        h_old = h.copy()
        use_block = rng.random() >= 0.6
        if not use_block:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
        else:
            L = rng.choice([2, 3, 4, 5])
            s1 = rng.randint(0, n - L)
            s2 = rng.randint(0, n - L)
            if abs(s1 - s2) < L:
                T *= cool_rate
                continue
            tmp = h[s1:s1+L].copy()
            h[s1:s1+L] = h[s2:s2+L]
            h[s2:s2+L] = tmp
        
        new_corr = np.correlate(h, 1 - h, mode='full').astype(float)
        nc = new_corr.max() * dx
        dc = nc - cur_c
        
        if dc < 0 or (T > 1e-14 and rng.random() < np.exp(-dc / T)):
            corr = new_corr
            cur_c = nc
            if nc < best_c - 1e-13:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h = h_old
            no_improve += 1
        
        T *= cool_rate
        if no_improve > patience:
            T = t_start * 0.4
            no_improve = 0
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            npert = rng.randint(5, 15)
            for _ in range(npert):
                pi = zeros[rng.randint(len(zeros))]
                pj = ones[rng.randint(len(ones))]
                h[pi], h[pj] = 1.0, 0.0
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
            cur_c = corr.max() * dx
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Quick SLSQP
    p1 = t0 + 0.05 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 5000, 50000]:
        if time.time() > p1:
            break
        h = slsqp_opt(h, n, alpha, 1500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Binary SA (main)
    p2 = t0 + 0.45 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    for attempt in range(4):
        at_end = p2 if attempt < 3 else p2
        if attempt == 0:
            sh = h_bin
            s = seed
        elif attempt == 1:
            sh = round_to_binary(best_h + 0.15 * rng.randn(n), n)
            s = seed + 100
        elif attempt == 2:
            sh = round_to_binary(best_h + 0.3 * rng.randn(n), n)
            s = seed + 200
        else:
            sh = round_to_binary(best_h + 0.5 * rng.randn(n), n)
            s = seed + 300
        # Split time among attempts
        chunk_end = t0 + (0.05 + 0.45 * (attempt+1)/4) * (budget_s - 5)
        hb, cb = binary_sa(sh, n, min(chunk_end, at_end), seed=s,
                         t_start=0.025, t_end=1e-5, patience=20000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 3: Projected gradient descent (new!)
    p3 = t0 + 0.78 * (budget_s - 5)
    h_start = project_feasible(best_h + 0.02 * rng.randn(n), n)
    hg, cg = proj_grad_desc(h_start, n, p3, lr_init=0.003, tau_init=0.08)
    if cg < best_c:
        best_c = cg
        best_h = hg.copy()
    
    # Phase 4: SLSQP polish from gradient result
    p4 = t0 + 0.88 * (budget_s - 5)
    if time.time() < p4 - 10:
        h = project_feasible(best_h + 0.01 * rng.randn(n), n)
        for alpha in [10000, 100000, 1000000]:
            if time.time() > p4:
                break
            h = slsqp_opt(h, n, alpha, 3000)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 5: Final SA + gradient from SLSQP result
    if time.time() < dl - 15:
        h_bin2 = round_to_binary(best_h, n)
        hb2, cb2 = binary_sa(h_bin2, n, dl - 15, seed=seed+400,
                            t_start=0.015, t_end=5e-5, patience=15000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    
    if time.time() < dl - 5:
        hg2, cg2 = proj_grad_desc(project_feasible(best_h + 0.01*rng.randn(n), n),
                                  n, dl - 3, lr_init=0.002, tau_init=0.05)
        if cg2 < best_c:
            best_c = cg2
            best_h = hg2.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
