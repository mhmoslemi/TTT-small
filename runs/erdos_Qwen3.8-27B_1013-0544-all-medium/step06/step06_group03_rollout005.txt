import numpy as np
import time
import math
from scipy.optimize import minimize

def project_feasible(y, n):
    y = np.asarray(y, dtype=float).reshape(-1)
    if y.size < n:
        y = np.concatenate([y, np.zeros(n - y.size)])
    elif y.size > n:
        y = y[:n]
    if not np.all(np.isfinite(y)):
        y = np.full(n, 0.5)
    S = 0.5 * float(n)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(y - mid, 0.0, 1.0))) > S:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        d = S - float(z.sum())
        if abs(d) < 1e-13:
            break
        if d > 0:
            i = int(np.argmax(1.0 - z))
            z[i] += min(d, 1.0 - z[i])
        else:
            i = int(np.argmin(z))
            z[i] += max(d, -z[i])
    return z

def raw_lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h, n):
    return float(np.max(raw_lags(h, n)) * (2.0 / float(n)))

def _add(G, h, lag, w):
    if w == 0.0:
        return
    n = len(h)
    if lag == 0:
        G += w * (1.0 - 2.0 * h)
    else:
        m = abs(lag)
        if lag > 0:
            G[m:] += w * (1.0 - h[:-m])
            G[:n - m] += -w * h[m:]
        else:
            G[:n - m] += w * (1.0 - h[m:])
            G[m:] += -w * h[:-m]

def _weighted_lag_gradient(h, n, idx, weights):
    G = np.zeros(n)
    cen = n - 1
    for i in range(len(idx)):
        w = weights[i]
        if w > 1e-15:
            _add(G, h, int(idx[i]) - cen, w)
    return G

def _soft_obj(x, n, alpha, idx):
    c = raw_lags(x, n)
    ac = c if idx is None else c[idx]
    m = float(ac.max())
    sh = float(alpha) * (ac - m)
    mx = float(sh.max())
    s = float(np.sum(np.exp(sh - mx)))
    return m + (mx + math.log(s)) / float(alpha)

def _soft_jac(x, n, alpha, idx):
    c = raw_lags(x, n)
    ac = c if idx is None else c[idx]
    m = float(ac.max())
    sh = float(alpha) * (ac - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    w = e / float(e.sum())
    G = np.zeros(n)
    cen = n - 1
    if idx is None:
        for j in range(2 * n - 1):
            if w[j] > 1e-15:
                _add(G, x, j - cen, w[j])
    else:
        for j in range(len(idx)):
            if w[j] > 1e-15:
                _add(G, x, int(idx[j]) - cen, w[j])
    return G

def _sum_eq(x, n):
    return float(np.sum(x) - 0.5 * n)

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp_smooth(h, n, alpha, idx, maxiter=None, ftol=1e-9):
    h = project_feasible(h, n)
    if maxiter is None:
        if idx is None:
            if n > 700:
                mi = 28
            elif n > 450:
                mi = 50
            else:
                mi = 85
        else:
            if n > 700:
                mi = 28
            elif n > 450:
                mi = 45
            else:
                mi = 70
    else:
        mi = int(maxiter)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_soft_obj, h, args=(n, alpha, idx), jac=_soft_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': mi, 'ftol': ftol})

        return h
    if res.x is None or not np.all(np.isfinite(res.x)):

    return project_feasible(res.x, n)

def _sum_eq_z(z, n):
    return float(np.sum(z[:-1]) - 0.5 * n)

def _sum_jac_z(z, n):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def _nlp_obj(z, n):
    return float(z[-1])

def _nlp_obj_jac(z, n):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _nlp_ieq(z, n, idx):
    h = z[:-1]
    return raw_lags(h, n)[idx] - z[-1]

def _nlp_ieq_jac(z, n, idx):
    h = z[:-1]
    n = len(h)
    J = np.empty((len(idx), n + 1))
    cen = n - 1
    for r in range(len(idx)):
        G = np.zeros(n)
        _add(G, h, int(idx[r]) - cen, 1.0)
        J[r, :n] = G
        J[r, n] = -1.0
    return J

def active_nlp(h, n, idx, maxiter=None, ftol=1e-9):
    h = project_feasible(h, n)
    idx = np.asarray(idx)
    if maxiter is None:

    z[-1] = t
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n) + 1.0)]

        {'type': 'ineq', 'fun': _nlp_ieq, 'jac': _nlp_ieq_jac, 'args': (n, idx)}
    ]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': mi, 'ftol': ftol})
    except Exception:
        return h
    if res.x is None or not np.all(np.isfinite(res.x)):
        return h
    return project_feasible(res.x[:-1], n)

def choose_variant(start, n):
    orig = project_feasible(start, n)
    best = orig.copy()
    bc = c5_of(best, n)
    for mode in (1, 2, 3):
        if mode == 1:
            cand = 1.0 - orig

            cand = orig[::-1]
        else:

        cc = c5_of(cand, n)
        if cc < bc - 1e-12:
            best = cand
            bc = cc
    return best

    best_c = c5_of(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(120.0, float(budget_s) - 45.0)
    rng = np.random.RandomState(seed)

    try:
        init = np.asarray(initial_h_values, dtype=float).reshape(-1)
        n0 = len(init)
    except Exception:
        init = None
        n0 = 300

    if n0 < 50:
        init = None
        n0 = 300

    if init is None:
        best_h = default_h(n0)
    else:
        best_h = project_feasible(init, n0)
    best_n = n0
    best_c = c5_of(best_h, best_n)

    rem = deadline - time.time() - 60
    if rem > 60:
        alloc = min(320.0, max(60.0, rem * 0.35))
        h_opt, c_opt = optimize_n(best_h, best_n, time.time() + alloc, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        else:
            best_h = project_feasible(best_h, best_n)

    extra = [400, 450, 500, 600, 720, 840]
    if best_n != 300:
        extra = [300] + extra
    done = {best_n}

    for n_try in extra:
        rem = deadline - time.time() - 35
        if rem < 90:
            break
        if n_try in done or n_try < 150 or n_try > 900:
            continue
        if n_try < best_n and n_try != 300:
            continue
        alloc = min(240.0, max(70.0, rem * 0.30))
        start = resample_h(best_h, best_n, n_try)
        h_opt, c_opt = optimize_n(start, n_try, time.time() + alloc, rng)
        done.add(n_try)
        if c_opt < best_c:
            best_h, best_n, best_c = h_opt, n_try, c_opt

    if time.time() < deadline - 25:
        rem = deadline - time.time()
        if rem > 60:
            h2, c2 = pair_polish(best_h, best_n, min(deadline - 1, time.time() + max(25.0, rem * 0.35)), rng)
            if c2 < best_c:
                best_h, best_c = h2, c2
        if time.time() < deadline - 20 and best_n <= 650:
            rc = raw_lags(best_h, best_n)
            idx = np.argsort(rc)[-24:]
            h2 = slsqp_smooth(best_h, best_n, 3000.0, idx, maxiter=40)
            c2 = c5_of(h2, best_n)
            if c2 < best_c:
                best_h, best_c = h2, c2
        if time.time() < deadline - 12:
            h2, c2 = pair_polish(best_h, best_n, deadline - 1, rng)
            if c2 < best_c:
                best_h, best_c = h2, c2

    best_h = project_feasible(best_h, best_n)
    best_c = c5_of(best_h, best_n)
    return best_h, best_c, best_n
