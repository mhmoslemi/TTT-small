            room = 1.0 - h
            tot = room.sum()

        else:

            if tot < 1e-15: break
            h += (diff / tot) * h

def grad_smooth_max(h, n, alpha):
    c = compute_c5_arr(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        if w[m] < 1e-13: continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 5.0), (2000, 20.0), (10000, 100.0),
               (100000, 500.0), (1000000, 2000.0), (5000000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2: break
        try:
            res = minimize(lambda x: _obj(x, n, alpha, mu), h,
                          jac=lambda x: _grad(x, n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters*10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new; h = h_new
        except Exception:
            pass
    return h, best_c

def _obj(x, n, alpha, mu):
    c = compute_c5_arr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx[:target]] = 1.0

def compute_c5_arr(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15: break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_arr(h, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

        else:

            g[kk:] += wm * (1.0 - h[:nv]); g[:nv] += wm * (-h[kk:])

def flip_delta_vec(h, n, i, delta):
    """Vectorized computation of correlation change when h[i] changes by delta."""
    dc = np.zeros(2 * n - 1)
    # For m >= n-1 (k >= 0): k = m - (n-1), ranges 0 to n-1
    k = np.arange(n)
    # Condition 1: k <= i (for the h[i-k] term)
    c1 = k <= i
    m1 = (n - 1) + k[c1]
    dc[m1] += delta * (1.0 - h[i - k[c1]])
    # Condition 2: i + k < n (for the h[i+k] term), i.e., k < n - i
    c2 = k < (n - i)
    m2 = (n - 1) + k[c2]
    dc[m2] += delta * (-h[i + k[c2]])
    # For m < n-1 (k < 0): p = (n-1) - m, ranges 1 to n-1
    p = np.arange(1, n)
    # Condition: i + p < n, i.e., p < n - i
    c1n = p < (n - i)
    m1n = (n - 1) - p[c1n]
    dc[m1n] += delta * (1.0 - h[i + p[c1n]])
    # Condition: p <= i (for the h[i-p] term)
    c2n = p <= i
    m2n = (n - 1) - p[c2n]
    dc[m2n] += delta * (-h[i - p[c2n]])
    return dc
