import time

def compute_c(h, n):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_c(h, n)) * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = compute_c(h, n) * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c(x, n) * (2.0 / n)
    mm = c.max()
    s = alpha * (c - mm)
    mx = s.max()
    val = mm + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_refine(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 10.0), (2000, 50.0), (10000, 200.0), (50000, 500.0), (200000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-14,
                                   'maxfun': max_iters * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def exact_pair_search(h, n, c, deadline, max_rounds=10000):
    dx = 2.0 / n
    best_c5 = float(np.max(c) * dx)
    m_arr = np.arange(2 * n - 1)
    stagnation = 0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        alpha = 200.0
        g = grad_smooth_max(h, n, alpha)

        # Score pairs
        diff_g = np.abs(g[:, None] - g[None, :])
        np.fill_diagonal(diff_g, 0.0)
        n_pairs = n * (n - 1) // 2
        k_pairs = min(n_pairs, 8000)

        flat_sorted = np.argsort(diff_g.ravel())[::-1]
        pairs = []
        seen = set()
        for idx in flat_sorted[:k_pairs * 3]:
            i, j = divmod(int(idx), n)
            if i == j:
                continue
            key = (min(i, j), max(i, j))
            if key not in seen:
                seen.add(key)
                pairs.append(key)
            if len(pairs) >= k_pairs:
                break

        best_improvement = 0.0
        best_pair = None
        best_delta = 0.0

        for i, j in pairs:
            idx1 = i - m_arr
            v1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i + m_arr
            v2 = (idx2 >= 0) & (idx2 < n)

            a = np.zeros(2 * n - 1)
            if v1.any():
                a[v1] += (1.0 - h[idx1[v1]])
            if v2.any():
                a[v2] -= h[idx2[v2]]

            idx1j = j - m_arr
            v1j = (idx1j >= 0) & (idx1j < n)
            idx2j = j + m_arr
            v2j = (idx2j >= 0) & (idx2j < n)
            if v1j.any():
                a[v1j] -= (1.0 - h[idx1j[v1j]])
            if v2j.any():
                a[v2j] += h[idx2j[v2j]]

            if np.abs(a).max() < 1e-15:
                continue

            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue

            lo, hi = d_lo, d_hi
            for _ in range(35):
                t1 = lo + (hi - lo) / 3.0
                t2 = hi - (hi - lo) / 3.0
                f1 = np.max(c + a * t1)
                f2 = np.max(c + a * t2)
                if f1 <= f2:
                    hi = t2
                else:
                    lo = t1
            opt_d = (lo + hi) / 2.0
            opt_c5 = float(np.max(c + a * opt_d) * dx)

            improvement = best_c5 - opt_c5
            if improvement > best_improvement:
                best_improvement = improvement
                best_pair = (i, j)
                best_delta = opt_d

        if best_pair is not None and best_improvement > 1e-11:
            i, j = best_pair
            h[i] += best_delta
            h[j] -= best_delta
            h = np.clip(h, 0.0, 1.0)
            c = compute_c(h, n)
            best_c5 = float(np.max(c) * dx)
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 15:
                # Perturb and retry
                rng = np.random.RandomState(rnd)
                h_pert = h + rng.randn(n) * 0.01
                h_pert = project_feasible(h_pert, n)
                c_pert = compute_c(h_pert, n)
                c5_pert = float(np.max(c_pert) * dx)
                if c5_pert < best_c5 + 0.005:
                    h = h_pert
                    c = c_pert
                    best_c5 = float(np.max(c) * dx)
                stagnation = 0
                k_pairs = n_pairs  # try all pairs next round
            if stagnation > 40:
                break

    return h, best_c5

    if c1 < best_c:
        best_c = c1
        best_h = h.copy()

    # Phase 2: Exact pair descent (main phase)
    c = compute_c(h, n)
    t2 = deadline - 150
    h, c2 = exact_pair_search(h, n, c, t2, max_rounds=50000)
    if c2 < best_c:
        best_c = c2
        best_h = h.copy()

    # Phase 3: L-BFGS on result
    if time.time() < deadline - 30:
        h3, c3 = lbfgs_refine(best_h, n, deadline - 20, max_iters=150)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: More pair descent
    if time.time() < deadline - 10:
        c4 = compute_c(best_h, n)
        h4, c4v = exact_pair_search(best_h, n, c4, deadline - 5, max_rounds=50000)
        if c4v < best_c:
            best_c = c4v
            best_h = h4.copy()

    # Phase 5: Try different n
    if time.time() < deadline - 40:
        for n_try in [200, 250, 350]:
            if time.time() > deadline - 20:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            c_i = compute_c(h_interp, n_try)
            ci5 = float(np.max(c_i) * (2.0 / n_try))
            # Quick pair descent
            c_end = min(time.time() + 40, deadline - 10)
            h_t, ct5 = exact_pair_search(h_interp, n_try, c_i, c_end, max_rounds=20000)
            if ct5 < best_c:
                best_c = ct5
                best_h = h_t.copy()
                n = n_try

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
