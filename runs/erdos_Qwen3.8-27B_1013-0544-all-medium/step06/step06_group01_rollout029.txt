import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, w, n):
    """Compute gradient of smooth max."""
    a = 1.0 - h
    grad = np.zeros(n)
    for p in range(n):
        k_min = max(0, n - 1 - p)
        k_max = min(2 * n - 2, n - 1 + p)
        if k_max >= k_min:
            ks = np.arange(k_min, k_max + 1)
            grad[p] += np.dot(w[ks], a[p + ks - (n-1)])
        k_min2 = max(0, p)
        k_max2 = min(2 * n - 2, p + n - 1)
        if k_max2 >= k_min2:
            ks2 = np.arange(k_min2, k_max2 + 1)
            grad[p] -= np.dot(w[ks2], h[p - ks2 + (n-1)])
    return grad

def adam_opt(h0, n, alpha, lr, max_iter, deadline, lam=50.0):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_h, best_c = h.copy(), compute_c5(h, n)
    
    for t in range(1, max_iter + 1):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        sh = alpha * (c - c.max())
        mx = sh.max()
        if mx > 500:
            w = np.zeros_like(c)
            w[mx > 400] = 1.0
            w /= w.sum()  # Hard max
        else:
            e = np.exp(sh - mx)
            w = e / e.sum()
        
        g = smooth_grad(h, w, n)
        g += 2.0 * lam * (h.sum() - n / 2.0)
        
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1**t)
        vhat = v / (1 - b2**t)
        h_new = np.clip(h - lr * mhat / (np.sqrt(vhat) + eps), 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def coord_descent(h, n, grid, max_passes, deadline, rng=None, penalty=5.0):
    # Same as before but cleaner
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    target = n / 2.0
    sumh = h.sum()
    
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        idx = np.arange(n)
        if rng: rng.shuffle(idx)
        for p in idx:
            dlo, dhi = -h[p], 1.0 - h[p]
            if dhi - dlo < 1e-14: continue
            klo = max(0, min(p, n-1-p))
            khi = min(2*n-2, max(p+n-1, 2*n-2-p))
            um = 0.0
            if klo > 0: um = max(um, c[:klo].max())
            if khi < 2*n-2: um = max(um, c[khi+1:].max())
            ks = np.arange(klo, khi+1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p+n-1)
            bwd = (ks >= n-1-p) & (ks <= 2*n-2-p)
            if fwd.any(): s[fwd] += 1.0 - h[p - ks[fwd] + n-1]
            if bwd.any(): s[bwd] -= h[p + ks[bwd] - (n-1)]
            
            best_d, best_v = 0.0, um
            for d in np.linspace(dlo, dhi, grid):
                if abs(d) < 1e-14: continue
                nc = c[ks] + d * s
                ci = (ks == n-1)
                if ci.any(): nc[ci] = c[n-1] + d*(1-2*h[p]) - d*d
                val = max(um, nc.max())
                if penalty: val += penalty * (sumh + d - target)**2
                if val < best_v - 1e-13:
                    best_v, best_d = val, d
            if abs(best_d) > 1e-12:
                for id in np.where(fwd)[0]:
                    c[ks[id]] += best_d * (1.0 - h[p - ks[id] + n-1])
                for id in np.where(bwd)[0]:
                    c[ks[id]] -= best_d * h[p + ks[id] - (n-1)]
                c[n-1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0, 1)
                sumh += best_d
                improved = True
        if not improved: break
    return h

def pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                lo, hi = max(0, i-radius), min(n, i+radius+1)
                for j in range(lo, hi):
                    if i == j: continue
                    # Try +d to i, -d to j
                    if h[i]+d <= 1 and h[j]-d >= 0:
                        h[i] += d; h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try -d to i, +d to j
                    if h[i]-d >= 0 and h[j]+d <= 1:
                        h[i] -= d; h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved: break
    return h, best_c

    n1 = n - 1
    grad = np.zeros(n)
    for p in range(n):
        k_lo1 = max(0, p)
        k_hi1 = min(2 * n1, n1 + p)
        if k_hi1 >= k_lo1:
            ks = np.arange(k_lo1, k_hi1 + 1)
            grad[p] += np.dot(w[ks], a[p - ks + n1])
        k_lo2 = max(0, n1 - p)
        k_hi2 = min(2 * n1, 2 * n1 - p)
        if k_hi2 >= k_lo2:
            ks = np.arange(k_lo2, k_hi2 + 1)
            grad[p] -= np.dot(w[ks], h[p + ks - n1])
    return grad

def adam_opt(h0, n, alpha, lr, max_iter, deadline, lam=50.0):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_h, best_c = h.copy(), compute_c5(h, n)
    stall = 0
    for t in range(1, max_iter + 1):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        sh = alpha * (c - c.max())
        mx = sh.max()
        if mx > 400:
            w = np.zeros_like(c)
            mask = sh > mx - 1.0
            w[mask] = 1.0
            w /= w.sum()
        else:
            e = np.exp(sh - mx)
            w = e / e.sum()
        g = smooth_grad(h, w, n)
        g += 2.0 * lam * (h.sum() - n / 2.0)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        h_new = np.clip(h - lr * mhat / (np.sqrt(vhat) + eps), 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 2000:
            break
    return best_h, best_c

def coord_descent(h, n, grid, max_passes, deadline, rng=None, penalty=5.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    target = n / 2.0
    sumh = float(h.sum())
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        idx = np.arange(n)
        if rng:
            rng.shuffle(idx)
        for p in idx:
            dlo, dhi = -h[p], 1.0 - h[p]
            if dhi - dlo < 1e-14:
                continue
            klo = max(0, min(p, n - 1 - p))
            khi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if klo > 0:
                um = max(um, float(c[:klo].max()))
            if khi < 2 * n - 2:
                um = max(um, float(c[khi + 1:].max()))
            ks = np.arange(klo, khi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if fwd.any():
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if bwd.any():
                s[bwd] -= h[p + ks[bwd] - (n - 1)]
            best_d, best_v = 0.0, um
            for d in np.linspace(dlo, dhi, grid):
                if abs(d) < 1e-14:
                    continue
                nc = c[ks] + d * s
                ci = (ks == n - 1)
                if ci.any():
                    nc[ci] = c[n - 1] + d * (1 - 2 * h[p]) - d * d
                val = max(um, float(nc.max()))
                if penalty:
                    val += penalty * (sumh + d - target) ** 2
                if val < best_v - 1e-13:
                    best_v, best_d = val, d
            if abs(best_d) > 1e-12:
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += best_d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= best_d * h[p + ks[ids_b] - (n - 1)]
                c[n - 1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh += best_d
                improved = True
        if not improved:
            break
    return h

def pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        progress = True
        while progress and time.time() < deadline:
            progress = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            progress = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            progress = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def optimize_block(h_start, n, t_end, rng):
    h = project_feasible(h_start, n)
    c = compute_c5(h, n)
    if time.time() < t_end - 80:
        h_a, c_a = adam_opt(h, n, alpha=5000, lr=0.02, max_iter=8000,
                            deadline=min(t_end - 100, time.time() + 150), lam=50.0)
        if c_a < c:
            h, c = h_a, c_a
    if time.time() < t_end - 50:
        h = coord_descent(h, n, grid=80, max_passes=3000,
                         deadline=min(t_end - 60, time.time() + 150), rng=rng, penalty=5.0)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
    if time.time() < t_end - 40:
        h_a, c_a = adam_opt(h, n, alpha=50000, lr=0.005, max_iter=5000,
                            deadline=min(t_end - 40, time.time() + 80), lam=100.0)
        if c_a < c:
            h, c = h_a, c_a
    if time.time() < t_end - 15:
        h = coord_descent(h, n, grid=120, max_passes=2000,
                         deadline=min(t_end - 20, time.time() + 80), rng=rng, penalty=0.5)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
    if time.time() < t_end - 5:
        h, c = pair_search(h, n, t_end - 2,
                           [0.01, 0.005, 0.001, 0.0005], radius=40)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Heavy optimization on initial n
    t1 = min(t0 + 350, deadline - 200)
    h, c = optimize_block(h_init, n_init, t1, rng)
    if c < best_c:
        best_h, best_c = h, c
    # Phase 2: Try different n values
    n_candidates = [200, 250, 350, 150]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        alloc = min(180, (deadline - time.time()) * 0.3)
        if alloc < 60:
            break
        t_end = time.time() + alloc
        h, c = optimize_block(h_interp, n_try, t_end, rng)
        if c < best_c:
            best_h, best_c = h, c
    # Phase 3: Multi-restart from best with varied noise
    if time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.02 + 0.05 * (r / max(1, n_restarts))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(65, (deadline - time.time() - 25) * 0.5)
            if alloc < 30:
                break
            h, c = optimize_block(hp, n, time.time() + alloc, rng)
            if c < best_c:
                best_h, best_c = h, c
    # Phase 4: Final aggressive polishing
    if time.time() < deadline - 20:
        n = len(best_h)
        h_f = coord_descent(best_h, n, grid=150, max_passes=3000,
                           deadline=min(deadline - 10, time.time() + 60), rng=rng, penalty=0.1)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, deadline - 2,
                                     [0.0005, 0.0002, 0.0001, 0.00005], radius=50)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
