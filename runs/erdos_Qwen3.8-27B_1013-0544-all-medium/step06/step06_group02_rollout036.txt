import time
import math
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        if h.size == 0:
            h = np.full(n, 0.5)
        else:
            h = np.interp(np.linspace(0.0, 1.0, n), np.linspace(0.0, 1.0, h.size), h)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    for _ in range(60):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h[mask] += diff * room[mask] / tot
        else:
            mask = h > 0.0
            if not mask.any():
                break
            tot = float(h[mask].sum())
            if tot <= 1e-15:
                break
            h[mask] += diff * h[mask] / tot
    for _ in range(25):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            idx = int(np.flatnonzero(mask)[0])
            val = h[idx] + diff
            if 0.0 <= val <= 1.0:
                h[idx] = val
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c_values(h, n):
    q = 1.0 - h
    c = np.empty(2 * n - 1, dtype=float)
    base = n - 1
    c[base] = float(np.dot(h, q))
    for k in range(1, n):
        nv = n - k
        c[base + k] = float(np.dot(h[:nv], q[k:]))
        c[base - k] = float(np.dot(h[k:], q[:nv]))
    return c


def harness_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def gradient_from_weight_pairs(h, w, indices, n):
    g = np.zeros(n, dtype=float)
    base = n - 1
    for pos, m in enumerate(indices):
        wm = float(w[pos])
        if wm <= 1e-15:
            continue
        k = int(m) - base
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += wm * (1.0 - h[k:])
                g[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += wm * (1.0 - h[:nv])
                g[:nv] -= wm * h[kk:]
    return g


def gradient_top(h, c, n, top_count=64, alpha=50.0):
    L = len(c)
    tc = min(top_count, max(1, L))
    idx = np.argsort(c)[-tc:]
    cmax = float(c.max())
    z = np.exp(np.clip(alpha * (c[idx] - cmax), -30.0, 0.0))
    s = float(z.sum())
    w = np.zeros(tc)
    if s > 0.0:
        w = z / s
    return gradient_from_weight_pairs(h, w, idx, n)


def _smooth_obj(x, n, alpha, mu):
    c = compute_c_values(x, n)
    cmax = float(c.max())
    z = np.exp(np.clip(alpha * (c - cmax), -700.0, 0.0))
    val = cmax + math.log(float(z.sum())) / alpha
    val += mu * float((x.sum() - 0.5 * n) ** 2)
    return float(val)


def _smooth_grad(x, n, alpha, mu):
    c = compute_c_values(x, n)
    cmax = float(c.max())
    z = np.exp(np.clip(alpha * (c - cmax), -700.0, 0.0))
    s = float(z.sum())
    g = np.zeros(n)
    if s > 0.0:
        w = z / s
        idx = np.nonzero(w > 1e-9)[0]
        if idx.size:
            g = g + gradient_from_weight_pairs(x, w[idx], idx, n)
    if mu:
        g += 2.0 * mu * (float(x.sum()) - 0.5 * n)
    return g


def _eq_fun(x, n):
    return float(x.sum() - 0.5 * n)


def _eq_jac(x, n):
    return np.ones(n)


def refine_lbfgs(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    c = compute_c_values(h, n)
    best = float(c.max()) * (2.0 / n)
    if time.time() >= deadline:
        return best_h, best
    bounds = [(0.0, 1.0)] * n
    scale = 1.0 if n <= 350 else (0.75 if n <= 700 else 0.55)
    configs = (
        (20.0, 1e-4, int(45 * scale)),
        (50.0, 1e-3, int(45 * scale)),
        (200.0, 1e-2, int(35 * scale)),
        (1000.0, 0.05, int(25 * scale)),
        (3000.0, 0.2, int(18 * scale)),
        (8000.0, 1.0, int(14 * scale)),
    )
    for alpha, mu, mx in configs:
        if time.time() > deadline - 6.0:
            break
        if mx < 8:
            mx = 8
        try:
            res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': mx, 'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxfun': max(25, mx * 6), 'maxls': 12})
            h2 = project_feasible(res.x, n)
            v2 = float(compute_c_values(h2, n).max()) * (2.0 / n)
            if v2 < best - 1e-13:
                h = h2
                best = v2
                best_h = h2.copy()
            elif v2 - best < 1e-7:
                h = h2
        except Exception:
            pass
    c = compute_c_values(h, n)
    v = float(c.max()) * (2.0 / n)
    if v < best - 1e-12:
        return h, v
    return best_h, best


def refine_slsqp(h, n, deadline, alpha=500.0, maxiter=80):
    if n > 500 or time.time() > deadline - 15.0:
        c = compute_c_values(h, n)
        return h, float(c.max()) * (2.0 / n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best = float(compute_c_values(h, n).max()) * (2.0 / n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(n, alpha, 0.0),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h2 = project_feasible(res.x, n)
        v2 = float(compute_c_values(h2, n).max()) * (2.0 / n)
        if v2 < best - 1e-13:
            return h2, v2
    except Exception:
        pass
    c = compute_c_values(h, n)
    v = float(c.max()) * (2.0 / n)
    if v < best - 1e-12:
        return h, v
    return best_h, best


def random_smooth_start(n, rng, amp):
    x = np.linspace(0.0, 1.0, n)
    f = np.zeros(n)
    mx = 1e-12
    for freq in (1, 2, 3, 5, 8):
        f = f + rng.normal(0.0, 1.0) * np.sin(2.0 * math.pi * freq * (x - 0.5) + rng.uniform(0.0, 2.0 * math.pi))
        m = float(np.max(np.abs(f)))
        if m > mx:
            mx = m
    h = np.clip(0.5 + amp * f / mx, 0.0, 1.0)
    return project_feasible(h, n)


def interpolate_h(h, n_new):
    n = len(h)
    if n == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)


def select_candidate_scores(h, c, n, rng=None, low_size=60, active_lags=8, rand_pairs=28):
    L = len(c)
    top_count = min(36, max(1, L))
    top_idx = np.argsort(c)[-top_count:]
    cmax = float(c.max())
    g = gradient_top(h, c, n, 70, 50.0)
    best = {}

    S = max(4, min(60, n // 2))
    if 2 * S <= n:
        order = np.argsort(g)
        low = order[:S]
        high = order[-S:]
        for aa in low:
            a = int(aa)
            for bb in high:
                b = int(bb)
                if a == b:
                    continue
                if g[a] <= g[b]:
                    room = float(min(1.0 - h[a], h[b]))
                    diff = float(g[b] - g[a])
                else:
                    room = float(min(1.0 - h[b], h[a]))
                    diff = float(g[a] - g[b])
                if room > 1e-8 and diff > 1e-12:
                    key = (a, b) if a < b else (b, a)
                    score = diff * room
                    old = best.get(key)
                    if old is None or score > old:
                        best[key] = score

    act = top_idx[-min(active_lags, top_idx.size):]
    for mi in act:
        m = int(mi)
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            if nv <= 1:
                continue
            contrib = h[:nv] * (1.0 - h[k:])
            tc = min(4, nv)
            if tc > 1:
                idxs = np.argpartition(contrib, tc - 1)[-tc:]
            else:
                idxs = np.array([int(np.argmax(contrib))])
            for pos in idxs:
                ii = int(pos)
                jj = int(ii + k)
                if 0 <= jj < n and ii != jj:
                    room = float(min(1.0 - h[jj], h[ii]))
                    if room > 1e-8:
                        score = float(contrib[ii]) * room
                        key = (ii, jj) if ii < jj else (jj, ii)
                        old = best.get(key)
                        if old is None or score > old:
                            best[key] = score
        else:
            kk = -k
            nv = n - kk
            if nv <= 1:
                continue
            contrib = h[kk:] * (1.0 - h[:nv])
            tc = min(4, nv)
            if tc > 1:
                idxs = np.argpartition(contrib, tc - 1)[-tc:]
            else:
                idxs = np.array([int(np.argmax(contrib))])
            for pos in idxs:
                pos = int(pos)
                ii = pos + kk
                jj = pos
                if 0 <= ii < n and 0 <= jj < n and ii != jj:
                    room = float(min(1.0 - h[jj], h[ii]))
                    if room > 1e-8:
                        score = float(contrib[pos]) * room
                        key = (ii, jj) if ii < jj else (jj, ii)
                        old = best.get(key)
                        if old is None or score > old:
                            best[key] = score

    if rng is not None and rand_pairs > 0 and n > 2:
        for _ in range(rand_pairs):
            a = int(rng.randint(n))
            b = int(rng.randint(n))
            if a == b:
                continue
            if g[a] <= g[b]:
                room = float(min(1.0 - h[a], h[b]))
                diff = float(g[b] - g[a])
            else:
                room = float(min(1.0 - h[b], h[a]))
                diff = float(g[a] - g[b])
            if room > 1e-8 and diff > 1e-12:
                key = (a, b) if a < b else (b, a)
                score = 0.5 * diff * room
                old = best.get(key)
                if old is None or score > old:
                    best[key] = score

    items = []
    for key, score in best.items():
        items.append((float(score), int(key[0]), int(key[1])))
    items.sort(reverse=True)
    return items


def candidate_line_search(h, c, i, j, best_val, samples, max_move):
    n = int(len(h))
    hi = float(h[i])
    hj = float(h[j])
    lo = max(-hi, hj - 1.0)
    hi_b = min(1.0 - hi, hj)
    if hi_b - lo < 1e-14:
        return None

    lo_k = min(min(-i, i - n + 1), min(-j, j - n + 1))
    hi_k = max(max(n - 1 - i, i), max(n - 1 - j, j))
    if lo_k < -(n - 1):
        lo_k = -(n - 1)
    if hi_k > n - 1:
        hi_k = n - 1

    aff = []
    A = []
    B = []
    C0 = []
    for k in range(lo_k, hi_k + 1):
        aff_k = (0 <= i + k < n) or (0 <= i - k < n) or (0 <= j + k < n) or (0 <= j - k < n)
        if not aff_k:
            continue
        a = 0.0
        if 0 <= i + k < n:
            a += 1.0 - h[i + k]
        if 0 <= i - k < n:
            a -= h[i - k]
        if 0 <= j + k < n:
            a -= 1.0 - h[j + k]
        if 0 <= j - k < n:
            a += h[j - k]
        b = 0.0
        if k == j - i and 0 <= i + k < n:
            b -= 1.0
        if k == i - j and 0 <= j + k < n:
            b += 1.0
        idx = n - 1 + k
        aff.append(idx)
        A.append(a)
        B.append(b)
        C0.append(float(c[idx]))

    if not aff:
        return None
    aff_idx = np.array(aff, dtype=int)
    A = np.array(A, dtype=float)
    B = np.array(B, dtype=float)
    C0 = np.array(C0, dtype=float)

    if aff_idx.size == c.size:
        base = -np.inf
    else:
        tmp = c.copy()
        tmp[aff_idx] = -np.inf
        base = float(tmp.max())

    l = max(lo, -max_move)
    r = min(hi_b, max_move)
    if r - l < 1e-15:
        return None

    ts = np.linspace(l, r, int(max(7, samples)))
    if l <= 0.0 <= r:
        ts = np.unique(np.concatenate((ts, np.array([0.0]))))

    best_f = np.inf
    best_d = 0.0
    for _ in range(3):
        t2 = ts * ts
        vals = C0[:, None] + A[:, None] * ts[None, :] + B[:, None] * t2[None, :]
        f = np.maximum(base, vals.max(axis=0))
        idx = int(np.argmin(f))
        fv = float(f[idx])
        if fv < best_f:
            best_f = fv
            best_d = float(ts[idx])
        if best_f >= best_val - 1e-13:
            break
        if len(ts) > 2:
            if idx == 0:
                l2 = l
                r2 = 0.5 * (l + float(ts[1]))
            elif idx == len(ts) - 1:
                l2 = 0.5 * (float(ts[-2]) + r)
                r2 = r
            else:
                l2 = float(ts[idx - 1])
                r2 = float(ts[idx + 1])
            if r2 - l2 < 1e-15:
                break
            ts = np.linspace(l2, r2, int(max(11, min(33, samples))))

    if not np.isfinite(best_f) or best_f >= best_val - 1e-13:
        return None

    d = best_d
    if d < lo:
        d = lo
    if d > hi_b:
        d = hi_b
    ni = hi + d
    nj = hj - d
    if ni < 0.0:
        d = -hi
        ni = 0.0
    elif ni > 1.0:
        d = 1.0 - hi
        ni = 1.0
    if nj < 0.0:
        d = hj
        nj = 0.0
    elif nj > 1.0:
        d = hj - 1.0
        nj = 1.0

    new_vals = C0 + A * d + B * d * d
    f_val = float(new_vals.max())
    if base > f_val:
        f_val = float(base)
    if not np.isfinite(f_val) or f_val >= best_val - 1e-13:
        return None
    return (d, f_val, aff_idx, A, B, C0)


def gradient_kick(h, c, n, best_val):
    g = gradient_top(h, c, n, 80, 50.0)
    for lr in (0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
        h2 = project_feasible(h - lr * g, n)
        c2 = compute_c_values(h2, n)
        v = float(c2.max())
        if v < best_val - 1e-13:
            return h2, c2, v, True
    return h, c, best_val, False


def polish_pairs(h, n, deadline, samples=33, max_candidates=110):
    h = project_feasible(h, n).copy()
    c = compute_c_values(h, n)
    best_val = float(c.max())
    rng = np.random.RandomState(123456)
    mm = 0.02
    acc = 0
    while time.time() < deadline:
        if acc >= 12:
            c = compute_c_values(h, n)
            best_val = float(c.max())
            acc = 0
        cand = select_candidate_scores(h, c, n, rng, rand_pairs=24)
        if not cand:
            break
        progress = False
        lim = max_candidates if mm < 0.12 else max(30, max_candidates // 2)
        for item in cand[:lim]:
            if time.time() >= deadline:
                break
            _, i, j = item
            res = candidate_line_search(h, c, i, j, best_val, samples, mm)
            if res is not None:
                d, new_val, aff_idx, A, B, C0 = res
                h[i] += d
                h[j] -= d
                if h[i] < 0.0:
                    h[i] = 0.0
                elif h[i] > 1.0:
                    h[i] = 1.0
                if h[j] < 0.0:
                    h[j] = 0.0
                elif h[j] > 1.0:
                    h[j] = 1.0
                c[aff_idx] = C0 + A * d + B * d * d
                best_val = float(c.max())
                acc += 1
                progress = True
                if acc >= 12:
                    c = compute_c_values(h, n)
                    best_val = float(c.max())
                    acc = 0
                    break
        if not progress:
            h, c, best_val, kicked = gradient_kick(h, c, n, best_val)
            if kicked:
                mm = min(0.05, max(0.02, 0.5 * mm))
                acc = 0
                continue
            mm = min(1.0, 2.0 * mm + 0.02)
            if mm >= 1.0:
                h2 = project_feasible(h + 0.003 * rng.normal(size=n), n)
                c2 = compute_c_values(h2, n)
                if float(c2.max()) < best_val - 1e-14:
                    h = h2
                    c = c2
                    best_val = float(c2.max())
                    mm = 0.02
                else:
                    break
    h = project_feasible(h, n)
    c = compute_c_values(h, n)
    return h, float(c.max()) * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = max(60.0, float(budget_s))
    deadline = t0 + B - 8.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 <= 0:
            raise ValueError
    except Exception:
        n0 = 300
        h0 = random_smooth_start(n0, rng, 0.45)

    if n0 > 950:
        h0 = interpolate_h(h0, 650)
        n0 = 650
    if n0 < 60:
        h0 = interpolate_h(h0, 180)
        n0 = 180
    h0 = project_feasible(h0, n0)

    cands = []
    c = compute_c_values(h0, n0)
    cands.append((float(c.max()) * (2.0 / n0), h0.copy(), n0))

    hs = project_feasible(0.5 * (h0 + h0[::-1]), n0)
    c = compute_c_values(hs, n0)
    cands.append((float(c.max()) * (2.0 / n0), hs, n0))

    for amp in (0.35, 0.50):
        hr = random_smooth_start(n0, rng, amp)
        c = compute_c_values(hr, n0)
        cands.append((float(c.max()) * (2.0 / n0), hr, n0))

    bi = 0
    for ci in range(1, len(cands)):
        if cands[ci][0] < cands[bi][0]:
            bi = ci
    best_h, best_n, best_c = cands[bi][1], cands[bi][2], cands[bi][0]

    endA = t0 + min(0.22 * B, 240.0)
    endB = t0 + min(0.55 * B, 560.0)
    endC = t0 + min(0.80 * B, 880.0)

    lb_dead = min(endA - 70.0, deadline - 300.0)
    if lb_dead > time.time() + 12.0:
        best_h, best_c = refine_lbfgs(best_h, best_n, lb_dead)
    slq_dead = min(endA - 10.0, time.time() + 75.0, deadline - 300.0)
    if best_n <= 450 and slq_dead > time.time() + 18.0:
        best_h, best_c = refine_slsqp(best_h, best_n, slq_dead)

    pair_dead = min(endB, deadline - 360.0)
    if pair_dead > time.time() + 25.0:
        best_h, best_c = polish_pairs(best_h, best_n, pair_dead, samples=33, max_candidates=110)

    if time.time() < endC - 30.0:
        n2 = 600 if best_n < 600 else 900
        if n2 <= 1000 and n2 > best_n and (endC - time.time()) > 130.0:
            h2 = interpolate_h(best_h, n2)
            rem = endC - time.time()
            lb2 = min(time.time() + 0.45 * rem, endC - 25.0, deadline - 50.0)
            if lb2 > time.time() + 10.0:
                h2, c2 = refine_lbfgs(h2, n2, lb2)
            if time.time() < endC - 15.0:
                h2, c2 = polish_pairs(h2, n2, min(endC - 2.0, deadline - 20.0), samples=29, max_candidates=80)
            if c2 < best_c - 1e-13:
                best_h, best_n, best_c = h2, n2, c2

        if time.time() < endC - 35.0 and best_n > 260:
            n3 = max(150, int(0.62 * best_n))
            if n3 < best_n:
                h3 = interpolate_h(best_h, n3)
                rem = max(0.0, endC - time.time())
                if rem > 70.0:
                    lb3 = min(time.time() + 0.35 * rem, endC - 20.0, deadline - 30.0)
                    if lb3 > time.time() + 8.0:
                        h3, c3 = refine_lbfgs(h3, n3, lb3)
                    if time.time() < endC - 10.0:
                        h3, c3 = polish_pairs(h3, n3, min(endC - 2.0, deadline - 20.0), samples=27, max_candidates=70)
                    if c3 < best_c - 1e-13:
                        best_h, best_n, best_c = h3, n3, c3

    if time.time() < deadline - 8.0:
        best_h, best_c = polish_pairs(best_h, best_n, deadline - 3.0, samples=31, max_candidates=95)

    best_h = project_feasible(best_h, int(best_n))
    best_c = harness_c5(best_h, int(best_n))
    return best_h, float(best_c), int(best_n)
