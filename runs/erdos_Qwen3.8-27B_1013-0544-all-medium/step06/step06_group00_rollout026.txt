    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in alpha_schedule:
        if time.time() > deadline:
            break
        # Compute initial weights
        corr = np.correlate(h, 1-h, mode='full') * (2.0/n)
        shifted = alpha * corr
        max_s = shifted.max()
        weights = np.exp(shifted - max_s)
        weights /= weights.sum()

        f_curr = (max_s + np.log(weights.sum())) / alpha
        step = 0.1
        for it in range(max_iters):
            if time.time() > deadline:
                break
            grad = lse_grad(h, n, weights)

            for _ in range(20):

                h_trial = project_feasible(h_trial, n)
                # Recompute objective and weights

            f_curr = f_trial

            shifted = alpha * corr
            max_s = shifted.max()
            weights = np.exp(shifted - max_s)
            weights /= weights.sum()

            if c < best_c:

Actually, let me think about this more carefully. The LSE function f_α(h) = (1/α) log(Σ exp(α g_k(h))) is:
- Smooth (C∞)
- g_k(h) = Σ_i h[i](1-h[i+k']) * dx is bilinear in h (quadratic, actually)
- Not convex in general

So we can't guarantee global convergence, but local convergence should be fine with a good step size.

For the step size, a safe choice is step ≤ 1/L where L is the Lipschitz constant of the gradient. The gradient of g_k is bounded (each component is in [-1, 1] effectively, times dx). The Hessian of g_k involves second derivatives which are bounded. With the LSE weighting, the overall Lipschitz constant is bounded by some constant times α (roughly).

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
        n_steps += 1
    return best_h, best_c
