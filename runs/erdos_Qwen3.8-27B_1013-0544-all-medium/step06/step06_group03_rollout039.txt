Let me also add a "heuristic" local search that's more directed:
- Find the lag with the maximum overlap
- Try to reduce that specific lag by moving mass from "bad" positions to "good" positions
- This is more directed than random swaps

For the directed search:
- At the maximum lag m, we have overlap_m = sum h[i]*(1-h[i+m])
- To reduce this, we want to either:
  a) Decrease h[i] for positions where h[i+m]=0 (reduce the "bad" source)
  b) Increase h[i+m] for positions where h[i]=1 (fill in the "gap")

For binary solutions, both (a) and (b) mean moving a 1 from position i to position i+m (or vice versa), which is exactly a swap!

So the directed search is: for the worst lag m, try all swaps (i, i+m) where h[i]=1 and h[i+m]=0.

This is much more focused than random swaps. Let me implement this:

```python

        h_dir, c_dir = directed_ls(best_h, n, min(deadline - 2, time.time() + 60))
        if c_dir < best_c:
            best_c = c_dir
            best_h = h_dir
    # Phase D: Annealing LS
    if time.time() < deadline - 3:
        h_an, c_an = anneal_ls(best_h, n, deadline - 2, niter=300000, seed=999)
        if c_an < best_c:
            best_c = c_an
            best_h = h_an
    # Phase E: Polish with SLSQP then directed
    if time.time() < deadline - 5:
        hn = slsqp(best_h, n, 1000000, maxit=3000)
        cn = compute_c5(hn, n)
        if cn < best_c:
            best_c = cn
            best_h = hn
        h_dir2, c_dir2 = directed_ls(best_h, n, deadline - 1)
        if c_dir2 < best_c:
            best_c = c_dir2
            best_h = h_dir2
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    best_h = None
    best_c = float('inf')
    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    # Main optimization at initial n
    alloc1 = min(350, (deadline - t0) * 0.35)
    h1, c1 = full_pipeline(h_init, n_init, t0 + alloc1)
    best_h, best_c = h1, c1
    # Try symmetric counterpart
    h_sym = 1.0 - h1[::-1].copy()
    h_sym = project(h_sym, n_init)
    if compute_c5(h_sym, n_init) < best_c:
        best_h = h_sym
        best_c = compute_c5(h_sym, n_init)
    # Directed LS on symmetric
    h_ds, c_ds = directed_ls(h_sym, n_init, min(deadline - 50, time.time() + 60))
    if c_ds < best_c:
        best_c = c_ds
        best_h = h_ds
    # Try other n values
    for n_try in [200, 250, 400, 500]:
        if n_try % 2 != 0:
            continue
        if time.time() > deadline - 120:
            break
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project(h_start, n_try)
        alloc = min(150, (deadline - time.time() - 60) * 0.4)
        if alloc < 30:
            break
        h_new, c_new = full_pipeline(h_start, n_try, time.time() + alloc)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Final intensive local search on best
    if best_h is not None and time.time() < deadline - 10:
        h_an, c_an = anneal_ls(best_h, len(best_h), deadline - 2, niter=500000, seed=seed + 1)
        if c_an < best_c:
            best_c = c_an
            best_h = h_an
        h_dir, c_dir = directed_ls(best_h, len(best_h), deadline - 1)
        if c_dir < best_c:
            best_c = c_dir
            best_h = h_dir
    if best_h is None:
        best_h = project(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    best_h = project(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
