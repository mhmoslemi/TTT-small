class SmoothObj:

    def __call__(self,x):
       c=np.correlate(x,1.0-x,'full')
       m=float(c.max())
       if self.alpha <=1e-3: return m
       sh=self.alpha*(c-m)
       if sh.max()>700: sh-=sh.max()
       val = np.sum(np.exp(sh))
       return m + (np.log(val)/self.alpha) if val>0 else m
    def grad(self,x):
       c=np.correlate(x,1.0-x,'full')
       m=float(c.max())

       ex=np.exp(sh-mx)
       S=ex.sum()

import numpy as np

ESTEPS = (0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.85, 1.0)

def compute_c5(h,n):
    return float(np.max(np.correlate(h,1.0-h,'full'))*(2.0/n))

def max_corr(h,b):
    return float(np.max(np.correlate(h,b,'full')))

def project_box_sum(h,n):
    x=np.asarray(h,dtype=float)
    if x.size != n:
        x=np.resize(x,n)
    if not np.all(np.isfinite(x)):
        x=np.full(n,0.5)
    target=0.5*n
    x=x-(x.sum()-target)/n
    lo=float(np.min(x))-1.0; hi=float(np.max(x))+1.0
    for _ in range(70):
        mid=0.5*(lo+hi)
        s=float(np.sum(np.clip(x-mid,0.0,1.0)))
        if s>target: lo=mid
        else: hi=mid
    x=np.clip(x-0.5*(lo+hi),0.0,1.0)
    for _ in range(4):
        diff=target-x.sum()
        if abs(diff)<1e-13: break
        mask=(x>1e-12)&(x<1.0-1e-12)
        if mask.sum()>0:
            x[mask]+=diff/float(mask.sum())
        else:
            idx=int(np.argmax(1.0-x)) if diff>0 else int(np.argmax(x))
            x[idx]+=diff
        x=np.clip(x,0.0,1.0)
    return x

def round_binary(h,n):
    out=np.zeros(n)
    idx=np.argsort(h)[::-1][:max(0,n//2)]
    out[idx]=1.0
    return out

def thue_morse_binary(n):
    h=np.empty(n)
    for i in range(n):
        x=i; p=0
        while x:
            p ^= (x & 1)
            x >>= 1
        h[i]=1.0 if p==0 else 0.0
    return round_binary(h,n)

def random_binary(n,rng):
    return round_binary(rng.rand(n),n)

def add_candidate(lst,seen,i,j):
    i=int(i); j=int(j)
    if i==j: return False
    key=(i,j)
    if key in seen: return False
    seen.add(key); lst.append(key)
    return True

def gradient_from_c(h,c,top,alpha):
    n=h.size
    top=min(top,c.size)
    idx=np.argsort(c)[::-1][:top]
    m=float(c[idx[0]])
    sc=alpha*(c[idx]-m)
    sc=sc-sc.max()
    w=np.exp(sc)
    s=w.sum()
    if s<=0: w=np.zeros_like(w); w[0]=1
    else: w=w/s
    G=np.zeros(n); center=n-1
    for ii,ww in zip(idx,w):
        ww=float(ww)
        if ww<1e-8: continue
        d=int(ii)-center
        if d>0:
            mm=n-d
            G[d:]+=ww*(1.0-h[:mm])
            G[:mm]-=ww*h[d:]
        elif d<0:
            r=-d; mm=n-r
            G[:mm]+=ww*(1.0-h[r:])
            G[r:]-=ww*h[:mm]
        else:
            G+=ww*(1.0-2.0*h)
    return G

def eval_pair_move(base,i,j,u):

    h2=base.copy(); b2=1.0-h2
    bi=base[i]; bj=base[j]

    for f in ESTEPS:

        if e<=1e-12: continue
        h2[i]=bi+e; h2[j]=bj-e
        b2[i]=1.0-h2[i]; b2[j]=1.0-h2[j]
        m=max_corr(h2,b2)
        if m<best_m:
            best_m=m; best_e=e
    if best_e>0.0:
        fr0=best_e/u
        for f in (0.25*fr0,0.5*fr0,0.75*fr0,1.25*fr0,1.5*fr0,2.0*fr0):

            b2[i]=1.0-h2[i]; b2[j]=1.0-h2[j]
            m=max_corr(h2,b2)
            if m<best_m:
                best_m=m; best_e=e
    return best_e, best_m

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s))
    dl = t0 + total - 6
    if dl < t0 + 20:
        dl = t0 + max(5.0, total - 2)
    rng = np.random.RandomState(seed)

    try:
        arr = initial_h_values
        if arr is None:
            raise ValueError
        h = np.array(arr, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 250
        h = np.where((np.arange(n) * 0.6180339887498949) % 1.0 < 0.5, 1.0, 0.0)

    if n <= 50 or n > 1000 or n % 2 == 1:
        no = n if n > 0 else 250
        ho = np.array(h, dtype=float).copy() if n > 0 else np.full(no, 0.5)
        n = 250
        if no != n and no > 1:
            h = np.interp(
                np.linspace(0.0, 2.0, n, endpoint=False),
                np.linspace(0.0, 2.0, no, endpoint=False),
                ho
            )
        else:
            h = np.full(n, 0.5)

    h = project_box_sum(h, n)
    bh = h.copy()
    bc = c5val(bh, n)

    reserve = 90.0 if (n < 500 and total > 250) else 0.0
    mdl = dl - reserve
    if mdl <= t0 + 60:
        mdl = dl - 5
        reserve = 0.0
    span = mdl - t0
    p1 = t0 + 0.30 * span
    p2 = t0 + 0.42 * span
    p3 = t0 + 0.70 * span

    hh, cc = cont_pair(h, n, p1, seed=seed, top=40, alpha=25)
    if cc < bc - 1e-12:
        bh, bc = hh, cc

    if time.time() < p2 - 5:
        h2 = project_box_sum(bh + 0.02 * rng.randn(n), n)
        hh, cc = cont_pair(h2, n, p2, seed=seed + 7, top=64, alpha=40)
        if cc < bc - 1e-12:
            bh, bc = hh, cc

    cand = bh
    if time.time() < p3 - 5:
        hb, cb = binary_local(bh, n, p3, seed=seed + 13)
        if cb < bc + 1e-4:
            cand = hb
        if cb < bc - 1e-12:
            bh, bc = hb, cb

    if time.time() < mdl - 5:
        hh, cc = cont_pair(cand, n, mdl, seed=seed + 23, top=50, alpha=35)
        if cc < bc - 1e-12:
            bh, bc = hh, cc

    if reserve == 0.0 and time.time() < dl - 5:
        hh, cc = cont_pair(
            project_box_sum(bh + 0.01 * rng.randn(n), n),
            n, dl - 3, seed=seed + 199, top=70, alpha=60
        )
        if cc < bc - 1e-12:
            bh, bc = hh, cc

    if reserve > 0.0 and time.time() < dl - 8:
        n2 = 500
        if n2 % n == 0:
            h5 = np.repeat(bh, n2 // n)
        else:
            h5 = np.interp(
                np.linspace(0.0, 2.0, n2, endpoint=False),
                np.linspace(0.0, 2.0, n, endpoint=False),
                bh
            )
        h5 = project_box_sum(h5, n2)
        hh, cc = cont_pair(h5, n2, dl - 4, seed=seed + 301, top=40, alpha=30)
        if cc < bc - 1e-12:
            bh, bc, n = hh, cc, n2

    bh = project_box_sum(bh, n)
    bc = c5val(bh, n)
    return bh, bc, n
