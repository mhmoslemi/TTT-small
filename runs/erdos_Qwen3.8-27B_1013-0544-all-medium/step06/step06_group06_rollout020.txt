import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def build_delta(h, n):
    base = compute_corr(h)
    m = 2 * n - 1
    delta = np.zeros((n, m))
    for p in range(n):
        h2 = h.copy()
        h2[p] += 1.0
        delta[p] = compute_corr(h2) - base
    return delta

def project_binary_sum(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on threshold
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def continuous_optimize(h, n, deadline, alpha_start=100, alpha_end=10000, lr=0.01, n_iter_per_alpha=200):
    """Projected gradient on log-sum-exp approximation"""
    dx = 2.0 / n
    delta = build_delta(h, n)

        if time.time() > deadline:
            break

            # Step

            h = project_binary_sum(h, n)

def build_delta_fast(h, n):
    m = 2 * n - 1
    delta = np.zeros((n, m))
    center = n - 1
    # d = 0
    delta[:, center] = 1.0 - 2.0 * h
    # d > 0
    for d in range(1, n):
        # For each p: delta[p, center+d] = (1-h[p-d])*(p>=d) - h[p+d]*(p+d<n)
        p = np.arange(n)
        # p >= d: indices d to n-1
        idx_pos = np.arange(d, n)
        delta[idx_pos, center + d] = (1.0 - h[idx_pos - d])
        # p + d < n: indices 0 to n-1-d
        idx_neg = np.arange(0, n - d)
        delta[idx_neg, center + d] -= h[idx_neg + d]
        # Symmetric: delta[p, center-d]
        # For d' = d: delta[p, center-d] = (1-h[p+d])*(p+d<n) - h[p-d]*(p>=d)
        idx_pos2 = np.arange(0, n - d)
        delta[idx_pos2, center - d] = (1.0 - h[idx_pos2 + d])
        idx_neg2 = np.arange(d, n)
        delta[idx_neg2, center - d] -= h[idx_neg2 - d]
    return delta
```

def find_best_swap(corr, delta, ones, zeros):
    m = len(corr)
    # A[i, k] = corr[k] + delta[zeros[i], k]
    A = corr[None, :] + delta[zeros]  # (n_zeros, m)
    # B[j, k] = delta[ones[j], k]
    B = delta[ones]  # (n_ones, m)
    # V[i, j] = max_k(A[i,k] - B[j,k])
    # = max_k of (A[i:None,:] - B[None,j,:])
    V = np.max(A[:, None, :] - B[None, :, :], axis=2)  # (n_zeros, n_ones)
    flat = V.argmin()
    ii, jj = flat // len(ones), flat % len(ones)
    return zeros[ii], ones[jj], V[ii, jj]
```

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def build_delta(h, n):
    m = 2 * n - 1
    delta = np.zeros((n, m))
    center = n - 1
    delta[:, center] = 1.0 - 2.0 * h
    for d in range(1, n):
        idx_pos = np.arange(d, n)
        delta[idx_pos, center + d] = 1.0 - h[idx_pos - d]
        idx_neg = np.arange(0, n - d)
        delta[idx_neg, center + d] -= h[idx_neg + d]
        idx_p2 = np.arange(0, n - d)
        delta[idx_p2, center - d] = 1.0 - h[idx_p2 + d]
        idx_n2 = np.arange(d, n)
        delta[idx_n2, center - d] -= h[idx_n2 - d]
    return delta

def find_best_swap(corr, delta, ones, zeros):
    A = corr[None, :] + delta[zeros]
    B = delta[ones]
    V = np.max(A[:, None, :] - B[None, :, :], axis=2)
    flat = V.argmin()
    jj = flat % len(ones)
    ii = flat // len(ones)
    return zeros[ii], ones[jj], V[ii, jj]

def greedy_improve(h, n, deadline, seed=42):
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    stall = 0
    no_improve = 0

    while time.time() < deadline:
        delta = build_delta(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i, j, best_val = find_best_swap(corr, delta, ones, zeros)
        new_c = best_val * dx

        if new_c < cur_c - 1e-12:
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta[i] - delta[j]
            cur_c = corr.max() * dx
            stall = 0
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 40:
                ri = rng.randint(len(zeros))
                rj = rng.randint(len(ones))
                ii = zeros[ri]
                jj = ones[rj]
                h[ii], h[jj] = h[jj], h[ii]
                corr = compute_corr(h)
                cur_c = corr.max() * dx
                stall = 0
                if cur_c < best_c - 1e-12:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                if no_improve > 500:
                    break

    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def continuous_optimize(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.astype(float).copy()
    dx = 2.0 / n
    alphas = [500, 2000, 8000, 30000, 100000, 300000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        iters = min(500, max(100, int((deadline - time.time()) * 3)))
        lr = 1.0 / (alpha * dx * np.sqrt(n))
        for it in range(iters):
            if time.time() > deadline - 1:
                break
            corr = compute_corr(h)
            shifted = alpha * (corr - corr.max())
            w = np.exp(shifted)
            w /= w.sum()
            delta = build_delta(h, n)
            grad = dx * (delta.T @ w)
            h = h - lr * grad
            h = project_feasible(h, n)
            lr *= 0.999
    return h

def binary_round(h_cont, n):
    target = n // 2
    h_bin = (h_cont > 0.5).astype(float)
    cur_ones = int(h_bin.sum())
    if cur_ones > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h_cont[idx])
        h_bin[idx[order[:cur_ones - target]]] = 0
    elif cur_ones < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(h_cont[idx])[::-1]
        h_bin[idx[order[:target - cur_ones]]] = 1
    return h_bin

def thue_morse_init(n, seed):
    rng = np.random.RandomState(seed)
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    deadline = t0 + total

    try:

    best_h = h0.copy()

    # Phase 1: Greedy from parent (35% time)
    d1 = t0 + 0.35 * total
    if time.time() < d1:
        hb, cb = greedy_improve(best_h, n, d1, seed=seed)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 2: Continuous optimization + rounding (20% time)
    d2 = t0 + 0.55 * total
    if time.time() < d2:
        h_cont = continuous_optimize(best_h, n, d2 - 20, seed=seed)
        h_bin = binary_round(h_cont, n)
        c = compute_c5(h_bin, n)
        if c < best_c:
            best_c = c
            best_h = h_bin.copy()
        if time.time() < d2 - 5:
            hb, cb = greedy_improve(h_bin, n, d2 - 5, seed=seed + 500)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    # Phase 3: Greedy from Thue-Morse (15% time)
    d3 = t0 + 0.70 * total
    if time.time() < d3:
        h_tm = thue_morse_init(n, seed + 1)
        hb, cb = greedy_improve(h_tm, n, d3, seed=seed + 1000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 4: Extended greedy polish on best (20% time)
    d4 = t0 + 0.90 * total
    if time.time() < d4:
        hb, cb = greedy_improve(best_h, n, d4, seed=seed + 2000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 5: Final micro-polish
    if time.time() < deadline - 3:
        hb, cb = greedy_improve(best_h, n, deadline - 3, seed=seed + 3000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
