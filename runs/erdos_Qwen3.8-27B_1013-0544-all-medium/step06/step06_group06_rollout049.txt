                wi = w[idx_i]
                if L >= 0:
                    p1 = np.arange(L, n)
                    g[p1] += wi * (1.0 - h[p1 - L])
                    if n - L > 0:
                        p2 = np.arange(0, n - L)
                        g[p2] -= wi * h[p2 + L]
                else:
                    M = -L
                    if n - M > 0:
                        p1 = np.arange(0, n - M)
                        g[p1] += wi * (1.0 - h[p1 + M])
                    p2 = np.arange(M, n)
                    g[p2] -= wi * h[p2 - M]
            
            # Gradient step
            h_new = project_hypersimplex(h - eta * g, target)
            c_new = compute_c5(h_new, n)
            
            if c_new < curr_val * dx:
                h = h_new
                eta = min(eta * 1.1, 1.0)
            else:
                eta = max(eta * 0.5, 0.001)
            
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        
        # Try rounding and re-optimizing
        h_rounded = (h > 0.5).astype(float)
        # Fix count
        num_ones = int(h_rounded.sum())
        if num_ones > target:
            thresh = h[h_rounded == 1]
            order = np.argsort(thresh)[:num_ones - int(target)]
            h_rounded[np.where(h_rounded == 1)[0][order]] = 0
        elif num_ones < target:
            thresh = h[h_rounded == 0]
            order = np.argsort(thresh)[::-1][:int(target) - num_ones]
            h_rounded[np.where(h_rounded == 0)[0][order]] = 1
        # Continue from rounded
        h = project_hypersimplex(h_rounded + 0.005 * np.random.randn(n), target)
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
