def binary_from_continuous(h, n):

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_g(h, n):
    """Compute all overlap values g(k) for k = -(n-1) to n-1."""
    v = 1.0 - h
    g = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        i0 = max(0, -k)
        i1 = min(n, n - k)
        g[k + n - 1] = np.dot(h[i0:i1], v[i0+k:i1+k])
    return g

def smooth_max_grad(h, n, alpha, top_k=0):

    v = 1.0 - h

    max_g_sub = g_sub.max()

    # Clip to avoid overflow
    exp_arg = np.clip(exp_arg, -500, 500)
    w = np.exp(exp_arg)
    w /= w.sum()

        grad[idx] += wm * v[idx + k]
        grad[idx + k] -= wm * h[idx]
