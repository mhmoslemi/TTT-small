def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:

            if s < 1e-14:

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]

    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def local_search(h, n, max_sweeps=100):
    """Intensive local search using single swaps."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()

                    improved = True

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk2 = (imn >= 0) & (imn < n)
    delta[msk2] -= h[imn[msk2]]
    jm = j + m_range
    msk3 = (jm >= 0) & (jm < n)
    delta[msk3] -= 1.0 - h[jm[msk3]]
    jmn = j - m_range
    msk4 = (jmn >= 0) & (jmn < n)
    delta[msk4] += h[jmn[msk4]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def local_search(h, n, max_sweeps=80):

def slsqp_smooth(h, n, alpha, maxiter):

        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n

    if time.time() < remaining and best_h is not None and best_n > 0:
        h_cont = project_feasible(best_h + 0.02 * rng.randn(best_n), best_n)
        for alpha in [10000, 50000, 200000]:
            if time.time() > remaining - 5:
                break
            h_cont = slsqp_smooth(h_cont, best_n, alpha, 800)
            c = compute_c5(h_cont, best_n)
            if c < best_c:
                best_c = c
                best_h = h_cont.copy()
                best_n = best_n

    if time.time() < deadline - 5 and best_h is not None:
        h_bin_final = round_to_binary(best_h, best_n)
        h_ils2, c_ils2 = ils_search(h_bin_final, best_n, min(deadline - 2, time.time() + 30),
                                     seed=seed + 9999, psize_min=1, psize_max=4)
        if c_ils2 < best_c:
            best_c = c_ils2
            best_h = h_ils2.copy()
            best_n = best_n

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
