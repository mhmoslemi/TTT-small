    m=float(corr.max())

from scipy.optimize import minimize

_target = 0.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n=None):
    h = np.asarray(h, float)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    if n is None:
        n = len(h)
    h = h[:n].copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(12):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            rs = float(room.sum())
            if rs > 1e-15:
                h += (diff / rs) * room
        else:
            if s > 1e-15:
                h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:

        if free.size:
            h[free[0]] = float(np.clip(h[free[0]] + diff, 0.0, 1.0))
    return h

def top_shifts(h, k=3):

        return [(h.copy(), compute_c5(h))]

    new = (np.arange(m) * n) / float(m)

def _soft_core(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    sh = beta * (corr - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        e[int(np.argmax(corr))] = 1.0
        s = 1.0
    w = e / s
    val = m + float(np.log(s)) / beta
    return corr, m, w, val

def gradient_from_weights(h, w):
    n = len(h)
    B = np.zeros(2 * n - 1)
    w0 = float(w[n - 1])
    B[n - 1] = -2.0 * w0
    if n > 1:
        if _LAG_SIGN > 0:
            pos = w[n - 2::-1]
            neg = w[n:2 * n - 1]
        else:
            pos = w[n:2 * n - 1]
            neg = w[n - 2::-1]
        val = -(pos + neg)
        B[:n - 1] = val[::-1]
        B[n:] = val
        c = np.full(n, w0, dtype=float)
        cum_pos = np.cumsum(pos)
        neg_pref = np.empty(n)
        neg_pref[0] = 0.0
        neg_pref[1:] = np.cumsum(neg)
        k = n - 1 - np.arange(n)
        suff = np.zeros(n)
        mpos = k > 0
        suff[mpos] = cum_pos[k[mpos] - 1]
        c += suff + neg_pref
    else:
        c = np.full(1, w0, dtype=float)
    q = np.convolve(h, B, mode='full')[n - 1:n - 1 + n]
    return c + q
