   def compute_c5(h, n):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full')
       return float(np.max(c) * dx)
   
   def project_feasible(h, n):
       target = n / 2.0
       h = np.clip(h, 0.0, 1.0)
       for _ in range(100):
           diff = target - h.sum()
           if abs(diff) < 1e-14:
               break
           if diff > 0:
               room = 1.0 - h
               s = room.sum()
               if s < 1e-14:
                   break
               h += diff / s * room
           else:
               s = h.sum()
               if s < 1e-14:
                   break
               h += diff / s * h
       return np.clip(h, 0.0, 1.0)
   
   def to_binary(h, n):
       idx = np.argsort(h)[::-1]
       h_b = np.zeros(n)
       h_b[idx[:n // 2]] = 1.0
       return h_b
   
   def smooth_obj_and_grad(h, alpha, lambda_pen=0.0):
       n = len(h)
       dx = 2.0 / n
       target = n / 2.0
       c_full = np.correlate(h, 1 - h, mode='full')
       c = c_full * dx
       max_c = c.max()
       shifted = alpha * (c - max_c)
       shifted = np.where(shifted > -500, shifted, -500)
       exp_c = np.exp(shifted)
       p = exp_c / exp_c.sum()
       f = max_c + np.log(exp_c.sum()) / alpha
       
       # Gradient
       s = 1.0 - 2.0 * h
       g = np.zeros(n)
       # g[j] = dx * sum_{r=0}^{n-1} s[r] * p[r+n-1-j]
       for j in range(n):
           g[j] = np.dot(s, p[n-1-j : 2*n-1-j])
       g = dx * g
       
       if lambda_pen > 0:
           f += lambda_pen * (h.sum() - target) ** 2
           g += 2 * lambda_pen * (h.sum() - target)
       
       return f, g

           return g

                      bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8, 'maxfun': 5000})
       return res.x
