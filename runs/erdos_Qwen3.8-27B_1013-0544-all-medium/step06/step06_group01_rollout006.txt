        res = minimize(obj_smooth, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       args=(n, alpha),
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

def fast_cd(h, n, max_passes=1000, grid=500, deadline=None, rng=None, penalty=10.0):

    n_corr = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(n_corr - 1, max(p + n - 1, n_corr - 1 - p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < n_corr - 1:

            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= n_corr - 1 - p)
            if np.any(fwd):

            if np.any(bwd):

            deltas = np.linspace(d_lo, d_hi, grid)

                    continue

                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:

                if val < best_val - 1e-14:

                    best_d = d

                    val = max(um, float(np.max(new_c)))
                    if penalty > 0:
                        val += penalty * (sumh + mid - n / 2.0) ** 2
                    if val < best_val:

                        best_d = mid
                        hi = mid
                    else:
                        lo = mid

            if abs(d) > 1e-12:
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]

                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:

def pair_search(h, n, deadline, deltas, radius=50):

                    return h, best_c

                        continue

                        h[j] -= d

                            improved = True

            if tot < 1e-15:

                            best_c = cv; improved = True
                        else:
                            h[i] -= dd; h[j] += dd
                    if h[i] - dd >= 0.0 and h[j] + dd <= 1.0:
                        h[i] -= dd; h[j] += dd
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv; improved = True
                        else:
                            h[i] += dd; h[j] -= dd
            if not improved:
                break
    return h, best_c

def stochastic_perturb(h, n, rng, n_perturb=10, scale=0.05):
    h2 = h.copy()
    idx = rng.choice(n, size=min(n_perturb, n), replace=False)
    h2[idx] += rng.normal(0, scale, size=len(idx))
    return project_feasible(h2, n)

def opt_run(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # Phase 1: CD with grid=300
    t_sub = min(250, (deadline - time.time()) * 0.45)
    if t_sub > 20:
        h2, c2 = fast_cd(h, n, max_passes=600, grid=300,
                         deadline=time.time() + t_sub, rng=rng, penalty=5.0)
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h, best_c = h2, c2
    # Phase 2: SLSQP with increasing alpha
    if time.time() < deadline - 80:
        for a in [10000, 50000, 200000, 1000000]:
            if time.time() > deadline - 40:
                break
            h2 = slsqp(h, n, alpha=a, maxiter=2000)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                h, best_c = h2, c2
    # Phase 3: Fine CD with larger grid
    if time.time() < deadline - 40:
        t_sub2 = min(120, deadline - time.time() - 30)
        if t_sub2 > 20:
            h2, c2 = fast_cd(h, n, max_passes=400, grid=600,
                             deadline=time.time() + t_sub2, rng=rng, penalty=0.5)
            h2 = project_feasible(h2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                h, best_c = h2, c2
    # Phase 4: Pair search
    if time.time() < deadline - 15:
        h, best_c = pair_search(h, n, deadline - 5,
                                [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
    # Phase 5: Final SLSQP
    if time.time() < deadline - 10:
        h2 = slsqp(h, n, alpha=5000000, maxiter=3000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h, best_c = h2, c2
    if time.time() < deadline - 5:
        h, best_c = pair_search(h, n, deadline - 2,
                                [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=50)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Optimize initial
    alloc1 = min(420, (deadline - t0) * 0.42)
    h_opt, c_opt = opt_run(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try n=400
    if time.time() < deadline - 250:
        x_init = np.linspace(0, 2, n_init)
        n_new = 400
        x_new = np.linspace(0, 2, n_new)
        h_interp = project_feasible(np.interp(x_new, x_init, h_init), n_new)
        alloc = min(220, deadline - time.time() - 120)
        if alloc > 60:
            h_opt, c_opt = opt_run(h_interp, n_new, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Stochastic restarts
    if time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restart = max(2, int(remaining / 80))
        for r in range(n_restart):
            if time.time() > deadline - 60:
                break
            scale = 0.015 + 0.035 * (r / max(1, n_restart - 1))
            hp = stochastic_perturb(best_h, n, rng, n_perturb=8, scale=scale)
            alloc = min(75, (deadline - time.time() - 35) * 0.5)
            if alloc < 30:
                break
            h_opt, c_opt = opt_run(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final polish
    if time.time() < deadline - 30:
        n = len(best_h)
        h_f = slsqp(best_h, n, alpha=10000000, maxiter=5000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, deadline - 2,
                                     [0.0005, 0.0002, 0.0001, 0.00005, 0.00002], radius=60)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
