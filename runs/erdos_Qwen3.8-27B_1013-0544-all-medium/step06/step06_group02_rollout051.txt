    grad = np.zeros(n)
    sig = np.where(w > 1e-12)[0]
    for m_idx in sig:

        wm = w[m_idx] * dx

            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

        improved_any = False
        while time.time() < deadline:
            improved = False

                if time.time() > deadline:
                    break

                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True; improved_any = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:

                # Try smaller radius or break
                if radius > 10:
                    radius = max(10, radius // 2)
                else:
                    break
        if time.time() > deadline:
            break

def binary_local_search(h, n, deadline):
    """Snap to binary and do local search"""
    # Create binary version: top n/2 values become 1, rest become 0
    h_bin = np.zeros(n)
    top_idx = np.argsort(h)[::-1][:n//2]
    h_bin[top_idx] = 1.0
    c = compute_c5(h_bin, n)
    best_c = c
    best_h = h_bin.copy()
    
    # Try flipping pairs
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h_bin[i] == 1.0:
                for j in range(i+1, n):
                    if h_bin[j] == 0.0:
                        # Swap: set i to 0, j to 1
                        h_bin[i] = 0.0; h_bin[j] = 1.0
                        c_new = compute_c5(h_bin, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new; best_h = h_bin.copy(); improved = True
                        else:
                            h_bin[i] = 1.0; h_bin[j] = 0.0
            if time.time() > deadline:
                break
        if not improved:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha, topk):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    nc = len(c)
    if topk is not None and topk < nc:
        idx = np.argpartition(c, -topk)[-topk:]
        c_top = c[idx]
        m = c_top.max()
        shifted = alpha * (c_top - m)
        mx = shifted.max()
        w_top = np.exp(shifted - mx)
        w_top /= w_top.sum()
        w = np.zeros(nc)
        w[idx] = w_top
    else:
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
    grad = np.zeros(n)
    sig = np.where(w > 1e-12)[0]
    for mi in sig:
        k = mi - (n - 1)
        wm = w[mi] * dx

            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])

        mx = shifted.max()
        val = m + np.log(np.sum(np.exp(shifted - mx))) / alpha
    else:
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        val = m + np.log(np.sum(np.exp(shifted - mx))) / alpha
    return float(val)

    best_h = h.copy()

                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True; improved_any = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                if improved_any and cur_radius > 15:
                    cur_radius = max(15, cur_radius // 2)
                else:
                    break
        if time.time() > deadline:
            break
    best_h = h.copy()
    return best_h, best_c

def coord_search(h, n, deadline, alpha=200000):
    """Gradient-guided coordinate descent with exact constraint."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.02
    for _ in range(50000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha, 1)
        gi = np.argmin(g)
        gj = np.argmax(g)
        if gi == gj:
            break
        step = min(lr, h[gi], 1.0 - h[gj])
        if step < 1e-10:
            step = min(lr, 1.0 - h[gi], h[gj])
            if step < 1e-10:
                lr *= 0.5
                if lr < 1e-10:
                    break
                continue
            h[gi] += step; h[gj] -= step
        else:
            h[gi] -= step; h[gj] += step
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            lr = min(lr * 1.05, 0.05)
        else:
            h = best_h.copy()
            lr *= 0.6
            if lr < 1e-10:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS with increasing alpha and decreasing topk
    t1 = min(t0 + 200, deadline - 500)
    configs1 = [(200, 30), (1000, 15), (5000, 8), (20000, 4), (100000, 2), (500000, 1)]
    best_h, best_c = lbfgs_optimize(best_h, n, t1, configs1)

    # Phase 2: SLSQP with constraint
    if time.time() < deadline - 350:
        t2 = min(time.time() + 80, deadline - 350)
        for a, tk in [(200000, 1), (1000000, 1)]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_optimize(best_h, n, t2, a, tk)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Projected gradient descent
    if time.time() < deadline - 300:
        t3 = min(time.time() + 80, deadline - 300)
        h3, c3 = proj_grad_descent(best_h, n, t3, alpha_init=50000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Coordinate search (gradient-guided)
    if time.time() < deadline - 250:
        t4 = min(time.time() + 80, deadline - 250)
        h4, c4 = coord_search(best_h, n, t4, alpha=500000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Pair search - main refinement
    if time.time() < deadline - 20:
        t5 = deadline - 5
        best_h, best_c = pair_search(best_h, n, t5,
                                      [0.02, 0.01, 0.005, 0.002, 0.001,
                                       0.0005, 0.0002, 0.0001, 0.00005],
                                      radius=100)

    # Phase 6: Try alternate resolutions
    if time.time() < deadline - 80:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 50:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            remain = deadline - time.time() - 40
            if remain < 30:
                break
            alloc1 = min(remain * 0.5, 60)
            alloc2 = min(remain * 0.3, 40)
            if time.time() < deadline - 40:
                h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc1,
                                           [(1000, 10), (10000, 3), (100000, 1)])
            else:
                h_t, c_t = h_interp, compute_c5(h_interp, n_try)
            if time.time() < deadline - 20:
                h_t, c_t = pair_search(h_t, n_try, time.time() + alloc2,
                                        [0.01, 0.005, 0.002, 0.001, 0.0005],
                                        radius=60)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try

    # Final: quick pair search polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, n, deadline,
                                      [0.001, 0.0005, 0.0002, 0.0001],
                                      radius=50)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
