    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    s = shifted.max()
    obj = m + (s + np.log(np.exp(shifted - s).sum())) / alpha
    weights = np.exp(shifted - s)

            grad[s2:] += w * (1.0 - x[:n - s2])
            grad[:n - s2] += w * (-x[s2:])
    grad *= dx
    return obj, grad

                       options={'maxiter': maxiter, 'ftol': 1e-15})

        pass
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    s = shifted.max()
    obj = m + (s + np.log(np.exp(shifted - s).sum())) / alpha
    weights = np.exp(shifted - s)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-9:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - x[shift:])
            grad[shift:] += w * (-x[:n - shift])
        else:
            s2 = -shift
            grad[s2:] += w * (1.0 - x[:n - s2])
            grad[:n - s2] += w * (-x[s2:])
    grad *= dx
    return obj, grad

def _slsqp(h, n, alpha, maxiter=500):
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: ones.copy()}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj_grad, h.copy(), args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def _subgrad(h, n, n_iter, alpha, eta0, deadline, mu=0.95):
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        s = shifted.max()
        w = np.exp(shifted - s)
        w = w / w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-9:
                continue
            shift = k - (n - 1)
            if shift >= 0:
                grad[:n - shift] += wk * (1.0 - h[shift:])
                grad[shift:] += wk * (-h[:n - shift])
            else:
                s2 = -shift
                grad[s2:] += wk * (1.0 - h[:n - s2])
                grad[:n - s2] += wk * (-h[s2:])
        grad *= dx
        eta = eta0 / (1.0 + 0.04 * np.sqrt(it))
        v = mu * v + (1.0 - mu) * grad
        h_new = project_feasible(h - eta * v, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _hill_climb(h, n, n_iter, deadline, rng, peak_alpha=1.0):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = corr.max()
        if peak_alpha >= 1e6:
            k_star = int(np.argmax(corr))
            kg = np.zeros(2 * n - 1)
            kg[k_star] = 1.0
        else:
            shifted = peak_alpha * (corr - m)
            kg = np.exp(shifted - shifted.max())
            kg = kg / kg.sum()
        shift = int(np.argmax(kg)) - (n - 1)
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n - shift] = 1.0 - h[shift:]
            grad[shift:] += -h[:n - shift]
        else:
            s2 = -shift
            grad[s2:] = 1.0 - h[:n - s2]
            grad[:n - s2] += -h[s2:]
        eta = 0.5
        improved = False
        for _ in range(20):
            h_try = project_feasible(h - eta * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h_try.copy()
                improved = True
                break
            eta *= 0.5
        if not improved:
            stall += 1
            if stall > 80:
                idx = rng.choice(n, size=min(30, n), replace=False)
                h = h.copy()
                h[idx] += 0.015 * rng.randn(len(idx))
                h = project_feasible(h, n)
                stall = 0
    return best_h, best_c

def _try_n_interpolated(h_src, n_src, n_dst, deadline):
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)
    h_dst = project_feasible(h_dst, n_dst)
    return h_dst

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    p1 = t0 + 0.38 * (budget_s - 5)
    for alpha in [200, 1000, 5000, 20000, 100000, 500000, 2000000]:
        if time.time() > p1 - 15:
            break
        hr = _slsqp(best_h, n, alpha, maxiter=500)
        c = compute_c5(hr, n)
        if c < best_c:
            best_c, best_h = c, hr.copy()

    # Phase 2: Subgradient with momentum
    p2 = t0 + 0.60 * (budget_s - 5)
    for alpha in [50, 200, 1000, 5000]:
        if time.time() > p2 - 15:
            break
        hs, cs = _subgrad(best_h, n, 30000, alpha, 0.04, p2, mu=0.96)
        if cs < best_c:
            best_c, best_h = cs, hs.copy()

    # Phase 3: Hill climbing (targeted at peak shift)
    p3 = t0 + 0.78 * (budget_s - 5)
    hh, ch = _hill_climb(best_h, n, 20000, p3, rng, peak_alpha=1e6)
    if ch < best_c:
        best_c, best_h = ch, hh.copy()

    # Phase 4: SLSQP again after hill climb
    p4 = t0 + 0.88 * (budget_s - 5)
    for alpha in [100000, 1000000, 5000000]:
        if time.time() > p4 - 10:
            break
        hr = _slsqp(best_h, n, alpha, maxiter=400)
        c = compute_c5(hr, n)
        if c < best_c:
            best_c, best_h = c, hr.copy()

    # Phase 5: Restart with perturbation
    p5 = t0 + 0.95 * (budget_s - 5)
    for r in range(8):
        if time.time() > p5 - 10:
            break
        scale = 0.02 + 0.01 * r
        hp = best_h + scale * rng.randn(n)
        hp = project_feasible(hp, n)
        hr = _slsqp(hp, n, 500000, maxiter=300)
        c = compute_c5(hr, n)
        if c < best_c:
            best_c, best_h = c, hr.copy()
        hh2, ch2 = _hill_climb(hr, n, 2000, min(time.time() + 5, p5 - 2), rng, peak_alpha=1e6)
        if ch2 < best_c:
            best_c, best_h = ch2, hh2.copy()

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        hr = _slsqp(best_h, n, 10000000, maxiter=600)
        c = compute_c5(hr, n)
        if c < best_c:
            best_c, best_h = c, hr.copy()
    if time.time() < deadline - 3:
        hh, ch = _hill_climb(best_h, n, 5000, deadline - 2, rng, peak_alpha=1e6)
        if ch < best_c:
            best_c, best_h = ch, hh.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
