def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    s = w.sum()
    w /= s
    
    grad = np.zeros(n)
    # For each lag l (indexed 0 to 2n-2 in the correlation array, corresponding to actual lag l-(n-1)):
    for idx in range(2 * n - 1):
        l = idx - (n - 1)  # actual lag
        if w[idx] < 1e-15:
            continue
        # Term 1: w[idx] * (1-h[j+l]) for j where 0 <= j+l < n
        # j ranges from max(0, -l) to min(n-1, n-1-l)
        j_start = max(0, -l)
        j_end = min(n, n - l)  # exclusive
        if j_end > j_start:
            grad[j_start:j_end] += w[idx] * (1 - h[j_start + l:j_end + l])
        # Term 2: -w[idx] * h[j-l] for j where 0 <= j-l < n
        # j ranges from max(0, l) to min(n, n + l)
        j_start2 = max(0, l)
        j_end2 = min(n, n + l)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= w[idx] * h[j_start2 - l:j_end2 - l]
    
    return grad * dx
