def project_fixed_sum(x,target):
    x = np.asarray(x,float).copy()
    target = float(target)
    if x.size ==0: return x
    if target <=0: return zeros; if >=n ones
    # bisection
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
       mid=(lo+hi)*0.5
       s=np.sum(np.clip(x-mid,0.0,1.0))
       if s > target: lo=mid
       else: hi=mid
    y=np.clip(x-((lo+hi)*0.5),0.0,1.0)
    # correct residual
    diff=target-y.sum()
    if abs(diff)>1e-12:
       mask=(y>0)&(y<1)
       if mask.sum()>0:
           y[mask]+=diff/mask.sum()
       else:
           y += diff/y.size
    return np.clip(y,0,1)

```python
import numpy as np
import time


def compute_C(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(compute_C(h)) * (2.0 / n))


def project_fixed_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    target = float(target)
    n = x.size
    if n == 0:
        return x
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    x = np.clip(x, -10.0, float(n) + 10.0)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    y = np.clip(x - theta, 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-12:
        mask = (y > 0.0) & (y < 1.0)
        if mask.sum() > 0:
            y[mask] += diff / float(mask.sum())
        else:
            y += diff / float(n)
    return np.clip(y, 0.0, 1.0)


def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def make_random_binary(n, rng):
    target = n // 2
    idx = rng.choice(n, target, replace=False)
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def perturb(h, n, rng, k):
    h = binarize(h, n)
    ones = np.nonzero(h >= 0.5)[0]
    zeros = np.nonzero(h < 0.5)[0]
    if ones.size == 0 or zeros.size == 0:
        return h
    for _ in range(int(k)):
        a = int(rng.choice(ones))
        b = int(rng.choice(zeros))
        h[a], h[b] = 0.0, 1.0
        ones = np.nonzero(h >= 0.5)[0]
        zeros = np.nonzero(h < 0.5)[0]
    return h


def compute_F(h, n):
    L = n - 1
    M = 2 * n - 1
    F = np.zeros((n, M), dtype=float)
    om_rev = (1.0 - h)[::-1]
    neg_h = -h
    for l in range(n):
        F[l, l:l + n] += om_rev
        F[l, L - l:L - l + n] += neg_h
    return F


def best_swap(C, F, h, n, L):
    zeros = np.nonzero(h < 0.5)[0]
    ones = np.nonzero(h >= 0.5)[0]
    if zeros.size == 0 or ones.size == 0:
        return -1, -1, 0.0

    M = 2 * n - 1
    FO = F[ones].copy()
    temp = np.empty((ones.size, M), dtype=float)
    cur = float(C.max())
    best_val = cur + 1.0
    best_i = -1
    best_j = -1

    for i in zeros:
        base = C.copy()
        base += F[i]
        base[L] -= 2.0
        np.subtract(FO, base[None, :], out=temp)
        vals = -temp.min(axis=1)

        d = np.abs(ones - int(i))
        p1 = L + d
        p2 = L - d

        v1 = 1.0 - temp[:, p1]
        v2 = 1.0 - temp[:, p2]

        np.maximum(vals, v1, out=vals)
        np.maximum(vals, v2, out=vals)

        pos = int(np.argmin(vals))
        val = float(vals[pos])
        if val < best_val - 1e-12:
            best_val = val
            best_i = int(i)
            best_j = int(ones[pos])

    return best_i, best_j, best_val


def binary_local_search(h, n, deadline, max_moves=8000, seed=42):
    h = binarize(h, n).copy()
    L = n - 1
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    moves = 0

    while time.time() < deadline and moves < max_moves:
        C = compute_C(h)
        F = compute_F(h, n)
        cur = float(C.max())
        bi, bj, val = best_swap(C, F, h, n, L)
        if bi < 0 or bj < 0 or val >= cur - 1e-9:
            break

        h[bi] = 1.0
        h[bj] = 0.0
        moves += 1
        cand = float(val) * dx
        if cand < best_c - 1e-10 or moves % 8 == 0:
            cnow = compute_c5(h, n)
            if cnow < best_c - 1e-10:
                best_c = cnow
                best_h = h.copy()

    cnow = compute_c5(h, n)
    if cnow < best_c - 1e-10:
        best_c = cnow
        best_h = h.copy()
    return best_h, best_c, moves


def interpolate_start(h0, n0, n):
    if n == n0:
        return h0.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    h1 = np.interp(x1, x0, h0, left=float(h0[0]), right=float(h0[-1]))
    return project_fixed_sum(np.clip(h1, 0.0, 1.0), 0.5 * n, n)


def eval_pair_values(C, v, L, p1, p2, ts, temp):
    np.multiply(ts[:, None], v[None, :], out=temp)
    temp += C

    temp[:, L] = -np.inf
    temp[:, p1] = -np.inf
    if p2 != p1:
        temp[:, p2] = -np.inf

    val = temp.max(axis=1)
    val = np.maximum(val, C[L] + ts * v[L] - 2.0 * ts * ts)
    val = np.maximum(val, C[p1] + ts * v[p1] + ts * ts)
    if p2 != p1:
        val = np.maximum(val, C[p2] + ts * v[p2] + ts * ts)
    return val


def try_gradient_step(h, C, F, g0, curmax, n):
    scale = float(np.max(np.abs(g0)))
    if scale < 1e-12:
        return None
    direction = -g0
    for step in (0.25, 0.10, 0.04, 0.015, 0.005):
        ht = h + (step / scale) * direction
        ht = project_fixed_sum(np.clip(ht, 0.0, 1.0), 0.5 * n, n)
        cnow = float(np.max(compute_C(ht)))
        if cnow < curmax - 1e-9:
            return ht
    return None


def continuous_pair(h, n, deadline, seed=42, max_iter=6000, kappa=20.0):
    L = n - 1
    M = 2 * n - 1
    h = project_fixed_sum(np.clip(h, 0.0, 1.0), 0.5 * n, n)

    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    no_improve = 0
    it = 0

    grid = np.linspace(0.0, 1.0, 40)[1:]
    temp40 = np.empty((grid.size, M), dtype=float)
    temp30 = np.empty((30, M), dtype=float)

    while time.time() < deadline and it < max_iter and no_improve < 4:
        C = compute_C(h)
        F = compute_F(h, n)
        curmax = float(C.max())
        cnow = (2.0 / n) * curmax
        if cnow < best_c - 1e-12:
            best_c = cnow
            best_h = h.copy()

        ex = np.exp(np.clip(kappa * (C - curmax), -40.0, 0.0))
        s = float(ex.sum())
        if s < 1e-15:
            active = C >= curmax - 1e-8
            W = active.astype(float)
            W /= max(1.0, float(W.sum()))
        else:
            W = ex / s

        g = F.T @ W
        g0 = g - float(np.mean(g))
        order = np.argsort(g0)

        K = 8 + min(6, it // 25)
        low = order[:K]
        high = order[-K:][::-1]

        if n > 2 * K:
            rnd = np.unique(rng.choice(n, size=min(n, 2 * K), replace=False))
            a_rand = rnd[np.argsort(g0[rnd])[:K]]
            b_rand = rnd[np.argsort(g0[rnd])[-K:][::-1]]
        else:
            rnd = np.arange(n)
            a_rand = rnd[:K]
            b_rand = rnd[-K:][::-1]

        a_list = np.unique(np.concatenate((low, a_rand)))
        b_list = np.unique(np.concatenate((high, b_rand)))

        best_new = curmax + 1.0
        best_t = 0.0
        ba = -1
        bb = -1
        bd = 0

        for a in a_list:
            ai = int(a)
            if h[ai] >= 1.0 - 1e-12:
                continue
            for b in b_list:
                bi = int(b)
                if ai == bi:
                    continue
                if h[bi] <= 1e-12:
                    continue
                tmax = min(1.0 - h[ai], h[bi])
                if tmax <= 1e-9:
                    continue

                d = abs(ai - bi)
                p1 = L + d
                p2 = L - d
                v = F[ai] - F[bi]
                ts = grid * tmax

                val = eval_pair_values(C, v, L, p1, p2, ts, temp40)
                pos = int(np.argmin(val))
                nm = float(val[pos])
                t = float(ts[pos])

                if nm < curmax - 1e-9:
                    width = max(tmax / 39.0, 1e-6)
                    lo = max(0.0, t - width)
                    hi = min(tmax, t + width)
                    if hi > lo:
                        ref = np.linspace(lo, hi, 30)
                        vr = eval_pair_values(C, v, L, p1, p2, ref, temp30)
                        rr = int(np.argmin(vr))
                        if float(vr[rr]) < nm:
                            nm = float(vr[rr])
                            t = float(ref[rr])

                if nm < best_new - 1e-10:
                    best_new = nm
                    best_t = t
                    ba = ai
                    bb = bi
                    bd = d

        if ba >= 0 and best_new < curmax - 1e-8:
            h[ba] += best_t
            h[bb] -= best_t
            h = project_fixed_sum(np.clip(h, 0.0, 1.0), 0.5 * n, n)
            it += 1
            no_improve = 0
            continue

        hnew = try_gradient_step(h, C, F, g0, curmax, n)
        if hnew is not None:
            h = hnew
            it += 1
            no_improve = 0
            continue

        moved = False
        for _ in range(10):
            a = int(rng.choice(n))
            b = int(rng.choice(n))
            if a == b:
                continue
            tmax = min(1.0 - h[a], h[b])
            if tmax > 1e-6:
                t = rng.uniform(0.004, 0.12) * tmax
                h[a] += t
                h[b] -= t
                h = project_fixed_sum(np.clip(h, 0.0, 1.0), 0.5 * n, n)
                moved = True
                break

        no_improve += 1
        if no_improve >= 4:
            break

    cnow = compute_c5(h, n)
    if cnow < best_c - 1e-12:
        best_c = cnow
        best_h = h.copy()
    return best_h, best_c


def process_candidate(h0, n, deadline, max_time, seed):
    stop = min(float(deadline), time.time() + float(max_time))
    h0 = project_fixed_sum(np.clip(h0, 0.0, 1.0), 0.5 * n, n)

    best_any_h = h0.copy()
    best_any_c = compute_c5(h0, n)

    best_bin_h = binarize(h0, n)
    best_bin_c = compute_c5(best_bin_h, n)
    current = best_bin_h.copy()

    l_rng = np.random.RandomState(seed)
    stag = 0
    cycle = 0

    while time.time() < stop:
        cycle += 1
        rem = stop - time.time()
        if rem < 12.0:
            break

        if cycle == 1:
            bin_dead = time.time() + min(0.40 * rem, 150.0)
            cont_dead = time.time() + min(0.60 * rem, 200.0)
        else:
            bin_dead = time.time() + min(0.35 * rem, 65.0)
            cont_dead = time.time() + min(0.60 * rem, 120.0)

        bin_dead = min(bin_dead, stop - 1.0)
        cont_dead = min(cont_dead, stop - 2.0)
        improved = False

        h_bin, c_bin, _ = binary_local_search(current, bin_dead, max_moves=10000, seed=seed + 13 * cycle)
        if c_bin < best_bin_c - 1e-11:
            best_bin_c = c_bin
            best_bin_h = h_bin.copy()
        if c_bin < best_any_c - 1e-11:
            best_any_c = c_bin
            best_any_h = h_bin.copy()
            improved = True

        if time.time() < cont_dead:
            if best_any_c <= best_bin_c + 1e-9:
                cont_start = best_any_h
            else:
                cont_start = best_bin_h
            h_cont, c_cont = continuous_pair(cont_start, n, cont_dead, seed=seed + 37 * cycle)
            if c_cont < best_any_c - 1e-11:
                best_any_c = c_cont
                best_any_h = h_cont.copy()
                improved = True

        if improved:
            stag = 0
        else:
            stag += 1

        if stag >= 5:
            break

        r = 3 + 2 * cycle
        if r > 38:
            r = 38
        current = perturb(best_bin_h, n, l_rng, r)

    best_any_h = project_fixed_sum(np.clip(best_any_h, 0.0, 1.0), 0.5 * n, n)
    best_any_c = compute_c5(best_any_h, n)
    return best_any_h, best_any_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)

    try:
        hpar = np.asarray(initial_h_values, dtype=float).copy()
        npar = len(hpar)
        if npar < 10:
            raise ValueError("too short")
        npar = 2 * (npar // 2)
        if npar < 10:
            npar = 250
        if len(hpar) != npar:
            hpar = interpolate_start(hpar, len(hpar), npar)
        npar = int(npar)
    except Exception:
        hpar = np.full(250, 0.5, dtype=float)
        npar = 250

    hpar = project_fixed_sum(np.clip(hpar, 0.0, 1.0), 0.5 * npar, npar)
    cpar = compute_c5(hpar, npar)

    best_h = hpar.copy()
    best_c = cpar
    best_n = npar

    def register(h, c, n):
        nonlocal_best_c = None
        return h, c, n

    parent_time = min(0.45 * float(budget_s), max(60.0, deadline - time.time() - 30.0))
    if parent_time > 10.0:
        hp, pc = process_candidate(hpar, npar, deadline, parent_time, seed)
        pc = compute_c5(hp, npar)
        if pc < best_c:
            best_c, best_h, best_n = pc, hp, npar

    base_n = best_n if 120 <= best_n <= 512 else 250
    seen = set()

    for i in range(7):
        if time.time() > deadline - 30.0:
            break
        hrand = make_random_binary(base_n, rng)
        htime = min(70.0, max(10.0, deadline - time.time() - 20.0))
        if htime < 12.0:
            break
        hr, hc = process_candidate(hrand, base_n, deadline, htime, seed + 1000 + 7 * i)
        hc = compute_c5(hr, base_n)
        if hc < best_c:
            best_c, best_h, best_n = hc, hr, base_n

    for nlarge in (256, 320, 384, 512):
        if nlarge in seen:
            continue
        if time.time() > deadline - 60.0:
            break
        htime = min(65.0, max(15.0, deadline - time.time() - 40.0))
        if htime < 15.0:
            break
        if best_h is not None:
            hstart = interpolate_start(best_h, best_n, nlarge)
        else:
            hstart = make_random_binary(nlarge, rng)
        hl, cl = process_candidate(hstart, nlarge, deadline, htime, seed + 2000 + nlarge)
        cl = compute_c5(hl, nlarge)
        if cl < best_c:
            best_c, best_h, best_n = cl, hl, nlarge
        seen.add(nlarge)

    if time.time() < deadline - 15.0 and best_h is not None:
        ftime = max(20.0, deadline - time.time() - 8.0)
        hg, fg = process_candidate(best_h, best_n, deadline, ftime, seed + 9000)
        fg = compute_c5(hg, best_n)
        if fg < best_c:
            best_c, best_h, best_n = fg, hg, best_n

    best_h = project_fixed_sum(np.clip(best_h, 0.0, 1.0), 0.5 * best_n, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
```
