def subgradient_optimize(h, n, deadline, seed, max_iter=5000):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    momentum = np.zeros(n)
    lr = 0.01
    best_momentum = 0.9
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find top lags
        top_m = min(10, len(corr))
        top_idx = np.argsort(corr)[-top_m:]
        # Softmax weights
        temps = np.array([5.0, 10.0, 20.0])
        # Use the temperature that gives most balanced weights among top lags
        for T in temps:
            w = np.exp(T * (corr[top_idx] - corr[top_idx].max()))
            w /= w.sum()
            if w.min() > 0.01:  # balanced enough
                break
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            grad += w[j] * gradient_for_lag(h, lag, n)
        
        momentum = best_momentum * momentum - lr * grad
        h_new = project_feasible(h + momentum, ts)
        c_new = compute_c5(h_new)
        
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
            lr = min(lr * 1.01, 0.05)
        else:
            momentum *= 0.5
            lr *= 0.9
            if lr < 1e-6:
                break
    return h, c5

def coordinate_descent(h, n, deadline, max_rounds=100):
    ts = n / 2.0
    c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        grad = gradient_for_lag(h, lag, n)
        
        # Try to decrease each coordinate in the direction that reduces the correlation
        # For coordinate j: if grad[j] > 0, decreasing h[j] helps; if grad[j] < 0, increasing h[j] helps
        best_improvement = 0
        best_j = -1
        best_delta = 0
        
        # Only try a subset for efficiency
        trial_indices = np.where(np.abs(grad) > 0.01)[0]
        if len(trial_indices) == 0:
            break
        # Sample at most 50
        if len(trial_indices) > 50:
            trial_indices = np.random.choice(trial_indices, 50, replace=False)
        
        for j in trial_indices:
            # Direction: if grad[j] > 0, decrease h[j]; if grad[j] < 0, increase h[j]
            direction = -np.sign(grad[j])
            # Try a small step
            for delta in [0.01, 0.005, 0.001]:
                h_try = h.copy()
                h_try[j] += direction * delta
                h_try = project_feasible(h_try, ts)
                c_try = compute_c5(h_try)
                if c_try < c5 - best_improvement:
                    best_improvement = c5 - c_try
                    best_j = j
                    best_delta = direction * delta
        
        if best_j >= 0:
            h[best_j] += best_delta
            h = project_feasible(h, ts)
            c5 = compute_c5(h)
        else:
            break
    return h, c5
