def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        step = diff / free.sum()
        # Limit step to avoid overshooting
        max_up = (1.0 - h[free]).min()
        max_dn = h[free].min()
        if step > 0:
            step = min(step, max_up)
        else:
            step = min(step, -max_dn)  # both negative
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
    return h

            if c5_new < c5_cur - 1e-15:

        except Exception:

    return h_cur, c5_cur

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_active(h, tol=1e-8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    active = np.where(corr > np.max(corr) - tol / dx)[0]
    return c5, active

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        step = diff / free.sum()
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(x)
    sum_exp = np.sum(exp_x)
    weights = exp_x / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    
    grad = np.zeros(n)
    thresh = 1e-12
    for k in np.where(weights > thresh)[0]:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += w * (1.0 - h[m2:])
                grad[m2:] -= w * h[:n - m2]
    return val, grad

def _s_obj(x):
    h = project_h(x, _state['target'])
    v, _ = smooth_max_obj_and_grad(h, _state['beta'])
    return v

def _s_grad(x):
    h = project_h(x, _state['target'])
    _, g = smooth_max_obj_and_grad(h, _state['beta'])
    return g

def _s_con(x):
    return x.sum() - _state['target']

def _s_con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter, time_fn):
    n = len(h0)
    target = n / 2.0
    _state['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_con, 'jac': _s_con_jac}]
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time_fn() < 5:
            break
        _state['beta'] = beta
        try:
            res = minimize(_s_obj, h_cur, method='SLSQP', bounds=bounds,
                         jac=_s_grad, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def subgrad_exact(h, target, n_iter, time_fn, top_k=5):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    lr = 0.005
    mom = 0.85
    c5_prev = compute_c5(h)
    stall = 0
    
    for it in range(n_iter):
        if time_fn() < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            else:
                m2 = -lag
                if m2 < n:
                    grad[:n - m2] += (1.0 - h[m2:])
                    grad[m2:] -= h[:n - m2]
        grad /= top_k
        
        v = mom * v - lr * grad
        h_new = project_h(h + v, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_prev - 1e-14:
            h = h_new
            c5_prev = c5_new
            stall = 0
            lr = min(lr * 1.01, 0.02)  # slowly increase
        else:
            v *= 0.3
            lr *= 0.7
            stall += 1
            if stall > 200:
                break
    
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

_state = {'beta': 100.0, 'target': 0.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(x)
    sum_exp = np.sum(exp_x)
    weights = exp_x / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    thresh = 1e-11
    for k in np.where(weights > thresh)[0]:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += w * (1.0 - h[m2:])
                grad[m2:] -= w * h[:n - m2]
    return val, grad

def _s_obj(x):
    v, _ = smooth_max_obj_and_grad(x, _state['beta'])
    return v

def _s_grad(x):
    _, g = smooth_max_obj_and_grad(x, _state['beta'])
    return g

def _s_con(x):
    return x.sum() - _state['target']

def _s_con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter, time_fn):
    n = len(h0)
    target = n / 2.0
    _state['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_con, 'jac': _s_con_jac}]
    h_cur = h0.copy()
    h_cur = project_h(h_cur, target)
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time_fn() < 5:
            break
        _state['beta'] = beta
        try:
            res = minimize(_s_obj, h_cur, method='SLSQP', bounds=bounds,
                         jac=_s_grad, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def subgrad_momentum(h, target, n_iter, time_fn, top_k=5, lr_init=0.005, mom=0.85):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    c5_prev = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if time_fn() < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            else:
                m2 = -lag
                if m2 < n:
                    grad[:n - m2] += (1.0 - h[m2:])
                    grad[m2:] -= h[:n - m2]
        grad /= top_k
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        v = mom * v - lr * grad
        h_new = project_h(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_prev - 1e-14:
            h = h_new
            c5_prev = c5_new
            stall = 0
            lr = min(lr * 1.02, 0.015)
        else:
            v *= 0.3
            lr *= 0.7
            stall += 1
            if stall > 300:
                break
    return h, compute_c5(h)

def coordinated_refine(h, target, time_fn, n_rounds=50):
    n = len(h)
    h = project_h(h.copy(), target)
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(n_rounds):
        if time_fn() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        active = np.where(corr > np.max(corr) - 1e-10 / dx)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for idx in active:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            else:
                m2 = -lag
                if m2 < n:
                    grad[:n - m2] += (1.0 - h[m2:])
                    grad[m2:] -= h[:n - m2]
        grad /= len(active)
        for step in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_t = project_h(h - step * grad, target)
            c5_t = compute_c5(h_t)
            if c5_t < c5_cur - 1e-15:
                h = h_t
                c5_cur = c5_t
                break
        else:
            if rnd > 20:
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial SLSQP refinement
    if tl() > 50:
        h1, c51 = do_slsqp(best_h, [500, 2000, 10000, 50000], 6000, tl)
        if c51 < best_c5:
            best_h, best_c5 = h1, c51

    # Phase 2: Multi-restart with diverse perturbations
    n = len(best_h)
    target = n / 2.0
    x_grid = np.linspace(0, 2, n)

    perturb_list = []
    for scale in [0.005, 0.01, 0.02, 0.04, 0.06]:
        perturb_list.append(np.random.randn(n) * scale)
    freq = np.random.randint(1, 12)
    perturb_list.append(0.02 * np.sin(2 * np.pi * freq * x_grid))
    perturb_list.append(0.03 * (np.random.random(n) - 0.5))
    # Binarization push
    perturb_list.append(np.sign(best_h - 0.5) * 0.05)
    # Anti-binarization (push toward 0.5)
    perturb_list.append((0.5 - best_h) * 0.1)

    for pert in perturb_list:
        if tl() < 60:
            break
        h_t = project_h(best_h + pert, target)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5 + 0.008:
            h_new, c5_new = do_slsqp(h_t, [2000, 10000, 50000], 4000, tl)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 3: Coordinated refinement (exact active set)
    if tl() > 30:
        n = len(best_h)
        target = n / 2.0
        h3, c53 = coordinated_refine(best_h, target, tl, n_rounds=80)
        if c53 < best_c5:
            best_h, best_c5 = h3, c53

    # Phase 4: Subgradient with momentum
    if tl() > 50:
        n = len(best_h)
        target = n / 2.0
        h4, c54 = subgrad_momentum(best_h, target, 20000, tl, top_k=5, lr_init=0.004, mom=0.88)
        if c54 < best_c5:
            best_h, best_c5 = h4, c54

    # Phase 5: SLSQP polish with very high beta
    if tl() > 30:
        h5, c55 = do_slsqp(best_h, [50000, 100000, 200000], 8000, tl)
        if c55 < best_c5:
            best_h, best_c5 = h5, c55

    # Phase 6: Resolution exploration
    for n_res in [300, 400, 700, 900]:
        if tl() < 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_r = project_h(np.interp(x_new, x_old, best_h), n_res / 2.0)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5 + 0.004:
            h_r2, c5_r2 = do_slsqp(h_r, [5000, 30000, 100000], 5000, tl)
            if c5_r2 < best_c5:
                best_h, best_c5 = h_r2, c5_r2

    # Phase 7: Final coordinated + subgrad
    if tl() > 15:
        n = len(best_h)
        target = n / 2.0
        h7, c57 = coordinated_refine(best_h, target, tl, n_rounds=60)
        if c57 < best_c5:
            best_h, best_c5 = h7, c57

    if tl() > 10:
        n = len(best_h)
        target = n / 2.0
        h8, c58 = subgrad_momentum(best_h, target, 8000, tl, top_k=3, lr_init=0.002, mom=0.9)
        if c58 < best_c5:
            best_h, best_c5 = h8, c58

    # Final SLSQP if time permits
    if tl() > 20:
        h9, c59 = do_slsqp(best_h, [100000, 200000], 6000, tl)
        if c59 < best_c5:
            best_h, best_c5 = h9, c59

    return best_h, best_c5, len(best_h)
