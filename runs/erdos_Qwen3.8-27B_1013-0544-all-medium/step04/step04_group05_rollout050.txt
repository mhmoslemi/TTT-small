import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad_single(h, lag):
    """Gradient of c_lag with respect to h, for a single lag."""
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad += (1.0 - 2.0 * h)
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_max(corr_vals, beta):
    m = np.max(corr_vals)
    x = np.clip(beta * (corr_vals - m), -500, 500)
    return m + np.log(np.sum(np.exp(x)) / len(corr_vals)) / beta

def smooth_max_grad(corr_vals, beta):
    """Return (value, weights) where weights are softmax(beta * (corr - max))."""
    m = np.max(corr_vals)
    x = np.clip(beta * (corr_vals - m), -500, 500)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    val = m + np.log(np.sum(ex) / len(corr_vals)) / beta
    return val, w

def slsqp_optimize(h, target, dx, beta, max_iter):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_fn(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full") * dx
        return smooth_max(corr, beta)
    
    def con_fn(h_in):
        return h_in.sum() - target
    
    def jac_fn(h_in):
        return np.ones(n)
    
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                      options={'maxiter': max_iter, 'ftol': 1e-18})
        return project_h(res.x, target)
    except Exception:
        return h

def subgrad_optimize(h, target, dx, time_fn, max_iters, mode='single'):
    """Subgradient descent with projection. mode='single' uses argmax, mode='top' uses top-m."""
    n = len(h)
    best = h.copy()
    corr_best, _ = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    momentum = 0.8
    
    for it in range(max_iters):
        if time_fn() < 3:
            break
        
        corr_full = np.correlate(best, 1.0 - best, mode="full") * dx
        
        if mode == 'single':
            k_star = np.argmax(corr_full)
            lag = int(k_star) - (n - 1)
            g = compute_subgrad_single(best, lag)
        else:
            m_act = max(3, n // 10)
            top = np.argsort(corr_full)[::-1][:m_act]
            g = np.zeros(n)
            for idx in top:
                lag = int(idx) - (n - 1)
                g += compute_subgrad_single(best, lag)
            g /= m_act
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        vel = momentum * vel + (1.0 - momentum) * g
        vn = np.linalg.norm(vel)
        if vn < 1e-15:

        d = -vel / vn
        
        # Line search
        found = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:

            c5t, _ = compute_c5(ht)
            if c5t < corr_best - 1e-16:

                corr_best = c5t

                if c5t < corr_best:
                    best = ht
                    corr_best = c5t
                    vel *= 0.3
                    no_improve = 0
            if no_improve > 3000:
                break
    
    return best, corr_best

            break

        violations = np.maximum(0, corr - T)
        if np.max(violations) < 1e-10:

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def smooth_max_val(corr_vals, beta):
    m = np.max(corr_vals)
    x = np.clip(beta * (corr_vals - m), -500, 500)
    return m + np.log(np.sum(np.exp(x) / len(corr_vals))) / beta

        return smooth_max_val(c, beta)
    def con(h_in):
        return h_in.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                      options={'maxiter': max_iter, 'ftol': 1e-18})

    except:
        return h
