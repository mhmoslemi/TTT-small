    """Gradient of g at shift s where g_s = sum_{i valid} h[i+s]*(1-h[i])*dx"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:

        # h[i+s] term: ∂/∂h[j] where j = i+s, so j ranges from s to n-1

def peak_peel(h, n, deadline, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        if m <= best_c - 1e-12:
            break
        k_star = int(np.argmax(g))
        s = k_star - (n - 1)
        grad = grad_g_at_shift(h, n, s)
        # Try to reduce g at k_star
        # Take step in direction -grad (reduces g at k_star)
        # But we need to project to feasible set
        # Use line search
        best_step = None
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < m:
                best_step = (lr, h_new, c_new)
                break
        if best_step is None:
            break
        h = best_step[1]
        c = best_step[2]
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

Actually, the problem has a nice structure. Let me think about it in terms of the autocorrelation:

g_k = sum_i h[i] * (1-h[i+k]) * dx (for the valid range)

We want to minimize max_k g_k.

The "autocorrelation" form is:
g_k = sum_i h[i]*dx - sum_i h[i]*h[i+k]*dx (approximately, ignoring boundary effects)
= 1 - R_k (where R_k is the autocorrelation at lag k)

So we want to maximize min_k R_k.

This is related to the "sidelobe suppression" problem in radar/communications, where you design a signal with low sidelobes. The optimal solution for that problem is known to be a "Zadoff-Chu" sequence or related.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_g_shift(h, n, s):
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            grad[s:] += (1.0 - h[:n-s]) * dx
            grad[:n-s] -= h[s:] * dx
    else:
        sp = -s
        if n - sp > 0:
            grad[:n-sp] += (1.0 - h[sp:]) * dx
            grad[sp:] -= h[:n-sp] * dx
    return grad

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, 'full') * dx
    m = g.max()
    sh = alpha * (g - m)
    if sh.max() > 400:
        k = int(np.argmax(g))
        s = k - (n - 1)
        val = m
        obj_grad = grad_g_shift(h, n, s)
    else:
        exp_sh = np.exp(sh)
        probs = exp_sh / exp_sh.sum()
        val = m + np.log(exp_sh.sum()) / alpha
        obj_grad = np.zeros(n)
        active = np.where(probs > 1e-12)[0]
        for k in active:
            s = k - (n - 1)
            obj_grad += alpha * probs[k] * grad_g_shift(h, n, s)
    return val, obj_grad

def lbfgs_opt(h, n, alpha, maxiter, deadline):
    target = n / 2.0
    def fun(x):
        hc = np.clip(x, 0, 1)
        val, _ = smooth_max_and_grad(hc, n, alpha)
        pen = 800.0 * (hc.sum() - target) ** 2
        return val + pen
    def fprime(x):
        hc = np.clip(x, 0, 1)
        _, og = smooth_max_and_grad(hc, n, alpha)
        pg = 1600.0 * (hc.sum() - target)
        xmask = (x >= 0) & (x <= 1)
        return np.where(xmask, og + pg, 0.0)
    try:
        res = minimize(fun, h, jac=fprime, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def peak_peel(h, n, deadline, max_iter=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        k_star = int(np.argmax(g))
        s = k_star - (n - 1)
        grad = grad_g_shift(h, n, s)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            step = lr * f
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < m - 1e-15:
                h = h_new
                found = True
                break
        if not found:
            lr *= 0.4
            stall += 1
            if stall > 30 or lr < 1e-13:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                stall = 0
    return best_h, best_c

def multipeak_peel(h, n, deadline, max_iter=30000, top_k=3):

        g = compute_all_g(best_h, n)

        k_star = int(np.argmax(g))
        s = k_star - (n - 1)
        cand_zeros = []
        cand_ones = []
        for i in sorted(zeros):
            if s >= 0:

                    cand_zeros.append(i)
            else:
                sp = -s
                if i < n - sp and i + sp < n:
                    cand_zeros.append(i)
            if len(cand_zeros) > 50:

        for i in sorted(ones, reverse=True):
            if s >= 0:

            else:
                sp = -s
                if i >= sp and i - sp >= 0:
                    cand_ones.append(i)
            if len(cand_ones) > 50:
                break
        for i in cand_zeros:
            if time.time() > deadline:
                break
            for j in cand_ones:
                best_h[i] = 1.0
                best_h[j] = 0.0
                c = compute_c5(best_h, n)

                    best_h = best_h.copy()
                    ones.remove(j)
                    ones.add(i)
                    zeros.remove(i)
                    zeros.add(j)
                    improved = True
                    break
                else:
                    best_h[i] = 0.0
                    best_h[j] = 1.0

    return best_h, best_c
