        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(8.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            grad += w[j] * subgrad_single_lag(h, lag, n)
        
        velocity = 0.85 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        
        if c_new < c5 - 1e-11:
            h = h_new
            c5 = c_new
            best_h = h.copy()
            stagnant = 0
            lr = min(lr * 1.003, 0.006)
        else:
            stagnant += 1
            velocity *= 0.3
            lr *= 0.93
            if stagnant > 200 and lr < 3e-6:
                break
    
    return best_h, c5

def try_resolution(h_parent, pn, n, deadline, seed):
    """Interpolate to new resolution and optimize."""
    if n == pn:
        return h_parent.copy(), compute_c5(h_parent)
    if n < 40 or n > 600:
        return None, float('inf')
    x_old = np.linspace(0, 2, pn, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h0 = np.interp(x_new, x_old, h_parent)
    h0 = project_feasible(h0, n / 2.0)
    c5 = compute_c5(h0)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    for beta in [200, 2000, 20000, 200000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h0 = h_new
                c5 = c5_new
        except Exception:
            pass
    return h0, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get parent
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)
    
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn
    
    # Phase 1: SLSQP on parent resolution (40% budget)
    d1 = t0 + budget_s * 0.40
    h1, c1 = slsqp_optimize(parent.copy(), pn, d1, seed,
                            betas=[50, 200, 1000, 5000, 20000, 100000],
                            perts=[0.0, 0.003, 0.010, 0.006],
                            maxiter=5000)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn
    
    # Phase 2: Directed polish on parent (20% budget)
    d2 = t0 + budget_s * 0.60
    h2, c2 = directed_polish(best_h, best_n, d2, seed + 10)
    if c2 < best_c5:
        best_h, best_c5, best_n = h2.copy(), c2, best_n
    h2b, c2b = nesterov_subgrad(best_h, best_n, d2, seed + 20)
    if c2b < best_c5:
        best_h, best_c5, best_n = h2b.copy(), c2b, best_n
    
    # Phase 3: Try other resolutions (20% budget)
    d3 = t0 + budget_s * 0.80
    resolutions = [pn - 10, pn - 5, pn + 5, pn + 10, pn + 20,
                   150, 200, 250, 300, 400, 450]
    resolutions = sorted(set(r for r in resolutions if 40 <= r <= 600))
    for n_trial in resolutions:
        if time.time() > d3:
            break
        h_t, c_t = try_resolution(best_h, best_n, n_trial, d3, seed + n_trial)
        if h_t is not None and c_t < best_c5:
            best_h, best_c5, best_n = h_t.copy(), c_t, n_trial
    
    # Phase 4: Final high-precision SLSQP (10% budget)
    d4 = t0 + budget_s * 0.92
    if best_h is not None:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
        for beta in [200000, 500000, 1000000]:
            if time.time() > d4:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 20000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass
    
    # Phase 5: Final micro-polish (8% budget)
    d5 = t0 + budget_s * 0.98
    if best_h is not None:
        h5, c5_5 = directed_polish(best_h, best_n, d5, seed + 500)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5
        h5b, c5_5b = nesterov_subgrad(best_h, best_n, d5, seed + 600)
        if c5_5b < best_c5:
            best_h, best_c5 = h5b, c5_5b
    
    # Final verification
    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
