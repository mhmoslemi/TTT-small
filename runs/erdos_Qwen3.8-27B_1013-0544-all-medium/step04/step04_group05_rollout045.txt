    return np.correlate(h, 1.0 - h, mode="full")

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)

            grad += (1.0 - 2.0 * h)

    return grad

        c = get_correlations(h_in)
        return smooth_max(c[active], 2000) * dx

    def con_fn(h_in):
        return h_in.sum() - target

    def jac_con(h_in):
        return np.ones(n)

    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_con}],
                      options={'maxiter': 4000, 'ftol': 1e-20, 'iprint': 0})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)

def subgrad_loop(h, target, rem_fn, max_iters, m_lags, momentum,
                 perturb_freq, perturb_scale, min_time=3):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    base_step = 0.025

    for it in range(max_iters):
        if rem_fn() < min_time:
            break
        corr = get_correlations(best)
        m_act = min(m_lags, max(5, n // 5))
        top = np.argsort(corr)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top]
        g = compute_subgrad(best, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        vel = momentum * vel + (1.0 - momentum) * g
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            break
        d = -vel / vn
        step = base_step / (1.0 + 0.002 * it)
        found = False
        for _ in range(15):
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
            step *= 0.55
        if not found:
            no_improve += 1
            if no_improve >= perturb_freq and it % perturb_freq == perturb_freq - 1:
                n = len(best)
                freqs = np.array([2, 5, 9, 14, 21, 31, 47])
                p = np.zeros(n)
                for f in freqs:
                    p += np.sin(2 * np.pi * f * np.arange(n) / float(n) + np.random.uniform(0, 2*np.pi))
                p += np.random.randn(n) * 0.7
                pn = np.linalg.norm(p)
                if pn > 1e-12:
                    p /= pn
                for ps in [perturb_scale, perturb_scale * 0.5, perturb_scale * 0.2]:
                    ht = project_h(best + ps * p, target)
                    c5t = compute_c5(ht)
                    if c5t < best_c5:
                        best = ht
                        best_c5 = c5t
                        vel *= 0.3
                        no_improve = 0
                        break
            if no_improve > perturb_freq * 15:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    n0 = len(best_h)
    target0 = n0 / 2.0
    dx0 = 2.0 / n0
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing sharpness
    for beta in [60, 200, 600, 2000, 6000]:
        if rem() < 25:
            break
        best_h, best_c5 = slsqp_step(best_h, target0, dx0, beta, 2500, rem, 15)

    # Phase 2: Subgradient refinement
    best_h, best_c5 = subgrad_loop(
        best_h, target0, rem, max_iters=40000, m_lags=30,
        momentum=0.86, perturb_freq=400, perturb_scale=0.012, min_time=5)

    # Phase 3: Focused SLSQP + subgradient cycle
    for cycle in range(3):
        if rem() < 30:
            break
        best_h, best_c5 = focused_slsqp(best_h, target0, dx0, rem, 10)
        best_h, best_c5 = subgrad_loop(
            best_h, target0, rem, max_iters=15000, m_lags=35,
            momentum=0.9, perturb_freq=300, perturb_scale=0.006, min_time=5)

    # Phase 4: Higher resolution
    if rem() > 120:
        n2 = n0 * 2
        if n2 <= 1200:
            target2 = n2 / 2.0
            dx2 = 2.0 / n2
            x_old = np.linspace(0, 2, n0)
            x_new = np.linspace(0, 2, n2)
            h2 = project_h(np.interp(x_new, x_old, best_h), target2)
            c5_2 = compute_c5(h2)
            for beta in [300, 1000, 4000]:
                if rem() < 30:
                    break
                h2, c5_2 = slsqp_step(h2, target2, dx2, beta, 2000, rem, 10)
            if rem() > 30:
                h2, c5_2 = subgrad_loop(
                    h2, target2, rem, max_iters=20000, m_lags=35,
                    momentum=0.88, perturb_freq=400, perturb_scale=0.008, min_time=5)
            if rem() > 20:
                h2, c5_2 = focused_slsqp(h2, target2, dx2, rem, 8)
            if rem() > 15:
                h2, c5_2 = subgrad_loop(
                    h2, target2, rem, max_iters=12000, m_lags=40,
                    momentum=0.92, perturb_freq=500, perturb_scale=0.004, min_time=3)
            if c5_2 < best_c5:
                best_h = h2
                best_c5 = c5_2

    # Phase 5: Also try n=600 if base was 400
    n_cur = len(best_h)
    if n_cur == n0 and rem() > 80:
        n3 = n0 + n0 // 2
        if n3 % 2 == 0 and n3 != n0 * 2:
            target3 = n3 / 2.0
            dx3 = 2.0 / n3
            x_old3 = np.linspace(0, 2, n0)
            x_new3 = np.linspace(0, 2, n3)
            h3 = project_h(np.interp(x_new3, x_old3, best_h), target3)
            c5_3 = compute_c5(h3)
            if rem() > 30:
                h3, c5_3 = slsqp_step(h3, target3, dx3, 1500, 2000, rem, 10)
            if rem() > 20:
                h3, c5_3 = subgrad_loop(
                    h3, target3, rem, max_iters=12000, m_lags=30,
                    momentum=0.89, perturb_freq=350, perturb_scale=0.007, min_time=5)
            if rem() > 10:
                h3, c5_3 = focused_slsqp(h3, target3, dx3, rem, 5)
            if c5_3 < best_c5:
                best_h = h3
                best_c5 = c5_3

    # Phase 6: Final high-beta SLSQP polish
    n_f = len(best_h)
    target_f = n_f / 2.0
    dx_f = 2.0 / n_f
    if rem() > 15:
        best_h, best_c5 = slsqp_step(best_h, target_f, dx_f, 10000, 4000, rem, 8)

    # Phase 7: Final focused + micro subgradient
    if rem() > 10:
        best_h, best_c5 = focused_slsqp(best_h, target_f, dx_f, rem, 5)
    best_h, best_c5 = subgrad_loop(
        best_h, target_f, rem, max_iters=20000, m_lags=40,
        momentum=0.93, perturb_freq=600, perturb_scale=0.003, min_time=1)

    # Final validation
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
