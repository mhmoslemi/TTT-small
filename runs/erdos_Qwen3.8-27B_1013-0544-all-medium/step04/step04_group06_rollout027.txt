    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    """Extract Fourier coefficients from h"""
    n = len(h)
    # Use real FFT
    H = np.real(np.fft.rfft(h))
    # Take first n_modes+1 coefficients (DC + n_modes harmonics)
    # Store as [a0, a1, b1, a2, b2, ...]
    params = np.zeros(2 * n_modes)
    for m in range(1, n_modes + 1):
        Hm = np.fft.rfft(h)[m]
        params[2*(m-1)] = 2 * np.real(Hm) / n  # a_m
        params[2*(m-1)+1] = -2 * np.imag(Hm) / n  # b_m
    return params

    """Reconstruct h from Fourier parameters"""
    x = np.arange(n)
    h = np.full(n, base)
    for m in range(1, n_modes + 1):
        a = params[2*(m-1)]
        b = params[2*(m-1)+1]
        h += a * np.cos(2*np.pi*m*x/n) + b * np.sin(2*np.pi*m*x/n)
    return h

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient_c5(h, n_active=10):
    """Compute subgradient of C5(h)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g, corr

def slsqp_obj_grad(h, beta):
    """Objective and gradient for SLSQP using soft-max"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    nc = len(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / nc) / beta) * dx
    w = e / se
    
    g = np.zeros(n)
    for j in range(nc):
        if w[j] < 1e-10:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

    def grad(x):
        return slsqp_obj_grad(np.clip(x, 0, 1), beta)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    def eqj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_c5(h, n_active=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g

def slsqp_og(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    nc = len(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / nc) / beta) * dx
    w = e / se
    g = np.zeros(n)
    for j in range(nc):
        if w[j] < 1e-10:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

def slsqp_opt(h_init, n, beta, maxiter=2500):
    target = n / 2.0
    bounds = [(0, 1)] * n

    def obj(x):
        return slsqp_og(np.clip(x, 0, 1), beta)[0]

    def grad(x):
        return slsqp_og(np.clip(x, 0, 1), beta)[1]

    def eq(x):
        return np.sum(x) - target

    def eqj(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

def mom_subgrad(h, target, n_active, max_iter, t0, budget, mom=0.9):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)
    step = 0.01
    stale = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        g = grad_c5(best, n_active)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        v = mom * v - step * g
        vn = np.linalg.norm(v)
        if vn > 1e-15:
            v /= vn
        improved = False
        for s in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            ht = project_sum(best + s * v, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if improved:
            stale = 0
        else:
            step *= 0.75
            stale += 1
            if stale > 150 and it % 80 == 79:
                pert = np.random.randn(n) * 0.008
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    stale = 0
                    v = np.zeros(n)
            if stale > 3000:
                break
    return best, best_c5

def pair_refine(h, target, n_pairs, t0, budget):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_pairs):
        if budget - (time.time() - t0) < 3:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        for delta in [0.005, 0.002, 0.001, 0.0004]:
            h_try = best.copy()
            h_try[i] += delta
            h_try[j] -= delta
            h_try = project_sum(h_try, target)
            c5t = compute_c5(h_try)
            if c5t < best_c5 - 1e-16:
                best = h_try
                best_c5 = c5t
                break
    return best, best_c5
