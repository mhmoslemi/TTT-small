        _, g = smooth_max_obj_and_grad(x, n, alpha)
        return g

    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Objective and gradient for smooth max of correlation values."""
    g = compute_all_g(h, n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    weights = exp_sh / exp_sh.sum()
    obj = m + np.log(exp_sh.sum()) / alpha
    
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for midx in range(ncorr):
        w = weights[midx]
        if w < 1e-10:
            continue
        # Term 1: j in [max(0, midx-n+1), min(n-1, midx)]
        lo1 = max(0, midx - n + 1)
        hi1 = min(n - 1, midx) + 1
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += w * (1.0 - h[idx - midx + n - 1])
        # Term 2: j in [max(0, n-1-midx), min(n-1, 2n-2-midx)]
        lo2 = max(0, n - 1 - midx)
        hi2 = min(n - 1, 2 * n - 2 - midx) + 1
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= w * h[idx + midx - n + 1]
    
    return obj, grad

def obj_and_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def slsqp_step(h, n, alpha, maxiter, deadline):
    def fun(x):
        val, _ = grad_smooth_max(x, n, alpha)
        return val
    
    def gfun(x):
        _, g = grad_smooth_max(x, n, alpha)
        return g
    
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_exact(h, n, deadline, max_iter=20000):
    """Subgradient method on exact max."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        ks = int(np.argmax(g))
        # Gradient of g[ks]
        dx = 2.0 / n
        grad = np.zeros(n)
        lo1 = max(0, ks - n + 1)
        hi1 = min(n - 1, ks) + 1
        idx = np.arange(lo1, hi1)
        grad[idx] += dx * (1.0 - h[idx - ks + n - 1])
        lo2 = max(0, n - 1 - ks)
        hi2 = min(n - 1, 2 * n - 2 - ks) + 1
        idx2 = np.arange(lo2, hi2)
        grad[idx2] -= dx * h[idx2 + ks - n + 1]
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        step = lr / (ng * np.sqrt(n))
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < m - 1e-14:
            h = h_new
            stall = 0
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            lr = min(lr * 1.02, 2.0)
        else:
            stall += 1
            lr *= 0.95
            if lr < 1e-8 or stall > 500:
                break
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=200):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(int(i) for i in np.where(h_bin == 1)[0])
    zeros = set(int(i) for i in np.where(h_bin == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            best_j = None
            best_trial = best_c
            for j in sorted(ones):
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                if c < best_trial - 1e-12:
                    best_trial = c
                    best_j = j
            if best_j is not None:
                h_bin[i] = 1.0
                h_bin[best_j] = 0.0
                ones.add(i)
                ones.remove(best_j)
                zeros.discard(i)
                zeros.add(best_j)
                best_c = best_trial
                improved = True
        if not improved:
            break
    return h_bin, best_c

Each correlation is O(n²) = O(62500) in the worst case (but numpy's correlate is optimized). Let's say each takes about 50 microseconds. Then one sweep takes about 0.8 seconds. A more efficient approach for the binary local search: instead of computing the full correlation each time, I can compute the change in each g[k] when I swap h[i] and h[j]. 

When we set h[i] from 0 to 1 and h[j] from 1 to 0:
- g[k] changes by: (contribution of h[i] to g[k]) - (contribution of h[j] to g[k])
- The contribution of h[i]=1 to g[k] is: (1-h[i-k+n-1]) if i-k+n-1 is valid, minus h[i+k-n+1] if i+k-n+1 is valid (and the index for h is in range)
- Similarly for h[j]
