    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    c = np.correlate(h, 1-h, mode='full')
    
    while time.time() < deadline:
        improved = False
        # Try all pairs (or a random subset for large n)
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng.shuffle(pairs)
        
        for i, j in pairs:
            if time.time() > deadline:
                return h, best_c
            
            a, b = h[i], h[j]
            s = a + b
