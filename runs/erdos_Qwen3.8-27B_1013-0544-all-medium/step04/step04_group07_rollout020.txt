    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_project(h, target_ones):
    """Project to nearest binary vector with exactly target_ones ones."""
    n = len(h)
    h = h.copy()
    # Sort by value descending, take top target_ones as 1
    idx = np.argsort(h)[::-1]
    z = np.zeros(n)
    z[idx[:int(target_ones)]] = 1.0
    return z

    curr_c5 = best_c5

def fast_c5(h):

    n = len(h)
    M = 1 << (2 * n - 1).bit_length()

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

def to_binary(h, target):

    n = len(h)
    # Use soft thresholding: sort by value, pick top target
    order = np.argsort(h)[::-1]
    z = np.zeros(n)
    z[order[:int(target)]] = 1.0
    return z

def local_refine_binary(h, n, target, t0, budget, max_sweeps=200):
    """Local search: try all 1→0 and 0→1 swaps, accept if improves."""
    best_c5 = fast_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fast_c5(h):
    n = len(h)
    M = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(h, M)
    fv = np.fft.rfft(1.0 - h, M)
    circ = np.fft.irfft(fa * np.conj(fv), M)
    out = np.concatenate([circ[M-n+1:M], circ[:n]])
    return float(np.max(out) * 2.0 / n)

def fast_corr(h):
    n = len(h)
    M = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(h, M)
    fv = np.fft.rfft(1.0 - h, M)
    circ = np.fft.irfft(fa * np.conj(fv), M)
    return np.concatenate([circ[M-n+1:M], circ[:n]])

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    z = np.zeros(n)
    z[order[:int(target)]] = 1.0
    return z

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)

def sa_binary(h0, n, target, t0, budget, seed=42, n_restarts=1):
    rng = np.random.RandomState(seed)
    h = to_binary(h0, target)
    best_h = h.copy()
    best_c5 = fast_c5(h)
    
    for restart in range(n_restarts):
        h = to_binary(h0 + 0.1 * rng.randn(n), target) if restart > 0 else best_h.copy()
        curr_c5 = fast_c5(h)
        T = 0.005
        alpha = 0.9997
        T_min = 1e-7
        
        while time.time() - t0 < budget - 10 and T > T_min:
            r = rng.random()
            h_new = h.copy()
            
            if r < 0.4:
                # Swap: move 1 from i to j
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    T *= alpha
                    continue
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
            elif r < 0.7:
                # Block flip with compensation
                bs = rng.choice([1, 2, 3, 5, 8])
                if bs >= n:
                    bs = n // 4
                start = rng.randint(0, n - bs)
                old_block = h[start:start+bs].copy()
                h_new[start:start+bs] = 1.0 - old_block
                # Compensate
                change = (bs - 2 * old_block.sum())
                if change > 0:
                    # Need to reduce sum by 'change': flip 'change' ones to zeros outside block
                    mask = np.ones(n, dtype=bool)
                    mask[start:start+bs] = False
                    ones_out = np.where((h == 1) & mask)[0]
                    if len(ones_out) >= int(change):
                        flip = rng.choice(ones_out, int(change), replace=False)
                        h_new[flip] = 0
                        change = 0
                    else:
                        h_new = h.copy()
                        T *= alpha
                        continue
                elif change < 0:
                    # Need to increase sum by -change: flip -change zeros to ones outside block
                    mask = np.ones(n, dtype=bool)
                    mask[start:start+bs] = False
                    zeros_out = np.where((h == 0) & mask)[0]
                    if len(zeros_out) >= int(-change):
                        flip = rng.choice(zeros_out, int(-change), replace=False)
                        h_new[flip] = 1
                        change = 0
                    else:
                        h_new = h.copy()
                        T *= alpha
                        continue
            else:
                # Adjacent swap: swap h[i] and h[i+1]
                i = rng.randint(0, n - 1)
                if h[i] != h[i+1]:
                    h_new[i], h_new[i+1] = h_new[i+1], h_new[i]
                else:
                    T *= alpha
                    continue
            
            if abs(h_new.sum() - target) > 1e-10:
                T *= alpha
                continue
            
            c5_new = fast_c5(h_new)
            delta = c5_new - curr_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                h = h_new
                curr_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
            T *= alpha
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fast_c5(h):
    n = len(h)
    M = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(h, M)
    fv = np.fft.rfft(1.0 - h, M)
    circ = np.fft.irfft(fa * np.conj(fv), M)
    out = np.concatenate([circ[M-n+1:M], circ[:n]])
    return float(np.max(out) * 2.0 / n)

def fast_corr(h):
    n = len(h)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
