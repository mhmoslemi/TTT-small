  - If j = n+m (i.e., m = j-n): h[j]*(1-h[n]) changes from (1-h[n]) to 0, so change is -(1-h[n])
  - If j = n: h[n+m]*(1-h[j]) changes from h[j+m]*0 to h[j+m]*1, so change is h[j+m]

So ΔC[m] for swapping i(0→1) and j(1→0):
- From i as "source" (i = n+m, so m = i-n): + (1-h[i-m]) for m such that n=i-m is valid
- From i as "target" (i = n, so m is any valid lag where i+m < N): - h[i+m]
- From j as "source" (j = n+m, so m = j-n): - (1-h[j-m]) for m such that n=j-m is valid  
- From j as "target" (j = n): + h[j+m] for m such that j+m < N

This is getting complex. Let me just compute it directly in code:

```python
def binary_swap_delta(h, N, i, j):
    """Compute delta in C[m] for all m when swapping h[i](0->1) and h[j](1->0)."""
    # Current overlap
    C = np.correlate(h, 1.0-h, 'full')
    # After swap
    h[i], h[j] = 1.0, 0.0
    C_new = np.correlate(h, 1.0-h, 'full')
    h[i], h[j] = 0.0, 1.0
    return C_new - C
```

But this is O(N) per swap (due to the correlate call being O(N²) actually). Hmm.

Actually np.correlate for 1D is O(N²) in the worst case. For N=250, that's 62500 operations per swap. With N²/4 swaps in a full sweep, that's about 10^9 operations per sweep. Too slow.

```python
def swap_delta_fast(h, N, i, j):
    """Fast delta for swapping h[i](0->1) and h[j](1->0) on C = correlate(h, 1-h, 'full')."""
    # C[m] = sum_n h[n+m]*(1-h[n])
    # Change at i (goes 0->1):
    #   As "source" (i = n+m): affects C[m] where n = i-m, valid when 0<=i-m<N and 0<=i<N
    #     C[m] += (1-h[i-m]), for m = i-n where 0<=n<N, 0<=n+m<N => m ranges: 0<=i-m<N => i-N<m<=i
    #     Also need 0<=i-m<N => i-N < m <= i, and m in [-(N-1), N-1]
    #   As "target" (i = n): affects C[m] where n = i, valid when 0<=i<N and 0<=i+m<N
    #     C[m] was h[i+m]*(1-h[i]) = h[i+m]*1, now h[i+m]*0
    #     C[m] -= h[i+m], for m such that 0<=i+m<N => 0<=m<N-i, m>=0 (and m can be negative? 
    #     No: n=i, so m can be anything such that 0<=i+m<N, i.e., -i <= m <= N-1-i)
    
    # Let me just enumerate:
    delta = np.zeros(2*N-1)  # C has indices 0 to 2N-2
    
    # i goes 0->1
    # i as target (n=i): C[m] = sum_n h[n+m](1-h[n]), term at n=i: h[i+m](1-h[i])
    #   before: h[i+m]*1, after: h[i+m]*0, change = -h[i+m]
    #   valid m: 0<=i+m<N => -i <= m <= N-1-i
    for m in range(-i, N-i):
        if 0 <= i+m < N:
            delta[N + m] -= h[i+m]  # C index is N+m for lag m
    
    # i as source (n+m=i, n=i-m): C[m] = ...h[n+m](1-h[n])...
    #   term: h[i]*(1-h[i-m]), before: 0, after: 1*(1-h[i-m])
    #   change: +(1-h[i-m])
    #   valid: 0<=i-m<N, 0<=m... well m = i-n, n=i-m, need 0<=i-m<N
    for m in range(max(-(N-1), i-(N-1)), min(N-1, i)+1):
        n = i - m
        if 0 <= n < N and 0 <= n+m < N:
            delta[N + m] += (1.0 - h[n])
    
    # j goes 1->0
    # j as target (n=j): C[m] term at n=j: h[j+m](1-h[j])
    #   before: h[j+m]*0, after: h[j+m]*1, change: +h[j+m]
    for m in range(-j, N-j):
        if 0 <= j+m < N:
            delta[N + m] += h[j+m]
    
    # j as source (n+m=j, n=j-m): 
    #   term: h[j]*(1-h[j-m]), before: 1*(1-h[j-m]), after: 0
    #   change: -(1-h[j-m])
    for m in range(max(-(N-1), j-(N-1)), min(N-1, j)+1):
        n = j - m
        if 0 <= n < N and 0 <= n+m < N:
            delta[N + m] -= (1.0 - h[n])
    
    return delta
```

Hmm, this is getting complicated and error-prone. Let me simplify by just computing the change directly:

When h changes at positions i and j:
- C[m] = sum_n h[n+m](1-h[n])
- The only terms that change are those where n or n+m equals i or j.

So for each m, C[m] changes only if:
- i = n (i is the "second" index) → m such that n=i is valid, i.e., 0≤i<N and 0≤i+m<N
- i = n+m (i is the "first" index) → m = i-n for some valid n
- j = n → similar
- j = n+m → similar

For a given m, the change in C[m] is:
- If i is in the valid range as n: old contribution was h[i+m]*(1-h[i]) = h[i+m]*1, new is h[i+m]*0. Change: -h[i+m]
- If i is in the valid range as n+m (i.e., m = i-n for some valid n, which means n=i-m is valid): old was 0*(1-h[i-m]) = 0, new is 1*(1-h[i-m]). Change: +(1-h[i-m])
- If j is in the valid range as n: old was h[j+m]*(1-h[j]) = h[j+m]*0, new is h[j+m]*1. Change: +h[j+m]
- If j is in the valid range as n+m (n=j-m valid): old was 1*(1-h[j-m]), new is 0. Change: -(1-h[j-m])

So for a given m:
ΔC[m] = -h[i+m]*1_{valid} + (1-h[i-m])*1_{valid} + h[j+m]*1_{valid} - (1-h[j-m])*1_{valid}

where "valid" means the corresponding index is in [0, N-1].

This is O(1) per m! So evaluating all m is O(N) per swap. Much better.

Let me code this cleanly:

```python
def evaluate_swap(h, N, i, j, C):
    """Return new max of C after swapping h[i](0->1) and h[j](1->0).
    C is the current correlate vector (length 2N-1).
    Returns new_C or just the new max."""
    dx = 2.0 / N
    new_max = 0.0
    for m_idx in range(2*N-1):
        m = m_idx - (N-1)  # actual lag
        C_new = C[m_idx]
        # i as target (n=i): valid if 0<=i<N and 0<=i+m<N
        if 0 <= i + m < N:
            C_new -= h[i+m]
        # i as source (n+m=i, n=i-m): valid if 0<=i-m<N
        if 0 <= i - m < N:
            C_new += (1.0 - h[i-m])
        # j as target (n=j): valid if 0<=j<N and 0<=j+m<N
        if 0 <= j + m < N:
            C_new += h[j+m]
        # j as source (n+m=j, n=j-m): valid if 0<=j-m<N
        if 0 <= j - m < N:
            C_new -= (1.0 - h[j-m])
        if C_new * dx > new_max:
            new_max = C_new * dx
    return new_max
```

This is O(N) per swap with a simple loop. In Python this might be slow for N=250 (250 iterations per swap, ~15625 swaps in a full sweep = ~4M iterations). Should be under a second.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, N):
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_C(h, N):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, N):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_analytical(h, N, k):
    dx = 2.0 / N
    grad = np.zeros(N)
    lo = max(0, -k)
    hi = min(N - 1, N - 1 - k)
    if lo > hi:
        return grad
    n_arr = np.arange(lo, hi + 1)
    j_arr = n_arr + k
    grad[j_arr] += (1.0 - h[n_arr])
    grad[n_arr] += (-h[j_arr])
    return grad * dx

def momentum_subgrad(h, N, deadline, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, N)
    v = np.zeros(N)
    lr = 0.5 / N
    beta = 0.85
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        C = compute_C(h, N)
        m = C.max()
        top_k = np.argsort(C)[-3:]
        s = np.zeros(N)
        for k in top_k:
            s += subgrad_analytical(h, N, int(k))
        s /= len(top_k)
        v = beta * v - lr * s
        h_new = project_feasible(h + v, N)
        c_new = compute_c5(h_new, N)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            patience += 1
            if patience > 30:
                lr *= 0.8
                v *= 0.5
                patience = 0
                if lr < 1e-10:
                    break
    return best_h, best_c

def eval_swap(h, N, i, j, C, mags):
    dx = 2.0 / N
    new_C = C.copy()
    idx = (mags >= -i) & (mags < N - i)
    if idx.any():
        new_C[idx] -= h[i + mags[idx]]
    idx = (mags >= i - (N - 1)) & (mags <= i)
    if idx.any():
        new_C[idx] += (1.0 - h[i - mags[idx]])
    idx = (mags >= -j) & (mags < N - j)
    if idx.any():
        new_C[idx] += h[j + mags[idx]]
    idx = (mags >= j - (N - 1)) & (mags <= j)
    if idx.any():
        new_C[idx] -= (1.0 - h[j - mags[idx]])
    return float(np.max(new_C) * dx)

def binary_local_search(h_init, N, deadline, max_sweeps=500):
    dx = 2.0 / N
    target = N // 2
    h = np.zeros(N)
    h[np.argsort(h_init)[::-1][:target]] = 1.0
    C = compute_C(h, N)
    best_c = float(np.max(C) * dx)
    mags = np.arange(2 * N - 1) - (N - 1)
    ones = set(np.where(h == 1)[0])
    zeros = set(np.arange(N)) - ones

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = sorted(zeros)
        o_list = sorted(ones)
        for i in z_list:
            if time.time() > deadline:
                break
            for j in o_list:
                if i == j:
                    continue
                c_new = eval_swap(h, N, i, j, C, mags)
                if c_new < best_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    best_c = c_new
                    C = compute_C(h, N)
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    o_list = sorted(ones)
                    z_list = sorted(zeros)
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def coord_descent(h, N, deadline, max_rounds=200):
    dx = 2.0 / N
    best_c = compute_c5(h, N)
    C = compute_C(h, N)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(N)
        for i in order:
            if time.time() > deadline:
                break
            C = compute_C(h, N)
            k_star = int(np.argmax(C))
            g = subgrad_analytical(h, N, k_star)
            if g[i] > 1e-10:
                h[i] = max(0.0, h[i] - 0.1)
            elif g[i] < -1e-10:
                h[i] = min(1.0, h[i] + 0.1)
            else:
                continue
            h = project_feasible(h, N)
            c_new = compute_c5(h, N)
            if c_new < best_c:
                best_c = c_new
                improved = True
            else:
                h = project_feasible(h, N)
        if not improved:
            break
    return h, best_c

def slsqp_smooth(h, N, alpha, maxiter, deadline):
    def obj(x):
        C = np.correlate(x, 1.0 - x, mode='full') * (2.0 / N)
        m = C.max()
        sh = alpha * (C - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - N / 2.0
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, N)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        N = len(h)
    except Exception:
        N = 200
        h = np.full(N, 0.5)

    best_h = h.copy()
    best_c = compute_c5(best_h, N)

    # Phase 1: SLSQP with increasing alpha (30% budget)
    p1 = t0 + 0.30 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, N, alpha, 800, p1)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Momentum subgradient (20% budget)
    p2 = t0 + 0.50 * (budget_s - 5)
    h2, c2 = momentum_subgrad(best_h.copy(), N, p2)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary local search (30% budget)
    p3 = t0 + 0.80 * (budget_s - 5)
    hb, cb = binary_local_search(best_h.copy(), N, p3)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Try binary from perturbed continuous
    hp = project_feasible(best_h + 0.15 * rng.randn(N), N)
    hb2, cb2 = binary_local_search(hp, N, p3)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 4: Try nearby n values (10% budget)
    p4 = t0 + 0.90 * (budget_s - 5)
    for nh in [N + 20, N - 20, N + 50]:
        if nh < 50 or time.time() > p4:
            continue
        xo = np.linspace(0, 2, N)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, ch = momentum_subgrad(hh, nh, p4)
        hb, cbh = binary_local_search(hh, nh, p4)
        if cbh < best_c:
            best_c = cbh
            best_h = hb.copy()
            N = nh

    # Phase 5: Coordinate descent + final SLSQP (10% budget)
    if time.time() < deadline - 10:
        hc, cc = coord_descent(best_h.copy(), N, deadline - 10, max_rounds=100)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
        if time.time() < deadline - 5:
            hf = slsqp_smooth(best_h.copy(), N, 200000, 500, deadline - 5)
            cf = compute_c5(hf, N)
            if cf < best_c:
                best_c = cf
                best_h = hf.copy()

    best_h = project_feasible(best_h, N)
    best_c = compute_c5(best_h, N)
    return best_h, best_c, N
```
