import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_with_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sml_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def sml_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-18:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += wk * (1.0 - h[lag:])
            g[lag:] += wk * (-h[:n-lag])
        else:
            la = -lag
            g[la:] += wk * (1.0 - h[:n-la])
            g[:n-la] += wk * (-h[la:])
    return g * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def run_slsqp(h0, target, beta, maxiter=3000, ftol=1e-14):
    n = len(h0)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, jac=lambda x: sml_grad(x, beta, dx),
                     args=(beta, dx), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()

    rng = np.random.RandomState(seed)  # need to pass seed

            break

            active_lags = np.where(corr >= thr)[0]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-11) & (h < 1.0 - 1e-11)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sml_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx

def sml_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(sh) / np.sum(np.exp(sh))
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += wk * (1.0 - h[lag:])
            g[lag:] += wk * (-h[:n-lag])
        else:
            la = -lag
            g[la:] += wk * (1.0 - h[:n-la])
            g[:n-la] += wk * (-h[la:])
    return g * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def run_slsqp(h0, target, beta, maxiter=4000, ftol=1e-14):
    n = len(h0)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, jac=lambda x: sml_grad(x, beta, dx),
                     args=(beta, dx), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def greedy_fw(h0, target, t0, budget, n_iter=20000, m_lags=25):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    rng = np.random.RandomState(12345)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        if it % 30 == 0:
            _, corr = compute_c5_corr(h)
            mx = np.max(corr)
            thr = mx - 0.08 * mx
            active = np.where(corr >= thr)[0]
            if len(active) > m_lags:
                active = active[np.argsort(corr[active])[::-1][:m_lags]]
            lag_grads = np.zeros((len(active), n))
            for li, k in enumerate(active):
                lag = int(k) - (n - 1)
                wg = np.zeros(n)
                if lag >= 0:
                    wg[:n-lag] += (1.0 - h[lag:])
                    wg[lag:] += (-h[:n-lag])
                else:
                    la = -lag
                    wg[la:] += (1.0 - h[:n-la])
                    wg[:n-la] += (-h[la:])
                lag_grads[li] = wg
        if it % 30 == 0 and len(active) > 0:
            # Find best swap direction using lag gradients
            # Weighted gradient: sum of active lag gradients
            wg = lag_grads.sum(axis=0)
            i_dec = np.argmax(wg)
            i_inc = np.argmin(wg)
            if i_dec == i_inc:
                i_inc = (i_inc + 1) % n
        else:
            i_inc, i_dec = rng.choice(n, 2, replace=False)
        for delta in (0.04, 0.02, 0.01, 0.005):
            if h[i_inc] + delta > 1.0 or h[i_dec] - delta < 0.0:
                continue
            trial = h.copy()
            trial[i_inc] += delta
            trial[i_dec] -= delta
            trial = project_h(trial, target)
            c = compute_c5(trial)
            if c < best_c5 - 1e-16:
                h, best_h, best_c5, stall = trial, trial.copy(), c, 0
                break
        else:
            stall += 1
            if stall > 3000:
                # Perturb to escape
                h = project_h(h + rng.normal(0, 0.01, n), target)
                stall = 0
    return best_h, best_c5

def binary_snap(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 0.5:
        zeros = np.where(z == 0)[0]
        order = np.argsort(h[zeros])[::-1][:int(diff)]
        z[zeros[order]] = 1.0
    elif diff < -0.5:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])[:int(-diff)]
        z[ones[order]] = 0.0
    return z

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)

def make_fourier_init(n, freq, amp, phase):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    target_parent = len(h_parent) / 2.0

    # Phase 1: Multi-start SLSQP at parent resolution
    n_pts = len(h_parent)
    target = n_pts / 2.0

    initializations = [best_h.copy()]
    # Fourier-based inits
    for freq, amp in [(3, 0.3), (5, 0.35), (7, 0.25), (11, 0.3)]:
        h_init = make_fourier_init(n_pts, freq, amp, 0)
        h_init = project_h(h_init, target)
        initializations.append(h_init)
    # Perturbed parent
    rng0 = np.random.RandomState(seed)
    h_init = project_h(h_parent + rng0.normal(0, 0.05, n_pts), target)
    initializations.append(h_init)

    for h_init in initializations:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        cur = h_init.copy()
        for beta in [200, 600, 2000]:
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                break
            cur = run_slsqp(cur, target, beta, maxiter=3000)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts

    # Phase 2: Greedy Frank-Wolfe refinement
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        cur = best_h.copy()
        cur, c = greedy_fw(cur, target, t0, budget_s, n_iter=15000)
        if c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c, n_pts

    # Phase 3: Binary snap + re-optimize
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        hb = binary_snap(best_h, target)
        cb = compute_c5(hb)
        cur = hb if cb < best_c5 else best_h.copy()
        cur = run_slsqp(cur, target, 1000, maxiter=3000)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c, n_pts

    # Phase 4: Try finer resolution
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        for n_fine in [512, 448]:
            rem = budget_s - (time.time() - t0)
            if rem < 60:
                break
            t_f = n_fine / 2.0
            cur = project_h(resample(best_h, n_fine), t_f)
            for beta in [500, 2000]:
                rem = budget_s - (time.time() - t0)
                if rem < 25:
                    break
                cur = run_slsqp(cur, t_f, beta, maxiter=3000)
                c = compute_c5(cur)
                if c < best_c5:
                    best_h, best_c5, best_n = cur.copy(), c, n_fine
            rem = budget_s - (time.time() - t0)
            if rem > 40:
                cur, c = greedy_fw(cur, t_f, t0, budget_s, n_iter=8000)
                if c < best_c5:
                    best_h, best_c5, best_n = cur.copy(), c, n_fine

    # Phase 5: Final polish at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        nf = len(best_h)
        tf = nf / 2.0
        best_h = run_slsqp(best_h, tf, 5000, maxiter=5000, ftol=1e-15)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 25:
        best_h, best_c5 = greedy_fw(best_h, len(best_h)/2.0, t0, budget_s, n_iter=10000, m_lags=40)

    # Final projection and verification
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
