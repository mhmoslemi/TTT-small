from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0

def weighted_subgrad(h, target, n, n_iter, t0, budget, top_k=8, decay=1.0):
    cur = h.copy()
    best_local = compute_c5(cur)
    stall = 0
    step = 0.03
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        # Weight by excess over mean
        mean_c = corr.mean()
        weights = np.clip(corr - mean_c, 0, None) ** decay
        total_w = weights.sum()
        if total_w < 1e-15:
            break
        weights /= total_w
        # Get top k weights
        top_idx = np.argsort(weights)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            w = weights[idx]
            if w < 1e-10:
                continue
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += w * (1.0 - cur[lag:])
                    grad[lag:] -= w * cur[:n - lag]
            else:
                la = -lag
                grad[la:] += w * (1.0 - cur[:n - la])
                grad[:n - la] -= w * cur[la:]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        trial = project_h(cur - step * grad, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.02, 0.08)
        else:
            stall += 1
            step *= 0.85
            if step < 0.001:
                step = 0.03
            trial = project_h(cur - step * grad, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                cur = trial
                best_local = c_try
                stall = 0
            else:

                    cur = trial
                    best_local = c_try
                    step *= 0.5
                else:
                    step *= 0.7
        if stall > 300:
            break
    return cur, best_local

    cur_max = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_max
    rng = np.random.default_rng(12345)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        # Try a swap
        a = list(ones)[rng.integers(len(ones))]
        b = list(zeros)[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        ones.discard(a)
        zeros.discard(b)
        ones.add(b)
        zeros.add(a)
        
        new_c5 = compute_c5(h)
        if new_c5 <= cur_max + 1e-10:
            cur_max = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[a] = 1.0
            h[b] = 0.0
            ones.discard(b)
            zeros.discard(a)
            ones.add(a)
            zeros.add(b)
            if it % 1000 == 0:
                # Occasionally accept worse solutions (simulated annealing style)
                temp = 0.005 * (0.5) ** (it / 5000.0)
                if rng.random() < np.exp(-(new_c5 - cur_max) / max(temp, 1e-15)):
                    h[a] = 0.0
                    h[b] = 1.0
                    ones.discard(a)
                    zeros.discard(b)
                    ones.add(b)
                    zeros.add(a)
                    cur_max = new_c5
    return best_h, best_c5

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

subject to: f_k(h) ≤ t for all k

This is a non-convex QCQP. However, we can use cvxpy's `Minimize` with a custom objective... actually cvxpy handles convex problems. Since f_k(h) = sum_i h[i] - h[i]*h[i+k], and the term -h[i]*h[i+k] is not concave (the Hessian of h[i]*h[i+k] is [[0,1],[1,0]] which has eigenvalues 1 and -1), this is non-convex.

But wait! If we fix t and check feasibility, we can use a different approach. Actually, let me think...

For the constraint f_k(h) ≤ t:
  sum_i h[i] - sum_i h[i]*h[i+k] ≤ t

The left side is sum_i h[i] * (1 - h[i+k]). Each term h[i]*(1-h[i+k]) is a product of two variables in [0,1]. This product is NOT convex.

However, we can use the McCormick envelope. For x, y ∈ [0,1]:
  xy ≥ 0
  xy ≤ x
  xy ≤ y
  xy ≥ x + y - 1

So h[i]*(1-h[i+k]) is bounded by:
  0 ≤ h[i]*(1-h[i+k]) ≤ min(h[i], 1-h[i+k])
  h[i]*(1-h[i+k]) ≥ h[i] + (1-h[i+k]) - 1 = h[i] - h[i+k]

Wait, that's: h[i]*(1-h[i+k]) ≥ h[i] - h[i+k] (from the McCormick lower bound with x=h[i], y=1-h[i+k])

And h[i]*(1-h[i+k]) ≤ h[i] (trivially)
And h[i]*(1-h[i+k]) ≤ 1-h[i+k]

So the McCormick upper bound is min(h[i], 1-h[i+k]) which isn't linear.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=10000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_subgrad(h, target, n, n_iter, t0, budget, top_k=6):
    cur = h.copy()
    y = h.copy()
    best_local = compute_c5(cur)
    best_h = cur.copy()
    step = 0.03
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        alpha = 2.0 / (it + 2.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        max_val = corr.max()
        threshold = max_val - (corr.std() if corr.std() > 0 else 0.1 * max_val)
        top_mask = corr >= max(0, threshold)
        if top_mask.sum() == 0:
            top_mask[np.argmax(corr)] = True
        if top_mask.sum() > top_k:
            idx = np.argsort(corr)[::-1][:top_k]
            top_mask = np.zeros(corr.shape, dtype=bool)
            top_mask[idx] = True
        grad = np.zeros(n)
        for idx in np.where(top_mask)[0]:
            lag = int(idx) - (n - 1)
            if 0 <= lag < n:
                grad[:n - lag] += (1.0 - y[lag:])
                grad[lag:] -= y[:n - lag]
            elif -n < lag < 0:
                la = -lag
                grad[la:] += (1.0 - y[:n - la])
                grad[:n - la] -= y[la:]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        new_y = project_h(y - step * grad, target)
        c_try = compute_c5(new_y)
        if c_try < best_local - 1e-15:
            cur = new_y
            best_local = c_try
            best_h = cur.copy()
            stall = 0
            step = min(step * 1.005, 0.08)
        else:
            stall += 1
            step *= 0.82
            if step < 0.003:
                step = 0.02
            new_y = project_h(y - step * grad, target)
            c_try = compute_c5(new_y)
            if c_try < best_local - 1e-15:
                cur = new_y
                best_local = c_try
                best_h = cur.copy()
            else:
                step *= 0.7
        y = new_y
        if stall > 400:
            break
    return best_h, best_local

def binary_anneal(h_bin, n, t0, budget, n_iter=100000):
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    ones = np.where(h == 1)[0]
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        a = ones[rng.integers(len(ones))]
        b = rng.integers(n)
        if b == a:
            continue
        if h[b] == 1:
            continue
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 <= cur_c5 + 1e-12:
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = 0.01 * (0.001) ** frac
            if rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-15)):
                cur_c5 = new_c5
                ones = np.where(h == 1)[0]
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[a] = 1.0
                h[b] = 0.0
                if it > 0 and it % 10000 == 0:
                    ones = np.where(h == 1)[0]
    return best_h, best_c5

def fine_cd(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    rng = np.random.default_rng(77)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        n_try = min(n, 80)
        indices = rng.choice(n, size=n_try, replace=False)
        for j in indices:
            for d in [0.001, -0.001, 0.003, -0.003, 0.007, -0.007]:
                trial = cur.copy()
                trial[j] += d
                if trial[j] < 0 or trial[j] > 1:
                    continue
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin
