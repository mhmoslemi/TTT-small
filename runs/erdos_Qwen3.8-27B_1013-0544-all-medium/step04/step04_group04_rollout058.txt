One idea I haven't tried: **Frank-Wolfe (conditional gradient) method**. This works by:
1. Compute gradient g
2. Find s = argmin_{s in feasible set} <g, s>
3. Take a step towards s

The feasible set here is {h : 0 <= h <= 1, sum(h) = n/2}. The linear minimization over this set can be solved analytically: sort the components of g, and the optimal s has s_i = 1 for the smallest g_i, s_i = 0 for the largest, and one intermediate value to satisfy the sum constraint.

This could be more effective than subgradient because it always moves towards a vertex of the feasible set, which might find better solutions.

def frank_wolfe_line(g, target):
    """Find s in {0<=s<=1, sum(s)=target} minimizing <g, s>"""
    n = len(g)
    sorted_idx = np.argsort(g)
    # We want s_i = 1 for small g_i, s_i = 0 for large g_i
    # With sum constraint: need exactly target ones (if target is integer)
    # For fractional: one intermediate value
    # Actually, the optimal solution to min <g,s> s.t. 0<=s<=1, sum(s)=target
    # is: sort g, set s=1 for the first floor(target) elements,
    # s=frac(target) for one more, s=0 for the rest.
    # But we also have the upper bound s<=1.
    # The solution is: sort g ascending, s_i = 1 for i < k, s_k = r, s_i = 0 for i > k
    # where k + r = target
    k = int(np.floor(target))
    r = target - k
    s = np.zeros(n)
    s[sorted_idx[:k]] = 1.0
    if r > 1e-10:
        s[sorted_idx[k]] = r
    return s

        for g_try in [gamma, min(1.0, gamma * 2), min(1.0, gamma * 3), 1.0, 0.5]:
            h_try = (1.0 - g_try) * h_cur + g_try * s
            h_try = project_h(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-15:
                h_cur = h_try
                c5_cur = c5_try
                break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    return best_h, best_c5

def do_nesterov(h, target, n_iter, start_time, budget, top_k=12):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    step = 0.05
    decay = 0.9997
    mom = 0.9
    for it in range(n_iter):
        if time.time() - start_time > budget - 10:
            break
        grad, _ = top_k_grad(h_cur, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        h_y = np.clip(h_cur + mom * v, 0.0, 1.0)
        h_new = h_y - step * grad
        v = h_new - h_cur
        h_cur = project_h(h_new, target)
        step *= decay
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
        if it % 800 == 0:
            v *= 0.5
    return best_h, best_c5

def do_perturb(h, target, n_trials, start_time, budget, scale):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(n_trials):
        if time.time() - start_time > budget - 5:
            break
        h_p = project_h(h + np.random.randn(n) * scale, target)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP on parent
    h1, c51 = do_slsqp(best_h, [50, 200, 1000, 5000, 20000, 60000], 8000, start_time, budget_s)
    if c51 < best_c5:
        best_h, best_c5 = h1, c51

    # Phase 2: Frank-Wolfe for global exploration
    n = len(best_h)
    h2, c52 = do_frank_wolfe(best_h, n / 2.0, 30000, start_time, budget_s, top_k=15)
    if c52 < best_c5:
        best_h, best_c5 = h2, c52

    # Phase 3: Nesterov subgradient
    n = len(best_h)
    h3, c53 = do_nesterov(best_h, n / 2.0, 40000, start_time, budget_s, top_k=15)
    if c53 < best_c5:
        best_h, best_c5 = h3, c53

    # Phase 4: Multi-resolution SLSQP
    for n_res in [150, 200, 250, 300, 400, 500, 600]:
        if time.time() - start_time > budget_s - 80:
            break
        n_cur = len(best_h)
        x_old = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        hr, c5r = do_slsqp(h_res, [500, 3000, 15000, 60000], 7000, start_time, budget_s)
        if c5r < best_c5:
            best_h, best_c5 = hr, c5r

    # Phase 5: Final FW + Nesterov
    n = len(best_h)
    h5, c55 = do_frank_wolfe(best_h, n / 2.0, 20000, start_time, budget_s, top_k=20)
    if c55 < best_c5:
        best_h, best_c5 = h5, c55
    n = len(best_h)
    h6, c56 = do_nesterov(best_h, n / 2.0, 30000, start_time, budget_s, top_k=20)
    if c56 < best_c5:
        best_h, best_c5 = h6, c56

    # Phase 6: Final SLSQP polish
    if time.time() - start_time < budget_s - 30:
        h7, c57 = do_slsqp(best_h, [30000, 80000, 200000], 10000, start_time, budget_s)
        if c57 < best_c5:
            best_h, best_c5 = h7, c57

    # Phase 7: Fine perturbation search
    n = len(best_h)
    for scale in [0.008, 0.004, 0.002, 0.001]:
        if time.time() - start_time > budget_s - 5:
            break
        hp, c5p = do_perturb(best_h, n / 2.0, 400, start_time, budget_s, scale)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    return best_h, best_c5, len(best_h)
