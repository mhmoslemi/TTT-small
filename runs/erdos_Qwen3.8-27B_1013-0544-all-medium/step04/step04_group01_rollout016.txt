        x_new = np.linspace(0, 2, n_try)

        if c < best_c:

            best_h = h
    
    # Phase 3: Final local search
    if time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = local_search(best_h, n, deadline - 2)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)

One idea: **annealing**. Start with low alpha (smooth, convex-like landscape), optimize, then increase alpha (sharper, more accurate to the true max). This is what the previous code does with the alpha schedule [1000, 5000, 20000, 100000, 500000].

Another idea: **random restarts with perturbation**. After getting a good solution, perturb it and re-optimize. This can escape local minima.

import numpy as np

def compute_corr_vec(h, n):

    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Numerically stable log-sum-exp
    a = shifted.max()
    if a > 500:
        return float(m + (a + np.log(np.sum(np.exp(shifted - a)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def soft_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    a = shifted.max()
    weights = np.exp(shifted - a)
    weights /= weights.sum()
    # Gradient of sum_k c[k] * w[k] where w[k] = exp(alpha*(c[k]-m)) / sum(...)
    # = sum_k w[k] * dc[k]/dx
    # dc[k]/dx is the gradient of the k-th correlation
    # For efficiency, compute the weighted gradient directly
    # c[k] = dx * corr[k] where corr = np.correlate(x, 1-x, 'full')
    # dc[k]/dx[j] = dx * d(corr[k])/dx[j]
    # For k corresponding to lag l:
    #   if l >= 0: corr = sum_i x[i+l]*(1-x[i]), d/dx[j] = (1-x[j-l])*[j>=l] - x[j+l]*[j+l<n]
    #   if l < 0: let l'=(-l). corr = sum_i x[i]*(1-x[i+l']), d/dx[j] = (1-x[j+l'])*[j+l'<n] - x[j-l']*[j>=l']
    
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        w = weights[l + n - 1]
        if w < 1e-15:
            continue
        if l >= 0:
            # (1-x[j-l]) for j=l..n-1
            grad[l:] += w * (1.0 - x[:n-l])
            # -x[j+l] for j=0..n-1-l
            grad[:n-l] -= w * x[l:]
        else:
            lp = -l
            # (1-x[j+lp]) for j=0..n-1-lp
            grad[:n-lp] += w * (1.0 - x[lp:])
            # -x[j-lp] for j=lp..n-1
            grad[lp:] -= w * x[:n-lp]
    return dx * grad
```

Wait, this gradient computation is O(n²) per evaluation (loop over all lags). For n=200, that's 40000 operations per gradient evaluation. SLSQP calls the gradient maybe 100-1000 times, so total is 4×10^6 to 4×10^7. That should be fine.

But actually, I realize SLSQP doesn't use the gradient directly - it uses numerical differentiation by default unless you provide a Jacobian. Let me provide the gradient for the objective but not for the constraint (which is just ones).

Actually, for SLSQP, I can provide the gradient of the objective. Let me do that.

But wait - there's an issue. The soft_max_obj and its gradient involve the full correlation which is O(n²) to compute. For n=200, that's 40000 operations per function evaluation. SLSQP might do 1000+ iterations, so total is 4×10^7. In Python with numpy, this should take a few seconds.

Let me also think about whether to provide the gradient or let SLSQP compute it numerically. Numerical gradient requires n+1 function evaluations per gradient, each O(n²). So O(n³) = 8×10^6 per gradient, times 1000 iterations = 8×10^9. That's too slow!

So I MUST provide the gradient analytically.

Let me also consider using L-BFGS-B with a penalty method for the equality constraint. The penalty method adds (sum(h) - n/2)² / (2*mu) to the objective, where mu is a penalty parameter. This avoids the need for equality constraint handling.

With L-BFGS-B:
- Box constraints [0,1] are handled natively
- The sum constraint is handled via penalty
- L-BFGS-B is much faster than SLSQP for large n

def penalty_obj(x, n, alpha, mu):
    c_val = soft_max_obj(x, n, alpha)
    penalty = (x.sum() - n/2.0) ** 2 / (2.0 * mu)
    return c_val + penalty

def penalty_grad(x, n, alpha, mu):
    g = soft_max_grad(x, n, alpha)
    s = x.sum() - n/2.0
    g += s / mu * np.ones(n)
    return g

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return float(np.max(c))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _corr_vec(x, n):
    dx = 2.0 / n
    return np.correlate(x, 1-x, mode='full') * dx

def soft_max_val(x, n, alpha):
    c = _corr_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    a = sh.max()
    if a > 500:
        return m + (a + np.log(np.sum(np.exp(sh - a)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def soft_max_grad(x, n, alpha):
    c = _corr_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    a = sh.max()
    w = np.exp(sh - a)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each lag l, the weight is w[l+n-1]
    # Gradient contribution for lag l:
    # l >= 0: grad[l:] += w_l * (1-x[:n-l]); grad[:n-l] -= w_l * x[l:]
    # l < 0 (lp=-l): grad[:n-lp] += w_l * (1-x[lp:]); grad[lp:] -= w_l * x[:n-lp]
    for l in range(-(n-1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        if l >= 0:
            grad[l:] += wl * (1.0 - x[:n-l])
            grad[:n-l] -= wl * x[l:]
        else:
            lp = -l
            grad[:n-lp] += wl * (1.0 - x[lp:])
            grad[lp:] -= wl * x[:n-lp]
    return dx * grad

def opt_lbgfs(x, n, alpha, mu, maxiter=5000):
    """Optimize using L-BFGS-B with penalty for sum constraint."""
    def obj(v):
        return soft_max_val(v, n, alpha) + (v.sum() - n/2.0)**2 / (2.0*mu)
    def grad(v):
        return soft_max_grad(v, n, alpha) + (v.sum() - n/2.0) / mu
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, x, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project_feasible(res.x, n)

def local_search(h, n, deadline):
    """Targeted local search focusing on critical lags."""
    best_c = compute_c5(h, n)
    c = compute_corr(h, n)
    
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        for _pass in range(100):
            if time.time() > deadline:
                return h, best_c
            c = compute_corr(h, n)
            c_max = c.max()
            # Find critical lags (within 1% of max)
            threshold = c_max * 0.99
            critical = np.where(c >= threshold)[0]
            improved = False
            for ci in critical[:10]:  # Limit to top 10 critical lags
                if time.time() > deadline:
                    return h, best_c
                lag = ci - (n - 1)  # Convert to lag
                lamp = abs(lag)
                if lamp == 0:
                    continue
                # Try pairs (i, i+lamp)
                for i in range(n - lamp):
                    j = i + lamp
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        new_c = compute_c5(h, n)
                        if new_c < best_c - 1e-12:
                            best_c = new_c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        new_c = compute_c5(h, n)
                        if new_c < best_c - 1e-12:
                            best_c = new_c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def full_local_search(h, n, deadline):

    best_c = compute_c5(h, n)

            improved = False

                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-12:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-12:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_one(h_start, n, deadline, rng):
    """Full optimization pipeline for a given n."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Stage 1: L-BFGS-B with increasing alpha and decreasing mu
    alphas = [200, 1000, 5000, 20000, 100000]
    mus = [1000, 100, 10, 1, 0.1]
    for a, mu in zip(alphas, mus):
        if time.time() > deadline - 10:
            break
        ht = opt_lbgfs(best_h, n, a, mu, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = ht
    
    # Stage 2: Local search
    if time.time() < deadline - 5:
        if n <= 100:
            best_h, best_c = full_local_search(best_h, n, deadline - 3)
        else:
            best_h, best_c = local_search(best_h, n, deadline - 3)
    
    # Stage 3: One more optimization round
    if time.time() < deadline - 5:
        ht = opt_lbgfs(best_h, n, 500000, 0.01, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 3:

                best_h, best_c = full_local_search(best_h, n, deadline - 1)
            else:
                best_h, best_c = local_search(best_h, n, deadline - 1)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
