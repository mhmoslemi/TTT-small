        # Only use top-k correlations
        idx = np.argsort(corr)[::-1][:topk]
        c_active = corr[idx]
        m = np.max(c_active)
        e = np.exp(np.clip(beta * (c_active - m), -500, 500))
        w = e / e.sum()
        g = np.zeros(n)
        for ii, wi in zip(idx, w):
            lag = int(ii) - (n - 1)
            g += wi * grad_corr_lag(h, n, lag)
        g *= beta / len(c_active)
        val = (m + np.log(np.sum(e) / len(c_active))) / beta
    else:
        m = np.max(corr)
        e = np.exp(np.clip(beta * (corr - m), -500, 500))
        w = e / e.sum()
        g = np.zeros(n)
        for ii, wi in enumerate(w):
            if wi < 1e-15:
                continue
            lag = ii - (n - 1)
            g += wi * grad_corr_lag(h, n, lag)
        g *= beta / len(corr)
        val = (m + np.log(np.sum(e) / len(corr))) / beta
    return val, g

def slsqp_smooth(h0, n, target, beta, maxiter, t0=None, budget=0, topk=None):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        v, _ = smooth_max_grad(x, n, beta, topk)
        return v * dx

    def grad(x):
        _, g = smooth_max_grad(x, n, beta, topk)
        return g * dx

    def cfun(x):
        return x.sum() - target

    def cjac(x):
        return np.ones(n)

    try:
        r = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except Exception:
        return h0

def lag_fw(h0, n, target, n_iter, topk, t0, budget):
    """Frank-Wolfe with lag focus."""
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        corr = get_corr(best_h)
        idx = np.argsort(corr)[::-1][:topk]
        g = np.zeros(n)
        for ii in idx:
            lag = int(ii) - (n - 1)
            g += grad_corr_lag(best_h, n, lag)
        g /= len(idx)
        # FW step
        order = np.argsort(g)
        z = np.zeros(n)
        z[order[:int(target)]] = 1.0
        diff = target - z.sum()
        if diff > 1e-10:
            z[order[int(target)]] = diff
        elif diff < -1e-10:
            z[order[int(target) - 1]] = 1.0 + diff
        alpha = 2.0 / (it + 2)
        h_new = project_h((1 - alpha) * best_h + alpha * z, target)
        c = compute_c5(h_new)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_new, c
    return best_h, best_c5

def binary_sa(h0, n, target, n_iter, t0, budget):
    """Simulated annealing on binary solutions."""
    zh = (h0 > 0.5).astype(float)
    diff = target - zh.sum()
    if diff > 1e-10:
        zeros = np.where(zh == 0)[0]
        zh[zeros[np.argsort(h0[zeros])[::-1]][:int(diff)]] = 1.0
    elif diff < -1e-10:
        ones = np.where(zh == 1)[0]
        zh[ones[np.argsort(h0[ones])][:int(-diff)]] = 0.0
    zh = project_h(zh, target)
    best_h = zh.copy()
    best_c5 = compute_c5(zh)
    rng = np.random.RandomState(123)
    T = 0.002
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        T = max(0.0001, T * 0.999)
        ones = np.where(zh >= 0.99)[0]
        zeros = np.where(zh <= 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        trial = zh.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        trial = project_h(trial, target)
        c = compute_c5(trial)
        delta = c - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            zh = trial
            if c < best_c5:
                best_h, best_c5 = trial.copy(), c
    return best_h, best_c5

def pair_refine(h, n, target, sweeps, t0, budget):
    """Pair coordinate descent."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(77)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for _ in range(n):
            if time.time() - t0 > budget - 1:
                break
            i, j = rng.choice(n, 2, replace=False)
            for d in (0.03, -0.03, 0.015, -0.015, 0.008, -0.008):
                trial = best_h.copy()
                trial[i] += d
                trial[j] -= d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = trial, c
                    improved = True
                    break
        if not improved and sw > 3:
            break
    return best_h, best_c5

def single_refine(h, n, target, sweeps, t0, budget):
    """Single-coordinate descent."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            for d in (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.04, -0.04):
                trial = best_h.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = trial, c
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)
    n_pts = best_n
    target = n_pts / 2.0

    # Phase 1: Lag-focused FW
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        cur, cc = lag_fw(best_h, n_pts, target, 5000, 15, t0, budget_s)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 2: SLSQP low beta
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        cur = slsqp_smooth(best_h, n_pts, target, 200, 3000, t0, budget_s, topk=30)
        cc = compute_c5(cur)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 3: Binary SA + relax
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        cur, cc = binary_sa(best_h, n_pts, target, 5000, t0, budget_s)
        if cc < best_c5:
            best_h, best_c5 = cur, cc
        cur = slsqp_smooth(best_h, n_pts, target, 300, 3000, t0, budget_s, topk=20)
        cc = compute_c5(cur)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 4: Lag-focused FW again
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        cur, cc = lag_fw(best_h, n_pts, target, 4000, 25, t0, budget_s)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 5: SLSQP high beta
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        cur = slsqp_smooth(best_h, n_pts, target, 800, 4000, t0, budget_s, topk=15)
        cc = compute_c5(cur)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 6: Pair refine
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        cur, cc = pair_refine(best_h, n_pts, target, 30, t0, budget_s)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 7: Single coordinate
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        cur, cc = single_refine(best_h, n_pts, target, 20, t0, budget_s)
        if cc < best_c5:
            best_h, best_c5 = cur, cc

    # Phase 8: Try alternate resolution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        for n_try in [320, 448]:
            if time.time() - t0 > budget_s - 30:
                break
            nt = n_try / 2.0
            cur = np.interp(np.linspace(0, 2, n_try), np.linspace(0, 2, n_pts), best_h)
            cur = project_h(cur, nt)
            cc = compute_c5(cur)
            if cc < best_c5:
                best_h, best_c5, best_n = cur.copy(), cc, n_try
                rem = budget_s - (time.time() - t0)
                if rem > 30:
                    cur = slsqp_smooth(best_h, n_try, nt, 500, 2000, t0, budget_s, topk=20)
                    cc = compute_c5(cur)
                    if cc < best_c5:
                        best_h, best_c5, best_n = cur, cc, n_try
                rem = budget_s - (time.time() - t0)
                if rem > 20:
                    cur, cc = pair_refine(best_h, n_try, nt, 15, t0, budget_s)
                    if cc < best_c5:
                        best_h, best_c5, best_n = cur, cc, n_try

    # Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        nf = len(best_h)
        tf = nf / 2.0
        best_h = slsqp_smooth(best_h, nf, tf, 1500, 3000, t0, budget_s, topk=10)
        best_c5 = compute_c5(best_h)

    # Final pair refine
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        nf = len(best_h)
        tf = nf / 2.0
        best_h, best_c5 = pair_refine(best_h, nf, tf, 10, t0, budget_s)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
