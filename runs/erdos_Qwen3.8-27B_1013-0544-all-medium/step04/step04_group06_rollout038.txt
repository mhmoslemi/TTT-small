    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx, w

def fista(h_init, beta, n_iter, t0, budget):
    """FISTA (accelerated projected gradient) on the smoothed objective."""
    n = len(h_init)
    target = n / 2.0
    x = h_init.copy()
    y = h_init.copy()
    t_k = 1.0
    best = h_init.copy()
    best_c5 = compute_c5(best)
    # Step size: for smooth max with parameter beta, L ~ beta * (max degree) / n
    # For our problem, each variable appears in at most n lags, each contributing O(1)
    # So L ~ beta * n / n = beta (roughly). Use conservative step.
    step = 2.0 / (beta * 2.0 / n)  # = n / beta
    step = min(step, 0.5)
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        val, grad, _ = soft_smooth(y, beta)
        # Backtracking line search
        for _bt in range(30):
            x_new = project_sum(y - step * grad, target)
            val_new, _, _ = soft_smooth(x_new, beta)
            if val_new <= val + np.dot(grad, x_new - y) + 0.5 * step * np.dot(x_new - y, x_new - y):
                break
            step *= 0.5
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + (t_k - 1.0) / t_new * (x_new - y)
        # Project y back
        y = project_sum(y, target)
        t_k = t_new
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best = x_new.copy()
            best_c5 = c5
    return best, best_c5

def subgrad_polish(h, n_active, max_iter, t0, budget):
    """Accelerated subgradient method on the exact max."""
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    y = best.copy()  # FISTA momentum point
    t_k = 1.0
    no_improve = 0
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        m = np.max(corr)
        # Weighted subgradient (softmax over top constraints)
        beta_local = 50.0
        bc = np.clip(beta_local * (corr - m), -500, 0)
        w = np.exp(bc)
        w /= w.sum()
        
        g = np.zeros(n)
        nc = len(corr)
        for j in range(nc):
            if w[j] < 1e-10:
                continue
            lag = j - (n - 1)
            if lag == 0:

            elif lag > 0:
                g[lag:] += w[j] * (1.0 - y[:-lag])
                g[:n - lag] -= w[j] * y[lag:]
            else:
                m2 = -lag
                g[:n - m2] += w[j] * (1.0 - y[m2:])
                g[m2:] -= w[j] * y[:n - m2]
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Line search with backtracking
        improved = False
        step = 0.1
        for _ in range(15):
            x_new = project_sum(y - step * g, target)
            c5_new = compute_c5(x_new)
            if c5_new < best_c5 - step * gn * 0.1:  # Armijo condition (approximate)
                best = x_new
                best_c5 = c5_new
                improved = True
                break
            step *= 0.5
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + (t_k - 1.0) / t_new * (x_new - y)
        y = project_sum(y, target)
        t_k = t_new
        
        if not improved:
            no_improve += 1
            if no_improve > 500:
                # Restart momentum
                y = best.copy()
                t_k = 1.0
                no_improve = 0
            if no_improve > 5000:
                break
        else:
            no_improve = 0
    
    return best, best_c5
