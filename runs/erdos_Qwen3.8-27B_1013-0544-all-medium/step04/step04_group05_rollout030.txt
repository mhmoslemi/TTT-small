def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    x_shift = np.clip(beta * (corr - mx), -500, 500)
    w = np.exp(x_shift)
    ws = w.sum()
    w /= ws
    val = (mx + np.log(ws) / beta) * dx
    grad = np.zeros(n)
    for k in range(n + 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:-lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += wk * (1.0 - h[m2:])
            grad[m2:] -= wk * h[:n - m2]
    grad *= dx
    return val, grad

def subgrad_active(h, m_act):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[::-1][:m_act]
    grad = np.zeros(n)
    for idx in top:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += (1.0 - h[m2:])
            grad[m2:] -= h[:n - m2]
    return grad

def gradient_phase(h, target, beta_schedule, iters_per_beta, momentum, time_fn):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    h_cur = h.copy()
    v = np.zeros(n)
    base_lr = 0.05
    for beta in beta_schedule:
        if time_fn() < 10:
            break
        for it in range(iters_per_beta):
            if time_fn() < 5:
                break
            val, g = smooth_max_and_grad(h_cur, beta)
            gn = np.linalg.norm(g)
            if gn < 1e-16:
                break
            g = g / gn
            v = momentum * v - base_lr / max(1.0, gn) * g
            candidate = project_h(h_cur + v, target)
            c5_val = compute_c5(candidate)
            if c5_val < best_c5:
                best = candidate
                best_c5 = c5_val
            h_cur = candidate
            if it % 200 == 0:
                base_lr *= 0.95
    return best, best_c5

def subgrad_phase(h, target, time_fn, max_iters, m_factor, step_list, perturb_every, perturb_scale, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        g = subgrad_active(best, m_act=min(m_factor, max(3, n // 8)))
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in step_list:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and it % perturb_every == perturb_every - 1:
                p = np.random.randn(n) * perturb_scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.3
                    no_improve = 0
            if no_improve > perturb_every * 15:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0

    # Phase 1: Smooth max gradient at base resolution
    best_h, best_c5 = gradient_phase(
        best_h, target0,
        beta_schedule=[30, 100, 400, 2000, 8000, 30000],
        iters_per_beta=1500, momentum=0.75, time_fn=rem
    )

    # Phase 2: Subgradient refinement
    best_h, best_c5 = subgrad_phase(
        best_h, target0, rem,
        max_iters=30000, m_factor=15,
        step_list=[0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002],
        perturb_every=200, perturb_scale=0.006, momentum=0.85
    )

    # Phase 3: Multi-restart with perturbations
    for attempt in range(3):
        if rem() < 60:
            break
        p = np.random.randn(n0) * 0.02
        ht = project_h(best_h + p, target0)
        c5t = compute_c5(ht)
        start_h = ht if c5t < best_c5 + 0.001 else best_h
        start_c5 = min(c5t, best_c5)
        ht2, c5t2 = gradient_phase(
            start_h, target0,
            beta_schedule=[200, 1000, 5000, 20000],
            iters_per_beta=800, momentum=0.7, time_fn=rem
        )
        if rem() < 20:
            break
        ht3, c5t3 = subgrad_phase(
            ht2, target0, rem,
            max_iters=8000, m_factor=15,
            step_list=[0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001],
            perturb_every=150, perturb_scale=0.005, momentum=0.85
        )
        if c5t3 < best_c5:
            best_h = ht3
            best_c5 = c5t3

    # Phase 4: Upscale to n*2 if time allows
    if rem() > 200:
        n2 = n0 * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        best_h2, best_c5_2 = gradient_phase(
            h2, target2,
            beta_schedule=[100, 500, 3000, 15000, 50000],
            iters_per_beta=1000, momentum=0.7, time_fn=rem
        )
        if rem() > 30:
            best_h2, best_c5_2 = subgrad_phase(
                best_h2, target2, rem,
                max_iters=20000, m_factor=20,
                step_list=[0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                perturb_every=250, perturb_scale=0.004, momentum=0.87
            )
        if best_c5_2 < best_c5:
            best_h = best_h2
            best_c5 = best_c5_2

    # Phase 5: Final SLSQP polish
    nf = len(best_h)
    targetf = nf / 2.0
    dxff = 2.0 / nf
    if rem() > 15:
        bounds_f = Bounds(np.zeros(nf), np.ones(nf))
        def obj_f(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            mx = np.max(c)
            xs = np.clip(50000.0 * (c - mx), -500, 500)
            return (mx + np.log(np.sum(np.exp(xs)) / len(c)) / 50000.0) * dxff
        def con_f(h_in):
            return h_in.sum() - targetf
        try:
            res = minimize(obj_f, best_h, method='SLSQP', bounds=bounds_f,
                          constraints=[{'type': 'eq', 'fun': con_f, 'jac': lambda x: np.ones(nf)}],
                          options={'maxiter': 8000, 'ftol': 1e-20})
            h_new = project_h(res.x, targetf)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 6: Ultra-fine subgradient
    best_h, best_c5 = subgrad_phase(
        best_h, targetf, rem,
        max_iters=10000, m_factor=25,
        step_list=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
        perturb_every=400, perturb_scale=0.003, momentum=0.92
    )

    return best_h, best_c5, len(best_h)
