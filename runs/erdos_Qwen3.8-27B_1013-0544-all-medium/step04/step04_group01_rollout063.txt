import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def c5_vec(h, n):
    """Return the correlation vector scaled by dx."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_max_obj(x, n, alpha, lam):
    c = c5_vec(x, n)
    sh = alpha * c
    sh -= sh.max()
    ex = np.exp(sh)
    val = (sh.max() + np.log(ex.sum())) / alpha
    val += lam * (x.sum() - n/2.0)**2
    # Gradient
    w = ex / ex.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for tau in range(-(n-1), n):
        wt = w[tau + n - 1]
        if wt < 1e-14:
            continue
        if tau >= 0:
            grad[:n-tau] += wt * (1 - x[tau:])
            grad[tau:] -= wt * x[:n-tau]
        else:
            t = -tau
            grad[t:] += wt * (1 - x[:n-t])
            grad[:n-t] -= wt * x[t:]
    grad *= dx
    grad += 2 * lam * (x.sum() - n/2.0)
    return val, grad

def lbfgsb_optimize(h, n, alpha, lam, maxiter=500, deadline=None):
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_max_obj, h, args=(n, alpha, lam), method='L-BFGS-B',
                   jac=True, bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, n)

def fista_optimize(h, n, alpha, deadline, max_iter=20000):
    """FISTA with smooth max objective and projection onto feasible set."""
    dx = 2.0 / n
    x = h.copy()
    xt = h.copy()
    t = 1.0
    
    # Estimate Lipschitz constant
    L = 10.0
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        # Compute gradient at xt
        c = c5_vec(xt, n)
        sh = alpha * c
        sh -= sh.max()
        ex = np.exp(sh)
        w = ex / ex.sum()
        
        grad = np.zeros(n)
        for tau in range(-(n-1), n):
            wt = w[tau + n - 1]
            if wt < 1e-14:
                continue
            if tau >= 0:
                grad[:n-tau] += wt * (1 - xt[tau:])
                grad[tau:] -= wt * xt[:n-tau]
            else:
                t2 = -tau
                grad[t2:] += wt * (1 - xt[:n-t2])
                grad[:n-t2] -= wt * xt[t2:]
        grad *= dx
        
        # FISTA update
        y = xt + ((t - 1) / (t + 1)) * (xt - x)
        x_new = y - grad / L
        
        # Project onto {0<=h<=1, sum(h)=n/2}
        x_new = project_feasible(x_new, n)
        
        # Update
        x, xt = xt, x_new
        t = (1 + np.sqrt(1 + 4*t*t)) / 2
    
    return project_feasible(xt, n)

For a pair (i, j), the feasible range of d is:

- h[j] - d ∈ [0, 1] → d ∈ [h[j]-1, h[j]]

where ΔC_k(d) is the change in C_k due to the move.

Since ΔC_k(d) is at most linear in d (ignoring the rare quadratic case), the new max is a piecewise-linear (or piecewise-quadratic) convex function of d. Its minimum can be found by bisection or by checking the breakpoints.

For practical purposes, I can just do a few evaluations (e.g., at d=0.001, 0.005, 0.01, 0.05 and the boundaries) and pick the best.

def best_step_size(h, n, i, j, c_vec, dx):
    """Find optimal step d for move h[i]+=d, h[j]-=d using the current c_vec."""
    # Feasible range
    d_lo = max(-h[i], h[j] - 1)
    d_hi = min(1 - h[i], h[j])
    if d_lo >= d_hi:
        return 0.0, c_vec.max()
    
    # Compute ΔC_k for active lags
    # For efficiency, only consider lags near the max
    max_c = c_vec.max()
    tol = 0.01  # consider lags within 0.01 of max
    
    # Find candidate d values: 0, boundaries, and where active lags cross
    candidates = [d_lo, d_hi, 0.0]
    
    # For each active lag, find where C_k + ΔC_k * d = max (i.e., where it crosses)
    for k_idx in range(len(c_vec)):
        if c_vec[k_idx] < max_c - tol:
            continue
        tau = k_idx - (n - 1)
        # Compute ΔC_k/d (the slope)
        slope = 0.0
        ki = i + tau
        kj = j + tau
        ik = i - tau
        jk = j - tau
        
        if 0 <= ki < n and ki != j:
            slope += (1 - h[ki])
        elif ki == j:
            slope += (1 - h[j]) + h[i]  # linear part (ignoring d²)
        
        if 0 <= kj < n and kj != i:
            slope -= (1 - h[kj])
        elif kj == i:
            slope -= (1 - h[i]) + h[j]
        
        if 0 <= ik < n and ik != i and ik != j:
            slope += h[ik]
        
        if 0 <= jk < n and jk != i and jk != j:
            slope -= h[jk]
        
        # C_k + slope * d = other_max → d = (other_max - C_k) / slope
        # But we don't know other_max, so just add the crossing point
        if abs(slope) > 1e-10:
            # The lag k is "decreasing" if slope < 0, "increasing" if slope > 0
            # We want to find d where it crosses the current max
            d_cross = (max_c - c_vec[k_idx]) / (slope * dx)
            if d_lo < d_cross < d_hi:
                candidates.append(d_cross)
    
    # Evaluate at candidates
    best_d = 0.0
    best_c = c_vec.max()
    for d in candidates:
        d = max(d_lo, min(d_hi, d))
        # Compute new max (approximate)
        new_c = c_vec.copy()

            kj = j + tau
            ik = i - tau
            jk = j - tau

                if ki == j:
                    delta += (h[i]+d)*(1-h[j]+d) - h[i]*(1-h[j])
                else:
                    delta += d * (1 - h[ki])
            
            if 0 <= kj < n:
                if kj == i:
                    delta += (h[j]-d)*(1-h[i]+d) - h[j]*(1-h[i])
                else:
                    delta += -d * (1 - h[kj])
            
            if 0 <= ik < n and ik != i and ik != j:
                delta += d * h[ik]
            
            if 0 <= jk < n and jk != i and jk != j:
                delta += -d * h[jk]
            
            new_c[k_idx] += delta * dx
        
        c_max = new_c.max()
        if c_max < best_c:
            best_c = c_max
            best_d = d
    
    return best_d, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def _c_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def _smooth_fg(h, n, alpha):
    c = _c_vec(h, n)
    sh = alpha * c
    sh_max = sh.max()
    ex = np.exp(sh - sh_max)
    val = (sh_max + np.log(ex.sum())) / alpha
    w = ex / ex.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for tau in range(-(n - 1), n):
        wt = w[tau + n - 1]
        if wt < 1e-13:
            continue
        if tau >= 0:
            grad[:n - tau] += wt * (1.0 - h[tau:])
            grad[tau:] -= wt * h[:n - tau]
        else:
            t = -tau
            grad[t:] += wt * (1.0 - h[:n - t])
            grad[:n - t] -= wt * h[t:]
    grad *= dx
    return val, grad

def _sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

        lr = 1.0 / (alpha * 4.0)
    for it in range(50000):
        if time.time() > deadline:
            break
        c = _c_vec(xt, n)
        sh = alpha * c
        sh_max = sh.max()
        ex = np.exp(sh - sh_max)
        w = ex / ex.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for tau in range(-(n - 1), n):
            wt = w[tau + n - 1]
            if wt < 1e-13:
                continue
            if tau >= 0:
                grad[:n - tau] += wt * (1.0 - xt[tau:])
                grad[tau:] -= wt * xt[:n - tau]
            else:
                tt = -tau
                grad[tt:] += wt * (1.0 - xt[:n - tt])
                grad[:n - tt] -= wt * xt[tt:]
        grad *= dx
        y = xt + ((t - 1) / (t + 1)) * (xt - x)
        x_new = y - lr * grad
        x_new = project_feasible(x_new, n)
        c_new = compute_c5(x_new, n)
        c_old = compute_c5(x, n)
        if c_new > c_old * 1.01:
            lr *= 0.5
            x_new = project_feasible(y - lr * grad, n)
        x, xt = xt, x_new
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2
    return project_feasible(xt, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(200, 0.5)
        n = 200

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with analytic gradient (main optimization)
    for alpha in [2000, 10000, 50000, 200000, 500000]:
        if time.time() > deadline - 200:
            break
        ht = _slsqp(best_h, n, alpha, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 2: L-BFGS-B (fast, penalty for constraint)
    if time.time() < deadline - 120:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 60:
                break
            ht = _lbfgs(best_h, n, alpha, lam=1000.0, maxiter=3000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

    # Phase 3: Guided pair search
    if time.time() < deadline - 100:
        best_h, best_c = _pair_search(best_h, n, deadline - 80, full=False, radius=40)

    # Phase 4: High-alpha SLSQP polish
    if time.time() < deadline - 40:
        for alpha in [1000000, 5000000]:
            if time.time() > deadline - 20:
                break
            ht = _slsqp(best_h, n, alpha, maxiter=1000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

    # Phase 5: FISTA for additional refinement
    if time.time() < deadline - 30:
        ht = _fista(best_h, n, 200000, deadline - 15)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 6: Full pair search (all pairs) for final polish
    if time.time() < deadline - 10:
        best_h, best_c = _pair_search(best_h, n, deadline - 5, full=True, radius=3)

    # Phase 7: Tiny-step local search
    if time.time() < deadline - 2:
        best_h, best_c = _pair_search(best_h, n, deadline - 0.5, full=False, radius=5)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
