from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

                              bounds=bounds, constraints=cons,
                              options={'maxiter': 10000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h, c5 = h_new, c5_new
            except Exception:
                pass
    return h, c5

def minimax_refine(h, n, deadline, seed, lr_w_init=0.08):
    np.random.seed(seed)
    ts = n / 2.0
    N_lags = 2 * n - 1
    c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(3.0 * (corr - corr.max()))
    w /= w.sum()
    lr_h = 0.004
    lr_w = lr_w_init
    velocity = np.zeros(n)
    best_c5 = c5
    best_h = h.copy()
    stagnant = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = w * np.exp(lr_w * (corr - corr.max()))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(N_lags) / N_lags
        else:
            w /= ws
        grad = np.zeros(n)
        active = np.where(w > 1e-9)[0]
        for li in active:
            wi = w[li]
            if wi < 1e-7:
                continue
            lag = li - (n - 1)
            grad += wi * subgrad_for_lag(h, lag, n)
        velocity = 0.84 * velocity - lr_h * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-13:
            h, c5 = h_new, c_new
            stagnant = 0
            lr_h = min(lr_h * 1.008, 0.01)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            stagnant += 1
            velocity *= 0.35
            lr_h *= 0.94
            if stagnant % 400 == 0:
                h = project_feasible(best_h + np.random.randn(n) * 0.004, ts)
                c5 = compute_c5(h)
                corr = np.correlate(h, 1.0 - h, mode='full')
                w = np.exp(3.0 * (corr - corr.max()))
                w /= w.sum()
        if stagnant > 1000 and lr_h < 3e-7:
            break
    if best_c5 < c5:
        h, c5 = best_h, best_c5
    return h, c5

def smooth_perturb(h, n, sigma):
    kernel_len = max(3, 2 * int(3 * n / 200) + 1)
    x = np.arange(kernel_len) - kernel_len // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    h_smooth = np.convolve(h, kernel, mode='same')
    return h_smooth

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h, best_c5, best_n = None, float('inf'), 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: Optimize parent (50% budget)
    d1 = t0 + budget_s * 0.50
    h1 = parent.copy()
    c1 = compute_c5(h1)
    h1, c1 = slsqp_optimize(h1, pn, min(t0 + budget_s * 0.22, d1), seed,
                            [50, 200, 1000, 5000])
    h1, c1 = minimax_refine(h1, pn, d1, seed + 7, lr_w_init=0.06)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Phase 2: Other resolutions (30% budget)
    d2 = t0 + budget_s * 0.82
    resolutions = [250, 300, 400, 450, 500]
    for n in resolutions:
        if time.time() > d2:
            break
        if n == pn:
            continue
        x_old = np.linspace(0, 2, pn, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h0, n / 2.0)
        subt = min(50.0, d2 - time.time())
        if subt < 10:
            continue
        h0, cn = slsqp_optimize(h0, n, time.time() + subt * 0.4, seed + n,
                                [200, 2000, 20000])
        h0, cn = minimax_refine(h0, n, time.time() + subt * 0.55, seed + n + 500)
        if cn < best_c5:
            best_h, best_c5, best_n = h0.copy(), cn, n

    # Phase 3: Final SLSQP polish (15% budget)
    d3 = t0 + budget_s * 0.96
    if best_h is not None and time.time() < d3:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
        for beta in [30000, 100000, 500000]:
            if time.time() > d3:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 15000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    # Phase 4: Final minimax micro-polish
    if best_h is not None:
        best_h, best_c5 = minimax_refine(best_h, best_n, t0 + budget_s * 0.99,
                                         seed + 9999, lr_w_init=0.15)

    if best_h is None:
        best_h, best_n = np.full(100, 0.5), 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
