import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_top_lags(h, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:K]
    return idx - (n - 1)

def subgrad_dir(h, lags, n):
    g = np.zeros(n)
    for L in lags:
        if L == 0:
            g += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:-L]
            g[:n-L] -= h[L:]
        else:
            m = -L
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    return g / len(lags)

def coord_opt(h, i, lags, n):
    lin_f = []
    quad = None
    for L in lags:
        if L == 0:
            rest = np.dot(h, 1.0 - h) - h[i] * (1.0 - h[i])
            quad = (-1.0, 1.0, rest)
        elif L > 0:
            a = 0.0
            b = 0.0
            if i >= L:
                a += 1.0 - h[i-L]
            if i + L < n:
                a -= h[i+L]
                b += h[i+L]
            full = np.dot(h[L:], 1.0 - h[:n-L])
            if i >= L:
                full -= h[i] * (1.0 - h[i-L])
            if i + L < n:
                full -= h[i+L] * (1.0 - h[i])
            lin_f.append((a, b + full))
        else:
            m = -L
            a = 0.0
            b = 0.0
            if i + m < n:
                a += 1.0 - h[i+m]
            if i >= m:
                a -= h[i-m]
                b += h[i-m]
            full = np.dot(h[:n-m], 1.0 - h[m:])
            if i + m < n:
                full -= h[i] * (1.0 - h[i+m])
            if i >= m:
                full -= h[i-m] * (1.0 - h[i])
            lin_f.append((a, b + full))
    
    cands = [0.0, 1.0]
    for p in range(len(lin_f)):
        for q in range(p+1, len(lin_f)):
            a1, b1 = lin_f[p]
            a2, b2 = lin_f[q]
            if abs(a1 - a2) > 1e-14:
                tc = (b2 - b1) / (a1 - a2)
                if -0.01 <= tc <= 1.01:
                    cands.append(tc)
    if quad is not None:
        aq, bq, cq = quad
        vtx = -bq / (2.0 * aq)
        if 0 <= vtx <= 1:
            cands.append(vtx)
        for al, bl in lin_f:
            A = aq
            B = bq - al
            C = cq - bl
            disc = B*B - 4.0*A*C
            if disc >= 0:
                sq = np.sqrt(disc)
                for t in [(-B+sq)/(2.0*A), (-B-sq)/(2.0*A)]:
                    if -0.01 <= t <= 1.01:
                        cands.append(t)
    
    best_v = np.inf
    best_t = h[i]
    for t in cands:
        t = max(0.0, min(1.0, t))
        mx = -1e30
        if quad is not None:
            mx = max(mx, quad[0]*t*t + quad[1]*t + quad[2])
        for al, bl in lin_f:
            v = al * t + bl
            if v > mx:
                mx = v
        if mx < best_v:
            best_v = mx
            best_t = t
    return best_t

def cd_sweep(h, lags, n, target):
    h = h.copy()
    for i in range(n):
        h[i] = coord_opt(h, i, lags, n)
    return project_sum(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(480, 0.5)
    
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    
    def elapsed():
        return time.time() - t0
    
    def remaining():
        return budget_s - elapsed()
    
    # Phase 1: Coordinate descent warmup
    for K in [8, 12, 16]:
        if remaining() < 120:
            break
        lags = get_top_lags(best_h, K)
        for swe in range(100):
            if remaining() < 120:
                break
            h_new = cd_sweep(best_h, lags, n, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            # Re-identify lags every 10 sweeps
            if swe % 10 == 9:
                lags = get_top_lags(best_h, K)
    
    # Phase 2: Subgradient with momentum
    if remaining() > 200:
        x = best_h.copy()
        v = np.zeros(n)
        K2 = 10
        for it in range(5000):
            if remaining() < 200:
                break
            lags = get_top_lags(x, K2)
            g = subgrad_dir(x, lags, n)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g /= gn
            v = 0.7 * v + 0.3 * g
            step = 0.05 / (1.0 + 0.05 * it)
            x_new = project_sum(x - step * v, target)
            c5_new = compute_c5(x_new)
            if c5_new < best_c5:
                best_h = x_new.copy()
                best_c5 = c5_new
                x = x_new
            elif it % 200 == 199:
                # Reset if stuck
                x = best_h.copy()
                v = np.zeros(n)
    
    # Phase 3: More CD with periodic subgrad
    if remaining() > 150:
        K3 = 15
        for sweep_num in range(500):
            if remaining() < 150:
                break
            lags = get_top_lags(best_h, K3)
            h_new = cd_sweep(best_h, lags, n, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            # Every 30 sweeps, do a subgradient step
            if sweep_num % 30 == 29:
                g = subgrad_dir(best_h, lags, n)
                gn = np.linalg.norm(g)
                if gn > 1e-14:
                    g /= gn
                    for step in [0.02, 0.01, 0.005, 0.002]:
                        h_try = project_sum(best_h - step * g, target)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5:
                            best_h = h_try
                            best_c5 = c5_try
                            break
    
    # Phase 4: SLSQP refinement
    if remaining() > 100:
        dx = 2.0 / n
        beta = 5000
        lags = get_top_lags(best_h, 8)
        
        def slsqp_obj(x):
            xc = np.clip(x, 0, 1)
            corr = np.correlate(xc, 1.0 - xc, mode="full")
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(np.clip(beta*(corr-m), -500, 500))))/beta) * dx
        
        def slsqp_grad(x):
            xc = np.clip(x, 0, 1)
            corr = np.correlate(xc, 1.0 - xc, mode="full")
            m = np.max(corr)
            bc = np.clip(beta*(corr-m), -500, 500)
            e = np.exp(bc)
            w = e / e.sum()
            g = np.zeros(n)
            for j in range(len(corr)):
                if w[j] < 1e-12:
                    continue
                lag = j - (n-1)
                if lag == 0:
                    g += w[j] * (1.0 - 2.0*xc)
                elif lag > 0:
                    g[lag:] += w[j] * (1.0 - xc[:-lag])
                    g[:n-lag] -= w[j] * xc[lag:]
                else:
                    m2 = -lag
                    g[:n-m2] += w[j] * (1.0 - xc[m2:])
                    g[m2:] -= w[j] * xc[:n-m2]
            return g * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(slsqp_obj, best_h, jac=slsqp_grad, method='SLSQP',
                       bounds=[(0,1)]*n, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-18})
        h_new = project_sum(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 5: Final CD polish
    if remaining() > 20:
        K5 = 20
        for swe in range(200):
            if remaining() < 20:
                break
            lags = get_top_lags(best_h, K5)
            h_new = cd_sweep(best_h, lags, n, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    return best_h, best_c5, n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_top_lags(h, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:K]
    return idx - (n - 1)

def subgrad_dir(h, lags, n):
    g = np.zeros(n)
    for L in lags:
        if L == 0:
            g += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:-L]
            g[:n-L] -= h[L:]
        else:
            m = -L
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    return g / len(lags)

def coord_opt(h, i, lags, n):
    lin_f = []
    quad = None
    for L in lags:
        if L == 0:
            rest = np.dot(h, 1.0 - h) - h[i] * (1.0 - h[i])
            quad = (-1.0, 1.0, rest)
        elif L > 0:
            a = 0.0
            b = 0.0
            if i >= L:
                a += 1.0 - h[i - L]
            if i + L < n:
                a -= h[i + L]
                b += h[i + L]
            full = np.dot(h[L:], 1.0 - h[:n - L])
            if i >= L:
                full -= h[i] * (1.0 - h[i - L])
            if i + L < n:
                full -= h[i + L] * (1.0 - h[i])
            lin_f.append((a, b + full))
        else:
            m = -L
            a = 0.0
            b = 0.0
            if i + m < n:
                a += 1.0 - h[i + m]
            if i >= m:
                a -= h[i - m]
                b += h[i - m]
            full = np.dot(h[:n - m], 1.0 - h[m:])
            if i + m < n:
                full -= h[i] * (1.0 - h[i + m])
            if i >= m:
                full -= h[i - m] * (1.0 - h[i])
            lin_f.append((a, b + full))
    cands = [0.0, 1.0]
    for p in range(len(lin_f)):
        for q in range(p + 1, len(lin_f)):
            a1, b1 = lin_f[p]
            a2, b2 = lin_f[q]
            if abs(a1 - a2) > 1e-14:
                tc = (b2 - b1) / (a1 - a2)
                if -0.01 <= tc <= 1.01:
                    cands.append(tc)
    if quad is not None:
        aq, bq, cq = quad
        vtx = -bq / (2.0 * aq)
        if 0.0 <= vtx <= 1.0:
            cands.append(vtx)
        for al, bl in lin_f:
            A = aq
            B = bq - al
            C = cq - bl
            disc = B * B - 4.0 * A * C
            if disc >= 0:
                sq = np.sqrt(disc)
                t1 = (-B + sq) / (2.0 * A)
                t2 = (-B - sq) / (2.0 * A)
                if -0.01 <= t1 <= 1.01:
                    cands.append(t1)
                if -0.01 <= t2 <= 1.01:
                    cands.append(t2)
    best_v = 1e30
    best_t = h[i]
    for t in cands:
        if t < 0.0:
            t = 0.0
        elif t > 1.0:
            t = 1.0
        mx = -1e30
        if quad is not None:
            vq = quad[0] * t * t + quad[1] * t + quad[2]
            if vq > mx:
                mx = vq
        for al, bl in lin_f:
            v = al * t + bl
            if v > mx:
                mx = v
        if mx < best_v:
            best_v = mx
            best_t = t
    return best_t

def cd_sweep(h, lags, n, target, order):
    h = h.copy()
    for i in order:
        h[i] = coord_opt(h, i, lags, n)
    return project_sum(h, target)

def cd_phase(h, target, n, K, n_sweeps, t0, budget, reserve):
    best = h.copy()
    best_c5 = compute_c5(best)
    order = np.arange(n)
    for sw in range(n_sweeps):
        if budget - (time.time() - t0) < reserve:
            break
        if sw % 20 == 0:
            np.random.shuffle(order)
            lags = get_top_lags(best, K)
        h_new = cd_sweep(best, lags, n, target, order)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best = h_new
            best_c5 = c5_new
    return best, best_c5

def subgrad_momentum(h, target, n, K, n_iters, t0, budget, reserve):
    best = h.copy()
    best_c5 = compute_c5(best)
    x = best.copy()
    v = np.zeros(n)
    stuck = 0
    for it in range(n_iters):
        if budget - (time.time() - t0) < reserve:
            break
        lags = get_top_lags(x, K)
        g = subgrad_dir(x, lags, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        v = 0.65 * v + 0.35 * g
        step = 0.08 / (1.0 + 0.03 * it)
        x_new = project_sum(x - step * v, target)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5 - 1e-15:
            best = x_new.copy()
            best_c5 = c5_new
            x = x_new
            stuck = 0
        elif c5_new < compute_c5(x) - 1e-15:
            x = x_new
        else:
            stuck += 1
            if stuck > 300:
                x = best.copy()
                v = np.zeros(n)
                stuck = 0
    return best, best_c5

def slsqp_polish(h, target, n, beta, maxiter, t0, budget, reserve):
    if budget - (time.time() - t0) < reserve:
        return h, compute_c5(h)
    dx = 2.0 / n

    def obj(x):
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode="full")
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500)))) / beta) * dx

    def grad(x):
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        w = e / e.sum()
        g = np.zeros(n)
        for j in range(len(corr)):
            if w[j] < 1e-12:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * xc)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - xc[:-lag])
                g[:n - lag] -= w[j] * xc[lag:]
            else:
                m2 = -lag
                g[:n - m2] += w[j] * (1.0 - xc[m2:])
                g[m2:] -= w[j] * xc[:n - m2]
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, jac=grad, method='SLSQP',
                   bounds=[(0, 1)] * n, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    h_new = project_sum(res.x, target)
    return h_new, compute_c5(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(480, 0.5)

    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: CD with increasing K
    for K in [6, 10, 14]:
        if rem() < 200:
            break
        best_h, best_c5 = cd_phase(best_h, target, n, K, 800, t0, budget_s, 200)

    # Phase 2: Subgradient with momentum
    if rem() > 150:
        best_h, best_c5 = subgrad_momentum(best_h, target, n, 10, 10000, t0, budget_s, 150)

    # Phase 3: CD with larger K
    if rem() > 150:
        for K in [16, 22]:
            if rem() < 150:
                break
            best_h, best_c5 = cd_phase(best_h, target, n, K, 1500, t0, budget_s, 150)

    # Phase 4: Multi-start with perturbations
    if rem() > 200:
        for restart in range(4):
            if rem() < 200:
                break
            if restart == 0:
                h_try = best_h.copy()
            elif restart == 1:
                h_try = project_sum(best_h + np.random.randn(n) * 0.03, target)
            elif restart == 2:
                h_try = project_sum(best_h + np.random.randn(n) * 0.08, target)
            else:
                blk = n // 4
                h_try = best_h.copy()
                h_try[:blk] = 1.0 - h_try[:blk]
                h_try = project_sum(h_try, target)
            h_try, c5_try = cd_phase(h_try, target, n, 15, 600, t0, budget_s, 200)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try

    # Phase 5: SLSQP refinement
    if rem() > 80:
        for beta in [2000, 10000]:
            if rem() < 80:
                break
            h_new, c5_new = slsqp_polish(best_h, target, n, beta, 3000, t0, budget_s, 80)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Final intensive CD polishing
    if rem() > 30:
        for K in [20, 30]:
            if rem() < 30:
                break
            best_h, best_c5 = cd_phase(best_h, target, n, K, 2000, t0, budget_s, 30)

    # Phase 7: Quick subgradient micro-polish
    if rem() > 15:
        best_h, best_c5 = subgrad_momentum(best_h, target, n, 8, 3000, t0, budget_s, 10)

    # Final: one more CD sweep for absolute best
    if rem() > 5:
        lags = get_top_lags(best_h, 15)
        h_new = cd_sweep(best_h, lags, n, target, np.random.permutation(n))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    return best_h, best_c5, n
