def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += diff / tot * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += diff / tot * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

    return x.sum() - n / 2.0

    return np.ones(n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)

        return h.copy()

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += diff / tot * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += diff / tot * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def obj_lse_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    e = np.exp(s - mx)
    se = e.sum()
    val = m + (mx + np.log(se)) / alpha
    w = e / se  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k] * dx
        if wk < 1e-13:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            if n - sh > 0:
                grad[sh:n] += wk * (1.0 - x[:n-sh])
                grad[:n-sh] -= wk * x[sh:]
        else:
            t = -sh
            if n - t > 0:
                grad[:n-t] += wk * (1.0 - x[t:])
                grad[t:] -= wk * x[:n-t]
    
    return float(val), grad

def _sc(x, n):
    return x.sum() - n / 2.0

def _sj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _sj, 'args': (n,)}]
    try:
        res = minimize(obj_lse_grad, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(300):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    hi, hj = h[i]+d, h[j]-d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi-d, hj+d
                    hi2, hj2 = h[i]-d, h[j]+d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e
