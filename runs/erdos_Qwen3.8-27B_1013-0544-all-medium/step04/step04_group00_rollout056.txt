        val[:n-kk] += hb[kk:]    # hb[p-k] = hb[p+kk] for p in [0, n-kk-1]

    zeros_set = set(np.where(hb == 0)[0].tolist())
    ones_list = sorted(ones_set)
    zeros_list = sorted(zeros_set)

    max_stall = 200

        improved = False

        top_lags = np.argsort(-c)[:min(3, len(c))]

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
