The key insight of exponential weights: instead of minimizing the max, we minimize a weighted sum where the weights are updated to focus on the currently worst lags. This is a well-known technique in minimax optimization and online learning.

Algorithm:
```
Initialize h, weights w_k = 1/(2n-1) for all k
For iteration t = 1, 2, ...:
    Compute overlaps c_k for all k
    Update weights: w_k ← exp(c_k / τ) / Σ exp(c_j / τ)
    Compute gradient of Σ w_k * c_k w.r.t. h
    Take a projected gradient step
    Project to feasible set (0 ≤ h ≤ 1, sum h = n/2)
```

The temperature τ controls how sharply we focus on the worst lags. Small τ = focus on the single worst lag. Large τ = uniform weighting.

For the implementation, I need to:
1. Compute all 2n-1 overlaps (O(n²) with direct correlation, or O(n log n) with FFT)
2. Compute the weighted gradient
3. Take a gradient step with line search

Let me vectorize the gradient computation. The gradient of overlap at lag k (where k is the actual shift, can be negative) with respect to h is:

For shift s ≥ 0 (lag index = s + n - 1):
∂c_s / ∂h[i] = (1 - h[i+s]) for i = 0,...,n-1-s
∂c_s / ∂h[i] = -h[i-s] for i = s,...,n-1

The total weighted gradient at position i is:
Σ_{s: 0≤s<i<n} w_{s+n-1} * (1-h[i+s])  [where i is the "source" for shift s]
+ Σ_{s: 0≤s<n-i} w_{s+n-1} * (-h[i-s])  [where i is the "target" for shift s]
+ similar terms for negative shifts

Let W_neg[s] = w_{n-1-s} for s = 0, ..., n-1 (weights for negative shifts)

Then:
grad[i] = (convolution of h with W_pos)(i) + (1 - (convolution of h with W_pos reversed))(i)

            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def weighted_gradient_fast(h, weights, n):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        elif -n < lag < 0:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    return grad

def exp_weights_optimize(h0, n, target, t0, budget, n_iter=5000, tau_init=0.05, momentum=0.7):
    h = h0.copy()
    tau = tau_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.04
    mom = np.zeros(n)
    stagnation = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        lags = compute_all_lags(h)
        lamax = np.max(lags)
        tau = max(tau * 0.9995, 0.008)
        weights = np.exp((lags - lamax) / tau)
        weights /= weights.sum()
        grad = weighted_gradient_fast(h, weights, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            s = step * factor
            mom = momentum * mom - s * grad
            trial = project_h(h + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-14:
                h = trial
                best_h = h.copy()
                best_c5 = c_try
                step = min(s * 1.08, 0.1)
                stagnation = 0
                accepted = True
                break
        if not accepted:
            step *= 0.65
            mom *= 0.3
            trial = project_h(h + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-14:
                h = trial
                best_h = h.copy()
                best_c5 = c_try
                stagnation = 0
            else:
                mom = np.zeros(n)
                stagnation += 1
        if stagnation > 300:
            break
    return best_h, best_c5

def binary_anneal(h_bin, n, t0, budget, n_iter=40000):
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        if len(ones_cache) == 0 or len(zeros_cache) == 0:
            break
        a = ones_cache[rng.integers(len(ones_cache))]
        b = zeros_cache[rng.integers(len(zeros_cache))]
        h[a], h[b] = 0.0, 1.0
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            frac = max(it, 1) / max(n_iter, 1)
            temp = 0.15 * (0.005) ** frac
            if rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-15)):
                cur_c5 = new_c5
                ones_cache = np.where(h == 1)[0]
                zeros_cache = np.where(h == 0)[0]
            else:
                h[a], h[b] = 1.0, 0.0
    return best_h, best_c5

def smooth_max_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(c)) / beta) * (2.0 / n)

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def slsqp_refine(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.004, -0.004, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 384
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = project_h(h, n / 2.0)

    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Binary anneal from parent
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        h_bin = binary_snap(h, n)
        h_bin, bin_c5 = binary_anneal(h_bin, n, t0, budget_s, n_iter=35000)
        h_cont = h_bin + 0.04 * np.random.randn(n)
        h_cont = project_h(h_cont, target)
        c_try = compute_c5(h_cont)
        if c_try < best_c5:
            best_h = h_cont.copy()
            best_c5 = c_try

    # Phase 2: Exponential weights from binary start
    remaining = budget_s - (time.time() - t0)
    if remaining > 120:
        h_bin = binary_snap(best_h, n)
        h_start = h_bin + 0.03 * np.random.randn(n)
        h_start = project_h(h_start, target)
        h_e, c5_e = exp_weights_optimize(h_start, n, target, t0, budget_s, n_iter=6000, tau_init=0.05, momentum=0.75)
        if c5_e < best_c5:
            best_h = h_e.copy()
            best_c5 = c5_e

    # Phase 3: Exponential weights from parent
    remaining = budget_s - (time.time() - t0)
    if remaining > 100:
        h_p = project_h(h.copy(), target)
        h_p, c5_p = exp_weights_optimize(h_p, n, target, t0, budget_s, n_iter=4000, tau_init=0.04, momentum=0.7)
        if c5_p < best_c5:
            best_h = h_p.copy()
            best_c5 = c5_p

    # Phase 4: SLSQP refinement
    remaining = budget_s - (time.time() - t0)
    for beta in [300, 1000, 5000, 20000]:
        if budget_s - (time.time() - t0) < 10:
            break
        best_h = slsqp_refine(best_h, n, target, beta, maxiter=6000)
        best_c5 = compute_c5(best_h)

    # Phase 5: Another exp-weights round with lower tau
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        best_h, best_c5 = exp_weights_optimize(best_h, n, target, t0, budget_s, n_iter=4000, tau_init=0.025, momentum=0.8)

    # Phase 6: SLSQP with very high beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h = slsqp_refine(best_h, n, target, 60000, maxiter=10000)
        best_c5 = compute_c5(best_h)

    # Phase 7: Coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        best_h, best_c5 = coord_descent(best_h, target, n, 300, t0, budget_s)

    # Phase 8: Perturbation restarts
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 77)
    for att in range(8):
        if budget_s - (time.time() - t0) < 20:
            break
        scale = 0.02 * (1.0 - att * 0.08)
        hp = best_h + rng.standard_normal(n) * max(scale, 0.005)
        hp = project_h(hp, target)
        hp, c_hp = exp_weights_optimize(hp, n, target, t0, budget_s, n_iter=1500, tau_init=0.03, momentum=0.75)
        if c_hp < best_c5:
            best_h = hp.copy()
            best_c5 = c_hp
            best_h = slsqp_refine(best_h, n, target, 10000, maxiter=4000)
            best_c5 = compute_c5(best_h)

    # Phase 9: Final coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = coord_descent(best_h, target, n, 400, t0, budget_s)

    # Phase 10: Ultra-fine SLSQP
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        best_h = slsqp_refine(best_h, n, target, 100000, maxiter=12000)
        best_c5 = compute_c5(best_h)

    # Final check
    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        best_h, best_c5 = coord_descent(best_h, target, n, 200, t0, budget_s)

    return best_h, best_c5, n
