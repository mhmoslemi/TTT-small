def subgrad_topk(h, n, num_lags=30):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.005 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > num_lags:
        idx = idx[np.argsort(corr[idx])[::-1][:num_lags]]
    g = np.zeros(n)

        best_h, best_c5 = pair_local(best_h, nf, tf, 20000, t0, budget_s, seed=seed+200)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
