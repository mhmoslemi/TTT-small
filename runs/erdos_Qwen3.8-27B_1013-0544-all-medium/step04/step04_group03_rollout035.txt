from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_grad(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    exp_s = np.exp(x_shift)
    weights = exp_s / np.sum(exp_s)
    return weights  # gradient of smooth_max w.r.t. x

def slsqp_obj(h_in, beta, n):

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-15):

def nesterov_pg(h, target, n, n_iter, beta, t0, budget, step_init=0.01):
    """Nesterov accelerated projected gradient on smooth-max."""
    cur = h.copy()
    y = h.copy()
    best_local = compute_c5(cur)
    best_h = cur.copy()
    gamma = 1.0  # momentum parameter (Nesterov)
    step = step_init
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        
        # Compute gradient of smooth-max objective
        corr = np.correlate(y, 1.0 - y, mode="full")
        sm_val = smooth_max(corr, beta) * (2.0 / n)
        
        # Gradient of smooth_max wrt corr
        m = np.max(corr)
        x_shift = np.clip(beta * (corr - m), -500, 500)
        exp_s = np.exp(x_shift)
        weights = exp_s / np.sum(exp_s)
        
        # Gradient of corr wrt h
        # corr[k] = sum_j y[j] * (1-y[j-(k-(n-1))])
        # This is complex. Let's use the top-k approach for efficiency.
        
        # Actually, let's compute gradient directly for the weighted lags
        # grad = sum_k weights[k] * d(corr[k])/d(y)
        # For efficiency, only compute for top-weighted lags
        
        top_k = 32
        top_idx = np.argsort(weights)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:

            w = weights[idx] * (2.0 / n)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += w * (1.0 - y[lag:])
                    grad[lag:] -= w * y[:n - lag]
            else:
                la = -lag
                grad[la:] += w * (1.0 - y[:n - la])
                grad[:n - la] -= w * y[la:]
        
        # Nesterov update
        t_val = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        y_new = cur + (gamma / t_val) * (y - cur) - step * grad
        y_new = project_h(y_new, target)
        
        cur = y_new
        gamma = t_val
        
        c_try = compute_c5(cur)
        if c_try < best_local:
            best_local = c_try
            best_h = cur.copy()
        
        # Decay step size slowly
        step = step_init / (1.0 + 0.01 * it)
    
    return best_h, best_local

        grad /= top_k

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def grad_smooth_max(h, n, beta, top_k=64):
    """Compute gradient of smooth_max(correlate(h,1-h)) * 2/n."""
    corr = np.correlate(h, 1.0 - h, mode="full")

    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_s = np.exp(x_shift)
    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    idx = np.argsort(weights)[::-1][:top_k]

                grad[lag:] -= w * h[:n - lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    return grad

    y = h.copy()

    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break

                cur_c5 = trial_c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def grad_smooth_max(h, n, beta, top_k=128):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_s = np.exp(x_shift)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    idx = np.argsort(weights)[::-1][:top_k]
    for i in idx:
        lag = int(i) - (n - 1)
        w = weights[i] * (2.0 / n)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:n - lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    return grad

def nesterov_pg(h, target, n, n_iter, beta, t0, budget, step_init=0.004, top_k=96):
    cur = h.copy()
    y = h.copy()
    best_local = compute_c5(cur)
    best_h = cur.copy()
    zeta = 1.0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        t_val = (1.0 + np.sqrt(1.0 + 4.0 * zeta * zeta)) / 2.0
        y = cur + (zeta / t_val) * (y - cur)
        y_clip = np.clip(y, 0.0, 1.0)
        grad = grad_smooth_max(y_clip, n, beta, top_k)
        step = step_init / (1.0 + 0.03 * it)
        cur = project_h(y_clip - step * grad, target)
        zeta = t_val
        c_try = compute_c5(cur)
        if c_try < best_local:
            best_local = c_try
            best_h = cur.copy()
    return best_h, best_local

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=24, momentum=0.85):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    best_h = cur.copy()
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = momentum * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            best_h = cur.copy()
            stall = 0
            step = min(step * 1.02, 0.06)
        else:
            stall += 1
            step *= 0.85
            mom *= 0.5
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                cur = trial
                best_local = c_try
                best_h = cur.copy()
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 200:
            break
    return best_h, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def continuous_anneal(h, target, n, t0, budget, n_iter=40000):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(7777)
    T_init = 0.004
    T_end = 0.0001
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        frac = it / max(n_iter, 1)
        T = T_init * (T_end / T_init) ** frac
        n_pert = int(1 + 4 * (1.0 - frac))
        idx = rng.choice(n, size=n_pert, replace=False)
        trial = cur.copy()
        trial[idx] += rng.standard_normal(n_pert) * T * 8.0
        trial = project_h(trial, target)
        trial_c5 = compute_c5(trial)
        if trial_c5 < cur_c5:
            cur = trial
            cur_c5 = trial_c5
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = cur.copy()
        else:
            if rng.random() < np.exp(-(trial_c5 - cur_c5) / T):
                cur = trial
                cur_c5 = trial_c5
    return best_h, best_c5

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.full(n_init, 0.5)
        h_parent[:n_init // 2] = 0.7
        h_parent[n_init // 2:] = 0.3
        h_parent = project_h(h_parent, n_init / 2.0)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    n_cur = best_n
    target = n_cur / 2.0
    # Phase 1: Nesterov PG with increasing sharpness
    for beta in [400, 1500, 8000, 30000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 12:
            break
        best_h, best_c5 = nesterov_pg(best_h, target, n_cur, 10000, beta, t0, budget_s, step_init=0.004, top_k=96)
    # Phase 2: SLSQP polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        for beta in [8000, 40000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 7:
                break
            best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=6000)
            c_try = compute_c5(best_h)
            if c_try < best_c5:
                best_c5 = c_try
    # Phase 3: Subgradient with momentum
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = subgrad_desc(best_h, target, n_cur, 25000, 0.012, t0, budget_s, top_k=20, momentum=0.85)
    # Phase 4: Nesterov refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = nesterov_pg(best_h, target, n_cur, 15000, 50000, t0, budget_s, step_init=0.002, top_k=64)
    # Phase 5: Continuous annealing for global exploration
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        best_h, best_c5 = continuous_anneal(best_h, target, n_cur, t0, budget_s, n_iter=35000)
    # Phase 6: SLSQP after annealing
    remaining = budget_s - (time.time() - t0)
    if remaining > 12:
        for beta in [20000, 80000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 6:
                break
            best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=5000)
            c_try = compute_c5(best_h)
            if c_try < best_c5:
                best_c5 = c_try
    # Phase 7: Basin-hopping restarts
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 300)
    for att in range(12):
        if remaining < 10:
            break
        scale = 0.012 * (0.75 ** att)
        hp = best_h + rng.standard_normal(n_cur) * scale
        hp = project_h(hp, target)
        hp, c_try = nesterov_pg(hp, target, n_cur, 5000, 10000, t0, budget_s, step_init=0.003, top_k=64)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try
            remaining = budget_s - (time.time() - t0)
        else:
            remaining = budget_s - (time.time() - t0)
    # Phase 8: Binary snap + refine (exploit bang-bang structure)
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        h_bin = binary_snap(best_h, n_cur)
        cur_bin = h_bin + 0.015 * np.random.randn(n_cur)
        cur_bin = project_h(cur_bin, target)
        cur_bin, c_try = nesterov_pg(cur_bin, target, n_cur, 8000, 5000, t0, budget_s, step_init=0.004, top_k=80)
        if c_try < best_c5:
            best_h = cur_bin.copy()
            best_c5 = c_try
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            best_h = do_slsqp(best_h, n_cur, target, 30000, maxiter=5000)
            c_try = compute_c5(best_h)
            if c_try < best_c5:
                best_c5 = c_try
    # Phase 9: Coordinate descent final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 400, t0, budget_s)
    # Phase 10: Ultra-fine SLSQP
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h = do_slsqp(best_h, n_cur, target, 100000, maxiter=6000)
        c_try = compute_c5(best_h)
        if c_try < best_c5:
            best_c5 = c_try
    # Final micro-polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 150, t0, budget_s)
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
