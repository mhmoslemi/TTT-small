    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-10:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

def subgrad_polish(h, target, n_active, max_iter, t0, budget_s):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    mom = np.zeros(n)
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < 4:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n - lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = 0.85 * mom + 0.15 * g
        mn = np.linalg.norm(mom)
        if mn > 1e-14:
            mom /= mn
        improved = False
        for direction in [g, mom]:
            for step in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
                ht = project_sum(best - step * direction, target)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-15:
                    best = ht
                    best_c5 = c5t
                    improved = True
                    break
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve > 400 and no_improve % 150 == 149:
                pert = np.random.randn(n) * 0.006
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
                    mom = np.zeros(n)
            if no_improve > 4000:
                break
        else:
            no_improve = 0
    return best, best_c5

def lbfgs_with_proj(h_init, beta, maxiter, t0, budget_s):
    n = len(h_init)
    target = n / 2.0
    bounds = [(0, 1)] * n
    def obj_fn(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[0]
    def grad_fn(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[1]
    res = minimize(obj_fn, h_init, jac=grad_fn, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxls': 30})
    return project_sum(np.clip(res.x, 0, 1), target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)

    # Phase 1: Aggressive subgradient with momentum
    for na in [4, 7, 12, 18]:
        if tl() < 25:
            break
        best_h, best_c5 = subgrad_polish(best_h, n0 / 2.0, na, 4000, t0, budget_s)

    # Phase 2: L-BFGS-B with increasing beta (much faster than SLSQP)
    for beta in [40, 150, 600, 2500, 10000]:
        if tl() < 50:
            break
        h_new = lbfgs_with_proj(best_h, beta, 300, t0, budget_s)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_polish(best_h, n0 / 2.0, 10, 2000, t0, budget_s)

    # Phase 3: Resolution increase with optimization
    for n_try in [560, 720]:
        if tl() < 140:
            break
        if n_try <= len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        for beta in [80, 400, 2000]:
            if tl() < 50:
                break
            h_new = lbfgs_with_proj(h_interp, beta, 250, t0, budget_s)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_interp = h_new
        best_h, best_c5 = subgrad_polish(best_h, len(best_h) / 2.0, 12, 4000, t0, budget_s)

    # Phase 4: Multi-start exploration
    n_best = len(best_h)
    rng = np.random.RandomState(seed + 7)
    for restart in range(8):
        if tl() < 80:
            break
        if restart == 0:
            h_try = best_h.copy()
        elif restart == 1:
            h_try = project_sum(best_h + rng.randn(n_best) * 0.02, n_best / 2.0)
        elif restart == 2:
            h_try = project_sum(best_h + rng.randn(n_best) * 0.08, n_best / 2.0)
        elif restart == 3:
            blk = max(4, n_best // 12)
            h_try = best_h.copy()
            h_try[:blk] = 1.0 - h_try[:blk]
            h_try = project_sum(h_try, n_best / 2.0)
        elif restart == 4:
            shift = rng.randint(3, n_best // 5)
            h_try = project_sum(np.roll(best_h, shift), n_best / 2.0)
        elif restart == 5:
            h_try = project_sum(best_h[::-1].copy(), n_best / 2.0)
        elif restart == 6:
            # Smooth perturbation via cosine
            freq = rng.randint(3, 15)
            phase = rng.uniform(0, 2 * np.pi)
            perturb = 0.03 * np.cos(freq * np.pi * np.arange(n_best) / n_best * 2 + phase)
            h_try = project_sum(best_h + perturb, n_best / 2.0)
        else:
            # Two-frequency perturbation
            f1, f2 = rng.randint(2, 8), rng.randint(9, 20)
            p = 0.02 * np.sin(f1 * np.pi * 2 * np.arange(n_best) / n_best) + \
                0.015 * np.cos(f2 * np.pi * 2 * np.arange(n_best) / n_best)
            h_try = project_sum(best_h + p, n_best / 2.0)

        for beta in [300, 3000]:
            if tl() < 35:
                break
            h_new = lbfgs_with_proj(h_try, beta, 200, t0, budget_s)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new
        best_h, best_c5 = subgrad_polish(best_h, n_best / 2.0, 12, 3000, t0, budget_s)

    # Phase 5: Final high-beta L-BFGS
    if tl() > 40:
        h_new = lbfgs_with_proj(best_h, 30000, 500, t0, budget_s)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: Ultra-fine subgradient polishing
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h, best_c5 = subgrad_polish(best_h, target_final, 20, 30000, t0, budget_s)

    # Phase 7: Micro-steps with very fine granularity
    for it in range(15000):
        if tl() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:6]
        g = np.zeros(n_final)
        for idx in top_idx:
            lag = int(idx) - (n_final - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best_h)
            elif lag > 0:
                g[lag:] += (1.0 - best_h[:-lag])
                g[:n_final - lag] -= best_h[lag:]
            else:
                m2 = -lag
                g[:n_final - m2] += (1.0 - best_h[m2:])
                g[m2:] -= best_h[:n_final - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        for step in [0.003, 0.001, 0.0004, 0.0001, 5e-05, 2e-05, 1e-05]:
            ht = project_sum(best_h - step * g, target_final)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht
                best_c5 = c5t
                break

    return best_h, best_c5, len(best_h)
