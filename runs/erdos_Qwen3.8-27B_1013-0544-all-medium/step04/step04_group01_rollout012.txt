```python
    # Compute gradient of c_{s_star}
    g = np.zeros(n)
    s = s_star - (n - 1)  # convert to shift
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo = -h.max() - 1.0
    hi = 1.0 - h.min()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return h.copy()

def soft_sg_optimize(h, n, deadline, alpha_start, alpha_end, n_iters, lr0):
    """Soft subgradient optimization"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Interpolate alpha
        frac = it / max(1, n_iters - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr0 / (1.0 + 0.1 * it)
        
        c = np.correlate(h, 1 - h, mode='full')
        c_shifted = c - c.max()
        w = np.exp(alpha * c_shifted)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        
        # Compute weighted gradient
        g = np.zeros(n)
        shift_len = n - 1
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            s = k - shift_len
            if s >= 0:
                g[0:n-s] += w[k] * (1.0 - h[s:n])
                g[s:n] -= w[k] * h[0:n-s]
            else:
                sp = -s
                g[sp:n] += w[k] * (1.0 - h[0:n-sp])
                g[0:n-sp] -= w[k] * h[sp:n]
        
        h_new = h - lr * g
        h_new = project_feasible(h_new, target_sum)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    lo = -h.max() - 1.0
    hi = 1.0 - h.min()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    lo = -h.max() - 1.0
    hi = 1.0 - h.min()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return h.copy()

def soft_sg(h, n, deadline, alpha, lr0, lr_decay, target_sum):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    shift_len = n - 1
    it = 0
    no_improve = 0
    while time.time() < deadline:
        lr = lr0 / (1.0 + lr_decay * it)
        c = np.correlate(h, 1 - h, mode='full')
        c_shifted = c - c.max()
        w = np.exp(alpha * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum
        thresh = w_sum * 1e-12
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < thresh:
                continue
            s = k - shift_len
            if s >= 0:
                g[0:n-s] += wk * (1.0 - h[s:n])
                g[s:n] -= wk * h[0:n-s]
            else:
                sp = -s
                g[sp:n] += wk * (1.0 - h[0:n-sp])
                g[0:n-sp] -= wk * h[sp:n]
        h_new = h - lr * g
        h_new = project_feasible(h_new, target_sum)
        it += 1
        if it % 50 == 0:
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                lr = lr * 0.5
                no_improve = 0
        h = h_new
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(80):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=35):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c
```
