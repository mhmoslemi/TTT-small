                free_mask = (best > 0.02) & (best < 0.98)
                if free_mask.sum() > 0:
                    p = np.zeros(n)
                    p[free_mask] = np.random.randn(int(free_mask.sum())) * perturb_scale
                    ht = project_h(best + p, target)
                    if compute_c5(ht) < best_c5:
                        best = ht
                        best_c5 = compute_c5(best)
                        vel *= 0.2
                        no_improve = 0
                        step = base_step
                if no_improve > perturb_every * 10:
                    break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    n0 = len(best_h)
    target0 = n0 / 2.0
    dx0 = 2.0 / n0
    best_h = project_h(best_h, target0)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing beta (use ~25% of budget)
    best_h, best_c5 = slsqp_optimize(
        best_h, target0, dx0,
        [40, 120, 400, 1200, 4000, 12000],
        6000, rem, margin_s=budget_s * 0.2
    )

    # Phase 2: Primary subgradient phase (~45% of budget)
    phase2_time = budget_s * 0.45
    best_h, best_c5 = subgrad_optimize(
        best_h, target0, rem,
        max_iters=80000, margin_s=budget_s * 0.15,
        eps_start=1e-4, eps_end=1e-6,
        base_step=0.012, min_step=1e-8,
        mom_start=0.55, mom_end=0.95,
        perturb_every=500, perturb_scale=0.006
    )

    # Phase 3: Fine subgradient with very tight active set (~15%)
    best_h, best_c5 = subgrad_optimize(
        best_h, target0, rem,
        max_iters=60000, margin_s=budget_s * 0.08,
        eps_start=5e-5, eps_end=1e-7,
        base_step=0.006, min_step=1e-9,
        mom_start=0.8, mom_end=0.97,
        perturb_every=800, perturb_scale=0.003
    )

    # Phase 4: Try alternate initial condition if time remains
    if rem() > budget_s * 0.2:
        n_alt = n0
        # Create a structured perturbation of the current best
        h_alt = best_h.copy()
        freqs = np.array([3, 5, 7, 11, 13])
        x = np.linspace(0, 2, n_alt, endpoint=False)
        for f in freqs:
            amp = 0.15 / len(freqs)
            h_alt += amp * np.sin(2 * np.pi * f * x / 2.0 + np.random.uniform(0, 2*np.pi))
        h_alt = project_h(h_alt, n_alt / 2.0)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5 + 0.02:
            h_alt, c5_alt = slsqp_optimize(
                h_alt, n_alt / 2.0, 2.0 / n_alt,
                [200, 1000, 5000], 4000, rem, margin_s=budget_s * 0.06
            )
            h_alt, c5_alt = subgrad_optimize(
                h_alt, n_alt / 2.0, rem,
                max_iters=20000, margin_s=budget_s * 0.03,
                eps_start=1e-5, eps_end=1e-7,
                base_step=0.005, min_step=1e-9,
                mom_start=0.7, mom_end=0.96,
                perturb_every=500, perturb_scale=0.004
            )
            if c5_alt < best_c5:
                best_h, best_c5 = h_alt, c5_alt

    # Phase 5: Final SLSQP polish with very high beta
    if rem() > budget_s * 0.05:
        best_h, best_c5 = slsqp_optimize(
            best_h, target0, dx0,
            [5000, 20000, 50000], 10000, rem, margin_s=budget_s * 0.01
        )

    # Phase 6: Final micro-subgradient
    best_h, best_c5 = subgrad_optimize(
        best_h, target0, rem,
        max_iters=50000, margin_s=2,
        eps_start=1e-6, eps_end=1e-8,
        base_step=0.003, min_step=1e-10,
        mom_start=0.9, mom_end=0.98,
        perturb_every=1000, perturb_scale=0.002
    )

    # Final constraint enforcement
    n_final = len(best_h)
    best_h = project_h(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
