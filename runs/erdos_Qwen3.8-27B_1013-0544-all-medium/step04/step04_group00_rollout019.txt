```python
def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    ex = alpha * (g - m)
    ex -= ex.max()
    w = np.exp(ex)
    w /= w.sum()  # length 2n-1
    
    # w[m] corresponds to shift s = m - (n-1)
    # grad[j] = dx * sum_m w[m] * ((1-h)[j+s_m] - h[j-s_m]) (with 0 for out of bounds)
    
    # Vectorize: for each j, we need sum over m of w[m] * (1-h)[j + m - n + 1]
    # Let's build a matrix M[j, m] = (1-h)[j + m - n + 1] if 0 <= j+m-n+1 < n else 0
    # Then grad[j] = dx * (M @ w - M2 @ w) where M2[j,m] = h[j - m + n - 1] if valid
    
    # M[j, m] = (1-h)[j + m - n + 1]
    # Let idx = j + m - n + 1, valid if 0 <= idx < n
    # M is an n x (2n-1) matrix
    # M[j, m] = (1-h)[j + m - (n-1)]
    
    j_idx = np.arange(n)[:, None]  # n x 1
    m_idx = np.arange(2*n-1)[None, :]  # 1 x (2n-1)
    idx1 = j_idx + m_idx - (n - 1)  # n x (2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    
    # (1-h)[idx1] where valid, 0 otherwise
    # We need to be careful with indexing
    idx1_clipped = np.clip(idx1, 0, n-1)
    M1 = np.where(valid1, 1 - h[idx1_clipped], 0.0)  # n x (2n-1)
    
    idx2 = j_idx - (m_idx - (n - 1))  # = j - m + n - 1
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx2_clipped = np.clip(idx2, 0, n-1)
    M2 = np.where(valid2, h[idx2_clipped], 0.0)  # n x (2n-1)
    
    grad = dx * ((M1 - M2) @ w)
    return grad
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            room = (1.0 - h)[mask]
            s = room.sum()
            if s < 1e-14:
                break
            h[mask] += (diff / s) * room
        else:
            mask = h > 0.0
            s = h[mask].sum()
            if s < 1e-14:
                break
            h[mask] += (diff / s) * h[mask]
    return np.clip(h, 0.0, 1.0)

def compute_lse_gradient(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    ex = alpha * (g - m)
    ex -= ex.max()
    w = np.exp(ex)
    w /= w.sum()
    
    # Compute gradient using the matrix formulation
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    shift = m_idx - (n - 1)
    
    idx1 = j_idx + shift
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    
    idx2 = j_idx - shift
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx2c = np.clip(idx2, 0, n - 1)
    
    # grad[j] = dx * sum_m w[m] * ((1-h)[j+s] - h[j-s])
    # = dx * (sum_m w[m]*(1-h)[j+s_m] - sum_m w[m]*h[j-s_m])
    
    # First term: for each j, sum over m of w[m] * (1-h)[j + s_m]
    # = (1 - h) @ W^T where W[j,m] = w[m] if valid else 0
    W1 = np.where(valid1, w[None, :], 0.0)  # n x (2n-1)
    term1 = (W1 * (1 - h)[idx1c]).sum(axis=1)
    
    W2 = np.where(valid2, w[None, :], 0.0)
    term2 = (W2 * h[idx2c]).sum(axis=1)
    
    return dx * (term1 - term2)

    v_v = np.zeros(n)
    t = 0
    best_h = h.copy()

            t += 1
            grad = compute_lse_gradient(h, n, alpha)
            m_v = beta1 * m_v + (1 - beta1) * grad
            v_v = beta2 * v_v + (1 - beta2) * grad**2
            m_hat = m_v / (1 - beta1**t)
            v_hat = v_v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                best_h = h_new.copy()
            h = h_new

        if t % 10000 == 0:
            alpha *= 2.0
        # Decay learning rate
        if t % 50000 == 0:
            lr *= 0.9
    
    return best_h, best_c

def coordinate_polish(h, n, deadline, max_sweeps=50):

                c = compute_c5(h, n)
                if c < best_c:

    return best_h, best_c
```

```python
    term2 = np.where(valid2, h[idx2c] * w[None, :], 0.0).sum(axis=1)
    
    return dx * (term1 - term2)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            room = (1.0 - h)[mask]
            s = room.sum()
            if s < 1e-14:
                break
            h[mask] += (diff / s) * room
        else:
            mask = h > 0.0
            s = h[mask].sum()
            if s < 1e-14:
                break
            h[mask] += (diff / s) * h[mask]
    return np.clip(h, 0.0, 1.0)

def precompute_grad_indices(n):
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    shift = m_idx - (n - 1)
    idx1 = j_idx + shift
    idx2 = j_idx - shift
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    return idx1c, idx2c, valid1, valid2

def grad_lse(h, n, alpha, pre):
    idx1c, idx2c, valid1, valid2 = pre
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    ex = alpha * (g - m)
    ex -= ex.max()
    w = np.exp(ex)
    w /= w.sum()
    t1 = (valid1 * (1.0 - h)[idx1c] * w[None, :]).sum(axis=1)
    t2 = (valid2 * h[idx2c] * w[None, :]).sum(axis=1)
    return dx * (t1 - t2)

def adam_run(h, n, deadline, alpha, lr, pre):
    h = h.copy()
    mv = np.zeros(n)
    vv = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    while time.time() < deadline:
        t += 1
        g = grad_lse(h, n, alpha, pre)
        mv = 0.9 * mv + 0.1 * g
        vv = 0.999 * vv + 0.001 * g * g
        mh = mv / (1 - 0.9**t)
        vh = vv / (1 - 0.999**t)
        step = lr * mh / (np.sqrt(vh) + 1e-8)
        h_new = project_feasible(h - step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 5000:
            h = project_feasible(best_h + 0.02 * np.random.randn(n), n)
            no_improve = 0
        if t % 20000 == 0:
            lr *= 0.9
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + np.log(np.exp(sh).sum()) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_polish(h, n, deadline, max_sweeps=100):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            for j in sorted(ones):
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c:
                    best_c = c
                    ones.remove(j)
                    ones.add(i)
                    zeros.remove(i)
                    zeros.add(j)
                    improved = True
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
        if not improved:
            break
    return h_bin, best_c

def pair_search(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -2):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c:
                                best_c = c
                                prog = True
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
    return h, best_c

    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    best_h = h0.copy()

    n = n0
    pre = precompute_grad_indices(n)

    # Phase 1: Adam with escalating alpha (45% budget)
    p1 = t0 + 0.45 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [2000, 10000, 50000, 200000]:
        phase_end = min(p1, time.time() + 100)
        h, c = adam_run(h, n, phase_end, alpha, 0.1, pre)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SLSQP polish (10% budget)
    p2 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p2 - 10:
        h2 = slsqp_polish(best_h.copy(), n, 500000, 800, p2 - 5)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Higher resolution attempts (25% budget)
    p3 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p3 - 20:
        for nh in [300, 350, 400]:
            if time.time() > p3 - 20:
                break
            xo = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            xn = np.linspace(0, 2, nh, endpoint=False) + 1.0 / nh
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            pre_nh = precompute_grad_indices(nh)
            hh, ch = adam_run(hh, nh, min(p3 - 10, time.time() + 80), 50000, 0.08, pre_nh)
            if ch < best_c:
                best_c = ch
                best_h = hh.copy()
                n = nh
                pre = pre_nh

    # Phase 4: Binary polish (10% budget)
    p4 = t0 + 0.90 * (budget_s - 5)
    if time.time() < p4 - 5:
        hb, cb = binary_polish(best_h.copy(), n, p4 - 3, max_sweeps=80)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        hp = project_feasible(best_h + 0.15 * rng.randn(n), n)
        hb2, cb2 = binary_polish(hp, n, p4 - 3, max_sweeps=50)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 5: Final Adam + pair (10% budget)
    if time.time() < dl - 15:
        pre_final = precompute_grad_indices(best_h.size)
        hf, cf = adam_run(best_h.copy(), best_h.size, dl - 8, 500000, 0.02, pre_final)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if time.time() < dl - 3:
        best_h, best_c = pair_search(best_h, best_h.size, dl - 1,
                                     [0.02, 0.01, 0.005, 0.002, 0.001])

    best_h = project_feasible(best_h, best_h.size)
    best_c = compute_c5(best_h, best_h.size)
    return best_h, best_c, best_h.size
```
