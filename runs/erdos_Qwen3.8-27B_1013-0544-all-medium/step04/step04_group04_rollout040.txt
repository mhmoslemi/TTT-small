    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    s = np.sum(exp_vals)
    weights = exp_vals / s
    val = (m + np.log(s) / beta) * dx
    grad = np.zeros(n)

            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]

    _, g = smooth_max_grad(x, _state['beta'])
    return g + 2.0 * _state['lam'] * (x.sum() - _state['target'])

def lbfgsb_optimize(h0, beta, lam, maxiter):
    n = len(h0)
    _state['beta'] = beta
    _state['lam'] = lam
    _state['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_lbfgsb, h0, method='L-BFGS-B', jac=_grad_lbfgsb,
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project_h(res.x, n / 2.0)

def subgrad_descent(h, n_iter, top_k, steps_list, pert_every, pert_scale):
    n = len(h)
    target = n / 2.0
    no_imp = 0
    for it in range(n_iter):
        if time_left_fn() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, max(3, n // 30))
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            elif lag < 0 and abs(lag) < n:
                m2 = -lag
                grad[:n - m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n - m2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        c5_cur = compute_c5(h)
        found = False
        for step in steps_list:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-15:
                h = ht
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % pert_every == pert_every - 1:
                sc = pert_scale * min(5, no_imp // pert_every + 1)
                ht = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(ht) < c5_cur:
                    h = ht
                    no_imp = 0
            if no_imp > max(80, n_iter // 12):
                break
    return h, compute_c5(h)

def try_resolution(h_best, n_best, c5_best, n_res, time_threshold):
    if time_left_fn() < time_threshold:
        return h_best, c5_best
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_res)
    h_res = np.interp(x_new, x_old, h_best)
    h_res = project_h(h_res, n_res / 2.0)
    for beta in [200, 2000, 20000]:
        if time_left_fn() < 25:
            break
        h_new = lbfgsb_optimize(h_res, beta, lam=10.0, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h_res):
            h_res = h_new
    # Upsample back to original resolution
    x_old2 = np.linspace(0, 2, n_res)
    x_new2 = np.linspace(0, 2, n_best)
    h_up = np.interp(x_new2, x_old2, h_res)
    h_up = project_h(h_up, n_best / 2.0)
    c5_up = compute_c5(h_up)
    if c5_up < c5_best:
        return h_up, c5_up
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    _time_state[0] = time.time()
    _time_state[1] = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    n = len(best_h)
    best_h = project_h(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing beta at native resolution
    for beta in [50, 200, 1000, 5000, 20000, 50000, 100000]:
        if time_left_fn() < 40:
            break
        h_new = lbfgsb_optimize(best_h, beta, lam=10.0, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Subgradient polish
    if time_left_fn() > 40:
        best_h, best_c5 = subgrad_descent(best_h, 15000, top_k=7,
            steps_list=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005],
            pert_every=200, pert_scale=0.005)

    # Phase 3: Multi-resolution exploration
    for n_res in [150, 250, 350, 450]:
        best_h, best_c5 = try_resolution(best_h, n, best_c5, n_res, 70)

    # Phase 4: Random perturbations + re-optimize
    for trial in range(10):
        if time_left_fn() < 50:
            break
        scale = 0.008 + 0.004 * trial
        h_pert = project_h(best_h + np.random.randn(n) * scale, n / 2.0)
        h_new = lbfgsb_optimize(h_pert, 8000, lam=10.0, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c5_new
        if time_left_fn() < 80:
            h_new = lbfgsb_optimize(h_pert, 30000, lam=10.0, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Very high beta L-BFGS-B
    if time_left_fn() > 40:
        for beta in [80000, 150000, 300000]:
            if time_left_fn() < 25:
                break
            h_new = lbfgsb_optimize(best_h, beta, lam=10.0, maxiter=600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h, best_c5 = h_new, c5_new

    # Phase 6: Final subgradient with fine steps
    if time_left_fn() > 15:
        best_h, best_c5 = subgrad_descent(best_h, 30000, top_k=5,
            steps_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
            pert_every=300, pert_scale=0.003)

    # Phase 7: One last L-BFGS-B if time remains
    if time_left_fn() > 20:
        h_new = lbfgsb_optimize(best_h, 200000, lam=10.0, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    return best_h, best_c5, len(best_h)
