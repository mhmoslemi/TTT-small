        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

        free = (h > 1e-12) & (h < 1.0 - 1e-12)

def _smograd(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)

    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp

        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]

    if c5_7 < best_c5:
        best_h, best_c5 = h7, c5_7

    # Phase 8: One more Nesterov pass
    if time.time() < deadline - 10:
        target = len(best_h) / 2.0
        best_h, best_c5 = do_nesterov(best_h, target, 20000, deadline,
                                       0.01, 0.98, 4, 0.80)

    return best_h, best_c5, len(best_h)
