    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    diff = target - int(np.sum(h == 1.0))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if diff > 0:
        h[zeros[:diff]] = 1.0
    elif diff < 0:
        h[ones[0:(-diff)]] = 0.0
    c5 = compute_c5(h)
    no_improve = 0
    for _ in range(max_iter):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
            if no_improve > 3000:
                break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Binary structure search at current n
    if tl() > 80:
        n_b = n_best
        best_bin = None
        best_bin_c5 = float('inf')
        n_starts = 6
        iters_per = 40000
        for s in range(n_starts):
            if tl() < 30:
                break
            if s == 0:
                bh = np.zeros(n_b)
                bh[np.random.choice(n_b, n_b//2, replace=False)] = 1.0
            elif s == 1:
                bh = np.zeros(n_b)
                per = max(4, n_b // 30)
                for i in range(n_b):
                    if (i % per) < per // 2:
                        bh[i] = 1.0
                bh = project_h(bh, n_b / 2.0)
            elif s == 2:
                bh = np.zeros(n_b)
                nb = 12
                bl = n_b // nb
                for b in range(nb):
                    if b % 3 != 1:
                        bh[b*bl:min(n_b, (b+1)*bl)] = 1.0
                bh = project_h(bh, n_b / 2.0)
            else:
                bh = np.zeros(n_b)
                bh[np.random.choice(n_b, n_b//2, replace=False)] = 1.0
            bh2, c5b = binary_hill_climb(bh, iters_per)
            if c5b < best_bin_c5:
                best_bin_c5 = c5b
                best_bin = bh2
        if best_bin is not None and best_bin_c5 < best_c5 + 0.05:
            h_rel = best_bin.copy().astype(float)
            h_rel = project_h(h_rel, n_best / 2.0)
            h_opt, c5_opt = do_slsqp(h_rel, [50, 500, 5000], 2000, tl)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Main SLSQP with beta annealing
    if tl() > 60:
        h_new, c5_new = do_slsqp(best_h, [50, 200, 1000, 5000, 20000], 4000, tl)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Subgradient with backtracking
    if tl() > 40:
        h_sg, c5_sg = do_subgrad(best_h, tl, 60000, 6, 0.025)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 4: Coordinate descent with penalty
    if tl() > 40:
        h_cd, c5_cd = coord_descent_pen(best_h, tl, 30, 50.0)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 5: Try different resolutions
    for n_res in [400, 300, 250]:
        if tl() < 80:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        h_opt, c5_opt = do_slsqp(h_res, [100, 1000, 8000], 3000, tl)
        h_sg2, c5_sg2 = do_subgrad(h_opt, tl, 20000, 5, 0.02)
        if c5_sg2 < best_c5:
            best_h, best_c5, n_best = h_sg2, c5_sg2, n_res

    # Phase 6: Final SLSQP polish at high beta
    if tl() > 30:
        h_new, c5_new = do_slsqp(best_h, [20000, 50000, 100000], 6000, tl)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 7: Final fine subgradient
    if tl() > 15:
        h_sg3, c5_sg3 = do_subgrad(best_h, tl, 40000, 4, 0.01)
        if c5_sg3 < best_c5:
            best_h, best_c5 = h_sg3, c5_sg3

    # Phase 8: Tiny SLSQP
    if tl() > 20:
        h_new, c5_new = do_slsqp(best_h, [50000, 100000], 8000, tl)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final subgradient nap
    if tl() > 5:
        h_sg4, c5_sg4 = do_subgrad(best_h, tl, 10000, 3, 0.005)
        if c5_sg4 < best_c5:
            best_h, best_c5 = h_sg4, c5_sg3 if c5_sg4 < best_c5 else h_sg4, c5_sg4

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
