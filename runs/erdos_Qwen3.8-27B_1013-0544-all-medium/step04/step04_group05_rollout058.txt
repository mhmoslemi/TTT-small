import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_exact(y, target):
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        if np.clip(y - tau, 0, 1).sum() > target:
            lo = tau
        else:
            hi = tau
    return np.clip(y - (lo + hi) / 2.0, 0, 1)

def soft_max(corr, beta):
    m = corr.max()
    s = np.clip(beta * (corr - m), -500, 500)
    es = np.exp(s)
    return m + np.log(es.sum()) / beta, es / es.sum()

def weighted_grad(h, W, dx):
    n = len(h)
    grad = np.zeros(n)
    w0 = W[n-1]
    grad[:] = w0 * (1.0 - 2.0 * h)
    for m in range(1, n):
        wk = W[n-1+m]
        wm = W[n-1-m]
        if wk > 1e-12:
            grad[m:] += wk * (1.0 - h[:n-m])
            grad[:n-m] -= wk * h[m:]
        if wm > 1e-12:
            grad[:n-m] += wm * (1.0 - h[m:])
            grad[m:] -= wm * h[:n-m]
    return grad * dx

def fista_smooth(h, target, dx, betas, iters_per_beta, time_fn):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 10:
            break
        t = 1.0
        y = best.copy()
        for it in range(iters_per_beta):
            if time_fn() < 5:
                break
            corr = np.correlate(y, 1.0 - y, mode="full")
            fval, W = soft_max(corr, beta)
            g = weighted_grad(y, W, dx)
            # Lipschitz constant estimate
            L = 4.0 * beta
            alpha = 1.0 / L
            z = y - alpha * g
            x_new = project_exact(z, target)
            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - best)
            t = t_new
            best = x_new
            c5 = compute_c5(best)
            if c5 < best_c5:
                best_c5 = c5
        best_c5 = compute_c5(best)
    return best, best_c5

def subgrad_softmax(h, target, dx, time_fn, max_iters, step_list, momentum, perturb_every, perturb_scale):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)

    beta = 500

        _, W = soft_max(corr, beta)
        g = weighted_grad(best, W, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:

        vel = momentum * vel + (1 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            break

        found = False

            c5t = compute_c5(ht)

                    best_c5 = c5t

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_exact(y, target):
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        tau = (lo + hi) * 0.5
        s = float(np.clip(y - tau, 0.0, 1.0).sum())
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def soft_max_weights(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    es = np.exp(s)
    W = es / es.sum()
    val = m + np.log(es.sum()) / beta
    return val, W

def weighted_grad(h, W, dx):
    n = len(h)
    grad = np.zeros(n)
    w0 = W[n - 1]
    if w0 > 1e-15:
        grad[:] = w0 * (1.0 - 2.0 * h)
    for m in range(1, n):
        wk = W[n - 1 + m]
        wm = W[n - 1 - m]
        nm = n - m
        if wk > 1e-12:
            grad[m:] += wk * (1.0 - h[:nm])
            grad[:nm] -= wk * h[m:]
        if wm > 1e-12:
            grad[:nm] += wm * (1.0 - h[m:])
            grad[m:] -= wm * h[:nm]
    return grad * dx

def slsqp_opt(h, target, dx, beta, max_iter, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj_fn(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        v, _ = soft_max_weights(c, beta)
        return v * dx
    def con_fn(h_in):
        return h_in.sum() - target
    def jac_fn(h_in):
        return np.ones(n)
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        h_new = project_exact(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h):
            return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)

def subgrad_all(h, target, dx, time_fn, max_iters, beta, step_list, momentum, perturb_every, perturb_scale):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_imp = 0
    for it in range(max_iters):
        if time_fn() < 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        _, W = soft_max_weights(corr, beta)
        g = weighted_grad(best, W, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            break
        d = -vel / vn
        found = False
        for step in step_list:
            ht = project_exact(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_imp = 0
                found = True
                break
        if not found:
            no_imp += 1
            if no_imp >= perturb_every and it % perturb_every == perturb_every - 1:
                p = np.random.randn(n) * perturb_scale
                ht = project_exact(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.5
                    no_imp = 0
            if no_imp > perturb_every * 12:
                break
    return best, best_c5

def interpolate_h(h_old, n_new, target):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_exact(h_new, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    n0 = len(best_h)

    # Phase 1: SLSQP at base resolution
    if rem() > 120:

    # Phase 2: Subgradient with all-lags softmax, low beta (broad)
    if rem() > 80:

    # Phase 3: Subgradient with higher beta (focused)

    # Phase 4: Resolution increase
    if rem() > 120:
        n2 = n0 * 2
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        h2 = interpolate_h(best_h, n2, target2)
        c5_2 = compute_c5(h2)
        if rem() > 50:
            h2, c5_2 = slsqp_opt(h2, target2, dx2, 5000, 6000, rem)
        if rem() > 40:

                h2, target2, dx2, rem,

    # Phase 5: Try alternative resolution (n0 or n0+100)
