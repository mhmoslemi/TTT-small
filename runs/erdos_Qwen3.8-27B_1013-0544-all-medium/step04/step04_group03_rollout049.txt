```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n=None):
    if n is None:
        n = len(h)
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    exp_shift = np.exp(x_shift)
    weights = exp_shift / exp_shift.sum()
    val = m + np.log(exp_shift.sum() / len(x)) / beta
    return val, weights

def compute_gradient(h, weights, n):
    """Compute gradient of smooth max C5 objective.
    weights: array of length 2n-1 (for all lags).
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for ell in range(-(n-1), n):
        w = weights[n-1+ell]
        if w < 1e-14:
            continue
        if ell >= 0:
            if n - ell > 0:
                grad[:n-ell] += w * one_minus_h[ell:]
            if ell > 0:
                grad[ell:] -= w * h[:n-ell]
            elif ell == 0:
                grad -= w * h  # j >= 0 always
        else:
            la = -ell
            grad[la:] += w * one_minus_h[:n-la]
            grad[:n-la] -= w * h[la:]
    return grad * dx

def projected_gradient(h, target, n, beta, n_iter, t0, budget, 
                        lr_init=0.01, lr_decay=0.999, momentum=0.9):
    """Nesterov-accelerated projected gradient descent."""
    cur = h.copy()
    v = np.zeros(n)  # momentum
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    lr = lr_init
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        
        # Compute objective and weights
        lags = compute_lags(cur)
        obj, weights = smooth_max_and_weights(lags, beta)
        
        # Compute gradient
        grad = compute_gradient(cur, weights, n)
        
        # Nesterov step
        v = momentum * v - lr * grad
        trial = project_h(cur + v, target)
        
        # Evaluate
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
            lr = min(lr * 1.01, 0.1)
        else:
            # Retry with smaller step
            v = 0.5 * v
            trial = project_h(cur + v, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                best_h = cur.copy()
            else:
                v = np.zeros(n)
                lr *= 0.8
        lr *= lr_decay
        
        if lr < 1e-6:
            break
    
    return best_h, best_c5

def binary_hill_climb(h_bin, n, t0, budget, n_iter=200000):
    """Hill climbing on binary sequences using single swaps."""
    rng = np.random.default_rng(42)
    cur = h_bin.copy()
    cur_lags = np.correlate(cur, 1.0 - cur, mode="full")
    cur_max = float(np.max(cur_lags))
    best_h = cur.copy()
    best_max = cur_max
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
