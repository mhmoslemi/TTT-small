    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    k = min(30, len(c_sorted))
    gap = c_sorted[0] - c_sorted[k - 1] if k > 1 else c_sorted[0]
    base_beta = max(gap * 0.4, max_c * 0.008)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.0075, 0.0037, 0.0018,
                                      0.0009, 0.00045, 0.0002, 0.0001,
                                      0.00005, 0.00002, 0.00001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06,
                                      0.03, 0.015, 0.007, 0.003,
                                      0.001, 0.0005, 0.0002, 0.0001,
                                      0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_per_iter = 0.003
        elif n <= 8192:
            est_per_iter = 0.007
        elif n <= 16384:
            est_per_iter = 0.018
        else:
            est_per_iter = 0.04 * (n / 32768.0)

        max_iter = min(8000, max(50, int(remaining / (est_per_iter * 12))))

        try:
            result = minimize_ls(z, beta, n, max_iter)
            a_new = _softmax(result)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def minimize_ls(z, beta, n, max_iter):
    from scipy.optimize import minimize as sp_minimize

    def obj(z_local):
        return _ls_objective(z_local, beta, n)

    def grad(z_local):
        return _ls_gradient(z_local, beta, n)

    result = sp_minimize(obj, z, jac=grad, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-18,
                                  'gtol': 1e-14, 'maxcor': 50, 'maxls': 80})
    return result.x


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:2 * n]
            active[top_idx] = True
            n_active = 2 * n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-5 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        x = np.zeros(n)
        r = -g_t - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for _ in range(300):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        u = x
        w = np.zeros(n)
        r2 = np.ones(n) - hess_matvec(w)
        p2 = r2.copy()
        rs2 = np.dot(r2, r2)
        for _ in range(300):
            Ap2 = hess_matvec(p2)
            pAp2 = np.dot(p2, Ap2)
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            w += alpha2 * p2
            r2_new = r2 - alpha2 * Ap2
            rs2_new = np.dot(r2_new, r2_new)
            if np.sqrt(rs2_new) < 1e-13:
                break
            p2 = r2_new + (rs2_new / rs2) * p2
            r2 = r2_new
            rs2 = rs2_new

        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for _ in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-10 * (1.0 + c_max):
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for _ in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _peak_targeted_refine(a_init, deadline, max_rounds=40):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        n_peaks = min(15, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        w = np.zeros(2 * n - 1)
        for k in top_peaks:
            w[k] = 1.0 / n_peaks

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)

        f_old = c_max
        step = 0.08 / (1.0 + 0.02 * rnd)
        for _ in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        k = min(25, max(5, len(c) // 20))
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c)
        w[top_k] = 1.0 / k

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.88
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent(a_init, deadline, max_sweeps=100):
    """Simple cyclic coordinate descent for final polish."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            # Try a few values for a[i]
            old_val = a[i]
            for frac in [0.5, 0.75, 1.25, 1.5]:
                a[i] = old_val * frac
                a[i] = max(0.0, a[i])
                s_try = a.sum()
                if s_try < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s_try
                c_try = _full_convolve(a_norm, a_norm)
                if c_try.max() < c_max - 1e-14:
                    a = a_norm
                    improved = True
                    break
            a[i] = a[i]  # keep last value
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        if not improved:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        a = 1.0 + 0.5 * np.exp(-((t - 0.25) ** 2) / 0.03) + 0.5 * np.exp(-((t - 0.75) ** 2) / 0.03)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining:
    - Multi-scale warm-starting (small n to large n)
    - Aggressive L-BFGS with log-sum-exp continuation
    - Active-set Newton with adaptive active set
    - Peak-targeted subgradient refinement
    - Mirror descent
    - Diversified random restarts
    - Final coordinate descent polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent found. Using uniform n={n_parent}")
        update_best(parent, _score(parent), "uniform_init")

    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 1: L-BFGS from parent] t={elapsed:.1f}s")
    if time.monotonic() < deadline - 400:
        a1 = _run_lbfgs(parent, deadline - 400, min_time_remaining=200.0, aggressive=True)
        update_best(a1, _score(a1), "lbfgs_parent")

    # Phase 2: Multi-scale approach
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 2: Multi-scale] t={elapsed:.1f}s")
    if time.monotonic() < deadline - 200:
        n_current = min(parent is not None and n_parent or 16384, 256)
        a_scale = _interpolate_to_n(parent, n_current) if parent is not None else np.ones(n_current) / n_current
        update_best(a_scale, _score(a_scale), f"multiscale_{n_current}_init")

        scale_sizes = [256, 512, 1024, 4096, 8192, 16384]
        for n_s in scale_sizes:
            if time.monotonic() > deadline - 50:
                break
            if n_s <= n_current:
                continue
            a_scale = _interpolate_to_n(a_scale, n_s)
            time_for_lbfgs = max(10, min(60, (deadline - time.monotonic() - 50) * 0.3))
            a_scale = _run_lbfgs(a_scale, time.monotonic() + time_for_lbfgs,
                                 min_time_remaining=5.0, aggressive=True)
            update_best(a_scale, _score(a_scale), f"multiscale_{n_s}")

    # Phase 3: Active-set Newton on best
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 3: Active-set Newton] t={elapsed:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 80, time.monotonic() + 80),
                                         max_iter=150, tol_active=0.001)
        update_best(a_as, sc_as, "active_newton_1")

        if time.monotonic() < deadline - 50:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 40, time.monotonic() + 40),
                                               max_iter=100, tol_active=0.0005)
            update_best(a_as2, sc_as2, "active_newton_2")

    # Phase 4: Peak-targeted refinement
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 4: Peak refinement] t={elapsed:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_pt, sc_pt = _peak_targeted_refine(best_seq, min(deadline - 40, time.monotonic() + 40),
                                            max_rounds=30)
        update_best(a_pt, sc_pt, "peak_refine")

    # Phase 5: Mirror descent
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 5: Mirror descent] t={elapsed:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_md, sc_md = _mirror_descent(best_seq, min(deadline - 20, time.monotonic() + 30),
                                      max_steps=3000, lr_init=0.03)
        update_best(a_md, sc_md, "mirror_descent")

    # Phase 6: Diversified restarts
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 6: Restarts] t={elapsed:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restarts = [
            (0.01, "restart_001"),
            (0.03, "restart_003"),
            (0.05, "restart_005"),
            (0.10, "restart_010"),
        ]
        for sigma, label in restarts:
            if time.monotonic() > deadline - 10:
                break
            p_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_pert, min(deadline - 5, time.monotonic() + 20),
                             min_time_remaining=3.0, aggressive=False)
            update_best(a_r, _score(a_r), label)

    # Phase 7: Structured initializations at best n
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 7: Structured inits] t={elapsed:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        for kind in ["sine", "bimodal", "cos_asym"]:
            if time.monotonic() > deadline - 10:
                break
            a_struct = _make_profile(n_best, kind)
            sc_struct = _score(a_struct)
            if sc_struct < best_score + 0.5:
                a_struct_opt = _run_lbfgs(a_struct, min(deadline - 5, time.monotonic() + 15),
                                          min_time_remaining=3.0, aggressive=False)
                update_best(a_struct_opt, _score(a_struct_opt), f"struct_{kind}")

    # Phase 8: Final active-set with tight tolerance
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 8: Final polish] t={elapsed:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1,
                                           max_iter=50, tol_active=0.0002)
        update_best(a_fin, sc_fin, "final_as_tight")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
