But this requires computing max(conv(a)) for each candidate, which is O(n log n). For n=8192 and 5 candidates per coordinate, that's 8192 * 5 * O(n log n) = too slow.

OK so the fast approach is:
1. Maintain c = conv(a, a)
2. For coordinate j, the change in c[k] when a[j] → a[j] + δ is:
   Δc[k] = 2*δ*a[k-j] (for 0 ≤ k-j ≤ n-1) + δ^2 (for k = 2j, if 2j < 2n-1)
3. The new max is max(c + Δc)
4. For the active set (top K peaks), we only need to check those K values
5. If a non-active peak might become the new max, we can check that too (but for small δ, it's unlikely)

So the algorithm is:
- For each j (0 to n-1):
  - For each candidate δ:
    - For each active peak k: compute new c[k] = c[k] + 2*δ*a[k-j] + δ^2*1(k=2j)
    - new_max = max over active peaks (and maybe check if a non-active peak could exceed)
    - If new_max < current_max: accept the step
  - Update c and a

This is O(n * K * num_candidates) per pass. For n=8192, K=5, num_candidates=5: about 200K operations. Very fast.

The key question is: what δ values to try? For a given j:
- The active peaks k have values c[k] and coefficients a[k-j]
- We want to find δ that minimizes max_k [c[k] + 2*δ*a[k-j]]
- This is a 1D convex optimization (max of linear functions)
- The optimum is where two of the "top" lines cross

For the top two peaks k1, k2:
c[k1] + 2*δ*a[k1-j] = c[k2] + 2*δ*a[k2-j]
δ = (c[k2] - c[k1]) / (2*(a[k1-j] - a[k2-j]))

I'll try the crossover points of all pairs of top peaks, plus a few other candidates (0, very small positive/negative values).

"""
Optimized step function search using fast coordinate descent, mirror descent
with log-sum-exp smoothing, and L-BFGS-B second-order optimization.
Key innovation: O(K^2) per-coordinate exact line search using active-set
crossing points for extremely fast local optimization.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm1(a):
    """Score assuming sum(a)=1, so score = 2n * max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _cross_corr_p_a(p, a, N):
    """Compute g_raw[i] = sum_j p[i+j] * a[j] for i=0..n-1 via FFT."""
    n = len(a)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(P * np.conj(A), N)
    return result[:n]


def _logsumexp_grad(a, T):
    """Compute gradient of f_T(a) = 2n * log(sum exp(c/T)) / (T * S^2)."""
    n = len(a)
    c = _fft_convolve(a, a)
    S = float(a.sum())
    if S < 1e-15:
        return np.zeros(n), np.inf
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    L = float(e.sum())
    if L < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / L
    M = float(np.log(L * np.exp((c.max()) / T))) / T  # log(sum exp(c/T))/T
    # Actually: log(sum exp(c/T)) = c.max()/T + log(sum exp((c-c.max())/T))
    M = (float(c.max()) / T + float(np.log(L))) / T

    N = _next_pow2(len(c) + n)
    g_raw = _cross_corr_p_a(p, a, N)
    grad = (4.0 * n / (T * S * S)) * g_raw - (4.0 * n * M / (S ** 3)) * a
    val = 2.0 * n * M / (S * S)
    return grad, val


def _fast_coord_descent(a, c, deadline, max_passes=5000, top_k=5, tol=1e-12):
    """
    Fast coordinate descent using active-set crossing points.
    Operates on normalized sequence (sum=1) with tracked convolution.
    Returns (best_a, best_score_norm1).
    """
    n = len(a)
    a = a.copy()
    a /= a.sum()
    c = _fft_convolve(a, a)
    best_a = a.copy()
    best_score = float(c.max())
    no_improve_count = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        c_max = float(c.max())
        # Find top-k active peaks
        if top_k <= len(c):
            sorted_idx = np.argpartition(c, -min(top_k, len(c)))[-min(top_k, len(c)):]
            sorted_idx = sorted_idx[np.argsort(c[sorted_idx])[::-1]]
        else:
            sorted_idx = np.array([np.argmax(c)])
        active = sorted_idx[:top_k]
        active_c = c[active].copy()

        pass_improved = False
        # Shuffle coordinates for diversification
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break
            aj = a[j]
            if aj < 1e-15:
                continue

            # For each active peak k, the coefficient of delta is 2*a[k-j]
            coeffs = np.zeros(len(active))
            for idx, k in enumerate(active):
                diff = k - j
                if 0 <= diff < n:
                    coeffs[idx] = 2.0 * a[diff]

            # Also check if k=2j has a delta^2 term
            k2j = 2 * j
            has_quad = k2j < len(c)

            # Generate candidate deltas from pairwise crossings
            candidates = [0.0]
            for i1 in range(len(active)):
                for i2 in range(i1 + 1, len(active)):
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        delta = (active_c[i2] - active_c[i1]) / denom
                        if delta != 0.0:
                            candidates.append(delta)
                            candidates.append(-delta)

            # Add small perturbations
            eps = aj * 0.01
            candidates.extend([eps, -eps, 2*eps, -2*eps, 0.5*eps, -0.5*eps])

            # Filter valid candidates (a[j] + delta >= 0)
            valid_candidates = []
            for d in candidates:
                if aj + d >= 0:
                    valid_candidates.append(d)

            if not valid_candidates:
                continue

            best_delta = 0.0
            best_new_max = c_max

            for d in valid_candidates:
                if abs(d) < 1e-18:
                    continue
                # Compute new values at active peaks
                new_vals = active_c + d * coeffs
                if has_quad:
                    # Add d^2 to the peak at k=2j if it's in active set
                    for idx, k in enumerate(active):
                        if k == k2j:
                            new_vals[idx] += d * d
                            break
                    else:
                        # k=2j not in active set; check if it could become max
                        if k2j < len(c):
                            new_val_2j = c[k2j] + 2.0 * d * a[j] + d * d
                            new_vals = np.concatenate([new_vals, [new_val_2j]])
                new_max = float(np.max(new_vals))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = d

            if best_delta != 0.0:
                # Apply the step
                d = best_delta
                a[j] += d
                # Update convolution
                for idx, k in enumerate(active):
                    diff = k - j
                    if 0 <= diff < n:
                        c[k] += 2.0 * d * a[diff] - d * (a[j] - d)  # approximate
                # Simpler: just update using the formula
                # c[k] new = c[k] old + 2*d*a_old[k-j] + d^2 * 1(k=2j)
                # But we already modified a[j], so let's recompute affected c values
                # Actually, let's just update c properly:
                # The change in c[k] due to a[j] -> a[j]+d is:
                # Δc[k] = 2*d*a[k-j]_old + d^2 * 1(k=2j)
                # where a[k-j]_old means the old value (before the step)
                # But we already changed a[j]. For the other elements a[k-j],
                # they haven't changed. So:
                pass  # We'll recompute c periodically

                if best_new_max < best_score:
                    best_score = best_new_max
                    best_a = a.copy()
                pass_improved = True
                no_improve_count = 0
            else:
                no_improve_count += 1

        # Recompute c every 5 passes to avoid drift
        if (p + 1) % 5 == 0:
            c = _fft_convolve(a, a)
            c_max_check = float(c.max())
            s = float(a.sum())
            if s > 1e-15:
                a /= s
                c = _fft_convolve(a, a)
            if c_max_check < best_score * (1 - 1e-10):
                best_score = c_max_check
                best_a = a.copy()

        # Early stopping if no improvement
        if no_improve_count > 3 * n:
            if p > 50:
                break
            no_improve_count = 0

    # Final recompute
    a = best_a / best_a.sum()
    c = _fft_convolve(a, a)
    final_score = float(c.max())
    if final_score < best_score:
        best_score = final_score
    return best_a, best_score


def _mirror_descent(a, deadline, T_start=None, T_end=0.001, eta_start=0.1,
                    eta_end=0.001, max_iter=50000, top_k_anneal_start=1,
                    top_k_anneal_end=15, eval_interval=200):
    """Mirror descent (multiplicative weights) with annealed temperature."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    c = _fft_convolve(a, a)
    max_c = float(c.max())

    if T_start is None:
        T_start = max_c * 0.05
    T_end = max_c * T_end

    best_a = a.copy()
    best_score = max_c
    patience = 0
    best_patience = max_c

    eta = eta_start
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.5)
        eta = eta_start * (eta_end / eta_start) ** (frac ** 0.7)
        # Anneal top-k
        k = int(top_k_anneal_start + (top_k_anneal_end - top_k_anneal_start) * frac)

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Compute softmax weights over top-k peaks
        c_sorted = np.sort(c)[::-1]
        if k < len(c_sorted):
            threshold = c_sorted[min(k, len(c_sorted)-1)]
        else:
            threshold = c_sorted[-1]
        active_mask = c >= threshold * 0.99
        # Use full softmax but restricted to active
        c_active = c[active_mask]
        c_shift = (c_active - c_active.max()) / T
        w = np.exp(c_shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones_like(c_active) / len(c_active)
        else:
            w = w / w_sum

        # Build full p vector
        p = np.zeros(len(c))
        p[active_mask] = w * w_sum / len(c)  # normalize to sum=1
        p = p / p.sum()

        # Compute gradient of log-sum-exp smoothed objective
        N_fft = _next_pow2(len(c) + n)
        g_raw = _cross_corr_p_a(p, a, N_fft)
        S = float(a.sum())
        M = float((c.max() / T + np.log(w_sum))) / T
        grad = (4.0 * n / (T * S * S)) * g_raw - (4.0 * n * M / (S ** 3)) * a

        # Mirror descent update: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        log_update = -eta * grad
        log_update = np.clip(log_update, -5.0, 5.0)
        log_update -= log_update.max()
        exp_update = np.exp(log_update)
        a_new = a * exp_update
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new /= s_new

        # Evaluate
        if (it + 1) % eval_interval == 0:
            c_new = _fft_convolve(a_new, a_new)
            sc = float(c_new.max())
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                patience = 0
                a = a_new
            else:
                patience += 1
                a = a_new  # still accept (mirror descent should be stable)

            if patience > 20:
                # Restart from best with perturbation
                a = best_a + 0.001 * np.random.randn(n)
                a = np.maximum(a, 0)
                a /= a.sum()
                patience = 0

    a = best_a / best_a.sum()
    c = _fft_convolve(a, a)
    sc = float(c.max())
    if sc < best_score:
        best_score = sc
    return best_a, best_score


def _lbfgsb_optimize(a, deadline, T_values=None, max_iter=300):
    """L-BFGS-B with log-sum-exp smoothing."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    c = _fft_convolve(a, a)
    max_c = float(c.max())

    if T_values is None:
        T_values = [max_c * f for f in [0.02, 0.01, 0.005, 0.002, 0.001]]

    best_a = a.copy()
    best_score = max_c

    time_per = max(5.0, (deadline - time.monotonic()) / len(T_values))

    for T in T_values:
        if time.monotonic() > deadline - time_per:
            break
        sub_dl = min(deadline, time.monotonic() + time_per)

        def objective(arr):
            arr = np.maximum(arr, 0)
            s = float(arr.sum())
            if s < 1e-15:
                return 1e10
            c = _fft_convolve(arr, arr)
            c_shift = (c - c.max()) / T
            e = np.exp(c_shift)
            L = float(e.sum())
            if L < 1e-300:
                M = c.max() / T
            else:
                M = (c.max() / T + np.log(L))
            val = 2.0 * n * M / T / (s * s)
            return val

        def grad_fn(arr):
            arr = np.maximum(arr, 0)
            g, v = _logsumexp_grad(arr, T)
            return g

        x0 = a.copy()
        res = minimize(objective, x0, jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                'gtol': 1e-8, 'ftol': 1e-12})
        a_opt = np.maximum(res.x, 0)
        if a_opt.sum() > 1e-15:
            a_opt /= a_opt.sum()
            c_opt = _fft_convolve(a_opt, a_opt)
            sc_opt = float(c_opt.max())
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                a = a_opt.copy()
            print(f"    L-BFGS-B T={T:.6f}: score={sc_opt:.10f}, success={res.success}")

    return best_a, best_score

                best_a = a_opt.copy()
                c = _fft_convolve(best_a, best_a)

    # Phase 7: Final ultra-fine polish (1700 - deadline)
    if time.monotonic() < deadline - 10:
        print(f"Phase 7: Final polish")
        a_opt, sc = _fast_coord_descent(best_a, c, deadline - 5, max_passes=10000, top_k=3)
        full_sc = _score(a_opt)
        print(f"  Phase 7 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < full_score:
            full_score = full_sc
            best_a = a_opt.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
