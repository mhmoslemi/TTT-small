    where H[i,j] = h[i+j] (Hankel matrix, h has length 2n-1, h >= 0)
    """
    a = np.ones(n) / n  # Initial guess
    
    for outer in range(max_outer):
        # Determine support
        support = a > 1e-15
        m = support.sum()
        if m == 0:
            return np.ones(n) / n
        if m == 1:
            a = np.zeros(n)
            a[support] = 1.0
            return a
        
        # Solve H_sub x = ones(m) where H_sub is H[support, support]
        # Use CG with the hankel_matvec restricted to support
        
        idx = np.where(support)[0]
        
        def restricted_matvec(v_sub):
            # v_sub has length m, corresponds to H[idx, idx] @ v_sub
            # Full vector: zero except at idx
            v_full = np.zeros(n)
            v_full[idx] = v_sub
            result_full = hankel_matvec(h, v_full, n)
            return result_full[idx]
        
        # CG to solve H_sub x = 1
        x = _cg_solve(restricted_matvec, np.ones(m), max_cg=max_cg, tol=tol)
        x_sum = x.sum()
        if x_sum < 1e-15:
            # Fallback
            a = np.zeros(n)
            a[support] = 1.0 / m
            return a
        x = x / x_sum
        
        # Check non-negativity
        a_new = np.zeros(n)
        a_new[idx] = x
        
        if (a_new >= -1e-12).all():

                    a[idx[i]] = 0.0
            # Renormalize remaining
            s = a.sum()
            if s > 0:
                a = a / s
    
    return a

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _cg_solve(matvec, rhs, max_iter=300, tol=1e-12):
    """Conjugate gradient solver."""
    n = len(rhs)

    r = rhs.copy()

    rs = np.dot(r, r)
    if rs < tol:
        return x
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_iter):

        pAp = np.dot(p, Ap)
        if pAp < 1e-30:

        r = r - alpha * Ap

        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

Primal-dual optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _hankel_matvec(h, v):
    n = len(v)
    v_rev = v[::-1].copy()
    total_len = len(h) + n - 1
    N = _next_pow2(total_len)
    fh = np.fft.rfft(h, N)
    fv = np.fft.rfft(v_rev, N)
    result = np.fft.irfft(fh * fv, N)
    return result[n - 1:2 * n - 1]


def _solve_equality_qp(h, n, max_cg=500, tol=1e-12):
    """Solve min a^T H a s.t. sum(a)=1 (no non-negativity) using CG.
