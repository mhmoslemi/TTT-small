"""
Optimized step function coefficient search.
Uses a combination of smooth optimization (log-sum-exp / power mean with L-BFGS),
FFT-based active-set Newton polishing with Schur complement reduction,
and multi-n exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, w, n):
    """Compute H @ v where H[j,m] = 2*w[j+m] using FFT."""
    v_rev = v[::-1]
    out_len = len(v_rev) + len(w) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(v_rev, N)
    F_W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _c_matvec(v, a, A_indices, n):
    """Compute C @ v where C[i,j] = 2*a[A_indices[i]-j]. O(n*|A|) direct."""
    m = len(A_indices)
    result = np.zeros(m)
    for i in range(m):
        k = A_indices[i]
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            aj = a[j_lo:k - j_lo + 1][::-1]
            vj = v[j_lo:j_hi + 1]
            result[i] = 2.0 * np.dot(aj, vj)
    return result


def _ct_matvec(v, a, A_indices, n):
    """Compute C^T @ v where C[i,j] = 2*a[A_indices[i]-j]. Result is length n."""
    result = np.zeros(n)
    for i in range(len(A_indices)):
        k = A_indices[i]
        if v[i] == 0.0:
            continue
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            result[j_lo:j_hi + 1] += 2.0 * v[i] * a[j_lo:k - j_lo + 1][::-1]
    return result


def _cg_hankel(w, n, rhs, max_iter=200, tol=1e-12):
    """Solve H @ x = rhs where H is the Hankel matrix with entries 2*w[j+m].
    Uses CG (H may be indefinite, so use BiCGSTAB-like restarts)."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        if rs_new < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

    """Power mean objective: 2*n*(sum c_k^p)^(1/p)"""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    val = float(np.sum(c ** p)) ** (1.0 / p)
    return 2.0 * n * val

    c = _full_convolve(a, a)

    wc = _full_convolve(w, a_rev)
