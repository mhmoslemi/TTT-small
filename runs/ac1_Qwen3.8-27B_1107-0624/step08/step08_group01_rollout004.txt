"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_remaining=15.0):
    """L-BFGS with aggressive LSE beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-18:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    # Extended annealing schedule - more gradual at the end
    betas = base_beta * np.array(
        [1.0, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03,
         0.018, 0.012, 0.008, 0.005, 0.003, 0.002,
         0.0012, 0.0008, 0.0005, 0.0003, 0.0002,
         0.0001, 0.00005, 0.00003, 0.00001, 0.000005, 0.000002]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3.0:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        # More iterations at smaller betas for finer convergence
        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 12))))

        def func(zv):
            return _lse_obj(zv, beta, n)

        def gfunc(zv):
            return _lse_grad(zv, beta, n)

        try:
            result = minimize(
                func, z, jac=gfunc, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 80,
                    'maxfun': max_iter * 3,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _run_adam(a_init, deadline, lr=0.005, min_remaining=15.0):
    """Adam optimizer with LSE temperature annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30)).copy()
    z = z - z.mean()

    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    beta = base_beta
    min_beta = base_beta * 0.00001
    stall_count = 0
    last_improve = time.monotonic()

    while time.monotonic() + min_remaining < deadline:
        obj, grad = _lse_obj(z, beta, n)

        t_adam += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t_adam)
        v_hat = v / (1 - 0.999 ** t_adam)

        z_new = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = _softmax(z_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc_new
            stall_count = 0
            last_improve = time.monotonic()
            # Slightly increase learning rate on improvement
            lr = min(0.02, lr * 1.01)
        else:
            stall_count += 1

        z = z_new

        # Adaptive beta decay
        if time.monotonic() - last_improve > 10.0 and beta > min_beta:
            beta *= 0.8
            stall_count = 0
            last_improve = time.monotonic()

        if stall_count > 5000:

            z = z + 0.01 * rng.standard_normal(n)
            z = z - z.mean()
            stall_count = 0
            lr = lr * 0.5

def _peak_manipulate(a, deadline, n_trials=500, min_remaining=10.0):
    """Direct peak manipulation: reduce top contributors to max peak."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(99999)

    for trial in range(n_trials):
        if time.monotonic() + min_remaining > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find contributing indices
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo >= k_hi:
            # Peak at boundary - just reduce the largest element
            idx = int(np.argmax(best_a))
            a_try = best_a.copy()
            red = a_try[idx] * 0.05
            a_try[idx] -= red
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = best_a[i_range] * best_a[j_range]
        total = contrib.sum()
        if total < 1e-18:
            continue

        # Strategy: reduce top contributors proportionally to their contribution
        n_reduce = max(1, int(len(contrib) * rng.uniform(0.05, 0.2)))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = best_a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * rng.uniform(0.02, 0.1)
            a_try[idx] -= red
            reduction += red

        # Distribute to low-contributing elements
        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
            jv = k_star - idx
            if 0 <= jv < n:
                low_mask[jv] = False

        if low_mask.sum() > 0:
            a_try[low_mask] += reduction / low_mask.sum()

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try

    return best_a, best_sc


def _active_gd(a_init, deadline, min_remaining=10.0):
    """Active-set gradient descent with backtracking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.02
    tol = 0.003
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0

        # Active set: top peaks
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > 20:
            top_idx = np.argsort(c)[::-1][:10]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 10

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Backtracking line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.3
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.3

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol = max(0.00001, tol * 0.6)
                lr *= 0.5
            if no_improve >= max_no_improve:
                a = best_a.copy()
                tol = max(0.00001, tol * 0.4)
                lr = 0.01
                no_improve = 0
                if tol <= 0.00002:
                    break
        else:
            if tol > 0.00001:
                tol = max(0.00001, tol * 0.98)
            lr = min(0.05, lr * 1.02)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.abs(t - 0.5)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

            print(f"  NEW BEST: {best_score:.10f} [final_gd] t={time.monotonic()-t_start:.1f}s", flush=True)

        # Tiny peak manipulation
        if time.monotonic() < deadline - 3.0:
            a_ps, sc_ps = _peak_manipulate(best_a, deadline - 2.0, n_trials=100, min_remaining=2.0)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  NEW BEST: {best_score:.10f} [final_peak] t={time.monotonic()-t_start:.1f}s", flush=True)

        # One more L-BFGS with very small beta
        remaining = deadline - 2.0 - time.monotonic()
        if remaining > 5.0:
            n = len(best_a)
            c_best = _full_convolve(best_a, best_a)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 0.00001
            max_iter = min(2000, max(50, int(remaining / 0.05)))

            def _final_func(zv):
                return _lse_obj(zv, beta_fin, n)

            def _final_grad(zv):
                return _lse_grad(zv, beta_fin, n)

            try:
                result = minimize(_final_func, z, jac=_final_grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 60, 'maxls': 50})
                a_fin2 = _softmax(result.x)
                a_fin2 = np.maximum(a_fin2, 0.0)
                s_fin2 = a_fin2.sum()
                if s_fin2 > 1e-10:
                    a_fin2 = a_fin2 / s_fin2
                    sc_fin2 = _score(a_fin2)
                    if sc_fin2 < best_score:
                        best_a = a_fin2.copy()
                        best_score = sc_fin2
                        print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s", flush=True)
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}", flush=True)
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * (0.1 / 256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # Verify with evaluate_sequence
    try:
        ev_score = evaluate_sequence([float(x) for x in final])
        if np.isfinite(ev_score) and ev_score < sc_final + 0.01:
            sc_final = ev_score
            print(f"Verified with evaluate_sequence: {ev_score:.10f}", flush=True)
    except Exception:
        pass

    return [float(x) for x in final]
