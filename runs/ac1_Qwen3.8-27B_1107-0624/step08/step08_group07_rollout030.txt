import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _subgradient(a, c, top_k=5, tol_frac=0.999):
    """Compute subgradient of max(c) w.r.t. a, averaged over top peaks."""
    n = len(a)
    c_max = c.max()
    # Find active peaks
    threshold = c_max * tol_frac
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(c[active])[::-1][:top_k]]
    
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(active)
    return g, c_max

def _mirror_descent_step(a, g, lr):
    """Multiplicative weights update."""
    # a_new[j] = a[j] * exp(-lr * g[j]) / Z
    # To avoid overflow, subtract max
    log_ratio = -lr * g
    log_ratio -= log_ratio.max()  # for numerical stability
    ratio = np.exp(log_ratio)
    a_new = a * ratio
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Find lambda such that sum(max(v - lambda, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Subgradient method with backtracking line search."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = 0.01
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        g, c_max = _subgradient(a, c, top_k=top_k)
        
        # Project gradient onto tangent space (remove component along 1-vector)
        g_proj = g - g.mean()
        
        # Try several step sizes
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * scale * g_proj
            a_cand = a - step
            a_cand = _project_simplex(a_cand)
            c_cand = _fft_convolve(a_cand, a_cand)
            sc = float(c_cand.max())

                best_a = a_cand.copy()
                a = a_cand

def _pairwise_exchange(a, deadline, max_rounds=500):

    n = len(a)

    best_a = a.copy()

        hi = min(kk, n - 1)
        if lo <= hi:
            A[i, lo:hi+1] = 2.0 * a[kk-hi:kk-lo+1][::-1]
        cur_vals[i] = c[kk]

    target = cur_vals.max()
    rhs = (target - cur_vals) / 2.0
    rhs_full = np.zeros(m + 1)
    rhs_full[:m] = rhs
    A[m, :] = 1.0

    G = A @ A.T
    reg = 1e-8 * np.eye(m + 1)
    try:
        d = A.T @ np.linalg.solve(G + reg, rhs_full)
    except np.linalg.LinAlgError:
        d, _, _, _ = np.linalg.lstsq(A, rhs_full, rcond=None)

    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return a.copy(), c_max

    best_a = a.copy()
    best_sc = c_max
    for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        a_cand = a + alpha * d / d_norm
        a_cand = _project_simplex(a_cand)
        c_cand = _fft_convolve(a_cand, a_cand)
        sc = float(c_cand.max())
        if sc < best_sc:
            best_sc = sc
            best_a = a_cand.copy()

    return best_a, best_sc


def _sgd_phase(a_init, deadline, max_iter=300000, top_k=3,
               lr_start=0.008, decay=0.9995,
               eq_every=800, patience=300):
    """
    Subgradient descent with adaptive step size and periodic equalization.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_start
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        if c_max < best_score * (1 - 1e-12):
            best_score = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            break

        g = _subgradient_topk(a, c, top_k)
        g_tan = g - g.mean()
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break

        step_dir = g_tan / g_norm
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.125]:
            a_cand = a - lr * scale * step_dir
            a_cand = _project_simplex(a_cand)
            c_cand = _fft_convolve(a_cand, a_cand)
            sc = float(c_cand.max())
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                c = c_cand
                improved = True
                break

        if improved:
            lr = min(lr * 1.002, 0.04)
        else:
            lr *= 0.92

        if it % eq_every == 0 and it > 0:
            a_eq, sc_eq = _equalize_peaks(a, c, top_k=5, tol_frac=0.999)
            if sc_eq < best_score:
                best_score = sc_eq
                best_a = a_eq.copy()
                a = a_eq

    return best_a, best_score


def _mirror_descent_phase(a_init, deadline, max_iter=100000, top_k=3,
                          lr_start=0.02, decay=0.999):
    """
    Multiplicative weights (mirror descent with KL) on the simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_start
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        if c_max < best_score * (1 - 1e-12):
            best_score = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 200:
            break

        g = _subgradient_topk(a, c, top_k)
        g_centered = g - g.mean()

        # Multiplicative update: a_new = a * exp(-lr * g_centered) / Z
        log_ratio = -lr * g_centered
        log_ratio -= log_ratio.max()
        ratio = np.exp(log_ratio)
        a_new = a * ratio
        sn = a_new.sum()
        if sn < 1e-300:
            break
        a_new = a_new / sn

        c_new = _fft_convolve(a_new, a_new)
        sc = float(c_new.max())

        if sc < best_score * (1 - 1e-10):
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.001, 0.08)
        else:
            lr *= 0.97

    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_rounds=2000):
    """
    Cyclic coordinate descent: try small multiplicative changes to each element.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)

    # Precompute which elements affect the current peak
    k_peak = int(np.argmax(c))

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Update peak location
        k_peak = int(np.argmax(c))
        lo = max(0, k_peak - n + 1)
        hi = min(k_peak, n - 1)

        improved = False
        # Cycle through elements
        start = (rnd * 32) % n
        for offset in range(min(32, n)):
            j = (start + offset) % n
            if best_a[j] < 1e-10:
                continue

            # Try scaling this element
            for scale in [0.9, 0.95, 1.05, 1.1]:
                a_cand = best_a.copy()
                a_cand[j] *= scale
                sc_r = a_cand.sum()
                if sc_r < 1e-15:
                    continue
                a_cand /= sc_r
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max())
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        if not improved and rnd % 50 == 0:
            #满身 try symmetric pair adjustments
            for j in range(min(64, n // 2)):
                j2 = n - 1 - j
                if j == j2:
                    continue
                if best_a[j] < 1e-10 or best_a[j2] < 1e-10:
                    continue
                diff = best_a[j] - best_a[j2]
                if abs(diff) < 1e-10:
                    continue
                for frac in [0.1, 0.2]:
                    a_cand = best_a.copy()
                    a_cand[j] -= diff * frac
                    a_cand[j2] += diff * frac
                    a_cand = np.maximum(0, a_cand)
                    sc_r = a_cand.sum()
                    if sc_r < 1e-15:
                        continue
                    a_cand /= sc_r
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max())
                    if sc < best_score * (1 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def _perturb_restart(a, rng, sigma):
    """Gaussian perturbation in log-space followed by normalization."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z += sigma * rng.standard_normal(n)
    z = z - z.mean()
    z = np.clip(z, -30, 30)
    a_new = np.exp(z)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: subgradient -> mirror descent -> coordinate descent
    -> perturbed restarts -> final polish.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent or default
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using uniform initialization, n=8192")

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Init: n={n}, score={best_score:.10f}")

    # Phase 1: SGD (0 - 400s)
    p1_end = min(deadline, t0 + 400.0)
    print(f"Phase 1: SGD [0-{p1_end-t0:.0f}s]")
    a1, s1 = _sgd_phase(best_a, p1_end, max_iter=200000, top_k=3,
                         lr_start=0.008, decay=0.9995, eq_every=600, patience=400)
    sc1 = _score(a1)
    print(f"  Phase 1: score={sc1:.10f}, t={time.monotonic()-t0:.0f}s")
    if sc1 < best_score:
        best_score = sc1
        best_a = a1.copy()

    # Phase 2: Mirror descent (400 - 700s)
    p2_end = min(deadline, t0 + 700.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: Mirror descent")
        a2, s2 = _mirror_descent_phase(best_a, p2_end, max_iter=80000, top_k=3,
                                       lr_start=0.015, decay=0.999)
        sc2 = _score(a2)
        print(f"  Phase 2: score={sc2:.10f}, t={time.monotonic()-t0:.0f}s")
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()

    # Phase 3: Coordinate descent (700 - 950s)
    p3_end = min(deadline, t0 + 950.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Coordinate descent")
        a3, s3 = _coordinate_descent(best_a, p3_end, max_rounds=3000)
        sc3 = _score(a3)
        print(f"  Phase 3: score={sc3:.10f}, t={time.monotonic()-t0:.0f}s")
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    # Phase 4: SGD with tighter settings (950 - 1200s)
    p4_end = min(deadline, t0 + 1200.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Refined SGD")
        a4, s4 = _sgd_phase(best_a, p4_end, max_iter=200000, top_k=2,
                            lr_start=0.004, decay=0.9998, eq_every=400, patience=500)
        sc4 = _score(a4)
        print(f"  Phase 4: score={sc4:.10f}, t={time.monotonic()-t0:.0f}s")
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()

    # Phase 5: Perturbed restarts (1200 - 1600s)
    p5_end = min(deadline, t0 + 1600.0)
    if time.monotonic() < p5_end:
        print(f"Phase 5: Perturbed restarts")
        sigmas = [0.003, 0.005, 0.008, 0.012, 0.02]
        time_per = max(30.0, (p5_end - time.monotonic()) / len(sigmas))
        for r, sigma in enumerate(sigmas):
            if time.monotonic() > p5_end - 60:
                break
            a_pert = _perturb_restart(best_a, rng, sigma)
            sub_dl = min(p5_end - 10, time.monotonic() + time_per * 0.8)
            # Quick SGD
            a_r, s_r = _sgd_phase(a_pert, sub_dl, max_iter=50000, top_k=3,
                                  lr_start=0.005, decay=0.999, eq_every=300, patience=150)
            sc_r = _score(a_r)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={sc_r:.10f} t={time.monotonic()-t0:.0f}s")
            if sc_r < best_score:
                best_score = sc_r
                best_a = a_r.copy()

    # Phase 6: Final polish (1600 - deadline)
    if time.monotonic() < deadline - 20:
        print(f"Phase 6: Final polish")
        sub_dl = deadline - 15
        # Equalization
        c_best = _fft_convolve(best_a, best_a)
        a_eq, sc_eq = _equalize_peaks(best_a, c_best, top_k=7, tol_frac=0.998)
        sc_eq_full = _score(a_eq)
        print(f"  Equalize: score={sc_eq_full:.10f}")
        if sc_eq_full < best_score:
            best_score = sc_eq_full
            best_a = a_eq.copy()

        # Quick mirror descent
        if time.monotonic() < sub_dl - 10:
            a_md, s_md = _mirror_descent_phase(best_a, sub_dl, max_iter=30000, top_k=2,
                                               lr_start=0.005, decay=0.9995)
            sc_md = _score(a_md)
            print(f"  Final MD: score={sc_md:.10f}, t={time.monotonic()-t0:.0f}s")
            if sc_md < best_score:
                best_score = sc_md
                best_a = a_md.copy()

    # Validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
