"""
Multi-strategy optimizer for step function coefficients.
Combines L-BFGS with LSE smoothing, active-set polishing, Nesterov accelerated
subgradient, and multi-length exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
Leverages the convex structure of the problem and uses adaptive time allocation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t ** 2)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Slight taper at edges
        edge = max(1, n // 20)
        taper = np.linspace(0.8, 1.0, edge)
        a[:edge] = taper
        a[-edge:] = taper[::-1]
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_lbfgs(a_init, deadline, beta_start=None, min_remaining=15.0):
    """L-BFGS optimization with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001

    if beta_start is None:
        beta_start = max(gap, max_c * 0.003)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Adaptive beta schedule: start large, decrease geometrically
    n_betas = 25
    ratio = 0.7
    betas = [beta_start * (ratio ** i) for i in range(n_betas)]
    # Extend with very small betas
    betas += [betas[-1] * 0.5 ** i for i in range(20)]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        # Scale iterations with n
        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        # More iterations for larger n (need more to converge)
        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 20))))

        def obj(z):
            return _lse_obj(z, beta, n)

        def grad(z):
            return _lse_grad(z, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    LSE beta={beta:.2e}: score={sc_new:.10f} (iter {i+1}/{len(betas)})")
        except Exception:
            pass

    return current_a


def _nesterov_subgrad(a_init, deadline, n_iter=None):
    """Nesterov accelerated projected subgradient method."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    y = a.copy()
    t_param = 1.0

    if n_iter is None:
        n_iter = 100000

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = y.copy()

        # Compute subgradient at active lag
        # c_k = sum_{i=0}^{n-1-k} a_i * a_{i+k} for k in [0, n-1]
        # For k >= n: c_k = c_{2n-2-k} by symmetry
        if k_star <= n - 1:
            k = k_star
        else:
            k = 2 * n - 2 - k_star

        grad = np.zeros(n)
        if k == 0:
            grad[:] = 2.0 * y
        else:
            grad[:n - k] += y[k:]
            grad[k:] += y[:n - k]
            grad *= 2.0

        # Step size: use a diminishing schedule
        # For convex problems, alpha_t = 1/(t+1) works but slowly
        # Better: use a constant step + averaging
        alpha = 0.5 / (np.sqrt(float(i + 1)) + 1.0)

        # Nesterov update
        y_new = y - alpha * grad
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / s_new

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        beta_mom = (t_param - 1.0) / t_new
        a_new = y_new + beta_mom * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new
        else:
            a_new = a_new / s_a

        y = a_new
        t_param = t_new

    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_start=0.003, tol_end=0.00001,
                       lr_start=0.02, top_k=5, momentum=0.9):
    """Active-set gradient descent with momentum and adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0
    max_stall = 200
    v = np.zeros(n)  # momentum buffer

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Select active set
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > top_k * 5:
            top_indices = np.argpartition(c, -top_k)[-top_k:]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = top_k

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Nesterov momentum
        v = momentum * v - lr * grad_t
        a_try = a + v
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            v *= 0.5
            a_try = a - lr * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                no_improve += 1
                continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < current_sc - 1e-14:
            a = a_try
            no_improve = 0
        else:
            # Backtracking
            improved = False
            for bt in range(10):
                lr_try = lr * (0.5 ** (bt + 1))
                v_bt = momentum * v - lr_try * grad_t
                a_bt = a + v_bt
                a_bt = np.maximum(a_bt, 0.0)
                s_bt = a_bt.sum()
                if s_bt < 1e-10:
                    continue
                a_bt = a_bt / s_bt
                sc_bt = _score(a_bt)
                if sc_bt < current_sc - 1e-14:
                    a = a_bt
                    v = v_bt
                    improved = True
                    break
            if not improved:
                no_improve += 1
                v *= 0.5
                if no_improve % 10 == 0:
                    tol = max(tol_end, tol * 0.8)
                    lr *= 0.7
                if no_improve >= max_stall:
                    a = best_a.copy()
                    v = np.zeros(n)
                    tol = max(tol_end, tol * 0.5)
                    lr = lr_start * 0.5
                    no_improve = 0
                    stall += 1
                    if stall > 15:
                        break
            else:
                if tol > tol_end:
                    tol = max(tol_end, tol * 0.99)
        # Slowly increase lr
        if lr < lr_start:
            lr = min(lr_start, lr * 1.005)

    return best_a, best_sc


def _peak_shave(a, deadline, n_trials=200):
    """Direct peak shaving on top contributors."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(99999)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Map to actual lag
        if k_star <= n - 1:
            k = k_star
        else:
            k = 2 * n - 2 - k_star

        if k == 0:
            # Peak at lag 0: reduce the largest elements
            top_idx = np.argsort(a)[::-1][:max(1, n // 50)]
            a_try = a.copy()
            for idx in top_idx:
                a_try[idx] *= 0.95
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            continue

        # Find contributing indices
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k + n - 1)
        # Actually for c_k = sum_i a[i] * a[i+k] where i ranges over valid indices:
        # i goes from 0 to n-1-k (for k >= 0)
        # Or equivalently, the pairs are (i, i+k) for i in [0, n-1-k]
        i_range = np.arange(0, n - k)
        j_range = i_range + k
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()

        if total < 1e-15:
            continue

        # Reduce top contributors
        n_reduce = max(1, int(len(contrib) * 0.05))
        top_local = np.argsort(contrib)[::-1][:n_reduce]

        a_try = a.copy()
        reduction = 0.0
        reduce_mask = np.zeros(n, dtype=bool)
        for li in top_local:
            i_idx = i_range[li]
            j_idx = j_range[li]
            red = min(a_try[i_idx], a_try[j_idx]) * 0.1
            a_try[i_idx] -= red * 0.5
            a_try[j_idx] -= red * 0.5
            reduction += red * 0.5
            reduce_mask[i_idx] = True
            reduce_mask[j_idx] = True

        # Redistribute to non-reduced elements
        add_mask = ~reduce_mask
        if add_mask.sum() > 0:
            a_try[add_mask] += reduction / add_mask.sum()

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _local_search(a_init, deadline, n_trials=5000, sigma=0.01):
    """Random perturbation local search with Gaussian noise."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(777)

    t_current = time.monotonic()
    t_start = t_current

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Adaptive sigma: decrease over time
        progress = (time.monotonic() - t_start) / max(1e-10, deadline - t_start)
        sig = sigma * (1.0 - 0.8 * progress)

        # Perturb in log-space
        z = np.log(np.maximum(a_init, 1e-30))
        z += sig * rng.standard_normal(n)
        z = z - z.mean()
        a_try = _softmax(z)

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a_init = a_try  # Reset exploration from new best

    return best_a, best_sc

"""
Multi-strategy convex optimizer for step function coefficient search.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with LSE smoothing,
Nesterov accelerated subgradient, active-set polishing with momentum,
peak shaving, multi-length exploration, and diverse restarts.
Exploits the convex structure of max of quadratic forms over the simplex.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-1.5 * t ** 2)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "broad_taper":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * t * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_lbfgs(a_init, deadline, beta_start=None, min_remaining=15.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001

    if beta_start is None:
        beta_start = max(gap, max_c * 0.004)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    phase2_betas = [phase1_betas[-1] * (0.5 ** i) for i in range(25)]
    all_betas = phase1_betas + phase2_betas

        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.0025 * scale
        max_iter = min(6000, max(50, int(remaining / (est_iter_time * 25))))

        def obj_fn(z):

        def grad_fn(z):
            return _lse_gradient(z, beta, n)

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

def _nesterov_subgrad(a_init, deadline):
    """Nesterov accelerated projected subgradient for the convex max problem."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    y = a.copy()
    t_param = 1.0
    iteration = 0

    # Use a slowly decreasing step with momentum
    base_lr = 2.0 / np.sqrt(float(n))

    while time.monotonic() < deadline - 1.0:
        iteration += 1

        c = _full_convolve(y, y)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = y.copy()

        # Map to actual lag
        if k_star <= n - 1:
            k = k_star
        else:
            k = 2 * n - 2 - k_star

        # Subgradient
        grad = np.zeros(n)
        if k == 0:
            grad[:] = 2.0 * y
        else:
            grad[:n - k] += y[k:]
            grad[k:] += y[:n - k]
            grad *= 2.0

        # Step size with decay
        alpha = base_lr / (np.sqrt(float(iteration)) + 1.0)

        # Nesterov update
        y_new = y - alpha * grad
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / s_new

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        beta_mom = (t_param - 1.0) / t_new
        a_new = y_new + beta_mom * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new
        else:
            a_new = a_new / s_a

        y = a_new
        t_param = t_new

    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_start=0.003, tol_end=0.00001,
                       lr_start=0.018, top_k=5, momentum=0.85):
    """Active-set gradient descent with Nesterov momentum and adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0
    max_stall = 250
    v = np.zeros(n)

    while time.monotonic() < deadline - 1.5:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Active set selection
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > top_k * 8:
            top_indices = np.argpartition(c, -top_k)[-top_k:]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = top_k

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-19:
            break

        # Nesterov momentum step
        v = momentum * v - lr * grad_t
        a_try = a + v
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            v = -lr * grad_t
            a_try = a + v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                no_improve += 1
                v *= 0.5
                continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < current_sc - 1e-14:
            a = a_try
            no_improve = 0
        else:
            improved = False
            for bt in range(12):
                lr_b = lr * (0.5 ** (bt + 1))
                v_b = momentum * v - lr_b * grad_t
                a_b = a + v_b
                a_b = np.maximum(a_b, 0.0)
                s_b = a_b.sum()
                if s_b < 1e-10:
                    continue
                a_b = a_b / s_b
                sc_b = _score(a_b)
                if sc_b < current_sc - 1e-14:
                    a = a_b
                    v = v_b
                    improved = True
                    break
            if not improved:
                no_improve += 1
                v *= 0.3
                if no_improve % 8 == 0:
                    tol = max(tol_end, tol * 0.75)
                    lr *= 0.7
                if no_improve >= max_stall:
                    a = best_a.copy()
                    v = np.zeros(n)
                    tol = max(tol_end, tol * 0.4)
                    lr = lr_start * 0.4
                    no_improve = 0
                    stall += 1
                    if stall > 15:
                        break
            else:
                if tol > tol_end:
                    tol = max(tol_end, tol * 0.985)
        if lr < lr_start and no_improve < 50:
            lr = min(lr_start, lr * 1.008)

    return best_a, best_sc


def _peak_shave(a, deadline, n_trials=300):
    """Targeted peak reduction by modifying top contributors to the convolution max."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(424242)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        if k_star <= n - 1:
            k = k_star
        else:
            k = 2 * n - 2 - k_star

        a_try = a.copy()

        if k == 0:
            # Reduce top elements
            n_reduce = max(1, n // 40)
            top_idx = np.argsort(a)[::-1][:n_reduce]
            for idx in top_idx:
                a_try[idx] *= rng.uniform(0.90, 0.97)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            continue

        # For lag k: c_k = sum_{i=0}^{n-1-k} a_i * a_{i+k}
        i_range = np.arange(0, n - k)
        j_range = i_range + k
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()

        if total < 1e-15:
            continue

        # Reduce top contributors
        frac = rng.uniform(0.02, 0.08)
        n_reduce = max(1, int(len(contrib) * frac))
        top_local = np.argsort(contrib)[::-1][:n_reduce]

        reduction = 0.0
        reduce_mask = np.zeros(n, dtype=bool)
        for li in top_local:
            i_idx = i_range[li]
            j_idx = j_range[li]
            red = min(a_try[i_idx], a_try[j_idx]) * rng.uniform(0.05, 0.15)
            a_try[i_idx] -= red * 0.5
            a_try[j_idx] -= red * 0.5
            reduction += red * 0.5
            reduce_mask[i_idx] = True
            reduce_mask[j_idx] = True

        # Redistribute to other positions
        add_mask = ~reduce_mask
        if add_mask.sum() > 0:
            a_try[add_mask] += reduction / add_mask.sum()

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _local_search(a_init, deadline, n_trials=3000, sigma_start=0.015):
    """Annealed random perturbation search in log-space."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(31337)
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        elapsed = time.monotonic() - t0
        total_time = max(1e-10, deadline - t0)
        progress = min(1.0, elapsed / total_time)
        sig = sigma_start * (1.0 - 0.85 * progress)

        z = np.log(np.maximum(a_init, 1e-30))
        z += sig * rng.standard_normal(n)
        z = z - z.mean()
        a_try = _softmax(z)
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a_init = a_try

    return best_a, best_sc


def _fourier_init(n, rng, n_modes=20):
    """Initialize using random Fourier features for diverse starting points."""
    freqs = rng.integers(1, n // 2, size=n_modes).astype(float)
    phases = rng.uniform(0, 2 * np.pi, size=n_modes)
    amps = rng.exponential(1.0, size=n_modes)
    t = np.arange(n)
    signal = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        signal += a * np.cos(2.0 * np.pi * f * t / n + p)
    signal = np.abs(signal) + 0.1
    return signal / signal.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS (LSE), Nesterov subgradient, active-set polish,
    peak shaving, multi-n exploration, diverse restarts, and final ultra-fine polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}", flush=True)
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}", flush=True)

    global_best_a = best_a.copy()
    global_best_score = best_score

    # === Phase 1: L-BFGS with LSE ===
    print(f"\n[P1] L-BFGS+LSE (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p1_end = min(deadline - 700.0, t_start + 450.0)
    a1 = _run_lbfgs(global_best_a, p1_end, min_remaining=30.0)
    sc1 = _score(a1)
    if sc1 < global_best_score:
        global_best_a = a1.copy()
        global_best_score = sc1
        print(f"  NEW BEST: {global_best_score:.10f} [lbfgs_p1] n={len(global_best_a)}", flush=True)
    print(f"  P1 done: best={global_best_score:.10f} (t={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 2: Nesterov accelerated subgradient ===
    print(f"\n[P2] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p2_end = min(deadline - 550.0, t_start + 650.0)
    a2, sc2 = _nesterov_subgrad(global_best_a, p2_end)
    if sc2 < global_best_score:
        global_best_a = a2.copy()
        global_best_score = sc2
        print(f"  NEW BEST: {global_best_score:.10f} [nesterov_p2]", flush=True)
    print(f"  P2 done: best={global_best_score:.10f} (t={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 3: Active-set polish with momentum ===
    print(f"\n[P3] Active-set polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p3_end = min(deadline - 450.0, t_start + 850.0)
    a3, sc3 = _active_set_polish(global_best_a, p3_end,
                                  tol_start=0.0025, tol_end=0.00001,
                                  lr_start=0.015, top_k=5, momentum=0.85)
    if sc3 < global_best_score:
        global_best_a = a3.copy()
        global_best_score = sc3
        print(f"  NEW BEST: {global_best_score:.10f} [polish_p3]", flush=True)
    print(f"  P3 done: best={global_best_score:.10f} (t={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_cur = len(global_best_a)
    n_targets = sorted(set([8192, 10240, 12288, 16384, 20480]))
    if n_cur not in n_targets:
        n_targets.insert(min(len(n_targets), n_targets.index(n_cur) if n_cur in n_targets else 2), n_cur)
    n_targets = sorted(set(n_targets))

    for n_try in n_targets:
        if time.monotonic() > deadline - 250.0:
            break
        if n_try == n_cur and abs(n_try - n_cur) < 1:
            continue
        a_init = _resample(global_best_a, n_try)
        remaining = deadline - time.monotonic() - 200.0
        if remaining < 40:
            break
        alloc = min(100.0, remaining / max(1, len(n_targets)))
        a_opt = _run_lbfgs(a_init, time.monotonic() + alloc, min_remaining=10.0)
        sc_opt = _score(a_opt)
        if sc_opt < global_best_score:
            global_best_a = a_opt.copy()
            global_best_score = sc_opt
            print(f"  NEW BEST: {global_best_score:.10f} (n={n_try}) [multi_n]", flush=True)
        if time.monotonic() < deadline - 120.0:
            a_pol, sc_pol = _active_set_polish(
                global_best_a if sc_opt < global_best_score else a_opt,
                min(deadline - 100.0, time.monotonic() + 50.0),
                tol_start=0.002, tol_end=0.00005, lr_start=0.01)
            if sc_pol < global_best_score:
                global_best_a = a_pol.copy()
                global_best_score = sc_pol
                print(f"  NEW BEST: {global_best_score:.10f} (n={n_try}) [multi_n_polish]", flush=True)
    print(f"  P4 done: best={global_best_score:.10f} (t={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 5: Peak shaving + re-optimize ===
    print(f"\n[P5] Peak shaving + re-opt (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p5_end = min(deadline - 200.0, t_start + 1250.0)
    a5, sc5 = _peak_shave(global_best_a, p5_end, n_trials=400)
    if sc5 < global_best_score:
        global_best_a = a5.copy()
        global_best_score = sc5
        print(f"  NEW BEST: {global_best_score:.10f} [peak_shave_p5]", flush=True)

    if time.monotonic() < deadline - 120.0:
        p5b_end = min(deadline - 100.0, time.monotonic() + 80.0)
        a5b = _run_lbfgs(global_best_a, p5b_end, min_remaining=10.0)
        sc5b = _score(a5b)
        if sc5b < global_best_score:
            global_best_a = a5b.copy()
            global_best_score = sc5b
            print(f"  NEW BEST: {global_best_score:.10f} [lbfgs_post_shave]", flush=True)
    print(f"  P5 done: best={global_best_score:.10f} (t={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_final = len(global_best_a)
    restarts = []
    for kind in ["cos_taper", "sin_taper", "gaussian", "raised_cos", "broad_taper"]:
        restarts.append((f"profile_{kind}", _make_profile(n_final, kind)))
    for sig in [0.01, 0.03, 0.06]:
        z = np.log(np.maximum(global_best_a, 1e-30))
        z += sig * rng.standard_normal(n_final)
        z = z - z.mean()
        restarts.append((f"pert_{sig}", _softmax(z)))
    # Fourier init
    restarts.append(("fourier", _fourier_init(n_final, rng, n_modes=30)))

    for label, a_r in restarts:
        if time.monotonic() > deadline - 60.0:
            break
        remaining = deadline - time.monotonic() - 40.0
        if remaining < 20.0:
            break
        alloc = min(35.0, remaining * 0.6)
        a_opt = _run_lbfgs(a_r, time.monotonic() + alloc, min_remaining=5.0)
        sc_opt = _score(a_opt)
        if sc_opt < global_best_score:
            global_best_a = a_opt.copy()
            global_best_score = sc_opt
            print(f"  NEW BEST: {global_best_score:.10f} [restart_{label}]", flush=True)
    print(f"  P6 done: best={global_best_score:.10f} (t={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 7: Local search (annealed) ===
    print(f"\n[P7] Local search (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 40.0:
        p7_end = min(deadline - 30.0, time.monotonic() + 60.0)
        a7, sc7 = _local_search(global_best_a, p7_end, n_trials=5000, sigma_start=0.01)
        if sc7 < global_best_score:
            global_best_a = a7.copy()
            global_best_score = sc7
            print(f"  NEW BEST: {global_best_score:.10f} [local_search_p7]", flush=True)
    print(f"  P7 done: best={global_best_score:.10f} (t={time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 8: Final ultra-fine polishing ===
    print(f"\n[P8] Final ultra-fine polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 5.0:
        a_fin, sc_fin = _active_set_polish(
            global_best_a, deadline - 3.0,
            tol_start=0.0004, tol_end=0.000001,
            lr_start=0.004, top_k=3, momentum=0.7)
        if sc_fin < global_best_score:
            global_best_a = a_fin.copy()
            global_best_score = sc_fin
            print(f"  NEW BEST: {global_best_score:.10f} [final_polish]", flush=True)

        # Tiny L-BFGS at the very end
        remaining = deadline - 3.0 - time.monotonic()
        if remaining > 8.0:
            n = len(global_best_a)
            c_best = _full_convolve(global_best_a, global_best_a)
            mb = float(c_best.max())
            beta_fin = mb * 0.00005
            z = np.log(np.maximum(global_best_a, 1e-30))
            z = z - z.mean()

            def final_obj(z):
                return _lse_objective(z, beta_fin, n)

            def final_grad(z):
                return _lse_gradient(z, beta_fin, n)

            try:
                max_it = min(3000, max(50, int(remaining / 0.04)))
                result = minimize(final_obj, z, jac=final_grad, method='L-BFGS-B',
                                  options={'maxiter': max_it, 'ftol': 1e-19,
                                           'gtol': 1e-16, 'maxcor': 60})
                a_fin2 = _softmax(result.x)
                a_fin2 = np.maximum(a_fin2, 0.0)
                s_fin2 = a_fin2.sum()
                if s_fin2 > 1e-10:
                    a_fin2 = a_fin2 / s_fin2
                    sc_fin2 = _score(a_fin2)
                    if sc_fin2 < global_best_score:
                        global_best_a = a_fin2.copy()
                        global_best_score = sc_fin2
                        print(f"  NEW BEST: {global_best_score:.10f} [final_lbfgs]", flush=True)
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}", flush=True)
    final = np.clip(global_best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final]
