Active-set QP optimizer with FFT-accelerated Hankel matvec, CG-based solves,
progressive tolerance tightening, and multi-n exploration.
Focuses on ultra-precise active-set optimization to minimize max autocorrelation.
"""
import numpy as np

# Global state for avoiding lambda/closures in scipy calls
_g_beta = 0.0
_g_n = 0
_g_best = None


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    L = n + m - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:L]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_mv(v, h, n):
    y = v[::-1]
    L = len(y) + len(h) - 1
    N = _next_pow2(L)
    Fy = np.fft.rfft(y, N)
    Fh = np.fft.rfft(h, N)
    conv = np.fft.irfft(Fy * Fh, N)[:L]
    return conv[n - 1:2 * n - 1]


def _cg_hankel(h, n, rhs, max_iter=2000, tol=1e-15):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hankel_mv(p, h, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        old_rs = rs
        rs = rs_new
        p = r + (rs / old_rs) * p
        if rs < tol * tol * rs0:
            break
    return x


def _activation_matrix(active, n):
    h = np.zeros(2 * n - 1)
    h[active] = 1.0
    return h


def _smooth_obj(z):
    n = _g_n
    beta = _g_beta
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_conv(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z):
    n = _g_n
    beta = _g_beta
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_conv(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _qp_solve_full(h, n, a_init, max_cg=2000, max_outer=50):
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    a = a / s

            oney = float(np.sum(y))

                break
            a_new = y / oney
            if np.any(a_new < -1e-8):

        else:
            idx = np.where(support)[0]
            h_sub = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                if k in set(idx + idx):
                    pass
            grad = _hankel_mv(a, h, n)
            lam = float(np.mean(grad[support]))
            step = 0.5 / (np.max(np.abs(grad)) + 1e-15)
            a_new = a.copy()
            a_new[support] -= step * (grad[support] - lam)
            a_new = np.maximum(a_new, 0)

        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        val_new = float(a_new @ _hankel_mv(a_new, h, n))
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:

            resid = np.max(np.abs(grad[support] - lam))

                resid = max(resid, max(0, viol))

            if not improved:

def _make_profile(n, kind):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos2":
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin2":
        a = np.sin(np.pi * (t + 0.5 / n)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_top":
        w = 0.4
        a = np.ones(n)
        mask = (t < 0.5 - w / 2) | (t > 0.5 + w / 2)
        a[mask] = 0.5 * (1 + np.cos(np.pi * (t[mask] - 0.5 - np.sign(t[mask] - 0.5) * w / 2) / (0.5 - w / 2)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    if best_seq is not None and time.monotonic() < deadline - 10:
        dq = deadline - 5
        a7, sc7 = _active_qp_polish(best_seq, dq, max_rounds=100)
        _update(a7, sc7, "final_qp")

    # Phase 8: Final flatten if time
    if best_seq is not None and time.monotonic() < deadline - 5:
        a8, sc8 = _flatten_polish(best_seq, deadline - 3, max_iters=100)
        _update(a8, sc8, "final_flatten")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "cos2")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
