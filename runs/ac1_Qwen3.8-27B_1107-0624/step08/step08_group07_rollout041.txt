```python
def compute_load(a, top_lags):
    n = len(a)
    # L[j] = sum_{k in top_lags} a[k-j] for valid k-j
    # This is the cross-correlation of a with the indicator of top_lags
    # indicator[k] = 1 if k in top_lags, 0 otherwise
    indicator = np.zeros(2*n-1)
    indicator[top_lags] = 1
    # L[j] = sum_k indicator[k] * a[k-j]
    # = cross_correlation(indicator, a)[?]
    # Actually, this is simpler to compute directly for small K
    L = np.zeros(n)
    for k in top_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        L[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    return L
```

For K = 100 and n = 8192, this takes about 100 * 8192 = 819K operations per iteration. That's fast.

```python
import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]
```

```python
"""
Optimized step function search using balanced peak-valley multiplicative updates,
smoothed subgradient with adaptive temperature, 1D coordinate descent exploiting
quadratic structure, and structured restarts.
Key innovations: (1) balanced load balancing using both peaks AND valleys,
(2) adaptive temperature for smoothed gradient, (3) exact 1D subproblem solving,
(4) multi-scale restart strategy.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_load_vector(a, lags):
    """Compute load[j] = sum_{k in lags} a[k-j] for valid indices."""
    n = len(a)
    load = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            load[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    return load


def _balanced_multiplicative(a_init, deadline, max_iter=200000,
                              eta0=0.05, eta_decay=0.99995,
                              K_start=3, K_end=150,
                              check_interval=200):
    """Phase 1: Balanced peak-and-valley multiplicative updates."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta0
    t = 0

    while t < max_iter and time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / max_iter)
        K = max(2, int(K_start + (K_end - K_start) * frac))

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        if t % check_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Find top-K and bottom-K lags
        sorted_idx = np.argsort(c)
        top_K = sorted_idx[-K:]
        bottom_K = sorted_idx[:K]

        # Compute loads
        peak_load = _compute_load_vector(a, top_K)
        valley_load = _compute_load_vector(a, bottom_K)

        # Balanced update: reduce high-peak-load coords, boost high-valley-load
        balance = valley_load / max(K, 1) - peak_load / max(K, 1)
        # Scale balance to have reasonable magnitude
        bal_scale = np.abs(balance).max()
        if bal_scale > 1e-15:
            balance = balance / bal_scale

        a = a * np.exp(eta * balance)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        eta *= eta_decay

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _smoothed_subgradient(a_init, deadline, max_iter=100000,
                           T_start=None, T_end=None,
                           lr_start=0.01, lr_end=0.001,
                           active_k=5, check_interval=100):
    """Phase 2: Smoothed subgradient in log-space with adaptive temperature."""
    n = len(a_init)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _fft_convolve(a_init, a_init)
    if T_start is None:
        T_start = float(c0.max()) * 0.01
    if T_end is None:
        T_end = T_start * 0.01

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr_mult = 1.0
    patience = 0

    while t < max_iter and time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.7) * lr_mult

        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Smoothed gradient via softmax weighting
        c_shifted = (c - c_max) / max(T, 1e-15)
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s

        # Compute gradient: g[m] = sum_k p[k] * 2*a[k-m]
        # Use only top active peaks for efficiency
        sorted_idx = np.argsort(c)[::-1]
        num_active = min(active_k + 5, len(sorted_idx))
        g = np.zeros(n)
        for idx in range(num_active):
            k = sorted_idx[idx]
            if p[k] < 1.0 / len(c):
                break
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += p[k] * 2.0 * a[k - hi:k - lo + 1][::-1]

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Log-space gradient with centering
        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Check and adapt
        if t % check_interval == 0:
            a_check = _softmax(z)
            sc = _score_norm(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                patience = 0
                lr_mult = min(lr_mult * 1.05, 3.0)
            else:
                patience += 1
                if patience > 5:
                    lr_mult *= 0.7
                    patience = 0
                    if lr_mult < 0.1:
                        # Reset from best
                        z = np.log(np.maximum(best_a, 1e-15))
                        z = z - z.mean()
                        z += 0.005 * np.random.randn(n)
                        z = z - z.mean()
                        m_buf = np.zeros(n)
                        v_buf = np.zeros(n)
                        lr_mult = 0.5

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=200):
    """Phase 3: 1D coordinate descent with exact subproblem awareness."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        improved = False

        # Focus on the peak region
        k_peak = int(np.argmax(c))
        lo = max(0, k_peak - n + 1)
        hi = min(k_peak, n - 1)
        span = hi - lo + 1

        # Subsample if too many coordinates
        if span > 200:
            step = max(1, span // 200)
            indices = list(range(lo, hi + 1, step))
        else:
            indices = list(range(lo, hi + 1))

        np.random.shuffle(indices)

        for m in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[m]
            if old_val < 1e-15:
                continue

            best_x = old_val
            best_local = c_max

            # Try candidate values (geometric spread)
            candidates = [old_val * f for f in [0.3, 0.5, 0.7, 1.4, 2.0, 3.0]]
            candidates.append(0.0)

            for x in candidates:
                a[m] = x
                s = a.sum()
                if s < 1e-15:
                    a[m] = old_val
                    continue
                c_new = _fft_convolve(a / s, a / s)
                sc = float(c_new.max())
                if sc < best_local:
                    best_local = sc
                    best_x = x
            a[m] = old_val

            if best_x != old_val:
                a[m] = best_x
                a = a / a.sum()
                sc_full = _score_norm(a)
                if sc_full < best_score:
                    best_score = sc_full
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try broader sweep
            any_improve = False
            for m in range(0, n, max(1, n // 100)):
                if time.monotonic() > deadline:
                    break
                old_val = a[m]
                if old_val < 1e-15:
                    continue
                for x in [old_val * 0.5, old_val * 1.5, 0.0]:
                    a[m] = x
                    s = a.sum()
                    if s < 1e-15:
                        a[m] = old_val
                        continue
                    c_new = _fft_convolve(a / s, a / s)
                    sc = float(c_new.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-10):
                        best_score = sc
                        a = a / s
                        best_a = a.copy()
                        any_improve = True
                        improved = True
                        break
                if not improved:
                    a[m] = old_val
            if not any_improve:
                break

    return best_a, _score_norm(best_a)


def _flat_top_optimize(a_init, deadline, max_iter=50000):
    """Try to equalize the top peaks by targeted adjustments."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = 0.002

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Find peaks that are within 0.5% of max
        tol = c_max * 0.995
        active = np.where(c >= tol)[0]
        if len(active) < 2:
            break
        if len(active) > 10:
            order = np.argsort(c[active])[::-1][:10]
            active = active[order]

        # Find the valley (minimum in the middle region)
        mid_region = c[n // 4:3 * n // 4]
        valley_idx = n // 4 + int(np.argmin(mid_region))

        # Compute gradient that reduces peaks and fills valley
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        # Add valley-filling term
        lo_v = max(0, valley_idx - n + 1)
        hi_v = min(valley_idx, n - 1)
        if lo_v <= hi_v:
            g[lo_v:hi_v + 1] -= 0.5 * a[valley_idx - hi_v:valley_idx - lo_v + 1][::-1]

        # Project to tangent space and apply
        W = float(np.dot(a, g))
        g_tan = g - W * a
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break

        a_new = a - lr * g_tan / g_norm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_new /= s

        sc_new = _score_norm(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.01)
        else:
            lr *= 0.95
            if lr < 1e-9:
                break

    return best_a, best_score
```
