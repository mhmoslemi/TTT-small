        # Switch to L-BFGS for finishing touches
        if time.time() - start_time + (evaluate_sequence([x.tolist()]) if _next_pow2 else None) < budget - 100:
            return minimize(score_func, result.x, method='L-BFGS-B', options={'disp': True, 'max_iter': 50})
        else:
            return result
    
    # --- Actual Search ---
    try:
        a_vector = np.log(a_init + 1e-8) if a_init[0] > 0 else np.ones(n_seq)
        # Check if we have evaluation time remaining
        deadline = time.time() + float(budget_s)
        # Primary optimization: cone with increasing similarity
        # We'll use a hybrid method with multiple restarts
        best_score = score_func(a_init)
        best_vector = a_init.copy()
        restarts = min(5, int(budget_s // 100))
        current_vector = a_vector.copy()
        for iter_num in range(restarts):
            if time.time() > deadline:
                break
            # Compute gradient / envelope for current state
            a_arr = np.exp(current_vector)
            conv_val = score_func(a_arr)
            grad = np.gradient(conv_val)  # Not full convolution, but safe for large n_seq
            # Backtracking along gradient directions
            alpha = min(0.001, max(5e-6, 0.01 * (1/iter_num))) if iter_num > 0 else 0.001
            new_vector = current_vector - alpha * grad
            new_vector = normalize_safe(np.exp(new_vector))  # Apply exponential then normalize
            # Check direct improvement
            new_score = score_func(new_vector)
            if new_score < best_score * (1 - 1e-4):
                print(f"Iteration {iter_num}: score improved to {new_score}")
                best_vector = new_vector
                best_score = new_score * (1 - 1e-5)
            else:
                # Revert and try a different strategy: simplex
                # This restart mechanism is different
                print("Restarting with simplex...")
                result_simplex = direct_search(current_vector, budget_s - iter_num*100)
                new_score = score_func(result_simplex.x)
                best_score = min(best_score, new_score)
                best_vector = result_simplex.x
        # Final normalization and transition to exponential space
        best_vector = best_vector.clip(0, 1000)
        best_vector = normalize_safe(np.exp(best_vector))
        # Re-evaluate to ensure
        final_score = score_func(best_vector)
        print(f"Optimized score: {final_score}")
        # Prepare for return: convert list
        return list(best_vector)
    except Exception as e:
        print(f"Error during optimization: {str(e)}")
        return list(provided_parent) if provided_parent is not None else list(np.ones(100)/100)
