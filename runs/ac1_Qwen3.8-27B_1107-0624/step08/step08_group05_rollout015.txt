"""
Optimized step function autocorrelation minimizer.
Uses adaptive active-set Newton with progressive tolerance tightening,

import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _log_space(a):
    z = np.log(np.maximum(a, 1e-30))
    return z - z.mean()

# ============================================================
# LSE objective and gradient
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val

def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    sum_cp = np.sum(np.power(c_safe, p))
    factor = 2.0 * n * sum_cp ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _peak_reduction(a_init, deadline, n_rounds=200, n_peaks=5, strength=0.01):
    """Directly reduce peaks by modifying contributing elements."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        peak_thresh = c[sorted_idx[-1]]

        # For each peak, find contributing pairs and reduce them
        delta = np.zeros(n)
        for pk in sorted_idx:
            weight = c[pk] / c_max  # prioritize highest peaks
            # Contributions to c[pk]: a[i] * a[pk-i] for valid i
            i_min = max(0, pk - (n - 1))
            i_max = min(n - 1, pk)
            for i in range(i_min, i_max + 1):
                j = pk - i
                if 0 <= j < n:
                    contrib = a[i] * a[j]
                    if contrib > 0:
                        # Reduce proportionally
                        delta[i] += weight * contrib / (a[i] + 1e-30) * strength
                        if i != j:
                            delta[j] += weight * contrib / (a[j] + 1e-30) * strength

        a_new = a - delta
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc:
            a = a_new
            best_a = a_new.copy()
            best_sc = sc_new
        else:
            # Try smaller step
            a_new2 = a - 0.5 * delta
            a_new2 = np.maximum(a_new2, 0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:
                a_new2 = a_new2 / s_new2
                sc_new2 = _score(a_new2)
                if sc_new2 < best_sc:
                    a = a_new2
                    best_a = a_new2.copy()
                    best_sc = sc_new2
                else:
                    strength *= 0.9
            else:
                strength *= 0.9

    return best_a, best_sc

    base_beta = max(gap * 0.3, max_c * 0.003)

    z = _log_space(current_a)
    betas = base_beta * np.array([1.0, 0.6, 0.35, 0.2, 0.12, 0.07,
                                   0.04, 0.02, 0.01, 0.005, 0.002,
                                   0.001, 0.0005, 0.0002])
    betas = betas[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = _log_space(a_new)
        except Exception:
            pass

    return current_a, current_sc

def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = _log_space(current_a)
    p_values = [2, 4, 8, 16, 32, 64, 128, 256, 512]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(200, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = _log_space(a_new)
        except Exception:
            pass

    return current_a, current_sc

# Subgradient with adaptive step
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = _log_space(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    no_improve = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                no_improve = 0
            else:
                no_improve += 1
                if no_improve % 3 == 0:
                    lr *= 0.9
                if no_improve > 40:
                    z = _log_space(best_a)
                    patience = 0
                    no_improve = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_log(a, sigma, rng):
    z = _log_space(a)
    z = z + sigma * rng.standard_normal(len(a))
    return _softmax(z)

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: exploration, LSE/power-sum L-BFGS,
    adaptive Newton active-set, direct peak reduction, and multi-n final check.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (90s) ===
    print(f"\n[Phase 1] Diverse exploration (90s)")
    expl_deadline = min(deadline, t_start + 90)
    expl_ns = [2048, 4096, 8192]
    if n_parent > 0:
        expl_ns.append(n_parent)
    expl_ns = sorted(set(expl_ns))

    for n_ex in expl_ns:
        for kind in ["uniform", "tapered", "cos2", "wheatstone", "triangle"]:
            if time.monotonic() > expl_deadline:
                break
            a_try = _make_profile(n_ex, kind)
            sc_try = _score(a_try)
            upd(a_try, sc_try, f"expl_{kind}_{n_ex}")

    # Quick L-BFGS on best from exploration
    if time.monotonic() < deadline - 600 and _best_seq is not None:
        a_q, sc_q = _lbfgs_lse(_best_seq.copy(), min(deadline - 550, time.monotonic() + 40), min_remaining=10, n_stages=4)
        upd(a_q, sc_q, "expl_quick_lbfgs")

    # === Phase 2: Main L-BFGS LSE (400s) ===
    print(f"\n[Phase 2] Main L-BFGS LSE (400s)")
    if time.monotonic() < deadline - 400:
        p2_dead = min(deadline - 380, time.monotonic() + 400)
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), p2_dead, min_remaining=30, n_stages=14)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 3: Power-sum refinement (200s) ===
    print(f"\n[Phase 3] Power-sum L-BFGS (200s)")
    if time.monotonic() < deadline - 200:
        p3_dead = min(deadline - 185, time.monotonic() + 200)
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), p3_dead, min_remaining=20)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 4: Adaptive Newton active-set (250s) ===
    print(f"\n[Phase 4] Adaptive Newton active-set (250s)")
    if time.monotonic() < deadline - 200:
        # Progressive tolerance tightening
        p4_dead = min(deadline - 185, time.monotonic() + 250)
        tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005]
        a_cur = _best_seq.copy()
        sc_cur = _score(a_cur)
        per_phase = (p4_dead - time.monotonic()) / len(tol_schedule)
        for tol_f in tol_schedule:
            phase_dead = time.monotonic() + max(10, per_phase)
            a_nw, sc_nw = _newton_active_set(a_cur, phase_dead, max_iter=100, tol_frac=tol_f, min_active=1)
            if sc_nw < sc_cur:
                a_cur = a_nw
                sc_cur = sc_nw
            upd(a_cur, sc_cur, f"newton_tol{tol_f}")

    # === Phase 5: Top-k Newton (150s) ===
    print(f"\n[Phase 5] Top-k Newton (150s)")
    if time.monotonic() < deadline - 130:
        p5_dead = min(deadline - 120, time.monotonic() + 150)
        a_topk, sc_topk = _newton_topk(_best_seq.copy(), p5_dead, max_iter_per_k=30)
        upd(a_topk, sc_topk, "newton_topk")

    # === Phase 6: Direct peak reduction (100s) ===
    print(f"\n[Phase 6] Direct peak reduction (100s)")
    if time.monotonic() < deadline - 80:
        p6_dead = min(deadline - 70, time.monotonic() + 100)
        a_pk, sc_pk = _peak_reduction(_best_seq.copy(), p6_dead, n_rounds=300, n_peaks=3, strength=0.005)
        upd(a_pk, sc_pk, "peak_reduction")

    # === Phase 7: Newton + subgradient alternating (150s) ===
    print(f"\n[Phase 7] Newton + SG alternating (150s)")
    if time.monotonic() < deadline - 120:
        a_cur = _best_seq.copy()
        for rnd in range(4):
            if time.monotonic() > deadline - 100:
                break
            nw_dead = min(deadline - 100, time.monotonic() + 25)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=50, tol_frac=0.001)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_nw_{rnd}")
            sg_dead = min(deadline - 100, time.monotonic() + 25)
            a_sg2, sc_sg2 = _subgrad_polish(a_cur, sg_dead, n_steps=2000, lr_init=0.004, tol_frac=0.0005)
            if sc_sg2 < _score(a_cur):
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_sg_{rnd}")

    # === Phase 8: Multi-n exploration (200s) ===
    print(f"\n[Phase 8] Multi-n exploration (200s)")
    n_candidates = [4096, 8192, 12288, 16384, 20480, 24576]
    if n_parent not in n_candidates and 1024 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        # LSE on this n
        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, deadline, min_remaining=30, n_stages=6)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        # Newton polish
        if time.monotonic() < deadline - 25:
            nw_dead = min(deadline - 15, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=60, tol_frac=0.001)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 9: Diverse restarts (100s) ===
    print(f"\n[Phase 9] Diverse restarts (100s)")
    if time.monotonic() < deadline - 50:
        restarts = [
            ("perturb_log_003", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("perturb_fourier", lambda: _perturb_fourier_phase(_best_seq, rng, n_modes=6, max_amp=0.3)),
            ("perturb_amp", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.06)),
            ("perturb_log_005", lambda: _perturb_log(_best_seq, 0.05, rng)),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=15, n_stages=5)
            upd(a_r, sc_r, f"restart_{label}")
            if time.monotonic() < deadline - 15:
                nw_dead = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=30, tol_frac=0.002)
                upd(a_r2, sc_r2, f"restart_{label}_nw")

    # === Phase 10: Final aggressive polish (80s) ===
    print(f"\n[Phase 10] Final aggressive polish (80s)")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        a_fin = _best_seq.copy()
        sc_fin = _score(a_fin)

        # Ultra-low beta LSE
        remaining = deadline - 10 - time.monotonic()
        if remaining > 15:
            c_fin = _full_convolve(a_fin, a_fin)
            mb = float(c_fin.max())
            z_fin = _log_space(a_fin)
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-17,
                             'gtol': 1e-15, 'maxcor': 80, 'maxls': 200}
                )
                a_ultra = _softmax(result.x)
                sc_ultra = _score(a_ultra)
                upd(a_ultra, sc_ultra, "final_ultra_lse")
                if sc_ultra < sc_fin:
                    a_fin = a_ultra
                    sc_fin = sc_ultra
            except Exception:
                pass

        # Final Newton with very tight tolerance
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_active_set(a_fin, deadline - 3, max_iter=100, tol_frac=0.0003, min_active=1)
            upd(a_fin2, sc_fin2, "final_newton_tight")

        # Final top-1 Newton
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _newton_topk(a_fin, deadline - 2, k_list=[1, 2, 3], max_iter_per_k=40)
            upd(a_fin3, sc_fin3, "final_topk")

        # Final subgradient with very tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _subgrad_polish(_best_seq, deadline - 1, n_steps=5000, lr_init=0.003, tol_frac=0.0003)
            upd(a_fin4, sc_fin4, "final_sg_tight")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print("FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
