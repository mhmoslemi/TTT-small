**IFT (Iterative Fourier Transform) / Gerchberg-Saxton:**

The standard GS algorithm alternates between:
1. Enforcing the magnitude constraint in the frequency domain
2. Enforcing the support constraint (non-negativity) in the time domain

For our problem:
- Frequency domain constraint: `|A[k]|^2` should be flat (i.e., `|A[k]| = constant` for `k > 0`)
- Time domain constraint: `a >= 0` and `sum(a) = 1`

The algorithm:
```
a = init
for it in range(max_iter):
    A = rfft(a, N)
    # Flatten the spectrum
    mag = |A|
    for k in 1..N/2:
        A[k] = A[k] / mag[k] * target_mag
    A[0] = A[0]  # keep DC
    a_new = irfft(A, N)[:n].real
    a_new = max(a_new, 0)
    a_new = a_new / sum(a_new)
    a = a_new
```

The `target_mag` can be set to the geometric mean of the current non-DC magnitudes, or to a fixed value.

Fixing `target_mag` to the geometric mean means we're preserving the total "energy" in the spectrum (approximately), while flattening it.

Let me also consider a "soft" version where we don't fully flatten but compress:
```
A[k] = A[k] / mag[k] * (mag[k])^alpha  # for alpha < 1, this compresses
     = mag[k]^(alpha-1) * A[k]
```

Wait, that's `A[k] * mag[k]^(alpha-1) = A[k] / mag[k]^(1-alpha)`. For `alpha < 1`, this reduces the dynamic range.

Actually, let me just use the standard GS with the geometric mean target, and also try a "soft" version.

Now, one issue with IFT: it might converge to a local minimum or even oscillate. To mitigate this, I can:
1. Use a cooling schedule for the "flattening strength"
2. Combine with gradient steps
3. Use multiple random starts

Let me now write the complete code. I'll focus on making it robust and efficient.

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses IFT (Gerchberg-Saxton), FISTA, and L-BFGS with adaptive smoothing.
Key innovation: frequency-domain spectrum flattening combined with 
time-domain gradient optimization, with multi-start strategy.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_1d(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _ift_step(a, N, alpha=1.0, n_iter=1):
    """Single IFT (Gerchberg-Saxton) step: flatten spectrum and project to non-negative."""
    n = len(a)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    # Target: geometric mean of non-DC, non-Nyquist magnitudes
    if N % 2 == 0:
        non_dc = slice(1, N // 2)
    else:
        non_dc = slice(1, N // 2 + 1)
    mags = mag[non_dc]
    mags_pos = mags[mags > 1e-30]
    if len(mags_pos) == 0:
        return a.copy()
    target = np.exp(np.mean(np.log(mags_pos)))
    # Apply: scale magnitudes towards target
    for it in range(n_iter):
        A_new = A.copy()
        A_new[non_dc] = A[non_dc] * (target / np.abs(A[non_dc]))**alpha
        A_new = (A_new + np.conj(A_new[::-1])) / 2  # ensure Hermitian... actually rfft handles this
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
        # Recompute for next iteration
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mags = mag[non_dc]
        mags_pos = mags[mags > 1e-30]
        if len(mags_pos) > 0:
            target = np.exp(np.mean(np.log(mags_pos)))
    return a

def _ift_optimize(a_init, deadline, n_iter_per_step=1, alpha=0.5):
    """Run IFT until deadline."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    if a.sum() < 1e-15:
        return a
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    count = 0
    while time.monotonic() < deadline:
        t0 = time.monotonic()
        a_new = _ift_step(a, N, alpha=alpha, n_iter=n_iter_per_step)
        sc = _score(a_new)
        if sc < best_sc:
            best_a = a_new.copy()
            best_sc = sc
        a = a_new
        count += 1
        if count % 50 == 0:
            print(f"  IFT iter {count}: score={best_sc:.10f}, t={time.monotonic()-t0:.3f}s/step")
    return best_a, best_sc

def _sob_smoothing(a, beta):
    """Compute S-plus log-sum-exp smoothing of max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _sob_gradient(a, beta):
    """Gradient of the S-plus smoothing."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: (2n/beta) * sum_k w[k] * dc[k]/da[j]
    # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
    # So grad[j] = (2n/beta) * 2 * sum_k w[k] * a[k-j] * 1_{0<=k-j<n}
    #            = (4n/beta) * h[j]
    # where h[j] = sum_k w[k] * a[k-j] (cross-correlation of w and a at lag j)
    # h = conv(w_rev, a)[n-1:2*n-1][::-1]
    w_rev = w[::-1]
    wc = _full_convolve(w_rev, a)
    h = wc[n-1:2*n-1][::-1]
    grad = (4.0 * n / beta) * h
    return grad, c_max

def _fista_smooth(a_init, beta, deadline, L0=None):
    """FISTA with S-plus smoothing for fixed beta."""
    n = len(a_init)
    a = a_init.copy()
    if a.sum() < 1e-15:
        return a
    a = a / a.sum()
    
    # Estimate L (Lipschitz constant of gradient)
    # grad = (4n/beta) * h where h = conv(w_rev, a)[...]
    # The operator norm of the gradient map is roughly (4n/beta) * 2 * max(a)
    # Actually, let's use a simpler bound: L = 8*n*max(a)/beta
    if L0 is None:
        L = 8.0 * n * max(a.max(), 1.0/n) / beta
    else:
        L = L0
    
    y = a.copy()
    t_param = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    count = 0
    stall = 0
    
    while time.monotonic() < deadline:
        # Compute gradient at y
        grad, c_max = _sob_gradient(y, beta)
        
        # Gradient step
        z = y - grad / L
        a_new = _proj_simplex(z)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = a_new + ((t_param - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_param = t_new
        
        count += 1
        if count % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if count % 1000 == 0:
                print(f"    FISTA beta={beta:.6f} iter={count}: score={best_sc:.10f}")
            if stall > 50:
                break
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, best_sc

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with multi-scale beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        # Objective and gradient in log-space
        def obj_fn(z_val):
            a_val = _softmax_1d(z_val)
            c_val = _full_convolve(a_val, a_val)
            c_max = float(c_val.max())
            shifted = (c_val - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def grad_fn(z_val):
            a_val = _softmax_1d(z_val)
            c_val = _full_convolve(a_val, a_val)
            c_max = float(c_val.max())
            shifted = (c_val - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c_val)
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s
            a_rev = a_val[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_val))
            # Grad of obj w.r.t. z: chain rule
            # d(obj)/d(a) * d(a)/d(z)
            # d(a)/d(z)[j,k] = a[j]*(delta[jk] - a[k])
            # d(obj)/d(a)[j] = (4n/beta) * h[j] ... 
            # Actually let me use the softmax-conv formulation
            # h[j] = sum_k w[k] * a_rev[...]*a[...][j]
            # Let me just use the direct formula
            W_total = float(np.dot(w, c_val))
            grad_a = 4.0 * n * a_val * (h - W_total)
            # Chain rule for softmax: d(loss)/dz = a * (grad_a - sum(a*grad_a))
            correction = float(np.dot(a_val, grad_a))
            return a_val * (grad_a - correction)
        
        est_time_per_iter = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 10))))
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 40,
                    'maxls': 30,
                }
            )
            a_new = _softmax_1d(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.6f}: score={current_sc:.10f}")
        except Exception as e:
            print(f"    L-BFGS beta={beta:.6f} failed: {e}")
    
    return current_a

def _make_ift_init(n, rng, n_ift_iters=200):
    """Create an initial sequence by running IFT from a random phase."""
    N = _next_pow2(2 * n - 1)
    # Random phases
    phases = rng.standard_normal(N // 2 + 1)
    # Target magnitude: uniform (flat spectrum)
    target_mag = np.ones(N // 2 + 1)
    target_mag[0] = np.sqrt(n)  # DC component (sum of a = 1, so A[0] = 1... 
    # Actually, for zero-padded FFT: A[0] = sum(a) = 1
    # And |A[k]| for k>0 should be roughly 1/sqrt(2n) for flat |A|^2
    # Since sum(|A[k]|^2) = N * sum(a[j]^2) and for uniform a, sum(a^2) = 1/n
    # So sum(|A|^2) = N/n, and |A[k]|^2 ≈ N/(n*(N-1)) ≈ 1/n for large N
    # So |A[k]| ≈ 1/sqrt(n) for non-DC bins
    target_mag[1:] = 1.0 / np.sqrt(n)
    
    A = target_mag * np.exp(1j * phases)
    A[0] = 1.0  # DC = sum(a) = 1
    
    a = np.fft.irfft(A, N)[:n].real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    
    # Run IFT to refine
    for _ in range(n_ift_iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mags = mag[1:]
        mags_pos = mags[mags > 1e-30]
        if len(mags_pos) == 0:
            break
        target = np.exp(np.mean(np.log(mags_pos)))
        A_new = A.copy()
        A_new[1:] = A[1:] * (target / mag[1:])
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a
