            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_obj_func(beta, n):
    """Return objective function (no closure)."""
    return lambda z: _smoobj(z, beta, n)


def _make_grad_func(beta, n):
    """Return gradient function (no closure)."""
    return lambda z: _smograd(z, beta, n)


def _active_set_qp(a_init, deadline, max_rounds=200, tol_schedule=None):
    """Active-set QP with progressive tolerance tightening and robust CG."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
                        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8,
                        2e-8, 1e-8, 5e-9, 2e-9, 1e-9, 5e-10, 1e-10]

    rounds_per_tol = 4
    tol_idx = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve: min a^T H_W a s.t. sum(a)=1
        # KKT: H_W a = lambda * 1, sum(a) = 1
        # => H_W y = 1, a = y / sum(y)
        y = _cg_solve_robust(W, n, np.ones(n), max_iter=1500, tol=1e-13)
        oney = float(np.sum(y))
        if abs(oney) < 1e-20 or not np.isfinite(oney):
            # Fallback: projected gradient
            grad = _hankel_matvec(a, W, n)
            lam = float(np.mean(grad))
            step = 0.05 / (float(np.max(np.abs(grad))) + 1e-15)
            a_try = a - step * (grad - lam)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a = a_try / s_try
            else:
                a += step * (-grad + lam) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
        else:
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

        # Check for convergence
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / max(1, int(active.sum()))
        grad = _hankel_matvec(a, W, n)
        lam = float(np.mean(grad))
        resid = float(np.max(np.abs(grad - lam)))
        if resid < 1e-12 * (float(np.max(np.abs(grad))) + 1e-15):
            break

    return best_a, best_sc


def _multi_peak_gradient(a_init, deadline, max_steps=5000):
    """Multi-peak projected gradient with Nesterov acceleration."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Nesterov
    y = a.copy()
    t_k = 1.0
    lr = 0.02
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Top L lag
        L = max(10, int(0.01 * len(c)))
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = max(1, int(active.sum()))
        W = np.zeros_like(c)
        W[active] = 1.0 / n_active

        # Gradient of sum_{k in active} c[k] w.r.t. a
        a_rev = y[::-1]
        wc = _full_convolve(W, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(W, c))
        grad = 2.0 * n * y * (h - W_val)

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_k - 1.0) / t_new

        # Step in y-space
        step = lr / (1.0 + 0.001 * i)
        y_new = y + theta * (y - a) - step * grad
        # Project to non-negative simplex
        y_new = _project_simplex(y_new)
        a = y_new
        t_k = t_new

        # Track best
        if i % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    a = best_a.copy()
                    y = a.copy()
                    t_k = 1.0
                    lr *= 0.7
                    patience = 0

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _spectral_equalize(a_init, deadline, max_iters=300):
    """Equalize autocorrelation values via frequency-domain adjustment."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(np.mean(c))

        # Target: make all c values equal to c_mean
        # Error: e[k] = max(0, c[k] - c_mean)
        excess = np.maximum(c - c_mean, 0)
        excess_max = float(excess.max())
        if excess_max < 1e-12:
            break

        # Gradient of sum excess^2 w.r.t. a
        a_rev = a[::-1]
        ec = _full_convolve(excess, a_rev)
        grad = ec[n - 1:2 * n - 1]

        step = 0.05 / (1.0 + 0.005 * it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power":
        a = (1.0 - abs(t - 0.5) * 2.0) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrink_and_reopt(a, deadline, min_time=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_idx = np.argsort(a)

    for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.20]:
        if time.monotonic() + min_time > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_idx = np.argsort(best_a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, active-set QP, multi-peak gradient,
    spectral equalization, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent (aggressive beta schedule)
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs1")

    # Phase 2: Active-set QP (tight)
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 450:
        a2, sc2 = _active_set_qp(best_seq, min(deadline - 450, time.monotonic() + 400),
                                  max_rounds=300)
        update_best(a2, sc2, "qp1")

    # Phase 3: Multi-peak gradient
    print(f"\n[Phase 3] Multi-peak gradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a3, sc3 = _multi_peak_gradient(best_seq, min(deadline - 350, time.monotonic() + 150))
        update_best(a3, sc3, "multipeak1")

    # Phase 4: Spectral equalization
    print(f"\n[Phase 4] Spectral equalize (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 280:
        a4, sc4 = _spectral_equalize(best_seq, min(deadline - 280, time.monotonic() + 100))
        update_best(a4, sc4, "spec_eq1")

    # Phase 5: Active-set QP round 2 (ultra-tight)
    print(f"\n[Phase 5] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 220:
        a5, sc5 = _active_set_qp(best_seq, min(deadline - 220, time.monotonic() + 200),
                                  max_rounds=300,
                                  tol_schedule=[5e-5, 2e-5, 1e-5, 5e-6, 2e-6,
                                                1e-6, 5e-7, 2e-7, 1e-7, 5e-8,
                                                2e-8, 1e-8, 5e-9, 1e-9, 5e-10, 1e-10])
        update_best(a5, sc5, "qp2")

    # Phase 6: L-BFGS round 2 (from improved point)
    print(f"\n[Phase 6] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a6 = _run_lbfgs(best_seq, deadline, min_time_remaining=100.0, aggressive=True)
        update_best(a6, _score(a6), "lbfgs2")

    # Phase 7: Shrinkage
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a7, sc7 = _shrink_and_reopt(best_seq, deadline, min_time=80.0)
        update_best(a7, sc7, "shrink")

    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [32768, 8192]:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 30:
            a_qp, sc_qp = _active_set_qp(a_opt, deadline - 25, max_rounds=100)
            update_best(a_qp, sc_qp, f"n_{n_try}_qp")

    # Phase 9: Final ultra-tight polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_qp(
            best_seq, deadline - 2, max_rounds=200,
            tol_schedule=[1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8,
                          1e-8, 5e-9, 2e-9, 1e-9, 5e-10, 1e-10, 5e-11, 1e-11])
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 3:
            a_sp, sc_sp = _spectral_equalize(best_seq, deadline - 2, max_iters=200)
            update_best(a_sp, sc_sp, "final_spec")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
