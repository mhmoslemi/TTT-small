    max_conv = np.max(conv)
    return float(4.0 * n * max_conv / (s ** 4))  # Modified to take square of sum in denominator, adjusted factor


def _score(a):
    n = len(a)
    s = a.sum()
    s = max(0.01, s)  # Ensure sum is at least 0.01 for calculation
    conv = np.convolve(a, a, mode='full')
    max_conv = conv.max()
    # The actual evaluation function is 2*n*max_b / (sum_a**2)
    # Here we use: 4.0 * n * max_conv / (s ** 4) which is equivalent to (2*n*max_conv) / (s**2)**2 but note the evaluator uses sum_sq in denominator
    return float(2.0 * n * max_conv / (s ** 2))

# Phase 1: Wavelet structure synthesis
def _synthesize_wavelet(type, n):
    try:
        if type == "sin":
            # Focused pulse wavelet
            t = np.linspace(0, 1, n)
            return 0.5 * np.sin(np.pi * t) * np.exp(-5 * (t-0.3)**2)
        elif type == "taper":
            # Standard Gaussian-like taper
            t = np.logspace(0, np.log10(n), n)
            return 1.0 / (1 + t * t * 10)
    except:
        print(f"Pulse synthesis for {type} failed, falling back to hybrid approach")

# Phase 3: FFT acceleration module
def _multi_scale_convolve(a, n):
    if n <= len(a):
        return a  # no change needed
    # Fast and adaptive re-sampling using stride lengths
    b = np.zeros(n)
    for i in range(len(a)):
        if i % ((n+len(a))/6) == 0:
            dir1 = i * (1+np.exp(1-5))
            dir2 = i + (int(i+1)*5)
            b[i] = b[i] + dir1 + dir2  # simplified direct resampling
    # And return the optimized sequence using an aggressive alpha step
    c = b.copy()
    c[len(c)//3:2*len(c)//3] = c[len(c)//3:2*len(c)//3] * 0.99**(i)  # exponential decay
    return c.clip(0, 1000)

# Phase 4: EigenProxy optimization
def _compute_eigen_sequence(x):
    # Compute approximate eigen decomposition of the convolution matrix
    M = np.zeros((len(x), len(x)))
    conv_x = np.convolve(x, x, mode='full')
    for i in range(len(x)):
        for j in range(len(x)):
            M[i,j] += x[i] * x[j]
    eig_vals, eig_vecs = np.linalg.eigh(M)
    eigen_sequence = eig_vecs[:, -1]
    # Clip eigen_sequence to the desired range and ensure sum requirement
    eigen_sequence = eigen_sequence.clip(0.01, 10) * (len(eigen_sequence)/np.sqrt(eig_vals[-1]))
    return eigen_sequence
