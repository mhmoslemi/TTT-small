def _adam_optimize(a_init, deadline, min_remaining=20.0):
    """Adam optimizer with LSE smoothing and annealing."""
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c_max = c.max()
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < 1e-15:
        gap = c_max * 0.001
    beta_init = max(gap * 5, c_max * 0.01)
    beta_final = c_max * 0.00001
    
    best_a = a.copy()
    best_score = _score(a)
    
    total_iters = 50000  # cap
    start_time = time.monotonic()
    
    while time.monotonic() < deadline - min_remaining:
        # Anneal beta and lr
        progress = min(1.0, (time.monotonic() - start_time) / max(1.0, deadline - min_remaining - start_time))
        beta_lse = beta_init * (beta_final / beta_init) ** progress
        lr = lr0 * (1.0 - 0.5 * progress)  # gentle decay

        z = z - z.mean()  # remove the gauge freedom

Let me also think about a key question: **Is it better to optimize at the parent's n or at a larger n?**

The score is 2n * max(conv(a,a)) / (sum(a))^2. If we resample from n1 to n2 > n1, the score changes because:
- The "2n" factor increases
- The max of conv might decrease (more "room" to spread out)
- The sum stays the same (if we normalize)

Typically, larger n can give better scores because the autocorrelation can be "smoother" (more degrees of freedom). But the "2n" factor works against us.

The optimal trade-off depends on how fast max(conv) decreases with n. If max(conv) ~ C/n^alpha, then score ~ 2n * C/n^alpha / 1 = 2C * n^(1-alpha). For alpha > 1, larger n is better. For alpha < 1, smaller n is better.

For a "flat" autocorrelation (ideal), max(conv) ~ 1/(2n), so score ~ 2n * 1/(2n) = 1. This is independent of n!

For uniform, max(conv) = 1/n, score = 2. Also independent of n.

So the choice of n mainly affects how close we can get to the ideal, not the asymptotic behavior. Larger n gives more "dial knobs" to turn, so it should help in practice.

"""
Optimized step-function search using Adam-based LSE-smoothed optimization,
active-set polishing, and multi-n exploration. Key innovation over previous
approach: uses Adam (adaptive per-parameter learning rates) instead of L-BFGS,
which is faster per iteration and more robust in high dimensions. Combines with
active-set refinement and diverse restarts to target sub-1.5027 scores.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj_and_grad(z, beta, n):
    """Compute LSE-smoothed objective and gradient w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    
    # Gradient computation
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * a * (h - W)
    
    # Convert to grad_z via softmax Jacobian
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z


def _adam_optimize(a_init, deadline, min_remaining=15.0, lr0=0.05, 
                   beta1=0.9, beta2=0.999, eps=1e-8, noise_std=0.0):
    """Adam optimizer with LSE-smoothed objective and temperature annealing."""
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Determine initial LSE beta
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted)-1)]
    if gap < c_max * 0.0001:
        gap = c_max * 0.0001
    beta_init = max(gap * 3.0, c_max * 0.005)
    beta_final = c_max * 0.000005
    
    best_a = a.copy()
    best_score = _score(a)
    
    start_time = time.monotonic()
    total_duration = max(1.0, deadline - min_remaining - start_time)
    iter_count = 0
    
    while time.monotonic() < deadline - min_remaining:
        progress = min(1.0, (time.monotonic() - start_time) / total_duration)
        
        # Anneal LSE temperature (exponential decay)
        beta_lse = beta_init * (beta_final / beta_init) ** progress
        
        # Learning rate: cosine decay
        lr = lr0 * 0.5 * (1.0 + np.cos(np.pi * progress))
        
        obj, grad_z = _lse_obj_and_grad(z, beta_lse, n)
        
        # Add noise for exploration (early phase only)
        if noise_std > 0 and progress < 0.3:
            grad_z = grad_z + noise_std * np.random.standard_normal(n)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > top_k * 10:
            top_indices = np.argsort(c)[::-1][:top_k]
            active = np.zeros_like(c, dtype=bool)

            n_active = top_k

        grad = 2.0 * h

        gnorm = np.linalg.norm(grad_t)

            break

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "quadratic_residues":
        # Use a prime close to n
        p = n
        while p > 1:
            is_prime = True
            for d in range(2, int(p**0.5) + 1):
                if p % d == 0:
                    is_prime = False
                    break
            if is_prime:
                break
            p -= 1
        if p == 1 or p < 10:
            return np.ones(n) / n
        # Create pattern based on quadratic residues mod p
        a = np.zeros(n)
        for i in range(p):
            if i == 0:
                continue
            if (i * i) % p < p // 2:
                a[i % n] = 1.0
            else:
                a[i % n] = 0.3
        a = a + 0.1  # small uniform background
        a = np.maximum(a, 0.0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)

        else:
            no_improve += 1

        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > top_k * 10:
            top_indices = np.argsort(c)[::-1][:top_k]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = top_k

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False

        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if no_improve % 10 == 0:
                tol = max(tol_end, tol * 0.6)
                lr *= 0.5
            if no_improve >= max_stall:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.4)
                lr = lr_start * 0.3
                no_improve = 0
                stall += 1
                if stall > 10:
                    break
        else:
            if tol > tol_end:
                tol *= 0.97
            lr = min(lr_start, lr * 1.02)

    return best_a, _compute_score_fast(best_a)


def _peak_balance(a_init, deadline, n_iter=200):
    """Try to equalize the top autocorrelation values via targeted adjustments."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _compute_score_fast(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find top contributors to the peak
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo >= k_hi:
            break

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Reduce top 20% contributors, add to bottom 20%
        n_items = len(i_range)
        n_reduce = max(1, n_items // 5)
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]
        bottom_idx = i_range[np.argsort(contrib)[:n_reduce]]

        reduction = 0.0
        for idx in top_idx:
            red = a[idx] * 0.05
            a[idx] -= red
            reduction += red

        a[bottom_idx] += reduction / len(bottom_idx)
        a = np.maximum(a, 0.0)
        s_try = a.sum()
        if s_try < 1e-10:
            a = best_a.copy()
            continue
        a = a / s_try

        sc = _compute_score_fast(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        else:
            a = best_a.copy()

    return best_a, _compute_score_fast(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "modulated":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: Adam LSE (exploration) → Active-set (exploitation) →
    Multi-n → Diverse restarts → Final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _compute_score_fast(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _compute_score_fast(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Adam on parent (main exploration) ===
    print(f"\n[P1] Adam on parent n={n_parent} t={time.monotonic()-t_start:.1f}s")
    p1_deadline = min(deadline - 850.0, t_start + 450.0)
    lr1 = 0.06 * np.sqrt(12288.0 / n_parent)
    a1, sc1, iters1 = _adam_optimize(best_a, p1_deadline, lr0=lr1,
                                      noise_std=0.005, restart_every=50000)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} (iters={iters1}) [adam_p1]")
    else:
        print(f"  No improvement: {sc1:.10f} vs {best_score:.10f} (iters={iters1})")

    # === Phase 2: Active-set polish ===
    print(f"\n[P2] Active-set polish t={time.monotonic()-t_start:.1f}s")
    p2_deadline = min(deadline - 550.0, t_start + 700.0)
    a2, sc2 = _active_set_refine(best_a, p2_deadline, tol_start=0.003,
                                  tol_end=0.00001, lr_start=0.015)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [polish_p2]")
    else:
        print(f"  No improvement: {sc2:.10f}")

    # === Phase 3: Second Adam with lower LR (finer search) ===
    print(f"\n[P3] Adam round 2 (fine) t={time.monotonic()-t_start:.1f}s")
    p3_deadline = min(deadline - 400.0, t_start + 900.0)
    lr3 = 0.015 * np.sqrt(12288.0 / len(best_a))
    a3, sc3, iters3 = _adam_optimize(best_a, p3_deadline, lr0=lr3,
                                      beta1=0.95, beta2=0.9999, restart_every=80000)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} (iters={iters3}) [adam_p3]")

    # === Phase 4: Peak balance ===
    print(f"\n[P4] Peak balance t={time.monotonic()-t_start:.1f}s")
    p4_deadline = min(deadline - 250.0, t_start + 1050.0)
    a4, sc4 = _peak_balance(best_a, p4_deadline, n_iter=500)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [peak_balance]")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration t={time.monotonic()-t_start:.1f}s")
    n_best = len(best_a)
    n_targets = [16384, 20480, 8192]
    for n_try in n_targets:
        if time.monotonic() > deadline - 120.0:
            break
        a_init = _resample(best_a, n_try)
        sc_init = _compute_score_fast(a_init)
        print(f"  n={n_try}, init={sc_init:.10f}")
        remaining = deadline - time.monotonic() - 80.0
        if remaining < 25:
            continue
        time_budget = min(120.0, remaining)
        p5_dl = time.monotonic() + time_budget
        lr5 = 0.05 * np.sqrt(12288.0 / n_try)
        a_opt, sc_opt, _ = _adam_optimize(a_init, p5_dl, lr0=lr5)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_adam]")
        if time.monotonic() < deadline - 50.0:
            a_pol, sc_pol = _active_set_refine(a_opt, min(deadline - 35.0, time.monotonic() + 40.0),
                                                tol_start=0.002, tol_end=0.00005, lr_start=0.01)
            if sc_pol < best_score:
                best_a = a_pol.copy()
                best_score = sc_pol
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish]")

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 50.0:
        n_r = len(best_a)
        restarts = []
        for kind in ["tapered", "raised_cos", "modulated"]:
            restarts.append((kind, _make_profile(n_r, kind)))
        restarts.append(("pert_01", _perturb(best_a, 0.02, rng)))
        restarts.append(("pert_05", _perturb(best_a, 0.06, rng)))

        for label, a_r in restarts:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 15.0:
                break
            lr_r = 0.05 * np.sqrt(12288.0 / n_r)
            a_opt, sc_opt, _ = _adam_optimize(a_r, deadline - 20.0, lr0=lr_r)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}]")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_refine(best_a, deadline - 2.0,
                                           tol_start=0.0002, tol_end=0.000002,
                                           lr_start=0.004, top_k=3, max_stall=100)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

        remaining = deadline - 2.0 - time.monotonic()
        if remaining > 8.0:
            lr_fin = 0.002 * np.sqrt(12288.0 / len(best_a))
            a_fin2, sc_fin2, _ = _adam_optimize(best_a, deadline - 1.0, lr0=lr_fin,
                                                beta1=0.95, beta2=0.9999)
            if sc_fin2 < best_score:
                best_a = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_adam]")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
