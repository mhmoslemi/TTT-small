        mod = 1.0 + 0.25 * np.cos(2.0 * np.pi * 3.0 * t)
        a = env * mod + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "uniform":
        return np.ones(n) / n
    elif kind == "bimodal_mod":
        env = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.05
        mod = 1.0 + 0.3 * np.cos(2.0 * np.pi * 4.0 * t)
        a = env * mod
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        idx = np.arange(1, n + 1).astype(float)
        a = 1.0 / np.sqrt(idx)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(99)
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _peak_shave_targeted(a, deadline, n_trials=500):
    """Targeted peak shaving: reduce specific elements contributing most to the peak."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find elements contributing most to c[k_star]
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        valid = (j_range >= 0) & (j_range < n)
        i_range = i_range[valid]
        j_range = j_range[valid]

        if len(i_range) == 0:
            continue

        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            continue

        # Reduce top contributors, add to low ones
        n_reduce = max(1, int(len(contrib) * 0.1))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.05
            a_try[idx] -= red
            reduction += red

        # Distribute to elements NOT in the contributing set
        contrib_set = set()
        for idx in top_idx:
            contrib_set.add(int(idx))
            jv = k_star - int(idx)
            if 0 <= jv < n:
                contrib_set.add(jv)

        low_mask = np.ones(n, dtype=bool)
        for idx in contrib_set:
            low_mask[idx] = False

        if low_mask.sum() == 0:
            continue

        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer targeting score < 1.5027.
    Phases:
    1. Load/initialize from parent
    2. L-BFGS refined optimization
    3. Active-set momentum
    4. Peak shaving
    5. Multi-n exploration
    6. Structured restarts
    7. KKT polishing
    8. Final micro-opt
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}", flush=True)
    else:
        n_parent = 12288
        best_a = _make_structured_init(n_parent, "envelope_mod")
        best_score = _score(best_a)
        print(f"No parent, structured init n={n_parent}, score={best_score:.10f}", flush=True)

    # === Phase 1: L-BFGS refined on parent ===
    print(f"\n[P1] L-BFGS refined parent (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p1_deadline = min(deadline - 1000.0, t_start + 400.0)
    a1 = _run_lbfgs_refined(best_a, p1_deadline, min_remaining=30.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [lbfgs_p1] t={time.monotonic()-t_start:.1f}s", flush=True)
    else:
        # Keep original if L-BFGS didn't help
        print(f"  No improvement: {sc1:.10f} vs {best_score:.10f}", flush=True)

    # === Phase 2: Active-set momentum ===
    print(f"\n[P2] Active-set momentum (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p2_deadline = min(deadline - 700.0, t_start + 650.0)
    a2, sc2 = _active_set_momentum(best_a, p2_deadline, tol_start=0.006, tol_end=0.00001, lr_start=0.015)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [momentum_p2] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 3: Peak shaving ===
    print(f"\n[P3] Targeted peak shaving (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p3_deadline = min(deadline - 500.0, t_start + 800.0)
    a3, sc3 = _peak_shave_targeted(best_a, p3_deadline, n_trials=400)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [peak_shave] t={time.monotonic()-t_start:.1f}s", flush=True)

    # Re-optimize after peak shaving
    if time.monotonic() < deadline - 350:
        p3b_deadline = min(deadline - 300.0, t_start + 900.0)
        a3b = _run_lbfgs_refined(best_a, p3b_deadline, min_remaining=20.0)
        sc3b = _score(a3b)
        if sc3b < best_score:
            best_a = a3b.copy()
            best_score = sc3b
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shave] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_best = len(best_a)
    n_targets = [16384, 20480, 24576, 32768, 11264, 13312]
    # Add parent n if not already there
    if n_parent not in n_targets and n_parent != n_best:
        n_targets.append(n_parent)
    # Sort by priority (closer to n_best first, then larger)
    n_targets = sorted(set(nt for nt in n_targets if nt != n_best),
                       key=lambda x: abs(x - n_best))

    for n_try in n_targets[:5]:
        if time.monotonic() > deadline - 200.0:
            break
        remaining = deadline - time.monotonic() - 50.0
        if remaining < 60:
            break

        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}, trying... (t={time.monotonic()-t_start:.1f}s)", flush=True)

        # Quick L-BFGS
        n_dl = min(deadline - 100.0, time.monotonic() + remaining * 0.4)
        a_opt = _run_lbfgs_refined(a_init, n_dl, min_remaining=10.0)
        sc_opt = _score(a_opt)

        # Quick momentum
        if time.monotonic() < deadline - 100.0:
            m_dl = min(deadline - 80.0, time.monotonic() + remaining * 0.3)
            a_opt2, sc_opt2 = _active_set_momentum(a_opt, m_dl, tol_start=0.004, tol_end=0.0001, lr_start=0.01, top_k_max=40)
            if sc_opt2 < sc_opt:
                a_opt = a_opt2
                sc_opt = sc_opt2

        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            n_best = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s", flush=True)
        else:
            print(f"  n={n_try}: best={sc_opt:.10f} (no improvement)", flush=True)

    # === Phase 5: Structured restarts ===
    print(f"\n[P5] Structured restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 100.0:
        kinds = ["envelope_mod", "envelope_mod2", "gaussian_mod", "bimodal_mod", "tapered", "random_smooth"]
        for kind in kinds:
            if time.monotonic() > deadline - 60.0:
                break
            remaining = deadline - time.monotonic() - 30.0
            if remaining < 30:
                break

            a_r = _make_structured_init(n_best, kind)
            # Quick optimize
            r_dl = min(deadline - 40.0, time.monotonic() + remaining * 0.5)
            a_opt = _run_lbfgs_refined(a_r, r_dl, min_remaining=10.0)
            sc_opt = _score(a_opt)

            if time.monotonic() < deadline - 40.0:
                a_opt2, sc_opt2 = _active_set_momentum(a_opt, min(deadline - 30.0, time.monotonic() + 30.0),
                                                       tol_start=0.003, tol_end=0.0001, lr_start=0.01, top_k_max=30)
                if sc_opt2 < sc_opt:
                    a_opt = a_opt2
                    sc_opt = sc_opt2

            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{kind}] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: KKT polishing ===
    print(f"\n[P6] KKT polishing (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 5.0:
        a_kkt, sc_kkt = _kkt_polish(best_a, min(deadline - 3.0, time.monotonic() + 30.0))
        if sc_kkt < best_score:
            best_a = a_kkt.copy()
            best_score = sc_kkt
            print(f"  NEW BEST: {best_score:.10f} [kkt_polish] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 7: Final L-BFGS micro-opt ===
    if time.monotonic() < deadline - 3.0:
        remaining = deadline - time.monotonic() - 2.0
        if remaining > 5.0:
            n = len(best_a)
            c_best = _full_convolve(best_a, best_a)
            mb = float(c_best.max())
            beta_fin = mb * 5e-6
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
            prob = _LseProblem(beta_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 60, 'maxls': 80})
                a_fin = _softmax(result.x)
                a_fin = np.maximum(a_fin, 0.0)
                s_fin = a_fin.sum()
                if s_fin > 1e-10:
                    a_fin = a_fin / s_fin
                    sc_fin = _score(a_fin)
                    if sc_fin < best_score:
                        best_a = a_fin.copy()
                        best_score = sc_fin
                        print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s", flush=True)
            except Exception as e:
                print(f"  Final L-BFGS error: {e}", flush=True)

    # Final validation
    print(f"\n{'='*60}", flush=True)
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256).tolist()
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)

    return [float(x) for x in final]
