    # To compute cross_corr(v, active) for all i simultaneously:
    # cross_corr(v, active)[i] = sum_j v[j] * active[j+i]
    # = ifft(fft(v) * conj(fft(active)))[i] ... NO, that's the other way.
    # 
    # Actually: cross_corr(v, active)[i] = sum_j v[j] * conj(active)[j - i] (if we use the standard definition)
    # Hmm, definitions vary. Let me just use the convolution relationship.
    #
    # cross_corr(v, active)[i] = sum_j v[j] * active[j+i] = conv(v, active_rev)[i + (len(active)-1)]
    # where active_rev = active[::-1]
    #
    # So: result[i] = 2 * conv(v, active_rev)[i + (2n-2)] + active[2i] * v[i]
    # 
    # In the FFT code:
    # fa = fft(v_rev) where v_rev = v[::-1]
    # fb = fft(active)
    # r = ifft(fa * fb) = conv(v_rev, active)
    # r has length n + (2n-1) - 1 = 3n
    # r[k] = sum_j v_rev[j] * active[k-j] = sum_j v[n-1-j] * active[k-j]
    # Let m = n-1-j: r[k] = sum_m v[m] * active[k - n + 1 + m]
    # = sum_m v[m] * active[m + (k - n + 1)]
    # = cross_corr(v, active)[k - n + 1]
    #
    # So cross_corr(v, active)[i] = r[i + n - 1]
    #
    # Therefore: (Hv)_i = 2 * r[i + n - 1] + active[2i] * v[i]
    #
    # The previous code computes: r[n-1 : 2n-1] which gives cross_corr(v, active)[0 : n]
    # And multiplies by 2. So it computes 2 * cross_corr(v, active)[0:n]
    # Which is 2 * sum_j v[j] * active[j+i] for i = 0, ..., n-1
    #
    # This matches our formula (Hv)_i = 2 * sum_j active[i+j] * v_j (ignoring the diagonal correction)
    #
    # So the previous code is correct (ignoring the small diagonal correction active[2i]*v[i]).
    # For large n, this correction is negligible.
    pass
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fv = np.fft.rfft(v_rev, N)
    fa = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fv * fa, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _cg_solve(hess_func, b, tol=1e-12, max_iter=200):
    """Conjugate gradient solver for the Newton step."""
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p_vec = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p_vec)
        pH = float(np.dot(p_vec, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p_vec
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p_vec = r + beta_cg * p_vec
        rs = r_new
    return x

"""
Step function optimizer minimizing autocorrelation peak using:
- Newton active-set with trust-region and adaptive tolerance
- Nesterov-accelerated exact subgradient (no surrogate bias)
- Randomized coordinate descent on 1D subproblems
- Multi-n exploration with Fourier-domain interconnection

import numpy as np

# ============================================================
# Core utilities
# ============================================================

def _npo2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _fconv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _npo2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _auto(a):
    return _fconv(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _auto(a)
    return float(2.0 * n * c.max() / (s * s))


def _norm1(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# Hessian-vector product for active set
# ============================================================

def _hvp(v, active, n):
    N = _npo2(4 * n)
    v_rev = v[::-1]
    fv = np.fft.rfft(v_rev, N)
    fa = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fv * fa, N)
    return 2.0 * r[n - 1:2 * n - 1]

# Conjugate gradient
# ============================================================

def _cg(hfunc, b, tol=1e-12, max_it=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(r @ r)
    bn = float(np.linalg.norm(b))
    if bn < 1e-30:
        return x
    for _ in range(max_it):
        Hp = hfunc(p)
        pH = float(p @ Hp)
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x += alpha * p
        r -= alpha * Hp
        r2 = float(r @ r)
        if np.sqrt(r2) < tol * bn:
            break
        beta = r2 / rs if rs > 1e-30 else 0.0
        p = r + beta * p
        rs = r2
    return x

def _newton(a0, dl, mx=300, tf=0.002):
    n = len(a0)
    a = _norm1(a0.copy())
    best_a, best_sc = a.copy(), _score(a)
    for it in range(mx):
        if time.monotonic() > dl:
            break
        c = _auto(a)
        cm = float(c.max())
        if cm < 1e-30:
            break
        tol = max(tf * cm, 1e-14)
        act = c >= (cm - tol)
        if act.sum() == 0:
            act[int(c.argmax())] = True
        g = _hvp(a, act, n)
        gt = g - np.mean(g)
        gn = float(np.linalg.norm(gt))
        if gn < 1e-14:
            break
        af = act.astype(np.float64)
        mu = 1e-6
        dx = _cg(lambda v: _hvp(v, af, n) + mu * v, -gt, tol=1e-12, max_it=300)
        dx -= np.mean(dx)
        fo = float(c[act].sum())
        alpha = 1.0
        a_new = a.copy()
        for _ in range(60):
            at = a + alpha * dx
            at = np.maximum(at, 0)
            st = float(at.sum())
            if st < 1e-15:
                alpha *= 0.5
                continue
            at /= st
            ct = _auto(at)
            ft = float(ct[act].sum())
            if ft < fo - 1e-10 * alpha * gn:
                a_new = at
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc

def _coord_desc(a0, dl, n_iter=50, batch=2000):
    n = len(a0)
    a = _norm1(a0.copy())
    best_a, best_sc = a.copy(), _score(a)
    rng = np.random.default_rng(12345)
    for it in range(n_iter):
        if time.monotonic() > dl:
            break
        c = _auto(a)
        idx = rng.choice(n, size=min(batch, n), replace=False)
        improved = False
        for i in idx:
            if time.monotonic() > dl:
                break
            lo, hi = i, min(i + n - 1, 2 * n - 2)
            seg = c[lo:hi + 1].copy()
            ai = a[i]
            B = seg - 2.0 * ai * a[:min(n - i, i) + 1]
            W = 2.0 * a[:min(n - i, i) + 1]
            W = np.concatenate([W, np.zeros(hi - lo - len(W) + 1)])[:hi - lo + 1]
            rest_max = float(np.delete(c, range(lo, hi + 1)).max()) if hi - lo < 2 * n - 2 else 0.0
            lo2, hi2 = 0.0, max(4.0 * ai, 4.0 / n)
            for _ in range(30):
                mid = (lo2 + hi2) / 2.0
                vals = B + W * mid
                v2 = mid * mid
                fv = max(float(vals.max()), v2, rest_max)
                g1 = W[vals == vals.max()]
                sub = np.concatenate([g1, [2.0 * mid]])
                if sub.max() < -1e-12:
                    lo2 = mid
                elif sub.min() > 1e-12:
                    hi2 = mid
                else:
                    break
            x_star = (lo2 + hi2) / 2.0
            if x_star > 1e-10:
                a[i] = x_star
                s = a.sum()
                if s > 1e-15:
                    a /= s
                    improved = True
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
        if not improved and it > 10:
            a = best_a.copy()
    return best_a, best_sc

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _auto(a)
    cx = float(c.max())
    sh = (c - cx) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cx
    return 2.0 * n * (cx + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _auto(a)
    cx = float(c.max())
    sh = (c - cx) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w = e / s
    wc = _fconv(w, a[::-1])
    h = wc[n - 1:2 * n - 1]
    W = float(w @ c)
    ga = 2.0 * n * (h - W)
    return a * (ga - float(a @ ga))


def _lbfgs_lse(a0, dl, min_rem=60):
    n = len(a0)
    ca = _norm1(a0.copy())
    sc = _score(ca)
    c = _auto(ca)
    mx = float(c.max())
    if mx < 1e-15:
        return ca, sc
    cs = np.sort(c)[::-1]
    gap = cs[0] - cs[min(50, len(cs) - 1)]
    bb = max(gap * 0.3, mx * 0.005)
    z = np.log(np.maximum(ca, 1e-30))
    z -= z.mean()
    betas = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002]
    for bf in betas:
        if time.monotonic() + min_rem > dl:
            break
        rem = dl - min_rem - time.monotonic()
        if rem < 5:
            break
        beta = bb * bf
        mi = min(3000, max(100, int(rem / 0.02)))
        try:
            res = minimize(_lse_obj, z, args=(beta, n), jac=_lse_grad,
                           method='L-BFGS-B',
                           options={'maxiter': mi, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxcor': 40})
            an = _softmax(res.x)
            sn = _score(an)
            if sn < sc - 1e-15:
                ca, sc = an, sn
                z = np.log(np.maximum(an, 1e-30))
                z -= z.mean()
        except Exception:
            pass
    return ca, sc

def _resize(a, nt):
    x = np.linspace(0, 1, len(a))
    xn = np.linspace(0, 1, nt)
    at = np.interp(xn, x, a)
    at = np.maximum(at, 0)
    s = at.sum()
    if s < 1e-10:
        return np.ones(nt) / nt
    return at / s

def _perturb_fourier(a, rng, amp=0.3):
    A = np.fft.rfft(a)
    pp = rng.uniform(-amp, amp, len(A))
    pp[0] = 0.0
    An = A * np.exp(1j * pp)
    an = np.fft.irfft(An, len(a))
    an = np.maximum(an, 0)
    s = an.sum()
    if s < 1e-10:
        return a.copy()
    return an / s

    z = np.log(np.maximum(a, 1e-30))

        a7, s7 = _lbfgs_lse(ps, dl, min_rem=10)
        up(a7, s7, f"rst{i}")
        if s7 < _best_score * 1.002 and time.monotonic() < dl - 15:
            ndl = min(dl - 10, time.monotonic() + 20)
            a7b, s7b = _newton(a7, ndl, mx=60, tf=0.002)
            up(a7b, s7b, f"rst{i}_nwt")

    # Phase 8: Final aggressive Newton
    print("\n[8] Final Newton")
    if _best_seq is not None and time.monotonic() < dl - 5:
        final_dl = max(time.monotonic() + 5, dl - 3)
        a8, s8 = _newton(_best_seq, final_dl, mx=500, tf=0.0001)
        up(a8, s8, "final_nwt")
        if time.monotonic() < dl - 2:
            a8b, s8b = _subgrad_nest(_best_seq, dl - 1, ns=5000, lr0=0.002, tf=0.0005)
            up(a8b, s8b, "final_sg")

    # Return
    print(f"\n{'='*60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            print("FALLBACK uniform")
        else:
            sc_final = _score(final)
            print(f"FINAL: score={sc_final:.10f} n={len(final)} t={time.monotonic()-t0:.1f}s")
        return [float(x) for x in final]
    fb = np.ones(256) / 256
    print(f"ULTIMATE FALLBACK: {_score(fb):.10f}")
    return [float(x) for x in fb]
