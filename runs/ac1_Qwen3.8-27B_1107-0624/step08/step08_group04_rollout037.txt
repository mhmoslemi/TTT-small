def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-12):

    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)

            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

def _make_asymmetric_profile(n, kind="ramp", param=0.3):
    t = np.linspace(0, 1, n)
    if kind == "ramp":
        # Linear ramp: more weight on one side
        a = t + 0.01
        return a / a.sum()
    elif kind == "exp_ramp":
        # Exponential ramp
        a = np.exp(param * n * t)
        return a / a.sum()
    elif kind == "gauss_offset":
        # Gaussian centered at 'param' * n
        mu = param
        sigma = 0.15
        a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_gauss":
        # Two Gaussians at different positions
        a = np.exp(-20 * (t - 0.3) ** 2) + 0.5 * np.exp(-30 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "power_law":
        # Power law: a ~ (i+1)^(-beta)
        beta = param
        a = (np.arange(n) + 1) ** (-beta)
        return a / a.sum()
    elif kind == "asym_sin":
        # Asymmetric sine: sin shifted
        phase = param
        a = np.sin(np.pi * t + phase * np.pi) ** 2 + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sawtooth":
        # Periodic sawtooth
        freq = 3
        a = ((t * freq) % 1.0) + 0.01
        return a / a.sum()
    elif kind == "sqrt_ramp":
        a = np.sqrt(t + 0.01)
        return a / a.sum()
    elif kind == "inv_ramp":
        a = (1 - t) + 0.01
        return a / a.sum()
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "uniform":
        return np.ones(n) / n
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.2:
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _obj_func(z):
    return _smoobj(z, _gstate['beta'], _gstate['n'])


def _grad_func(z):
    return _smograd(z, _gstate['beta'], _gstate['n'])


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.0008)

    if aggressive:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078,
                  0.0039, 0.00195, 0.00098, 0.00049, 0.00024, 0.00012,
                  0.00006, 0.00003, 0.000015, 0.000008, 0.000004, 0.000002,
                  0.000001, 0.0000005, 0.00000025, 0.0000001]
    else:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                  0.00002, 0.00001]
    betas = [base_beta * r for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    _gstate['n'] = n

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        _gstate['beta'] = beta
        est_iter_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(4000, max(80, int(remaining / (est_iter_time * 20))))

        try:
            result = minimize(
                _obj_func,
                z,
                jac=_grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, max_rounds=100):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.003, 0.0015, 0.0007, 0.0003, 0.00015, 0.00007,
                    0.00003, 0.000015, 0.000007, 0.000003, 0.000001,
                    0.0000005, 0.0000002, 0.0000001]

    tol_idx = 0
    rounds_per_tol = 5

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        n_active = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_mean = g.mean()
        g_t = g - g_mean

        Hg = _cg_solve(W, n, g, max_iter=400)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=400)
        H1_sum = float(np.sum(H1))

        if abs(H1_sum) > 1e-20 and np.isfinite(H1_sum):
            lam = float(np.sum(Hg)) / H1_sum
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
        else:
            gn = float(np.linalg.norm(g_t))
            dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        improved = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not improved:
            # Try a smaller fixed step
            for trial_alpha in [0.3, 0.1, 0.05, 0.01]:
                a_try = a + trial_alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < _score(a) - 1e-16:
                    a = a_try
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a.copy()
                    break

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _momentum_subgrad(a_init, deadline, max_steps=8000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.06
    momentum = 0.85
    velocity = np.zeros(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol_frac = max(0.0005, 0.005 * (1.0 - i / float(max_steps)))
        active = c >= (c_max - tol_frac * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0005 * i)
        velocity = momentum * velocity + (1.0 - momentum) * grad
        z_new = z - step * velocity
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if i % 15 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    velocity = np.zeros(n)
                    patience = 0
                    lr *= 0.75
                    momentum = min(0.97, momentum + 0.03)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=300):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_mean = float(np.sum(c) / len(c))
        c_max = float(c.max())

        # Target peaks significantly above mean
        threshold = c_mean + 0.3 * (c_max - c_mean)
        excess = np.maximum(c - threshold, 0)
        excess_max = float(excess.max())
        if excess_max < 1e-15:
            break

        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]
        grad_part = grad_part - grad_part.mean()

        step = 0.08 / (1.0 + 0.003 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            for frac in [0.5, 0.25, 0.1]:
                a_try = a - frac * step * grad_part
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < _score(a) - 1e-16:
                    a = a_try
                    break

    return best_a, best_sc


def _make_profile(n, kind, param1=0.0, param2=0.0):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = t + 0.01
        return a / a.sum()
    elif kind == "inv_ramp":
        a = (1.0 - t) + 0.01
        return a / a.sum()
    elif kind == "exp_ramp":
        a = np.exp(param1 * n * t) if param1 != 0 else np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "gauss_offset":
        mu = param1 if param1 != 0 else 0.3
        sigma = param2 if param2 != 0 else 0.15
        a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_gauss":
        a = np.exp(-25 * (t - 0.3) ** 2) + 0.5 * np.exp(-35 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "asym_sin":
        phase = param1 if param1 != 0 else 0.3
        p = param2 if param2 != 0 else 2.0
        a = (np.sin(np.pi * t + phase * np.pi) ** 2 + 0.01) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        beta = param1 if param1 != 0 else 0.5
        a = (np.arange(n, dtype=float) + 1) ** (-beta)
        return a / a.sum()
    elif kind == "modulated_gauss":
        # Gaussian envelope * cosine modulation
        mu = 0.5
        sigma = 0.25
        freq = param1 if param1 != 0 else 2.0
        a = np.exp(-0.5 * ((t - mu) / sigma) ** 2) * (1.0 + 0.5 * np.cos(2 * np.pi * freq * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic":
        # Cubic profile: a = (t-t^2)(1-t+t^2) type shapes
        a = t * (1 - t) * (4 * t - 1) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "shifted_taper":
        # Tapered but shifted: more mass on one side
        shift = param1 if param1 != 0 else 0.2
        a = np.sin(np.pi * np.clip(t + shift * (1 - t), 0, 1)) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_reopt(a, deadline, min_time_remaining=40.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.03, 0.06, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer exploiting non-symmetry of optimal solutions.
    Uses diverse asymmetric profiles, adaptive L-BFGS smoothing,
    active-set QP, momentum subgradient, and flattening polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Diverse profile exploration ===
    print(f"\n[Phase 1] Diverse asymmetric profiles (t={time.monotonic() - t_start:.1f}s)")
    n_base = n_parent
    candidates = []

    # Key asymmetric profiles
    profile_specs = [
        ("ramp", 0, 0),
        ("inv_ramp", 0, 0),
        ("exp_ramp", 2.0, 0),
        ("exp_ramp", -2.0, 0),
        ("gauss_offset", 0.25, 0.12),
        ("gauss_offset", 0.35, 0.15),
        ("gauss_offset", 0.6, 0.18),
        ("gauss_offset", 0.4, 0.10),
        ("asym_sin", 0.3, 2.0),
        ("asym_sin", 0.5, 1.5),
        ("asym_sin", 0.7, 2.5),
        ("power_law", 0.3, 0),
        ("power_law", 0.7, 0),
        ("power_law", 1.0, 0),
        ("two_gauss", 0, 0),
        ("modulated_gauss", 2.0, 0),
        ("modulated_gauss", 3.0, 0),
        ("modulated_gauss", 5.0, 0),
        ("shifted_taper", 0.2, 0),
        ("shifted_taper", 0.4, 0),
        ("cubic", 0, 0),
        ("tapered", 0, 0),
    ]

    for kind, p1, p2 in profile_specs:
        if time.monotonic() > deadline - 1200:
            break
        prof = _make_profile(n_base, kind, p1, p2)
        sc = _score(prof)
        candidates.append((sc, prof, f"{kind}_{p1}_{p2}"))

    # Also try perturbing the parent
    if parent is not None:
        for sigma in [0.05, 0.1, 0.15]:
            pert = _perturb(parent, sigma, rng)
            sc = _score(pert)
            candidates.append((sc, pert, f"perturb_{sigma}"))

    # Sort by score, keep top ones
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best={candidates[0][0]:.10f} [{candidates[0][2]}]")

    # Take top 5 (including parent if it's good)
    top_candidates = candidates[:5]
    # Make sure parent is included
    if parent is not None:
        parent_sc = _score(parent)
        if parent_sc < top_candidates[-1][0]:
            top_candidates[4] = (parent_sc, parent, "parent")
        elif parent is not None and parent_sc < 1.51:
            top_candidates.append((parent_sc, parent, "parent"))
            top_candidates.sort(key=lambda x: x[0])

    # === Phase 2: L-BFGS optimization on top candidates ===
    print(f"\n[Phase 2] L-BFGS optimization (t={time.monotonic() - t_start:.1f}s)")
    n_opt = min(4, len(top_candidates))
    time_per_opt = min(400.0, (deadline - time.monotonic() - 600.0) / max(1, n_opt))

    for idx in range(n_opt):
        if time.monotonic() > deadline - 500:
            break
        sc_init, prof_init, label = top_candidates[idx]
        print(f"  L-BFGS on candidate {idx} (init score={sc_init:.10f}, {label}), "
              f"budget={time_per_opt:.0f}s")
        t_phase_start = time.monotonic()
        remaining_for_phase = deadline - 500 - time.monotonic()
        alloc = min(time_per_opt, remaining_for_phase / (n_opt - idx))
        if alloc < 30:
            break
        a_opt = _run_lbfgs(prof_init, t_phase_start + alloc, min_time_remaining=40.0, aggressive=True)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"lbfgs_{idx}_{label}")
        print(f"    Result: score={sc_opt:.10f}, time={time.monotonic() - t_phase_start:.1f}s")

    # === Phase 3: Active-set QP polish ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        qp_deadline = min(deadline - 300, time.monotonic() + 350)
        a_qp, sc_qp = _active_set_polish(best_seq, qp_deadline, max_rounds=120)
        update_best(a_qp, sc_qp, "active_qp_1")

    # === Phase 4: Flatten polish ===
    print(f"\n[Phase 4] Flatten polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        fl_deadline = min(deadline - 200, time.monotonic() + 120)
        a_fl, sc_fl = _flatten_polish(best_seq, fl_deadline, max_iters=300)
        update_best(a_fl, sc_fl, "flatten_1")

    # === Phase 5: Another active-set QP round (tighter) ===
    print(f"\n[Phase 5] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        qp2_deadline = min(deadline - 150, time.monotonic() + 180)
        a_qp2, sc_qp2 = _active_set_polish(best_seq, qp2_deadline, max_rounds=100)
        update_best(a_qp2, sc_qp2, "active_qp_2")

    # === Phase 6: Momentum subgradient ===
    print(f"\n[Phase 6] Momentum subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        sg_deadline = min(deadline - 80, time.monotonic() + 120)
        a_sg, sc_sg = _momentum_subgrad(best_seq, sg_deadline, max_steps=6000)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 7: Shrinkage + re-optimize ===
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_shr, sc_shr = _shrinkage_reopt(best_seq, deadline, min_time_remaining=60.0)
        update_best(a_shr, sc_shr, "shrinkage")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[Phase 8] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
