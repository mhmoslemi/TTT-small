        return np.inf

    e = np.exp(np.clip(z, -700, 700))

    N = _next_pow2(3 * n)
    Fw = np.fft.rfft(w, N)
    Fa = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(Fw * Fa, N)[:3 * n - 2]
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(w, n, rhs, max_iter=300, tol=1e-14):
    """Solve H_w @ x = rhs where H_w is the Hessian of sum_k w[k]*C[k].
    H_w[i,j] = w[i+j] + w[i+j] (symmetrized) = effectively 2*w[i+j] for i!=j, w[i+j] for i=j.
    We use the unsymmetrized version: (H_w @ x)[i] = sum_j 2*w[i+j]*x[j] for the off-diagonal part.
    Actually for the convex combination of quadratics, the Hessian of C[k] is:
    d2C[k]/da[i]da[j] = delta(i+j,k) + delta(j+i,k) = 2*delta(i+j,k) for i!=k-i
    For i = k-i (i.e. 2i=k): d2C[k]/da[i]da[i] = 2
    So the Hessian matrix H_k has H_k[i,j] = 1 if i+j=k, plus an extra 1 on diagonal when 2i=k.
    For simplicity in the CG, we use: (H_w @ x)[i] = sum_j w[i+j]*x[j] * 2 (approximation)
    This is a common approximation that works well in practice.
    """
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for it in range(max_iter):
        Hp = _hankel_matvec(p, w * 2.0, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoothed_obj(z, beta, n):
    """Log-sum-exp smoothed max of convolution."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    """Gradient of smoothed objective w.r.t. z (log-parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of sum_k w[k] * C[k] w.r.t. a[i] = sum_k w[k]*a[k-i]
    # = hankel matvec
    grad_a = _hankel_matvec(a, w, n)
    # Chain rule: d/dz_i = sum_j (d/d a_j)(a_j * delta) ... 
    # a = softmax(z), so da_j/dz_i = a_j*(delta_ij - a_i)
    # d_obj/dz_i = sum_j grad_a[j] * da_j/dz_i = sum_j grad_a[j]*a_j*delta_ij - sum_j grad_a[j]*a_j*a_i
    # = a_i * grad_a[i] - a_i * sum_j grad_a[j]*a_j
    # = a_i * (grad_a[i] - dot(grad_a, a))
    val = float(np.dot(grad_a, a))
    return a * (grad_a - val)


class _SmoObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoothed_obj(z, self.beta, self.n)


class _SmoGrad:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoothed_grad(z, self.beta, self.n)


def _newton_active_set(a_init, deadline, max_rounds=100):
    """Newton method with active set identification for KKT conditions."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    stall = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: start tight, expand if needed
        active = None
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3]:
            tol = tol_frac * c_max
            trial = c >= (c_max - tol)
            n_act = int(trial.sum())
            if n_act >= 2:
                active = trial
                break
        if active is None or active.sum() < 1:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True

        n_active = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute KKT gradient: g[i] = sum_k W[k]*a[k-i]
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Newton step: solve H @ dx = -g_t (with constraint sum(dx)=0)
        Hg = _cg_solve_hankel(W, n, g, max_iter=300)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=200)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search with sufficient decrease
        alpha = 1.0
        step_ok = False
        c_max_old = float(c.max())
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo condition
            if c_max_try < c_max_old - 0.001 * alpha * gn * 0.1 - 1e-16:
                a = a_try
                step_ok = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_ok:
            # Fallback: subgradient step
            for trial_lr in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial_lr * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max_old - 1e-16:
                    a = a_try
                    step_ok = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if not step_ok and stall > 8:
                break

    return best_a, best_sc


def _kkt_projection(a_init, deadline, max_iters=2000):
    """Iteratively project to satisfy KKT conditions using fixed-point iteration."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.5

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with moderate tolerance
        tol = 1e-4 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_act

        # KKT residual: g[i] - mean(g)
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-15 * n:
            break

        # Damped update to reduce KKT residual
        a_try = a - lr * g_t / max(np.max(np.abs(g_t)), 1e-30)
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        c_max_try = float(c_try.max())

        if c_max_try < c_max - 1e-18:
            a = a_try
            lr = min(lr * 1.02, 2.0)
            sc = _score(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
        else:
            lr *= 0.7

    return best_a, best_sc


def _run_lbfgs_adaptive(a_init, deadline, min_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())

    if max_c < 1e-15:
        return current_a

    # Initial beta from gap
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.3, max_c * 0.001)
    min_beta = max_c * 5e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_per_iter = 0.002 * scale * 20  # rough time per L-BFGS iteration
        max_iter = min(10000, max(50, int(remaining / max(est_per_iter, 0.01))))

        z = np.log(np.maximum(current_a, 1e-30))
        z = z - z.mean()

        try:
            obj_fn = _SmoObj(beta, n)
            grad_fn = _SmoGrad(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-18,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]

        sc_impr = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_impr > 1e-14:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 2e-4)

    return current_a


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        a = np.sin(np.pi * t ** 0.7) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "gaussian":
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        a = np.exp(-5.0 * ((t - 0.5) / 0.3) ** 4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        a = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_freq(a, rng, strength=0.3):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    A = np.fft.rfft(a)
    # Add random phase perturbation (preserves magnitude roughly)
    phase_noise = strength * rng.standard_normal(len(A))
    A_new = A * np.exp(1j * phase_noise * 0.1)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _perturb_time(a, sigma, rng):
    """Perturb in time domain via log-normal."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _cycle_optimize(a_init, deadline, min_remaining=50.0):
    """Alternate L-BFGS bursts with Newton polishing."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    cycle_dur = min(45.0, (deadline - min_remaining - time.monotonic()) / 8.0)
    if cycle_dur < 5:
        return current_a

    for cyc in range(10):
        if time.monotonic() + min_remaining > deadline:
            break
        t_start = time.monotonic()
        t_end = min(t_start + cycle_dur, deadline - min_remaining)

        # L-BFGS phase
        if time.monotonic() < t_end - 5:
            a_lb = _run_lbfgs_adaptive(current_a, t_end - 5, min_remaining=10.0)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb

        # Newton phase
        if time.monotonic() + 8 < min(t_end, deadline - min_remaining):
            a_np, sc_np = _newton_active_set(
                current_a, min(t_end, deadline - min_remaining), max_rounds=60)
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np

        # KKT projection phase
        if time.monotonic() + 5 < min(t_end, deadline - min_remaining):
            a_kkt, sc_kkt = _kkt_projection(
                current_a, min(t_end, deadline - min_remaining), max_iters=1000)
            if sc_kkt < current_sc:
                current_a = a_kkt
                current_sc = sc_kkt

        cycle_dur = min(cycle_dur * 1.15, 90.0)

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # === Phase 1: Quick profile sweep (100s budget) ===
    print(f"\n[Phase 1] Profile sweep (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 100.0
    profiles = ["tapered", "tapered2", "cos", "asym", "bimodal", "gaussian", "plateau", "cos_taper"]
    best_prof_a = parent.copy()
    best_prof_sc = _score(parent)

    for pk in profiles:
        if time.monotonic() > p1_deadline:
            break
        a_p = _make_profile(n_parent, pk)
        sc_p = _score(a_p)
        if sc_p < best_prof_sc:
            best_prof_sc = sc_p
            best_prof_a = a_p.copy()
            print(f"  Profile '{pk}': {sc_p:.10f}")

    # Perturbed parent
    for sigma in [0.05, 0.1]:
        if time.monotonic() > p1_deadline:
            break
        a_pt = _perturb_time(parent, sigma, rng)
        sc_pt = _score(a_pt)
        if sc_pt < best_prof_sc:
            best_prof_sc = sc_pt
            best_prof_a = a_pt.copy()

    # Quick L-BFGS on best
    if time.monotonic() < p1_deadline - 5:
        a_q = _run_lbfgs_adaptive(best_prof_a, p1_deadline - 5, min_remaining=80.0)
        _update_best(a_q, _score(a_q), "phase1_lbfgs")

    # === Phase 2: Main cycle optimization (800s budget) ===
    print(f"\n[Phase 2] Main cycle optimization (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = t_start + 900.0
    start_a = best_seq if best_seq is not None else a_q
    a_main = _cycle_optimize(start_a, p2_deadline, min_remaining=600.0)
    _update_best(a_main, _score(a_main), "main_cycle")

    # === Phase 3: Dedicated Newton polish (300s budget) ===
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = t_start + 1200.0
    if best_seq is not None:
        a_np, sc_np = _newton_active_set(best_seq, p3_deadline, max_rounds=200)
        _update_best(a_np, sc_np, "newton_p3")

        # KKT projection
        if time.monotonic() < p3_deadline - 10:
            a_kkt, sc_kkt = _kkt_projection(best_seq, p3_deadline - 10, max_iters=3000)
            _update_best(a_kkt, sc_kkt, "kkt_p3")

    # === Phase 4: More L-BFGS at very small beta (200s budget) ===
    print(f"\n[Phase 4] Ultra-fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = t_start + 1400.0
    if best_seq is not None:
        a_fine = _run_lbfgs_adaptive(best_seq, p4_deadline, min_remaining=300.0)
        _update_best(a_fine, _score(a_fine), "lbfgs_fine")

        # Follow up with Newton
        if time.monotonic() < p4_deadline - 5:
            a_fnp, sc_fnp = _newton_active_set(best_seq, p4_deadline - 5, max_rounds=80)
            _update_best(a_fnp, sc_fnp, "newton_fine")

    # === Phase 5: Multi-n exploration (150s budget) ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = t_start + 1550.0
    n_tries = [32768, 8192]
    for n_try in n_tries:
        if time.monotonic() > p5_deadline - 20:
            break
        src = best_seq if best_seq is not None else parent
        a_rs = _resample(src, n_try)
        a_opt = _run_lbfgs_adaptive(a_rs, min(p5_deadline, time.monotonic() + 60),
                                     min_remaining=20.0)
        _update_best(a_opt, _score(a_opt), f"n_{n_try}")

        if _score(a_opt) < best_score + 0.002 and time.monotonic() < p5_deadline - 10:
            a_onp, sc_onp = _newton_active_set(a_opt, p5_deadline - 10, max_rounds=50)
            _update_best(a_onp, sc_onp, f"n_{n_try}_np")

    # === Phase 6: Frequency-domain diverse restarts (80s budget) ===
    print(f"\n[Phase 6] Freq restarts (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = t_start + 1630.0
    if best_seq is not None:
        for i in range(4):
            if time.monotonic() > p6_deadline:
                break
            a_div = _perturb_freq(best_seq, rng, strength=0.2 + 0.1 * i)
            sc_div = _score(a_div)
            if sc_div < best_score + 0.01:
                a_do = _run_lbfgs_adaptive(a_div, p6_deadline, min_remaining=10.0)
                _update_best(a_do, _score(a_do), f"freq_{i}")

    # === Phase 7: Final ultra-fine Newton (60s budget) ===
    print(f"\n[Phase 7] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 2, max_rounds=200)
        _update_best(a_fin, sc_fin, "final_newton")

        # One more KKT
        if time.monotonic() < deadline - 1:
            a_fk, sc_fk = _kkt_projection(best_seq, deadline - 0.5, max_iters=500)
            _update_best(a_fk, sc_fk, "final_kkt")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
