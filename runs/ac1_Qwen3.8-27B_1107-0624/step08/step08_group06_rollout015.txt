```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _corr_a_A(a, A_vec, n):
    """Compute sum_k A_vec[k] * a[k-j] for j = 0, ..., n-1
    This is the gradient of C[k] w.r.t. a, weighted by A_vec."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]
```

```python
def _multiplicative_reweight(a_init, deadline, n_iter=10000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()

        return a_init

        c_max = float(c.max())

        sorted_c = np.sort(c)[::-1]
        # Use the "knee" or top m values
        m = max(10, min(len(c), int(0.1 * len(c))))
        threshold = sorted_c[m - 1]
        active = c >= threshold
        
        # Compute weights
        w = np.zeros_like(c)
        active_ones = active.astype(float)
        w[active] = 1.0 / active.sum()
        
        # Compute pressure P[j] = sum_k w[k] * a[k-j]
        P = _corr_a_A(a, w, n)
        
        # Target: make P uniform
        P_pos = P[P > 1e-15]

            break

        # Multiplicative update
        if target > 1e-15:
            ratio = target / np.maximum(P, 1e-15)
            ratio = np.clip(ratio, 0.5, 2.0)  # clip for stability
            a = a * (1.0 - eta + eta * ratio)
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 100 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        
        # Adaptive eta
        if it % 500 == 0 and it > 0:
            eta *= 0.95
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    
    return best_a
```
