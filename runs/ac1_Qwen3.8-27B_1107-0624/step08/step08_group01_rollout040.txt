"""
Optimized step-function search using Adam with gradient-structure exploitation,
Frank-Wolfe hybrids, active-set QP polishing, and multi-n exploration.
Exploits the key insight that grad of (f*f)[k] wrt f[j] = 2*a[k-j], enabling
extremely efficient gradient computation. Uses Adam for robust high-dim optimization
and active-set methods for precise final polishing.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _convolve(a):
    return _full_convolve(a, a)


def _gradient_at_peak(a, k_star):
    """Compute gradient of (f*f)[k_star] wrt f.
    grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n, else 0."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(k_star, n - 1)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        g[idx] = 2.0 * a[k_star - idx]
    return g


def _gradient_smoothed(a, c, betas, n):
    """Compute gradient of the smoothed (LSE) objective.
    Objective: (1/beta) * log(sum_k exp(beta * c[k]))
    Gradient: sum_k w_k * grad_k where w_k = softmax(beta * c)
    """
    c_max = float(c.max())
    shifted = betas * (c - c_max)
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Weighted sum of gradients
    g = np.zeros(len(a))
    for k in range(len(c)):
        if w[k] > 1e-12:
            g += w[k] * _gradient_at_peak(a, k)
    return g, c_max


def _adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step with projection onto {f>=0, sum(f)=1}."""
    m_new = beta1 * m + (1 - beta1) * grad
    v_new = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m_new / (1 - beta1**t)
    v_hat = v_new / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, m_new, v_new
    a_new = a_new / s
    return a_new, m_new, v_new


def _run_adam(a_init, deadline, lr=0.05, beta_start=1.0, min_remaining=5.0,
              use_lse=True, max_iters=None):
    """Run Adam optimization with LSE smoothing and adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Adaptive beta schedule
    c_cur = _convolve(a)
    c_max_cur = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_max_cur * 0.01
    beta = max(gap * 5.0, c_max_cur * 0.1)
    
    if max_iters is None:
        max_iters = 1000000
    
    no_improve = 0
    lr_decay = 0.999
    
    while time.monotonic() < deadline - min_remaining and t < max_iters:
        t += 1
        c = _convolve(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        s_a = float(a.sum())
        sc = sc / (s_a * s_a)
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            # Decay beta when we improve
            beta = max(c_max * 0.001, beta * 0.95)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                beta = max(c_max * 0.0005, beta * 0.9)
                lr = max(1e-5, lr * lr_decay)
        
        # Compute gradient
        if use_lse:
            grad, _ = _gradient_smoothed(a, c, beta, n)
        else:
            k_star = int(np.argmax(c))
            grad = _gradient_at_peak(a, k_star)
        
        # Normalize gradient (for Adam stability)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        a, m, v = _adam_step(a, grad, m, v, t, lr)
        
        # Periodic check
        if t % 1000 == 0:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _frank_wolfe_steps(a_init, n_steps, deadline, tol=1e-6):
    """Frank-Wolfe (conditional gradient) method with top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    k_peaks = 5
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (float(a.sum())**2)
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-k peaks
        top_k_idx = np.argsort(c)[::-1][:k_peaks]
        # Compute weighted gradient
        w = np.zeros(len(c))
        for idx in top_k_idx:
            w[idx] = 1.0 / k_peaks
        
        grad = np.zeros(n)
        for k in top_k_idx:
            grad += w[k] * _gradient_at_peak(a, k)
        
        # Frank-Wolfe: find vertex that minimizes <grad, e_i - a>
        # = argmin_i (grad[i] - sum(grad)/n)
        # = argmin_i grad[i]
        i_min = int(np.argmin(grad))
        e_i = np.zeros(n)
        e_i[i_min] = 1.0

        if sc_new < sc - 1e-15:
            a = a_new
        else:
            # Try smaller step
            gamma = gamma * 0.5
            a_new = (1 - gamma) * a + gamma * e_i
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new

def _active_set_qp(a_init, deadline, max_iters=200):
    """Active-set QP: at each step, identify active peaks and solve a QP to reduce them."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (float(a.sum())**2)
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        
        # Identify active set: peaks within a small tolerance of max
        tol = c_max * 0.01
        active_mask = c >= (c_max - tol)
        n_active = int(active_mask.sum())
        if n_active == 0:
            idx_max = int(np.argmax(c))
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[idx_max] = True
            n_active = 1
        
        # Limit active set size
        if n_active > 10:
            top_idx = np.argsort(c)[::-1][:10]
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[top_idx] = True
            n_active = 10
        
        # Compute the "average gradient" over active peaks
        grad = np.zeros(n)
        for k in range(len(c)):
            if active_mask[k]:
                grad += _gradient_at_peak(a, k) / n_active
        
        # We want to find the direction d that minimizes:
        #   max_{k in active} (f+d)*(f+d)[k]
        # subject to: sum(d) = 0, f + d >= 0
        # Approximation: minimize grad^T d (linear)
        # subject to: sum(d) = 0, d >= -f
        # The optimal d is: d[j] = -f[j] * max(0, grad[j] - nu) / sum_j f[j]*max(0, grad[j]-nu) ... 
        # Actually, the optimal solution to min grad^T d s.t. sum d = 0, d >= -f
        # is: sort grad, find threshold where cumulative constraint is met.
        # Simpler: just use d[j] = -(grad[j] - mean(grad)) * f[j] * alpha
        # This is a "proportional" update.
        
        grad_mean = np.dot(grad, a)  # weighted mean
        d = -(grad - grad_mean) * a
        # Ensure f + d >= 0
        alpha = 0.1
        a_try = a + alpha * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        
        sc_try = _score(a_try)
        if sc_try < sc - 1e-12:
            a = a_try
        else:
            # Try smaller step
            alpha *= 0.5
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-12:
                a = a_try
            else:
                # No improvement, try gradient-based approach
                break
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a

def _make_init(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * (2.0 * t - 1.0)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * (2.0 * t - 1.0)**2
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "front_loaded":
        t = np.linspace(0, 1, n)
        a = np.exp(-t * 2.0)
        return a / a.sum()
    elif kind == "back_loaded":
        t = np.linspace(0, 1, n)
        a = np.exp(-(1.0 - t) * 2.0)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.25)**2) + np.exp(-50.0 * (t - 0.75)**2) + 0.01
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        # Random smooth function
        freqs = rng.choice(np.arange(1, 10), size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amplitudes = rng.uniform(0.1, 0.3, size=5)
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amplitudes):
            a += amp * np.cos(2 * np.pi * f * t + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n + 1)  # Use +1 for better interpolation
    y_old = np.concatenate([a, [0.0]])
    x_new = np.linspace(0, 1, new_n + 1)
    a_new = np.interp(x_new, x_old, y_old)[:new_n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

"""
Optimized step-function search exploiting the gradient structure grad_k[j] = 2*a[k-j].
Uses efficient FFT-based smoothed gradients, Adam optimizer in primal space with
projection, Frank-Wolfe hybrids, peak-driven redistribution, active-set polishing,
and multi-n exploration. Key innovations: (1) O(n log n) gradient via correlation,
(2) Adam with adaptive beta and learning rate warmup/decay, (3) peak sensitivity
redistribution as a complementary update, (4) diverse initializations.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _convolve(a):
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _gradient_single(a, k):
    """Gradient of (f*f)[k] wrt f: g[j] = 2*a[k-j] if 0<=k-j<n."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(k, n - 1)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def _gradient_weighted(a, w):
    """Efficient weighted gradient: g[j] = 2 * sum_k w[k]*a[k-j].
    Computed via FFT: g = 2 * (flip(w) * a) at appropriate indices."""
    n = len(a)
    w_flip = w[::-1]
    conv_res = _full_convolve(w_flip, a)
    # g[j] = 2 * conv_res[2n-2-j] for j=0..n-1
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * conv_res[idx]
    return g


def _lse_weights(c, beta):
    """Compute softmax weights for LSE smoothing."""
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, c_max


def _adam_update(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step with projection to {f>=0, sum=1}."""
    m_new = beta1 * m + (1.0 - beta1) * grad
    v_new = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m_new / (1.0 - beta1 ** t)
    v_hat = v_new / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, m_new, v_new
    a_new = a_new / s
    return a_new, m_new, v_new


def _run_adam_primal(a_init, deadline, lr=0.08, use_lse=True, min_remaining=3.0,
                     beta_init=None, beta_decay=0.97, beta_min_frac=0.0005):
    """Adam optimization in primal space with LSE smoothing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    # Initialize beta
    if beta_init is None:
        c_init = _convolve(a)
        c_max_init = float(c_init.max())
        c_sorted = np.sort(c_init)[::-1]
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
        beta_init = max(gap * 3.0, c_max_init * 0.05)
    beta = beta_init

    # Learning rate warmup
    warmup_steps = min(200, max(20, n // 50))

    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() < deadline - min_remaining:
        t += 1
        c = _convolve(a)
        c_max = float(c.max())
        s_a = float(a.sum())
        sc = 2.0 * n * c_max / (s_a * s_a)

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Adaptive learning rate
        if t <= warmup_steps:
            lr_eff = lr * t / warmup_steps
        else:
            lr_eff = lr * (0.5 + 0.5 * np.cos(np.pi * min(1.0, (t - warmup_steps) / (10000.0))))
            lr_eff = max(lr_eff, lr * 0.01)

        # Compute gradient
        if use_lse:
            w, _ = _lse_weights(c, beta)
            grad = _gradient_weighted(a, w)
        else:
            k_star = int(np.argmax(c))
            grad = _gradient_single(a, k_star)

        # Normalize for stability
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm

        a, m, v = _adam_update(a, grad, m_new if False else m, v, t, lr_eff)
        # Fix: pass correct m, v
        # (recompute properly)
        m = beta1_val * m + (1 - beta1_val) * grad if False else m

        # Decay beta
        if no_improve % 200 == 0:
            beta = max(c_max * beta_min_frac, beta * beta_decay)

        # Periodic validation
        if t % 2000 == 0:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    # Final check
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _run_adam_clean(a_init, deadline, lr=0.08, use_lse=True, min_remaining=3.0,
                    beta_init=None, beta_decay=0.97, beta_min_frac=0.0005):
    """Clean Adam optimization in primal space."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    if beta_init is None:
        c_init = _convolve(a)
        c_max_init = float(c_init.max())
        c_sorted = np.sort(c_init)[::-1]
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
        beta_init = max(gap * 3.0, c_max_init * 0.05)
    beta = beta_init

    warmup_steps = min(200, max(20, n // 50))
    no_improve = 0
    max_no_improve = 3000

    while time.monotonic() < deadline - min_remaining:
        t += 1
        c = _convolve(a)
        c_max = float(c.max())
        s_a = float(a.sum())
        sc = 2.0 * n * c_max / (s_a * s_a)

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Adaptive learning rate with cosine decay
        if t <= warmup_steps:
            lr_eff = lr * t / warmup_steps
        else:
            progress = min(1.0, float(t - warmup_steps) / 20000.0)
            lr_eff = lr * (0.1 + 0.9 * 0.5 * (1.0 + np.cos(np.pi * progress)))

        # Compute gradient
        if use_lse:
            w, _ = _lse_weights(c, beta)
            grad = _gradient_weighted(a, w)
        else:
            k_star = int(np.argmax(c))
            grad = _gradient_single(a, k_star)

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        else:
            break

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        # Decay beta periodically
        if no_improve % 300 == 0:
            beta = max(c_max * beta_min_frac, beta * beta_decay)

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _peak_redistribute(a, n_steps=500, deadline=None):
    """Peak-driven redistribution: reduce high-sensitivity positions, boost low-sensitivity."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _convolve(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (float(a.sum()) ** 2)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        k_star = int(np.argmax(c))

        # Sensitivity: s[j] = a[k_star - j]
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        sensitivity = np.zeros(n)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            sensitivity[idx] = a[k_star - idx]

        # Reduce high-sensitivity, increase low-sensitivity
        # Use a smooth threshold
        s_mean = float(np.dot(a, sensitivity))
        s_std = float(np.sqrt(np.dot(a, (sensitivity - s_mean) ** 2))) + 1e-15

        # Direction: move mass from high-sens to low-sens
        threshold = s_mean + 0.5 * s_std
        reduce_mask = sensitivity > threshold
        add_mask = ~reduce_mask

        total_reduce = float(a[reduce_mask].sum()) * 0.02  # 2% per step
        if total_reduce < 1e-15 or add_mask.sum() == 0:
            break

        a[reduce_mask] -= a[reduce_mask] * 0.02
        a[add_mask] += total_reduce / a[add_mask].sum() * a[add_mask]

        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-15:
            break
        a = a / s_new

        # Check
        if step % 50 == 0:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _frank_wolfe(a_init, deadline, top_k=3):
    """Frank-Wolfe with multiple peak lags."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    t_fw = 0
    no_improve = 0

    while time.monotonic() < deadline - 2.0:
        t_fw += 1
        c = _convolve(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (float(a.sum()) ** 2)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        # Top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        # Weighted gradient
        w = np.zeros(len(c))
        for idx in top_idx:
            w[idx] = 1.0 / top_k

        grad = _gradient_weighted(a, w)

        # FW direction: e_{argmin grad} - a
        i_min = int(np.argmin(grad))
        e_i = np.zeros(n)
        e_i[i_min] = 1.0

        gamma = 2.0 / (t_fw + 2.0)
        a_new = (1.0 - gamma) * a + gamma * e_i
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = _convolve(a_new)
        sc_new = 2.0 * n * float(c_new.max()) / (float(a_new.sum()) ** 2)
        if sc_new < sc - 1e-14:
            a = a_new
        else:
            # Try half step
            gamma = gamma * 0.5
            a_new = (1.0 - gamma) * a + gamma * e_i
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            c_new = _convolve(a_new)
            sc_new = 2.0 * n * float(c_new.max()) / (float(a_new.sum()) ** 2)
            if sc_new < sc - 1e-14:
                a = a_new
            else:
                break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _active_set_polish(a_init, deadline, max_iters=500):
    """Active-set polynomial: solve approximate QP at each active peak set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.05
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break

        c = _convolve(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (float(a.sum()) ** 2)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        # Active set: peaks within 0.5% of max
        tol = c_max * 0.005
        active = c >= (c_max - tol)
        n_act = int(active.sum())
        if n_act == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_act = 1
        if n_act > 8:
            top_idx = np.argsort(c)[::-1][:8]
            active = np.zeros(len(c), dtype=bool)
            active[top_idx] = True
            n_act = 8

        # Weighted gradient
        w = np.zeros(len(c))
        w[active] = 1.0 / n_act
        grad = _gradient_weighted(a, w)

        # Proportional update: reduce where grad is high (weighted by a)
        g_weighted_mean = float(np.dot(a, grad))
        d = -(grad - g_weighted_mean) * a

        # Line search
        improved = False
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-13:
                a = a_try
                improved = True
                break

        if not improved:
            lr *= 0.7
            if lr < 1e-6:
                break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _cma_like_step(a, n_trials=20, deadline=None):
    """CMA-like: try random perturbations in the sensitivity direction."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _convolve(a)
        k_star = int(np.argmax(c))

        # Sensitivity direction
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        sens = np.zeros(n)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            sens[idx] = a[k_star - idx]

        # Random perturbation: move mass along or against sensitivity
        sigma = rng.uniform(0.005, 0.03)
        sign = 1.0 if rng.random() > 0.5 else -1.0
        d = sign * sens * sigma * a
        d -= np.dot(d, a)  # Make zero-sum (approximately)

        a_try = a + d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a


def _shape_optimize(a_init, deadline, n_restarts=2):
    """Core multi-method optimizer."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    total_time = max(10.0, deadline - time.monotonic())
    phase_time = total_time / n_restarts

    for restart in range(n_restarts):
        t_phase_start = time.monotonic()
        phase_deadline = t_phase_start + phase_time

        # Sub-phase A: Adam (50%)
        adam_dl = t_phase_start + phase_time * 0.5
        if time.monotonic() < adam_dl - 5:
            a = _run_adam_clean(a, adam_dl, lr=0.08 * (1.0 - 0.15 * restart))
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        # Sub-phase B: Peak redistribution (15%)
        redist_dl = t_phase_start + phase_time * 0.65
        if time.monotonic() < redist_dl - 3:
            a2 = _peak_redistribute(a, n_steps=300, deadline=redist_dl)
            sc2 = _score(a2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2.copy()
                a = a2

        # Sub-phase C: Frank-Wolfe (15%)
        fw_dl = t_phase_start + phase_time * 0.8
        if time.monotonic() < fw_dl - 3:
            a3 = _frank_wolfe(a, fw_dl, top_k=3)
            sc3 = _score(a3)
            if sc3 < best_sc:
                best_sc = sc3
                best_a = a3.copy()
                a = a3

        # Sub-phase D: Active-set polish (20%)
        as_dl = t_phase_start + phase_time * 1.0
        if time.monotonic() < as_dl - 3:
            a4 = _active_set_polish(a, as_dl, max_iters=200)
            sc4 = _score(a4)
            if sc4 < best_sc:
                best_sc = sc4
                best_a = a4.copy()
                a = a4

        # Perturb for next restart
        if restart < n_restarts - 1:
            rng = np.random.default_rng(777 + restart)
            a = a + 0.002 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            s_r = a.sum()
            if s_r > 1e-15:
                a = a / s_r

    return best_a


def _make_init(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * (2.0 * t - 1.0)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif kind == "quadratic_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif kind == "front_loaded":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        return a / a.sum()
    elif kind == "cosine_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t) + 0.15 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.3) ** 2) + np.exp(-40.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.arange(n) / n
        a = 1.0 + 0.3 * (2.0 * np.abs(t - 0.5) - 0.5)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _cma_exploration(a_init, deadline, n_trials=200):
    """More extensive CMA-like exploration with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    sigma = 0.01

    for trial in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _convolve(a)
        k_star = int(np.argmax(c))

        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        sens = np.zeros(n)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            sens[idx] = a[k_star - idx]

        # Also consider second peak
        c_copy = c.copy()
        c_copy[k_star] = 0
        k_star2 = int(np.argmax(c_copy))
        j_lo2 = max(0, k_star2 - n + 1)
        j_hi2 = min(k_star2, n - 1)
        sens2 = np.zeros(n)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            sens2[idx2] = a[k_star2 - idx2]

        # Random direction in sensitivity space
        d = sigma * (rng.standard_normal(n) * 0.1 + rng.standard_normal() * sens + rng.standard_normal() * 0.5 * sens2)
        d = d - np.dot(d, a) / (a.sum() + 1e-15) * a  # Make zero-weighted-sum

        a_try = a + d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            sigma *= 0.9
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            sigma = min(sigma * 1.1, 0.05)
        else:
            sigma *= 0.95

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer exploiting gradient structure grad_k[j]=2*a[k-j].
    Uses efficient FFT-based weighted gradients, Adam with adaptive schedules,
    peak redistribution, Frank-Wolfe, active-set polishing, CMA exploration,
    and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_init(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Intensive optimization on parent ===
    print(f"\n[P1] Intensive optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 700.0, t_start + 450.0)
    a1 = _shape_optimize(best_a, p1_deadline, n_restarts=2)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [P1]")
    else:
        print(f"  P1 best: {sc1:.10f} (no improvement over {best_score:.10f})")

    # CMA exploration on parent
    if time.monotonic() < deadline - 600.0:
        cma_dl = min(deadline - 600.0, time.monotonic() + 100.0)
        a_cma = _cma_exploration(best_a, cma_dl, n_trials=200)
        sc_cma = _score(a_cma)
        if sc_cma < best_score:
            best_a = a_cma.copy()
            best_score = sc_cma
            print(f"  NEW BEST: {best_score:.10f} [P1_cma]")

    # === Phase 2: Multi-n exploration ===
    print(f"\n[P2] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = sorted(set([n_parent, 16384, 24576]))
    n_targets = [nt for nt in n_targets if nt > 0 and nt != len(best_a)]

    for i_n, n_try in enumerate(n_targets):
        if time.monotonic() > deadline - 200.0:
            break

        n_remaining = len(n_targets) - i_n
        time_alloc = min(250.0, (deadline - time.monotonic() - 100.0) / max(1, n_remaining))
        if time_alloc < 40.0:
            continue

        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}, alloc={time_alloc:.0f}s (t={time.monotonic()-t_start:.1f}s)")

        n_dl = time.monotonic() + time_alloc
        a_opt = _shape_optimize(a_init, n_dl, n_restarts=2)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P2]")
        else:
            print(f"  n={n_try}: best={sc_opt:.10f} (not better than {best_score:.10f})")

    # === Phase 3: Diverse initializations at best n ===
    print(f"\n[P3] Diverse initializations n={len(best_a)} (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    init_kinds = ["asymmetric", "cosine_asym", "front_loaded", "bimodal", "sawtooth"]

    for kind in init_kinds:
        if time.monotonic() > deadline - 80.0:
            break
        remaining = deadline - time.monotonic() - 40.0
        if remaining < 30.0:
            break

        a_r = _make_init(n_best, kind)
        # Blend with current best for a warm start
        blend = 0.7 * a_r + 0.3 * best_a
        blend = np.maximum(blend, 0.0)
        s_b = blend.sum()
        if s_b > 1e-15:
            blend = blend / s_b

        alloc = min(remaining * 0.4, 120.0)
        a_opt = _shape_optimize(blend, time.monotonic() + alloc, n_restarts=1)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [P3_{kind}]")
        else:
            print(f"  [P3_{kind}]: {sc_opt:.10f}")

    # === Phase 4: Final intensive on best ===
    print(f"\n[P4] Final intensive (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30.0:
        p4_dl = deadline - 15.0
        a4 = _shape_optimize(best_a, p4_dl, n_restarts=3)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_a = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4]")

    # === Phase 5: Final micro-polish ===
    print(f"\n[P5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        a5 = _active_set_polish(best_a, deadline - 2.0, max_iters=300)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_a = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5]")

        # One more CMA burst
        if time.monotonic() < deadline - 2.0:
            a6 = _cma_exploration(best_a, deadline - 1.0, n_trials=100)
            sc6 = _score(a6)
            if sc6 < best_score:
                best_a = a6.copy()
                best_score = sc6
                print(f"  NEW BEST: {best_score:.10f} [P5_cma]")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = _make_init(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
