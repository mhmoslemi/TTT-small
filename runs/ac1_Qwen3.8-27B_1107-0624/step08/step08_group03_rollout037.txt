"""
Optimized step function search combining IFT (Iterative Fourier Transformation)
for flat autocorrelation initialization with L-BFGS log-sum-exp smoothing and
active-set Newton refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# IFT for flat autocorrelation
def _ift_flat_autocorr(n, n_iter=5000, rng=None, target_type='flat'):
    """
    IFT to find a non-negative sequence of length n with flat autocorrelation.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    N = _next_pow2(2 * n - 1)
    
    # Initialize
    a_pad = np.zeros(N)
    a_pad[:n] = rng.random(n)
    a_pad[:n] = np.maximum(a_pad[:n], 1e-10)
    a_pad = a_pad / a_pad.sum()
    
    best_sc = np.inf
    best_a = a_pad[:n].copy()
    
    for it in range(n_iter):
        # Forward FFT
        A = np.fft.rfft(a_pad)
        
        # Compute power spectrum
        power = A.real**2 + A.imag**2
        
        # Target: flat power spectrum
        # But we need to handle the DC component carefully
        # A[0] = sum(a) which we want to keep
        # For flat target: |A[k]| = sqrt(mean(power)) for all k
        target_amp = np.sqrt(np.mean(power))
        
        # Create new spectrum with flat amplitude, keep phase
        phase = A / np.maximum(np.abs(A), 1e-30)  # unit phase
        A_new = target_amp * phase
        A_new[0] = np.abs(A[0])  # Keep DC as positive real
        
        # Inverse FFT
        a_pad = np.fft.irfft(A_new, N)
        
        # Project: non-negative support in [0, n)
        a_pad[n:] = 0
        a_pad[:n] = np.maximum(a_pad[:n], 0)
        
        # Renormalize
        s = a_pad.sum()
        if s < 1e-15:
            break
        a_pad = a_pad / s
        
        # Check every 100 iterations

            a_check = a_pad[:n]
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    
    return best_a, best_sc

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    if rng is None:
        rng = np.random.default_rng()
    N = _next_pow2(2 * n - 1)

    a_pad = np.zeros(N)
    a_pad[:n] = rng.random(n)
    a_pad[:n] = np.maximum(a_pad[:n], 1e-10)
    a_pad = a_pad / a_pad.sum()

    best_sc = np.inf
    best_a = a_pad[:n].copy()

    for it in range(n_iter):
        A = np.fft.rfft(a_pad)
        power = A.real ** 2 + A.imag ** 2
        target_amp = np.sqrt(np.mean(power))
        abs_A = np.maximum(np.abs(A), 1e-30)
        phase = A / abs_A
        A_new = target_amp * phase
        A_new[0] = np.abs(A[0])
        a_pad = np.fft.irfft(A_new, N)
        a_pad[n:] = 0
        a_pad[:n] = np.maximum(a_pad[:n], 0)
        s = a_pad.sum()
        if s < 1e-15:
            break
        a_pad = a_pad / s

        if it % 200 == 0 or it == n_iter - 1:
            a_check = a_pad[:n]
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    return best_a, best_sc


def _ift_concentrated(n, n_iter=3000, rng=None):
    """
    IFT variant targeting concentrated power (mostly DC),
    which corresponds to flat autocorrelation.
    """
    if rng is None:
        rng = np.random.default_rng()
    N = _next_pow2(2 * n - 1)

    a_pad = np.zeros(N)
    a_pad[:n] = rng.random(n)
    a_pad[:n] = np.maximum(a_pad[:n], 1e-10)
    a_pad = a_pad / a_pad.sum()

    best_sc = np.inf
    best_a = a_pad[:n].copy()

    # Target: power concentrated at low frequencies
    # |A[k]|^2 should be large near k=0 and small elsewhere
    # This gives a smoother (more spread) autocorrelation
    k_vals = np.arange(N // 2 + 1)
    # Target shape: decaying exponential in frequency
    freq_decay = np.exp(-k_vals / (N / max(n, 4)))

    for it in range(n_iter):
        A = np.fft.rfft(a_pad)
        abs_A = np.maximum(np.abs(A), 1e-30)
        phase = A / abs_A

        # Set target amplitude: use the current DC value scaled by frequency decay
        target_amp = np.abs(A[0]) * freq_decay / freq_decay[0]
        A_new = target_amp * phase
        A_new[0] = np.abs(A[0])

        a_pad = np.fft.irfft(A_new, N)
        a_pad[n:] = 0
        a_pad[:n] = np.maximum(a_pad[:n], 0)
        s = a_pad.sum()
        if s < 1e-15:
            break
        a_pad = a_pad / s

        if it % 200 == 0 or it == n_iter - 1:
            a_check = a_pad[:n]
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                      0.015, 0.0075, 0.0037, 0.00185, 0.0009,
                                      0.00045, 0.00022, 0.00011, 0.00005,
                                      0.000025, 0.00001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + max(min_time_remaining, 5) > deadline:
            break
        remaining = deadline - max(min_time_remaining, 5) - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.006
        elif n <= 16384:
            est_time_per_iter = 0.015
        else:
            est_time_per_iter = 0.03 * (n / 32768.0)

        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 20))))

        try:
            res = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            n_active = 2 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def _hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def _cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - _hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(max_cg):
                Ap = _hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = _cg_solve(g_t)
        w = _cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for _ in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for _ in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-12 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _peak_shave(a_init, deadline, max_rounds=200):
    """Directly target and reduce the highest convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks
        n_peaks = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Weighted average of top peaks
        w = np.zeros(len(c))
        weights = np.exp(-np.arange(n_peaks) * 0.3)
        weights /= weights.sum()
        for i, idx in enumerate(top_idx):
            w[idx] = weights[i]

        # Gradient of weighted peak w.r.t. a
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)

        # Line search
        f_old = c_max
        step = 0.05 / (1.0 + 0.005 * rnd)
        improved = False
        for _ in range(30):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "rimmed":
        a = np.ones(n)
        edge = max(2, n // 32)
        a[:edge] *= 2.0
        a[-edge:] *= 2.0
        return a / a.sum()
    elif kind == "rimmed_strong":
        a = np.ones(n)
        edge = max(2, n // 16)
        a[:edge] *= 3.0
        a[-edge:] *= 3.0
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_window":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.2) ** 2) / 0.03) + 0.5 * np.exp(-((t - 0.8) ** 2) / 0.03)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.standard_normal(n)
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Mirror descent in log-space with active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    - IFT-based initialization for diverse starting points
    - L-BFGS with log-sum-exp smoothing
    - Active-set Newton refinement
    - Peak shaving
    - Multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: IFT initialization (diverse starting points)
    print(f"\n[Phase 1] IFT initialization (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        n_if = n_parent if n_parent <= 16384 else 16384
        # Flat power IFT
        a_if1, sc_if1 = _ift_flat_power(n_if, n_iter=2000, rng=rng)
        update_best(a_if1, sc_if1, "ift_flat")
        # Concentrated power IFT
        a_if2, sc_if2 = _ift_concentrated(n_if, n_iter=2000, rng=rng)
        update_best(a_if2, sc_if2, "ift_conc")

    # Phase 2: Structured initializations
    print(f"\n[Phase 2] Structured initializations (t={time.monotonic() - t_start:.1f}s)")
    structured_profiles = ["rimmed", "rimmed_strong", "sine", "cos_window", "bimodal", "gaussian_center"]
    for kind in structured_profiles:
        if time.monotonic() > deadline - 400:
            break
        a_struct = _make_profile(n_parent, kind, rng)
        sc_struct = _score(a_struct)
        if sc_struct < best_score + 0.05:
            # Only optimize if promising
            a_opt = _run_lbfgs(a_struct, deadline, min_time_remaining=200.0)
            update_best(a_opt, _score(a_opt), f"struct_{kind}")

    # Phase 3: Main L-BFGS from parent (most time)
    print(f"\n[Phase 3] Main L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    t_remaining = deadline - time.monotonic()
    if t_remaining > 100:
        a_main = _run_lbfgs(parent, deadline - 150, min_time_remaining=100.0, aggressive=True)
        update_best(a_main, _score(a_main), "lbfgs_parent_aggr")

    # Phase 4: L-BFGS from IFT results if they were good
    print(f"\n[Phase 4] L-BFGS from best IFT (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100 and best_seq is not None:
        a_main2 = _run_lbfgs(best_seq, deadline - 80, min_time_remaining=60.0, aggressive=True)
        update_best(a_main2, _score(a_main2), "lbfgs_best_aggr")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200 and best_seq is not None:
        for n_try in [8192, 32768]:
            if time.monotonic() > deadline - 100:
                break
            a_interp = _interpolate_to_n(best_seq, n_try)
            sc_interp = _score(a_interp)
            print(f"  n={n_try} interpolated score: {sc_interp:.10f}")
            if sc_interp < best_score + 0.02:
                a_opt = _run_lbfgs(a_interp, deadline - 80, min_time_remaining=50.0)
                update_best(a_opt, _score(a_opt), f"interp_n{n_try}")

    # Phase 6: Active-set Newton
    print(f"\n[Phase 6] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 30, time.monotonic() + 60),
                                         max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "active_set_1e-3")

        if time.monotonic() < deadline - 30:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 15, time.monotonic() + 40),
                                               max_iter=150, tol_active=0.0003)
            update_best(a_as2, sc_as2, "active_set_3e-4")

    # Phase 7: Peak shaving
    print(f"\n[Phase 7] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_ps, sc_ps = _peak_shave(best_seq, min(deadline - 20, time.monotonic() + 60), max_rounds=200)
        update_best(a_ps, sc_ps, "peak_shave")

    # Phase 8: Subgradient polish
    print(f"\n[Phase 8] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 10, time.monotonic() + 60),
                                      n_max_steps=3000, lr_init=0.02)
        update_best(a_sg, sc_sg, "subgrad")

    # Phase 9: Diverse restarts from best
    print(f"\n[Phase 9] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for label, sigma in [("r002", 0.02), ("r005", 0.05), ("r010", 0.10)]:
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline - 5, min_time_remaining=5.0)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # Phase 10: Final tight active-set
    print(f"\n[Phase 10] Final tight Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 2, max_iter=200, tol_active=0.0001)
        update_best(a_fin, sc_fin, "final_newton_tight")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
