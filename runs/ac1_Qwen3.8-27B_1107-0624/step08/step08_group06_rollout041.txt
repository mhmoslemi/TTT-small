import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr_a_A(a, A_vec, n):
    """Compute the correlation of a with A_vec, extracted to the right length."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[50]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap, max_c * 0.005)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                  0.015, 0.0075, 0.004, 0.002, 0.001, 
                                  0.0005, 0.00025, 0.000125, 0.00006,
                                  0.00003, 0.000015, 0.0000075])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration based on n
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 5))))
        
        try:
            result = minimize(
                _lsobj_wrapper,
                z,
                args=(beta, n),
                jac=_lsgrad_wrapper,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception:
            pass
    
    return current_a

def _deflation_step(a, deadline, n_attempts=30):
    """Targeted deflation of top convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top k positions
        n_top = max(5, min(len(c), int(0.02 * len(c))))
        top_positions = np.argsort(c)[::-1][:n_top]
        
        # For each top position, find and reduce the largest contributing pair
        a_new = a.copy()
        reduced = False
        for k_pos in top_positions:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 1e-15:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= 0.99
                else:
                    a_new[j] *= 0.99
                reduced = True

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2*block:] = 1.0
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif kind == "block3":
        a = np.zeros(n)
        block = n // 6
        a[:block] = 1.0
        a[2*block:3*block] = 1.0
        a[5*block:] = 1.0
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t * (1-t) * 4  # Parabolic
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

Multi-phase optimizer combining L-BFGS (log-sum-exp with aggressive beta annealing),
adaptive active-set Newton with CG, iterative deflation, and multi-n exploration
to minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_vec(a, A_vec, n):
    """Compute cross-correlation of a with A_vec, extracting positions [n-1, 2n-2]."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    a = _softmax(z)
    c = _full_convolve(a, a)

    s = e.sum()

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

                    accepted = True
                    break
                t *= 0.5

            if accepted:
                break

        if best_a_try is None:
            # Fallback: projected gradient
            g_all = (2.0 / n) * _corr_vec(a, np.ones(2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-15:
                break
            t = 0.01
            for ls in range(40):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_a_try = a_try.copy()
                    best_sc_try = sc_try
                    break
                t *= 0.5

        if best_a_try is None:
            break

        a = best_a_try
        if best_sc_try < best_sc:
            best_sc = best_sc_try
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14 * max(c_max, 1e-15):
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(100, len(c_sorted) - 1)
    gap = max_c - c_sorted[idx]
    base_beta = max(gap * 0.5, max_c * 0.003)

    # Very aggressive beta schedule
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625,
        0.0078, 0.004, 0.002, 0.001, 0.0005, 0.00025,
        0.000125, 0.00006, 0.00003, 0.000015, 0.0000075,
        0.000004, 0.000002, 0.000001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 4))))

        try:
            result = minimize(
                _lsobj_call,
                z,
                args=(beta, n),
                jac=_lsgrad_call,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-19,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 300,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    LBFGS beta={beta:.2e}: score={sc_new:.12f}")
        except Exception:
            pass

    return current_a


def _deflation_step(a, deadline, n_attempts=30):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        n_top = max(5, min(len(c), int(0.03 * len(c))))
        top_positions = np.argsort(c)[::-1][:n_top]

        a_new = a.copy()
        reduced = False
        for k_pos in top_positions:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 1e-18:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= 0.995
                else:
                    a_new[j] *= 0.995
                reduced = True

        if not reduced:
            break

        a_new = np.maximum(a_new, 0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try gentler reduction
            a_small = a.copy()
            for k_pos in top_positions[:5]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0.0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_small[i] * a_small[k_pos - i]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    if a_small[max_i] >= a_small[j]:
                        a_small[max_i] *= 0.9995
                    else:
                        a_small[j] *= 0.9995
            a_small = np.maximum(a_small, 0)
            s_small = float(a_small.sum())
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-14:
                    best_sc = sc_small
                    best_a = a_small.copy()
                else:
                    break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2 * block:] = 1.0
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif kind == "block3":
        a = np.zeros(n)
        block = max(1, n // 7)
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[4 * block:5 * block] = 1.0
        a[6 * block:] = 1.0
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif kind == "parab":
        t = np.linspace(0, 1, n)
        a = 4.0 * t * (1.0 - t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tri":
        t = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _kkt_polish(a, deadline, max_rounds=50):
    """
    KKT-based polishing: find weights w over active set such that
    sum_k w[k] * a[k-i] is constant for all i with a[i] > 0.
    Use iterative proportional fitting to find such w.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with tight tolerance
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        active_idx = np.where(active)[0]
        m = len(active_idx)

        # Solve for w: find w >= 0 on active set such that
        # (w * a_rev)[i + n - 1] is as constant as possible
        # Use iterative method: initialize w uniform, then adjust

        w = np.ones(m) / m
        for sub_iter in range(200):
            # Compute the "effective gradient" at each position
            w_full = np.zeros(2 * n - 1)
            w_full[active_idx] = w
            g_eff = _corr_vec(a, w_full, n)
            # We want g_eff to be constant (equal to lambda) on support of a
            support = a > 1e-10
            if not support.any():
                break
            g_supp = g_eff[support]
            if g_supp.max() < 1e-15:
                break
            # Multiplicative update: increase w where g_eff is low
            # For each active position k, its contribution to position i is a[k-i]
            # We want to equalize g_eff over support
            target = np.mean(g_supp)
            if target < 1e-15:
                break
            # Compute how much each active position contributes to the imbalance
            adjust = np.zeros(m)
            for j, k_pos in enumerate(active_idx):
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                # Contribution of w[j] to g_eff[i] is a[k_pos - i] for valid i
                # We want to reduce w[j] if it's contributing to high g_eff positions
                contrib_imbalance = 0.0
                for i in range(i_lo, i_hi + 1):
                    if a[i] > 1e-10:
                        contrib_imbalance += (g_eff[i] - target) * a[k_pos - i]
                adjust[j] = contrib_imbalance
            # Normalize and apply
            if np.max(np.abs(adjust)) < 1e-15:
                break
            eta = 0.1
            w = w * np.exp(-eta * adjust / (np.max(np.abs(adjust)) + 1e-15))
            w = np.maximum(w, 1e-15)
            w = w / w.sum()

        # Now use w as subgradient direction
        w_full = np.zeros(2 * n - 1)
        w_full[active_idx] = w

        # Newton step with this fixed active set
        g = (2.0 / m) * _corr_vec(a, w_full, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        mu = 1e-7 * c_max / m + 1e-9
        hv = _hess_vp_factory(w_full, m, mu, n)
        x = _cg_solve(hv, g_t, max_iter=200, tol=1e-12)
        w_sol = _cg_solve(hv, np.ones(n), max_iter=100, tol=1e-10)
        w_sum = float(w_sol.sum())

        if abs(w_sum) < 1e-15:
            dx = -x
        else:
            lam = float(x.sum()) / w_sum
            dx = -x + lam * w_sol
        dx = dx - np.mean(dx)

        # Line search
        neg = dx < -1e-12
        if neg.any():
            t_max = float(np.min(a[neg] / (-dx[neg]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        t = min(1.0, t_max)
        accepted = False
        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                a = a_try
                best_sc = sc_try
                best_a = a.copy()
                accepted = True
                break
            t *= 0.5

        if not accepted:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS with aggressive beta annealing
    Phase 2: Active-set Newton (medium tolerance)
    Phase 3: KKT polish
    Phase 4: Deflation + re-optimize cycle
    Phase 5: Active-set Newton (tight tolerance)
    Phase 6: Multi-n exploration
    Phase 7: Final polish (tiny beta L-BFGS + Newton)
    Phase 8: Diverse restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Parent: n={n_parent}, score={best_score:.12f}")
    else:
        n_parent = 8192
        best_seq = np.ones(n_parent) / n_parent
        best_score = _score(best_seq)
        print(f"No parent: uniform n={n_parent}, score={best_score:.12f}")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS (budget reserved: 350s)")
    a1 = _run_lbfgs(best_seq, deadline, min_time_remaining=350)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  Phase 1 best: {best_score:.12f}")

    # Phase 2: Active-set Newton (medium)
    print(f"\n[Phase 2] Newton medium tolerance")
    if time.monotonic() < deadline - 250:
        dl = min(deadline - 250, time.monotonic() + 180)
        a2, sc2 = _active_set_newton(best_seq, dl, max_iter=400,
                                     tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  Phase 2 best: {best_score:.12f}")

    # Phase 3: KKT polish
    print(f"\n[Phase 3] KKT polish")
    if time.monotonic() < deadline - 200:
        dl = min(deadline - 200, time.monotonic() + 80)
        a3, sc3 = _kkt_polish(best_seq, dl, max_rounds=40)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  Phase 3 best: {best_score:.12f}")

    # Phase 4: Deflation + re-optimize
    print(f"\n[Phase 4] Deflation + Newton")
    if time.monotonic() < deadline - 150:
        dl_def = min(deadline - 150, time.monotonic() + 30)
        a_def, sc_def = _deflation_step(best_seq, dl_def, n_attempts=15)
        if sc_def < best_score:
            best_score = sc_def
            best_seq = a_def.copy()
            print(f"  After deflation: {best_score:.12f}")
        # Re-optimize after deflation
        if time.monotonic() < deadline - 100:
            dl = min(deadline - 100, time.monotonic() + 60)
            a4, sc4 = _active_set_newton(best_seq, dl, max_iter=200,
                                         tol_list=[0.002, 0.001, 0.0005, 0.0002, 0.0001])
            if sc4 < best_score:
                best_score = sc4
                best_seq = a4.copy()
                print(f"  Phase 4 best: {best_score:.12f}")

    # Phase 5: Tight Newton
    print(f"\n[Phase 5] Newton tight tolerance")
    if time.monotonic() < deadline - 80:
        dl = min(deadline - 80, time.monotonic() + 60)
        a5, sc5 = _active_set_newton(best_seq, dl, max_iter=300,
                                     tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  Phase 5 best: {best_score:.12f}")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n")
    n_try_list = [4096, 8192, 16384]
    if n_parent not in n_try_list and 1024 <= n_parent <= 32768:
        n_try_list.append(n_parent)
    n_try_list.sort()

    for n_try in n_try_list:
        if time.monotonic() > deadline - 40:
            break
        if n_try == n_parent:
            continue

        # Best initialization for this n
        a_init = None
        sc_init = np.inf
        for kind in ["uniform", "tapered", "block3", "cos2"]:
            ai = _make_profile(n_try, kind)
            sci = _score(ai)
            if sci < sc_init:
                sc_init = sci
                a_init = ai
        if parent is not None:
            ai_parent = _interpolate_to_n(parent, n_try)
            sci_parent = _score(ai_parent)
            if sci_parent < sc_init:
                sc_init = sci_parent
                a_init = ai_parent

        if a_init is None:
            continue

        a6 = _run_lbfgs(a_init, deadline, min_time_remaining=40)
        sc6 = _score(a6)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  n={n_try}: {best_score:.12f}")

        # Quick Newton
        if sc6 < best_score + 0.01 and time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + 20)
            a6b, sc6b = _active_set_newton(a6, dl, max_iter=80,
                                           tol_list=[0.002, 0.001, 0.0005, 0.0002])
            if sc6b < best_score:
                best_score = sc6b
                best_seq = a6b.copy()
                print(f"  n={n_try} newton: {best_score:.12f}")

    # Phase 7: Final polish
    print(f"\n[Phase 7] Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Tiny beta L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        for beta_frac in [0.00001, 0.000005, 0.000002, 0.000001]:
            if time.monotonic() > deadline - 3:
                break
            beta = mb * beta_frac
            remaining = deadline - 4 - time.monotonic()
            if remaining < 3:
                break
            max_iter = min(5000, max(30, int(remaining / 0.02)))
            try:
                result = minimize(
                    _lsobj_call, z, args=(beta, n),
                    jac=_lsgrad_call,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-19,
                             'gtol': 1e-15, 'maxcor': 120, 'maxls': 400}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_score = sc_fin
                    best_seq = a_fin.copy()
                    z = np.log(np.maximum(a_fin, 1e-30))
                    z = z - z.mean()
                    print(f"  Final LBFGS beta={beta:.2e}: {best_score:.12f}")
            except Exception:
                pass

        # Final tight Newton
        if time.monotonic() < deadline - 3:
            dl = min(deadline - 2, time.monotonic() + 20)
            a_fn, sc_fn = _active_set_newton(best_seq, dl, max_iter=200,
                                             tol_list=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
            if sc_fn < best_score:
                best_score = sc_fn
                best_seq = a_fn.copy()
                print(f"  Final Newton: {best_score:.12f}")

        # Final deflation
        if time.monotonic() < deadline - 2:
            a_fd, sc_fd = _deflation_step(best_seq, min(deadline - 1, time.monotonic() + 10),
                                          n_attempts=10)
            if sc_fd < best_score:
                best_score = sc_fd
                best_seq = a_fd.copy()
                print(f"  Final deflation: {best_score:.12f}")

    # Phase 8: Diverse restarts (if time remains)
    print(f"\n[Phase 8] Restarts")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for sigma, lab in [(0.02, "s"), (0.05, "m"), (0.1, "L")]:
            if time.monotonic() > deadline - 3:
                break
            p_st = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_st, deadline, min_time_remaining=2)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {lab}: {best_score:.12f}")

    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            ev = evaluate_sequence([float(x) for x in final])
            print(f"Verified evaluate_sequence: {ev:.12f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
