def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

Mirror-descent step-function optimizer for minimum-peak autocorrelation.
Uses multiplicative weights with LSE-smoothed gradients, FFT-based convolution,
adaptive step sizes, multi-length exploration, and active-set polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 25 == 0:
                tol = max(tol_end, tol * 0.6)
                lr *= 0.5
            if no_improve >= 150:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.3)
                lr = lr_start * 0.5
                no_improve = 0
                stall += 1
                if stall > 8:
                    break
        else:
            tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.005)
    
    return best_a


def _peak_shave_targeted(a, deadline, n_trials=500, min_remaining=3.0):
    """Targeted peak shaving: reduce top contributors to the peak lag."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)
    
    for trial in range(n_trials):
        if time.monotonic() + min_remaining > deadline:
            break
        
        c = _autocorr(a)
        c_max = float(c.max())
        
        # Find top few lags
        top_lags = np.argsort(c)[::-1][:3]
        
        for k_star in top_lags:
            k_star = int(k_star)
            # Contributors: a[i] * a[k_star - i] for valid i
            k_lo = max(0, k_star - n + 1)
            k_hi = min(n - 1, k_star)
            if k_lo > k_hi:
                continue
            i_range = np.arange(k_lo, k_hi + 1)
            j_range = k_star - i_range
            contrib = a[i_range] * a[j_range]
            total = contrib.sum()
            if total < 1e-15:
                continue
            
            # Reduce top contributors
            n_red = max(1, min(5, len(contrib) // 4))
            top_idx_local = np.argsort(contrib)[::-1][:n_red]
            top_i = i_range[top_idx_local]
            
            a_try = a.copy()
            reduction_total = 0.0
            for idx in top_i:
                red = a_try[idx] * 0.05
                a_try[idx] -= red
                reduction_total += red
            
            # Distribute to non-contributing elements
            contrib_set = set(int(x) for x in top_i)
            contrib_set.update(int(x) for x in k_star - top_i)
            non_contrib = np.array([i for i in range(n) if i not in contrib_set])
            
            if len(non_contrib) > 0:
                a_try[non_contrib] += reduction_total / len(non_contrib)
            
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break  # Move to next lag
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
