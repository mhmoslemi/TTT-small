    for _ in range(max_outer):
        neg_mask = x < -1e-8
        if not neg_mask.any():
            x = np.maximum(x, 0)
            s = x.sum()
            if s > 1e-15:
                return x / s
            break
        
        # Zero out the most negative components
        A_new = A_indicator.copy()
        # We need to also update the "support" - remove coordinates where x < 0
        # This is approximate: we just re-solve with the same H but project
        # Actually, for the removed coordinates, the constraint is a_i = 0.
        # The reduced problem has H_sub @ x_sub = 1_sub.
        # This is complex to implement with FFT. Let's just clip and renormalize.
        x = np.maximum(x, 0)
        s = x.sum()
        if s > 1e-15:
            return x / s
        break
    
    return a_init.copy()

    s = e.sum()

        w[int(np.argmax(c))] = 1.0

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Active-set QP with FFT-based conjugate gradients (Hankel matvec via cross-correlation)
2. L-BFGS with log-sum-exp smoothing (continuation method)
3. Alternating direction refinement (dual weights + QP)
4. Multi-n exploration and diverse restarts
Key innovation: The QP Hessian matvec is an FFT cross-correlation, giving O(n log n) iterates.
"""
import numpy as np
import time
from scipy.optimize import minimize

    import cvxpy as cp

except Exception:
    HAVE_CVXPY = False

# Module-level params for scipy optimizer (avoids closures/lambdas)

_g_n = 1


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s

    return 2.0 * n * (c_max + beta * np.log(s))

def _hankel_matvec(v, fA, N, n, mu=0.0):
    """Compute H @ v where H[i,j] = A[i+j], using precomputed fA."""
    fv = np.fft.rfft(v, N)
    result = np.fft.irfft(fA * np.conj(fv), N)[:n]
    if mu > 0:
        result = result + mu * v
    return result


def _cg_solve_hankel(rhs, A, n, max_iter=800, tol=1e-10, mu=0.0):
    """Solve (H + mu*I) @ x = rhs via CG, where H[i,j] = A[i+j]."""
    N = _next_pow2(3 * n - 2)
    fA = np.fft.rfft(A.astype(np.float64), N)

    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    rhs_norm = np.linalg.norm(rhs)

    for i in range(max_iter):
        Hv = _hankel_matvec(p, fA, N, n, mu)
        pH = float(np.dot(p, Hv))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hv
        rs_new = float(np.dot(r, r))
        if not np.isfinite(rs_new):
            break
        if rs_new < 0:
            rs_new = 0.0
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new

    return x


def _solve_qp(a_init, c, tol_frac, n, mu=1e-8):
    """
    Solve QP: min (1/m) sum_{k in S} c_k(a) s.t. sum(a)=1.
    S = {k : c[k] >= c_max - tol_frac * c_max}
    Uses CG with FFT-based Hankel matvec.
    Returns updated a.
    """
    c_max = float(c.max())
    if c_max < 1e-15:
        return a_init.copy()

    threshold = c_max - tol_frac * c_max
    A = (c >= threshold).astype(np.float64)
    m = float(A.sum())

    if m < 2:
        A[int(np.argmax(c))] = 1.0
        m = 1.0
    if m > 4 * n:
        top_idx = np.argsort(c)[::-1][:int(4 * n)]
        A = np.zeros_like(c)
        A[top_idx] = 1.0
        m = float(A.sum())

    # Solve H @ x = 1
    rhs = np.ones(n)
    x = _cg_solve_hankel(rhs, A, n, max_iter=1000, tol=1e-10, mu=mu)

    # Handle non-negativity
    x = np.maximum(x, 0)
    s = float(x.sum())
    if s < 1e-15:
        return a_init.copy()
    return x / s

    n = len(a_init)

            break

            if c_max < prev_c_max * 0.999:
                tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
        prev_c_max = c_max

        tol_frac = tol_schedule[tol_idx]

        # Solve QP
        a_qp = _solve_qp(a, c, tol_frac, n, mu=1e-8)
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-14:

            best_a = a_qp.copy()

        accepted = False

                    best_sc = sc_try

                else:

            else:
                break

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:

        if remaining < 5:
            break

        _g_beta = beta
        est_time = 0.006 * max(1.0, n / 2048.0)
        max_iter = min(10000, max(50, int(remaining / (est_time * 4))))

        try:
            result = minimize(
                _lsobj_func,
                z,
                jac=_lsgrad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _deflation(a, deadline, n_attempts=30):
    """Targeted deflation of top convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        top_k = np.argsort(c)[::-1]
        n_top = max(3, min(len(c), int(0.03 * len(c))))
        top_positions = top_k[:n_top]

        a_new = a.copy()
        for k_pos in top_positions[:min(n_top, 15)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= 0.998
                else:
                    a_new[j] *= 0.998

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller reduction
            a_small = a.copy()
            for k_pos in top_positions[:8]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_small[i] * a_small[k_pos - i]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    if a_small[max_i] >= a_small[j]:
                        a_small[max_i] *= 0.9995
                    else:
                        a_small[j] *= 0.9995
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-12:
                    best_sc = sc_small
                    best_a = a_small.copy()
                else:
                    break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        bs = n // 4
        a[:bs] = 1.0
        a[3 * bs:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining active-set QP (FFT-CG),
    L-BFGS log-sum-exp, alternating direction, and deflation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline - 800, time.monotonic() + 400)
    a_main = _run_lbfgs(parent, p1_dl, min_time_remaining=700.0)
    update_best(a_main, _score(a_main), "lbfgs_1")

    # Phase 2: Active-set QP
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        p2_dl = min(deadline - 500, time.monotonic() + 350)
        a_qp, sc_qp = _active_set_qp_loop(best_seq, p2_dl, max_iter=3000)
        update_best(a_qp, sc_qp, "as_qp_1")

    # Phase 3: Alternating direction
    print(f"\n[Phase 3] Alternating direction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        p3_dl = min(deadline - 400, time.monotonic() + 250)
        a_alt, sc_alt = _alternating_direction(best_seq, p3_dl, max_iter=500)
        update_best(a_alt, sc_alt, "alternating_1")

    # Phase 4: Second active-set QP with tighter tolerance
    print(f"\n[Phase 4] Fine QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        p4_dl = min(deadline - 300, time.monotonic() + 200)
        a_qp2, sc_qp2 = _active_set_qp_loop(best_seq, p4_dl, max_iter=2000)
        update_best(a_qp2, sc_qp2, "as_qp_2")

    # Phase 5: Deflation + QP
    print(f"\n[Phase 5] Deflation + QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        p5_dl = min(deadline - 200, time.monotonic() + 60)
        a_def, sc_def = _deflation(best_seq, p5_dl, n_attempts=40)
        update_best(a_def, sc_def, "deflation")
        if sc_def < best_score and time.monotonic() < deadline - 120:
            p5b_dl = min(deadline - 120, time.monotonic() + 80)
            a_def2, sc_def2 = _active_set_qp_loop(a_def, p5b_dl, max_iter=1000)
            update_best(a_def2, sc_def2, "deflation_qp")

    # Phase 6: L-BFGS again with very small beta
    print(f"\n[Phase 6] Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        p6_dl = min(deadline - 100, time.monotonic() + 150)
        a_lb = _run_lbfgs(best_seq, p6_dl, min_time_remaining=80.0)
        update_best(a_lb, _score(a_lb), "lbfgs_fine")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue
        a_init = _interpolate_to_n(parent if parent is not None else best_seq, n_try)
        sc_init = _score(a_init)

        p7_dl = min(deadline - 50, time.monotonic() + 120)
        a_opt = _run_lbfgs(a_init, p7_dl, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _active_set_qp_loop(a_opt, min(deadline - 25, time.monotonic() + 30),
                                                   max_iter=300)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_qp")

    # Phase 8: Restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma in [0.02, 0.05]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _active_set_qp_loop(p_start, min(deadline - 10, time.monotonic() + 40),
                                             max_iter=500)
            update_best(a_r, sc_r, f"restart_s{sigma}")

    # Phase 9: Final polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        p9_dl = min(deadline - 2, time.monotonic() + 50)
        a_fin, sc_fin = _active_set_qp_loop(best_seq, p9_dl, max_iter=500)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _alternating_direction(best_seq, min(deadline - 1, time.monotonic() + 20),
                                                     max_iter=100)
            update_best(a_fin2, sc_fin2, "final_alt")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
