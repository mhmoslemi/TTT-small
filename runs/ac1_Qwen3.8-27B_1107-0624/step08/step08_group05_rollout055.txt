"""
Advanced multi-phase optimizer for minimizing step function autocorrelation peak.
Key improvements over previous: Nesterov-accelerated stochastic subgradient,
top-K adaptive active-set Newton, multi-K exploration, aggressive LSE annealing
with restarts, and Barzilai-Borwein adaptive step sizes.

import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# LSE smoothing
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    val = np.sum(np.power(c_safe, p)) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    sum_cp = np.sum(np.power(c_safe, p))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def _subgrad_max(a, c, n, top_k=50, random_frac=0.0, rng=None):
    """Compute subgradient of max(c) w.r.t. a, optionally using random subset."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    # Find top-k lags
    k_eff = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k_eff]
    
    if random_frac > 0 and rng is not None and k_eff > 2:
        n_use = max(2, int(k_eff * (1.0 - random_frac)))
        chosen = rng.choice(top_indices, size=n_use, replace=False)
    else:
        chosen = top_indices
    
    w = np.zeros_like(c)
    w[chosen] = 1.0 / len(chosen)
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _nesterov_subgrad(a_init, deadline, max_steps=20000, top_k=50,
                       random_frac=0.3, base_lr=0.005, rng=None):
    """Nesterov-accelerated stochastic subgradient method on log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    momentum = 0.9
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        z_y = z + momentum * (z - z_prev)
        a_y = _softmax(z_y)
        c_y = _full_convolve(a_y, a_y)
        
        # Stochastic subgradient
        grad_a = _subgrad_max(a_y, c_y, n, top_k=top_k,
                              random_frac=random_frac, rng=rng)
        grad_z = a_y * (grad_a - np.dot(a_y, grad_a))
        
        # Adaptive step size (opacity method)
        lr = base_lr / (1.0 + 0.01 * i)
        
        z_new = z - lr * grad_z
        z_new = np.clip(z_new, -100.0, 100.0)
        z_new = z_new - z_new.mean()
        
        # Check improvement periodically
        if i % 100 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
        
        z_prev = z
        z = z_new
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _bb_subgrad(a_init, deadline, max_steps=10000, top_k=100, rng=None):
    """Barzilai-Borwein adaptive step size subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    z_prev = z.copy()
    grad_prev = None
    lr = 0.005
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _subgrad_max(a, c, n, top_k=top_k, rng=rng)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        # BB step size
        if grad_prev is not None:
            y = grad_z - grad_prev
            s = z - z_prev
            y_norm_sq = float(np.dot(y, y))
            s_dot_y = float(np.dot(s, y))
            if y_norm_sq > 1e-30 and abs(s_dot_y) > 1e-15:
                lr_bb = s_dot_y / y_norm_sq
                lr = max(1e-6, min(lr_bb, 0.05))
        
        z_new = z - lr * grad_z
        z_new = np.clip(z_new, -100.0, 100.0)
        z_new = z_new - z_new.mean()
        
        if i % 200 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
        
        z_prev = z
        grad_prev = grad_z
        z = z_new
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha < 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

    """Newton method with top-K active set selection. Tries multiple K values."""

    best_a = a_init.copy()
    best_sc = _score(a_init)

            break

        a = np.maximum(a, 0)

        if s < 1e-15:
            continue

        for it in range(max_iter):
            if time.monotonic() > deadline:
                break

            # Top-K active set

            active_idx = np.argsort(c)[::-1][:k_eff]

            # Gradient = H*a where H is the active-set Hessian

            active_f = active.astype(np.float64)

                return _hess_matvec(v, active_f, n) + mu * v

            alpha = 1.0
            a_new = a.copy()

                    continue

                c_try = _full_convolve(a_try, a_try)

                    a_new = a_try
                    break
                alpha *= 0.5
            else:
                if alpha < 1e-12:
                    break
                a_new = a
            
            a = a_new
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a, best_sc

def _lbfgs_lse_aggressive(a_init, deadline, min_remaining=30.0):
    """L-BFGS with LSE, aggressive beta schedule with restarts."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    # More aggressive schedule with more stages
    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18,
                                   0.12, 0.08, 0.05, 0.03, 0.02,
                                   0.012, 0.007, 0.004, 0.002,
                                   0.001, 0.0005, 0.0002])
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 6))))
        
        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):

    current_a = a_init.copy()
    current_sc = _score(current_a)

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = np.maximum(amp * (1.0 + sigma * rng.standard_normal(len(amp))), 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_flatten(a, rng, strength=0.1):
    """Mix with uniform to flatten."""
    a_flat = np.ones_like(a) / len(a)
    a_new = (1 - strength) * a + strength * a_flat
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_concave(a, rng, strength=0.05):
    """Add a concave bump to encourage spreading."""
    n = len(a)
    t = np.linspace(0, 1, n)
    bump = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
    bump = bump - bump.mean()
    a_new = a + strength * bump
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t) + 0.1 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

# Stochastic mirror descent with adaptive step
# ============================================================

def _stochastic_mirror_descent(a_init, deadline, max_steps=30000, top_k=30,
                               random_frac=0.4, base_lr=0.003, rng=None):
    """Stochastic mirror descent with adaptive learning rate and restarts."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = base_lr
    patience = 0
    patience_limit = 50
    restart_count = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _subgrad_max(a, c, n, top_k=top_k,
                              random_frac=random_frac, rng=rng)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        # Adaptive step size
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.95
                if patience > patience_limit:
                    # Restart from best with perturbation
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z = z + 0.02 * rng.standard_normal(n)
                    z = z - z.mean()
                    patience = 0
                    lr = base_lr * 0.5
                    restart_count += 1
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: LSE-LBFGS, Nesterov subgradient, BB subgradient,
    top-K Newton, stochastic mirror descent, multi-restart with perturbations.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic()-t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init_uniform")

    n_main = max(n_parent, 4096) if n_parent > 0 else 16384

    # === Phase 1: Quick diverse exploration (45s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 45)
    for kind in ["uniform", "tapered", "cos2", "gauss", "bimodal", "wheatstone", "raised_cos"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_main // 4, kind)
        sc_try = _score(a_try)
        if _best_seq is not None and len(_best_seq) != len(a_try):
            a_try_full = _resize_sequence(a_try, n_main)
            sc_try = _score(a_try_full)
            upd(a_try_full, sc_try, f"expl_{kind}")
        else:
            upd(a_try, sc_try, f"expl_{kind}")

    # Quick LSE on best to establish a good starting point
    if time.monotonic() < deadline - 600:
        a_up = _resize_sequence(_best_seq, n_main) if len(_best_seq) != n_main else _best_seq.copy()
        q_dead = min(deadline - 500, t_start + 90)
        a_q, sc_q = _lbfgs_lse_aggressive(a_up, q_dead, min_remaining=50)
        upd(a_q, sc_q, "quick_lse")

    # === Phase 2: Main LSE optimization (400s) ===
    print(f"\n[Phase 2] Main LSE-LBFGS")
    if time.monotonic() < deadline - 350:
        lse_dead = min(deadline - 300, time.monotonic() + 400)
        a_lse, sc_lse = _lbfgs_lse_aggressive(_best_seq.copy(), lse_dead, min_remaining=250)
        upd(a_lse, sc_lse, "lse_main")

    # === Phase 3: Power-sum L-BFGS (150s) ===
    print(f"\n[Phase 3] Power-sum L-BFGS")
    if time.monotonic() < deadline - 150:
        ps_dead = min(deadline - 130, time.monotonic() + 150)
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), ps_dead, min_remaining=120)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 4: Nesterov subgradient (200s) ===
    print(f"\n[Phase 4] Nesterov subgradient")
    if time.monotonic() < deadline - 150:
        ns_dead = min(deadline - 130, time.monotonic() + 200)
        a_ns, sc_ns = _nesterov_subgrad(_best_seq, ns_dead, max_steps=50000,
                                         top_k=80, random_frac=0.3,
                                         base_lr=0.004, rng=rng)
        upd(a_ns, sc_ns, "nesterov_sg")

    # === Phase 5: Top-K Newton (200s) ===
    print(f"\n[Phase 5] Top-K Newton")
    if time.monotonic() < deadline - 120:
        nw_dead = min(deadline - 100, time.monotonic() + 200)
        a_nw, sc_nw = _newton_topk(_best_seq, nw_dead,
                                    k_values=[20, 50, 100, 300, 1000],
                                    max_iter=150)
        upd(a_nw, sc_nw, "newton_topk")

    # === Phase 6: BB subgradient (150s) ===
    print(f"\n[Phase 6] BB subgradient")
    if time.monotonic() < deadline - 100:
        bb_dead = min(deadline - 80, time.monotonic() + 150)
        a_bb, sc_bb = _bb_subgrad(_best_seq, bb_dead, max_steps=20000,
                                   top_k=100, rng=rng)
        upd(a_bb, sc_bb, "bb_sg")

    # === Phase 7: Stochastic mirror descent (150s) ===
    print(f"\n[Phase 7] Stochastic mirror descent")
    if time.monotonic() < deadline - 100:
        smd_dead = min(deadline - 80, time.monotonic() + 150)
        a_smd, sc_smd = _stochastic_mirror_descent(_best_seq, smd_dead,
                                                    max_steps=50000, top_k=30,
                                                    random_frac=0.4,
                                                    base_lr=0.003, rng=rng)
        upd(a_smd, sc_smd, "smd")

    # === Phase 8: Alternating Newton + Nesterov (150s) ===
    print(f"\n[Phase 8] Alternating Newton + Nesterov")
    if time.monotonic() < deadline - 80:
        a_cur = _best_seq.copy()
        for rnd in range(4):
            if time.monotonic() > deadline - 60:
                break
            # Newton
            nw_d = min(deadline - 60, time.monotonic() + 30)
            a_nw2, sc_nw2 = _newton_topk(a_cur, nw_d,
                                          k_values=[50, 200, 500], max_iter=60)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_nw_{rnd}")
            # Nesterov
            ns_d = min(deadline - 60, time.monotonic() + 25)
            a_ns2, sc_ns2 = _nesterov_subgrad(a_cur, ns_d, max_steps=20000,
                                               top_k=50, random_frac=0.3,
                                               base_lr=0.003, rng=rng)
            if sc_ns2 < _score(a_cur):
                a_cur = a_ns2
            upd(a_cur, _score(a_cur), f"alt_ns_{rnd}")

    # === Phase 9: Multi-restart with perturbations (200s) ===
    print(f"\n[Phase 9] Multi-restart")
    if time.monotonic() < deadline - 60:
        perturbations = [
            ("log_003", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("log_005", lambda: _perturb_log(_best_seq, 0.05, rng)),
            ("fourier", lambda: _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)),
            ("block", lambda: _perturb_block_swap(_best_seq, rng, n_swaps=4)),
            ("amp", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.08)),
            ("flatten", lambda: _perturb_flatten(_best_seq, rng, strength=0.05)),
            ("concave", lambda: _perturb_concave(_best_seq, rng, strength=0.03)),
        ]
        for name, pfunc in perturbations:
            if time.monotonic() > deadline - 40:
                break
            try:
                p_start = pfunc()
            except Exception:
                continue
            
            # Quick LSE on perturbed point
            qd = min(deadline - 35, time.monotonic() + 20)
            a_r, sc_r = _lbfgs_lse_aggressive(p_start, qd, min_remaining=10)
            upd(a_r, sc_r, f"restart_{name}")
            
            # Quick Newton
            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 20:
                nwd = min(deadline - 15, time.monotonic() + 15)
                a_r2, sc_r2 = _newton_topk(a_r, nwd, k_values=[50, 200], max_iter=40)
                upd(a_r2, sc_r2, f"restart_{name}_nw")
            
            # Quick Nesterov
            if time.monotonic() < deadline - 10:
                nsd = min(deadline - 8, time.monotonic() + 10)
                a_r3, sc_r3 = _nesterov_subgrad(_best_seq, nsd, max_steps=10000,
                                                 top_k=50, random_frac=0.3,
                                                 base_lr=0.003, rng=rng)
                upd(a_r3, sc_r3, f"restart_{name}_ns")

    # === Phase 10: Multi-n exploration (150s) ===
    print(f"\n[Phase 10] Multi-n exploration")
    n_candidates = [4096, 8192, 12288, 16384, 20480]
    if n_parent not in n_candidates and 2048 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)
        
        # LSE
        lse_d = min(deadline - 25, time.monotonic() + 40)
        a_on, sc_on = _lbfgs_lse_aggressive(a_init, lse_d, min_remaining=15)
        upd(a_on, sc_on, f"n_{n_try}_lse")
        
        # Newton
        if sc_on < _best_score * 1.005 and time.monotonic() < deadline - 10:
            nwd = min(deadline - 8, time.monotonic() + 20)
            a_on2, sc_on2 = _newton_topk(a_on, nwd, k_values=[50, 200], max_iter=40)
            upd(a_on2, sc_on2, f"n_{n_try}_nw")

    # === Phase 11: Final aggressive polish (60s) ===
    print(f"\n[Phase 11] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-low beta LSE
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())

        z_fin = z_fin - z_fin.mean()

                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),

                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")

                pass

        # Final Newton with tight active set
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_topk(_best_seq, deadline - 2,
                                            k_values=[10, 30, 100], max_iter=50)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final Nesterov with very small random_frac (nearly deterministic)
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _nesterov_subgrad(_best_seq, deadline - 1,
                                                 max_steps=30000, top_k=200,
                                                 random_frac=0.1,
                                                 base_lr=0.002, rng=rng)
            upd(a_fin3, sc_fin3, "final_nesterov")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
