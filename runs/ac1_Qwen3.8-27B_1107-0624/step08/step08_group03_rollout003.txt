Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: multiplicative subgradient, frequency-domain flattening, 
asymmetric perturbations, and tight active-set Newton polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_act = int(active.sum())
    if n_act == 0:

        n_act = 1

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h

            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            top_idx = np.argsort(c)[::-1][:2 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 2 * n
        
        A_vec = active.astype(np.float64)
        m = n_active
        
        # Gradient of the average of active peaks
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break
        
        # Regularized Hessian
        mu = 1e-8 * (c_max / max(m, 1)) + 1e-10
        
        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v
        
        # CG solve
        def cg_solve(rhs, max_cg=100):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha_cg = rs / pAp
                x += alpha_cg * p
                r_new = r - alpha_cg * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x
        
        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)
        
        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)
        
        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(30):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5
        
        if not accepted:
            # Fallback: steepest descent
            t = 0.01
            a_try = a - t * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    accepted = True
        
        if not accepted:
            break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _corr_a_A(a, A_vec, n):
    """Compute the 'correlation' of a with A_vec, extracted to length n."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _asymmetric_perturb(a, epsilon, kind="linear", n_mod=1):
    """Apply an asymmetric perturbation to break symmetry."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    if kind == "linear":
        pert = 1.0 + epsilon * (t - 0.5) * 2.0
    elif kind == "sine":
        pert = 1.0 + epsilon * np.sin(np.pi * t * n_mod)
    elif kind == "sine2":
        pert = 1.0 + epsilon * np.sin(2.0 * np.pi * t * n_mod)
    elif kind == "cos":
        pert = 1.0 + epsilon * np.cos(np.pi * t)
    else:
        pert = np.ones(n)
    a_pert = a * pert
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
