"""
Optimized subgradient method for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key ideas:
1. Exact subgradient of the max convolution (top-k smoothing with continuation)
2. Projected gradient onto the simplex with backtracking line search
3. Simulated annealing for escaping local minima
4. Multiple n values (16384, 32768, 65536)
5. FFT-based convolution for large n
6. Direct gradient computation for small k (sparse peak weighting)

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    tau = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - tau, 0)


def _compute_subgradient(a, c, k):
    """Compute subgradient of max(conv) using top-k peaks."""
    n = len(a)
    top_k = np.argsort(c)[::-1][:k]
    c_max = c[top_k[0]]
    # Softmax weights over top-k
    shifts = (c[top_k] - c_max) / max(c_max * 0.01, 1e-15)
    w = np.exp(np.clip(shifts, -50, 0))
    sw = w.sum()
    if sw < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / sw

    if k <= 32:
        # Direct computation
        g = np.zeros(n)
        for i in range(k):
            kj = int(top_k[i])
            wi = w[i]
            # g[i] += 2 * wi * a[kj - i] for valid indices
            # a[kj - idx] is valid when 0 <= kj - idx < n
            lo = max(0, kj - (n - 1))
            hi = min(n - 1, kj)
            for idx in range(lo, hi + 1):
                g[idx] += 2.0 * wi * a[kj - idx]
    else:
        # FFT-based
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_k] = w
        a_rev = a[::-1]
        full = _fft_convolve(w_vec, a_rev)
        g = 2.0 * full[n - 1:2 * n - 1]
    return g

            n_active = 4 * n

        A_vec = active.astype(np.float64)
        a_rev = a[::-1]
        full = _fft_convolve(A_vec, a_rev)
        g = (2.0 / n_active) * full[n - 1:2 * n - 1]
        g_tan = g - np.mean(g)
        gnorm = np.linalg.norm(g_tan)

            break

        # H is the Hessian of the objective, which involves the active set

            full_h = _fft_convolve(A_vec, vect[::-1])
            return (2.0 / n_active) * full_h[n - 1:2 * n - 1] + mu * vect

        rhs = -g_tan
        rhs = rhs - np.mean(rhs)

            if np.sqrt(rs_new) < 1e-12:
                x = x_new
                break
            p = r_new + (rs_new / rs) * p
            x = x_new
            r = r_new
            rs = rs_new

            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:

                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            f_try = float(c_try.max())

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_asymmetric_init(n, rng, kind="ramp"):

        a = 0.01 + 0.99 * t ** 3

        a = 0.1 + 0.9 * np.exp(-((t - 0.2) ** 2) / 0.05)
    elif kind == "bump_right":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * np.exp(-((t - 0.8) ** 2) / 0.05)
    elif kind == "two_bump":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.6 * np.exp(-((t - 0.2) ** 2) / 0.03) + 0.6 * np.exp(-((t - 0.8) ** 2) / 0.03)

    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
