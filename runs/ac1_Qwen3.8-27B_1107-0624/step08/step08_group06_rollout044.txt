"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Aggressive log-sum-exp L-BFGS with geometric beta schedule
2. Proper convex active-set Newton with FFT-based Hessian application
3. Asymmetric perturbations to escape symmetric local minima
4. Multi-n exploration (16384, 32768)
5. FISTA acceleration for subgradient steps

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the gradient contribution: cross-correlation of a with A_vec, sliced to [n-1, 2n-2]."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_vec(v, a, w, beta, n):
    """Apply Hessian of the smoothed objective to vector v (for Newton/CG)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w_vec = np.zeros_like(c)
        w_vec[int(np.argmax(c))] = 1.0
    else:
        w_vec = e / s

    # First term: sum_k w_k * H_k * v
    # (H_k * v)_j = 2 * v_{k-j} (correlation)
    # sum_k w_k * (H_k * v) = 2 * cross_corr(w_vec, v)
    a_rev = a[::-1]
    v_rev = v[::-1]
    corr_wv = _full_convolve(w_vec, v_rev)
    corr_wv = corr_wv[n - 1:2 * n - 1]
    term1 = 2.0 * corr_wv

    # Second term: (1/beta) * [sum_k w_k * (grad_qk^T v) * grad_qk - g * (g^T v)]
    # grad_qk = 2 * a shifted (correlation of a with delta at k)
    # grad_qk^T v = 2 * (a * v)[k] (convolution of a and v)
    av = _full_convolve(a, v)
    gkv = 2.0 * av  # length 2n-1 = len(c)

    # sum_k w_k * gkv[k] * grad_qk
    # grad_qk is the k-th "shifted" version of 2a
    # sum_k w_k * gkv[k] * 2a_{k-i} = 2 * cross_corr(w_vec * gkv, a_rev)
    wgkv = w_vec * gkv
    sum_term = _full_convolve(wgkv, a_rev)
    sum_term = sum_term[n - 1:2 * n - 1]
    term2 = (2.0 / beta) * sum_term

    # g = sum_k w_k * grad_qk
    g = 2.0 * _corr_a_A(a, w_vec, n)
    gv = float(np.dot(g, v))
    term3 = -(2.0 / beta) * gv * g

    return term1 + term2 + term3


def _newton_step(a, beta, n, max_cg=50):
    """Solve H*d = -grad + nu*1 using CG, return d and nu."""
    g = _lsgrad(np.log(np.maximum(a, 1e-30)), beta, n)
    g_t = g - np.mean(g)

    # CG to solve H * d = -g_t
    # (projected: H * d is in tangent space approximately)
    r = -g_t
    p = r.copy()
    d = np.zeros(n)
    rs = np.dot(r, r)

    for _ in range(max_cg):
        Hp = _hess_vec(p, a, None, beta, n)
        Hp = Hp - np.mean(Hp)  # project to tangent space
        pHp = np.dot(p, Hp)
        if abs(pHp) < 1e-30:
            break
        alpha = rs / pHp
        d = d + alpha * p
        r = r - alpha * Hp
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < 1e-12 * np.sqrt(rs):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new

    # Project d to tangent space
    d = d - np.mean(d)
    return d, g

def _fista_v2(a_init, deadline, n_steps=5000):
    """FISTA with adaptive step size for the convex problem."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    a = a / s

    y = a.copy()
    t_fista = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.1

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())

        # Adaptive tolerance for active set
        tol = max(0.005 * c_max, 1e-10)
        active = c_y >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[np.argmax(c_y)] = True
            n_active = 1

        # Subgradient: average gradient over active set
        A_vec = active.astype(np.float64)
        grad = (2.0 / n_active) * _corr_a_A(y, A_vec, n)
        g_t = grad - np.mean(grad)

        # Adaptive step size (barzilai-borwein style)
        a_new = y - step_size * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        y_old = y.copy()
        y = a_new + ((t_fista - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t_fista = t_new

        # Check improvement
        if step % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, best_sc

    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Estimate iterations possible
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                _make_lsobj_wrapper,
                z,
                jac=_make_lsgrad_wrapper,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_lsobj_wrapper(z, beta, n):
    return _lsobj(z, beta, n)


def _make_lsgrad_wrapper(z, beta, n):
    return _lsgrad(z, beta, n)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_asymmetric(a, sigma, rng):
    """Create an asymmetric perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    noise = rng.standard_normal(n)
    # Add a linear trend to break symmetry
    trend = np.linspace(-1, 1, n) * sigma * 0.5
    z = z + sigma * noise + trend
    z = z - z.mean()
    return _softmax(z)


def _perturb_symmetric(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    noise = rng.standard_normal(n)
    # Symmetrize noise
    noise = (noise + noise[::-1]) / 2
    z = z + sigma * noise
    z = z - z.mean()
    return _softmax(z)


def _peak_shaving(a, deadline, n_rounds=50):
    """Directly shave the highest peaks of the convolution."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find the top peak
        k_peak = int(np.argmax(c))

        # The peak is contributed by pairs (i, k_peak - i)
        # Reduce the larger contributor
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(k_peak, n - 1)

        if i_lo > i_hi:
            break

        # Find the pair with maximum product
        best_i = -1
        best_prod = -1
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            if 0 <= j < n:
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod = prod
                    best_i = i

        if best_i < 0:
            break

        j = k_peak - best_i
        # Reduce the larger element
        if a[best_i] >= a[j]:
            a[best_i] *= 0.999
        else:
            a[j] *= 0.999

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(0.5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy convex optimizer:
    1. L-BFGS with aggressive log-sum-exp beta schedule
    2. Active-set Newton with CG-based Hessian application
    3. Asymmetric perturbations
    4. Multi-n exploration
    5. Peak shaving as final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        # Check symmetry
        sym_diff = np.max(np.abs(parent - parent[::-1]))
        print(f"  Symmetry deviation: {sym_diff:.6e}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent with aggressive beta schedule
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 400, time.monotonic() + 600)
    a_main = _run_lbfgs_v2(parent, lbfgs_dl, min_time_remaining=400.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton polish
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        as_dl = min(deadline - 300, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton_v2(best_seq, as_dl, max_iter=300)
        update_best(a_as, sc_as, "as_newton")

    # Phase 3: FISTA from current best
    print(f"\n[Phase 3] FISTA (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        fista_dl = min(deadline - 200, time.monotonic() + 100)
        a_fista, sc_fista = _fista_v2(best_seq, fista_dl, n_steps=20000)
        update_best(a_fista, sc_fista, "fista")

    # Phase 4: Asymmetric perturbation + optimize
    print(f"\n[Phase 4] Asymmetric perturbation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        for sigma in [0.05, 0.1]:
            if time.monotonic() > deadline - 50:
                break
            a_sym = _perturb_asymmetric(best_seq, sigma, rng)
            a_opt = _run_lbfgs_v2(a_sym, min(deadline - 80, time.monotonic() + 100),
                                   min_time_remaining=60.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"asym_{sigma}")

            if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
                a_as2, sc_as2 = _active_set_newton_v2(a_opt,
                    min(deadline - 30, time.monotonic() + 50), max_iter=100)
                update_best(a_as2, sc_as2, f"asym_{sigma}_as")

    # Phase 5: Try n=32768
    print(f"\n[Phase 5] n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if parent is not None and time.monotonic() < deadline - 80:
        n_big = 32768
        a_big = _interpolate_to_n(parent, n_big)
        sc_big_init = _score(a_big)
        print(f"  Interpolated to n={n_big}: score={sc_big_init:.10f}")
        update_best(a_big, sc_big_init, "interp_32768")

        if time.monotonic() < deadline - 40:
            a_big_opt = _run_lbfgs_v2(a_big, min(deadline - 20, time.monotonic() + 120),
                                       min_time_remaining=15.0)
            sc_big_opt = _score(a_big_opt)
            update_best(a_big_opt, sc_big_opt, "lbfgs_32768")

    # Phase 6: Final polish on best
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Peak shaving
        ps_dl = min(deadline - 3, time.monotonic() + 20)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_dl, n_rounds=100)
        update_best(a_ps, sc_ps, "peak_shave")

        # Very tight L-BFGS
        if time.monotonic() < deadline - 3:
            a_tight = _run_lbfgs_v2(best_seq, min(deadline - 2, time.monotonic() + 30),
                                     min_time_remaining=2.0)
            update_best(a_tight, _score(a_tight), "final_lbfgs")

        # One more active-set Newton
        if time.monotonic() < deadline - 2:
            a_fin, sc_fin = _active_set_newton_v2(best_seq,
                min(deadline - 1, time.monotonic() + 20), max_iter=100)
            update_best(a_fin, sc_fin, "final_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
