    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 0.3 + t ** 2
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "spectral":
        if rng is None:
            rng = np.random.default_rng(0)
        N_fft = _next_pow2(2 * n)
        n_half = N_fft // 2 + 1
        phases = rng.uniform(0, 2.0 * np.pi, n_half)
        mags = np.ones(n_half)
        mags[0] = 0.0
        F = np.zeros(N_fft, dtype=complex)
        F[:n_half] = mags * np.exp(1j * phases)
        for k in range(1, N_fft // 2 + 1):
            F[N_fft - k] = np.conj(F[k])
        a_raw = np.fft.ifft(F).real
        offset = (N_fft - n) // 2
        a = np.abs(a_raw[offset:offset + n])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "spectral_mix":
        if rng is None:
            rng = np.random.default_rng(1)
        N_fft = _next_pow2(2 * n)
        n_half = N_fft // 2 + 1
        phases = rng.uniform(0, 2.0 * np.pi, n_half)
        mags = np.ones(n_half)
        mags[0] = 0.0
        # Add some low-frequency structure
        for k in range(1, min(10, n_half)):
            mags[k] = 1.0 + 0.5 * rng.random()
        F = np.zeros(N_fft, dtype=complex)
        F[:n_half] = mags * np.exp(1j * phases)
        for k in range(1, N_fft // 2 + 1):
            F[N_fft - k] = np.conj(F[k])
        a_raw = np.fft.ifft(F).real
        offset = (N_fft - n) // 2
        a = np.abs(a_raw[offset:offset + n])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _run_lbfgs(a_init, deadline, aggressive=True, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted)) - 1]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                      0.031, 0.015, 0.0075, 0.00375,
                                      0.00187, 0.00094, 0.00047,
                                      0.00024, 0.00012, 0.00006,
                                      0.00003, 0.00001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001,
                                      0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.006
        elif n <= 16384:
            est_time_per_iter = 0.015
        else:
            est_time_per_iter = 0.03

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        _G['beta'] = float(beta)
        _G['n'] = n

        try:
            result = minimize(
                _lsobj,
                z,
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            n_active = 2 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1.0 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _peak_targeted_refine(a_init, deadline, max_rounds=80):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        n_peaks = min(20, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]

        w = np.zeros(2 * n - 1)
        for k in top_peaks:
            w[k] = 1.0 / n_peaks

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        grad = 2.0 * a * (h - W)

        f_old = c_max
        step = 0.15 / (1.0 + 0.005 * rnd)
        improved = False
        for ls in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            c_new = _full_convolve(a, a)
            if abs(float(c_new.max()) - f_old) < 1e-13:
                break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.88
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _solve_lp_reduced(b, n, active_indices, time_limit=10):
    """Solve LP: min t s.t. conv(a,b)_i <= t for i in active_indices, sum(a)=1, a>=0."""
    K = len(active_indices)
    if K == 0:
        return None, None

    A = lil_matrix((K, n + 1), dtype=np.float64)
    for idx, i in enumerate(active_indices):
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        for j in range(j_start, j_end + 1):
            A[idx, j] = b[i - j]
        A[idx, n] = -1.0

    A_csr = A.tocsr()
    b_vec = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_csr, b_ub=b_vec, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            a_sol = np.maximum(res.x[:n], 0)
            s = a_sol.sum()
            if s > 1e-15:
                a_sol = a_sol / s
            return a_sol, float(res.x[n])
    except Exception:
        pass
    return None, None


def _lp_refine(a_init, deadline, max_iters=5, k_peaks=50):
    """LP-based refinement: alternate solving LP with top-k active constraints."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        active_indices = np.where(active)[0]
        if len(active_indices) > k_peaks:
            top_vals = c[active_indices]
            n_keep = k_peaks
            top_in_active = np.argsort(top_vals)[::-1][:n_keep]
            active_indices = active_indices[top_in_active]

        a_lp, t_val = _solve_lp_reduced(a, n, active_indices, time_limit=15)
        if a_lp is None:
            break

        sc_lp = _score(a_lp)
        if sc_lp < best_sc:
            best_sc = sc_lp
            best_a = a_lp.copy()

        # Move toward LP solution
        a_new = 0.5 * a + 0.5 * a_lp
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-10:
            break

    return best_a, best_sc


def _random_peak_targeting(a_init, n_restarts, deadline, rng):
    """Try optimizing with random subsets of peaks to escape local minima."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for restart in range(n_restarts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_init, a_init)
        c_max = float(c.max())
        # Pick random peaks (mix of top and random)
        n_total = len(c)
        top_10 = np.argsort(c)[::-1][:10]
        rand_peaks = rng.integers(0, n_total, size=20)
        all_peaks = np.concatenate([top_10, rand_peaks])
        # Remove duplicates
        all_peaks = np.unique(all_peaks)
        # Shuffle and take a subset
        rng.shuffle(all_peaks)
        selected = all_peaks[:30]

        # Optimize with these peaks as "active set"
        a = a_init.copy()
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()

        for sub_it in range(50):
            if time.monotonic() > deadline:
                break
            a_cur = _softmax(z)
            c_cur = _full_convolve(a_cur, a_cur)
            # Weight by selected peaks
            w = np.zeros(2 * n - 1)
            for idx in selected:
                w[idx] = 1.0 / len(selected)
            a_rev = a_cur[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_cur))
            grad = 4.0 * n * a_cur * (h - W)
            step = 0.05 / (1.0 + 0.01 * sub_it)
            z = z - step * grad
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if sub_it % 10 == 0:
                a_check = _softmax(z)
                sc = _score(a_check)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a_check.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining spectral initialization, LP refinement,
    adaptive L-BFGS, active-set Newton, random peak targeting, and
    peak-targeted coordinate refinement across multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init_uniform")

    # === Phase 1: L-BFGS from parent with aggressive beta schedule ===
    print(f"\n[Phase 1] Aggressive L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_main = _run_lbfgs(parent, deadline - 500, aggressive=True, min_time_remaining=300.0)
        update_best(a_main, _score(a_main), "lbfgs_parent")

    # === Phase 2: LP-based refinement ===
    print(f"\n[Phase 2] LP refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_lp, sc_lp = _lp_refine(best_seq, min(deadline - 200, time.monotonic() + 120),
                                  max_iters=8, k_peaks=50)
        update_best(a_lp, sc_lp, "lp_refine")

    # === Phase 3: Spectral initializations ===
    print(f"\n[Phase 3] Spectral initializations (t={time.monotonic() - t_start:.1f}s)")
    n_main = len(best_seq) if best_seq is not None else 16384
    spectral_kinds = ["spectral", "spectral_mix", "ramp", "exp_ramp", "quadratic"]
    for kind in spectral_kinds:
        if time.monotonic() > deadline - 150:
            break
        a_spec = _make_profile(n_main, kind, rng)
        sc_spec = _score(a_spec)
        print(f"  {kind}: score={sc_spec:.10f}")

        if sc_spec < best_score + 0.1:
            a_spec_opt = _run_lbfgs(a_spec, min(deadline - 60, time.monotonic() + 120),
                                     aggressive=True, min_time_remaining=60.0)
            update_best(a_spec_opt, _score(a_spec_opt), f"struct_{kind}")

            # LP on spectral result
            if time.monotonic() < deadline - 50:
                a_spec_lp, sc_spec_lp = _lp_refine(a_spec_opt, min(deadline - 40, time.monotonic() + 40),
                                                    max_iters=3, k_peaks=30)
                update_best(a_spec_lp, sc_spec_lp, f"struct_{kind}_lp")

    # === Phase 4: Active-set Newton on current best ===
    print(f"\n[Phase 4] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 100, time.monotonic() + 120),
                                          max_iter=300, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton")

        # Tighter tolerance round
        if time.monotonic() < deadline - 80:
            a_as2, sc_as2 = _active_set_newton(a_as, min(deadline - 60, time.monotonic() + 60),
                                                max_iter=200, tol_active=0.0003)
            update_best(a_as2, sc_as2, "as_newton_tight")

    # === Phase 5: Peak-targeted refinement ===
    print(f"\n[Phase 5] Peak-targeted refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_pt, sc_pt = _peak_targeted_refine(best_seq, min(deadline - 50, time.monotonic() + 60),
                                             max_rounds=80)
        update_best(a_pt, sc_pt, "peak_refine")

    # === Phase 6: Random peak targeting for basin hopping ===
    print(f"\n[Phase 6] Random peak targeting (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_rpt, sc_rpt = _random_peak_targeting(best_seq, n_restarts=5,
                                                deadline=min(deadline - 40, time.monotonic() + 50),
                                                rng=rng)
        update_best(a_rpt, sc_rpt, "random_peak")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        for n_try in [8192, 4096]:
            if time.monotonic() > deadline - 40:
                break
            if best_seq is not None:
                a_interp = _interpolate_to_n(best_seq, n_try)
            else:
                a_interp = _make_profile(n_try, "uniform")
            sc_interp = _score(a_interp)
            a_opt = _run_lbfgs(a_interp, min(deadline - 30, time.monotonic() + 60),
                                aggressive=True, min_time_remaining=30.0)
            update_best(a_opt, _score(a_opt), f"n{n_try}_lbfgs")

            if time.monotonic() < deadline - 25:
                a_as_n, sc_as_n = _active_set_newton(a_opt, min(deadline - 20, time.monotonic() + 20),
                                                      max_iter=100, tol_active=0.001)
                update_best(a_as_n, sc_as_n, f"n{n_try}_as")

    # === Phase 8: Diverse restarts from best ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            a_r = _run_lbfgs(p_start, min(deadline - 10, time.monotonic() + 25),
                              aggressive=False, min_time_remaining=10.0)
            update_best(a_r, _score(a_r), f"restart_s{sigma}")

            if time.monotonic() < deadline - 8:
                a_r_as, sc_r_as = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 10),
                                                      max_iter=50, tol_active=0.001)
                update_best(a_r_as, sc_r_as, f"restart_s{sigma}_as")

    # === Phase 9: Final subgradient polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 3, n_max_steps=5000, lr_init=0.002)
        update_best(a_fin, sc_fin, "final_sg")

        # Final active-set with very tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _active_set_newton(a_fin, deadline - 1, max_iter=100, tol_active=0.0001)
            update_best(a_fin2, sc_fin2, "final_as_tight")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
