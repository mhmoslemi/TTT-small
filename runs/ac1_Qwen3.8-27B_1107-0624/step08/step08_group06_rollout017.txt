"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-start at low resolution (n=256) with structured shapes
2. Upscaling + adaptive L-BFGS with log-sum-exp smoothing
3. Peak-targeting equalization via averaged active-set gradient
4. Multi-n exploration (16384, 32768)
5. Final tight active-set Newton polish
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cross_conv(a, w, n):
    """Compute 2 * sum_k w[k] * a[|k-j|] for all j (gradient of active peaks)."""
    a_flip = a[::-1]
    out_len = len(w) + n - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_flip, N)
    result = np.fft.irfft(fw * fa, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad_c = 2.0 * _cross_conv(a, w, n)
    W = float(np.dot(a, grad_c))
    return a * (grad_c - W)


def _cg_solve(matvec_fn, rhs, max_cg=200, tol=1e-12, n_args=None):
    """Conjugate gradient solver. matvec_fn(v, *n_args) returns v @ matrix."""
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x, *n_args) if n_args else rhs
    p = r.copy()
    rs = np.dot(r, r)
    for _ in range(max_cg):
        if n_args:
            Ap = matvec_fn(p, *n_args)
        else:
            Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _hess_vp(v, A_vec, m, mu, n):
    """Hessian vector product for active-set Newton."""
    return (2.0 / m) * _cross_conv(v, A_vec, n) + mu * v


def _active_set_newton(a_init, deadline, max_iter=300, tol_list=None):
    """Adaptive active-set Newton with multiple tolerance levels."""
    if tol_list is None:
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005]
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None
        best_t = 0.0
        best_A = None
        best_sc_dir = np.inf

        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 8 * n:
                if m > 8 * n:
                    top_idx = np.argsort(c)[::-1][:8 * n]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                continue

            A_vec = active.astype(np.float64)
            g = (2.0 / m) * _cross_conv(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

            u = _cg_solve(_hess_vp, g_t, max_cg=150,
                          n_args=(A_vec, m, mu, n))
            w_vec = _cg_solve(_hess_vp, np.ones(n), max_cg=150,
                              n_args=(A_vec, m, mu, n))
            w_sum = float(w_vec.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w_vec
            dx = dx - np.mean(dx)

            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t = min(1.0, max(t_max, 1e-10))

            a_try = np.maximum(a + t * dx, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_dir:
                best_sc_dir = sc_try
                best_dir = dx
                best_t = t
                best_A = A_vec.copy()

        if best_dir is not None:
            f_old = c_max
            t = best_t
            accepted = False
            for ls in range(40):
                a_try = np.maximum(a + t * best_dir, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break
        else:
            # Fallback gradient step
            g_all = (2.0 / n) * _cross_conv(a, np.ones(2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            if np.linalg.norm(g_t) < 1e-14:
                break
            t = 0.05
            for ls in range(20):
                a_try = np.maximum(a - t * g_t, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                if _score(a_try) < best_sc - 1e-13:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _peak_shave(a, deadline, n_rounds=50):
    """Targeted peak shaving using averaged gradient of top peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks (within 2% of max)
        thresh = c_max * 0.98
        active = c >= thresh
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())
        if m < 1:
            break

        # Averaged gradient direction
        A_vec = active.astype(np.float64)
        g = (2.0 / m) * _cross_conv(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Line search
        t = 1.0
        for ls in range(30):
            t_try = t / (1.0 + 0.1 * rnd)
            a_try = np.maximum(a - t_try * g_t, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                a = a_try
                best_sc = sc_try
                best_a = a.copy()
                break
            t *= 0.5
        else:
            if t < 1e-8:
                break

    return best_a, best_sc


def _lbfgs_run(a_init, deadline, min_remaining=60.0, n_betas=None):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    if n_betas is None:
        n_betas = 14

    base_beta = max_c * 0.03
    ratios = np.concatenate([
        np.array([1.0, 0.7, 0.5, 0.35]),
        np.array([0.25, 0.18, 0.12, 0.08, 0.05, 0.035, 0.025]),
        np.array([0.015, 0.01, 0.006, 0.004, 0.002, 0.001, 0.0006])
    ])
    if len(ratios) > n_betas:
        ratios = ratios[:n_betas]
    betas = base_beta * ratios

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        est_per_iter = 0.0025 * max(1.0, n / 8192.0)
        max_iter = min(10000, max(30, int(remaining / (est_per_iter * 10))))

        try:
            result = minimize(
                _smooth_obj, z, args=(beta, n),
                jac=_smooth_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_shape(n, kind, param=1.0):
    """Generate a structured shape of length n, normalized to sum=1."""
    x = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "raised_cos":
        a = np.sin(np.pi * x) ** param
        a = np.maximum(a, 1e-10)
    elif kind == "gaussian":
        a = np.exp(-((x - 0.5) ** 2) / (2 * param ** 2))
        a = np.maximum(a, 1e-10)
    elif kind == "two_hump":
        a = np.exp(-((x - 0.3) ** 2) / (2 * param ** 2)) + \
            np.exp(-((x - 0.7) ** 2) / (2 * param ** 2))
        a = np.maximum(a, 1e-10)
    elif kind == "w_shape":
        a = 1.0 + param * np.cos(4 * np.pi * x)
        a = np.maximum(a, 1e-10)
    elif kind == "block":
        a = np.zeros(n)
        b = n // 4
        a[:b] = 1.0
        a[3 * b:] = 1.0
    elif kind == "three_block":
        a = np.zeros(n)
        b = n // 6
        a[:b] = 1.0
        a[2 * b:3 * b] = 1.0
        a[5 * b:] = 1.0
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def _interpolate(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_shape(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_shape(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
