def _cg_solve_hankel(S_ind, n, rhs, iterations=500, tol=1e-14):
    """Solve H*x = rhs where H[i,j] = 2*S_ind[i+j] using CG."""
    def matvec(v):
        return _hankel_matvec(v, S_ind, n)

    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(iterations):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

    best_sc = _score(a)

    # Adaptive tolerance schedule
    tol_schedule = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    tol_idx = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]

        # Active set: lags within tol of max
        active = c >= (c_max - tol * c_max)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())

        # Adaptive: if active set too large, tighten; if too small, loosen
        if m > 2 * n:
            top_idx = np.argsort(c)[::-1][:n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = int(active.sum())
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
        elif m < 3:
            tol_idx = max(0, tol_idx - 1)
            active = c >= (c_max - tol * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())

        S_ind = active.astype(np.float64)
        S_ind = S_ind / m  # uniform weights on active set

        # Gradient projected to tangent space
        g_raw = _hankel_matvec(a, S_ind, n)
        g_t = g_raw - np.mean(g_raw)
        gnorm = np.linalg.norm(g_t)

        if gnorm < 1e-15:
            break

        # Newton direction: solve (H + mu*I) * u = g_t, then project
        # H[v] = _hankel_matvec(v, S_ind, n) (which already includes 2*S_ind)
        # Add regularization
        mu = 1e-6 * c_max / m + 1e-10

        def hess_vp(v):
            return _hankel_matvec(v, S_ind, n) + mu * v

        # CG for u
        u = _cg_solve_hankel(S_ind, n, g_t, iterations=300, tol=1e-13)
        # But we need to add mu*I, so re-run with the modified operator
        # Actually let's just do CG with the full operator inline
        x = np.zeros(n)
        r = g_t - hess_vp(x)
        p = r.copy()
        rs = np.dot(r, r)
        for _ in range(300):
            Ap = hess_vp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x = x + alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-13 * (1.0 + np.dot(g_t, g_t)):
                break
            p = r + (rs_new / rs) * p
            rs = rs_new
        u = x

        # Also solve for the sum constraint
        w_vec = np.ones(n)
        x2 = np.zeros(n)
        r2 = w_vec - hess_vp(x2)
        p2 = r2.copy()
        rs2 = np.dot(r2, r2)
        for _ in range(300):
            Ap2 = hess_vp(p2)
            pAp2 = np.dot(p2, Ap2)
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            x2 = x2 + alpha2 * p2
            r2 = r2 - alpha2 * Ap2
            rs2_new = np.dot(r2, r2)
            if np.sqrt(rs2_new) < 1e-13:
                break
            p2 = r2 + (rs2_new / rs2) * p2
            rs2 = rs2_new
        w_sol = x2
        w_sum = float(w_sol.sum())

        if abs(w_sum) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / w_sum
            dx = -u + lam * w_sol

        dx = dx - np.mean(dx)

        # Line search
        t_max = 1.0
        neg = dx < -1e-12
        if neg.any():
            t_max = np.min(a[neg] / (-dx[neg])) * 0.99
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old - 1e-13 or ls >= 40:
                if f_try <= f_old + 1e-9:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback gradient step
            t = 0.05
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old - 1e-13:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                # Adjust tolerance and continue
                tol_idx = max(0, tol_idx - 1)
                continue

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            # Check KKT residual
            c_final = _full_convolve(a, a)
            cm = float(c_final.max())
            active_f = c_final >= (cm - 1e-8 * cm)
            if active_f.sum() > 0:
                g_check = _hankel_matvec(a, active_f.astype(float) / active_f.sum(), n)
                g_check_t = g_check - np.mean(g_check)
                if np.linalg.norm(g_check_t) < 1e-10:
                    break

        # Periodically tighten tolerance
        if it % 30 == 0 and it > 0:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation using log-sum-exp."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 5.0, max_c * 0.05)

    # More gradual beta schedule
    n_steps = 20
    betas = base_beta * np.power(0.7, np.arange(n_steps))
    betas = np.append(betas, [max_c * 0.001, max_c * 0.0005, max_c * 0.0002,
                               max_c * 0.0001, max_c * 0.00005, max_c * 0.00002,
                               max_c * 0.00001, max_c * 0.000005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Estimate time per iteration
        est_time = 0.05 * max(1, n / 4096.0)
        max_iter = min(4000, max(20, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad_simple(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_equalization(a, deadline, max_rounds=50):
    """Directly equalize top peaks by solving small LSQ systems."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks
        top_k = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        peak_vals = c[top_idx]

        # Target: bring all top peaks to the average of all peaks
        avg_peak = float(c.mean())
        target = max(avg_peak * 1.01, c_max * 0.999)

        # Identify which a[i] contribute most to the top peaks
        # For each peak at position k, the main contributors are pairs (i, k-i)
        # Reduce the larger contributor
        a_new = a.copy()
        reduction = 0.001
        for k in top_idx[:10]:
            if c[k] <= target:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            best_i = -1
            best_prod = 0
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k - i]
                if prod > best_prod:
                    best_prod = prod
                    best_i = i
            if best_i >= 0 and best_prod > 0:
                j = k - best_i
                if a_new[best_i] >= a_new[j]:
                    a_new[best_i] *= (1.0 - reduction)
                else:
                    a_new[j] *= (1.0 - reduction)

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try smaller reduction
            reduction *= 0.5
            if reduction < 1e-8:
                break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_structured_profile(n, kind="random_flat"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "random_flat":
        # Random sequence with values close to flat
        rng = np.random.default_rng(12345)
        a = 1.0 + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "chirp":
        t = np.arange(n) / n
        a = np.abs(np.sin(2 * np.pi * 0.5 * n * t + np.pi * n * t * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "windowed":
        t = np.linspace(0, 1, n)
        a = np.sinc(10 * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_scale":
        # Two-scale pattern: large values at sparse positions, small elsewhere
        a = np.full(n, 0.1)
        step = max(1, n // 100)
        a[::step] = 1.0
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. L-BFGS continuation from parent
    2. Active-set Newton with adaptive tolerance
    3. Peak equalization
    4. Multi-n exploration
    5. Final KKT polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_structured_profile(n_parent, "random_flat")
        print(f"No parent, using random_flat n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent (allocate ~40% of time)
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = deadline - 0.55 * (deadline - t_start)
    a_main = _run_lbfgs(parent, lbfgs_deadline, min_time_remaining=0.3 * (deadline - t_start))
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_phase1")

    # Phase 2: Active-set Newton (allocate ~25% of time)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_deadline = min(deadline - 0.2 * (deadline - t_start), time.monotonic() + 0.25 * (deadline - t_start))
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=500)
        update_best(a_as, sc_as, "as_newton_1")

        # Second round with tighter tolerance
        if time.monotonic() < deadline - 0.15 * (deadline - t_start):
            as_dl2 = min(deadline - 0.15 * (deadline - t_start), time.monotonic() + 0.1 * (deadline - t_start))
            a_as2, sc_as2 = _active_set_newton(a_as, as_dl2, max_iter=300)
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: Peak equalization
    print(f"\n[Phase 3] Peak equalization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 0.1 * (deadline - t_start):
        peq_dl = min(deadline - 0.1 * (deadline - t_start), time.monotonic() + 0.05 * (deadline - t_start))
        a_peq, sc_peq = _peak_equalization(best_seq, peq_dl, max_rounds=100)
        update_best(a_peq, sc_peq, "peak_eq")

        # Newton after equalization
        if sc_peq < best_score + 0.001 and time.monotonic() < deadline - 0.08 * (deadline - t_start):
            as_dl3 = min(deadline - 0.08 * (deadline - t_start), time.monotonic() + 0.04 * (deadline - t_start))
            a_peq2, sc_peq2 = _active_set_newton(a_peq, as_dl3, max_iter=200)
            update_best(a_peq2, sc_peq2, "peak_eq_newton")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent and best_seq is not None:
            continue

        # Try different initial profiles
        best_for_n = None
        best_sc_n = np.inf
        for kind in ["random_flat", "chirp"]:
            if time.monotonic() > deadline - 50:
                break
            if parent is not None and kind == "random_flat":
                a_init = _interpolate_to_n(parent, n_try)
            else:
                a_init = _make_structured_profile(n_try, kind)
            sc_init = _score(a_init)
            if sc_init < best_sc_n:
                best_sc_n = sc_init
                best_for_n = a_init

        if best_for_n is None:
            best_for_n = _make_structured_profile(n_try, "uniform")

        # L-BFGS on this n
        a_opt = _run_lbfgs(best_for_n, deadline, min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick Newton polish
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 20:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 10, time.monotonic() + 30),
                                                  max_iter=100)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_ns")

    # Phase 5: Restart and final polish
    print(f"\n[Phase 5] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Perturbation restarts
        for sigma in [0.01, 0.03]:
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_s{sigma}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 3, time.monotonic() + 15),
                                                  max_iter=100)
                update_best(a_r2, sc_r2, f"restart_s{sigma}_ns")

        # Final aggressive Newton
        if time.monotonic() < deadline - 3:
            a_fin, sc_fin = _active_set_newton(best_seq, deadline - 2, max_iter=300)
            update_best(a_fin, sc_fin, "final_ns")

        # Very final peak equalization
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _peak_equalization(best_seq, deadline - 1, max_rounds=50)
            update_best(a_fin2, sc_fin2, "final_peq")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        # Also validate with evaluate_sequence
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
