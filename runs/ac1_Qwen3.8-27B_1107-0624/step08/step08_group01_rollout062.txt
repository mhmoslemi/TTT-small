"""
Step-function coefficient optimization using LSE-smoothed L-BFGS with aggressive
annealing, coordinate-wise peak equalization, perturbation-based restarts, and
multi-length exploration. Targets improvement below 1.5027.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)

        return a_init

    # Use a wider gap for better conditioning

    gap = c_sorted[0] - c_sorted[idx]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.003)

    # Very aggressive schedule with more steps at the low end

        0.015, 0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
        0.000005, 0.000002, 0.000001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))

def _peak_equalize(a_init, deadline, max_sweeps=200):
    """Coordinate-wise peak equalization using the analytical update."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Compute initial convolution
    c = _full_convolve(a, a)
    
    no_improve = 0
    sweep = 0
    
    while time.monotonic() < deadline - 3.0 and sweep < max_sweeps:
        sweep += 1
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0
    else:

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        return _lse_grad(z, self.beta, self.n)

def _resample(a, new_n):

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    print(f"\n[P4] Perturbation restarts (t={time.monotonic()-t_start:.1f}s)")
    n_restarts = 6
    sigmas = [0.015, 0.025, 0.04, 0.01, 0.06, 0.035]

    for i in range(n_restarts):
        if time.monotonic() > deadline - 350.0:
            break
        remaining = deadline - 350.0 - time.monotonic()
        if remaining < 30.0:
            break

        # Alternate between random and structured perturbations
        if i % 2 == 0:
            a_pert = _perturb(best_a, sigmas[i % len(sigmas)], rng)
            label = f"rand_{sigmas[i % len(sigmas)]}"
        else:
            a_pert = _perturb_structured(best_a, rng, n_modes=2 + i % 3)
            label = f"struct_{i}"

        # Short L-BFGS from perturbed point
        a_opt = _lbfgs_aggressive(a_pert, deadline - 350.0, min_remaining=15.0, reheat=False)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  P4[{label}]: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            # Try active-set polish on the perturbed point too
            if remaining > 60.0:
                a_pol, sc_pol = _active_set_polish(a_opt, deadline - 350.0,
                                                   tol_start=0.002, tol_end=0.00005,
                                                   lr_start=0.015, top_k=4)
                if sc_pol < best_score:
                    best_a = a_pol.copy()
                    best_score = sc_pol
                    print(f"  P4[{label}_polish]: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [16384, 12288, 14336, 10240]
    if n_best not in n_targets:
        n_targets.append(n_best)
    n_targets = [x for x in n_targets if x != n_best]

    for n_try in sorted(n_targets):
        if time.monotonic() > deadline - 200.0:
            break
        remaining = deadline - 200.0 - time.monotonic()
        if remaining < 60.0:
            break

        a_init_n = _resample(best_a, n_try)
        sc_init = _score(a_init_n)
        print(f"  n={n_try}: init score={sc_init:.10f}, remaining={remaining:.0f}s")

        a_opt = _lbfgs_aggressive(a_init_n, deadline - 200.0, min_remaining=20.0, reheat=False)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  n={n_try}: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

        # Quick polish
        if time.monotonic() < deadline - 150.0:
            a_pol, sc_pol = _active_set_polish(a_opt, min(deadline - 150.0, time.monotonic() + 50.0),
                                               tol_start=0.003, tol_end=0.0001, lr_start=0.015, top_k=4)
            if sc_pol < best_score:
                best_a = a_pol.copy()
                best_score = sc_pol
                print(f"  n={n_try} polish: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        # Active-set with very fine tolerance
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 4.0,
                                           tol_start=0.0005, tol_end=0.000003,
                                           lr_start=0.006, top_k=3)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  P6 active: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

        # Final L-BFGS with tiny beta
        remaining = deadline - 4.0 - time.monotonic()
        if remaining > 5.0:
            n = len(best_a)
            c_best = _full_convolve(best_a, best_a)
            mb = float(c_best.max())
            beta_final = mb * 0.000008
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
            prob = _LseProblem(beta_final, n)
            max_iter = min(3000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-17, 'maxcor': 80, 'maxls': 80})
                a_fin2 = _softmax(result.x)
                a_fin2 = np.maximum(a_fin2, 0.0)
                s_fin2 = a_fin2.sum()
                if s_fin2 > 1e-10:
                    a_fin2 = a_fin2 / s_fin2
                    sc_fin2 = _score(a_fin2)
                    if sc_fin2 < best_score:
                        best_a = a_fin2.copy()
                        best_score = sc_fin2
                        print(f"  P6 lbfgs: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # One last tiny perturbation + optimization
        remaining = deadline - 3.0 - time.monotonic()
        if remaining > 8.0:
            a_tiny = _perturb(best_a, 0.005, rng)
            a_tiny_opt = _lbfgs_aggressive(a_tiny, deadline - 3.0, min_remaining=2.0, reheat=False)
            sc_tiny = _score(a_tiny_opt)
            if sc_tiny < best_score:
                best_a = a_tiny_opt.copy()
                best_score = sc_tiny
                print(f"  P6 tiny: NEW BEST {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
