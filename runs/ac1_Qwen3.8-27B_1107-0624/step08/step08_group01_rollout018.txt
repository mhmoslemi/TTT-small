3. Update: $a \leftarrow \text{project}(a - \eta \cdot g)$ where project means project onto $\{a \geq 0, \sum a = 1\}$
4. Use Polyak step size: $\eta_t = (f^* - f_t) / \|g_t\|^2$ where $f^*$ is the best known value

    g = wc[n - 1:2 * n - 1]

    g = g - np.mean(g)
    
    if eta is None:
        eta = 1.0 / (np.linalg.norm(g) + 1e-15)
    
    # Update and project
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, g
    a_new = a_new / s
    return a_new, g


def _subgradient_with_averaging(a_init, deadline, top_k=10, max_iter=100000):
    """
    Subgradient method with Polyak averaging.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Running average
    a_avg = np.zeros_like(a)
    count = 0
    
    # Polyak step size parameters
    f_lower = 2.0  # theoretical lower bound
    f_upper = best_score  # best known
    
    t = 0
    initial_a = a.copy()
    
    while time.monotonic() < deadline - 5.0 and t < max_iter:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max
        
        # Update best
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            f_upper = best_score
        
        # Compute subgradient
        k_eff = min(top_k, len(c))
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff:
            top_idx = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = k_eff
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        g = wc[n - 1:2 * n - 1]
        g = g - np.mean(g)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Polyak step size
        if f_upper > f_lower:
            eta = (f_upper - f_lower) / (gnorm * gnorm + 1e-30)
        else:
            eta = 1.0 / (np.sqrt(t + 1) * gnorm)
        
        # Cap the step
        eta = min(eta, 1.0 / (gnorm + 1e-15))
        
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Update average
        a_avg += a_new
        count += 1
        a_avg /= count
        
        a = a_new
        t += 1
        
        # Occasionally check the average
        if t % 1000 == 0:
            sc_avg = _score(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
    
    # Final check on average
    if count > 0:
        sc_avg = _score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()
    
    return best_a


def _adaptive_lbfgs(a_init, deadline, min_remaining=20.0):
    """Adaptive L-BFGS with LSE approximation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Initial beta based on the gap
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = max_c * 0.01
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Adaptive beta annealing
    beta = base_beta
    beta_min = max_c * 0.00001
    beta_decay = 0.7
    
    # We'll do multiple rounds with decreasing beta
    round_time = max(10.0, (deadline - min_remaining - time.monotonic()))
    n_rounds_est = int(np.log(beta_min / base_beta) / np.log(beta_decay)) + 1
    time_per_round = round_time / max(1, n_rounds_est)
    
    for rnd in range(max(30, n_rounds_est * 2)):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10.0:
            break
        
        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(3000, max(30, int(remaining / (est_iter_time * 10))))
        
        def obj_func(z_val):
            return _lse_obj(z_val, beta, n)
        
        def grad_func(z_val):
            return _lse_grad(z_val, beta, n)
        
        try:
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, reduce beta more aggressively
                beta *= beta_decay * 0.8
                if beta < beta_min:
                    break
            beta *= beta_decay
            if beta < beta_min:
                break
        except Exception:
            beta *= 0.5
            if beta < beta_min:
                break
    
    return current_a


def _active_set_polish(a_init, deadline, tol_start=0.005, tol_end=0.00001,
                       lr_start=0.03, top_k=5):
    """Active-set GD with top-k peaks and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    lr = lr_start
    tol = tol_start
    no_improve = 0
    max_stall = 200
    
    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 5:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False
        
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 10 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.005)
    
    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + np.abs(t - 0.5)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(42)
        a = rng.standard_normal(n)
        # Smooth it
        from scipy.ndimage import uniform_filter
        a = uniform_filter(a, size=max(2, n // 100))
        a = np.maximum(a, 0.0) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

"""
Optimized step-function search combining adaptive LSE-LBFGS, subgradient with
Polyak averaging, mirror descent, active-set polishing, and multi-n exploration.
Uses FFT-based convolution for speed and multiple diverse initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LSEFunctor:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _compute_subgradient(a, top_k=10, tol_frac=0.001):
    """Compute a subgradient of max(a*a) at point a, averaged over top-k active lags."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    k_eff = min(top_k, len(c))
    tol_val = tol_frac * c_max
    active = c >= (c_max - tol_val)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    if n_active > k_eff:
        top_idx = np.argsort(c)[::-1][:k_eff]
        active = np.zeros_like(c, dtype=bool)
        active[top_idx] = True
        n_active = k_eff
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    g = wc[n - 1:2 * n - 1]
    g = g - np.mean(g)
    return g, c_max, active


def _adaptive_lbfgs(a_init, deadline, min_remaining=20.0):
    """Adaptive L-BFGS with LSE approximation and geometric beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = max_c * 0.01
    base_beta = max(gap * 0.5, max_c * 0.005)
    beta_min = max_c * 0.000005

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    beta_decay = 0.65
    no_improve_count = 0

    while time.monotonic() + min_remaining < deadline and beta > beta_min:
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 8.0:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.0025 * scale
        max_iter = min(4000, max(50, int(remaining / (est_iter_time * 8))))

        functor = _LSEFunctor(beta, n)
        try:
            result = minimize(
                functor, z, jac=functor.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                beta *= 0.5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                no_improve_count = 0
                beta *= beta_decay
            else:
                no_improve_count += 1
                beta *= beta_decay * (0.7 if no_improve_count > 3 else 1.0)
                if no_improve_count > 8:
                    break
        except Exception:
            beta *= 0.5

    return current_a


def _subgradient_averaging(a_init, deadline, top_k=15, max_iter=80000):
    """Subgradient method with exponential averaging."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    a_avg = a.copy()
    weight_sum = 1.0
    decay = 0.9995

    t = 0
    eta_scale = 1.0 / np.sqrt(n)

    while time.monotonic() < deadline - 3.0 and t < max_iter:
        g, c_max, _ = _compute_subgradient(a, top_k=top_k, tol_frac=0.002)
        current_score = 2.0 * n * c_max

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        eta = eta_scale / (gnorm * np.sqrt(t + 1) + 1e-15)
        eta = min(eta, 0.5 / (gnorm + 1e-15))

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Exponential averaging
        a_avg = decay * a_avg + (1 - decay) * a_new
        weight_sum = decay * weight_sum + (1 - decay)

        a = a_new
        t += 1

        if t % 5000 == 0:
            sc_avg = _score(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()

    if t > 0:
        sc_avg = _score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

    return best_a


def _mirror_descent(a_init, deadline, top_k=10, max_iter=50000):
    """Mirror descent (exponentiated gradient) for the non-negativity constrained problem."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    z = np.log(np.maximum(a, 1e-30))

    best_a = a.copy()
    best_score = _score(a)

    t = 0
    while time.monotonic() < deadline - 3.0 and t < max_iter:
        g, c_max, _ = _compute_subgradient(a, top_k=top_k, tol_frac=0.002)
        current_score = 2.0 * n * c_max

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        # Mirror descent update in log-space
        eta = 2.0 / (np.linalg.norm(g) ** 2 * (t + 1) + 1e-15)
        eta = min(eta, 1.0)

        z_new = z - eta * g
        a_new = _softmax(z_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        z = np.log(np.maximum(a_new, 1e-30))

        a = a_new
        t += 1

    return best_a


def _active_set_polish(a_init, deadline, tol_start=0.004, tol_end=0.000005,
                       lr_start=0.025, top_k=5):
    """Active-set projected gradient with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    max_stall = 300

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 5:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(tol_end, tol * 0.65)
                lr *= 0.55
            if no_improve >= max_stall:
                break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.985)
            lr = min(lr_start, lr * 1.008)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + np.abs(t - 0.5)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-25.0 * (t - 0.15) ** 2) + np.exp(-25.0 * (t - 0.85) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = (1.0 + 0.25 * np.cos(2.0 * np.pi * 7.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 13.0 * t / n)
             + 0.08 * np.cos(2.0 * np.pi * 3.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(7)
        a = rng.standard_normal(n)
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.0) + 0.05
        return a / a.sum()
    elif kind == "pseudo_random":
        rng = np.random.default_rng(123)
        a = rng.uniform(0.5, 1.5, size=n)
        kernel_size = max(3, n // 80)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.0) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: adaptive LSE-LBFGS, subgradient averaging, mirror descent,
    active-set polishing, multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Adaptive L-BFGS on parent ===
    print(f"\n[P1] Adaptive L-BFGS parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 800.0, t_start + 350.0)
    a1 = _adaptive_lbfgs(best_a, p1_deadline, min_remaining=25.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [lbfgs_p1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Subgradient with averaging ===
    print(f"\n[P2] Subgradient averaging (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 550.0, t_start + 600.0)
    a2 = _subgradient_averaging(best_a, p2_deadline, top_k=20, max_iter=60000)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [subgrad_p2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS refinement after subgradient ===
    if time.monotonic() < deadline - 350.0:
        print(f"\n[P3] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
        p3_deadline = min(deadline - 250.0, t_start + 800.0)
        a3 = _adaptive_lbfgs(best_a, p3_deadline, min_remaining=20.0)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_a = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_p3] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Active-set polish ===
    print(f"\n[P4] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 200.0, t_start + 1000.0)
    a4, sc4 = _active_set_polish(best_a, p4_deadline, tol_start=0.003,
                                  tol_end=0.000005, lr_start=0.02, top_k=5)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [polish_p4] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Mirror descent exploration ===
    if time.monotonic() < deadline - 150.0:
        print(f"\n[P5] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
        p5_deadline = min(deadline - 100.0, t_start + 1150.0)
        a5 = _mirror_descent(best_a, p5_deadline, top_k=15, max_iter=40000)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_a = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [mirror_p5] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_current = len(best_a)
    n_candidates = sorted(set([8192, 12288, 16384, 20480, 24576]))
    n_candidates = [x for x in n_candidates if x != n_current]

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 80.0:
            break
        remaining_for_n = min(120.0, deadline - time.monotonic() - 40.0)
        if remaining_for_n < 25:
            continue

        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # L-BFGS on new n
        rem1 = remaining_for_n * 0.55
        a_opt = _adaptive_lbfgs(a_init, time.monotonic() + rem1, min_remaining=10.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            n_current = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs] t={time.monotonic()-t_start:.1f}s")

        # Quick subgradient
        rem2 = min(40.0, deadline - time.monotonic() - 20.0)
        if rem2 > 12:
            a_sg = _subgradient_averaging(a_opt, time.monotonic() + rem2, top_k=10, max_iter=20000)
            sc_sg = _score(a_sg)
            if sc_sg < best_score:
                best_a = a_sg.copy()
                best_score = sc_sg
                n_current = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_sg] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40.0:
        n_best = len(best_a)
        restarts = [
            ("pert_s", _perturb(best_a, 0.015, rng)),
            ("pert_m", _perturb(best_a, 0.04, rng)),
            ("tapered", _make_profile(n_best, "tapered")),
            ("cos_mod", _make_profile(n_best, "cosine_mod")),
            ("pseudo", _make_profile(n_best, "pseudo_random")),
        ]

        for label, a_r in restarts:
            if time.monotonic() > deadline - 25.0:
                break
            remaining = deadline - time.monotonic() - 15.0
            if remaining < 12.0:
                break
            rem_opt = min(remaining * 0.7, 50.0)
            a_opt = _adaptive_lbfgs(a_r, time.monotonic() + rem_opt, min_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 2.0:
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 1.5,
                                           tol_start=0.0002, tol_end=0.000002,
                                           lr_start=0.004, top_k=3)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
