"""

2*n*max(conv(a,a))/sum(a)^2. Key improvements: Nesterov momentum for faster

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_single(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a. Vectorized O(n)."""
    n = len(a)
    grad = np.zeros(n)
    # First part: a[k-j] for valid j
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += a[idx]
    # Second part: a[j-k] for valid j
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        idx = np.arange(j_lo2, j_hi2 + 1) - k
        grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _conv_grad_multi(a, active_indices):
    """Compute gradient of sum_{k in active} conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += a[idx]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += a[idx]
    return grad


def _nesterov_subgrad(a_init, deadline, min_time_remaining=60.0,
                      K_init=5, K_max=200, seed=0):
    """Nesterov-accelerated subgradient method with adaptive K."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.85
    base_lr = 0.005
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K: grow as peaks flatten
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.005 and K < K_max:
            K = min(K_max, int(K * 1.3) + 1)

        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Compute subgradient
        grad = _conv_grad_multi(a, active)
        # Project to tangent space of simplex (sum = 0)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Adaptive lr
        lr = min(base_lr, lr * 1.003)
        effective_lr = lr / (gnorm + 1e-18) * 0.1

        # Nesterov step
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
        else:
            # Backtrack
            v = 0.5 * v
            lr *= 0.9
            a = a_proj  # still stay on simplex

        # Periodic restart to escape local minima
        if no_improve % 200 == 0 and no_improve > 0:
            stall_count += 1
            if stall_count > 6:
                break
            # Small random perturbation
            perturb = 0.001 * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0

        if t % 500 == 0:
            print(f"    Nesterov t={t}: best={best_sc:.10f}, cur={sc_new:.10f}, K={K}, lr={lr:.6f}")

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_fine(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very fine beta schedule and more iterations."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    # Finer beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03125, 0.015, 0.0075, 0.00375,
                                  0.0019, 0.00095, 0.0005, 0.00025,
                                  0.00012, 0.00006, 0.00003, 0.000015])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # More iterations for smaller beta
        if i < len(betas) // 2:
            est_time_per_iter = 0.004 * max(1.0, n / 8192.0)
            max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 10))))
        else:
            est_time_per_iter = 0.005 * max(1.0, n / 8192.0)
            max_iter = min(4000, max(30, int(remaining / (est_time_per_iter * 8))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 4 == 0:
                    print(f"    L-BFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _direct_peak_equalize(a_init, deadline, min_time_remaining=30.0):
    """Directly equalize top peaks by moving mass from peak contributors
    to non-contributors. Uses 1D water-filling logic."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 0.02
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            stall_count += 1
            if stall_count > 10:
                break
            lr *= 0.5
            no_improve = 0

        # Find top peaks
        K = max(3, int(n * 0.005))
        c_idx = np.argsort(c)[::-1]
        active_k = c_idx[:K]
        peak_vals = c[active_k]

        # For each active peak, find which a[j] contribute most
        # and which contribute least
        high_contrib = np.zeros(n)
        low_contrib = np.zeros(n)
        for k in active_k:
            g = _conv_grad_single(a, k)
            high_contrib += g
        # Normalize
        high_contrib = high_contrib / (np.max(high_contrib) + 1e-30)

        # Move mass from high contributors to low contributors
        # Low contributors are those with low total gradient across all peaks
        all_grad = np.zeros(n)
        for k in active_k:
            all_grad += _conv_grad_single(a, k)
        low_contrib_idx = np.argsort(all_grad)[:max(5, int(n * 0.01))]
        high_contrib_idx = np.argsort(all_grad)[::-1][:max(5, int(n * 0.01))]

        # Compute how much to move
        total_grad_high = all_grad[high_contrib_idx].sum()
        total_grad_low = all_grad[low_contrib_idx].sum()
        if total_grad_high < 1e-15:
            continue
        move_amount = lr * (c_max - peak_vals.min()) / total_grad_high
        move_amount = min(move_amount, 0.01 * a.max())

        # Apply
        n_high = len(high_contrib_idx)
        n_low = len(low_contrib_idx)
        for j in high_contrib_idx:
            a[j] -= move_amount / n_high
        for j in low_contrib_idx:
            a[j] += move_amount * n_high / (n_low * a[j] + 1e-30) * 0.01

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 50)
        for i in range(taper):
            a[i] = (i + 1) / taper
            a[n - 1 - i] = (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Nesterov + L-BFGS hybrid optimizer with adaptive active sets."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Nesterov subgradient (main workhorse) ===
    print(f"\n[P1] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        p1_deadline = min(deadline - 800, t_start + 400)
        a_nes, sc_nes = _nesterov_subgrad(best_seq, p1_deadline,
                                          min_time_remaining=60.0,
                                          K_init=10, K_max=300, seed=seed)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Nesterov: no improvement (best={best_score:.10f})")

    # === Phase 2: Fine L-BFGS ===
    print(f"\n[P2] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p2_deadline = min(deadline - 500, t_start + 600)
        a_lb = _run_lbfgs_fine(best_seq, p2_deadline, min_time_remaining=60.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Second Nesterov pass (finer) ===
    print(f"\n[P3] Nesterov round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, t_start + 800)
        a_nes2, sc_nes2 = _nesterov_subgrad(best_seq, p3_deadline,
                                            min_time_remaining=60.0,
                                            K_init=5, K_max=100, seed=seed + 1)
        if sc_nes2 < best_score:
            best_seq = a_nes2.copy()
            best_score = sc_nes2
            print(f"  NEW BEST: {best_score:.10f} [nesterov2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS with even finer beta ===
    print(f"\n[P4] Ultra-fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, t_start + 1000)
        a_lb2 = _run_lbfgs_fine(best_seq, p4_deadline, min_time_remaining=60.0)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Direct peak equalization ===
    print(f"\n[P5] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p5_deadline = min(deadline - 200, t_start + 1100)
        a_eq, sc_eq = _direct_peak_equalize(best_seq, p5_deadline, min_time_remaining=60.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [equalize] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 12288, 8192, 24576]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        if time.monotonic() < deadline - 80:
            a_opt = _run_lbfgs_fine(a_init, deadline, min_time_remaining=60.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        if time.monotonic() < deadline - 60:
            a_nes_n, sc_nes_n = _nesterov_subgrad(a_opt if 'a_opt' in dir() else a_init,
                                                  deadline, min_time_remaining=50.0,
                                                  K_init=5, K_max=100, seed=seed + n_try)
            if sc_nes_n < best_score:
                best_seq = a_nes_n.copy()
                best_score = sc_nes_n
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_nest] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Shrinkage ===
    print(f"\n[P7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 60)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 50:
                a_shr2 = _run_lbfgs_fine(best_seq, deadline, min_time_remaining=30.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final Nesterov + L-BFGS polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        a_fin_n, sc_fin_n = _nesterov_subgrad(best_seq, deadline - 10,
                                              min_time_remaining=10.0,
                                              K_init=3, K_max=50, seed=seed + 100)
        if sc_fin_n < best_score:
            best_seq = a_fin_n.copy()
            best_score = sc_fin_n
            print(f"  NEW BEST: {best_score:.10f} [final_nest] t={time.monotonic()-t_start:.1f}s")

    # Final: tiny L-BFGS on best
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(10, len(c_sorted)-1)]
        beta_fin = max(gap * 0.01, mb * 0.00001)
        prob = _LbfgsProb(beta_fin, n)
        max_iter = min(2000, max(20, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 60})
            a_fin2 = _softmax(result.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
