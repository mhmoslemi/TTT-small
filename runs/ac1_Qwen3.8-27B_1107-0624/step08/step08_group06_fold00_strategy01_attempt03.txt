            if new_score < self.best_score:
                self.update_best(new_sequence)
                # Early exit if significant improvement
                if abs(new_score - self.best_score_o) < 1e-4:
                    break
            self.elapsed += 0.1
        
        return self.best_sequence

    def __call__(self, seed=None):
        self.seed = seed
        if seed is None:
            self.n = 8192
            self.x0 = self._initial_uniform_sequence()
        else:
            self.x0 = seed
        
        self.best_sequence = self.x0
        self.best_score = np.inf
        self.elapsed = 0
        
        # Start the brainstorm
        self.progress = []
        for _ in range(100):
            self.x0_log += 0.001 * np.random.standard_normal(len(self.x0_log))
            self.elapsed += time.time()
            
            # Ensure we keep track of improvements
            new_sequence = self.x0_log - 1.0
            if new_sequence.sum() < 1e-4:
                self.elapsed -= time.time()
                self.perturb_sequence()
            else:
                self.x0 = self._ensure_valid_sequence(new_sequence)
                new_score = evaluate_sequence(list(self.x0))
                if new_score < self.best_score:
                    self.best_sequence = self.x0
                    self.best_score = new_score
                if (time.time() - self.elapsed) > 10 * 60:  # 10 mins passed
                    break
        
        # Final pass to ensure we find the best candidate
        self.x0_log = self.perturb_sequence()
        self.x0 = self.x0_log
        
        # If no significant improvement, return the initial one
        if self.elapsed < self.deadline - 10.0 and self.best_score < evaluate_sequence(list(self.x0[:self.n])):
            self.x0 = self.x0_log * (1 if self.n < 4*len(self.x0) else 4)
            self.x0 = self._ensure_valid_sequence(self.x0)
            self.best_score = evaluate_sequence(list(self.x0))
            self.best_sequence = self.x0
        
        return self.best_sequence.tolist()
