The "ideal" solution would have all `c[k] = 1/(2n-1)`. This requires:

- `sum_i a[i]*a[n-1-i] = 1/(2n-1)`

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Diverse structured initializations (cosine tapers, periodic patterns, blocks)
2. Multi-resolution optimization (coarse to fine)
3. Adaptive log-sum-exp smoothed L-BFGS with entropy-based beta scheduling
4. Iterative sparsification (zero out small elements, re-optimize)
5. Active-set Newton polish with adaptive tolerance

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind="uniform", **kwargs):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine_taper":
        p = kwargs.get('p', 0.5)
        i = np.arange(n)
        a = np.sin(np.pi * (i + 0.5) / n) ** p
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "periodic":
        freq = kwargs.get('freq', 1)
        eps = kwargs.get('eps', 0.3)
        i = np.arange(n)
        a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "cos_periodic":
        freq = kwargs.get('freq', 1)
        eps = kwargs.get('eps', 0.3)
        p = kwargs.get('p', 1.0)
        i = np.arange(n)
        base = np.sin(np.pi * (i + 0.5) / n) ** p
        a = base * (1.0 + eps * np.cos(2.0 * np.pi * freq * i / n))
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "block":
        n_blocks = kwargs.get('n_blocks', 3)
        gap = kwargs.get('gap', 0.0)
        a = np.zeros(n)
        block_total = n / (n_blocks * (1 + gap))
        for blk in range(n_blocks):
            start = int(blk * n / n_blocks)
            end = int(start + block_total)
            end = min(end, n)
            a[start:end] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "gaussian":
        sigma = kwargs.get('sigma', 0.3)
        i = np.arange(n)
        t = (i - n/2) / (sigma * n)
        a = np.exp(-t**2)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta based on effective support."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Initial beta based on the gap
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    else:
        gap = c_sorted[0]
    base_beta = max(gap, max_c * 0.05)
    
    # Beta schedule: start large, decrease geometrically
    n_betas = 20
    betas = base_beta * (0.7 ** np.arange(n_betas))
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(8000, max(20, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 150,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, break early
                if i > 3:
                    break
        except Exception:
            pass

    return current_a


def _sparsify_and_reoptimize(a, deadline, min_time_remaining=30.0, sparsity_levels=(0.05, 0.1, 0.15, 0.2)):
    """Zero out small elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for frac in sparsity_levels:
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        a_sp = a.copy()
        n_zero = int(n * frac)
        if n_zero < 1:
            continue
        # Zero out the smallest elements
        thresh = np.sort(a_sp)[n_zero]
        a_sp[a_sp <= thresh] = 0
        s = a_sp.sum()
        if s < 1e-15:
            continue
        a_sp = a_sp / s
        
        sc_sp = _score(a_sp)
        if sc_sp < best_sc - 1e-12:
            # Re-optimize with L-BFGS
            a_opt = _run_adaptive_lbfgs(a_sp, deadline, min_time_remaining)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()
    
    return best_a, best_sc


def _active_set_newton_v2(a_init, deadline, max_iter=300):
    """Active-set Newton with improved line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None
        best_t = 0
        best_sc_try = np.inf

        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 6 * n:
                if m > 6 * n:
                    top_idx = np.argsort(c)[::-1][:6 * n]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                continue

            A_vec = active.astype(np.float64)

            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                continue

            mu = 1e-6 * (c_max / max(m, 1)) + 1e-9

            # Conjugate gradient solve
            def _hvp(v):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

            # CG for direction
            x = np.zeros(n)
            r = g_t - _hvp(x)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(150):
                Ap = _hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13:
                    r = r_new
                    rs = rs_new
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new

            # Solve for the normalization constraint
            w = np.zeros(n)
            r = np.ones(n) - _hvp(w)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(150):
                Ap = _hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                w = w + alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13:
                    r = r_new
                    rs = rs_new
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new

            w_sum = float(w.sum())
            if abs(w_sum) < 1e-15:
                dx = -x
            else:
                lam = float(x.sum()) / w_sum
                dx = -x + lam * w

            dx = dx - np.mean(dx)

            # Line search
            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.999
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)

            t = min(1.0, t_max)
            for _ in range(40):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc_try:
                    best_sc_try = sc_try
                    best_dir = dx
                    best_t = t
                if sc_try <= c_max / (a.sum()**2) - 1e-12:  # sufficient decrease
                    break
                t *= 0.5

        if best_dir is None or best_sc_try >= best_sc - 1e-15:
            # No improvement from Newton, try a simple gradient step
            g_all = (2.0 / n) * _corr_a_A(a, np.ones(2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            t = 0.1
            improved = False
            for _ in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                    improved = True
                    break
                t *= 0.5
            if not improved:
                break
        else:
            a = a + best_t * best_dir
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
            elif sc < best_sc + 1e-12:
                a = a  # accept anyway for progress
            else:
                break

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _multi_fstart(a_init, deadline, min_time_remaining=60.0, n_f_starts=5):
    """Multiple frequency perturbations followed by optimization."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    i = np.arange(n)
    for f in range(1, n_f_starts + 1):
        if time.monotonic() + min_time_remaining > deadline:
            break

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Diverse structured initializations (cosine tapers, periodic patterns)
2. Momentum-based projected gradient warm-up

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind="uniform", p=0.5, freq=1, eps=0.3, n_blocks=3, gap=0.0, sigma=0.3):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine_taper":
        i = np.arange(n)
        a = np.sin(np.pi * (i + 0.5) / n) ** p
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "periodic":
        i = np.arange(n)
        a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "cos_periodic":
        i = np.arange(n)
        base = np.sin(np.pi * (i + 0.5) / n) ** p
        a = base * (1.0 + eps * np.cos(2.0 * np.pi * freq * i / n))
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        for blk in range(n_blocks):
            start = int(blk * n / n_blocks)
            length = int(n / (n_blocks * (1.0 + gap)))
            end = min(start + length, n)
            a[start:end] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:

            a = np.ones(n) / n

        t = (i - n / 2.0) / (sigma * n)

    elif kind == "triangle":

                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 8 * n:
                if m > 8 * n:
                    top_idx = np.argsort(c)[::-1][:8 * n]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                continue

            A_vec = active.astype(np.float64)
            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                continue

            mu = 1e-5 * (c_max / max(m, 1)) + 1e-9

            def _hvp(v):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

            # CG for direction
            x = np.zeros(n)
            r = g_t
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(200):
                Ap = _hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12:
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new

            # CG for normalization
            w = np.zeros(n)
            r_w = np.ones(n)
            p_w = r_w.copy()
            rs_w = np.dot(r_w, r_w)
            for _ in range(200):
                Ap_w = _hvp(p_w)
                pAp_w = np.dot(p_w, Ap_w)
                if pAp_w < 1e-30:
                    break
                alpha_w = rs_w / pAp_w
                w = w + alpha_w * p_w
                r_w_new = r_w - alpha_w * Ap_w
                rs_w_new = np.dot(r_w_new, r_w_new)
                if np.sqrt(rs_w_new) < 1e-12:
                    break
                p_w = r_w_new + (rs_w_new / rs_w) * p_w
                r_w = r_w_new
                rs_w = rs_w_new

            w_sum = float(w.sum())
            if abs(w_sum) < 1e-15:
                dx = -x
            else:
                lam = float(x.sum()) / w_sum
                dx = -x + lam * w

            dx = dx - np.mean(dx)
            dnorm = np.linalg.norm(dx)
            if dnorm < 1e-15:
                continue

            # Backtracking line search
            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.999
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)

            t = min(1.0, t_max)
            for _ in range(50):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc_try:
                    best_sc_try = sc_try
                    best_dir = dx.copy()
                    best_t = t
                if sc_try < sc_cur * (1.0 - 1e-10):
                    break
                t *= 0.5

        if best_dir is not None and best_sc_try < best_sc - 1e-14:
            a = a + best_t * best_dir
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # Gradient fallback
            g_all = (2.0 / n) * _corr_a_A(a, np.ones(2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            t = 0.05
            improved = False
            for _ in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                    improved = True
                    stagnation = 0
                    break
                t *= 0.5
            if not improved:
                stagnation += 1
                if stagnation > 10:
                    break

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _sparsify_and_reoptimize(a, deadline, min_time_remaining=30.0):
    """Zero out small elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for frac in [0.02, 0.05, 0.08, 0.12, 0.15, 0.2]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_sp = a.copy()
        n_zero = int(n * frac)
        if n_zero < 1:
            continue
        sorted_vals = np.sort(a_sp)
        thresh = sorted_vals[n_zero]
        a_sp[a_sp <= thresh] = 0.0
        s = a_sp.sum()
        if s < 1e-15:
            continue
        a_sp = a_sp / s
        sc_sp = _score(a_sp)
        if sc_sp < best_sc - 1e-12:
            a_opt = _run_adaptive_lbfgs(a_sp, deadline, min_time_remaining)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a, best_sc


def _freq_perturb_optimize(a_init, deadline, min_time_remaining=60.0, n_freqs=6):
    """Try frequency perturbations and optimize each."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    i = np.arange(n)

    for f in range(1, n_freqs + 1):
        if time.monotonic() + min_time_remaining > deadline:
            break
        eps = 0.15 + 0.08 * f
        pert = eps * np.cos(2.0 * np.pi * f * i / n)
        a_pert = a_init * (1.0 + pert)
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        # Quick momentum PG first
        a_pg = _momentum_pg(a_pert, min(deadline - 30, time.monotonic() + 60), max_steps=2000)
        # Then L-BFGS
        a_opt = _run_adaptive_lbfgs(a_pg, deadline, min_time_remaining)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc
