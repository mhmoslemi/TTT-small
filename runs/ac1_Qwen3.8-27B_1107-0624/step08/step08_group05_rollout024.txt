import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

    c_safe = np.maximum(c, 1e-30)

def _power_sum_grad(z, p, n):

    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = np.power(c_safe, p - 1)

    a_rev = a[::-1]

    sum_cp = np.sum(np.power(c_safe, p))
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p_vec = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    old_pH = 0.0
    for _ in range(max_iter):
        Hp = hess_func(p_vec)
        pH = float(np.dot(p_vec, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p_vec
        r_new = r - alpha * Hp
        r_norm = float(np.dot(r_new, r_new))
        if np.sqrt(r_norm) < tol * b_norm:
            r = r_new
            break
        # Barzilai-Borwein step size for CG
        if pH > 1e-30 and abs(pH - old_pH) > 1e-30:
            gamma = abs(pH) / abs(pH - old_pH)
        else:
            gamma = 1.0
        old_pH = pH
        r = r_new
        rs_new = float(np.dot(r, r))
        beta_cg = (rs_new / rs) * (1.0 + gamma * 0.1) if rs > 1e-30 else 0.0
        beta_cg = min(beta_cg, 2.0)
        p_vec = r + beta_cg * p_vec
        rs = rs_new
    return x

        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-14:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        if it > 5 and abs(sc - best_sc) < 1e-12 and gnorm < 1e-10:
            break

    return best_a, best_sc


def _lbfgs_lse_staged(a_init, deadline, min_remaining=30.0, n_stages=25):
    """L-BFGS with LSE smoothing, 25-stage geometric schedule."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    # Estimate spread of top convolution values
    c_sorted = np.sort(c_cur)[::-1]
    k_ref = min(100, len(c_sorted) - 1)
    gap = max(c_sorted[0] - c_sorted[k_ref], max_c * 0.001)
    base_beta = max(gap * 0.5, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # 25-stage geometric schedule with ratio 0.85
    ratio = 0.85
    betas = base_beta * np.array([ratio ** i for i in range(n_stages)])

    for stage, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        # Estimate time per iteration
        est_time = 0.004 * max(1.0, n / 4096.0)
        # More iterations for earlier (higher beta) stages
        iter_factor = max(0.3, 1.0 - 0.03 * stage)
        max_iter = min(5000, max(200, int(remaining * iter_factor / (est_time * 6))))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                jac=_lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, try with slightly different beta
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lbfgs_power_sum_staged(a_init, deadline, min_remaining=30.0):
    """L-BFGS with power-sum smoothing."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [2, 4, 8, 16, 32, 64, 128, 256, 512]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(200, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _subgrad_polish(a_init, deadline, n_steps=8000, lr_init=0.01, tol_frac=0.001):
    """Subgradient method on log-softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = _normalize(a_init.copy())
    best_sc = _score(best_a)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 300 == 0 or i == n_steps - 1:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                best_patience_sc = sc
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = max(lr_init * 0.3, lr * 0.3)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _spectral_perturb(a, rng, target_lags=None, n_frequencies=20, max_shift=0.3):
    """Perturb in frequency domain, targeting specific lag bottlenecks."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    # Perturb amplitude at selected frequency bins
    n_bins = len(A)
    if target_lags is not None:
        # Focus perturbation on frequencies that affect target lags
        freq_indices = rng.choice(n_bins, size=min(n_frequencies, n_bins), replace=False)
    else:
        freq_indices = np.arange(1, min(n_frequencies + 1, n_bins))
    A_new = A.copy()
    for fi in freq_indices:
        A_new[fi] *= (1.0 + max_shift * rng.standard_normal())
    a_new = np.fft.irfft(A_new, N)[:n]
    return _normalize(a_new)


def _log_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _block_perturb(a, rng, n_blocks=8):
    """Randomly permute blocks of the log-transformed sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = n // n_blocks
    if block_size < 4:
        return a.copy()
    for _ in range(min(4, n_blocks)):
        i = int(rng.integers(0, n_blocks))
        j = int(rng.integers(0, n_blocks))
        if i == j:
            continue
        si, si_end = i * block_size, (i + 1) * block_size
        sj, sj_end = j * block_size, (j + 1) * block_size
        tmp = z[si:si_end].copy()
        z[si:si_end] = z[sj:sj_end]
        z[sj:sj_end] = tmp
    return _softmax(z)


def _asymmetric_perturb(a, rng, strength=0.1):
    """Perturb to break symmetry - add a linear ramp component."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    ramp = np.linspace(-1, 1, n)
    z = z + strength * ramp * rng.uniform(0.5, 1.5)
    z = z - z.mean()
    return _softmax(z)


def _resize_sequence(a, n_target):
    """Resize sequence using linear interpolation, preserving sum=1."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    return _normalize(a_target)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "step":
        a = np.ones(n)
        a[n // 2:] = 1.5
        return a / a.sum()
    elif kind == "asym_step":
        a = np.ones(n)
        a[n // 3:] = 1.3
        return a / a.sum()
    elif kind == "smooth_ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        return a / a.sum()
    elif kind == "cos_ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(np.pi * t)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.03) + 0.7 * np.exp(-((t - 0.7) ** 2) / 0.03)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _variance_obj(a):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    return float(np.mean((c - mean_c) ** 2))


def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    return g - np.mean(g)


def _variance_refine(a_init, deadline, n_steps=500):
    """Secondary optimization: minimize variance of convolution."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.002

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad = _variance_grad(a, n)
        a_new = a - lr * grad
        a_new = _normalize(a_new)
        v_new = _variance_obj(a_new)
        v_old = _variance_obj(a)
        if v_new < v_old:
            a = a_new
            lr = min(lr * 1.01, 0.008)
        else:
            lr *= 0.8
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _identify_bottlenecks(a, top_k=10):
    """Identify the lags where convolution is highest (bottlenecks)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = 0.01 * c_max
    active = np.where(c >= c_max - tol)[0]
    return active[:top_k], c, c_max


def _propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-resolution optimizer with adaptive LSE, spectral perturbations,
    bottleneck-informed restarts, and Barzilai-Borwein Newton.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = float('inf')

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init")

    # === Phase 1: Multi-resolution warm-up (120s) ===
    print(f"\n[Phase 1] Multi-resolution warm-up")
    phase1_deadline = min(deadline, t_start + 120)

    if _best_seq is not None:
        # Coarse optimization at n/4
        n_coarse = max(1024, len(_best_seq) // 4)
        a_coarse = _resize_sequence(_best_seq, n_coarse)
        a_coarse, sc_coarse = _lbfgs_lse_staged(a_coarse, phase1_deadline, min_remaining=60, n_stages=12)
        upd(a_coarse, sc_coarse, "coarse_lse")

        # Medium optimization at n/2
        if time.monotonic() < phase1_deadline - 30:
            n_med = max(2048, len(_best_seq) // 2)
            a_med = _resize_sequence(_best_seq, n_med)
            a_med, sc_med = _lbfgs_lse_staged(a_med, phase1_deadline, min_remaining=25, n_stages=10)
            upd(a_med, sc_med, "med_lse")

        # Upscale coarse back to full size and check
        if time.monotonic() < phase1_deadline - 10:
            a_up = _resize_sequence(a_coarse, len(_best_seq))
            a_up, sc_up = _newton_active_set(a_up, phase1_deadline, max_iter=50, tol_frac=0.003)
            upd(a_up, sc_up, "upscale_newton")

    # === Phase 2: Main LSE optimization (400s) ===
    print(f"\n[Phase 2] Main LSE (25 stages)")
    phase2_deadline = min(deadline, time.monotonic() + 400)
    if _best_seq is not None:
        a_main, sc_main = _lbfgs_lse_staged(_best_seq, phase2_deadline, min_remaining=30, n_stages=25)
        upd(a_main, sc_main, "main_lse_25")

    # === Phase 3: Power-sum refinement (150s) ===
    print(f"\n[Phase 3] Power-sum L-BFGS")
    phase3_deadline = min(deadline, time.monotonic() + 150)
    if _best_seq is not None:
        a_ps, sc_ps = _lbfgs_power_sum_staged(_best_seq, phase3_deadline, min_remaining=20)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 4: Newton active-set (200s) ===
    print(f"\n[Phase 4] Newton active-set (tight)")
    phase4_deadline = min(deadline, time.monotonic() + 200)
    if _best_seq is not None:
        # Run with progressively tighter tolerance
        a_cur = _best_seq.copy()
        for tol_i, tol_v in enumerate([0.003, 0.001, 0.0005, 0.0002]):
            if time.monotonic() > phase4_deadline - 20:
                break
            nw_dead = min(phase4_deadline - 15, time.monotonic() + 50)
            a_nw, sc_nw = _newton_active_set(a_cur, nw_dead, max_iter=150, tol_frac=tol_v)
            upd(a_nw, sc_nw, f"newton_tol{tol_v}")

    # === Phase 5: Subgradient polish (150s) ===
    print(f"\n[Phase 5] Subgradient polish")
    phase5_deadline = min(deadline, time.monotonic() + 150)
    if _best_seq is not None:
        a_sg, sc_sg = _subgrad_polish(_best_seq, phase5_deadline, n_steps=10000, lr_init=0.008, tol_frac=0.0008)

    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:

            break
        if len(_best_seq) == n_try:
            a_init_n = _best_seq.copy()

            a_init_n = _resize_sequence(_best_seq, n_try)

        # Quick LSE
        nse_dead = min(phase6_deadline - 30, time.monotonic() + 80)
        a_opt_n, sc_opt_n = _lbfgs_lse_staged(a_init_n, nse_dead, min_remaining=20, n_stages=15)
        upd(a_opt_n, sc_opt_n, f"n{n_try}_lse")

        # Newton polish
        if sc_opt_n < _best_score * 1.01 and time.monotonic() < phase6_deadline - 20:
            nw_dead = min(phase6_deadline - 15, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=100, tol_frac=0.001)
            upd(a_opt_n2, sc_opt_n2, f"n{n_try}_newton")

    # === Phase 7: Bottleneck-informed restarts (200s) ===
    print(f"\n[Phase 7] Bottleneck-informed restarts")
    phase7_deadline = min(deadline, time.monotonic() + 200)
    if _best_seq is not None:
        # Identify bottlenecks in current best
        bn_lags, c_bn, c_max_bn = _identify_bottlenecks(_best_seq, top_k=20)
        print(f"  Bottleneck lags (offset from center): {bn_lags - len(_best_seq) + 1}")

        for i in range(8):
            if time.monotonic() > phase7_deadline - 30:
                break
            if i == 0:
                p_start = _asymmetric_perturb(_best_seq, rng, strength=0.05)
            elif i == 1:
                p_start = _spectral_perturb(_best_seq, rng, target_lags=bn_lags, n_frequencies=30, max_shift=0.2)
            elif i == 2:
                p_start = _log_perturb(_best_seq, 0.03, rng)
            elif i == 3:
                p_start = _block_perturb(_best_seq, rng, n_blocks=16)
            elif i == 4:
                p_start = _asymmetric_perturb(_best_seq, rng, strength=-0.05)
            elif i == 5:
                # Reflect + perturb
                p_start = _log_perturb(_best_seq[::-1], 0.02, rng)
            elif i == 6:
                p_start = _spectral_perturb(_best_seq, rng, n_frequencies=50, max_shift=0.15)
            else:
                p_start = _log_perturb(_best_seq, 0.05, rng)

            # Quick LSE on perturbed
            r_dead = min(phase7_deadline - 20, time.monotonic() + 30)
            a_r, sc_r = _lbfgs_lse_staged(p_start, r_dead, min_remaining=10, n_stages=8)
            upd(a_r, sc_r, f"restart_{i}")

            # Newton if promising
            if sc_r < _best_score * 1.005 and time.monotonic() < phase7_deadline - 10:
                nw_dead = min(phase7_deadline - 8, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=60, tol_frac=0.002)
                upd(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 8: Variance-based secondary refinement (80s) ===
    print(f"\n[Phase 8] Variance refinement")
    phase8_deadline = min(deadline, time.monotonic() + 80)
    if _best_seq is not None:
        a_var, sc_var = _variance_refine(_best_seq, phase8_deadline, n_steps=300)
        upd(a_var, sc_var, "variance")

    # === Phase 9: Final Newton + SG polish (120s) ===
    print(f"\n[Phase 9] Final polish")
    phase9_deadline = min(deadline, time.monotonic() + 120)
    if _best_seq is not None:
        a_cur = _best_seq.copy()
        # Ultra-tight Newton

        upd(a_final_nw, sc_final_nw, "final_newton_tight")
        if sc_final_nw < _score(a_cur):
            a_cur = a_final_nw

        # Final subgradient
        sg_dead = min(phase9_deadline - 10, time.monotonic() + 50)
        a_final_sg, sc_final_sg = _subgrad_polish(a_cur, sg_dead, n_steps=5000, lr_init=0.004, tol_frac=0.0003)
        upd(a_final_sg, sc_final_sg, "final_sg_tight")

        # One more Newton
        if time.monotonic() < phase9_deadline - 5:
            a_final2, sc_final2 = _newton_active_set(a_final_sg, phase9_deadline - 3, max_iter=100, tol_frac=0.0001)
            upd(a_final2, sc_final2, "final_newton2")

    # === Phase 10: Ultra-final LSE with very low beta (60s) ===
    print(f"\n[Phase 10] Ultra-low beta LSE")
    phase10_deadline = min(deadline - 5, time.monotonic() + 60)
    if _best_seq is not None and time.monotonic() < phase10_deadline:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        beta_final = mb * 0.00003

                    _lse_obj, z_fin, args=(beta_final, n_fin),
                    jac=_lse_grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_ultra = _softmax(result.x)
                sc_ultra = _score(a_ultra)
                upd(a_ultra, sc_ultra, "ultra_lse")
            except Exception:
                pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print("FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
