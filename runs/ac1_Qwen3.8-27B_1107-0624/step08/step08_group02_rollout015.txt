import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14, restarts=5):
    """CG with periodic restarts for better convergence on ill-conditioned systems."""
    x = np.zeros(n)
    r = rhs.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    p = r.copy()
    iters_since_restart = 0
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        x = x_new
        r = r_new
        if rs_new < tol * tol * rs0:
            break
        iters_since_restart += 1
        if iters_since_restart >= restarts:
            # Full restart
            p = r.copy()
            iters_since_restart = 0
        else:
            p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < 1e-30 * rs0:
            break
    return x


def _smoothed_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _newton_active_polish(a_init, deadline, max_rounds=200, tight=False):
    """Newton active set method with adaptive tolerance and multi-pass refinement."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    tol_fracs = [0.000005, 0.00002, 0.0001, 0.0005, 0.002] if tight else [0.00005, 0.0002, 0.001, 0.005]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set tolerance
        tol_frac = tol_fracs[min(rnd // 20, len(tol_fracs) - 1)]
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=600, tol=1e-14)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600, tol=1e-14)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with sufficient decrease
        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.4
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 0.001 * alpha * gn / n:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:

                c_try = _full_convolve(a_try, a_try)

                    step_done = True

        else:

def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):

    min_beta = max_c * 1e-11

            break

        scale = max(1.0, n / 16384.0)

                obj_fn, z, jac=grad_fn,

                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400, 'maxfun': max_iter * 3}

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-17:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnant = 0
            else:
                stagnant += 1
        except Exception:
            stagnant += 1

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())

        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        else:

    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)

    elif kind == "step_like":

    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

        coeffs = 0.05 * rng.standard_normal(blocks)
        pert = np.zeros(n)
        for b in range(blocks):
            pert[b * block:(b + 1) * block] = coeffs[b]
    elif mode == 2:
        # Mirror perturbation
        half = n // 2
        pert = np.zeros(n)
        coeffs = 0.03 * rng.standard_normal(half)
        pert[:half] = coeffs
        pert[half:] = coeffs[::-1]
    else:
        # Random smooth perturbation via random Fourier features
        n_freq = 8
        freqs = rng.integers(1, n // 2 + 1, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        amps = 0.03 * rng.standard_normal(n_freq)
        t = np.arange(n) / n
        pert = sum(amps[k] * np.cos(2 * np.pi * freqs[k] * t + phases[k]) for k in range(n_freq))

    z = z + pert
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgradient_polish(a_init, deadline, max_steps=2000):
    """Projection-free subgradient method in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01
    step = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.0003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.03)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.008

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Profile exploration (60s) ===
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 60
    profiles = ["tapered", "asym_taper", "left_heavy", "bimodal", "exp_decay",
                "step_like", "flat_top", "gaussian_center", "cos"]
    best_profile_a = parent.copy()
    best_profile_sc = best_score

    for prof_kind in profiles:
        if time.monotonic() > phase1_deadline:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': score={sc_p:.10f} *")

    # Quick L-BFGS on best profile
    if time.monotonic() < phase1_deadline - 5:
        a_quick = _run_lbfgs_continuation(best_profile_a, phase1_deadline, min_time_remaining=10.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_profile")

    # === Phase 2: Main L-BFGS continuation (500s) ===
    print(f"\n[Phase 2] Main L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = t_start + 560
    start_a = best_seq if best_seq is not None else best_profile_a
    a_main = _run_lbfgs_continuation(start_a, phase2_deadline, min_time_remaining=50.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # === Phase 3: Newton polish pass 1 (300s) ===
    print(f"\n[Phase 3] Newton polish pass 1 (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(t_start + 860, deadline - 300)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_np1, sc_np1 = _newton_active_polish(best_seq, phase3_deadline, max_rounds=100, tight=False)
        update_best(a_np1, sc_np1, "newton_1")

    # === Phase 4: L-BFGS refinement from Newton result (250s) ===
    print(f"\n[Phase 4] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(t_start + 1110, deadline - 250)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_ref = _run_lbfgs_continuation(best_seq, phase4_deadline, min_time_remaining=200.0)
        sc_ref = _score(a_ref)
        update_best(a_ref, sc_ref, "refine")

    # === Phase 5: Newton polish pass 2 (tight) (250s) ===
    print(f"\n[Phase 5] Newton polish pass 2 (tight) (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(t_start + 1360, deadline - 200)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_np2, sc_np2 = _newton_active_polish(best_seq, phase5_deadline, max_rounds=200, tight=True)
        update_best(a_np2, sc_np2, "newton_2_tight")

    # === Phase 6: Diversification (200s) ===
    print(f"\n[Phase 6] Diversification (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(t_start + 1560, deadline - 150)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        n_div = len(best_seq)
        # Try structured perturbations
        for mode in range(4):
            if time.monotonic() > phase6_deadline - 30:
                break
            a_div = _perturb_structured(best_seq, rng, mode)
            sc_div = _score(a_div)
            # Quick Newton from perturbed point
            if time.monotonic() < phase6_deadline - 15:
                a_div_np, sc_div_np = _newton_active_polish(a_div, phase6_deadline - 5,
                                                             max_rounds=30, tight=True)
                update_best(a_div_np, sc_div_np, f"div_np_m{mode}")
                update_best(a_div, sc_div, f"div_m{mode}")

        # Try random gaussian perturbation
        for sig in [0.02, 0.05, 0.1]:
            if time.monotonic() > phase6_deadline - 20:
                break
            a_pert = _perturb(best_seq, sig, rng)
            if time.monotonic() < phase6_deadline - 10:
                a_pert_np, sc_pert_np = _newton_active_polish(a_pert, phase6_deadline - 5,
                                                               max_rounds=25, tight=True)
                update_best(a_pert_np, sc_pert_np, f"pert_np_s{sig}")

    # === Phase 7: Subgradient + Newton (150s) ===
    print(f"\n[Phase 7] Subgradient + Newton (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(t_start + 1710, deadline - 80)
    if best_seq is not None and time.monotonic() < phase7_deadline - 30:
        a_sg, sc_sg = _subgradient_polish(best_seq, phase7_deadline - 30, max_steps=1000)
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.002 and time.monotonic() < phase7_deadline - 10:
            a_sg_np, sc_sg_np = _newton_active_polish(a_sg, phase7_deadline - 5,
                                                       max_rounds=50, tight=True)
            update_best(a_sg_np, sc_sg_np, "subgrad_newton")

    # === Phase 8: Multi-length exploration (100s) ===
    print(f"\n[Phase 8] Multi-length (t={time.monotonic() - t_start:.1f}s)")
    phase8_deadline = deadline - 30
    if best_seq is not None and time.monotonic() < phase8_deadline - 20:
        n_best = len(best_seq)
        n_options = [n_best * 2, n_best // 2, int(n_best * 1.5)]
        n_options = [n for n in n_options if 1024 <= n <= 65536]
        for n_try in n_options[:2]:
            if time.monotonic() > phase8_deadline - 10:
                break
            a_rs = _resample(best_seq, n_try)
            if time.monotonic() < phase8_deadline - 5:
                a_rs_np, sc_rs_np = _newton_active_polish(a_rs, phase8_deadline - 3,
                                                           max_rounds=30, tight=True)
                update_best(a_rs_np, sc_rs_np, f"n_{n_try}")

    # === Phase 9: Final ultra-tight Newton (remaining time) ===
    print(f"\n[Phase 9] Final ultra-tight Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=300, tight=True)
        update_best(a_fin, sc_fin, "final_newton")

        if best_seq is not None and time.monotonic() < deadline - 1:
            a_vf, sc_vf = _newton_active_polish(best_seq, deadline - 0.5, max_rounds=100, tight=True)
            update_best(a_vf, sc_vf, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
