def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    N = _next_pow2(N)  # Make sure N is power of 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s

        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _soft_active_gradient(a, c, n, temperature=0.01):
    """Gumbel-softmax style soft active set."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    logits = c / max(temperature, 1e-10)
    logits = logits - logits.max()
    w = np.exp(np.clip(logits, -700, 0))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002, mu_init=1e-6):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    mu = mu_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            if alpha < 1e-12:
                mu *= 10  # Regularize more
                if mu > 1.0:
                    break
                continue
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            mu = max(mu * 0.5, mu_init)  # Reduce regularization on progress

    return best_a, best_sc

# Polyak-accelerated Mirror Descent
# ============================================================

def _polyak_mirror_descent(a_init, deadline, n_steps=10000, tol_frac=0.001):
    """Mirror descent on the simplex with Polyak step size and Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Theoretical lower bound
    lb = 2.0 * n / (2 * n - 1)
    
    # Nesterov momentum
    beta_mom = 0.9
    v = np.zeros(n)
    
    patience = 0
    lr = 0.1
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        sc = _score(a)
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 50:
            lr *= 0.9
            if patience > 200:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                v = np.zeros(n)
                patience = 0
                lr = 0.05
                beta_mom = 0.8
        
        # Compute subgradient
        grad_a = _soft_active_gradient(a, c, n, temperature=0.005)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        # Polyak step size
        gap = best_sc - lb
        gnorm_z = float(np.dot(grad_z, grad_z))
        if gnorm_z > 1e-30 and gap > 0:
            step_polyak = gap / gnorm_z
        else:
            step_polyak = lr
        step = min(step_polyak, lr)
        
        # Nesterov update
        v = beta_mom * v + (1 - beta_mom) * grad_z
        z = z - step * v
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        # Slowly increase beta for Nesterov
        beta_mom = min(0.99, beta_mom + 0.0001)
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

# Two-phase L-BFGS with adaptive LSE
# ============================================================

def _adaptive_lbfgs(a_init, deadline, min_remaining=30.0):
    """Adaptive L-BFGS that adjusts beta based on the spectral gap of the convolution."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    # Find the "knee" - where the sorted values stop dropping rapidly
    gaps = np.diff(c_sorted)
    # Look at the first 100 elements
    k = min(100, len(gaps))
    gap_ratios = np.abs(gaps[:k]) / np.maximum(c_sorted[:k], 1e-30)
    knee = int(np.argmax(gap_ratios < gap_ratios[0] * 0.5)) if k > 1 else 10
    
    base_beta = max(c_sorted[knee] * 0.1, max_c * 0.001)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Adaptive beta schedule
    betas = []
    b = base_beta
    while b > max_c * 0.00001 and len(betas) < 20:
        betas.append(b)
        b *= 0.5
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time_per_iter = 0.002 * n / 4096.0
        max_iter = min(5000, max(200, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If LSE didn't help, try power-sum
                p = max(2, int(1.0 / (beta / max_c + 1e-10)))
                p = min(p, 512)
                try:
                    result2 = minimize(
                        _power_sum_obj, z, args=(float(p), n),
                        jac=_power_sum_grad, method='L-BFGS-B',
                        options={'maxiter': max_iter // 2, 'ftol': 1e-16,
                                 'gtol': 1e-14, 'maxcor': 50, 'maxls': 100}
                    )
                    a_new2 = _softmax(result2.x)
                    sc_new2 = _score(a_new2)
                    if sc_new2 < current_sc - 1e-15:
                        current_a = a_new2
                        current_sc = sc_new2
                        z = np.log(np.maximum(a_new2, 1e-30))
                        z = z - z.mean()
                except Exception:
                    pass
        except Exception:
            pass

    return current_a, current_sc

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)

def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_nonneg":
        # Random non-negative with some structure
        a = np.abs(np.fft.irfft(
            np.exp(1j * np.random.uniform(-np.pi, np.pi, n // 2 + 1)), n
        ))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_flat":
        # Try to make a sequence with flat autocorrelation
        # Start with random, then iterate
        a = np.random.exponential(1.0, n)
        for _ in range(20):
            c = _full_convolve(a, a)
            c_max = c.max()
            active = c >= c_max * 0.95
            w = np.zeros_like(c)
            w[active] = 1.0 / active.sum()
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = np.dot(w, c)
            grad = 2.0 * n * (h - W)
            a = a - 0.01 * grad
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
        return a
    else:
        return np.ones(n) / n

# ============================================================
# Resize utility
# ============================================================

def _resize_sequence(a, n_target):
    if len(a) == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

# ============================================================
# Two-stage optimization: coarse then fine
# ============================================================

def _two_stage_optimize(a_init, deadline, min_remaining=100):
    """First optimize on a compressed version, then expand and refine."""
    n = len(a_init)
    if n <= 4096:
        return _adaptive_lbfgs(a_init, deadline, min_remaining)
    
    # Compress to 1/4 size
    n_small = n // 4
    a_small = _resize_sequence(a_init, n_small)
    
    # Optimize on small
    a_small_opt, sc_small = _adaptive_lbfgs(a_small, deadline, min_remaining // 2)
    
    # Expand back
    a_expanded = _resize_sequence(a_small_opt, n)
    
    # Refine on full size
    a_final, sc_final = _adaptive_lbfgs(a_expanded, deadline, min_remaining // 2)
    
    return a_final, sc_final
