"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set Newton with adaptive tolerance, exact 1D coordinate
descent, and multi-scale refinement.
"""

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:out_n]
    return result


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]

def _coordinate_step(a, idx, c):
    """Find optimal value of a[idx] given all others. Returns new a and score."""
    n = len(a)
    i = idx
    S_others = a.sum() - a[i]
    if S_others < 1e-15:
        return a, _score(a)

    ai = a[i]
    # Compute c_no_i[k] = c[k] minus contributions of a[i]
    c_no_i = c.copy()
    ki_range_start = max(0, 0)  # k-i >= 0 => k >= i
    ki_range_end = min(2 * n - 2, i + n - 1)  # k-i < n => k < i+n
    for k in range(ki_range_start, ki_range_end + 1):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                c_no_i[k] = c[k] - ai * ai
            else:
                c_no_i[k] = c[k] - 2.0 * ai * a[ki]

    # Now find x >= 0 minimizing 2n * max_k g_k(x) / (S_others + x)^2
    # where g_k(x) = c_no_i[k] + 2*x*a[k-i] (linear) or c_no_i[2i]+x^2 (quadratic)
    
    # Precompute the "slope" for each k: slope_k = 2*a[k-i] for valid k, 0 otherwise
    slopes = np.zeros(2 * n - 1)
    for k in range(ki_range_start, ki_range_end + 1):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                slopes[k] = 0  # handled separately (quadratic)
            else:
                slopes[k] = 2.0 * a[ki]
    
    # The quadratic term at k=2i: g_{2i}(x) = c_no_i[2i] + x^2
    has_quad = (2 * i < 2 * n - 1)
    c_quad = c_no_i[2 * i] if has_quad else 0.0

    # Evaluate f(x) = 2n * max(c_no_i + x*slopes, c_quad + x^2 if has_quad) / (S_others + x)^2
    def _eval_f(x_val):
        g = c_no_i + x_val * slopes
        if has_quad:
            g[2 * i] = c_quad + x_val * x_val
        f_val = 2.0 * n * g.max() / (S_others + x_val) ** 2
        return f_val

    # Golden section search on [0, upper]
    # Upper bound: try a few multiples of current ai
    upper = max(ai * 5.0, S_others * 0.5, 1e-10)
    
    # First, check a few candidate points to narrow the range
    candidates = [0.0, ai * 0.1, ai * 0.5, ai, ai * 2.0, ai * 5.0, upper]
    candidates = [max(0.0, min(c, upper)) for c in candidates]
    fvals = [_eval_f(x) for x in candidates]
    best_x = candidates[int(np.argmin(fvals))]
    best_f = min(fvals)
    
    # Golden section on [max(0, best_x - ai), min(upper, best_x + ai)]
    lo = max(0.0, best_x - ai)
    hi = min(upper, best_x + ai)
    if hi - lo < ai * 0.01:
        hi = lo + ai * 0.1
        hi = max(hi, 1e-10)
    
    golden = 0.6180339887498949
    for _ in range(40):
        if hi - lo < 1e-15 * max(1.0, best_x):
            break
        x1 = hi - golden * (hi - lo)
        x2 = lo + golden * (hi - lo)
        f1 = _eval_f(x1)
        f2 = _eval_f(x2)
        if f1 < f2:
            hi = x2
        else:
            lo = x1
        mid = (lo + hi) / 2.0
        fm = _eval_f(mid)
        if fm < best_f:
            best_x = mid
            best_f = fm

    # Apply the update
    a_new = a.copy()
    a_new[i] = best_x
    a_new = _normalize(a_new)
    sc = _score(a_new)
    return a_new, sc

    gap = sorted_c[min(20, len(sorted_c) - 1)] if len(sorted_c) > 20 else sorted_c[-1]
    base_beta = max(max_c - float(gap), max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time > deadline:
            break

        def obj_fn(z_local):
            a_local = np.exp(z_local - z_local.max())
            a_local = a_local / a_local.sum()
            c_local = _full_convolve(a_local, a_local)
            cm = c_local.max()
            shifted = (c_local - cm) / beta
            e = np.exp(np.clip(shifted, -500, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s))

        def grad_fn(z_local):
            a_local = np.exp(z_local - z_local.max())
            a_local = a_local / a_local.sum()
            c_local = _full_convolve(a_local, a_local)
            cm = c_local.max()
            shifted = (c_local - cm) / beta
            e = np.exp(np.clip(shifted, -500, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s
            h = _cross_corr_w_a(w, a_local, n)
            W = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W)

        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(3000, max(50, int(remaining / 0.05)))

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 80}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_avoidance_step(a_init, deadline, max_rounds=100):
    """Aggressively target the single largest peak and reduce it."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_peak = int(np.argmax(c))

        # Gradient of c[k_peak] w.r.t. a[i] is 2*a[k_peak - i] (for valid k_peak-i)
        grad = np.zeros(n)
        for i in range(n):
            ki = k_peak - i
            if 0 <= ki < n:
                grad[i] = 2.0 * a[ki]

        # We want to reduce c[k_peak]. The direction is -grad.
        # But we need to maintain sum(a) = 1, so subtract the mean.
        g_t = grad - grad.mean()

        # Line search
        step = 0.5 / (1.0 + 0.1 * rnd)
        for ls in range(30):
            a_try = a - step * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Active-set Newton with progressively tighter tolerances
    2. Exact coordinate descent
    3. Peak avoidance
    4. Multi-scale refinement
    5. Shrinkage polish
    6. Final precision pass
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is None:
        parent = np.ones(16384) / 16384
        print("No parent available, using uniform init")

    n_parent = len(parent)
    print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
    update_best(parent, _score(parent), "parent")

    # Phase 1: Active-set Newton with progressively tighter tolerances
    print(f"\n[Phase 1] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    a_work = parent.copy()
    for i, tol in enumerate([0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]):
        if time.monotonic() > deadline - 600:
            break
        time_budget = min(120.0, deadline - 600 - time.monotonic() + 120.0)
        a_work, sc_work = _active_set_newton(a_work, min(deadline - 300, time.monotonic() + time_budget),
                                             max_iter=300, tol_active=tol)
        update_best(a_work, sc_work, f"as_tol{tol}")

    # Phase 2: Coordinate descent (exact 1D optimization)
    print(f"\n[Phase 2] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_cd, sc_cd = _coordinate_descent(a_work, min(deadline - 200, time.monotonic() + 300),
                                          top_k=300, max_passes=30)
        update_best(a_cd, sc_cd, "coord_descent")
        a_work = a_cd if a_cd is not None else a_work

    # Phase 3: Peak avoidance
    print(f"\n[Phase 3] Peak avoidance (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_peak, sc_peak = _peak_avoidance_step(a_work, min(deadline - 150, time.monotonic() + 120),
                                                max_rounds=100)
        update_best(a_peak, sc_peak, "peak_avoid")
        if sc_peak < _score(a_work):
            a_work = a_peak

    # Phase 4: Active set refinement (try different active sets)
    print(f"\n[Phase 4] Active set refinement (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_asr, sc_asr = _active_set_refine(a_work, min(deadline - 100, time.monotonic() + 120))
        update_best(a_asr, sc_asr, "active_refine")
        if sc_asr < _score(a_work):
            a_work = a_asr

    # Phase 5: Multi-scale
    print(f"\n[Phase 5] Multi-scale (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_ms, sc_ms = _multi_scale_refine(a_work, min(deadline - 60, time.monotonic() + 90))
        update_best(a_ms, sc_ms, "multi_scale")
        if sc_ms < _score(a_work):
            a_work = a_ms

    # Phase 6: L-BFGS smoothed (for different basin exploration)
    print(f"\n[Phase 6] L-BFGS smoothed (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_lb, sc_lb = _run_lbfgs_smoothed(a_work, min(deadline - 50, time.monotonic() + 80), min_time=40)
        update_best(a_lb, sc_lb, "lbfgs")
        if sc_lb < _score(a_work):
            a_work = a_lb

    # Phase 7: More coordinate descent (finer)
    print(f"\n[Phase 7] Fine coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_cd2, sc_cd2 = _coordinate_descent(a_work, min(deadline - 40, time.monotonic() + 60),
                                            top_k=500, max_passes=20)
        update_best(a_cd2, sc_cd2, "coord_fine")
        if sc_cd2 < _score(a_work):
            a_work = a_cd2

    # Phase 8: Final active-set Newton (very tight)
    print(f"\n[Phase 8] Final precision Newton (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        a_fin, sc_fin = _active_set_newton(a_work, min(deadline - 10, time.monotonic() + 30),
                                           max_iter=500, tol_active=0.00003, cg_tol=1e-14)
        update_best(a_fin, sc_fin, "final_newton")
        if sc_fin < _score(a_work):
            a_work = a_fin

    # Phase 9: Shrinkage
    print(f"\n[Phase 9] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_sh, sc_sh = _shrinkage_polish(a_work, deadline - 5)
        update_best(a_sh, sc_sh, "shrinkage")
        if sc_sh < _score(a_work):
            a_work = a_sh
        # After shrinkage, do one more Newton pass
        if time.monotonic() < deadline - 5:
            a_sh2, sc_sh2 = _active_set_newton(a_sh, deadline - 3, max_iter=200, tol_active=0.0001)
            update_best(a_sh2, sc_sh2, "shrink_newton")

    # Phase 10: Final coordinate descent
    if time.monotonic() < deadline - 3:
        a_fin2, sc_fin2 = _coordinate_descent(a_work, deadline - 2, top_k=1000, max_passes=10)
        update_best(a_fin2, sc_fin2, "final_coord")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    return [float(x) for x in fallback]
