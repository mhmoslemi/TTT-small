import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the gradient contribution for active set A_vec."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _spectrum_flatten(a, alpha=0.5):
    """Flatten the power spectrum of a and invert."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    # Interpolate between current power spectrum and flat
    P_target = np.full_like(P, P.sum() / len(P))
    P_new = (1 - alpha) * P + alpha * P_target
    # Ensure non-negative
    P_new = np.maximum(P_new, 1e-30)
    # Take square root to get magnitude
    mag = np.sqrt(P_new)
    # Keep original phase (for real sequence)
    phase = np.angle(A)
    A_new = mag * np.exp(1j * phase)
    # Invert
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_new / s

def _targeted_deflation(a, deadline, n_attempts=30):

    n = len(a)

            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1]
        n_top = max(5, min(len(c), 20))
        top_positions = top_idx[:n_top]
        
        # For each top position, find and reduce the largest contributing pair
        a_new = a.copy()
        reduction = 0.99 - 0.5 * (attempt / max(1, n_attempts))  # decrease reduction over time

            max_i = -1

                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:

                # Reduce the larger element
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= reduction
                else:
                    a_new[j] *= reduction

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Spectral sharpening (power-domain) to flatten autocorrelation
2. Continuation-based active-set Newton with adaptive regularization
3. Log-sum-exp smoothed L-BFGS with adaptive temperature schedule
4. Targeted deflation of top peaks
5. Multi-n exploration (16384, 32768) with interpolation
"""
import numpy as np
import time
from scipy.optimize import minimize
from functools import partial


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lsobj_wrapper(z, beta, n):
    return _lsobj(z, beta, n)


def _lsgrad_wrapper(z, beta, n):
    return _lsgrad(z, beta, n)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _spectrum_sharpen(a, power=2.0):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    P_max = P.max()
    if P_max < 1e-30:
        return a.copy()
    P_norm = P / P_max
    P_sharp = (P_norm ** power) * P_max
    P_sharp = np.maximum(P_sharp, 0)
    mag = np.sqrt(P_sharp)
    phase = np.angle(A)
    A_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_new / s


def _hess_vp_wrapper(v, A_vec, m, mu, n):
    return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v


def _active_set_newton(a_init, deadline, max_iter=400, phases=None, iters_per_phase=30):
    if phases is None:
        phases = [0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004]
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    phase = 0
    stale = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        frac = phases[min(phase, len(phases) - 1)]
        n_active_target = max(3, int(frac * (2 * n - 1)))

        sorted_idx = np.argsort(c)[::-1]
        active = np.zeros(2 * n - 1, dtype=bool)
        active[sorted_idx[:n_active_target]] = True
        m = n_active_target

        A_vec = active.astype(np.float64)
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)

        if gnorm < 1e-14:
            stale += 1
            if stale >= iters_per_phase:
                phase += 1
                stale = 0
                if phase >= len(phases):
                    break
            continue

        mu = 1e-6 * c_max / max(m, 1) * (1.0 + 30.0 / m) + 1e-10

        def hess_vp(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        u = _cg_solve(hess_vp, g_t, max_cg=250)
        w_vec = _cg_solve(hess_vp, np.ones(n), max_cg=250)
        w_sum = float(w_vec.sum())

        if abs(w_sum) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / w_sum
            dx = -u + lam * w_vec

        dx = dx - np.mean(dx)

        f_old = c_max
        accepted = False
        t = 1.0
        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old - 1e-5 * t * np.linalg.norm(dx) or ls >= 50:
                if f_try <= f_old + 1e-11:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = 0.01
            for ls in range(30):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old - 1e-14:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5

        if not accepted:
            g_all = _corr_a_A(a, np.ones(2 * n - 1), n)
            g_t_all = g_all - np.mean(g_all)
            t = 0.005
            for ls in range(30):
                a_try = a - t * g_t_all
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old - 1e-14:
                    a = a_try
                    break
                t *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-12 * c_max:
            stale += 1
            if stale >= iters_per_phase and phase < len(phases) - 1:
                phase += 1
                stale = 0
        else:
            stale = 0

    return best_a, best_sc


def _log_sum_exp_optimize(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx1 = min(3, len(c_sorted) - 1)
    idx2 = min(10, len(c_sorted) - 1)
    gap = c_sorted[0] - (c_sorted[idx1] + c_sorted[idx2]) / 2.0
    base_beta = max(gap * 0.15, max_c * 0.008)

    betas = base_beta * np.array([1.0, 0.75, 0.55, 0.4, 0.3, 0.22, 0.16, 0.12, 0.09,
                                  0.07, 0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.012,
                                  0.009, 0.007, 0.005, 0.004, 0.003, 0.002, 0.0015,
                                  0.001, 0.0008, 0.0006, 0.0005, 0.0004, 0.0003])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 16384.0)
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 10))))

        try:
            func = partial(_lsobj_wrapper, beta=beta, n=n)
            jac = partial(_lsgrad_wrapper, beta=beta, n=n)
            result = minimize(
                func,
                z,
                jac=jac,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-19,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 300,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _targeted_deflation(a, deadline, n_attempts=30):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        top_idx = np.argsort(c)[::-1]
        n_top = max(5, min(len(c), int(0.02 * len(c))))
        top_positions = top_idx[:n_top]

        reduction = 0.995 - 0.003 * (attempt / max(1, n_attempts))

        a_new = a.copy()
        for k_pos in top_positions[:min(n_top, 30)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= reduction
                else:
                    a_new[j] *= reduction

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _update_best(state, p, sc=None, label=""):
    if p is None:
        return
    p_arr = np.asarray(p, dtype=np.float64).ravel()
    p_arr = np.clip(p_arr, 0.0, 1000.0)
    s_val = float(p_arr.sum())
    if s_val < 0.01:
        return
    if sc is None:
        sc = _score(p_arr)
    if np.isfinite(sc) and sc < state['best_score']:
        prev = state['best_score']
        state['best_seq'] = p_arr.copy()
        state['best_score'] = float(sc)
        print(f"  NEW BEST: {state['best_score']:.10f} (n={len(p_arr)}) [{label}] "
              f"improve={prev - state['best_score']:.2e} t={time.monotonic() - state['t_start']:.1f}s")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
