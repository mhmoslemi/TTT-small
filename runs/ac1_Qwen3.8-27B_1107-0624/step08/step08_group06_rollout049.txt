"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive active-set Newton with dual-variable-based active set management
2. Multi-beta log-sum-exp L-BFGS continuation
3. Multi-n exploration (8192, 16384, 32768)
4. Diverse restarts with structural priors
5. Banded Hessian CG solver with adaptive preconditioning

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=500, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    rs_norm = np.sqrt(rs)
    if rs_norm < tol * (1 + rhs_norm):
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x_new = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if rs_new >= rs * (1 + 1e-10):
            break
        x = x_new
        r = r_new
        rs = rs_new
        if np.sqrt(rs) < tol * (1 + rhs_norm):
            break
        p = r + (rs / (rs_new + 1e-300)) * p
    return x


def _active_set_newton_v2(a_init, deadline, max_iter=500):
    """Enhanced active-set Newton with dual-variable-based active set management."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_list = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)] if len(c_sorted) > 50 else c_sorted[0]

        best_dir = None
        best_dir_score = np.inf
        best_t = 1.0
        best_A_vec = None

        for tol_a in tol_list:
            threshold = c_max - tol_a * c_max
            active = c >= threshold
            n_active = int(active.sum())
            if n_active < 2:
                active[np.argmax(c)] = True
                n_active = 1
            if n_active > 8 * n:
                top_idx = np.argsort(c)[::-1][:8 * n]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                n_active = int(active.sum())

            A_vec = active.astype(np.float64)
            m = float(n_active)

            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            mu = 1e-8 * (c_max / max(n_active, 1)) + 1e-10

            def hess_vp(v, A_vec=A_vec, m=m, mu=mu, n=n):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

            u = _cg_solve(hess_vp, g_t, max_cg=300, tol=1e-13)
            w = _cg_solve(hess_vp, np.ones(n), max_cg=300, tol=1e-13)
            w_sum = float(w.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w

            dx = dx - np.mean(dx)

            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)

            t = min(1.0, t_max)
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)

            if sc_try < best_dir_score:
                best_dir_score = sc_try
                best_dir = dx
                best_t = t
                best_A_vec = A_vec.copy()

        if best_dir is None:
            g_all = (2.0 / n) * _corr_a_A(a, np.ones(2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            t = 0.02
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        f_old = c_max
        accepted = False
        t = best_t
        for ls in range(60):
            a_try = a + t * best_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old - 1e-10 * t * f_old or ls >= 50:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14 * c_max:
            break

    return best_a, best_sc


def _run_lbfgs_v2(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation and smarter schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
        0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00004
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        break

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * ((t - 0.5) / 0.3) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block_size = n // 5
        a[:block_size] = 1.0
        a[2 * block_size:3 * block_size] = 1.0
        a[4 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "ramp":
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fista_projected_v2(a_init, deadline, n_steps=5000):
    """FISTA for min max_k c[k] s.t. a >= 0, sum(a) = 1."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    y = a.copy()
    t_val = 1.0
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad = 2.0 * _corr_a_A(y, w, n)
        g_t = grad - np.mean(grad)

        step_size = 0.02 / (1.0 + 0.005 * step)
        a_new = y - step_size * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_new + ((t_val - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t_val = t_new

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def _multi_newton_refine(a_init, deadline, max_rounds=5):
    """Multiple rounds of active-set Newton with decreasing tolerance."""
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedules = [
        [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005],
        [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002],
        [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
        [0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
        [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
    ]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        tol_list = tol_schedules[min(rnd, len(tol_schedules) - 1)]
        a_out, sc_out = _active_set_newton_v2(a, deadline, max_iter=300)
        if sc_out < best_sc - 1e-15:
            best_sc = sc_out
            best_a = a_out.copy()
        a = a_out

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer with enhanced active-set Newton,
    multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Strategy: Focus optimization on 2-3 n values with different time budgets
    # Primary: parent n, Secondary: a larger n, Tertiary: a smaller n
    primary_n = n_parent if n_parent in [4096, 8192, 16384, 32768] else 16384
    n_options = sorted(set([primary_n, 16384, 32768]))
    # Ensure we have at least 2 options
    if len(n_options) < 2:
        n_options.append(8192)

    print(f"n options: {n_options}")

    # Phase 1: Primary n optimization
    print(f"\n{'='*60}")
    print(f"[Phase 1] Primary n={primary_n} optimization")
    print(f"{'='*60}")

    if primary_n == n_parent and parent is not None:
        a_main = parent.copy()
    else:

    # L-BFGS coarse
    dl1 = min(deadline - 800, time.monotonic() + 200)
    a_lbfgs = _run_lbfgs_v2(a_main, dl1, min_time_remaining=500)
    sc_lbfgs = _score(a_lbfgs)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_primary")

    # Active-set Newton refinement
    if time.monotonic() < deadline - 500:
        dl2 = min(deadline - 500, time.monotonic() + 300)
        a_newton, sc_newton = _multi_newton_refine(a_lbfgs, dl2, max_rounds=4)
        update_best(a_newton, sc_newton, "newton_primary")

        # Final L-BFGS polish on the Newton result
        if time.monotonic() < deadline - 400:
            dl3 = min(deadline - 400, time.monotonic() + 100)
            a_polish = _run_lbfgs_v2(a_newton, dl3, min_time_remaining=200)
            sc_polish = _score(a_polish)
            update_best(a_polish, sc_polish, "polish_primary")

            # One more Newton round
            if time.monotonic() < deadline - 300:

        # Also try different profiles

            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 5: Try asymmetric initializations

        # Create an asymmetric profile: ramp-like

        m2 = n_asym // 3
        a_bump[:m1] = 1.0
        a_bump[n_asym - m2:] = 0.6

            update_best(a_opt, sc_opt, f"asym_{lbl}")

                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
