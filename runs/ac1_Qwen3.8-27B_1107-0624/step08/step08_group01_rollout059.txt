"""
Step-function optimization targeting score < 1.5027.
Uses LSE-regularized L-BFGS with verified gradients, adaptive temperature
annealing, direct peak deflation, and iterated perturbation-reoptimization.
Employs softmax parameterization for non-negativity, FFT-based convolution,
and multi-scale exploration across diverse initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # H[m] = sum_k w[k] * a[k-m] (for valid indices)
    # H = full_convolve(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    H = wc[n - 1:2 * n - 1]
    P = float(np.dot(a, H))
    grad = 4.0 * n * a * (H - P)
    return grad


def _run_lbfgs_lse(a_init, deadline, min_remaining=15.0):
    """L-BFGS on LSE objective with aggressive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    beta_ratios = [1.0, 0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.025,
                   0.015, 0.009, 0.005, 0.003, 0.0018, 0.001,
                   0.0006, 0.00035, 0.0002, 0.00012, 0.00007, 0.00004,
                   0.00002, 0.00001, 0.000005]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, ratio in enumerate(beta_ratios):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        beta = base_beta * ratio
        # Estimate time per iteration
        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.0015 * scale
        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 20))))

        def obj_func(zv):
            return _lse_obj(zv, beta, n)

        def grad_func(zv):
            return _lse_grad(zv, beta, n)

        try:
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-12,
                    'maxcor': 100,
                    'maxls': 80,
                    'iprint': -1,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _deflate_peaks(a_init, deadline, min_remaining=3.0):
    """Direct peak deflation via active-set gradient steps."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.02
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        # Active set: top peaks within tolerance
        tol_val = 0.002 * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        # Cap active set size
        if n_active > 20:
            top_idx = np.argsort(c)[::-1][:20]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 20

        # Average gradient over active set
        # grad_a[j] = (1/|K|) * sum_{k in K} 2*a[k-j]
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        H = wc[n - 1:2 * n - 1]
        grad_a = 2.0 * H

        # Center to preserve sum
        grad_centered = grad_a - np.mean(grad_a)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15)) * gnorm
        improved = False

        for _ in range(50):
            a_try = a - alpha * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.85
            if lr < 1e-6:
                break
        else:
            lr = min(0.02, lr * 1.02)

    return best_a, _score(best_a)


def _perturb_reoptimize(a_best, deadline, rng, n_cycles=5, min_remaining=30.0):
    """Perturb and re-optimize to escape local minima."""
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score(a_best)

    for cycle in range(n_cycles):
        if time.monotonic() + min_remaining > deadline:
            break

        # Structured perturbation
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()

        # Mix of Gaussian and structured perturbations
        sigma = 0.03 + 0.02 * cycle
        noise = sigma * rng.standard_normal(n)

        # Add a small oscillatory component to break symmetry
        freq = rng.integers(3, max(4, n // 500) + 1)
        phase = rng.uniform(0, 2 * np.pi)
        osc_amp = 0.01 * rng.uniform(0.5, 1.0)
        noise += osc_amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)

        z_pert = z + noise
        z_pert = z_pert - z_pert.mean()
        a_pert = _softmax(z_pert)
        a_pert = a_pert / a_pert.sum()

        # Quick L-BFGS from perturbed start
        c_pert = _full_convolve(a_pert, a_pert)
        max_c = float(c_pert.max())
        c_sorted = np.sort(c_pert)[::-1]
        gap = max_c - c_sorted[min(50, len(c_sorted) - 1)]
        base_beta = max(gap, max_c * 0.002)

        sub_deadline = min(deadline - min_remaining, time.monotonic() + 60.0)
        betas_sub = [base_beta * r for r in [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]]

        z_opt = z_pert.copy()
        for beta in betas_sub:
            if time.monotonic() > sub_deadline:
                break
            remaining = sub_deadline - time.monotonic()
            if remaining < 3:
                break
            scale = max(1.0, n / 8192.0)
            max_iter = min(3000, max(30, int(remaining / (0.002 * scale * 10))))

            def obj_f(zv):
                return _lse_obj(zv, beta, n)

            def grad_f(zv):
                return _lse_grad(zv, beta, n)

            try:

                                  options={'maxiter': max_iter, 'ftol': 1e-16,

                a_new = _softmax(result.x)

                if s_new < 1e-15:

                a_new = a_new / s_new

            except Exception:
                pass

        # Final deflation on best found

def _make_asymmetric_profile(n, kind="gauss_shift"):
    if kind == "gauss_shift":
        # Asymmetric Gaussian: peak off-center

    elif kind == "exp_decay":

        a = np.exp(-0.3 * t / n) + 0.05

    elif kind == "sin_asym":

        return a / a.sum()
    elif kind == "cos_2f":
        t = np.arange(n, dtype=float) / n

        return a / a.sum()
    elif kind == "bimodal_asym":

        return a / a.sum()
    elif kind == "power_asym":
        t = np.arange(1, n + 1, dtype=float)
        a = 1.0 / (t ** 0.3)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        print(f"\n[P4] Second deflation (t={time.monotonic()-t_start:.1f}s)")
        p4_deadline = min(deadline - 350.0, t_start + 800.0)
        a4, sc4 = _deflate_peaks(best_a, p4_deadline, min_remaining=15.0)
        if sc4 < best_score:
            best_a = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [deflate_p4] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Perturbation-reoptimization cycles ===
    print(f"\n[P5] Perturbation-reoptimization (t={time.monotonic()-t_start:.1f}s)")
    n_cycles = 8
    p5_deadline = min(deadline - 200.0, t_start + 1200.0)
    a5, sc5 = _perturb_reoptimize(best_a, p5_deadline, rng, n_cycles=n_cycles, min_remaining=50.0)
    if sc5 < best_score:
        best_a = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} [perturb_p5] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Try different n values ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_candidates = [16384, 24576, 20480, 18432]
    n_candidates = [n for n in n_candidates if n != n_best]
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 150.0:
            break
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  Trying n={n_try}, init score={sc_init:.10f}")

        sub_deadline = min(deadline - 100.0, time.monotonic() + 180.0)
        a_opt = _run_lbfgs_lse(a_init, sub_deadline, min_remaining=20.0)
        sc_opt = _score(a_opt)

        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < deadline - 80.0:
            sub_def = min(deadline - 60.0, time.monotonic() + 30.0)
            a_def, sc_def = _deflate_peaks(a_opt, sub_def, min_remaining=5.0)
            if sc_def < best_score:
                best_a = a_def.copy()
                best_score = sc_def
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_deflate] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Try diverse asymmetric initializations at best n ===
    print(f"\n[P7] Diverse starts (t={time.monotonic()-t_start:.1f}s)")
    n_use = len(best_a)
    profiles = ["exp_decay", "sin_asym", "cos_2f", "bimodal_asym", "power_asym"]
    for prof in profiles:
        if time.monotonic() > deadline - 60.0:
            break
        a_init = _make_asymmetric_profile(n_use, prof)
        sc_init = _score(a_init)
        if sc_init > best_score + 0.1:
            continue  # Skip if too far from best
        sub_deadline = min(deadline - 40.0, time.monotonic() + 60.0)
        a_opt = _run_lbfgs_lse(a_init, sub_deadline, min_remaining=10.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_use}) [profile_{prof}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        a_fin, sc_fin = _deflate_peaks(best_a, deadline - 3.0, min_remaining=2.0)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_deflate] t={time.monotonic()-t_start:.1f}s")

    # One final L-BFGS with very small beta
    remaining = deadline - 4.0 - time.monotonic()
    if remaining > 10.0:
        n = len(best_a)
        c_best = _full_convolve(best_a, best_a)
        mb = float(c_best.max())
        beta_fin = mb * 0.000005
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()
        scale = max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (0.002 * scale * 5))))
        try:
            result = minimize(lambda zv: _lse_obj(zv, beta_fin, n), z,
                              jac=lambda zv: _lse_grad(zv, beta_fin, n),
                              method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 80, 'iprint': -1})
            a_fin2 = _softmax(result.x)
            a_fin2 = np.maximum(a_fin2, 0.0)
            s_fin2 = a_fin2.sum()
            if s_fin2 > 1e-10:
                a_fin2 = a_fin2 / s_fin2
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_a = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation and output
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
