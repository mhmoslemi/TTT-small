def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]

"""

gradients, adaptive temperature scheduling, active-set peak equalization, and

import numpy as np

def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]


def _fft_xcorr(w, a):
    """Compute cross-correlation R[j] = sum_m w[m+j]*a[m] for j=0..len(a)-1.
    Uses full FFT with zero-padding for linear (non-circular) correlation."""
    n = len(a)
    lw = len(w)
    N = _next_pow2(lw + n - 1)
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    # R[j] = ifft(conj(W) * A)[j]
    R = np.fft.ifft(np.conj(W) * A, N)
    return R[:n].real


def _score_norm(a):
    """Score = 2*n*max(c) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

    """Actual evaluate_sequence score."""
    n = len(a)
    s = float(a.sum())

    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _mirror_descent(a_init, deadline, T_start, T_end, eta_start, max_iter,

    a /= a.sum()

    patience = 0
    t = 0

            break
        t += 1

        eta = eta_start * (1.0 + t / 5000.0) ** (-0.7)

def _active_set_equalize(a, deadline, K=5, max_iter=500):
    """Equalize top-K peaks using linear algebra."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find top K peaks
        sorted_idx = np.argsort(c)[::-1][:K]
        active = sorted_idx
        
        # Build constraint matrix: for each active lag k, the gradient of c[k] w.r.t. a
        # grad c[k] = 2 * a[k-j] (for valid j)
        # We want to find direction d (sum(d)=0) that reduces all active peaks equally
        # Linearize: c[k] + 2 * sum_j d[j] * a[k-j] = target for each active k
        
        # M[i, j] = 2 * a[active[i] - j] if 0 <= active[i]-j < n, else 0
        M = np.zeros((len(active), n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Add sum constraint
        M_full = np.vstack([M, np.ones(n)])
        
        # Target: reduce all active peaks to their minimum (or equalize them)
        target = np.min(c[active])  # We want to bring all down to the min
        b = target - c[active]
        b_full = np.concatenate([b, [0.0]])
        
        # Minimum norm solution: d = M_full^T @ pinv(M_full @ M_full^T) @ b_full
        G = M_full @ M_full.T
        try:
            G_inv = np.linalg.inv(G + 1e-10 * np.eye(G.shape[0]))
            d = M_full.T @ (G_inv @ b_full)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b_full, rcond=None)[0]
        
        d = np.maximum(d, -a * 0.5)  # Don't make a negative
        d -= d.sum() / n  # Ensure sum = 0 (approximate)
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        
        d = d / d_norm
        
        # Try multiple step sizes
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.001]:
            a_cand = a + alpha * d
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=50000, top_k=3, eta_initial=0.01):
    """Subgradient method with multiplicative updates on the simplex."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-30)
    a /= a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta_initial
    patience = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: top-k peaks
        sorted_idx = np.argsort(c)[::-1][:top_k]
        active_val = c[sorted_idx]
        
        # Softmax over active set (sharp)
        w_active = np.exp((active_val - c_max) / (c_max * 0.01 + 1e-15))
        w_active /= w_active.sum()
        
        # Compute gradient using only active peaks
        g = np.zeros(n)
        for idx, k in enumerate(sorted_idx):
            if w_active[idx] < 1e-6:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += w_active[idx] * 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Multiplicative update
        log_step = -eta * g
        log_step = np.clip(log_step, -1.0, 1.0)
        a_new = a * np.exp(log_step)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new /= s
        
        if it % 10 == 0:
            c_new = _fft_convolve(a_new, a_new)
            sc_new = float(2.0 * n * c_new.max())
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = a_new.copy()
                patience = 0
                eta *= 0.999
            else:
                patience += 1
                eta *= 0.95
            a = a_new
        else:
            a = a_new
        
        if patience > 200:
            break
    
    return best_a, best_score


def _optimize_single_n(n, a_init, deadline):
    """Full optimization pipeline for a single n and initialization."""
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = _init_sine(n)
        s = a.sum()
    a /= s
    a = a * (s)  # keep same sum for score computation
    
    best_a = a.copy()
    best_score = _score_full(a)
    
    # Normalized (sum=1) for optimization
    a_norm = a / a.sum()
    
    # Adaptive mirror descent
    ml_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.55)
    a_opt, sc_opt = _adaptive_mirror_descent(a_norm, ml_deadline)
    # Convert back to original scale
    a_opt_scale = a_opt * a.sum()
    sc_full = _score_full(a_opt_scale)
    if sc_full < best_score:
        best_score = sc_full
        best_a = a_opt_scale.copy()
    
    # Active set equalization
    as_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.2)
    a_as, sc_as = _active_set_equalize(a_opt, as_deadline, K=5, max_iter=300)
    a_as_scale = a_as * a.sum()
    sc_as_full = _score_full(a_as_scale)
    if sc_as_full < best_score:
        best_score = sc_as_full
        best_a = a_as_scale.copy()
    
    # Subgradient polish
    sg_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.2)
    a_best_norm = best_a / best_a.sum()
    a_sg, sc_sg = _subgradient_polish(a_best_norm, sg_deadline, max_iter=30000, top_k=3, eta_initial=0.005)
    a_sg_scale = a_sg * a.sum()
    sc_sg_full = _score_full(a_sg_scale)
    if sc_sg_full < best_score:
        best_score = sc_sg_full
        best_a = a_sg_scale.copy()
    
    # Final active-set polish
    if time.monotonic() < deadline:
        a_final_norm = best_a / best_a.sum()
        a_fp, sc_fp = _active_set_equalize(a_final_norm, deadline, K=3, max_iter=200)
        a_fp_scale = a_fp * a.sum()
        sc_fp_full = _score_full(a_fp_scale)
        if sc_fp_full < best_score:
            best_score = sc_fp_full
            best_a = a_fp_scale.copy()
    
    return best_a, best_score


def _try_n(n, inits, timeline_deadline, label):
    """Try all initializations for a given n, within the time budget."""
    best_a = None
    best_score = np.inf
    
    time_per_init = max(5.0, (timeline_deadline - time.monotonic()) / len(inits))
    
    for i, (name, a_init) in enumerate(inits):
        if time.monotonic() > timeline_deadline - 5:
            break
        sub_dl = min(timeline_deadline - 2, time.monotonic() + time_per_init)
        try:
            a_opt, sc = _optimize_single_n(n, a_init, sub_dl)
            print(f"    {label} [{name}]: score={sc:.10f}, t={time.monotonic():.0f}s")
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
        except Exception as ex:
            print(f"    {label} [{name}] failed: {ex}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Multiple initializations (sine, cosine, Gaussian, parent, random)
    2. Mirror descent with FFT-based gradients and adaptive temperature
    3. Active-set peak equalization
    4. Subgradient polish
    5. Multiple sequence lengths (4096, 8192, 16384)
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Try to load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            print(f"Parent loaded: n={len(parent)}, score={_score_full(parent):.10f}")
        else:
            parent = None
    except Exception:
        pass
    
    # Strategy: allocate time across different n values and initializations
    # Focus primarily on n=8192 (proven to work), with some time for others
    
    global_best_a = None
    global_best_score = np.inf
    
    def _update_best(a, sc):
        nonlocal global_best_a, global_best_score
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = a.copy()
            print(f"  NEW BEST: {sc:.10f} (n={len(a)})")
    
    # ---- n = 8192 (primary) ----
    n_primary = 8192
    inits_8192 = [
        ("sine", _init_sine(n_primary)),
        ("cosine", _init_cosine(n_primary)),
        ("gaussian", _init_gaussian(n_primary)),
    ]
    if parent is not None and len(parent) == n_primary:
        inits_8192.insert(0, ("parent", parent / parent.sum()))
    elif parent is not None:
        # Resample parent to n_primary
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n_primary)
        a_resampled = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')(x_new)
        a_resampled = np.maximum(a_resampled, 0.0)
        if a_resampled.sum() > 1e-10:
            inits_8192.insert(0, ("parent_resampled", a_resampled / a_resampled.sum()))
    
    # Random init
    a_rand = rng.random(n_primary) ** 2
    a_rand += 0.1 * _init_sine(n_primary)
    inits_8192.append(("random", a_rand / a_rand.sum()))
    
    print(f"\n=== n={n_primary} ({len(inits_8192)} inits) ===")
    t0 = time.monotonic()
    n_primary_time = min(deadline - 300, t_start + 1200)  # Give 1200s to primary
    a_best_8192, sc_best_8192 = _try_n(n_primary, inits_8192, n_primary_time, "n8192")
    if a_best_8192 is not None and sc_best_8192 < global_best_score:
        _update_best(a_best_8192, sc_best_8192)
    print(f"  n=8192 done: best={global_best_score:.10f}, elapsed={time.monotonic()-t_start:.0f}s")
    
    # ---- n = 4096 ----
    if time.monotonic() < deadline - 100:
        n_alt = 4096
        inits_4096 = [
            ("sine", _init_sine(n_alt)),
            ("gaussian", _init_gaussian(n_alt)),
        ]
        if parent is not None:
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_alt)
            a_r = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')(x_new)
            a_r = np.maximum(a_r, 0.0)
            if a_r.sum() > 1e-10:
                inits_4096.insert(0, ("parent", a_r / a_r.sum()))
        
        print(f"\n=== n={n_alt} ({len(inits_4096)} inits) ===")
        n_alt_time = min(deadline - 50, time.monotonic() + 300)
        a_alt, sc_alt = _try_n(n_alt, inits_4096, n_alt_time, "n4096")
        if a_alt is not None and sc_alt < global_best_score:
            _update_best(a_alt, sc_alt)
        print(f"  n=4096 done: best={global_best_score:.10f}, elapsed={time.monotonic()-t_start:.0f}s")
    
    # ---- n = 16384 ----
    if time.monotonic() < deadline - 80:
        n_large = 16384
        inits_16384 = [
            ("sine", _init_sine(n_large)),
        ]
        if parent is not None:
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_large)
            a_r = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')(x_new)
            a_r = np.maximum(a_r, 0.0)
            if a_r.sum() > 1e-10:
                inits_16384.insert(0, ("parent", a_r / a_r.sum()))
        
        print(f"\n=== n={n_large} ===")
        n_large_time = min(deadline - 20, time.monotonic() + 250)
        a_large, sc_large = _try_n(n_large, inits_16384, n_large_time, "n16384")
        if a_large is not None and sc_large < global_best_score:
            _update_best(a_large, sc_large)
        print(f"  n=16384 done: best={global_best_score:.10f}, elapsed={time.monotonic()-t_start:.0f}s")
    
    # ---- Final polish on the best solution found ----
    if global_best_a is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final polish ===")
        a_polish = global_best_a / global_best_a.sum()
        orig_sum = global_best_a.sum()
        
        # Quick active-set
        a_as2, sc_as2 = _active_set_equalize(a_polish, deadline - 5, K=3, max_iter=100)
        sc_as2_full = _score_full(a_as2 * orig_sum)
        if sc_as2_full < global_best_score:
            _update_best(a_as2 * orig_sum, sc_as2_full)
        
        # Quick subgradient
        a_sg2, sc_sg2 = _subgradient_polish(a_polish, deadline - 3, max_iter=10000, top_k=2, eta_initial=0.002)
        sc_sg2_full = _score_full(a_sg2 * orig_sum)
        if sc_sg2_full < global_best_score:
            _update_best(a_sg2 * orig_sum, sc_sg2_full)
    
    # Return the best
    if global_best_a is None:
        # Fallback
        global_best_a = _init_sine(8192)
        global_best_score = _score_full(global_best_a)
    
    # Final validation
    final = np.clip(global_best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
        s = final.sum()
    
    sc_final = _score_full(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval check error: {ex}")
    
    return [float(x) for x in final]
