from scipy.optimize import minimize


def _next_pow2(n):

    while k < n:
        k <<= 1

def _full_convolve(a, b):

def _score(a):

        return np.inf

def _softmax(z):

def _corr_a_A(a, A_vec, n):
    """Compute correlation of a with A_vec, returning the n elements centered."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

def _spectral_init(n, n_iters=200, target_flatness=0.5):
    """
    Gerchberg-Saxton style spectral shaping.
    Alternates between:
    - Time domain: project onto non-negative, sum=1
    - Frequency domain: flatten the magnitude spectrum
    """
    rng = np.random.default_rng(12345)
    a = np.ones(n) / n

    for it in range(n_iters):
        # FFT
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)

        # Flatten magnitude: blend current with flat
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: raise the low-magnitude bins, keep high ones
        mag_target = np.maximum(mag, target_flatness * np.mean(mag))
        mag_new = 0.5 * mag + 0.5 * mag_target
        # Renormalize to preserve DC
        A_new = mag_new * np.exp(1j * phase)
        A_new[0] = A[0]  # preserve DC (sum)

        # IFFT back to time domain
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project onto non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a_new / s

    return a

def _adapt_lbfgs(a_init, deadline, min_time_remaining=30.0, max_total_iters=20000):
    """
    Adaptive L-BFGS with stall detection and automatic beta reduction.
    Returns the best (a, score) found.
    """
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return best_a, best_sc

    # Initial beta: proportional to the spread of conv values
    c_sorted = np.sort(c_cur)[::-1]
    n_top = min(20, len(c_sorted))
    gap = c_sorted[0] - c_sorted[n_top - 1]
    base_beta = max(gap, max_c * 0.05)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    stall_count = 0
    total_iters = 0
    prev_sc = current_sc

    while time.monotonic() + min_time_remaining < deadline and total_iters < max_total_iters:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda zz: _lsobj(zz, beta, n),
                z,
                jac=lambda zz: _lsgrad(zz, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 80,
                }

            total_iters += result.nit

            a_new = _softmax(result.x)

                best_sc = sc_new
                best_a = a_new.copy()
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0

                    z = np.log(np.maximum(current_a, 1e-30))

                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()

                stall_count = 0
            elif sc_new < prev_sc - 1e-10:

            active[top_idx] = True
            m = int(active.sum())

        A_vec = active.astype(np.float64)

        # Compute gradient
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Newton direction with Tikhonov regularization
        mu = 1e-6 * (c_max / max(m, 1)) + 1e-10

        def hess_vp(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve for Newton direction
        x = np.zeros(n)
        r = g_t.copy()
        p = r.copy()
        rs = np.dot(r, r)
        Ap = hess_vp(p)
        for _ in range(200):
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x = x + alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                r = r_new
                break
            p_new = r_new + (rs_new / rs) * p
            r = r_new
            p = p_new
            rs = rs_new
            Ap = hess_vp(p)

        # Solve for the scaling to preserve sum
        w_x = np.zeros(n)
        r_w = np.ones(n)
        p_w = r_w.copy()
        rs_w = np.dot(r_w, r_w)
        Ap_w = hess_vp(p_w)
        for _ in range(100):
            pAp_w = np.dot(p_w, Ap_w)
            if pAp_w < 1e-30:
                break
            alpha_w = rs_w / pAp_w
            w_x = w_x + alpha_w * p_w
            r_w_new = r_w - alpha_w * Ap_w
            rs_w_new = np.dot(r_w_new, r_w_new)
            if np.sqrt(rs_w_new) < 1e-10:
                r_w = r_w_new
                break
            p_w_new = r_w_new + (rs_w_new / rs_w) * p_w
            r_w = r_w_new
            p_w = p_w_new
            rs_w = rs_w_new
            Ap_w = hess_vp(p_w)

        w_sum = float(w_x.sum())
        u_sum = float(x.sum())

        if abs(w_sum) < 1e-15:
            dx = -x
        else:
            lam = u_sum / w_sum
            dx = -x + lam * w_x

        dx = dx - np.mean(dx)

        # Line search
        f_old = c_max
        accepted = False
        t = 1.0
        neg = dx < -1e-12
        if neg.any():
            t = min(t, np.min(a[neg] / (-dx[neg])) * 0.99)
        t = max(t, 1e-12)

        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Try gradient descent step
            t_g = 0.02
            for ls in range(30):
                a_try = a - t_g * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t_g *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old - 1e-14:
                    a = a_try
                    accepted = True
                    break
                t_g *= 0.5
            if not accepted:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _peak_deflation_v2(a, deadline, n_rounds=50):
    """
    Targeted peak deflation: identify top peaks and reduce them
    using an analytical direction that minimizes the peak while
    preserving the sum constraint.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find the top peaks (up to 10)
        n_top = min(10, len(c))
        top_indices = np.argsort(c)[::-1][:n_top]
        # Only keep peaks that are within 2% of max
        top_indices = top_indices[c[top_indices] >= 0.98 * c_max]
        if len(top_indices) == 0:
            top_indices = top_indices[:1]

        # Build direction: for each peak position k, the "partner" vector is a[k-i]
        # We want to reduce a at positions where they contribute to peaks
        direction = np.zeros(n)
        for k in top_indices:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                # Weight by the product (contribution)
                direction[i] += a[j]
                if j != i:
                    direction[j] += a[i]

        # Normalize direction
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break
        direction /= dnorm

        # Project onto sum-preserving subspace
        direction_t = direction - np.mean(direction)

        # Line search
        f_old = c_max
        accepted = False
        t = 0.01
        for ls in range(40):
            a_try = a - t * direction_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Try a different approach: reduce the largest elements at peak-contributing positions
            a_try = a.copy()
            max_contrib_idx = -1
            max_contrib_val = 0
            for k in top_indices[:3]:
                i_lo = max(0, k - n + 1)
                i_hi = min(k, n - 1)
                for i in range(i_lo, i_hi + 1):
                    j = k - i
                    prod = a[i] * a[j]
                    if prod > max_contrib_val:
                        max_contrib_val = prod
                        max_contrib_idx = i if a[i] >= a[j] else j
            if max_contrib_idx >= 0:
                a_try[max_contrib_idx] *= 0.99
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    f_try = float(_full_convolve(a_try, a_try).max())
                    if f_try < f_old - 1e-14:
                        a = a_try
                        accepted = True

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2*block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "sqrt_triangle":
        t = np.linspace(0, 1, n)
        a = np.sqrt(1.0 - np.abs(2.0 * t - 1.0))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp_amp":
        # Amplitude-modulated chirp (non-negative)
        t = np.linspace(0, 1, n)
        freq = 1.0 + 10.0 * t
        a = np.abs(np.sin(2.0 * np.pi * np.cumsum(freq) / n)) 
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _fista_peak(a_init, deadline, n_steps=5000):
    """
    FISTA-style accelerated method for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Soft active set
        tol = max(0.02 * c_max, 1e-10)
        active = c >= (c_max - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient
        grad = 2.0 * _corr_a_A(y, w, n)
        grad_t = grad - np.mean(grad)

        # Adaptive step size
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break
        step_size = 0.5 / (gnorm * (1.0 + 0.001 * step))
        step_size = min(step_size, 0.1)

        a_new = y - step_size * grad_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t = t_new

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    Phase 1: Spectral shaping initialization (multiple variants)
    Phase 2: Multi-beta L-BFGS optimization
    Phase 3: Active-set Newton polish
    Phase 4: Targeted peak deflation
    Phase 5: FISTA for additional polish
    Phase 6: Multi-n exploration
    Phase 7: Diversified restarts
    Phase 8: Final micro-optimization
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Generate diverse initializations via spectral shaping
    print(f"\n[Phase 1] Spectral shaping (t={time.monotonic()-t_start:.1f}s)")
    n_main = n_parent if n_parent > 0 else 16384

    inits = {}
    inits["parent"] = parent.copy()

    # Spectral shaping variants
    if time.monotonic() < deadline - 60:
        try:
            inits["spec1"] = _spectral_init(n_main, n_iters=100, target_flatness=0.3)
            inits["spec2"] = _spectral_init(n_main, n_iters=100, target_flatness=0.7)
            inits["spec3"] = _spectral_init_v2(n_main, n_iters=200)
        except Exception:
            pass

    # Structured profiles
    inits["tapered"] = _make_profile(n_main, "tapered")
    inits["bell"] = _make_profile(n_main, "bell")
    inits["sqrt_tri"] = _make_profile(n_main, "sqrt_triangle")

    # Evaluate and pick top initializations
    init_scores = {}
    for name, init in inits.items():
        sc = _score(init)
        init_scores[name] = (sc, init)
        print(f"    {name}: score={sc:.10f}")

    # Sort by score
    sorted_inits = sorted(init_scores.items(), key=lambda x: x[1][0])
    top_inits = [x[1][1] for x in sorted_inits[:4]]  # Take top 4

    # Phase 2: Multi-beta L-BFGS on best initializations
    print(f"\n[Phase 2] L-BFGS optimization (t={time.monotonic()-t_start:.1f}s)")
    for i, init in enumerate(top_inits):
        if time.monotonic() > deadline - 120:
            break
        remaining_budget = deadline - time.monotonic() - 120
        min_rem = max(10, remaining_budget // (len(top_inits) - i))
        a_opt, sc_opt = _adapt_lbfgs(init, min(deadline - 100, time.monotonic() + min_rem),
                                      min_time_remaining=min(60, min_rem // 3),
                                      max_total_iters=10000)
        update_best(a_opt, sc_opt, f"lbfgs_{i}")

    # Phase 3: Active-set Newton on best
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        newton_dl = min(deadline - 60, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton_v2(best_seq, newton_dl, max_iter=400)
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 50:
            newton_dl2 = min(deadline - 40, time.monotonic() + 60)
            a_as2, sc_as2 = _active_set_newton_v2(best_seq, newton_dl2, max_iter=200)
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 4: Peak deflation
    print(f"\n[Phase 4] Peak deflation (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        def_dl = min(deadline - 40, time.monotonic() + 60)
        a_def, sc_def = _peak_deflation_v2(best_seq, def_dl, n_rounds=100)
        update_best(a_def, sc_def, "deflation_1")

        if sc_def < best_score and time.monotonic() < deadline - 30:
            # Follow up with L-BFGS
            a_def2, sc_def2 = _adapt_lbfgs(a_def, deadline, min_time_remaining=20,
                                            max_total_iters=5000)
            update_best(a_def2, sc_def2, "lbfgs_post_def")

    # Phase 5: FISTA
    print(f"\n[Phase 5] FISTA (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        fista_dl = min(deadline - 20, time.monotonic() + 50)
        a_fista = _fista_peak(best_seq, fista_dl, n_steps=8000)
        sc_fista = _score(a_fista)
        update_best(a_fista, sc_fista, "fista")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 40:
            break
        if n_try == n_main and best_seq is not None:
            continue  # Already optimized at this n

        # Interpolate best to new n
        if best_seq is not None:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")

        # Also try spectral init at this n
        if time.monotonic() < deadline - 30:
            try:
                a_spec = _spectral_init(n_try, n_iters=50, target_flatness=0.5)
                if _score(a_spec) < _score(a_init):
                    a_init = a_spec
            except Exception:
                pass

        # Optimize
        a_opt, sc_opt = _adapt_lbfgs(a_init, deadline, min_time_remaining=20,
                                      max_total_iters=5000)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick Newton polish
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 15:
            a_opt2, sc_opt2 = _active_set_newton_v2(a_opt,
                                                     min(deadline - 10, time.monotonic() + 30),
                                                     max_iter=100)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 7: Diversified restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_strategies = [
            ("perturb_0.02", 0.02),
            ("perturb_0.05", 0.05),
            ("perturb_0.1", 0.1),
            ("perturb_0.2", 0.2),
        ]
        for label, sigma in restart_strategies:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _adapt_lbfgs(p_start, deadline, min_time_remaining=10,
                                      max_total_iters=3000)
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 8: Final micro-optimization
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 3:
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(100, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda zz: _lsobj(zz, beta_final, n),
                        z,
                        jac=lambda zz: _lsgrad(zz, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17,
                                 'gtol': 1e-13, 'maxcor': 60, 'maxls': 100}
                    )
                    a_fin = _softmax(result.x)
                    update_best(a_fin, _score(a_fin), "final_tiny_beta")
                except Exception:
                    pass

        # Final Newton
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _active_set_newton_v2(
                best_seq, min(deadline - 1, time.monotonic() + 20), max_iter=200)
            update_best(a_fin2, sc_fin2, "final_newton")

        # Final deflation
        if time.monotonic() < deadline - 1:
            a_fin3, sc_fin3 = _peak_deflation_v2(
                best_seq, min(deadline - 0.5, time.monotonic() + 15), n_rounds=50)
            update_best(a_fin3, sc_fin3, "final_deflation")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
