Multi-temperature L-BFGS optimizer for minimizing max autocorrelation of
non-negative step functions. Uses soft-max smoothing of the max convolution
peak with adaptive temperature cooling, followed by active-set equalization
and subgradient polish. Works directly on the probability simplex.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_obj_grad(x, T):
    """
    Objective and gradient for L-BFGS-B.
    x: raw non-negative vector (will be normalized internally).
    Returns (objective_value, gradient_wrt_x).
    """
    n = len(x)
    a = np.maximum(x, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return 0.0, np.zeros(n)
    a = a / s
    
    c = _fft_convolve(a, a)
    c_scaled = c / T
    log_s = float(logsumexp(c_scaled))
    obj = T * log_s
    
    p = np.exp(c_scaled - log_s)
    
    # Gradient of soft-max w.r.t. a: g_a[j] = 2 * sum_k p[k] * a[k-j]
    g_a = np.zeros(n)
    thresh = p.max() * 1e-12
    active = np.where(p > thresh)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g_a[lo:hi+1] += p[k] * 2.0 * a[k-hi:k-lo+1][::-1]
    
    # Chain rule for normalization: a = x/s
    sum_g = float(g_a.sum())
    g_x = (g_a - x * (sum_g / s)) / s
    
    return obj, g_x


def _lbfgs_optimize(a_init, T, max_iter=3000):
    """Run L-BFGS-B on the soft-max objective."""
    n = len(a_init)
    x0 = a_init.copy()
    
    def func(x):
        return _softmax_obj_grad(x, T)
    
    bounds = [(0.0, 1.0)] * n
    result = minimize(func, x0, method='L-BFGS-B', jac=True,
                      bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                               'maxfun': max_iter * 3})
    x_opt = np.maximum(result.x, 0.0)
    s = x_opt.sum()
    if s < 1e-15:
        return a_init.copy()
    return x_opt / s


def _active_set_refine(a, deadline, max_iter=20000, tol=0.005):
    """Active-set method to equalize top convolution peaks."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks
        tol_val = c_max * (1.0 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        elif len(active) > 20:
            sorted_a = active[np.argsort(c[active])[::-1][:20]]
            active = sorted_a
        
        # Build gradient: average of gradients at active peaks
        g = np.zeros(n)
        K = len(active)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= K
        
        # Project to tangent space of simplex: subtract <g, a> * a / ... 
        # Actually for the simplex {a >= 0, sum = 1}, the tangent space is
        # {d: sum d = 0}. So project: g_proj = g - (sum(g)/n) * 1
        # But we also need to respect a >= 0.
        # Simple projection: g_proj = g - mean(g)
        g_proj = g - g.mean()
        
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-15:
            break
        
        g_proj /= g_norm
        
        # Backtracking line search
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = a - alpha * g_proj
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand = a_cand / s
            sc = _score_norm(a_cand)
            if sc < best_score * (1.0 - 1e-12):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for alpha in [0.0001, 0.00005, 0.00002]:
                a_cand = a - alpha * g_proj
                a_cand = np.maximum(a_cand, 0.0)
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand = a_cand / s
                sc = _score_norm(a_cand)
                if sc < best_score * (1.0 - 1e-14):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    improved = True
                    break
            
            if not improved:
                # Perturb and retry
                a += 0.001 * np.random.randn(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                if it > 100:
                    break
    
    return best_a, best_score


def _coordinate_polish(a, deadline, max_rounds=500):
    """Coordinate descent on the convolution peak."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        
        c = _fft_convolve(best_a, best_a)
        k_max = int(np.argmax(c))
        
        # Find contributing elements
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        improved = False
        for j in range(lo, hi + 1, max(1, (hi - lo + 1) // 50)):
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1.0 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random coordinates
            any_imp = False
            for _ in range(20):
                j = np.random.randint(n)
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.7, 0.9, 1.1, 1.3]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        any_imp = True
                        improved = True
                        break
                if improved:
                    break
            if not any_imp:
                break
    
    return best_a, best_score
