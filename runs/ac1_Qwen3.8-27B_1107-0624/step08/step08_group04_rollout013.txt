"""
Hybrid optimizer for step function inequality: combines L-BFGS with continuation
method, frequency-domain structured perturbations, multi-n resampling, and

import numpy as np
import time
from scipy.optimize import minimize


_next_pow2_cache = {}


def _next_pow2(n):
    if n in _next_pow2_cache:
        return _next_pow2_cache[n]
    k = 1
    while k < n:
        k <<= 1
    result = max(2, k)
    _next_pow2_cache[n] = result
    return result


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=2000, tol=1e-16):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new < 0:
            break
        x = x_new
        r = r_new
        if rs < tol * rs0:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _topk_smooth(c, k):

    if s < 1e-300:

        idx = np.argsort(c)[::-1][:k]
        w[idx] = 1.0 / k
    else:
        w = e / s
    val = float(threshold + (c > threshold).sum() * 0) + float(np.dot(w, c))

    m = top.max()
    lse = m + np.log(np.sum(np.exp(top - m)))
    # Return weighted average that approximates max for k=1
    return float(np.dot(w, c)), w


def _soft_max_obj(z, beta, n, k_ratio=0.0):

    c = _full_convolve(a, a)
    c_max = float(c.max())

        lse = m + np.log(np.sum(np.exp(top_k - m)))

        return obj
    else:

def _soft_max_grad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

    base_beta = max(gap * 0.5, max_c * 0.001)

    if aggressive:
        ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                  0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002,
                  0.00001, 0.000005, 0.000002, 0.000001, 0.0000005]
    else:
        ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                  0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    betas = [base_beta * r for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        est_iter_time = 0.001 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_iter_time * 5))))

        try:
            result = minimize(
                _soft_max_obj,
                z,
                args=(beta, n, 0.0),
                jac=_soft_max_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _qp_solve(W, n, a_init, max_outer=200, cg_iter=3000):
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n) / 2.0)

    for outer in range(max_outer):
        support = a > 1e-12
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            n_sup = n

        if n_sup == n:
            y = _cg_solve(W, n, np.ones(n), max_iter=cg_iter)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                break
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            grad = _hankel_matvec(a, W, n)
            lam = float(grad[support].mean()) if n_sup > 0 else 0.0
            step = 1.0 / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.2)
            a_new = a - step * (grad - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new

        val_new = float(a_new @ _hankel_matvec(a_new, W, n) / 2.0)
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new

        grad = _hankel_matvec(a, W, n)
        support = a > 1e-12
        if support.any():
            lam = float(grad[support].mean())
            resid = float(np.max(np.abs(grad[support] - lam)))
            off_s = ~support
            if off_s.any():
                resid = max(resid, float(np.max(lam - grad[off_s])))
        else:
            resid = np.inf
        if resid < 1e-12 * (np.max(np.abs(grad)) + 1e-15):
            break

    return best_a


def _newton_polish(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002,
                    0.00001, 0.000005, 0.000002, 0.000001, 0.0000005,
                    0.0000002, 0.0000001, 0.00000005, 0.00000002, 0.00000001]
    tol_idx = 0
    rounds_per_tol = 3

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        a_qp = _qp_solve(W, n, a, max_outer=80, cg_iter=2000)
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-15:
                Hg = _cg_solve(W, n, g, max_iter=1000)
                H1 = _cg_solve(W, n, np.ones(n), max_iter=1000)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20 and np.isfinite(H1_norm):
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t / (gn + 1e-30)
            else:
                dx = np.zeros(n)

            alpha = 1.0
            improved = False
            for ls in range(60):
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < c_max * (1 - 1e-12):
                    a = a_try
                    sc_try = _score(a)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a.copy()
                    improved = True
                    break
                alpha *= 0.5
                if alpha < 1e-16:
                    break

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _dft_phase_perturb(a, rng, num_modes=None):
    """Perturb the DFT phase of a while keeping amplitude, then project to non-negative."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.fft(a, N)
    if num_modes is None:
        num_modes = n // 4
    freqs = np.arange(N)
    # Perturb phases for non-DC, non-Nyquist frequencies
    mask = (freqs > 0) & (freqs < N // 2)
    # Only perturb a subset of frequencies
    active_freqs = np.where(mask)[0]
    n_perturb = min(num_modes, len(active_freqs))
    chosen = rng.choice(active_freqs, size=n_perturb, replace=False)
    phase_perturb = 2 * np.pi * rng.standard_normal(n_perturb) * 0.1
    A[chosen] *= np.exp(1j * phase_perturb)
    # Enforce Hermitian symmetry
    for i in chosen:
        if N - i < N and N - i != i:
            A[N - i] = np.conj(A[i])
    a_new = np.real(np.fft.ifft(A))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _amplitude_perturb(a, rng, sigma=0.05):
    """Perturb the DFT amplitude (log-scale) and reconstruct."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.fft(a, N)
    amp = np.abs(A)
    phase = np.angle(A)
    log_amp = np.log(np.maximum(amp, 1e-30))
    log_amp += sigma * rng.standard_normal(N)
    # Maintain Hermitian symmetry
    log_amp[N - np.arange(N)] = log_amp[np.arange(N)]
    log_amp[0] = np.max(log_amp)  # Keep DC dominant
    amp_new = np.exp(np.clip(log_amp, -30, 30))
    A_new = amp_new * np.exp(1j * phase)
    # Enforce Hermitian
    for i in range(1, N // 2):
        A_new[N - i] = np.conj(A_new[i])
    a_new = np.real(np.fft.ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _half_mixed_perturb(a, rng, mix=0.3):
    """Mix the first half with the reversed second half."""
    n = len(a)
    half = n // 2
    a_rev = a[::-1].copy()
    # Blend: a_new[i] = (1-mix)*a[i] + mix*a_rev[i] for i < half
    a_new = a.copy()
    a_new[:half] = (1 - mix) * a[:half] + mix * a_rev[:half]
    a_new[half:] = (1 - mix) * a[half:] + mix * a_rev[half:]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.3) ** 2) + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "comb":
        a = np.ones(n)
        a[::3] *= 1.5
        a[1::3] *= 0.7
        return a / a.sum()
    else:
        return np.ones(n) / n


def _flatten_direct(a, deadline, max_iters=300):
    """Direct gradient descent on the sum of squared excess above mean."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.5

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_mean = float(c.sum()) / len(c)
        c_max = float(c.max())
        excess = np.maximum(c - c_mean, 0)
        excess_sq = float(np.sum(excess ** 2))
        if excess_sq < 1e-30:
            break

        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad = excess_conv[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.8
            if lr < 1e-6:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: L-BFGS continuation, DFT perturbations,
    direct flatten, and progressive Newton polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc)
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}, score={_score(parent):.4f}")
        update_best(parent)

    n = len(parent)

    # === Phase 1: L-BFGS from parent ===
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        t1_deadline = min(deadline - 350, t_start + 400)
        a1 = _lbfgs_optimize(parent, t1_deadline, min_remaining=50, aggressive=True)
        update_best(a1)

    # === Phase 2: Direct flatten ===
    print(f"\n[Phase 2] Direct flatten (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        t2_deadline = min(deadline - 280, time.monotonic() + 120)
        a2, sc2 = _flatten_direct(best_seq, t2_deadline, max_iters=500)
        update_best(a2, sc2)

    # === Phase 3: L-BFGS again from best ===
    print(f"\n[Phase 3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        t3_deadline = min(deadline - 230, time.monotonic() + 200)
        a3 = _lbfgs_optimize(best_seq, t3_deadline, min_remaining=60, aggressive=True)
        update_best(a3)

    # === Phase 4: Structured perturbations ===
    print(f"\n[Phase 4] Structured perturbations (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        perturbations = []

        # DFT phase perturbations
        for i in range(3):
            p = _dft_phase_perturb(best_seq, rng, num_modes=n // 8)
            perturbations.append(p)

        # Amplitude perturbations
        for i in range(2):
            p = _amplitude_perturb(best_seq, rng, sigma=0.03)
            perturbations.append(p)

        # Half-mix perturbations
        for mix in [0.2, 0.4, 0.6]:
            p = _half_mixed_perturb(best_seq, rng, mix=mix)
            perturbations.append(p)

        # Log-space random perturbations
        for sigma in [0.02, 0.05, 0.1]:
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            z += sigma * rng.standard_normal(n)
            z = z - z.mean()
            perturbations.append(_softmax(z))

        # Different initial profiles resampled to n
        for prof in ["cos", "bimodal", "skewed", "comb"]:
            p = _make_profile(n, prof)
            # Blend with best: 70% best + 30% profile
            p = 0.7 * best_seq + 0.3 * p
            p = np.maximum(p, 0)
            p = p / p.sum()
            perturbations.append(p)

        # Optimize each perturbation with short L-BFGS
        per_perturb_time = min(15.0, (deadline - 200 - time.monotonic()) / max(1, len(perturbations)))
        for i, p in enumerate(perturbations):
            if time.monotonic() > deadline - 180:
                break
            sc_p = _score(p)
            if sc_p < best_score + 0.01:
                p_opt = _lbfgs_optimize(p, time.monotonic() + per_perturb_time,
                                         min_remaining=2, aggressive=False)
                update_best(p_opt)
            if i % 3 == 0:
                print(f"  Perturbation {i+1}/{len(perturbations)} done, "
                      f"best={best_score:.8f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Newton polish ===
    print(f"\n[Phase 5] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        t5_deadline = min(deadline - 80, time.monotonic() + 200)
        a5, sc5 = _newton_polish(best_seq, t5_deadline, max_rounds=300)
        update_best(a5, sc5)

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        for n_try in [n * 2, n // 2]:
            if time.monotonic() > deadline - 50:
                break
            if n_try < 256 or n_try > 65536:
                continue
            if n_try == n:
                continue
            a_resamp = _resample(best_seq, n_try)
            sc_res = _score(a_resamp)
            print(f"  n={n_try}: resampled score={sc_res:.8f}")
            if time.monotonic() < deadline - 40:
                a_opt = _lbfgs_optimize(a_resamp, min(deadline - 35, time.monotonic() + 80),
                                         min_remaining=10, aggressive=False)
                update_best(a_opt)
                if _score(a_opt) < best_score + 0.005 and time.monotonic() < deadline - 20:
                    a_np, sc_np = _newton_polish(a_opt, min(deadline - 15, time.monotonic() + 60),
                                                  max_rounds=100)
                    update_best(a_np, sc_np)

    # === Phase 7: Final ultra-tight Newton ===
    print(f"\n[Phase 7] Final Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        t7_deadline = deadline - 3
        a7, sc7 = _newton_polish(best_seq, t7_deadline, max_rounds=500)
        update_best(a7, sc7)

    # === Phase 8: Final flatten + L-BFGS squeeze ===
    print(f"\n[Phase 8] Final squeeze (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fl, sc_fl = _flatten_direct(best_seq, deadline - 3, max_iters=200)
        update_best(a_fl, sc_fl)
        if time.monotonic() < deadline - 3:
            a_s = _lbfgs_optimize(best_seq, deadline - 1, min_remaining=0, aggressive=False)
            update_best(a_s)

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
