"""
Multi-phase optimizer for minimizing 2*n*max_conv/sum^2 over non-negative sequences.
Key improvements over previous version:
1. More aggressive L-BFGS beta continuation with adaptive scheduling
2. New iterative peak-deflation polish (direct gradient on active peaks)
3. Multi-resolution warm start
4. Better time management
5. Block coordinate descent as an additional polish phase
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=800, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.2:
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    if aggressive:
        ratios = np.array([1.0, 0.4, 0.16, 0.064, 0.0256, 0.01, 0.004, 0.0016,
                           0.00064, 0.000256, 0.0001, 0.00004, 0.000016,
                           0.0000064, 0.00000256, 0.000001, 0.0000004,
                           0.00000016, 0.000000064, 0.0000000256,
                           0.00000001, 0.000000004, 0.0000000016,
                           0.00000000064])
    else:
        ratios = np.array([1.0, 0.4, 0.16, 0.064, 0.0256, 0.01, 0.004,
                           0.0016, 0.00064, 0.000256, 0.0001, 0.00004,
                           0.000016, 0.0000064, 0.00000256, 0.000001])

    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_iter_time * 5))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _flatten_polish(a_init, deadline, max_iters=300):
    """Convolution flattening: reduce variance of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_mean = np.median(c)  # use median instead of mean for robustness
        c_max = float(c.max())
        
        # Excess above median
        excess = np.maximum(c - c_mean, 0)
        excess_max = excess.max()
        if excess_max < 1e-15:
            break

        # Gradient of sum(excess^2)
        a_rev = a[::-1]
        excess_conv = _full_convolve(a_rev, excess)
        grad = 2.0 * excess_conv[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)
        
        step = 0.05 / (1.0 + 0.005 * it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_half = a - 0.5 * step * grad
            a_half = np.maximum(a_half, 0)
            s_half = a_half.sum()
            if s_half > 1e-15:
                a_half = a_half / s_half
                sc_half = _score(a_half)

                    best_sc = sc_half
                    best_a = a_half.copy()
                    a = a_half
                else:
                    break  # Stuck
            else:
                break

    return best_a, best_sc


def _block_coordinate_descent(a_init, deadline, max_rounds=200):
    """Block coordinate descent: optimize blocks of coordinates at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    block_size = max(8, n // 64)
    n_blocks = (n + block_size - 1) // block_size
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for blk in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = blk * block_size
            end = min(start + block_size, n)
            
            # Fix all coordinates except the block
            # Optimize the block to minimize the max convolution
            # This is a small subproblem
            
            # Simple approach: for each coordinate in the block, try to decrease it
            # if it reduces the peak
            c = _full_convolve(a, a)
            c_max = float(c.max())
            
            for i in range(start, end):
                if a[i] < 1e-15:
                    continue
                # Try reducing a[i] by a small amount
                delta = a[i] * 0.1
                a_try = a.copy()
                a_try[i] -= delta
                # Redistribute to maintain sum
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                else:
                    # Try increasing (if it's currently small)
                    pass
            
            if not improved and rnd > 50:
                break
    
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = t ** 0.3 + (1-t) ** 0.3 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2` over non-negative sequences. The key insight is that the convolution peak determines the score, and we want the self-convolution to be as flat as possible.

My approach focuses on:
1. **Aggressive L-BFGS** with very smooth → very sharp beta continuation (log-sum-exp smoothing of the max)
2. **New iterative peak deflation** using direct gradient on active peak lags with line search
3. **Adaptive flattening** that targets the upper quantile of the convolution
4. **Multi-n exploration** with upsampling
5. **Interleaved optimization phases** to avoid getting stuck

"""
Multi-phase optimizer: aggressive L-BFGS beta continuation, iterative peak
deflation with line search, adaptive convolution flattening, and multi-n
exploration. Targets score < 1.5027 from parent ~1.5057.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    if aggressive:
        ratios = np.array([1.0, 0.4, 0.16, 0.064, 0.0256, 0.01, 0.004, 0.0016,
                           0.00064, 0.000256, 0.0001, 0.00004, 0.000016,
                           0.0000064, 0.00000256, 0.000001, 0.0000004,
                           0.00000016, 0.000000064, 0.0000000256,
                           0.00000001, 0.000000004, 0.0000000016,
                           0.00000000064, 0.000000000256])
    else:
        ratios = np.array([1.0, 0.4, 0.16, 0.064, 0.0256, 0.01, 0.004,
                           0.0016, 0.00064, 0.000256, 0.0001, 0.00004,
                           0.000016, 0.0000064])

    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_iter_time * 5))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_deflation(a_init, deadline, max_iters=500):
    """
    Iteratively reduce the peak convolution value using projected gradient
    steps on the set of active (near-max) lags.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_c_max = float(_full_convolve(a, a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        if c_max < best_c_max * (1 - 1e-13):
            best_c_max = c_max
            best_a = a.copy()
            best_sc = _score(a)

        # Find active peaks
        tol = max(0.0005 * c_max, 1e-15)
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Weight vector: uniform over active peaks
        w_arr = np.zeros_like(c)
        w_arr[active] = 1.0 / n_active

        # Gradient: grad[i] = 2 * sum_m a[m] * w[i+m]
        # = 2 * convolve(a[::-1], w)[n-1:2n-1]
        a_flipped = a[::-1]
        bw = _full_convolve(a_flipped, w_arr)
        grad = 2.0 * bw[n - 1:2 * n - 1]

        # Center the gradient (preserve sum constraint)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Adaptive step with backtracking line search
        step_base = 0.05 / (1.0 + 0.005 * it)
        alpha = step_base
        found_step = False
        for _ in range(25):
            a_try = a - alpha * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1 - 1e-12):
                a = a_try
                found_step = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not found_step and alpha < 1e-14:
            # Try a much smaller step
            alpha = 1e-8 / (1.0 + 0.01 * it)
            a_try = a - alpha * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try

    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=300):
    """Reduce upper tail of the convolution distribution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        # Use 75th percentile as threshold
        threshold = np.percentile(c, 75)
        c_max = float(c.max())

        excess = np.maximum(c - threshold, 0)
        if excess.max() < 1e-15:
            break

        # Gradient of sum(excess^2) w.r.t. a
        a_flipped = a[::-1]
        excess_conv = _full_convolve(a_flipped, excess)
        grad = 2.0 * excess_conv[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = 0.03 / (1.0 + 0.005 * it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try half step
            a_half = a - 0.5 * step * grad
            a_half = np.maximum(a_half, 0)
            s_half = a_half.sum()
            if s_half > 1e-15:
                a_half = a_half / s_half
                sc_half = _score(a_half)
                if sc_half < best_sc:
                    best_sc = sc_half
                    best_a = a_half.copy()
                    a = a_half
                elif it > 50:
                    break
            elif it > 50:
                break

    return best_a, best_sc


def _subgradient_nesterov(a_init, deadline, max_steps=3000):
    """Subgradient method with Nesterov momentum on smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Smoothed gradient with adaptive beta
        beta = c_max * 0.01
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_flipped = a[::-1]
        wc = _full_convolve(w, a_flipped)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)

        step = lr / (1.0 + 0.0003 * i)
        z_new = z - step * grad
        # Nesterov
        z_new = z_prev + 0.9 * (z_new - z) + 0.1 * (-step * grad)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    patience = 0
                    lr *= 0.7

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = t ** 0.3 + (1.0 - t) ** 0.3 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0, aggressive=True)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 40:
            a_pd, sc_pd = _peak_deflation(a_opt, min(deadline - 30, time.monotonic() + 50))
            update_best(a_pd, sc_pd, f"n_{n_try}_pd")

    # Phase 8: Final L-BFGS
    print(f"\n[Phase 8] Final L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a8 = _run_lbfgs(best_seq, deadline - 5, min_time_remaining=5.0, aggressive=True)
        update_best(a8, _score(a8), "final_lbfgs")

    # Phase 9: Final peak deflation + flatten
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a9, sc9 = _peak_deflation(best_seq, deadline - 3, max_iters=100)
        update_best(a9, sc9, "final_pd")
        if time.monotonic() < deadline - 2:
            a10, sc10 = _flatten_polish(best_seq, deadline - 1, max_iters=50)
            update_best(a10, sc10, "final_flatten")

    # Final
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
