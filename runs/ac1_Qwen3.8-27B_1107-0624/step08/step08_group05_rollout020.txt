        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _newton_adaptive(a_init, deadline, max_iter=300, tol_start=0.05, tol_end=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    log_tol_start = np.log(tol_start)
    log_tol_end = np.log(tol_end)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iter - 1)
        tol_frac = np.exp(log_tol_start + frac * (log_tol_end - log_tol_start))

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8 * (1.0 + 0.1 * frac)
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha_step = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha_step * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha_step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha_step * gnorm:
                a_new = a_try
                break
            alpha_step *= 0.5
        else:
            if alpha_step < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _proj_grad_qp(a_init, active, n, deadline, max_iter=500, lr_init=0.01):
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    active_f = active.astype(np.float64)
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        a_new = a - lr * grad_t
        a_new = _project_simplex(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        c_old = _full_convolve(a, a)
        c_new = _full_convolve(a_new, a_new)
        obj_old = c_old[active].sum()
        obj_new = c_new[active].sum()
        if obj_new < obj_old:
            a = a_new
            lr = min(lr * 1.05, 0.1)
            patience = 0
        else:
            lr *= 0.8
            patience += 1
            if patience > 30:
                break
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _variance_obj(a):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    return float(np.mean((c - mean_c) ** 2))


def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    return g - np.mean(g)


def _variance_warm_start(a_init, deadline, n_steps=500):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.002
    lr_decay = 0.999

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad = _variance_grad(a, n)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
        lr *= lr_decay
        if i % 50 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=None):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    if n_stages is None:
        n_stages = max(10, min(14, int((deadline - time.monotonic() - min_remaining) / 20)))

    log_betas = np.linspace(np.log(base_beta), np.log(max(base_beta * 0.0001, 1e-10)), n_stages)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in np.exp(log_betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _lse_objective, z, args=(float(beta), n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-13, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-13, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_anti_corr(a, rng, sigma=0.05):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    active = c > 0.95 * c_max
    a_rev = a[::-1]
    wc = _full_convolve(active.astype(float), a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(active.astype(float), c))
    grad = h - W
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - sigma * grad / (np.linalg.norm(grad) + 1e-15)
    z = z - z.mean()
    return _softmax(z)


def _perturb_mirror_mix(a, rng, alpha=0.3):
    a_mirror = a[::-1].copy()
    noise = rng.standard_normal(len(a)) * 0.01
    a_new = (1 - alpha) * a + alpha * a_mirror + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "tukey":
        t = np.linspace(0, 1, n)
        r = 0.5
        a = np.ones(n)
        m = int(r * n / 2)
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (r * n) - 1)))
            a[n - 1 - i] = a[i]
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "kaiser":
        t = np.linspace(-1, 1, n)
        beta_k = 5.0
        i0 = np.ones(n)
        from scipy.special import i0 as _i0
        a = _i0(beta_k * np.sqrt(1 - t**2)) / _i0(beta_k)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _multi_start_explore(n, deadline, rng):
    best_a = None
    best_sc = np.inf
    profiles = ["uniform", "tapered", "cos2", "wheatstone", "gauss", "bimodal", "ramp", "tukey"]
    for kind in profiles:
        if time.monotonic() > deadline:
            break
        a_try = _make_profile(n, kind)
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    if best_a is None:
        best_a = _make_profile(n, "uniform")
        best_sc = _score(best_a)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-stage adaptive optimizer combining variance warm-start,
    LSE with geometric temperature, adaptive active-set Newton,
    projected-gradient QP, subgradient polish, and structured restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init_uniform")

    n_work = n_parent if n_parent >= 1024 else 8192

    # === Phase 1: Variance warm start (40s) ===
    print(f"\n[Phase 1] Variance warm start")
    p1_deadline = min(deadline, t_start + 40)
    if _best_seq is not None and n_work > 0:
        a_work = _resize_sequence(_best_seq, n_work) if len(_best_seq) != n_work else _best_seq.copy()
        a_var, sc_var = _variance_warm_start(a_work, p1_deadline, n_steps=200)
        upd(a_var, sc_var, "variance_warm")

    # === Phase 2: Quick diverse exploration (30s) ===
    print(f"\n[Phase 2] Diverse exploration")
    p2_deadline = min(deadline, t_start + 70)
    n_expl = max(1024, n_work // 8)
    for kind in ["tukey", "kaiser", "gauss", "bimodal", "tapered"]:
        if time.monotonic() > p2_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 3: Main LSE with geometric temperature (300s) ===
    print(f"\n[Phase 3] Main L-BFGS LSE (geometric temp)")
    if time.monotonic() < deadline - 300:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=400, n_stages=14)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 4: Adaptive active-set Newton (400s) ===
    print(f"\n[Phase 4] Adaptive Newton (geometric tol)")
    if time.monotonic() < deadline - 250:
        nw_dead = min(deadline - 200, time.monotonic() + 400)
        a_nw, sc_nw = _newton_adaptive(_best_seq, nw_dead, max_iter=400, tol_start=0.05, tol_end=0.001)
        upd(a_nw, sc_nw, "newton_adaptive")

    # === Phase 5: Projected gradient QP (100s) ===
    print(f"\n[Phase 5] Projected Gradient QP")
    if time.monotonic() < deadline - 150:
        a_cur = _best_seq.copy()
        c_cur = _full_convolve(a_cur, a_cur)
        c_max = c_cur.max()
        active = c_cur > 0.98 * c_max
        pg_dead = min(deadline - 120, time.monotonic() + 100)
        a_pg, sc_pg = _proj_grad_qp(a_cur, active, len(a_cur), pg_dead, max_iter=300, lr_init=0.005)
        upd(a_pg, sc_pg, "proj_grad_qp")

        # Second round with tighter active set
        if sc_pg < _best_score and time.monotonic() < deadline - 80:
            c_cur2 = _full_convolve(a_pg, a_pg)
            c_max2 = c_cur2.max()
            active2 = c_cur2 > 0.99 * c_max2
            pg_dead2 = min(deadline - 60, time.monotonic() + 50)
            a_pg2, sc_pg2 = _proj_grad_qp(a_pg, active2, len(a_cur), pg_dead2, max_iter=200, lr_init=0.003)
            upd(a_pg2, sc_pg2, "proj_grad_qp_2")

    # === Phase 6: Subgradient polish (100s) ===
    print(f"\n[Phase 6] Subgradient polish")
    if time.monotonic() < deadline - 100:
        sg_dead = min(deadline - 80, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=4000, lr_init=0.006, tol_frac=0.0008)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 7: Power-sum L-BFGS (150s) ===
    print(f"\n[Phase 7] Power-sum L-BFGS")
    if time.monotonic() < deadline - 120:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=80)
        upd(a_ps, sc_ps, "lbfgs_power_sum")

    # === Phase 8: Newton + SG alternating (150s) ===
    print(f"\n[Phase 8] Newton + SG alternating")
    if time.monotonic() < deadline - 120:
        a_cur = _best_seq.copy()
        for round_i in range(3):
            if time.monotonic() > deadline - 80:
                break
            nw_dead = min(deadline - 80, time.monotonic() + 40)
            a_nw2, sc_nw2 = _newton_adaptive(a_cur, nw_dead, max_iter=100, tol_start=0.03, tol_end=0.002)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")
            sg_dead2 = min(deadline - 80, time.monotonic() + 30)
            a_sg2, sc_sg2 = _subgrad_polish(a_cur, sg_dead2, n_steps=1500, lr_init=0.004, tol_frac=0.0008)
            if sc_sg2 < _score(a_cur):
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 9: Multi-n exploration (250s) ===
    print(f"\n[Phase 9] Multi-n exploration")
    n_candidates = [4096, 8192, 12288, 16384, 20480]
    if n_parent not in n_candidates and n_parent >= 2048:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if len(_best_seq) == n_try:
            a_init_n = _best_seq.copy()
        else:
            a_init_n = _resize_sequence(_best_seq, n_try)

        # LSE at this size
        a_opt_n, sc_opt_n = _lbfgs_lse(a_init_n, deadline, min_remaining=50, n_stages=8)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        # Newton polish
        if time.monotonic() < deadline - 50:
            nw_dead = min(deadline - 40, time.monotonic() + 60)
            a_opt_n2, sc_opt_n2 = _newton_adaptive(a_opt_n, nw_dead, max_iter=80, tol_start=0.03, tol_end=0.001)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 10: Diverse restarts (250s) ===
    print(f"\n[Phase 10] Diverse restarts")
    if time.monotonic() < deadline - 60:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.4)
            elif i == 2:
                p_start = _perturb_block_swap(_best_seq, rng, n_swaps=6)
            elif i == 3:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.1)
            elif i == 4:
                p_start = _perturb_anti_corr(_best_seq, rng, sigma=0.03)
            elif i == 5:
                p_start = _perturb_mirror_mix(_best_seq, rng, alpha=0.2)
            elif i == 6:
                p_start = _perturb_log(_best_seq, 0.06, rng)
            else:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=15, max_amp=0.2)

            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=25, n_stages=8)
            upd(a_r, sc_r, f"restart_{i}")

            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 20:
                nw_dead = min(deadline - 15, time.monotonic() + 30)
                a_r2, sc_r2 = _newton_adaptive(a_r, nw_dead, max_iter=60, tol_start=0.03, tol_end=0.002)
                upd(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 11: Final aggressive polish (80s) ===
    print(f"\n[Phase 11] Final aggressive polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.00003
            max_iter = min(2000, max(50, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(float(beta_final), n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-14, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton with very tight tolerance
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_adaptive(_best_seq, deadline - 2, max_iter=60, tol_start=0.01, tol_end=0.0005)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=4000, lr_init=0.003, tol_frac=0.0003)
            upd(a_fin3, sc_fin3, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
