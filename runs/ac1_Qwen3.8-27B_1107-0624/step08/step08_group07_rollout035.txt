"""
Optimized step function search using FFT-based smooth gradient descent,
active-set equalization, and multiplicative updates.
Key innovations: (1) efficient FFT-based gradient for log-sum-exp objective,
(2) line search with proper descent direction, (3) combined additive/multiplicative
updates, (4) aggressive active-set equalization with Newton steps.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    """Compute 2*n*max(c) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())

def _score(a):
    """Full score computation matching evaluate_sequence."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _compute_smooth_grad(a, c, T):
    """Compute gradient of log-sum-exp(T*c) w.r.t. a using FFT.
    Returns grad[j] = 2 * sum_k p[k] * a[k-j]
    where p = softmax(T*c) (normalized to sum=1)."""
    n = len(a)
    c_shifted = c - c.max()
    e = np.exp(T * c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    
    # grad[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * conv(p, a_reversed)[j + n - 1]
    a_rev = a[::-1].copy()
    N = _next_pow2(len(p) + n - 1)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(P * A, N)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad

def _active_set_direction(a, c, top_k=10, tol_frac=0.001):
    """Find the direction that equalizes the top peaks.
    Solves: minimize ||d|| subject to M*d = -1, sum(d) = 0
    where M[i,j] = 2*a[k_i - j] for the active peaks k_i."""
    n = len(a)
    c_max = float(c.max())
    
    # Find active peaks
    tol = c_max * (1 - tol_frac)
    active = np.where(c >= tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    elif len(active) > top_k:
        # Keep top_k
        sorted_active = active[np.argsort(c[active])[::-1][:top_k]]
        active = sorted_active
    
    K = len(active)
    # Build M (K x n)
    M = np.zeros((K, n))
    for i, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                M[i, j] = 2.0 * a[m]
    
    # Add sum constraint row
    M_full = np.vstack([M, np.ones(n)])
    b = np.concatenate([-np.ones(K), [0.0]])
    
    # Minimum norm solution: d = M_full^T * inv(M_full * M_full^T) * b
    G = M_full @ M_full.T  # (K+1) x (K+1)
    try:
        G_inv = np.linalg.inv(G + 1e-10 * np.eye(G.shape[0]))
        d = M_full.T @ (G_inv @ b)
    except np.linalg.LinAlgError:
        d = np.linalg.lstsq(M_full, b, rcond=None)[0]
    
    return d, active

def _line_search(a, d, c, current_score, max_iter=30):
    """Backtracking line search to find a step size that decreases the score."""
    n = len(a)
    best_alpha = 1.0
    best_sc = current_score
    alpha = 1.0
    
    for _ in range(max_iter):
        a_new = a + alpha * d
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s
        
        c_new = _fft_convolve(a_new, a_new)
        sc_new = float(c_new.max())
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_alpha = alpha
            best_a = a_new.copy()
            best_c = c_new
        else:
            best_a = a_new  # still return the best
            # Armitage condition: sufficient decrease
            if sc_new < best_sc + 1e-12:
                break
        
        alpha *= 0.5
        if alpha < 1e-10:
            break
    
    return best_a, best_sc, best_alpha

                              lr_start=0.01, max_iter=100000):

    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    c = _fft_convolve(a, a)
    best_a = a.copy()
    best_sc = float(c.max())
    
    t = 0
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

"""
Optimized step function search using FFT-based smooth gradient descent with
adaptive temperature, line search, active-set Newton equalization, and
multiplicative weight updates. Designed to improve upon the 1.505685 baseline.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1: 2*n*max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_gradient(a, c, T):
    """Compute gradient of (1/T)log(sum(exp(T*c))) w.r.t. a.
    Uses FFT for O(n log n) computation.
    grad_a[j] = 2 * sum_k p[k] * a[k-j]
    where p[k] = exp(T*(c[k]-max(c))) / sum(exp(T*(c-max(c))))"""
    n = len(a)
    c_shifted = c - c.max()
    e = np.exp(T * c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    
    # grad_a[j] = 2 * conv(p, a_rev)[j + n - 1]
    # conv has length len(p) + n - 1 = 3n - 2
    # We need indices [n-1, ..., 2n-2]
    a_rev = a[::-1].copy()
    N = _next_pow2(3 * n - 2)
    P = np.fft.rfft(p, N)
    AR = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(P * AR, N)
    grad_a = 2.0 * conv[n - 1:2 * n - 1]
    return grad_a
