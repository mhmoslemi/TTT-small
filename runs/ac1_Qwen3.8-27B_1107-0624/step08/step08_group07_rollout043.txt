```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _fft_convolve(a_norm, a_norm)
    return 2.0 * n * float(c.max())

def _proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    k = np.sum(cond)
    theta = (cssv[k-1] - 1) / k
    return np.maximum(v - theta, 0)
```

    k = 1
```python
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    c = _fft_convolve(a, a)
```

```python
def _proj_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cv - 1.0) / rho
    k = int(np.sum(cond))
    if k == 0:
        return np.ones(n) / n
    theta = (cv[k - 1] - 1.0) / k
    return np.maximum(v - theta, 0.0)


def _smooth_grad(a, c, T):
    """Smoothed gradient of max(c) via softmax weighting + FFT cross-correlation."""
    n = len(a)
    z = (c - c.max()) / T
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    # g[j] = 2 * sum_k p[k+j]*a[k] via FFT
    N = _next_pow2(3 * n - 2)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(P) * A, N)
    return 2.0 * cross[:n]


def _active_grad(a, c, top_k=5):
    """Average subgradient over top-k peaks."""
    n = len(a)
    idx = np.argpartition(c, -top_k)[-top_k:]
    g = np.zeros(n)
    for k in idx:

        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= top_k
    return g
```

The advantage of FW is that it maintains the simplex constraint exactly (by convex combination) and doesn't require projection. The disadvantage is that it can converge slowly (O(1/t)).

```python
import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]
```

Step function optimization using FFT-accelerated smoothed gradient descent

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]
