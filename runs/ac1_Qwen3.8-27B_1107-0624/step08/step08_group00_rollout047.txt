"""
Hybrid momentum-subgradient + L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s


def _subgradient_topk(a, c, K):
    """Compute subgradient of max(conv(a,a)) averaged over top-K peaks.
    Returns gradient vector of length n."""
    n = len(a)
    top_k = np.argsort(c)[::-1][:K]
    grad = np.zeros(n)
    for k in top_k:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # grad[j] += 2*a[k-j] for j in [j_min, j_max]
        # a[k-j] for j from j_min to j_max:
        # = a[k-j_min], a[k-j_min-1], ..., a[k-j_max]
        # = a[k-j_max : k-j_min+1][::-1]
        lo = k - j_max
        hi = k - j_min + 1
        grad[j_min:j_max+1] += 2.0 * a[lo:hi][::-1]
    grad /= float(K)
    return grad


def _momentum_optimize(a_init, deadline, min_time_remaining=60.0):
    """Momentum-based subgradient optimization with adaptive K."""
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.05 / np.sqrt(n)  # scale with n
    K = max(3, n // 100)  # start with top 1%
    K_min = 1
    K_max = max(5, n // 50)
    
    no_improve = 0
    max_no_improve = 200
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive K: reduce K over time (more focus on main peak)
        progress = min(1.0, (time.monotonic() - t_start) / 300.0)
        K = max(K_min, int(K_max * (1 - progress * 0.9) + K_min))
        K = min(K, len(c) - 1)
        
        # Compute subgradient
        grad = _subgradient_topk(a, c, K)
        # Center to preserve sum
        grad -= grad.mean()
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project to simplex (a >= 0, sum = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s

            s_try = a_try.sum()

                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            a = a_new  # accept anyway to make progress
            # Reduce learning rate
            lr *= 0.8
            if no_improve > max_no_improve:
                # Restart from best
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.05 / np.sqrt(n)
                momentum = 0.9
                no_improve = 0
        
        # Adapt learning rate

        if (time.monotonic() - t_start) % 10 < 0.1:

def _lbfgs_optimize(a_init, deadline, min_time_remaining=30.0, beta_scale=0.003):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a, best_sc
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    betas = [max_c * beta_scale * f for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]]
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        # Local objective
        def obj(z_val):
            a_loc = _softmax(z_val)
            c_loc = _full_convolve(a_loc, a_loc)
            c_max_loc = float(c_loc.max())
            shifted = (c_loc - c_max_loc) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max_loc
            return 2.0 * n * (c_max_loc + beta * np.log(s_e))
        
        def grad(z_val):
            a_loc = _softmax(z_val)
            c_loc = _full_convolve(a_loc, a_loc)
            c_max_loc = float(c_loc.max())
            shifted = (c_loc - c_max_loc) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_loc)
                w[int(np.argmax(c_loc))] = 1.0
            else:
                w = e / s_e
            a_rev = a_loc[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c_loc))
            return 4.0 * n * a_loc * (h - W)
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(20, int(remaining / (est_time * 15))))
        
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-15,
                                      'gtol': 1e-12, 'maxcor': 50, 'maxls': 30})
            a_new = _softmax(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a, _score_normalized(best_a)


def _deflation_step(a, c, n_peaks=5, reduction_frac=0.01):
    """Deflation: reduce elements contributing most to top peaks.
    Returns modified a."""
    n = len(a)
    a_new = a.copy()
    
    top_k = np.argsort(c)[::-1][:n_peaks]
    
    for k in top_k:
        # Find elements contributing most to c[k]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        
        # Contribution of a[j] to c[k] is a[j]*a[k-j]
        contributions = np.zeros(n)
        for j in range(j_min, j_max + 1):
            contributions[j] = a[j] * a[k-j]
        
        # Reduce top contributors
        total_contrib = contributions.sum()
        if total_contrib < 1e-30:
            continue
        
        # Target: reduce c[k] by reduction_frac
        target_reduction = c[k] * reduction_frac
        n_reduce = max(1, int(n * 0.002))
        top_idx = np.argsort(contributions)[::-1][:n_reduce]
        
        for idx in top_idx:
            if contributions[idx] > 1e-30:
                # Reduce a[idx] to reduce contribution
                reduction = target_reduction * contributions[idx] / total_contrib
                a_new[idx] -= reduction / (2.0 * a[k-idx] + 1e-30)
    
    a_new = np.maximum(a_new, 0.0)
    return _normalize(a_new)


def _structured_perturb(a, freq, amplitude, phase, rng):
    """Apply a sinusoidal perturbation to a."""
    n = len(a)
    t = np.arange(n)
    perturbation = 1.0 + amplitude * np.sin(2.0 * np.pi * freq * t / n + phase)
    perturbation = np.maximum(perturbation, 0.01)
    a_new = a * perturbation
    return _normalize(a_new)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.3) ** 2) + 0.1 * np.exp(-5.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    return _normalize(a_new)
