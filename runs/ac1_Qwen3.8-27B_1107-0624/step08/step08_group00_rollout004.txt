5. Compute g = irfft(2*A*W, N)[:n]
6. Update a using mirror descent: a *= exp(-lr*g); a /= sum(a)

"""
Nesterov-accelerated mirror descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: (1) FFT-based gradient computation in O(n log n), (2) adaptive
temperature schedule for log-sum-exp smoothing, (3) interleaved exact subgradient
steps, (4) QP-based peak reduction for final refinement, (5) multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _fft_convolve_aa(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r, N, fa


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_conv_max(a, T, N=None, fa=None, C=None):
    """
    Compute gradient of smoothed max of conv(a,a) with respect to a.
    Returns (gradient, C, c_max, N, fa) for reuse.
    T: temperature (0 means exact subgradient at argmax)
    """
    n = len(a)
    out_n = 2 * n - 1
    if C is None:
        C, N, fa = _fft_convolve_aa(a)
    c_max = float(C.max())

    if T <= 1e-15:
        # Exact subgradient: indicator of argmax
        peak_idx = int(np.argmax(C))
        w = np.zeros(out_n)
        w[peak_idx] = 1.0
    else:
        # Softmax weights
        shifted = (C - c_max) / T
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(out_n)
            w[int(np.argmax(C))] = 1.0
        else:
            w = e / s

    # Gradient: g = 2 * IFFT(A * W)[:n]
    if fa is None:
        fa = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    g = np.fft.irfft(2.0 * fa * W, N)[:n]

    return g, C, c_max, N, fa


def _mirror_descent_step(a, g, lr):
    """Multiplicative weights update: a *= exp(-lr*g), then normalize."""
    # For stability, center g
    gc = g - g.mean()
    a_new = a * np.exp(np.clip(-lr * gc, -50, 50))
    s = a_new.sum()
    if s < 1e-300:
        return a
    a_new = a_new / s
    return a_new


def _run_mirror_descent(a_init, deadline, min_time_remaining=200.0):
    """
    Main optimization loop: mirror descent with adaptive temperature.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Temperature schedule: geometric decay from T_max to T_min
    C0, _, _ = _fft_convolve_aa(a)
    c_max0 = float(C0.max())
    C_sorted = np.sort(C0)[::-1]
    gap = C_sorted[0] - C_sorted[min(50, len(C_sorted) - 1)]
    if gap < 1e-15:
        gap = c_max0 * 0.01
    T_max = max(gap * 5, c_max0 * 0.05)
    T_min = max(c_max0 * 0.0001, 1e-8)

    lr = 1.0 / (np.linalg.norm(a) + 1e-10)
    momentum = 0.5
    g_smooth = None
    T = T_max
    t_start = time.monotonic()
    stall_count = 0
    max_stall = 100
    total_iters = 0
    last_improve_time = t_start

    while time.monotonic() + min_time_remaining < deadline:
        # Compute gradient
        g, C, c_max, N, fa = _grad_conv_max(a, T, N=None, fa=None, C=None)

        # Score check
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            last_improve_time = time.monotonic()
            stall_count = 0
            momentum = min(0.95, momentum * 1.005)
        else:
            stall_count += 1

        # Adaptive temperature: decay when improving, increase when stalling
        if stall_count % 20 == 0:
            if stall_count > max_stall:
                # Restart with higher temperature
                T = min(T_max, T * 3.0)
                momentum = 0.3
                lr = lr * 0.5
                stall_count = 0
            else:
                T = max(T_min, T * 0.85)
                lr = max(lr * 0.9, 1e-6)

        # Normal temperature decay based on time
        elapsed_frac = (time.monotonic() - t_start) / max(1e-10, deadline - t_start - min_time_remaining)
        T_time = T_max * (T_min / T_max) ** (elapsed_frac ** 0.7)
        T = min(T, T_time)

        # Gradient smoothing (momentum)
        if g_smooth is None:
            g_smooth = g.copy()
        else:
            g_smooth = momentum * g_smooth + (1.0 - momentum) * g

        # Mirror descent step
        a_new = _mirror_descent_step(a, g_smooth, lr)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        # Line search: verify improvement
        C_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(C_new.max())
        if sc_new < sc - 1e-14:
            a = a_new
        else:
            # Halve learning rate
            lr *= 0.7
            # Try with smaller step
            for _ls in range(5):
                a_try = _mirror_descent_step(a, g_smooth, lr)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    break
                a_try = a_try / s_try
                C_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(C_try.max())
                if sc_try < sc - 1e-14:
                    a = a_try
                    break
                lr *= 0.5

        total_iters += 1
        if total_iters % 500 == 0:
            print(f"    MD iter {total_iters}: score={best_sc:.10f}, T={T:.2e}, lr={lr:.2e}, t={time.monotonic()-t_start:.1f}s")

        # Periodic exact subgradient step (every 100 iterations)
        if total_iters % 100 == 0 and T > T_min * 10:
            g_exact, C_e, c_max_e, _, fa_e = _grad_conv_max(a, 0.0)
            a_exact = _mirror_descent_step(a, g_exact, lr * 0.5)
            a_exact = np.maximum(a_exact, 0.0)
            s_e = a_exact.sum()
            if s_e > 1e-10:
                a_exact = a_exact / s_e
                sc_e = _score(a_exact)
                if sc_e < 2.0 * n * c_max:
                    a = a_exact
                    if sc_e < best_sc:
                        best_sc = sc_e
                        best_a = a.copy()

    # Final score
    final_sc = _score(best_a)
    if final_sc < best_sc:
        best_sc = final_sc
    return best_a, best_sc


def _exact_subgradient_polish(a_init, deadline, min_time_remaining=50.0):
    """
    Polish using exact subgradient (no smoothing) with backtracking.
    Focuses exclusively on reducing the single peak.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.05
    momentum = 0.7
    g_hist = None
    t_start = time.monotonic()
    iter_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        C = _full_convolve(a, a)
        c_max = float(C.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            lr = min(0.2, lr * 1.05)
        else:
            lr *= 0.85

        # Exact subgradient
        g, _, _, _, _ = _grad_conv_max(a, 0.0)

        # Momentum
        if g_hist is None:
            g_hist = g.copy()
        else:
            g_hist = momentum * g_hist + (1.0 - momentum) * g

        # Step
        a_new = _mirror_descent_step(a, g_hist, lr)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        # Check
        C_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(C_new.max())
        if sc_new < sc - 1e-13:
            a = a_new
        else:
            # Backtrack
            found = False
            for _bt in range(8):
                a_try = _mirror_descent_step(a, g_hist, lr)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    break
                a_try = a_try / s_try
                C_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(C_try.max())
                if sc_try < sc - 1e-13:
                    a = a_try
                    found = True
                    break
                lr *= 0.5
            if not found:
                # Reset momentum
                g_hist = g.copy()

        iter_count += 1
        if iter_count % 200 == 0:
            print(f"    Subgrad iter {iter_count}: score={best_sc:.10f}, lr={lr:.4f}, t={time.monotonic()-t_start:.1f}s")

        if lr < 1e-8:
            break

    return best_a, best_sc


def _solve_qp_reduced(a, n_active, target_frac, deadline):
    """
    Solve QP to reduce top peaks. Uses the structure to build a small dual problem.
    Returns (direction, status).
    """
    n = len(a)
    C = _full_convolve(a, a)
    c_max = float(C.max())
    target = c_max * target_frac

    # Select top peaks
    top_idx = np.argsort(C)[::-1][:n_active]

    # Build constraint matrix B (m x n): B[k, j] = 2*a[k-j] if valid
    # conv(a+d, a+d)[k] ≈ C[k] + 2*conv(a,d)[k]
    # conv(a,d)[k] = sum_i a[i]*d[k-i]
    # So constraint: C[k] + 2*sum_i a[i]*d[k-i] <= target
    # i.e., 2*sum_i a[i]*d[k-i] <= target - C[k]

    m = n_active
    B = np.zeros((m, n))
    rhs = np.zeros(m)
    for idx, k in enumerate(top_idx):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            if 0 <= j < n:
                B[idx, j] = 2.0 * a[i]
        rhs[idx] = target - C[k]

    if not HAS_CVXPY:
        return None, 'no_cvxpy'

    try:
        d_var = cp.Variable(n)
        constraints = [cp.sum(d_var) == 0, B @ d_var <= rhs]

        # Non-negativity: only for elements close to zero
        zero_tol = 1e-8 * a.max()
        neg_mask = a < zero_tol
        if neg_mask.any() and neg_mask.sum() < n // 2:
            constraints.append(d_var[neg_mask] >= -a[neg_mask])

        # Bound on d to keep it reasonable
        constraints.append(d_var <= 0.5 * a.max())
        constraints.append(d_var >= -0.5 * a.max())

        objective = cp.Minimize(cp.sum_squares(d_var))
        prob = cp.Problem(objective, constraints)

        for solver in ['ECOS', 'SCS', 'CLARABEL']:
            try:
                prob.solve(solver=solver)
                break
            except Exception:
                continue
        else:
            prob.solve()

        if prob.status not in ['optimal', 'optimal_inaccurate']:
            return None, f'status_{prob.status}'

        return d_var.value.copy(), 'ok'
    except Exception as e:
        return None, f'error: {str(e)[:50]}'


def _qp_phase(a_init, deadline, min_time_remaining=30.0):
    """QP-based peak reduction phase."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    n_active = max(5, int(n * 0.01))
    target_frac = 0.98
    consecutive_failures = 0

    while time.monotonic() + min_time_remaining < deadline:
        d, status = _solve_qp_reduced(a, n_active, target_frac,
                                      deadline - min_time_remaining)
        if d is None:
            consecutive_failures += 1
            if consecutive_failures > 5:
                n_active = min(n // 2, n_active * 2)
                target_frac = min(0.999, target_frac + 0.02)
                consecutive_failures = 0
                if target_frac > 0.998:
                    break
            continue

        # Line search
        C_cur = _full_convolve(a, a)
        sc_cur = 2.0 * n * float(C_cur.max())
        improved = False
        alpha = 1.0
        for _ in range(30):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            C_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(C_try.max())
            if sc_try < sc_cur - 1e-13:
                a = a_try
                sc_cur = sc_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            consecutive_failures = 0
            if sc_cur < best_sc - 1e-14:
                best_sc = sc_cur
                best_a = a.copy()
            # Tighten
            target_frac = max(0.90, target_frac * 0.99)
            n_active = min(n // 4, max(5, int(n_active * 1.2)))
        else:
            consecutive_failures += 1
            target_frac = min(0.999, target_frac + 0.005)
            n_active = max(3, int(n_active * 0.7))

        if time.monotonic() - (deadline - min_time_remaining) > 300:
            break

    return best_a, best_sc


def _active_set_gradient(a_init, deadline, min_time_remaining=30.0):
    """
    Active-set gradient method: focus on peaks within adaptive tolerance of max.
    Uses uniform weights on active peaks (not just the single max).
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.03
    tol_frac = 0.01
    t_start = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        C = _full_convolve(a, a)
        c_max = float(C.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            tol_frac = max(0.0001, tol_frac * 0.95)
            lr = min(0.15, lr * 1.02)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(0.0001, tol_frac * 0.7)
                lr *= 0.7

        # Active set: peaks within tol_frac of max
        tol = tol_frac * c_max
        active = C >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros_like(C, dtype=bool)
            active[int(np.argmax(C))] = True
            n_active = 1

        w = np.zeros_like(C)
        w[active] = 1.0 / n_active

        # Gradient via FFT
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w, N)
        g = np.fft.irfft(2.0 * fa * fw, N)[:n]

        # Center gradient (sum constraint)
        g = g - g.mean()

        # Step with line search
        improved = False
        for _ in range(40):
            a_new = _mirror_descent_step(a, g, lr)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                lr *= 0.5
                break
            a_new = a_new / s_new
            C_new = _full_convolve(a_new, a_new)
            sc_new = 2.0 * n * float(C_new.max())
            if sc_new < sc - 1e-13:
                a = a_new
                improved = True
                break
            lr *= 0.5

        if not improved and no_improve > 200:
            # Reset
            tolerance_reset = min(0.05, tol_frac * 5)
            tol_frac = tolerance_reset
            lr = 0.03 * np.random.default_rng(iter_count).uniform(0.5, 1.5)
            no_improve = 0

        iter_count += 1
        if iter_count % 300 == 0:
            print(f"    AS-grad iter {iter_count}: score={best_sc:.10f}, tol={tol_frac:.5f}, n_act={n_active}, t={time.monotonic()-t_start:.1f}s")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.5) ** 2) + 0.001
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _lbfgs_smoothing(a_init, deadline, min_time_remaining=30.0):
    """
    L-BFGS with log-sum-exp smoothing as a final polish.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    C = _full_convolve(a, a)
    c_max = float(C.max())
    if c_max < 1e-15:
        return a

    # Adaptive beta
    C_sorted = np.sort(C)[::-1]
    gap = C_sorted[0] - C_sorted[min(30, len(C_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap * 0.5, c_max * 0.001)

    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()

    def obj(z_in):
        a_in = _softmax_helper(z_in)
        C_in = _full_convolve(a_in, a_in)
        c_max_in = float(C_in.max())
        shifted = (C_in - c_max_in) / base_beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * c_max_in
        return 2.0 * n * (c_max_in + base_beta * np.log(s))

    def grad(z_in):
        a_in = _softmax_helper(z_in)
        C_in = _full_convolve(a_in, a_in)
        c_max_in = float(C_in.max())
        shifted = (C_in - c_max_in) / base_beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(C_in)
            w[int(np.argmax(C_in))] = 1.0
        else:
            w = e / s
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a_in, N)
        fw = np.fft.rfft(w, N)
        h = np.fft.irfft(fa * fw, N)[:n]
        W_val = float(np.dot(w, C_in))
        return 4.0 * n * a_in * (h - W_val)

    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 5:
        return a

    max_iter = min(5000, max(50, int(remaining / 0.02)))
    try:
        result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-15,
                                   'gtol': 1e-13, 'maxcor': 40})
        a_new = _softmax_helper(result.x)
        sc_new = _score(a_new)
        sc_cur = _score(a)
        if sc_new < sc_cur:
            return a_new
    except Exception:
        pass
    return a


def _softmax_helper(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining mirror descent, exact subgradient,
    QP refinement, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: Mirror descent (main workhorse) ===
    print(f"\n[P1] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 600)
    a_md, sc_md = _run_mirror_descent(best_seq, p1_deadline, min_time_remaining=550.0)
    if sc_md < best_score:
        best_seq = a_md.copy()
        best_score = sc_md
        print(f"  NEW BEST: {best_score:.10f} [mirror_descent] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  MD: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Exact subgradient polish ===
    print(f"\n[P2] Exact subgradient (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, time.monotonic() + 200)
    a_sg, sc_sg = _exact_subgradient_polish(best_seq, p2_deadline, min_time_remaining=350.0)
    if sc_sg < best_score:
        best_seq = a_sg.copy()
        best_score = sc_sg
        print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  SubGrad: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS smoothing ===
    print(f"\n[P3] L-BFGS smoothing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 250, time.monotonic() + 150)
        a_lb = _lbfgs_smoothing(best_seq, p3_deadline, min_time_remaining=230.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  L-BFGS: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Active-set gradient ===
    print(f"\n[P4] Active-set gradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 150, time.monotonic() + 120)
        a_as, sc_as = _active_set_gradient(best_seq, p4_deadline, min_time_remaining=130.0)
        if sc_as < best_score:
            best_seq = a_as.copy()
            best_score = sc_as
            print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: QP refinement ===
    print(f"\n[P5] QP refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100 and HAS_CVXPY:
        p5_deadline = min(deadline - 80, time.monotonic() + 80)
        a_qp, sc_qp = _qp_phase(best_seq, p5_deadline, min_time_remaining=70.0)
        if sc_qp < best_score:
            best_seq = a_qp.copy()
            best_score = sc_qp
            print(f"  NEW BEST: {best_score:.10f} [qp] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = []
    all_ns = [8192, 16384, 32768, 12288, 24576]
    for nn in all_ns:
        if nn != len(best_seq):
            n_candidates.append(nn)

    for n_try in n_candidates[:4]:  # Try at most 4 different n
        if time.monotonic() > deadline - 60:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        print(f"  n={n_try}: resampled score={sc_res:.10f}")

        if sc_res < best_score:
            best_seq = a_res.copy()
            best_score = sc_res
            print(f"  NEW BEST: {best_score:.10f} [n={n_try} resample] t={time.monotonic()-t_start:.1f}s")
            continue

        # Optimize at this n
        if time.monotonic() < deadline - 80:
            p_deadline = min(deadline - 50, time.monotonic() + 40)
            a_opt, sc_opt = _run_mirror_descent(a_res, p_deadline, min_time_remaining=40.0)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try} MD] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final mirror descent + subgradient refinement ===
    print(f"\n[P7] Final refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        p7_deadline = min(deadline - 5, time.monotonic() + 30)
        a_fin, sc_fin = _run_mirror_descent(best_seq, p7_deadline, min_time_remaining=20.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_MD] t={time.monotonic()-t_start:.1f}s")
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _exact_subgradient_polish(best_seq, deadline - 2, min_time_remaining=10.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_subgrad] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
