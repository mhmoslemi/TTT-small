"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) L-BFGS with log-sum-exp smoothing, (2) active-set projected
gradient on the simplex, (3) multi-start with diverse profiles,
(4) fine polish with adaptive step sizes.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        est_time_per_eval = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_eval * 15))))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _active_set_gradient(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Parameters
    lr = 0.01
    tol_frac = 0.01
    min_tol_frac = 0.00001
    no_improve = 0
    max_stall = 200
    stall_count = 0
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[idx] = True

        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_gradient(a_init, deadline, min_time_remaining=30.0,
                         tol_start=0.01, tol_end=0.00005, lr_init=0.02):
    """Fast active-set projected gradient using FFT-based O(n log n) gradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_init
    tol_frac = tol_start
    no_improve = 0
    max_stall = 300
    stall_count = 0
    a_rev = a[::-1]

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active_mask = c >= (c_max - tol)
        n_active = int(active_mask.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[idx] = True
            n_active = 1

        # Weighted gradient: g_j = sum_k w_k * (a[k-j] + a[j])
        # = a_j * W + h_j where h_j = sum_k w_k * a[k-j]
        # h = (a_rev * v)[n-1 : 2n-1] where v[k] = w_k
        w = np.zeros(2 * n - 1)
        w[active_mask] = 1.0 / n_active

        v = w
        h = _full_convolve(a_rev, v)[n - 1:2 * n - 1]
        grad = a + h

        # Remove mean for simplex tangency
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        # Adaptive line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                a_rev = a[::-1]
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 40 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.85
            if no_improve >= max_stall:
                a = best_a.copy()
                a_rev = a[::-1]
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.4
                no_improve = 0
                stall_count += 1
                if stall_count > 12:
                    break
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.97)
            lr = min(0.08, lr * 1.015)

    return best_a, _score(best_a)


def _coordinate_descent(a_init, deadline, min_time_remaining=20.0):
    """Coordinate descent: optimize individual coordinates to reduce peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    iteration = 0
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Pick top peak
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Find coordinates that contribute most to this peak
        # c[k_star] = sum_i a[i]*a[k_star-i]
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        contributions = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_star - i
            contributions[i] = a[j]
            contributions[j] += a[i]

        # Try perturbing the top-contributing coordinate
        n_try = max(5, int(n * 0.002))
        top_idx = np.argsort(contributions)[::-1][:n_try]

        improved_this_round = False
        for idx in top_idx:
            if time.monotonic() + min_time_remaining > deadline:
                break
            # Try decreasing a[idx]
            old_val = a[idx]
            for frac in [0.001, 0.0005, 0.0001, 0.00005]:
                a[idx] = old_val * (1 - frac)
                s_new = a.sum()
                if s_new < 1e-10:
                    a[idx] = old_val
                    break
                a = a / s_new
                c_new = _full_convolve(a, a)
                sc_new = 2.0 * n * float(c_new.max())
                if sc_new < current_sc - 1e-14:
                    improved_this_round = True
                    current_sc = sc_new
                    break
                a[idx] = old_val  # restore
            else:
                a[idx] = old_val  # restore if no improvement

        if not improved_this_round:
            # Try random perturbation
            idx = rng.integers(n)
            old_val = a[idx]
            delta = old_val * rng.uniform(-0.002, 0.002)
            a[idx] = max(0, old_val + delta)
            s_new = a.sum()
            if s_new > 1e-10:
                a = a / s_new
                c_new = _full_convolve(a, a)
                sc_new = 2.0 * n * float(c_new.max())
                if sc_new < current_sc - 1e-14:
                    current_sc = sc_new
                else:
                    a[idx] = old_val  # revert
            else:
                a[idx] = old_val

        if iteration % 500 == 0:
            print(f"    CD iter {iteration}: sc={best_sc:.10f}")

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "osc1":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "osc2":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 13.0 * t / n) + 0.1 * np.sin(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "osc3":
        t = np.arange(n)
        alpha = 0.6180339887
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * alpha * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.3) ** 2) + np.exp(-50.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _smoothed_peak_method(a_init, deadline, min_time_remaining=30.0):
    """Minimize a smoothed version of max using iterative reweighting."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    a_rev = a[::-1]

    # Start with broad smoothing, narrow down
    beta_frac = 0.05
    lr = 0.03
    no_improve = 0
    stall = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Softmax weights
        shifted = (c - c_max) / (beta_frac * c_max)
        e = np.exp(np.clip(shifted, -700, 0))
        s_w = e.sum()
        if s_w < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_w

        # FFT-based gradient
        h = _full_convolve(a_rev, w)[n - 1:2 * n - 1]
        grad = a + h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                a_rev = a[::-1]
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                beta_frac = max(0.001, beta_frac * 0.7)
                lr *= 0.85
            if no_improve >= 200:
                a = best_a.copy()
                a_rev = a[::-1]
                beta_frac = max(0.001, beta_frac * 0.5)
                lr = 0.01
                no_improve = 0
                stall += 1
                if stall > 10:
                    break
        else:
            if beta_frac > 0.001:
                beta_frac *= 0.985
            lr = min(0.06, lr * 1.01)

    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, active-set gradient, smoothed peak
    method, coordinate descent, and multi-start with diverse profiles.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    n = 16384

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size == n:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        parent = _make_profile(n, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n}, score={best_score:.10f}")

    # === Phase 1: L-BFGS on parent ===
    print(f"\n[P1] L-BFGS on parent (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = min(deadline - 900, t_start + 500)
    a1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=60.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (score={sc1:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set gradient ===
    print(f"\n[P2] Active-set gradient (t={time.monotonic()-t_start:.1f}s)")
    phase2_deadline = min(deadline - 600, time.monotonic() + 450)
    a2, sc2 = _active_set_gradient(best_seq, phase2_deadline, min_time_remaining=100.0,
                                   tol_start=0.01, tol_end=0.0001, lr_init=0.015)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (score={sc2:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Smoothed peak method ===
    print(f"\n[P3] Smoothed peak method (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        phase3_deadline = min(deadline - 400, time.monotonic() + 300)
        a3, sc3 = _smoothed_peak_method(best_seq, phase3_deadline, min_time_remaining=100.0)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [smoothed] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (score={sc3:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-start with diverse profiles ===
    print(f"\n[P4] Multi-start exploration (t={time.monotonic()-t_start:.1f}s)")
    profiles = ["osc1", "osc3", "tapered2", "osc2"]
    per_profile_time = 180.0
    for prof in profiles:
        if time.monotonic() > deadline - 200:
            break
        a_init = _make_profile(n, prof)
        sc_init = _score(a_init)
        print(f"  Profile '{prof}': init score={sc_init:.10f}")

        # Quick L-BFGS
        sub_deadline = min(deadline - 150, time.monotonic() + per_profile_time * 0.6)
        a_opt = _run_lbfgs(a_init, sub_deadline, min_time_remaining=30.0)
        sc_opt = _score(a_opt)

        # Active-set polish
        if time.monotonic() < deadline - 100:
            sub2_deadline = min(deadline - 100, time.monotonic() + per_profile_time * 0.4)
            a_opt, sc_opt = _active_set_gradient(a_opt, sub2_deadline,
                                                 min_time_remaining=60.0,
                                                 tol_start=0.005, tol_end=0.0001,
                                                 lr_init=0.01)

        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [multi_start:{prof}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  '{prof}': best={sc_opt:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: L-BFGS again on best (with tighter beta) ===
    print(f"\n[P5] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        phase5_deadline = min(deadline - 150, time.monotonic() + 250)
        a5 = _run_lbfgs(best_seq, phase5_deadline, min_time_remaining=100.0)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Active-set gradient with tight tolerance ===
    print(f"\n[P6] Tight active-set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        phase6_deadline = min(deadline - 100, time.monotonic() + 200)
        a6, sc6 = _active_set_gradient(best_seq, phase6_deadline, min_time_remaining=60.0,
                                       tol_start=0.003, tol_end=0.00002, lr_init=0.008)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [tight_active] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Coordinate descent ===
    print(f"\n[P7] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        phase7_deadline = min(deadline - 50, time.monotonic() + 120)
        a7, sc7 = _coordinate_descent(best_seq, phase7_deadline, min_time_remaining=30.0)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [coord_desc] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Very tight active set
        final_deadline = min(deadline - 3, time.monotonic() + 80)
        a8, sc8 = _active_set_gradient(best_seq, final_deadline, min_time_remaining=5.0,
                                       tol_start=0.001, tol_end=0.00001, lr_init=0.004)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  NEW BEST: {best_score:.10f} [final_active] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            n_seq = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            c_sorted = np.sort(c_best)[::-1]
            gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
            beta_fin = max(mb * 0.0001, gap * 0.5, 1e-10)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n_seq)
            max_iter = min(3000, max(50, int(remaining / 0.01)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 80})
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin.copy()
                    best_score = sc_fin
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
