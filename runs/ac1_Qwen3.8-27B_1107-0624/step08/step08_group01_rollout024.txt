    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cosh_init(n, gamma):
    """Initialize with cosh(gamma * (2i/(n-1) - 1)) shape - high at edges, low in middle."""
    if n == 1:
        return np.ones(1)
    t = 2.0 * np.arange(n) / (n - 1) - 1.0
    a = np.cosh(gamma * t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _cos_taper_init(n, beta):
    """Initialize with (cos(pi*i/(n-1)))^beta - zero at edges."""
    if n == 1:
        return np.ones(1)
    t = np.pi * np.arange(n) / (n - 1)
    a = np.cos(t) ** beta
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _inv_cos_taper_init(n, beta):
    """Initialize with (1 + beta*(1-cos(pi*i/(n-1))))^(-1) - high at edges."""
    if n == 1:
        return np.ones(1)
    t = np.pi * np.arange(n) / (n - 1)
    a = 1.0 / (1.0 + beta * (1.0 - np.cos(t)))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _spectral_flat_init(n, num_iters=50):
    """Try to flatten the power spectrum iteratively."""
    a = np.ones(n) / n
    for _ in range(num_iters):
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        # Try to flatten P: raise the bottom, lower the top
        P_flat = P ** 0.5
        P_flat = P_flat / P_flat.sum()
        # Inverse: want |A|^2 proportional to P_flat
        # A_new = A * sqrt(P_flat / P)
        mask = P > 1e-30
        A_new = A.copy()
        A_new[mask] = A[mask] * np.sqrt(P_flat[mask] / P[mask])
        # Make real and non-negative
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        if np.allclose(a, a_new, atol=1e-12):
            break
        a = a_new
    return a


def _gradient_multipeak(a, c, top_k=5, frac=0.01):
    """Compute multi-peak subgradient. Returns gradient vector."""
    n = len(a)
    c_max = float(c.max())
    threshold = c_max * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    k_act = min(top_k, len(active))
    top_indices = active[np.argsort(c[active])[::-1][:k_act]]
    
    grad = np.zeros(n)
    for k in top_indices:
        k_lo = max(0, k - n + 1)
        k_hi = min(k, n - 1)
        if k_lo <= k_hi:
            grad[k_lo:k_hi+1] += 2.0 * a[k_hi:k_lo-1:-1] if k_lo == 0 else a[k_hi:k_lo-1:-1]
    return grad


def _gradient_multipeak_v2(a, c, top_k=5, frac=0.02):
    """More careful multi-peak gradient computation."""
    n = len(a)
    c_max = float(c.max())
    threshold = c_max * (1.0 - frac)
    active = c >= threshold
    n_active = int(active.sum())
    if n_active == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_active = 1
    
    # Use top_k active peaks with equal weight
    if n_active > top_k:
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.ones(top_k) / top_k
    else:
        top_idx = np.where(active)[0]
        weights = np.ones(len(top_idx)) / len(top_idx)
    
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        # (grad c_k)_j = 2 * a[k-j] for j in [max(0,k-n+1), min(k,n-1)]
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            # a[k-j] for j in [j_lo, j_hi] = a[k-j_hi : k-j_lo : -1]
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += w * 2.0 * a[idx]
    return grad


def _project_normalize(a):
    """Project onto a_i >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _proj_grad_descent(a_init, deadline, top_k=5, frac_start=0.05, frac_end=0.001,
                       lr_start=0.05, max_iters=5000):
    """Projected gradient descent with multi-peak gradient and backtracking."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_normalize(a)
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    
    lr = lr_start
    frac = frac_start
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max
        
        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Compute gradient
        grad = _gradient_multipeak_v2(a, c, top_k=top_k, frac=frac)
        
        # Normalize gradient (remove mean to stay on simplex)
        grad = grad - grad.mean()
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad
            a_try = _project_normalize(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            
            if sc_try < cur_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
            no_improve += 5
        else:
            lr = min(lr_start, lr * 1.05)
        
        # Anneal frac (focus on fewer peaks over time)
        frac = max(frac_end, frac * 0.995)
    
    return best_a, best_sc


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Gradient of 2n * sum_k w_k * c_k wrt a
    # = 2n * sum_k w_k * grad(c_k)
    # grad(c_k)_j = 2 * a[k-j] for valid j
    grad_c = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            grad_c[j_lo:j_hi+1] += w[k] * 2.0 * a[idx]
    
    # Now we need d/dz of 2n * sum w_k c_k where a = softmax(z)
    # da_i/dz_j = a_i (delta_ij - a_j)
    # dF/dz_j = sum_i (dF/da_i)(da_i/dz_j) = (sum_i dF/da_i * a_i) - a_j * (sum_i dF/da_i * a_i)
    # = a_j * (gradF . a) ... wait
    # dF/dz_j = sum_i gradF_i * a_j (delta_ij - a_i) = gradF_j * a_j - a_j * sum_i gradF_i * a_i
    # = a_j * (gradF_j - sum_i gradF_i * a_i)
    
    F = 2.0 * n * grad_c  # dF/da
    dot = float(np.dot(F, a))
    grad_z = a * (F - dot)
    
    return grad_z


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, betas=None, min_remaining=15.0):
    """L-BFGS with log-sum-exp annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    if betas is None:
        betas = base_beta * np.array(
            [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.007,
             0.003, 0.0015, 0.0007, 0.0003, 0.0001, 0.00005,
             0.00002, 0.00001, 0.000005, 0.000002, 0.000001]
        )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_iter_time * 20))))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_refine(a_init, deadline, n_iters=100):
    """Iteratively refine in the spectral domain to flatten the power spectrum."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        
        # Compute the linear convolution values
        c = _full_convolve(a, a)
        
        # In spectral domain, c values are inverse FFT of P (approximately)
        # We want to reduce the peak of c, which means reducing the peak 
        # of the inverse FFT of P
        
        # Strategy: attenuate frequencies that contribute most to the peak
        c_max = float(c.max())
        k_peak = int(np.argmax(c))
        
        # The peak at k_peak is dominated by low frequencies (especially DC)
        # Attenuate the contribution of frequencies that align with the peak
        # Use a soft threshold on P
        p_max = float(P.max())
        if p_max < 1e-30:
            break
        
        # Softly reduce the dynamic range of P
        alpha = 0.1 + 0.1 * (it / n_iters)
        P_new = P ** (1 - alpha)
        
        # Reconstruct A with modified magnitude, keep phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a smaller step
            alpha2 = alpha * 0.3
            P_new2 = P ** (1 - alpha2)
            A_new2 = np.sqrt(P_new2) * np.exp(1j * phase)
            a_new2 = np.fft.irfft(A_new2, N)[:n]
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 < 1e-15:
                break
            a_new2 = a_new2 / s2
            sc_new2 = _score(a_new2)
            if sc_new2 < best_sc - 1e-14:
                best_sc = sc_new2
                best_a = a_new2.copy()
                a = a_new2
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fine_polish(a_init, deadline, top_k=3, tol=0.001):
    """Very fine active-set polishing for the final phase."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_normalize(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Focus on the single peak
    k_star = int(np.argmax(c))
    
    lr = 0.005
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max
        
        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        
        # Find peak contributors
        k_star = int(np.argmax(c))
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        
        if j_lo > j_hi:
            break
        
        # Gradient at the peak
        idx = k_star - np.arange(j_lo, j_hi + 1)
        # Peak value: sum of a[j] * a[k_star - j]
        peak_contrib = a[j_lo:j_hi+1] * a[idx]
        total = peak_contrib.sum()
        
        if total < 1e-15:
            break
        
        # Compute effective gradient: reduce peak by moving mass from high-contrib to low-contrib
        # The "blame" for each a_i is a[k_star - i]
        blame = np.zeros(n)
        blame[j_lo:j_hi+1] = a[idx]
        
        # Also consider: if we increase a_i, the peak changes by a[k_star-i]
        # If we decrease a_i, the peak changes by -a[k_star-i]
        # But changing a_i also changes other c_k values
        
        # Simple approach: projected gradient with single peak
        grad = np.zeros(n)
        grad[j_lo:j_hi+1] = blame
        grad = grad - grad.mean()
        
        # Line search
        alpha = lr
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = _project_normalize(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < cur_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            lr *= 0.7
            if no_improve > 200 or lr < 1e-12:
                break
        else:
            lr = min(0.005, lr * 1.02)
            no_improve = 0
    
    return best_a, best_sc


def proposal_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
    except Exception:
        pass
    
    best_a = None
    best_score = np.inf
    
    # === Phase 0: Generate initial candidates from multiple families ===
    print(f"[P0] Generating initial candidates (t={time.monotonic()-t_start:.1f}s)")
    
    candidates = []
    n_base = len(parent) if parent is not None else 12288
    
    # Cosh family with various gammas
    for gamma in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 8.0, 12.0]:
        candidates.append(_cosh_init(n_base, gamma))
    
    # Inverse cos taper
    for beta in [0.5, 1.0, 2.0, 5.0, 10.0]:
        candidates.append(_inv_cos_taper_init(n_base, beta))
    
    # Spectral flat
    candidates.append(_spectral_flat_init(n_base, 30))
    
    # Parent resample
    if parent is not None:
        candidates.append(parent.copy())
    
    # Evaluate all candidates
    for i, cand in enumerate(candidates):
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
        if i % 5 == 0:
            print(f"  Candidate {i}: score={sc:.10f}")
    
    print(f"  Best initial: score={best_score:.10f}")
    
    # === Phase 1: Projected gradient descent on best initial ===
    print(f"\n[P1] Projected gradient descent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200.0, t_start + 300.0)
    a1, sc1 = _proj_grad_descent(best_a, p1_deadline, top_k=5, frac_start=0.05)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [pgd] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: L-BFGS with log-sum-exp ===
    print(f"\n[P2] L-BFGS log-sum-exp (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 800.0, t_start + 600.0)
    a2 = _run_lbfgs(best_a, p2_deadline)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Spectral refinement ===
    print(f"\n[P3] Spectral refinement (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 500.0, t_start + 800.0)
    a3, sc3 = _spectral_refine(best_a, p3_deadline, n_iters=50)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: L-BFGS again after spectral ===
    if time.monotonic() < deadline - 400.0:
        print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
        p4_deadline = min(deadline - 400.0, t_start + 1000.0)
        a4 = _run_lbfgs(best_a, p4_deadline)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_a = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_targets = [n_best]
    for n_try in [8192, 12288, 16384, 24576, 6144, 32768]:
        if n_try not in n_targets and n_try != n_best:
            n_targets.append(n_try)
    
    for n_try in sorted(n_targets, key=lambda x: abs(x - n_best)):
        if time.monotonic() > deadline - 200.0:
            break
        if n_try == n_best:
            continue
        
        # Try both resampled best and cosh init for this n
        inits = [_resample(best_a, n_try), _cosh_init(n_try, 2.0), _cosh_init(n_try, 4.0)]
        
        for init in inits:
            if time.monotonic() > deadline - 150.0:
                break
            sc_init = _score(init)
            if sc_init >= best_score * 1.1:
                continue  # Skip if too far from best
            
            # Quick optimize
            sub_deadline = min(deadline - 100.0, time.monotonic() + 80.0)
            a_opt = _run_lbfgs(init, sub_deadline, min_remaining=10.0)
            sc_opt = _score(a_opt)
            
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                n_best = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100.0:
        n_cur = len(best_a)
        restarts = [
            ("pert_small", _perturb(best_a, 0.01, rng)),
            ("pert_med", _perturb(best_a, 0.03, rng)),
            ("cosh_var", _cosh_init(n_cur, 1.0)),
            ("cosh_var2", _cosh_init(n_cur, 5.0)),
            ("inv_cos", _inv_cos_taper_init(n_cur, 3.0)),
        ]
        
        for label, a_r in restarts:
            if time.monotonic() > deadline - 50.0:
                break
            sub_deadline = min(deadline - 40.0, time.monotonic() + 60.0)
            a_opt = _run_lbfgs(a_r, sub_deadline, min_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Fine polishing ===
    print(f"\n[P7] Fine polishing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        a_fin, sc_fin = _fine_polish(best_a, deadline - 3.0, top_k=2)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [fine_polish] t={time.monotonic()-t_start:.1f}s")
    
    # Final tiny L-BFGS
    remaining = deadline - 2.0 - time.monotonic()
    if remaining > 5.0:
        n = len(best_a)
        c_best = _full_convolve(best_a, best_a)
        mb = float(c_best.max())
        beta_fin = mb * 0.000005
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()
        prob = _LseProblem(beta_fin, n)
        max_iter = min(500, max(10, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 40})
            a_fin2 = _softmax(result.x)
            a_fin2 = np.maximum(a_fin2, 0.0)
            s_fin2 = a_fin2.sum()
            if s_fin2 > 1e-10:
                a_fin2 = a_fin2 / s_fin2
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_a = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass
    
    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _cosh_init(256, 1.0)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
