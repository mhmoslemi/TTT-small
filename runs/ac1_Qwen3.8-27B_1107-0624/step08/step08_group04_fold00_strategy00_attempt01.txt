while time.time() < deadline:
    a = manage_time_and_search()
    if a is None or not np.isfinite(score):
        a = fallback_sequence()
    else:
        a = aggressive_optimize(a, deadline - 3)
        a = flatten_polish(a)
        if compare_score(a) < overall_best:
            overall_best = a.copy()
            overall_score = compare_score(a)
            
        # Try to resample to bigger length
        a_large = resample(overall_best, deadline - 30)
        a_final = aggressive_optimize(a_large, deadline - 300)
        if np.isfinite(evaluate_sequence(a_final)) and evaluate_sequence(a_final) < overall_score * 0.997:
            overall_best = a_final.copy()
            overall_score = evaluate_sequence(overall_best)
            break
            
        # Then try to find better with different kernel
        a_tuned = apply_kernel_tuning(overall_best)
        score_tuned = evaluate_sequence(a_tuned)
        if score_tuned < overall_score:
            overall_best = a_tuned.copy()
            overall_score = score_tuned

    print("Best sequence score: %.8f" % overall_score)

final_sequence = manage_time_budget_for_return(overall_best, deadline)[0]
print("Final best score: %.8f" % evaluate_sequence(final_sequence))
