def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _self_convolve(a):
    """Efficient self-convolution using a single FFT."""
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _corr(a, A_vec, n):
    """Compute correlation of a with A_vec (for gradient computation)."""
    flip_a = a[::-1]

    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()

def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1, n / 8192.0)
        max_iter = min(8000, max(20, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 3 == 0:
                print(f"    L-BFGS stage {i+1}/{n_stages}: beta={beta:.3e}, score={current_sc:.10f}, "
                      f"iters={result.nit}, time={time.monotonic() - (deadline - remaining - min_time_remaining):.1f}s")
        except Exception as e:
            print(f"    L-BFGS stage {i+1} error: {e}")

    return current_a, current_sc


def _peak_shaving(a_init, deadline, n_attempts=50):
    """Targeted peak shaving: directly reduce contributions to top peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _self_convolve(best_a)
        c_max = float(c.max())

        # Find top peaks
        n_top = max(3, min(len(c), int(0.02 * len(c))))
        top_k = np.argsort(c)[::-1][:n_top]

        # For each top peak, find the dominant contributing pair and reduce it
        a_new = best_a.copy()
        reduction = 0.998 - 0.003 * min(attempt, 10)  # gradually stronger

        for k_pos in top_k:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(int(k_pos), n - 1)
            if i_lo > i_hi:
                continue

            # Find pair with largest product
            best_prod = 0
            best_i = -1
            best_j = -1
            for i in range(i_lo, i_hi + 1):
                j = int(k_pos) - i
                if 0 <= j < n:
                    prod = a_new[i] * a_new[j]
                    if prod > best_prod:
                        best_prod = prod
                        best_i = i
                        best_j = j

            if best_i >= 0 and best_prod > 0:
                # Reduce the larger element
                if a_new[best_i] >= a_new[best_j]:
                    a_new[best_i] *= reduction
                else:
                    a_new[best_j] *= reduction

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try weaker reduction
            a_weak = best_a.copy()
            for k_pos in top_k[:5]:
                i_lo = max(0, int(k_pos) - n + 1)
                i_hi = min(int(k_pos), n - 1)
                if i_lo > i_hi:
                    continue
                best_prod = 0
                best_i = -1
                best_j = -1
                for i in range(i_lo, i_hi + 1):
                    j = int(k_pos) - i
                    if 0 <= j < n:
                        prod = a_weak[i] * a_weak[j]
                        if prod > best_prod:
                            best_prod = prod
                            best_i = i
                            best_j = j
                if best_i >= 0:
                    if a_weak[best_i] >= a_weak[best_j]:
                        a_weak[best_i] *= 0.9999
                    else:
                        a_weak[best_j] *= 0.9999
            a_weak = np.maximum(a_weak, 0)
            s_weak = a_weak.sum()
            if s_weak > 1e-15:
                a_weak = a_weak / s_weak
                sc_weak = _score(a_weak)
                if sc_weak < best_sc - 1e-13:
                    best_sc = sc_weak
                    best_a = a_weak.copy()
                else:
                    break

    return best_a, best_sc


def _gradient_descent_subgrad(a_init, deadline, n_steps=5000):
    """Subgradient method with averaging for the max constraint."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    avg_a = np.zeros_like(a)
    step_base = 0.1

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _self_convolve(a)
        c_max = float(c.max())

        # Use soft active set
        tol = max(0.005, 0.02 * (1 - step / n_steps)) * c_max
        active = c >= (c_max - tol)
        m = max(1, int(active.sum()))
        w = np.zeros_like(c)
        w[active] = 1.0 / m

        grad = 2.0 * _corr(a, w, n)
        g_t = grad - np.mean(grad)

        # Adaptive step size
        gamma = step_base / (1.0 + 0.05 * step)
        a_new = a - gamma * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        # Averaging
        avg_a += a_new
        avg_candidate = avg_a / (step + 1)
        sc_avg = _score(avg_candidate)
        if sc_avg < best_sc - 1e-14:
            best_sc = sc_avg
            best_a = avg_candidate.copy()

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = max(1, n // 4)
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = max(1, n // 6)
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _alternating_refine(a_init, deadline, min_time_remaining):
    """Alternate between L-BFGS and active-set Newton for refinement."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    rounds = 4
    round_time = (deadline - min_time_remaining - time.monotonic()) / max(1, rounds)

    for r in range(rounds):
        target_time = time.monotonic() + round_time * 0.6
        if time.monotonic() > deadline - min_time_remaining:
            break

        # L-BFGS stage
        c_cur = _self_convolve(current_a)
        max_c = float(c_cur.max())
        c_sorted = np.sort(c_cur)[::-1]
        idx = min(10, len(c_sorted) - 1)
        gap = max(c_sorted[0] - c_sorted[idx], max_c * 0.005)
        beta = gap * (0.01 ** (r + 1))

        z = np.log(np.maximum(current_a, 1e-30))
        z = z - z.mean()
        max_iter = min(3000, max(50, int(round_time * 0.5 / 0.01)))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
        except Exception:
            pass

        # Active-set Newton stage
        target_time2 = time.monotonic() + round_time * 0.4
        if time.monotonic() < deadline - min_time_remaining:
            a_as, sc_as = _active_set_newton(current_a, min(target_time2, deadline - min_time_remaining),
                                             max_iter=200,
                                             tol_list=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
            if sc_as < current_sc - 1e-15:
                current_a = a_as
                current_sc = sc_as

        print(f"    Round {r+1}/{rounds}: score={current_sc:.10f}, time={time.monotonic() - (deadline - (time.monotonic() - round_time * (r+1))):.1f}s")

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp,
    active-set Newton, peak shaving, and subgradient methods.
    Uses alternating refinement and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    a_main, sc_main = _run_lbfgs_stages(parent, deadline, min_time_remaining=600, n_stages=14)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_dl = min(deadline - 500, time.monotonic() + 200)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=400,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 400:
            as_dl2 = min(deadline - 400, time.monotonic() + 120)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=300,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: Peak shaving
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        ps_dl = min(deadline - 350, time.monotonic() + 60)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_dl, n_attempts=50)
        update_best(a_ps, sc_ps, "peak_shave")

        # L-BFGS after peak shaving
        if sc_ps < best_score + 0.001 and time.monotonic() < deadline - 250:
            a_ps2, sc_ps2 = _run_lbfgs_stages(a_ps, deadline, min_time_remaining=200, n_stages=8)
            update_best(a_ps2, sc_ps2, "lbfgs_post_shave")

    # Phase 4: Alternating refinement
    print(f"\n[Phase 4] Alternating refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_alt, sc_alt = _alternating_refine(best_seq, deadline, min_time_remaining=150)
        update_best(a_alt, sc_alt, "alternating")

    # Phase 5: Subgradient with averaging
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        sg_dl = min(deadline - 100, time.monotonic() + 60)
        a_sg, sc_sg = _gradient_descent_subgrad(best_seq, sg_dl, n_steps=10000)
        update_best(a_sg, sc_sg, "subgrad")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        if n_try <= 32768:
            a_opt, sc_opt = _run_lbfgs_stages(a_init, deadline, min_time_remaining=60, n_stages=10)
            update_best(a_opt, sc_opt, f"n_{n_try}")

            if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
                a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 30, time.monotonic() + 40),
                                                     max_iter=150, tol_list=[0.002, 0.001, 0.0005, 0.0002])
                update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 7: Random restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts = [(0.01, "small"), (0.03, "med"), (0.08, "large")]
        for sigma, label in restarts:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _run_lbfgs_stages(p_start, deadline, min_time_remaining=15, n_stages=6)
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 8: Final polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        as_final = min(deadline - 3, time.monotonic() + 50)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=300,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_newton")

        # Peak shave
        if time.monotonic() < deadline - 3:
            a_ps3, sc_ps3 = _peak_shaving(best_seq, min(deadline - 2, time.monotonic() + 20), n_attempts=30)
            update_best(a_ps3, sc_ps3, "final_shave")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _self_convolve(best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 3 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 50, 'maxls': 80}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_lbfgs")
                except Exception:
                    pass

        # One more Newton after tiny L-BFGS
        if time.monotonic() < deadline - 2 and best_seq is not None:
            a_fin4, sc_fin4 = _active_set_newton(best_seq, min(deadline - 1, time.monotonic() + 10),
                                                 max_iter=100, tol_list=[0.0001, 0.00005, 0.00002, 0.00001])
            update_best(a_fin4, sc_fin4, "final_newton2")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
