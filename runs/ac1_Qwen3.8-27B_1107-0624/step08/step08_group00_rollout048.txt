The key insight: `c[0] = sum(a^2)` is minimized when `a` is uniform (by Cauchy-Schwarz), giving `c[0] = 1/n`. But we can make `c[n-1]` smaller than `1/n` by making the sequence "anti-symmetric" in some sense.

Wait, `c[n-1] = sum_{i=0}^{n-1} a[i]*a[n-1-i]`. If `a[i] = a[n-1-i]` (symmetric), then `c[n-1] = sum a[i]^2_{sym} = sum(a^2)` for the symmetric part. If `a` is anti-symmetric (impossible for non-negative), then `c[n-1] = 0`.

For non-negative `a`, the minimum of `c[n-1]` is 0 (achieved when `a[i]*a[n-1-i] = 0` for all `i`, e.g., `a` is supported on `{0, ..., n/2}`). But then `c[0] = sum(a^2)` would be large (since the mass is concentrated).

So there's a trade-off between `c[0]` and `c[n-1]` (and the other intermediate lags).

**Key algorithmic idea**: The problem is to balance all the `c[k]` values. The "bottleneck" is typically the largest `c[k]`. A good approach is:

1. Identify which `c[k]` are the largest
2. Compute the gradient direction that reduces the largest `c[k]` while keeping others from increasing too much
3. Take a step in that direction

This is essentially what the active-set method does, but I want to make it more efficient.

**Efficient active-set method:**

Given the current `a` and the set of "active" lags (where `c[k]` is close to the max), we want to find a perturbation `d` such that:
- `sum(d) = 0` (to maintain normalization)
- `a + d >= 0`
- The active `c[k]` values are reduced

The first-order condition for `c[k]` is:
`c[k](a+d) ≈ c[k](a) + 2 * sum_j a[k-j] * d[j]` (linearized)

So we want `sum_j a[k-j] * d[j] < 0` for all active `k`.

This is a linear feasibility problem: find `d` such that `A @ d <= b` where `A[k,j] = a[k-j]` (for active `k`) and `b[k] = -(c[k] - target)`.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _active_set_direct(a_init, deadline, min_remain=30.0):
    """Direct active-set peak reduction: solve linear system to cancel top peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    iteration = 0
    stall = 0
    
    while time.monotonic() + min_remain < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            break
        
        # Select active set: top peaks
        n_active = max(3, min(20, n // 100))
        active_idx = np.argsort(c)[::-1][:n_active]
        
        # Build linear system: for each active k, sum_j a[k-j]*d[j] = -target_reduction
        # We want to reduce each active c[k] by a fraction
        reduction = c_max * 0.01
        target = c_max - reduction
        
        # B[k, j] = a[k-j] for valid j (this gives the linearized change in c[k])
        B = np.zeros((n_active, n))
        rhs = np.zeros(n_active)
        
        for ii, k in enumerate(active_idx):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    B[ii, j] = a[idx] + a[idx]  # factor of 2 from chain rule
            rhs[ii] = target - c[k]
        
        # Solve: minimize ||d||^2 s.t. B@d = rhs, sum(d) = 0
        # Using Lagrange: d = (B^T B + lambda * 1*1^T)^{-1} B^T rhs
        # For efficiency, use least squares with the sum constraint
        
        # Add sum(d)=0 as an extra constraint
        B_ext = np.vstack([B, np.ones(n_active) * 0.0 + np.eye(1, n)[0] * 0.0])
        # Actually, let's use a simpler approach: solve B@d = rhs in least squares,
        # then project to sum(d)=0
        
        # Least squares solution
        try:
            B_pinv = np.linalg.pinv(B)
            d = B_pinv @ rhs
        except np.linalg.LinAlgError:
            d = np.zeros(n)
        
        # Project to sum(d) = 0
        d = d - np.mean(d)
        
        # Ensure a + d >= 0: clip and re-project
        alpha_max = 1.0
        min_ratio = np.inf
        neg_mask = d < 0
        if neg_mask.any():
            min_ratio = np.min(a[neg_mask] / (-d[neg_mask]))
            alpha_max = max(0.0, min_ratio * 0.99)
        
        # Line search
        improved = False
        for alpha in [alpha_max, alpha_max * 0.5, alpha_max * 0.25, alpha_max * 0.1]:
            if alpha <= 0:
                continue
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
        
        if not improved:
            # Try reducing active set size
            n_active = max(2, n_active // 2)
            if n_active < 2:
                break
    
    return best_a, best_sc

        A = fft(a, N)

            P_mod = np.maximum(P_mod, 0.0)
        
        # Reconstruct a

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        a3, sc3 = _active_set_direct(best_seq, deadline, min_remain=300.0)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic()-t_start:.1f}s")
    
    t_check = time.monotonic()
    print(f"\n=== Phase 4: Spectral flattening (t={t_check-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 250:
        a4, sc4 = _spectral_flatten(best_seq, deadline, min_remain=250.0)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")
    
    t_check = time.monotonic()
    print(f"\n=== Phase 5: Perturbation + re-optimize (t={t_check-t_start:.1f}s) ===")
    for sigma in [0.05, 0.1, 0.03]:
        if time.monotonic() > deadline - 150:
            break
        a_pert = _perturb_restart(best_seq, sigma, rng)
        a5, sc5 = _lbfgs_optimize(a_pert, deadline, min_remain=150.0)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [perturb sigma={sigma}] t={time.monotonic()-t_start:.1f}s")
    
    t_check = time.monotonic()
    print(f"\n=== Phase 6: Multi-n exploration (t={t_check-t_start:.1f}s) ===")
    n_candidates = [32768, 24576, 16384]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq):
            continue
        print(f"  Trying n={n_try}")
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"    init score={sc_init:.10f}")
        
        a6, sc6 = _lbfgs_optimize(a_init, deadline, min_remain=120.0)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"    NEW BEST: {best_score:.10f} [multi_n lbfgs] t={time.monotonic()-t_start:.1f}s")
        
        if time.monotonic() < deadline - 80:
            a6b, sc6b = _active_set_direct(a6, deadline, min_remain=80.0)
            if sc6b < best_score:
                best_seq = a6b.copy()
                best_score = sc6b
                print(f"    NEW BEST: {best_score:.10f} [multi_n active] t={time.monotonic()-t_start:.1f}s")
    
    t_check = time.monotonic()
    print(f"\n=== Phase 7: Shrinkage (t={t_check-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        a7, sc7 = _shrinkage(best_seq, deadline - 50)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 30:
                a7b, sc7b = _lbfgs_optimize(best_seq, deadline, min_remain=10.0)
                if sc7b < best_score:
                    best_seq = a7b.copy()
                    best_score = sc7b
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink] t={time.monotonic()-t_start:.1f}s")
    
    t_check = time.monotonic()
    print(f"\n=== Phase 8: Final polish (t={t_check-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 5:
        # Quick Nesterov with very small beta
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.0001
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        
        def obj_fin(z_val):
            return _smoobj(z_val, beta_fin, n)
        def grad_fin(z_val):
            return _smograd(z_val, beta_fin, n)
        
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(2000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(obj_fin, z, jac=grad_fin, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin.copy()
                    best_score = sc_fin
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
        
        # Final active-set
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_direct(best_seq, deadline - 1, min_remain=1.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_active] t={time.monotonic()-t_start:.1f}s")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
