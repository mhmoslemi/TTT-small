def _next_pow2(n):
    k = 1

    """Compute gradient of c_k w.r.t. a. Result is 2*a[k-j] for j in S_k."""
    n = len(a)

    lo = max(0, k - n + 1)
    hi = min(k, n - 1)

        # a[k-j] for j in [lo, hi] = a[k-hi : k-lo+1] reversed
        g[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    return g

**Alternative: Use scipy's QP solver**

For the QP:
$$\min_d \frac{1}{2} d^T d + q^T d \quad \text{s.t.} \quad Ad \leq b, \quad \mathbf{1}^T d = 0$$

where $q = 0$ (we just want minimum norm) and the constraints ensure the peaks are reduced.

Actually, let me try a different formulation. We want to find $d$ such that:
- $\mathbf{1}^T d = 0$
- $A d \leq -\epsilon$ where $A[i,j] = 2 a[k_i - j] \cdot \mathbf{1}[j \in S_{k_i}]$ for the $i$-th active peak
- $\|d\|$ is minimized

This is: $\min \frac{1}{2} d^T d$ s.t. $[A; \mathbf{1}^T] d \leq [-\epsilon; 0]$

(And also $[0; -\mathbf{1}^T] d \leq [0; 0]$ for the equality constraint)

For $n = 8192$ and $m \approx 10$ active peaks, this is a QP with 8192 variables and ~21 constraints. scipy can handle this.

But actually, the solution to this QP has a simple form. By KKT:
$d = A^T \lambda + \mu \mathbf{1}$

where $\lambda \geq 0$ and $\mu$ are Lagrange multipliers. The active constraints determine $\lambda$.

For the equality constraint $\mathbf{1}^T d = 0$:
$\mathbf{1}^T A^T \lambda + n\mu = 0$
$\mu = -\frac{1}{n} (A \mathbf{1})^T \lambda$

For each active inequality constraint $i$ (where $\lambda_i > 0$):
$(A d)_i = -\epsilon$
$(A A^T)_{i\cdot} \lambda + (A \mathbf{1})_i \mu = -\epsilon$

This gives a system in $\lambda$ that's only $m \times m$. We can solve it using the active set method for QPs.

def _qp_min_norm_direction(a, c, active_k, eps=0.01):
    """
    Find minimum-norm direction d such that:
    - sum(d) = 0
    - For each active peak k: sum_j d_j * 2*a[k-j] <= -eps
    Returns d or None if infeasible.
    """
    n = len(a)
    m = len(active_k)
    
    # Build A matrix (m x n): A[i, j] = 2 * a[k_i - j] if j in S_{k_i}
    A = np.zeros((m, n))
    for i, k in enumerate(active_k):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    
    # A1 = A @ ones(n) = sum_j A[i, j]
    A1 = A.sum(axis=1)  # (m,)
    
    # We want d = A^T @ lambda + mu * ones(n)
    # Constraint: sum(d) = 0 => A1^T @ lambda + n*mu = 0 => mu = -A1^T @ lambda / n
    # For active constraints i: (A @ d)_i = -eps
    # A @ A^T @ lambda + A @ (mu * ones) = -eps
    # A @ A^T @ lambda + mu * A1 = -eps
    # A @ A^T @ lambda - (A1^T @ lambda / n) * A1 = -eps
    # (A @ A^T - outer(A1, A1)/n) @ lambda = -eps
    
    G = A @ A.T - np.outer(A1, A1) / n  # (m x m)
    b = -np.full(m, eps)
    
    # Solve G @ lambda = b
    try:
        lambda_full = np.linalg.solve(G + 1e-12 * np.eye(m), b)
    except np.linalg.LinAlgError:
        lambda_full = np.linalg.lstsq(G, b, rcond=None)[0]
    
    # Check complementary slackness: lambda_i >= 0 for all i
    # If some lambda_i < 0, that constraint is not active
    # Use active set method
    active_set = set(range(m))
    for _ in range(m + 1):
        if not active_set:
            return None
        idx = list(active_set)
        G_sub = G[np.ix_(idx, idx)]
        b_sub = b[idx]
        try:
            lambda_sub = np.linalg.solve(G_sub + 1e-12 * np.eye(len(idx)), b_sub)
        except np.linalg.LinAlgError:
            return None
        
        if np.all(lambda_sub >= -1e-10):
            # All multipliers non-negative, this is the solution
            lambda_vec = np.zeros(m)
            for i, j in enumerate(idx):
                lambda_vec[j] = max(0, lambda_sub[i])
            break
        else:
            # Remove the most negative multiplier
            min_idx = np.argmin(lambda_sub)
            active_set.remove(idx[min_idx])
    
    else:
        return None
    
    mu = -float(np.dot(A1, lambda_vec)) / n
    d = A.T @ lambda_vec + mu * np.ones(n)
    
    return d

import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _get_active_peaks(c, tol=1e-5, max_count=30):
    c_max = float(c.max())
    active = np.where(c >= c_max * (1.0 - tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    elif len(active) > max_count:
        top_idx = np.argsort(c[active])[::-1][:max_count]
        active = active[top_idx]
    return active, c_max

        vals = c_act + 2.0 * alpha * g_act + alpha * alpha * h_act
        return float(np.max(vals))

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = alpha_max
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj(c_pt)
        fd = obj(d_pt)
        for _ in range(max_iter):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj(d_pt)
        alpha = 0.5 * (lo + hi)
    else:
        alpha = alpha_max

    return alpha


def _subgradient_round(a, deadline, tol_peaks=1e-5, max_iter=20000,
                       alpha_decay=0.95, min_alpha=1e-8):
    """Precise subgradient with exact line search."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    current_alpha_cap = 0.3
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        active, c_max = _get_active_peaks(c, tol=tol_peaks)

        g = _compute_subgradient(a, active)
        # Project to {d: sum(d)=0}
        g = g - float(np.mean(g))
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        d = -g / g_norm

        # Alpha max from non-negativity
        neg = d < -1e-15
        if np.any(neg):
            alpha_max = float(np.min(a[neg] / (-d[neg])))
            alpha_max = min(alpha_max, current_alpha_cap)
        else:
            alpha_max = current_alpha_cap

        if alpha_max < min_alpha:
            break

        # Exact line search
        alpha = _line_search(a, d, c, active, alpha_max)

        if alpha < min_alpha:
            current_alpha_cap *= 0.5
            if current_alpha_cap < min_alpha:
                break
            continue

        a_new = a + alpha * d
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        sc = _score_norm(a_new)
        if sc < best_score * (1.0 - 1e-10):
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            current_alpha_cap = max(current_alpha_cap * 0.98, 0.05)
        else:
            stall_count += 1
            current_alpha_cap *= alpha_decay
            if stall_count > 100:
                break

    return best_a, best_score


def _qp_equalize(a, deadline, max_rounds=500):
    """
    QP-based active set equalization.
    Solves min ||d||^2 s.t. A*d <= -eps, 1^T d = 0
    where A[i,j] = 2*a[k_i - j] for active peaks.
    """
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    eps = 0.005

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Use a tighter tolerance for "active" peaks in QP
        active, _ = _get_active_peaks(c, tol=1e-4, max_count=15)
        m = len(active)
        if m < 2:
            break

        # Build A (m x n)
        A = np.zeros((m, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]

        A1 = A.sum(axis=1)  # (m,)

        # QP: min 0.5*d^T*d s.t. A*d <= -eps*1, 1^T*d = 0
        # KKT: d = A^T*lambda + mu*1, mu = -(A1^T*lambda)/n
        # Active constraints: (A*A^T - 1/n * A1*A1^T) * lambda_active = -eps
        G = A @ A.T - np.outer(A1, A1) / n  # (m x m)
        b = -np.full(m, eps)

        # Active set method for QP
        in_set = list(range(m))
        lambda_vec = None
        for _ in range(m + 2):
            if not in_set:
                break
            idx = np.array(in_set)
            G_sub = G[np.ix_(idx, idx)]
            b_sub = b[idx]
            try:
                lam_sub = np.linalg.solve(G_sub + 1e-10 * np.eye(len(idx)), b_sub)
            except np.linalg.LinAlgError:
                lam_sub = np.linalg.lstsq(G_sub, b_sub, rcond=None)[0]

            if np.all(lam_sub >= -1e-8):
                lambda_vec = np.zeros(m)
                for i, j in enumerate(idx):
                    lambda_vec[j] = max(0.0, lam_sub[i])
                break
            else:
                min_i = int(np.argmin(lam_sub))
                in_set.remove(int(idx[min_i]))

        if lambda_vec is None:
            # Fallback: use the subgradient direction
            g = _compute_subgradient(a, active)
            g = g - np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            d = -g / gn
            neg = d < -1e-15
            alpha_max = 0.1
            if np.any(neg):
                alpha_max = min(alpha_max, float(np.min(a[neg] / (-d[neg]))))
            a_new = a + alpha_max * d
            a_new = np.maximum(0, a_new)
            a_new /= a_new.sum()
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            continue

        mu = -float(np.dot(A1, lambda_vec)) / n
        d = A.T @ lambda_vec + mu  # mu * ones(n)

        # Normalize d
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break

        # Line search
        c = _fft_convolve(a, a)
        cross_da = _fft_convolve(d, a)
        dd = _fft_convolve(d, d)

        c_act = c[active]
        g_act = cross_da[active]
        h_act = dd[active]

        neg = d < -1e-15
        if np.any(neg):
            alpha_max = float(np.min(a[neg] / (-d[neg])))
            alpha_max = min(alpha_max, 0.1)
        else:
            alpha_max = 0.1

        def qp_obj(alpha):
            vals = c_act + 2.0 * alpha * g_act + alpha * alpha * h_act
            return float(np.max(vals))

        # Try decreasing alphas
        alpha = alpha_max
        improved = False
        for _ in range(20):
            a_new = a + alpha * d
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_new /= s
            sc = _score_norm(a_new)
            if sc < best_score * (1.0 - 1e-10):
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            alpha *= 0.7

        if not improved:
            # Try with smaller eps
            eps *= 0.5
            if eps < 1e-6:
                break

    return best_a, best_score


def _coordinate_polish(a, deadline, max_rounds=500):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        k_max = int(np.argmax(c))

        # Elements contributing to the peak
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        # Also consider the second peak
        c_sorted = np.sort(c)[::-1]
        k_second = int(np.argsort(c)[::-1][1]) if len(c) > 1 else k_max
        lo2 = max(0, k_second - n + 1)
        hi2 = min(k_second, n - 1)

        all_coords = set(range(lo, hi + 1))
        all_coords.update(range(lo2, hi2 + 1))
        all_coords = sorted(all_coords)

        if not all_coords:
            break

        improved = False
        for j in all_coords:
            old_val = a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.85, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a_cand = a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = 2.0 * n * float(c_cand.max())
                if sc < best_score * (1.0 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try randomized coordinates
            rng = np.random.default_rng(rnd * 7919 + 13)
            try_coords = rng.choice(n, size=min(200, n), replace=False)
            for j in try_coords:
                old_val = a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.7, 0.8, 1.2, 1.3]:
                    a_cand = a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def _adam_logspace(a_init, deadline, lr_start=0.05, lr_end=0.001,
                   T_ratio=0.01, max_iter=100000, eval_interval=50):
    """Adam in log-space with softmax parameterization."""
    n = len(a_init)
    a0 = a_init.copy() / a_init.sum()
    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()

    c0 = _fft_convolve(a0, a0)
    T = float(c0.max()) * T_ratio

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_a = a0.copy()
    best_score = _score_norm(a0)

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac
        T_cur = T * (1.0 - 0.9 * frac)

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        active, _ = _get_active_peaks(c, tol=0.001, max_count=8)

        # Smoothed subgradient
        c_act = c[active]
        w = np.exp((c_act - c_act.max()) / (T_cur + 1e-10))
        w = w / w.sum()

        g = np.zeros(n)
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += w[idx] * 2.0 * a[k-hi:k-lo+1][::-1]

        # Center gradient for log-space
        W = float(np.dot(a, g))
        zg = a * (g - W)

        m_buf = b1 * m_buf + (1.0 - b1) * zg
        v_buf = b2 * v_buf + (1.0 - b2) * (zg * zg)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 2.0:
            update *= 2.0 / update_norm

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Adam log-space warm-up
    2. Precise subgradient with exact line search
    3. QP-based active set equalization
    4. More subgradient rounds with tighter tolerance
    5. Coordinate descent polish
    6. Structured restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Phase 1: Adam log-space (0 - 300s)
    p1_end = min(deadline, t_start + 300.0)
    print(f"Phase 1: Adam log-space, budget={p1_end - t_start:.0f}s")
    a_opt, sc = _adam_logspace(a, p1_end, lr_start=0.04, lr_end=0.002,
                                T_ratio=0.02, max_iter=80000, eval_interval=40)
    if sc < best_score:
        best_score = sc
        best_a = a_opt.copy()
    print(f"Phase 1 done: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Subgradient with moderate tolerance (300 - 600s)
    p2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: Subgradient (tol=1e-4)")
        a_sg, sc_sg = _subgradient_round(best_a, p2_end, tol_peaks=1e-4,
                                          max_iter=30000, alpha_decay=0.97,
                                          min_alpha=1e-6)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
        print(f"Phase 2 done: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: QP equalization (600 - 900s)
    p3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: QP equalization")
        a_qp, sc_qp = _qp_equalize(best_a, p3_end, max_rounds=300)
        if sc_qp < best_score:
            best_score = sc_qp
            best_a = a_qp.copy()
        print(f"Phase 3 done: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 4: Subgradient with tight tolerance (900 - 1200s)
    p4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Subgradient (tol=1e-6)")
        a_sg2, sc_sg2 = _subgradient_round(best_a, p4_end, tol_peaks=1e-6,
                                            max_iter=30000, alpha_decay=0.96,
                                            min_alpha=1e-9)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_a = a_sg2.copy()
        print(f"Phase 4 done: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 5: QP again with tighter eps (1200 - 1450s)
    p5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < p5_end:
        print(f"Phase 5: QP (tight)")
        a_qp2, sc_qp2 = _qp_equalize(best_a, p5_end, max_rounds=400)
        if sc_qp2 < best_score:
            best_score = sc_qp2
            best_a = a_qp2.copy()
        print(f"Phase 5 done: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 6: Structured restarts (1450 - 1680s)
    p6_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < p6_end:
        print(f"Phase 6: Structured restarts")
        n_restarts = 6
        time_per = max(15.0, (p6_end - time.monotonic()) / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > p6_end - time_per * 0.5:
                break

            # Different perturbation strategies
            if r % 3 == 0:
                # Gaussian perturbation
                sigma = 0.003 + 0.002 * r
                z = np.log(np.maximum(best_a, 1e-15))
                z += sigma * rng.standard_normal(n)
                z = z - z.mean()
                a_pert = np.exp(z - z.max())
                a_pert = a_pert / a_pert.sum()
            elif r % 3 == 1:
                # Targeted: reduce top contributors to peak
                c = _fft_convolve(best_a, best_a)
                k_max = int(np.argmax(c))
                lo = max(0, k_max - n + 1)
                hi = min(k_max, n - 1)
                a_pert = best_a.copy()
                a_pert[lo:hi+1] *= 0.9
                a_pert /= a_pert.sum()
            else:
                # Spread: increase small values
                a_pert = best_a.copy()
                median = np.median(a_pert)
                small_mask = a_pert < median * 0.5
                a_pert[small_mask] *= 1.5
                a_pert /= a_pert.sum()

            # Quick subgradient optimization from perturbation
            sub_dl = min(p6_end - 5, time.monotonic() + time_per * 0.8)
            a_opt_r, sc_r = _subgradient_round(a_pert, sub_dl, tol_peaks=1e-5,
                                                 max_iter=15000, alpha_decay=0.95,
                                                 min_alpha=1e-7)
            # Draw the best from both the original and perturbed
            sc_orig = _score_norm(best_a)
            if sc_r < best_score:
                best_score = sc_r
                best_a = a_opt_r.copy()
            print(f"  Restart[{r}]: score={min(sc_r, sc_orig):.10f} t={time.monotonic()-t_start:.0f}s")

    # Phase 7: Final polish (1680 - deadline)
    if time.monotonic() < deadline - 10:
        print(f"Phase 7: Final polish")
        # Ultra-fine subgradient
        p7a_end = min(deadline - 5, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        a_final, sc_final = _subgradient_round(best_a, p7a_end, tol_peaks=1e-7,
                                                max_iter=20000, alpha_decay=0.95,
                                                min_alpha=1e-10)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()

        # Coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd, sc_cd = _coordinate_polish(best_a, deadline - 3, max_rounds=200)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cd.copy()

        print(f"Phase 7 done: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01

    sc_check = _score_full(final)
    print(f"FINAL: score={sc_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
