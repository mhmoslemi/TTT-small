import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# Smoothing-based L-BFGS
def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    sv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (sv[-1] - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (sv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _mirror_gradient(a, peak_positions):
    """Compute the averaged mirror gradient for given peak positions.
    Returns g[j] = mean over p in peak_positions of a[p-j] (with bounds)."""
    n = len(a)
    g = np.zeros(n)
    K = len(peak_positions)
    for p in peak_positions:
        # a[p-j] is valid when 0 <= p-j < n, i.e., p-n+1 <= j <= p
        j_lo = max(0, p - (n - 1))
        j_hi = min(n - 1, p)
        # a[p-j] for j in [j_lo, j_hi]: this is a[p-j_hi : p-j_lo+1][::-1]
        vals = a[j_lo: j_hi + 1][::-1]
        g[j_lo:j_hi + 1] += vals
    g /= K
    return g


def _mirror_gradient_descent(a_init, deadline, min_time_remaining=60.0):
    """Mirror gradient method with projected steps on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_c = _full_convolve(a, a)
    best_cmax = float(best_c.max())

    # Parameters
    top_k_frac_start = 0.005
    top_k_frac_end = 0.05
    alpha_init = 0.1
    no_improve = 0
    max_no_improve = 200
    stall_restarts = 0
    max_restarts = 15
    iteration = 0
    t_start = time.monotonic()

    # Barzilai-Borwein state
    prev_a = a.copy()
    prev_g = None

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            best_c = c.copy()
            best_cmax = c_max
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive top_k
        elapsed_frac = min(1.0, (time.monotonic() - t_start) / max(1, deadline - min_time_remaining - t_start))
        top_k_frac = top_k_frac_start + (top_k_frac_end - top_k_frac_start) * elapsed_frac
        top_k = max(3, int(n * top_k_frac))

        # Find top peak positions
        top_indices = np.argsort(c)[::-1][:top_k]

        # Compute mirror gradient
        g = _mirror_gradient(a, top_indices)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)

        if gnorm < 1e-15:
            break

        # Step size: try Barzilai-Borwein or default
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            sy = float(np.dot(s_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            yy = float(np.dot(y_vec, y_vec))
            if ss > 1e-15 and yy > 1e-15 and sy > 0:
                alpha_bb = ss / sy
                alpha_bb = np.clip(alpha_bb, 0.01 * n, 10.0 * n)
            else:
                alpha_bb = alpha_init * n
        else:
            alpha_bb = alpha_init * n

        # Backtracking line search
        alpha = alpha_bb
        improved = False
        for _ in range(30):
            a_try = _project_simplex(a - alpha * g_centered)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 5
            if no_improve >= max_no_improve:
                # Restart from best with perturbation
                a = best_a.copy()
                sigma = 0.01 * (1 + stall_restarts * 0.5)
                z = np.log(np.maximum(a, 1e-30)) + sigma * np.random.standard_normal(n)
                z -= z.mean()
                a = _softmax(z)
                no_improve = 0
                stall_restarts += 1
                prev_g = None
                if stall_restarts >= max_restarts:
                    break

        prev_a = a.copy()
        prev_g = g_centered.copy()
        iteration += 1

        if iteration % 500 == 0:
            print(f"    MG iter {iteration}: score={best_sc:.10f}, top_k={top_k}, alpha={alpha:.2f}, t={time.monotonic()-t_start:.1f}s")

    return best_a, best_sc


def _water_filling_descent(a_init, deadline, min_time_remaining=60.0):
    """Water-filling: equalize top peaks by moving mass from high to low sensitivity."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    rng = np.random.default_rng(123)
    top_k = max(10, int(n * 0.02))
    alpha = 0.05 * n
    no_improve = 0
    max_no_improve = 300
    iteration = 0
    t_start = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        # Find top peaks and their values
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]

        # Soft weights: softmax with temperature based on spread
        spread = max(top_vals[0] - top_vals[-1], c_max * 1e-6)
        temps = spread / 5.0
        w = np.exp(-(top_vals - top_vals[0]) / temps)
        w /= w.sum()

        # Weighted mirror gradient
        g = np.zeros(n)
        for ki, p in enumerate(top_indices):
            j_lo = max(0, p - (n - 1))
            j_hi = min(n - 1, p)
            vals = a[j_lo:j_hi + 1][::-1]
            g[j_lo:j_hi + 1] += w[ki] * vals
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)

        if gnorm < 1e-15:
            break

        # Backtracking
        alpha_try = alpha
        improved = False
        for _ in range(25):
            a_try = _project_simplex(a - alpha_try * g_centered)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                alpha = min(alpha * 1.05, 2.0 * n)
                improved = True
                break
            alpha_try *= 0.5

        if not improved:
            alpha *= 0.7
            if alpha < 0.01:
                break

        iteration += 1
        if iteration % 500 == 0:
            print(f"    WF iter {iteration}: score={best_sc:.10f}, alpha={alpha:.2f}, t={time.monotonic()-t_start:.1f}s")

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with aggressive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # More aggressive schedule: start larger, go smaller faster, more iterations at small beta
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # More iterations for small beta values
        if beta < base_beta * 0.01:
            max_iter = min(6000, max(50, int(remaining / (0.003 * max(1.0, n / 16384.0) * 10))))
        else:
            max_iter = min(3000, max(30, int(remaining / (0.003 * max(1.0, n / 16384.0) * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _coordinate_peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Direct coordinate descent: for the peak position, find best (j,k) swap to reduce it."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    t_start = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        p = int(np.argmax(c))
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        # Sensitivity: how much does a[j] contribute to peak at p?
        # d(conv[a,a][p])/d(a[j]) = a[p-j] + a[p-j] (if j != p-j) or a[j] (if j == p-j)
        # Simplified: sensitivity[j] = a[p-j] (clipped to valid range)
        sens = np.zeros(n)
        j_lo = max(0, p - (n - 1))
        j_hi = min(n - 1, p)
        sens[j_lo:j_hi + 1] = a[j_lo:j_hi + 1][::-1]

        # We want to move mass from high-sens to low-sens positions
        # Try a random batch of swaps
        n_swaps = min(n // 10, 1000)
        high_idx = np.where(sens > np.percentile(sens, 75))[0]
        low_idx = np.where(sens < np.percentile(sens, 25))[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            break

        improved = False
        for _ in range(n_swaps):
            j = high_idx[rng.integers(len(high_idx))]
            k = low_idx[rng.integers(len(low_idx))]
            if j == k:
                continue
            # Move small amount from j to k
            delta = min(a[j], c_max * 0.001) * rng.uniform(0.01, 0.1)
            if delta < 1e-15:
                continue
            a_try = a.copy()
            a_try[j] -= delta
            a_try[k] += delta
            if a_try[j] < 0:
                a_try[j] = 0
                a_try[k] += a[j]
                continue
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                current_sc = sc_try
                improved = True
                break

        if not improved and no_improve > 100:
            # Try a mirror gradient step instead
            sens_c = sens - sens.mean()
            alpha = 0.1 * n
            a_try = _project_simplex(a - alpha * sens_c)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                current_sc = sc_try
                improved = True

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n

        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [mg1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  MG1: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n[P2] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    lbfgs_deadline = min(deadline - 500, time.monotonic() + 250)
    a_lb = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=100.0)
    sc_lb = _score(a_lb)
    if sc_lb < best_score:
        best_seq = a_lb.copy()
        best_score = sc_lb
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  L-BFGS: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Water-filling descent ===
    print(f"\n[P3] Water-filling descent (t={time.monotonic()-t_start:.1f}s)")
    wf_deadline = min(deadline - 400, time.monotonic() + 200)
    a_wf, sc_wf = _water_filling_descent(best_seq, wf_deadline, min_time_remaining=80.0)
    if sc_wf < best_score:
        best_seq = a_wf.copy()
        best_score = sc_wf
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [wf] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  WF: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Mirror gradient round 2 (with perturbation) ===
    print(f"\n[P4] Mirror gradient round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_pert = _perturb(best_seq, 0.005, rng)
        mg_deadline2 = min(deadline - 300, time.monotonic() + 150)
        a_mg2, sc_mg2 = _mirror_gradient_descent(a_pert, mg_deadline2, min_time_remaining=60.0)
        if sc_mg2 < best_score:
            best_seq = a_mg2.copy()
            best_score = sc_mg2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [mg2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 12288, 8192, 24576]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        if n_try == len(best_seq):
            continue
        # Check feasibility
        if n_try > 32768:
            continue

        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # Quick mirror gradient
        mg_d = min(deadline - 150, time.monotonic() + 80)
        a_opt, sc_opt = _mirror_gradient_descent(a_init, mg_d, min_time_remaining=30.0)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_mg] t={time.monotonic()-t_start:.1f}s")

        # Quick L-BFGS
        if time.monotonic() < deadline - 60:
            lb_d = min(deadline - 55, time.monotonic() + 40)
            a_opt2 = _run_lbfgs(a_opt, lb_d, min_time_remaining=20.0)
            sc_opt2 = _score(a_opt2)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Coordinate peak shaving ===
    print(f"\n[P6] Coordinate peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        ps_deadline = min(deadline - 80, time.monotonic() + 80)
        a_ps, sc_ps = _coordinate_peak_shave(best_seq, ps_deadline, min_time_remaining=30.0)
        if sc_ps < best_score:
            best_seq = a_ps.copy()
            best_score = sc_ps
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peak_shave] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final L-BFGS polish ===
    print(f"\n[P7] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        fin_deadline = deadline - 10
        a_fin = _run_lbfgs(best_seq, fin_deadline, min_time_remaining=10.0)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final mirror gradient micro-polish ===
    print(f"\n[P8] Final mirror gradient micro-polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_mg_final, sc_mg_final = _mirror_gradient_descent(best_seq, deadline - 3, min_time_remaining=3.0)
        if sc_mg_final < best_score:
            best_seq = a_mg_final.copy()
            best_score = sc_mg_final
            print(f"  NEW BEST: {best_score:.10f} [final_mg] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
