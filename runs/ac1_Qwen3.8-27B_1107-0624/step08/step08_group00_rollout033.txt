```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based peak reduction, L-BFGS with log-sum-exp smoothing,
and multi-start strategies.
"""
import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
```

Given current point a (normalized to sum=1), current convolution c = conv(a,a), and max value M = c.max():

We want to find d such that:

- a_new >= 0  →  d >= -a
- sum(a_new) = 1  →  sum(d) = 0
- conv(a_new, a_new)[k] <= M for all k (to not increase the max)

Linearizing: conv(a_new, a_new)[k] ≈ c[k] + 2*conv(a, d)[k]

So: 2*conv(a, d)[k] <= M - c[k] for all k.

For k where c[k] < M, the RHS is positive, so the constraint is not tight.
For k where c[k] = M, the RHS is 0, so conv(a, d)[k] <= 0.

To ALSO reduce the peaks, we can set a target < M:
2*conv(a, d)[k] <= target - c[k] for k in active set (where c[k] is close to M).

And to fill the valleys:
We want to maximize the minimum of c + 2*conv(a,d) over the non-active set.

The LP:
maximize: t
subject to:

  c[k] + 2*(conv(a,d))[k] <= target  for k in active set
  d >= -a
  sum(d) = 0

Where:
- Active set: top 5-10% of c values
- Valley set: bottom 5-10% of c values
- target: slightly below M (e.g., M * 0.99)

The constraint "c[k] + 2*(conv(a,d))[k] >= t" can be rewritten as:
2*(conv(a,d))[k] >= t - c[k]

And "conv(a,d)[k] = sum_i a[i]*d[k-i]" is linear in d.

So the constraint matrix is:
For each k in valley set: row = 2 * (the k-th row of the "conv(a, .)" operator)
For each k in active set: row = -2 * (the k-th row of the "conv(a, .)" operator) (flipped for <=)

The "conv(a, .)" operator: (conv(a, d))[k] = sum_{i: valid} a[i] * d[k-i]

So the k-th row has entries: row[j] = a[k-j] if 0 <= k-j < n, else 0.

This is a banded matrix (each row has at most n nonzeros, but in practice, for k in 0..2n-2, the number of nonzeros is min(k+1, n, 2n-1-k)).

For n=16384, the constraint matrix has about 1600 constraints (800 active + 800 valley) and 16384 columns. Each row has at most 16384 nonzeros (but typically Much fewer for k near the edges).

"""
Multi-phase optimizer combining LP-based peak reduction, L-BFGS with
log-sum-exp smoothing, active-set gradient descent, and multi-start
strategies for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: uses convex LP subproblems to find directions that
simultaneously reduce peaks and fill valleys in the autocorrelation.
"""
import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.arange(n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 3.0
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.arange(n)
        edge = n // 10
        a = np.ones(n)
        for i in range(edge):
            a[i] *= 0.5 * (1 - np.cos(np.pi * i / edge))
            a[n - 1 - i] *= 0.5 * (1 - np.cos(np.pi * i / edge))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    a_sum = a.sum()
    grad = grad - a * (grad @ a) / (a_sum * a_sum)
    return grad


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005])

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 15))))

        obj = _smoobj
        grad = _smograd
        try:
            result = minimize(
                lambda zz: _smoobj(zz, beta, n),
                z,
                jac=lambda zz: _smograd(zz, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return a, current_sc


def _build_conv_rows(a, indices):
    n = len(a)
    m = len(indices)
    rows = np.zeros((m, n))
    for idx, k in enumerate(indices):
        if k < n:
            rows[idx, :k + 1] = a[k::-1]
        else:
            j_lo = k - n + 1
            j_hi = n - 1
            rows[idx, j_lo:j_hi + 1] = a[n - 1:k - n:-1]
    return rows


def _lp_direction(a, active_indices, valley_indices, target, deadline):
    n = len(a)
    c = _full_convolve(a, a)
    n_active = len(active_indices)
    n_valley = len(valley_indices)
    if n_active == 0 or n_valley == 0:
        return None

    all_indices = list(active_indices) + list(valley_indices)
    B = _build_conv_rows(a, all_indices)

    d_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [cp.sum(d_var) == 0]

    zero_tol = 1e-10
    small_mask = a < zero_tol
    if small_mask.any():
        constraints.append(d_var[small_mask] >= -a[small_mask])

    for i in range(n_active):
        k = active_indices[i]
        constraints.append(2.0 * (B[i] @ d_var) <= target - c[k])

    for i in range(n_active, n_active + n_valley):
        k = valley_indices[i]
        constraints.append(2.0 * (B[i] @ d_var) >= t_var - c[k])

    max_step = 0.5 * a.max()
    constraints.append(d_var <= max_step)
    constraints.append(d_var >= -max_step)

    obj = cp.Maximize(t_var - 1e-4 * cp.sum_squares(d_var))
    prob = cp.Problem(obj, constraints)

    solved = False
    for solver in ['GUROBI', 'MOSEK', 'ECOS', 'SCS', 'CLARABEL']:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved:
        try:
            prob.solve()
            solved = True
        except Exception:
            return None

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None

    d = d_var.value.copy()
    if d is None or not np.all(np.isfinite(d)):
        return None
    return d


def _active_set_gradient_step(a, tol_frac, step_size):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1

    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / n_active

    N = _next_pow2((2 * n - 1) + n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    avg_grad = 2.0 * cross[:n]

    avg_grad = avg_grad - np.mean(avg_grad)
    a_new = a - step_size * avg_grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), c_max
    a_new = a_new / s
    c_new = _full_convolve(a_new, a_new)
    return a_new, float(c_new.max())


def _line_search(a, d, current_sc, max_alphas=30):

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-15:

            best_a = a_try.copy()
        alpha *= 0.7
        if alpha < 1e-10:
            break
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Multi-start initialization ===
    print(f"\n[P1] Multi-start init (t={time.monotonic()-t_start:.1f}s)")
    n_main = len(best_seq)
    profiles_to_try = ["hann", "gaussian", "flat_taper", "cos_mod", "bimodal"]
    time_per_profile = 40.0

        if time.monotonic() + time_per_profile > deadline - 1200:
