        K_act = len(active)
        
        # Build Jacobian J[i,j] = d(a*a)[active[i]]/d a[j] = 2*a[active[i]-j]
        J = np.zeros((K_act, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                J[i, lo:hi + 1] = 2.0 * best_a[k - hi:k - lo + 1][::-1]
        
        # Solve: minimize ||J @ d - (-1)||^2 + lambda * (sum(d))^2
        # -> d = (J^T J + lambda * 1*1^T)^{-1} J^T (-1)
        # For efficiency with n >> K_act: d = J^T (J J^T + lambda * J 1 1^T J^T)^{-1} (-1)
        # Simpler: use the KKT system
        # Minimize 0.5*d^T G d + b^T d s.t. 1^T d = 0
        # where G = J^T J + eps*I, b = -J^T 1
        # KKT: [G  1] [d]   [(-b)]
        #       [1^T 0] [nu] = [ 0  ]
        
        # Use the smaller system since K_act << n:
        # d = J^T @ alpha, where alpha solves (J J^T + mu * J @ 1 @ 1^T @ J^T) alpha = -J^T @ 1
        # This is getting complex; let's just use lstsq with the sum constraint
        
        sum_vec = np.ones(n)
        M = np.vstack([J, 0.01 * sum_vec])
        b = np.concatenate([np.full(K_act, -1.0 / K_act), [0.0]])
        
        try:
            d, _, _, _ = np.linalg.lstsq(M, b, rcond=1e-12)
        except np.linalg.LinAlgError:
            break
        
        # Scale to trust region
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        d = d * min(1.0, trust_radius / d_norm)
        
        # Try multiple step sizes
        improved = False
        for alpha_step in [1.0, 0.5, 0.25, 0.1]:
            a_cand = best_a + alpha_step * d
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand = a_cand / s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cand.copy()
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            # Adapt trust region
            trust_radius *= 0.7
            if no_improve > 20 and trust_radius < 1e-5:
                break
    
    return best_a, best_score


def _subgradient_adam(a_init, deadline, lr_start=0.003, active_frac=0.995, max_iter=200000):
    """Adam in log-space with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    no_improve = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * lr_mult * (1.0 - 0.5 * frac)
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Active set
        tol = c_max * active_frac
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > 50:
            top_idx = np.argsort(c[active])[::-1][:50]
            active = active[top_idx]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)
        
        # Center for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update *= 0.5 / update_norm
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        # Evaluate periodically
        if t % 300 == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adapt
            if no_improve > 15 and t > 500:
                lr_mult *= 0.7
                no_improve = 0
                if lr_mult < 0.05:
                    # Restart from best with small perturbation
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.2
    
    a_final = _softmax_stable(z)
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score


def _direct_subgradient(a_init, deadline, max_iter=50000, K=10, lr_start=0.01):
    """Direct subgradient on the simplex (no log parameterization)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_start
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: top K peaks within 99.5% of max
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:K]:
            if c[idx] >= c_max * 0.995:
                active.append(idx)
        
        # Subgradient (average over active)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)
        
        # Project gradient to tangent space of simplex
        # Subtract the component along 'a' (the normal to the face of simplex)
        g_proj = g - np.dot(g, a) * a
        
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-15:
            break
        
        # Polyak step size or fixed decreasing
        step = lr / (1.0 + it * 0.01)
        a_new = a - step * g_proj / g_norm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score_norm(a_new)
        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr *= 0.999
        else:
            no_improve += 1
            lr *= 0.995
            if no_improve > 200:
                break
    
    return best_a, best_score


def _coordinate_refine(a, deadline, max_rounds=200, n_coords=None):
    """Coordinate descent: try scaling individual elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    
    if n_coords is None:
        n_coords = 500
    
    # Focus on elements that contribute to the peak
    k_max_arr = np.argsort(c)[::-1][:10]
    contributing = set()
    for k in k_max_arr:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        contributing.update(range(lo, hi + 1))
    
    if not contributing:
        contributing = set(range(0, n, max(1, n // n_coords)))
    
    # Convert to list, shuffle for diversity
    coord_list = list(contributing)
    if len(coord_list) > n_coords:
        coord_list = coord_list[:n_coords]
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        any_improve = False
        
        for j in coord_list:
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    any_improve = True
                    break
            if any_improve:
                break
        
        if not any_improve:
            # Try a random subset of coordinates
            rng_local = np.random.default_rng(round_num)
            random_coords = rng_local.choice(n, size=min(200, n), replace=False)
            any_improve2 = False
            for j in random_coords:
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.5, 0.7, 1.4, 2.0]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        any_improve = True
                        any_improve2 = True
                        break
                if any_improve2:
                    break
            if not any_improve2:
                break
    
    return best_a, best_score


def _two_point_swap(a, deadline, max_swaps=5000):
    """Try pairwise swaps: move mass from one element to another."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    rng = np.random.default_rng(12345)
    
    # Focus on elements contributing to peaks
    k_max_arr = np.argsort(c)[::-1][:5]
    contributing = set()
    for k in k_max_arr:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        contributing.update(range(lo, hi + 1))
    coord_list = list(contributing)[:300]
    
    swap_count = 0
    while swap_count < max_swaps:
        if time.monotonic() > deadline:
            break
        swap_count += 1
        
        # Pick two random elements from contributing set
        i = rng.choice(coord_list)
        j = rng.choice(coord_list)
        if i == j:
            continue
        
        # Try moving a small fraction of mass from i to j (and vice versa)
        base = best_a[i] + best_a[j]
        if base < 1e-12:
            continue
        
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            a_cand = best_a.copy()
            a_cand[i] = base * frac
            a_cand[j] = base * (1 - frac)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            c_cand = _fft_convolve(a_cand, a_cand)
            sc = float(c_cand.max()) * 2.0 * n
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cand.copy()
                c = c_cand
                break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Gentle spectral shaping (periodogram flattening)
    2. Active-set Newton with large K
    3. Adam in log-space with adaptive active sets
    4. Direct subgradient on simplex
    5. Coordinate descent + two-point swaps
    6. Random restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")
    
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")
    
    # Phase 1: Gentle spectral shaping (periodogram compression)
    phase1_end = min(deadline, t_start + 100.0)
    print(f"Phase 1: Spectral shaping (power), deadline={phase1_end-t_start:.0f}s")
    a_s1, sc_s1 = _spectral_shaping(best_a, phase1_end, alpha=0.997, max_iter=5000)
    full_s1 = _score(a_s1)
    print(f"  Power shaping: score={full_s1:.10f}")
    if full_s1 < best_score:
        best_score = full_s1
        best_a = a_s1.copy()
    
    # Phase 1b: Threshold spectral shaping
    if time.monotonic() < deadline and best_score > 1.50:
        phase1b_end = min(deadline, t_start + 180.0)
        a_s2, sc_s2 = _spectral_shaping_threshold(best_a, phase1b_end, max_iter=3000)
        full_s2 = _score(a_s2)
        print(f"  Threshold shaping: score={full_s2:.10f}")
        if full_s2 < best_score:
            best_score = full_s2
            best_a = a_s2.copy()
    
    # Phase 2: Active-set Newton (large K)
    phase2_end = min(deadline, t_start + 450.0)
    print(f"Phase 2: Active-set Newton, deadline={phase2_end-t_start:.0f}s")
    a_as, sc_as = _active_set_newton(best_a, phase2_end, K=50, trust_radius=0.008, max_iter=4000)
    full_as = _score(a_as)
    print(f"  Active-set: score={full_as:.10f}")
    if full_as < best_score:
        best_score = full_as
        best_a = a_as.copy()
    
    # Phase 3: Adam in log-space
    phase3_end = min(deadline, t_start + 900.0)
    print(f"Phase 3: Adam log-space, deadline={phase3_end-t_start:.0f}s")
    a_ad, sc_ad = _subgradient_adam(best_a, phase3_end, lr_start=0.004, active_frac=0.993, max_iter=300000)
    full_ad = _score(a_ad)
    print(f"  Adam: score={full_ad:.10f}")
    if full_ad < best_score:
        best_score = full_ad
        best_a = a_ad.copy()
    
    # Phase 4: Direct subgradient
    phase4_end = min(deadline, t_start + 1150.0)
    print(f"Phase 4: Direct subgradient, deadline={phase4_end-t_start:.0f}s")
    a_sg, sc_sg = _direct_subgradient(best_a, phase4_end, max_iter=30000, K=20, lr_start=0.008)
    full_sg = _score(a_sg)
    print(f"  Subgradient: score={full_sg:.10f}")
    if full_sg < best_score:
        best_score = full_sg
        best_a = a_sg.copy()
    
    # Phase 5: Active-set Newton again (smaller K, smaller step)
    phase5_end = min(deadline, t_start + 1350.0)
    print(f"Phase 5: Active-set Newton (fine), deadline={phase5_end-t_start:.0f}s")
    a_as2, sc_as2 = _active_set_newton(best_a, phase5_end, K=20, trust_radius=0.003, max_iter=5000)
    full_as2 = _score(a_as2)
    print(f"  Fine active-set: score={full_as2:.10f}")
    if full_as2 < best_score:
        best_score = full_as2
        best_a = a_as2.copy()
    
    # Phase 6: Coordinate descent + two-point swaps
    phase6_end = min(deadline, t_start + 1550.0)
    print(f"Phase 6: Coordinate + swaps, deadline={phase6_end-t_start:.0f}s")
    a_cd, sc_cd = _coordinate_refine(best_a, phase6_end, max_rounds=150, n_coords=400)
    full_cd = _score(a_cd)
    print(f"  Coordinate: score={full_cd:.10f}")
    if full_cd < best_score:
        best_score = full_cd
        best_a = a_cd.copy()
    
    # Two-point swaps
    if time.monotonic() < deadline - 100:
        a_sw, sc_sw = _two_point_swap(best_a, min(deadline - 80, time.monotonic() + 100), max_swaps=3000)
        full_sw = _score(a_sw)
        print(f"  Swaps: score={full_sw:.10f}")
        if full_sw < best_score:
            best_score = full_sw
            best_a = a_sw.copy()
    
    # Phase 7: Final Adam polish
    if time.monotonic() < deadline - 60:
        print("Phase 7: Final Adam polish")
        a_fin, sc_fin = _subgradient_adam(best_a, deadline - 30, lr_start=0.001, active_frac=0.99, max_iter=100000)
        full_fin = _score(a_fin)
        print(f"  Final Adam: score={full_fin:.10f}")
        if full_fin < best_score:
            best_score = full_fin
            best_a = a_fin.copy()
    
    # Phase 8: Quick random restarts
    if time.monotonic() < deadline - 50:
        print("Phase 8: Random restarts")
        for r in range(3):
            if time.monotonic() > deadline - 60:
                break
            sigma = 0.003 + 0.002 * r
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            # Quick active-set polish
            a_pert_opt, sc_pert = _active_set_newton(a_pert, min(deadline - 40, time.monotonic() + 40), K=30, trust_radius=0.005, max_iter=2000)
            full_pert = _score(a_pert_opt)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_pert:.10f}")
            if full_pert < best_score:
                best_score = full_pert
                best_a = a_pert_opt.copy()
    
    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
