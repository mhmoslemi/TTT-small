Let me also think about the **pairwise swap** more carefully:

Given a, c = conv(a,a), and the active set S:
- The "pressure" on coordinate i is p[i] = 2 * (1/|S|) * sum_{k in S} a[k-i]
- To reduce the peak, we want to move mass from high-pressure coordinates to low-pressure coordinates.
- Pick i with lowest p[i] and j with highest p[j]
- Transfer delta from j to i: a[i] += delta, a[j] -= delta
- The change in c[k] is: delta * (2*a[k-i] - 2*a[k-j])
- Find optimal delta: the largest delta such that max_k (c[k] + delta*(2*a[k-i] - 2*a[k-j])) is minimized

For the optimal delta:
- Let g[k] = 2*(a[k-i] - a[k-j])
- f(delta) = max_k (c[k] + delta * g[k])
- We want to minimize f(delta) over [0, a[j]]
- f is piecewise linear, convex
- The minimum is where the derivative changes sign (if it does), otherwise at a boundary

The derivative of f at delta is: max over k in argmax_k (c[k] + delta*g[k]) of g[k].
f is decreasing while all active k have g[k] < 0, and increasing while all have g[k] > 0.
The optimal delta is where the "switching" happens.

To find this efficiently:
- Sort peaks by (c[k] + delta*g[k]) / (-g[k]) for k where g[k] < 0 (these are the "covering" peaks)
- The optimal delta is the smallest such value that makes all "covering" peaks at least as high as all "rising" peaks

This is O(n log n) per swap, which for n=16384 is about 16384 * 14 ≈ 230K operations. In 400 seconds, I can do maybe 10K-50K swaps. That should be enough for fine-tuning.

Actually, for the optimal delta in a pairwise swap, I can use a simpler approach:
- Binary search on delta in [0, a[j]]
- For each delta, compute max_k (c[k] + delta * g[k]) in O(n)
- Total: O(n * log(1/eps)) per swap

For n=16384 and 50 iterations of bisection: 16384 * 50 ≈ 800K operations per swap. In 400s, maybe 5K-10K swaps. Should be fine.

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Assumes sum(a) = 1."""
    return float(2.0 * len(a) * _full_convolve(a, a).max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0))

    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)

def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """
    Direct (non-smoothed) active-set optimization.
    Uses subgradient of the max with adaptive step size.
    """
    n = len(a_init)
    a = (a_init / a_init.sum()).copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    tol_frac = 0.005  # active set: peaks within this fraction of max
    lr = 0.02
    no_improve = 0
    max_stall = 200
    
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Subgradient: g[i] = 2 * mean_{k in active} a[k-i]
        # This is 2 * conv(active/n_active, a_reversed)[n-1:2n-1]
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - grad.mean()  # Project to maintain sum=1
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Adaptive step size
        alpha = lr / (gnorm + 1e-15)
        alpha = min(alpha, 0.5)
        
        # Line search
        a_old_sc = current_sc
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(0.0001, tol_frac * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(0.00005, tol_frac * 0.3)
                lr = 0.01
                no_improve = 0
                if tol_frac < 0.00005 and lr < 0.001:
                    break
        else:
            if tol_frac > 0.0001:
                tol_frac = max(0.0001, tol_frac * 0.98)
            lr = min(0.05, lr * 1.01)
    
    return best_a, _score_normalized(best_a)

    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.004)

    betas = base_beta * np.array(
        [1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.025, 0.015,
         0.008, 0.005, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    )

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 12))))

        prob = _SmoProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score_norm(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"      L-BFGS beta={beta:.6g}: score={current_sc:.10f}, nit={result.nit}")
        except Exception:
            pass

    return current_a, _score_norm(current_a)


def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """
    Direct subgradient method on the exact max objective.
    Uses adaptive active set and step size.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)

    tol_frac = 0.004
    lr = 0.03
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Subgradient: g[i] = 2 * sum_{k in active} w_k * a[k-i]
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        alpha = min(lr / (gnorm + 1e-15), 0.5)

        improved = False
        a_old_sc = current_sc
        for _ in range(100):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol_frac = max(0.00005, tol_frac * 0.6)
                lr *= 0.5
            if no_improve >= 300:
                a = best_a.copy()
                tol_frac = max(0.00003, tol_frac * 0.3)
                lr = 0.005
                no_improve = 0
                if tol_frac < 0.00003 and lr < 0.002:
                    break
        else:
            if tol_frac > 0.00005:
                tol_frac = max(0.00005, tol_frac * 0.985)
            lr = min(0.08, lr * 1.005)

    return best_a, _score_norm(best_a)


def _pairwise_swap(a_init, deadline, min_time_remaining=30.0):
    """
    Gradient-guided pairwise mass transfer optimization.
    Moves mass from high-pressure to low-pressure coordinates.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)
    t_start = time.monotonic()
    swap_count = 0
    stall = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0

        # Compute pressure vector: p[i] = sum of a[k-i] over active peaks
        n_peaks = max(10, min(200, int(0.005 * n)))
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        p = np.zeros(n)
        for k in peak_idx:
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            idx_i = np.arange(i_lo, i_hi + 1)
            idx_j = k - idx_i
            p[idx_i] += a[idx_j]
            mask = idx_i != idx_j
            if mask.any():
                p[idx_i[mask]] += a[idx_j[mask]]

        # Find best swap: i with low p, j with high p
        nz_mask = a > 1e-8 * a.max()
        p_j = p.copy()
        p_j[~nz_mask] = np.inf
        j_candidates = np.argsort(p_j)[::-1][:20]
        i_candidates = np.argsort(p)[:20]

        best_delta_this_iter = 0.0
        best_sc_this_iter = np.inf
        best_i = -1
        best_j = -1

        for jc in j_candidates[:5]:
            if a[jc] < 1e-12:
                continue
            for ic in i_candidates[:5]:
                if ic == jc:
                    continue
                # g[k] = 2*(a[k-i] - a[k-j])
                g = np.zeros(2 * n - 1)
                # Add 2*a shifted by i
                lo_i = max(0, ic)
                hi_i = min(2 * n - 2, ic + n - 1)
                g[lo_i:hi_i + 1] += 2.0 * a[n - 1 - (hi_i - ic): n - 1 - (lo_i - ic)][::-1]
                # Subtract 2*a shifted by j
                lo_j = max(0, jc)
                hi_j = min(2 * n - 2, jc + n - 1)
                g[lo_j:hi_j + 1] -= 2.0 * a[n - 1 - (hi_j - jc): n - 1 - (lo_j - jc)][::-1]

                # Find optimal delta by trying candidates
                hi_d = a[jc]
                cands = hi_d * np.array([0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 0.9])
                best_d = 0.0
                best_f = np.inf
                for d in cands:
                    if d > hi_d:
                        break
                    f_val = float(np.max(c + d * g))
                    if f_val < best_f:
                        best_f = f_val
                        best_d = d

                if best_d > 0 and best_f < best_sc_this_iter:
                    best_sc_this_iter = best_f
                    best_delta_this_iter = best_d
                    best_i = ic
                    best_j = jc

        if best_i < 0:
            stall += 1
            if stall > 50:
                break
            continue

        # Apply best swap
        a[best_i] += best_delta_this_iter
        a[best_j] -= best_delta_this_iter
        a[best_j] = max(0.0, a[best_j])
        s_now = a.sum()
        if s_now < 1e-15:
            break
        a = a / s_now
        swap_count += 1
        stall = 0

        if swap_count % 500 == 0:
            sc_now = _score_norm(a)
            print(f"      Swap {swap_count}: cur={sc_now:.10f}, best={best_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    return best_a, _score_norm(best_a)


def _coord_descent(a_init, deadline, min_time_remaining=30.0):
    """
    Coordinate descent: for each coordinate, find optimal value via
    1D minimax of the convolution peaks.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(12345)
    stagnant = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Pick coordinates to optimize: those contributing to top peaks
        top_k = np.argsort(c)[::-1][:max(10, n // 100)]
        involved = set()
        for k in top_k:
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            involved.update(range(i_lo, i_hi + 1))
        involved_list = list(involved)

        # Also add some random coordinates
        n_rand = max(5, n // 50)
        involved_list.extend(rng.integers(0, n, size=n_rand).tolist())

        # Shuffle
        rng2 = np.random.default_rng(int(time.monotonic() * 1000000) % 100000)
        rng2.shuffle(involved_list)

        iters_this_round = 0
        for idx in involved_list:
            if time.monotonic() + min_time_remaining > deadline:
                break
            iters_this_round += 1

            # Try perturbing a[idx]
            # f(delta) = max_k (c[k] + delta * g[k]) where g[k] = 2*a[k-idx]
            # But we must maintain sum=1, so decreasing a[idx] by delta means
            # the rest must increase by delta/(n-1) each (uniform redistribution)
            # Simpler: just try a few multipliers
            
            base = a[idx]
            if base < 1e-15:
                # Try small increase
                candidates = [1e-5 / n, 1e-4 / n]
            else:
                candidates = base * np.array([0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.7])
                candidates = np.concatenate([[0.0], candidates])

            best_val = base
            best_f = current_sc
            for new_val in candidates:
                a_try = a.copy()
                delta = new_val - base
                a_try[idx] = new_val
                # Redistribute: scale others to maintain sum=1
                if delta != 0:
                    others_sum = 1.0 - new_val
                    if others_sum > 1e-15:
                        a_try[np.arange(n) != idx] *= others_sum / a_try[np.arange(n) != idx].sum()
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = 2.0 * n * float(c_try.max())
                if f_try < best_f - 1e-15:
                    best_f = f_try
                    best_val = new_val

            if best_val != base:
                delta = best_val - base
                a[idx] = best_val
                others_sum = 1.0 - best_val
                if others_sum > 1e-15:
                    mask = np.arange(n) != idx
                    a[mask] *= others_sum / a[mask].sum()
                s_now = a.sum()
                if s_now > 1e-15:
                    a = a / s_now

        if stagnant > 50 and iters_this_round > 100:
            break

    return best_a, _score_norm(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 19.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(777)
        a = np.abs(rng.standard_normal(n))
        # Smooth
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    a_sorted = np.argsort(a)
    for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score_norm(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted = np.argsort(a)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS + subgradient + pairwise swaps + coord descent
    + shrinkage + multi-n. Designed to beat 1.5057.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score_norm(best_seq)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    def _update_best(a, sc, tag):
        nonlocal best_seq, best_score
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} n={len(a)} [{tag}] t={time.monotonic()-t_start:.0f}s")

    # === Phase 1: L-BFGS from parent (200s) ===
    print(f"\n[P1] L-BFGS from parent (t={time.monotonic()-t_start:.0f}s)")
    p1_dead = min(deadline - 1400, t_start + 200)
    a1, sc1 = _run_lbfgs(best_seq, p1_dead, min_time_remaining=50)
    _update_best(a1, sc1, "lbfgs_parent")

    # === Phase 1b: L-BFGS from perturbed parent (100s) ===
    print(f"\n[P1b] L-BFGS from perturbations (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() < deadline - 1200:
        for sigma in [0.3, 0.15]:
            if time.monotonic() > deadline - 1200:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            p1b_dead = min(deadline - 1200, time.monotonic() + 60)
            a_bp, sc_bp = _run_lbfgs(a_pert, p1b_dead, min_time_remaining=30)
            _update_best(a_bp, sc_bp, f"lbfgs_pert_{sigma}")

    # === Phase 2: Active-set subgradient refinement (400s) ===
    print(f"\n[P2] Active-set subgradient (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() < deadline - 800:
        p2_dead = min(deadline - 800, time.monotonic() + 400)
        a2, sc2 = _active_set_refine(best_seq, p2_dead, min_time_remaining=60)
        _update_best(a2, sc2, "active_set")

    # === Phase 3: Pairwise swap optimization (350s) ===
    print(f"\n[P3] Pairwise swaps (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() < deadline - 450:
        p3_dead = min(deadline - 450, time.monotonic() + 350)
        a3, sc3 = _pairwise_swap(best_seq, p3_dead, min_time_remaining=40)
        _update_best(a3, sc3, "pairwise")

    # === Phase 4: Coordinate descent (250s) ===
    print(f"\n[P4] Coordinate descent (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() < deadline - 250:
        p4_dead = min(deadline - 250, time.monotonic() + 250)
        a4, sc4 = _coord_descent(best_seq, p4_dead, min_time_remaining=30)
        _update_best(a4, sc4, "coord_desc")

    # === Phase 5: Active-set round 2 on best (200s) ===
    print(f"\n[P5] Active-set round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() < deadline - 150:
        p5_dead = min(deadline - 150, time.monotonic() + 200)
        a5, sc5 = _active_set_refine(best_seq, p5_dead, min_time_remaining=30)
        _update_best(a5, sc5, "active_set2")

    # === Phase 6: Shrinkage + re-optimize (150s) ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() < deadline - 100:
        a_sh, sc_sh = _shrinkage(best_seq, deadline - 80)
        _update_best(a_sh, sc_sh, "shrink")
        if time.monotonic() < deadline - 60:
            a_sh2, sc_sh2 = _active_set_refine(best_seq, min(deadline - 40, time.monotonic() + 60),
                                                min_time_remaining=20)
            _update_best(a_sh2, sc_sh2, "post_shrink")

    # === Phase 7: Multi-n exploration (200s) ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.0f}s)")
    n_options = [32768, 24576, 12288, 6144]
    if n_parent not in n_options:
        n_options.insert(0, n_parent)

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        if n_try == len(best_seq):
            continue
        init_time = deadline - time.monotonic()
        if init_time < 80:
            break
        a_rs = _resample(best_seq, n_try)
        sc_rs = _score_norm(a_rs)
        print(f"    n={n_try}: resampled score={sc_rs:.10f}")

        # Quick L-BFGS
        p7_dead = min(deadline - 40, time.monotonic() + min(100, init_time * 0.5))
        a_7, sc_7 = _run_lbfgs(a_rs, p7_dead, min_time_remaining=30)
        _update_best(a_7, sc_7, f"multi_n_{n_try}")

        # Quick active-set
        if time.monotonic() < deadline - 30:
            p7b_dead = min(deadline - 20, time.monotonic() + min(50, init_time * 0.3))
            a_7b, sc_7b = _active_set_refine(a_7, p7b_dead, min_time_remaining=15)
            _update_best(a_7b, sc_7b, f"multi_n_{n_try}_refine")

    # === Phase 8: Final ultra-fine polish (remaining time) ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.0f}s, best={best_score:.10f})")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fin, sc_fin = _active_set_refine(
            best_seq, deadline - 5,
            min_time_remaining=5
        )
        _update_best(a_fin, sc_fin, "final")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
