def _projected_subgrad(a, deadline, max_iters=5000, top_k=5, lr=0.02, momentum=0.9):
    """Projected subgradient descent with Nesterov momentum on normalized a."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)

    z = a.copy()
    v = np.zeros(n)
    patience = 0
    max_patience = 200

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it % 50 == 0 and patience > max_patience:
            break

        a_cur = _normalize(z)
        sg = _subgradient_max(a_cur, n, top_k=top_k)
        # Center the gradient (remove mean to preserve sum)
        sg = sg - sg.mean()
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        step = lr / (1.0 + 0.01 * it)
        v = momentum * v + (1 - momentum) * (-step * sg / sg_norm)
        z = z + v
        z = np.maximum(z, 0.0)

        a_cur = _normalize(z)
        sc = _score(a_cur)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_cur.copy()
            patience = 0
        else:
            patience += 1

    return best_a, best_sc

def _smoothed_max_grad(a, n, beta):
    """Compute gradient of smooth max: beta * log(sum(exp((c - c_max)/beta))) + c_max."""
    c = _full_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    a_rev = a[::-1]
    wc = _full_conv(w, a_rev)
    # gradient w.r.t. a: 2 * (w_correlate_a)
    # Using the same formula as _subgradient_max but with general weights
    N = _next_pow2(3 * n)
    Fw = np.fft.rfft(w, N)
    Fa = np.fft.rfft(a, N)
    R = np.fft.irfft(Fw * np.conj(Fa), N)[:2 * n - 1]
    grad_c = 2.0 * R[:n]

    # Total gradient of smooth_max = c_max_derivative + beta * log(s) part
    # d/di [c_max + beta*log(sum exp((c_k - c_max)/beta))]
    # = (1 - sum_k w_k) * dc_max/da_i + beta * sum_k w_k * (1/beta) * dc_k/da_i
    # = (1-1)*dc_max/da_i + sum_k w_k * dc_k/da_i  (since sum w_k = 1)
    # = sum_k w_k * dc_k/da_i
    # So the gradient is just the weighted sum of individual gradients:
    grad = grad_c
    # Center it
    grad = grad - grad.mean()
    # Multiply by 2n / sum(a)^2 for the actual score gradient
    s_a = a.sum()
    grad = grad * (2.0 * n / (s_a * s_a))
    return grad, c_max, w

        wc = _full_conv(w, a_rev)
        N = _next_pow2(3 * n)
        Fw = np.fft.rfft(w, N)
        Fa = np.fft.rfft(a_local, N)
        R = np.fft.irfft(Fw * np.conj(Fa), N)[:2 * n - 1]
        grad_c = 2.0 * R[:n]
        # Chain rule: d/dz = d/da * da/dz
        # a = exp(z - max(z)) / sum(exp(z - max(z)))
        # da/dz_i = a_i * (1 - a_i) (for softmax-like, but we normalize)
        # Actually a = softmax(z) approximately (since we normalize)
        # da_i/dz_j = a_i*(delta_ij - a_j)
        # So grad_z = a * (grad_c - sum(a * grad_c))
        grad_z = a_local * (grad_c - np.dot(a_local, grad_c))
        return 2.0 * n * grad_z

    beta_curr = betas[0]
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta_curr = beta
        max_iter = min(3000, max(30, int(remaining / 0.05)))

        try:
            result = minimize(
                obj, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _normalize(np.exp(result.x - result.x.max()))
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def _water_filling(a, deadline, max_iters=500):
    """Iterative water-filling: equalize top correlation taps."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_conv(a, a)
        c_max = float(c.max())
        c_mean = float(c.mean())

        # Identify taps above target
        target = c_mean * 1.5  # try to keep max below 1.5 * mean
        excess = np.maximum(c - target, 0)
        if excess.max() < 1e-15:
            break

        # Gradient of sum(excess^2) w.r.t. a
        N = _next_pow2(3 * n)
        Fe = np.fft.rfft(excess, N)
        Fa = np.fft.rfft(a, N)
        R = np.fft.irfft(Fe * np.conj(Fa), N)[:2 * n - 1]
        grad = 2.0 * R[:n]
        grad = grad - grad.mean()
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        alpha = 0.05 / (1.0 + 0.1 * it)
        for ls in range(15):
            a_try = _normalize(a - alpha * grad)
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break
            alpha *= 0.6

    return best_a, best_sc


def _direct_cg_solve(W, n, rhs, max_iter=500):
    """CG solve for H*x = rhs where H is Hankel with symbol W."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        # H*p: Hankel matvec
        V = p[::-1]
        out_len = n + len(W) - 1
        N = _next_pow2(out_len)
        conv = np.fft.irfft(np.fft.rfft(V, N) * np.fft.rfft(W, N), N)[:out_len]
        Hp = 2.0 * conv[n - 1:2 * n - 1]
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30:
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if rs_new < 1e-24 * rs0:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _qp_round(a, deadline, max_rounds=50):
    """Active-set QP: minimize a^T H a s.t. sum(a)=1, a>=0."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_conv(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max * (0.95 ** rnd)
        active = c >= c_max - tol
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()

        # Solve: min (1/2) a^T (2*H_W) a s.t. sum(a)=1, a>=0
        # KKT: 2*H_W*a = lambda*1 - mu, mu_i >= 0, a_i*mu_i = 0
        # Try: assume all a_i > 0: H_W*a = (lambda/2)*1
        y = _direct_cg_solve(W, n, np.ones(n), max_iter=300)
        oney = float(y.sum())
        if abs(oney) > 1e-20:
            a_new = y / oney
            if (a_new >= -1e-10).all():
                a_new = np.maximum(a_new, 0)
                a_new = _normalize(a_new)
                sc_new = _score(a_new)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new
                continue

        # Fallback: projected gradient
        grad = np.zeros(n)
        V = a[::-1]
        out_len = n + len(W) - 1
        N = _next_pow2(out_len)
        conv = np.fft.irfft(np.fft.rfft(V, N) * np.fft.rfft(W, N), N)[:out_len]
        grad = 2.0 * conv[n - 1:2 * n - 1]
        grad = grad - grad.mean()
        step = 0.5 / (np.max(np.abs(grad)) + 1e-10)
        a_new = _normalize(a - step * grad)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flattop":
        t = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5 + 0.5 * np.cos(np.pi * t) ** 4
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 30.0
    rng = np.random.default_rng(seed % (2**63 - 1))

    best_seq = None
    best_sc = np.inf

    def update(p, label=""):
        nonlocal best_seq, best_sc
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_seq = p.copy()
            print(f"  BEST: {best_sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t0:.0f}s")

    # Load parent
    parent = None
    try:
        pr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if pr.size > 0:
            pr = np.clip(pr, 0, 1000)
            if pr.sum() > 0.01:
                parent = pr / pr.sum()
                print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
                update(parent, "parent")
    except Exception:
        pass

    if parent is None:
        n_init = 16384
        parent = _make_profile(n_init, "tapered")
        print(f"No parent. Using tapered n={n_init}, score={_score(parent):.10f}")
        update(parent, "init_tapered")

    n_cur = len(parent)

    # Phase 1: L-BFGS smoothed max
    print(f"\n[Phase 1] L-BFGS smoothed max (n={n_cur})")
    if time.monotonic() < deadline - 600:
        a1, s1 = _lbfgs_smoothed(parent, deadline - 600, min_time_remaining=300)
        update(a1, "lbfgs_1")

    # Phase 2: Active set flatten
    print(f"\n[Phase 2] Active set flatten")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, s2 = _active_set_flatten(best_seq, deadline - 400, max_rounds=150)
        update(a2, "active_flatten")

    # Phase 3: Water filling
    print(f"\n[Phase 3] Water filling")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a3, s3 = _water_filling(best_seq, deadline - 300, max_iters=300)
        update(a3, "water_fill")

    # Phase 4: QP rounds
    print(f"\n[Phase 4] QP active set")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, s4 = _qp_round(best_seq, deadline - 250, max_rounds=40)
        update(a4, "qp")

    # Phase 5: Projected subgradient
    print(f"\n[Phase 5] Projected subgradient")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, s5 = _projected_subgrad(best_seq, deadline - 200, max_iters=3000, top_k=8)
        update(a5, "subgrad")

    # Phase 6: Another L-BFGS round (tighter)
    print(f"\n[Phase 6] L-BFGS round 2")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a6, s6 = _lbfgs_smoothed(best_seq, deadline - 150, min_time_remaining=80)
        update(a6, "lbfgs_2")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n exploration")
    for n_try in [2048, 4096, 8192, 32768]:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_cur:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt, s_opt = _lbfgs_smoothed(a_init, deadline - 80, min_time_remaining=40)
        update(a_opt, f"n={n_try}")

    # Phase 8: Restarts from different profiles
    print(f"\n[Phase 8] Diverse restarts")
    for prof in ["flattop", "bump", "uniform"]:
        if time.monotonic() > deadline - 50:
            break
        n_r = len(best_seq) if best_seq is not None else 8192
        a_init = _make_profile(n_r, prof)
        # Add small perturbation
        a_init = _normalize(a_init + 0.01 * rng.standard_normal(n_r))
        a_opt, s_opt = _lbfgs_smoothed(a_init, deadline - 30, min_time_remaining=15)
        update(a_opt, f"restart_{prof}")

    # Phase 9: Final polish
    print(f"\n[Phase 9] Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fs, s_fs = _active_set_flatten(best_seq, deadline - 3, max_rounds=100)
        update(a_fs, "final_flatten")
        a_wf, s_wf = _water_filling(best_seq, deadline - 2, max_iters=100)
        update(a_wf, "final_water")

    # Final
    print(f"\n{'='*50}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(8192, "uniform")
    print(f"FALLBACK: uniform n=8192, score={_score(fb):.10f}")
    return [float(x) for x in fb]
