def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_objective_and_grad(z, p, n):
    """Compute L_p^p norm of conv(softmax(z), softmax(z)) and its gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)

        obj = float(c.max())
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        obj = float(np.sum(np.power(c_safe, p)))
        w = p * np.power(c_safe, p - 1)

    W = np.dot(g, a)

    return obj, grad_z

            w[active] = 1.0 / n_active
        else:
            w = excess / total_excess
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        
        if g_norm < 1e-14:
            break
        
        # Momentum step
        alpha = lr / (g_norm + 1e-15)
        v = momentum * v + alpha * g
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            v = np.zeros(n)
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
        a = a_new / s_new
        
        # Adaptive adjustments
        if it % 200 == 0:
            current_sc2 = _score(a)
            if current_sc2 < best_sc - 1e-14:
                active_frac = min(0.05, active_frac * 1.2)
                lr = min(1.0, lr * 1.05)
            else:
                active_frac = max(0.001, active_frac * 0.8)
                lr = max(0.01, lr * 0.85)
                momentum = max(0.3, momentum * 0.9)
    
    return best_a, best_sc


def _nesterov_mirror_descent(a_init, deadline, min_remaining=10.0):
    """Nesterov-accelerated mirror descent in log domain."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    
    active_frac = 0.02
    t = 1
    max_iter = 3000
    
    for it in range(max_iter):
        if time.monotonic() + min_remaining > deadline:
            break
        t += 1
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        # Nesterov extrapolation
        z_y = z + (1.0 / beta_t) * (z - z_prev)
        a_y = _softmax(z_y)
        
        # Compute gradient at y
        c = _full_convolve(a_y, a_y)
        c_max = float(c.max())
        
        threshold = c_max * (1.0 - active_frac)
        active = c >= threshold
        n_active = max(1, int(active.sum()))
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        
        a_y_rev = a_y[::-1]
        wc = _full_convolve(w, a_y_rev)
        h = wc[n - 1:2 * n - a_y.size]
        g = 2.0 * h
        g = g - np.mean(g)
        
        # Mirror descent step
        lr_t = 1.0 / (np.sqrt(t) * (np.linalg.norm(g) + 1e-15))
        z_new = z_y - lr_t * g
        a_new = _softmax(z_new)
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
        
        z_prev = z.copy()
        z = z_new
        
        if it % 500 == 0:
            active_frac = min(0.1, active_frac * 1.3)
    
    return best_a, best_sc


def _lbfgs_lse(a_init, deadline, min_remaining=30.0):
    """Standard L-BFGS with log-sum-exp smoothing (baseline for comparison)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        def obj_fn(zz, beta=beta, n=n):
            a = _softmax(zz)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def grad_fn(zz, beta=beta, n=n):
            a = _softmax(zz)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            return 4.0 * n * a * (h - W)
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(20, int(remaining / (est_time * 12))))
        
        try:
            result = minimize(obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-13, 'maxcor': 60, 'maxls': 40})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 3.0 * t / n) + \
            0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_center":
        t = np.arange(n) / (n - 1)
        a = 1.0 - 0.3 * (2 * t - 1) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    """Try zeroing out the smallest elements to see if it helps."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):

    except Exception:
