"""
Optimized step function search minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp L-BFGS with aggressive beta continuation,
diverse restarts, QP-based IRLS for medium n, and Newton active-set polish.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z, grad=False):
        if not grad:
            return _smoobj(z, self.beta, self.n)
        return _smoobj(z, self.beta, self.n), _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=80):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        active = None
        for tol_frac in [1e-5, 5e-5, 1e-4, 5e-4, 2e-3, 1e-2]:
            tol = tol_frac * c_max
            candidate = c >= (c_max - tol)
            if candidate.sum() >= 1:
                active = candidate
                break
        if active is None:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        n_active = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-12 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.003 * alpha) - 1e-14:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    stall_count = 0

    while beta > min_beta:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.004 * scale
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 15))))

        adapter = _SmoAdapter(beta, n)
        try:
            result = minimize(
                adapter, z, jac=True,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 2:
                    z = z + 0.01 * np.random.default_rng(int(time.monotonic()*1e6)).standard_normal(n)
                    z = z - z.mean()
                    stall_count = 0
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.12
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.18
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.45

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = (np.linspace(0, 1, n) - 0.5)
        a = np.exp(-4.0 * t * t)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.5) ** 0.7
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "double_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4 * (1.0 + 0.3 * np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = n // 8
        if taper > 2:
            for i in range(taper):
                frac = (i + 1) / taper
                a[i] = frac ** 1.5
                a[n - 1 - i] = frac ** 1.5
        return a / a.sum()
    elif kind == "optimal_like":
        t = np.linspace(0, 1, n)
        a = (1 + np.cos(np.pi * t)) ** 1.2 * (1 + 0.1 * np.sin(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _qp_irls_step(a, beta, n):
    """One IRLS step: compute weights from softmax of conv, solve QP via projected grad."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return a
    w = e / s

    # Solve min a^T H_w a s.t. sum(a)=1, a>=0 using projected gradient
    # H_w * a = _hankel_matvec(a, w, n) / 2 (since _hankel returns 2*conv)
    # Actually _hankel_matvec(v, W, n) returns 2 * (W * v_rev)[n-1:2n-1]
    # And a^T H a = 2 * sum_k W_k * (a*a)_k = sum_k W_k * c_k * 2... 
    # Let me just use the gradient: grad = 4*n*a*(h - W_val) but for the QP it's different
    # For QP: minimize a^T H_w a where (H_w)_{ij} = w_{i+j}
    # grad of a^T H_w a = 2 * H_w * a
    # H_w * a at index i = sum_j w_{i+j} * a_j
    # This is the "autocorrelation" of w with a, restricted to valid range
    
    # Use projected gradient with BB step size
    x = a.copy()
    s_prev = None
    g_prev = None
    
    for pg_iter in range(50):
        # Compute gradient of a^T H_w a: = 2 * (H_w a)
        # (H_w a)_i = sum_j w_{i+j} a_j = (w * a_rev)_{n-1+i} ... 
        # Actually: (w * a)_k = sum_j w_j a_{k-j}, so (H_w a)_i = sum_j w_{i+j} a_j
        # Let's compute via FFT: reverse w, convolve with a
        w_rev = w[::-1]
        conv_wa = _full_convolve(w_rev, a)
        # (H_w a)_i = sum_j w_{i+j} a_j = for i in 0..n-1, this is conv_wa at index...
        # w_rev[m] = w[n-1-m-1]... let me think more carefully
        # _full_convolve(w_rev, a)[k] = sum_j w_rev[j] * a[k-j] = sum_j w[n-1-j] * a[k-j]
        # Let m = n-1-j, then j = n-1-m: sum_m w[m] * a[k-n+1+m]
        # For (H_w a)_i = sum_j w_{i+j} a_j, substitute the above with appropriate indices
        # This is getting complex. Let me use _hankel_matvec directly.
        grad = _hankel_matvec(x, w, n)  # This gives 2*(w*x_rev)[n-1:2n-1] which is 2*H_w*x
        
        # Projected gradient step with BB step size
        if s_prev is not None and g_prev is not None:
            ys = float(x - s_prev)
            yg = float(grad - g_prev)
            if abs(yg) > 1e-30:
                step = float(np.dot(ys, ys)) / max(abs(float(np.dot(ys, yg))), 1e-30)
                step = min(max(step, 1e-10), 10.0 / (float(np.linalg.norm(grad)) + 1e-30))
            else:
                step = 1.0 / (float(np.linalg.norm(grad)) + 1e-30)
        else:
            step = 1.0 / (float(np.linalg.norm(grad)) + 1e-30)
        
        s_prev = x.copy()
        g_prev = grad.copy()
        
        x = x - step * grad
        # Project onto simplex {a >= 0, sum = 1}
        # Use bisection for the simpler projection
        x = np.maximum(x, 0)
        s = x.sum()
        if s > 0:
            x = x / s
        else:
            x = a.copy()
            break
    
    return x

        update_best(parent, _score(parent), "init")

    # Phase 1: Quick profile exploration
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic()-t_start:.1f}s)")
    profiles = ["tapered", "raised_cos", "gaussian", "asym", "double_taper", "plateau", "optimal_like"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for pk in profiles:
        if time.monotonic() > deadline - 800:
            break
        a_p = _make_profile(n_parent, pk)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{pk}': {sc_p:.10f} (new best profile)")

    # Quick L-BFGS on best profile (short)
    if time.monotonic() < deadline - 700:
        a_q = _run_lbfgs(best_profile_a, min(deadline - 700, time.monotonic() + 90), min_remaining=600.0)
        sc_q = _score(a_q)
        update_best(a_q, sc_q, "quick_profile")
        if sc_q < best_profile_sc:
            best_profile_a = a_q
            best_profile_sc = sc_q

    # Phase 2: Main L-BFGS optimization
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_profile_a
    a_main = _run_lbfgs(start_a, deadline, min_remaining=600.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 3: Restart from perturbed best + different profiles
    print(f"\n[Phase 3] Restarts (t={time.monotonic()-t_start:.1f}s)")
    restart_configs = [
        ("perturb_01", lambda: _perturb(best_seq, 0.03, rng)),
        ("perturb_03", lambda: _perturb(best_seq, 0.1, rng)),
        ("gaussian", lambda: _make_profile(n_parent, "gaussian")),
        ("optimal_like", lambda: _make_profile(n_parent, "optimal_like")),
    ]
    for label, init_fn in restart_configs:
        if time.monotonic() > deadline - 200:
            break
        a_r = init_fn()
        a_opt = _run_lbfgs(a_r, deadline, min_remaining=150.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"restart_{label}")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_parent * 2, n_parent // 2, n_parent * 4]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try < 512 or n_try > 131072:
            continue
        a_init_n = _resample(best_seq, n_try)
        a_opt_n = _run_lbfgs(a_init_n, deadline, min_remaining=80.0)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")

    # Phase 5: Subgradient + Newton hybrid
    print(f"\n[Phase 5] Subgradient + Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 150, time.monotonic() + 100))
        update_best(a_sg, sc_sg, "subgrad")
        if time.monotonic() < deadline - 50:
            a_np, sc_np = _newton_active_polish(a_sg, min(deadline - 50, time.monotonic() + 60), max_rounds=50)
            update_best(a_np, sc_np, "subgrad_newton")

    # Phase 6: Newton polish on best
    print(f"\n[Phase 6] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 100, time.monotonic() + 120), max_rounds=100)
        update_best(a_np, sc_np, "newton_main")

    # Phase 7: Another L-BFGS round from Newton result
    print(f"\n[Phase 7] L-BFGS after Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_final_lb = _run_lbfgs(best_seq, min(deadline - 60, time.monotonic() + 80), min_remaining=30.0)
        update_best(a_final_lb, _score(a_final_lb), "lbfgs_final")

    # Phase 8: Final Newton
    print(f"\n[Phase 8] Final Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=60)
        update_best(a_fin, sc_fin, "final_newton")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
