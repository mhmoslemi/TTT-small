"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses QP-based alternating optimization (new), combined with smoothed L-BFGS
and Newton active-set polish. The QP approach solves min a^T Q a subject to
sum(a)=1, a>=0 where Q = sum_k w_k B_k, enabling aggressive "best response"
updates that go beyond gradient steps.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _q_matvec(v, W, n):
    """Compute (Q v) where Q[i,j] = W[i+j]. No factor of 2."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _grad_weighted(v, W, n):
    """Compute gradient of sum_k W[k]*c_k w.r.t. v: 2*(Q v)."""
    return 2.0 * _q_matvec(v, W, n)


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    """Solve (Q)x = rhs using conjugate gradient, where Q is defined by W."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Qp = _q_matvec(p, W, n)
        pHp = float(p @ Qp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            # Add ridge for numerical stability
            pHp = pHp + 1e-10
        if pHp <= 0:
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Qp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _cg_solve_ridge(W, n, rhs, max_iter=800, tol=1e-14, ridge=1e-10):
    """CG with explicit ridge for robustness."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Qp = _q_matvec(p, W, n) + ridge * p
        pHp = float(p @ Qp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp <= 1e-30:
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Qp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.0:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _qp_solve(W, n, active_mask=None, max_iter=800):
    """Solve min a^T Q a s.t. sum(a)=1, a>=0 on active set.
    Returns solution on full space (zeros on inactive)."""
    if active_mask is None:
        active_mask = np.ones(n, dtype=bool)
    idx = np.where(active_mask)[0]
    n_act = len(idx)
    if n_act == 0:
        return np.ones(n) / n
    if n_act == 1:
        a = np.zeros(n)
        a[idx[0]] = 1.0
        return a

    # Restrict W to active indices
    # (Q_restricted)[i,j] = W[idx[i] + idx[j]]
    # We need to solve Q_restricted * x = ones(n_act)
    # Then scale so sum = 1, and check non-negativity

    # Build restricted W product from original W
    # Q_restricted[i,j] = W[idx[i]+idx[j]]
    # For CG, we need matvec: (Q_restricted v)[i] = sum_j W[idx[i]+idx[j]] * v[j]

    # We can compute this by: for each i, sum over j of W[idx[i]+idx[j]]*v[j]
    # This is like a "gather" operation on W

    # Simpler approach: if active set is the full set, use _cg_solve directly
    if n_act == n:
        ones = np.ones(n)
        x = _cg_solve_ridge(W, n, ones, max_iter=max_iter)
        s = x.sum()
        if s < 1e-30 or not np.isfinite(s):
            return np.ones(n) / n
        a = x / s
        a = np.maximum(a, 0)
        s2 = a.sum()
        if s2 < 1e-30:
            return np.ones(n) / n
        return a / s2

    # For restricted active sets, do direct matvec
    # (Q_act v)[i] = sum_j W[idx[i]+idx[j]] * v[j]
    # Build the matrix W_act (n_act x n_act) if n_act is small enough
    if n_act <= 2048:
        W_act = np.array([[W[idx[i] + idx[j]] for j in range(n_act)] for i in range(n_act)])
        W_act = (W_act + W_act.T) / 2.0
        ones = np.ones(n_act)
        try:
            x = np.linalg.solve(W_act + 1e-10 * np.eye(n_act), ones)
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(W_act + 1e-10 * np.eye(n_act), ones, rcond=None)[0]
        s = x.sum()
        if s < 1e-30:
            a = np.zeros(n)
            a[idx] = 1.0 / n_act
            return a
        a_full = np.zeros(n)
        a_full[idx] = x / s
        a_full = np.maximum(a_full, 0)
        s2 = a_full.sum()
        if s2 < 1e-30:
            a = np.zeros(n)
            a[idx] = 1.0 / n_act
            return a
        return a_full / s2
    else:
        # Large active set: use iterative approach
        # Define matvec for restricted Q
        def matvec_restricted(v):
            # (Q_act v)[i] = sum_j W[idx[i]+idx[j]] * v[j]
            # Use the trick: expand to full space, use _q_matvec, restrict
            v_full = np.zeros(n)
            v_full[idx] = v
            result_full = _q_matvec(v_full, W, n)
            return result_full[idx]

        ones = np.ones(n_act)
        x = np.zeros(n_act)
        r = ones.copy()
        p = r.copy()
        rs = float(r @ r)
        rs0 = rs
        for i in range(max_iter):
            Qp = matvec_restricted(p) + 1e-10 * p
            pHp = float(p @ Qp)
            if abs(pHp) < 1e-30 or not np.isfinite(pHp) or pHp <= 0:
                break
            alpha = rs / pHp
            x_new = x + alpha * p
            r_new = r - alpha * Qp
            rs_new = float(r_new @ r_new)
            if not np.isfinite(rs_new) or rs_new > rs:
                break
            x = x_new
            r = r_new
            p = (rs_new / rs) * p + r
            rs = rs_new
            if rs < 1e-28 * rs0:
                break

        a_full = np.zeros(n)
        s = x.sum()
        if s < 1e-30:
            a_full[idx] = 1.0 / n_act
            return a_full
        a_full[idx] = x / s
        a_full = np.maximum(a_full, 0)
        s2 = a_full.sum()
        if s2 < 1e-30:
            a_full[idx] = 1.0 / n_act
            return a_full
        return a_full / s2


def _alternating_qp(a_init, deadline, max_rounds=200, tol_fracs=None, min_time=30.0):
    """Alternating QP optimization: fix w, solve QP for a; fix a, update w."""
    if tol_fracs is None:
        tol_fracs = [0.01, 0.005, 0.001, 0.0005, 0.0001, 0.00005, 0.00001]

    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - min_time:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Choose tolerance (progressively tighter)
        tol_idx = min(rnd // 20, len(tol_fracs) - 1)
        tol = tol_fracs[tol_idx] * c_max

        # Active set
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Weighted active set (uniform over active)
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve QP: min a^T Q a s.t. sum(a)=1, a>=0
        a_qp = _qp_solve(W, n)

        # Line search: blend old and new
        best_alpha = 0.0
        best_obj = c_max
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_try = (1 - alpha) * a + alpha * a_qp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < best_obj:
                best_obj = c_max_try
                best_alpha = alpha

        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * a_qp
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a


def _newton_active_polish(a_init, deadline, max_rounds=150, min_time=30.0):
    """Newton polish with progressively tighter active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.001, 0.0005, 0.0001, 0.00005, 0.00001, 0.000005, 0.000001, 0.0000005]
    tol_idx = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - min_time:
            break
        # Progressively tighten tolerance
        if rnd > 0 and rnd % 15 == 0 and tol_idx < len(tol_schedule) - 1:
            tol_idx += 1

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[tol_idx] * c_max

        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute gradient: 2 * Q * a
        g = _grad_weighted(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # CG for H^{-1} g and H^{-1} 1
        Hg = _cg_solve_ridge(W, n, g, max_iter=500, ridge=1e-12)
        H1 = _cg_solve_ridge(W, n, np.ones(n), max_iter=500, ridge=1e-12)

        denom = float(np.ones(n) @ H1)
        if abs(denom) < 1e-30:
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-30:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_init=None):
    """Adaptive L-BFGS on smoothed objective with geometric beta schedule."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    if beta_init is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
        beta_init = max(gap * 0.3, max_c * 0.002)

    beta = beta_init
    min_beta = max_c * 1e-10

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        max_iter = min(6000, max(20, int(remaining / (est_iter_time * 30))))

        try:
            class _ObjFn:
                def __init__(self, b, nn):
                    self.b = b
                    self.nn = nn
                def __call__(self, zz):
                    return _smoobj(zz, self.b, self.nn)

            class _GradFn:
                def __init__(self, b, nn):
                    self.b = b
                    self.nn = nn
                def __call__(self, zz):
                    return _smograd(zz, self.b, self.nn)

            obj_fn = _ObjFn(beta, n)
            grad_fn = _GradFn(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        if new_gap < 1e-9 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.0 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Load parent and initialize
    print("=" * 60)
    print(f"Phase 1: Initialization (t=0s)")
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"  Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"  No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init_default")

    start_n = len(parent)

    # Phase 2: Quick L-BFGS to warm up
    print(f"\nPhase 2: Quick L-BFGS warm-up (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_warm = _run_adaptive_lbfgs(parent, min(deadline - 600, time.monotonic() + 180),
                                      min_time_remaining=500.0, beta_init=None)
        update_best(a_warm, _score(a_warm), "lbfgs_warm")

    # Phase 3: QP-based alternating optimization (NEW)
    print(f"\nPhase 3: QP alternating (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_qp = _alternating_qp(best_seq, min(deadline - 500, time.monotonic() + 300),
                                max_rounds=100, min_time=480.0)
        update_best(a_qp, _score(a_qp), "qp_1")

    # Phase 4: Newton polish from QP result
    print(f"\nPhase 4: Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 200),
                                             max_rounds=100, min_time=380.0)
        update_best(a_np, sc_np, "newton_1")

    # Phase 5: Fine L-BFGS
    print(f"\nPhase 5: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        c_best = _full_convolve(best_seq, best_seq)
        max_c_best = float(c_best.max())
        a_lb = _run_adaptive_lbfgs(best_seq, min(deadline - 300, time.monotonic() + 200),
                                    min_time_remaining=280.0, beta_init=max_c_best * 0.0005)
        update_best(a_lb, _score(a_lb), "lbfgs_fine")

    # Phase 6: More QP alternating with tighter tolerance
    print(f"\nPhase 6: QP alternating tight (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_qp2 = _alternating_qp(best_seq, min(deadline - 200, time.monotonic() + 150),
                                 max_rounds=80,
                                 tol_fracs=[0.0005, 0.0001, 0.00005, 0.00001, 0.000005, 0.000001, 0.0000005],
                                 min_time=180.0)
        update_best(a_qp2, _score(a_qp2), "qp_tight")

    # Phase 7: Intensive Newton polish
    print(f"\nPhase 7: Intensive Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_np2, sc_np2 = _newton_active_polish(best_seq, min(deadline - 150, time.monotonic() + 130),
                                               max_rounds=200, min_time=130.0)
        update_best(a_np2, sc_np2, "newton_2")

    # Phase 8: Ultra-fine L-BFGS
    print(f"\nPhase 8: Ultra-fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        c_best = _full_convolve(best_seq, best_seq)
        max_c_best = float(c_best.max())
        a_ultra = _run_adaptive_lbfgs(best_seq, min(deadline - 80, time.monotonic() + 70),
                                       min_time_remaining=70.0, beta_init=max_c_best * 1e-6)
        update_best(a_ultra, _score(a_ultra), "lbfgs_ultra")

    # Phase 9: Multi-n exploration
    print(f"\nPhase 9: Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 32768]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        if n_try == len(best_seq):
            continue
        a_init_n = _resample(best_seq, n_try)
        # Quick QP step
        if time.monotonic() < deadline - 25:
            a_qp_n = _alternating_qp(a_init_n, min(deadline - 25, time.monotonic() + 20),
                                      max_rounds=30, min_time=10.0)
            update_best(a_qp_n, _score(a_qp_n), f"qp_n{n_try}")
        # Quick Newton
        if time.monotonic() < deadline - 10:
            a_np_n, sc_np_n = _newton_active_polish(
                best_seq if len(best_seq) == n_try else a_init_n,
                min(deadline - 10, time.monotonic() + 10),
                max_rounds=30, min_time=5.0)
            update_best(a_np_n, sc_np_n, f"newton_n{n_try}")

    # Phase 10: Final ultra-tight Newton
    print(f"\nPhase 10: Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=100, min_time=1.0)
        update_best(a_fin, sc_fin, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
