"""
Optimized step-function search using LSE smoothing, Adam optimization,
peak shaving, and multi-n exploration. Targets score below 1.5027.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

def _peak_shave_adaptive(a, deadline, n_trials=500):
    """

        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find contributing indices
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            break
        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = best_a[i_range] * best_a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            break

        # Reduce top contributors
        n_reduce = max(1, int(len(contrib) * 0.05))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.05
            a_try[idx] -= red
            reduction += red

        if reduction < 1e-15:
            continue

        c_peaks = np.sort(c)[::-1][:10]

def _direct_peak_reduction(a, deadline, n_reductions=200):
    """
    Direct subgradient step on the max: compute which a[i] values most
    increase the max convolution, and reduce them with backtracking.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for _ in range(n_reductions):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        # Active set: all peaks within 1% of max
        active = c >= c_max * 0.999
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        # Gradient of max (approximated by active set average)
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = best_a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * h  # gradient of max_b w.r.t. a (up to normalization)

        # Center the gradient (to maintain sum approximately)
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-16:
            break

        # Backtracking
        alpha = 0.01 * n / (gnorm * gnorm + 1e-10)
        base_sc = best_sc
        improved = False
        for attempt in range(50):
            a_try = best_a - alpha * grad_centered / n
            a_try = np.maximum(a_try, 0.0)

                alpha *= 0.5

            sc_try = _score(a_try)
            if sc_try < base_sc - 1e-14:
                best_a = a_try
                best_sc = sc_try
                improved = True
                break

def _resample_linear(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_diverse_init(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (t - 0.5) ** 2
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.arange(n) / n
        a = t + 0.1
        return a / a.sum()
    elif kind == "noise":
        a = rng.random(n) + 0.1
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-50 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


_t_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Aggressive L-BFGS LSE on parent
    P2: Adam multi-peak polishing
    P3: Direct peak reduction (subgradient)
    P4: Peak shaving with smart redistribution
    P5: Try alternative n values
    P6: Diverse restarts with quick optimization
    P7: Final ultra-fine polish
    """
    global _t_global
    _t_global = time.monotonic()
    t_start = _t_global
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_best = len(best_a)
        print(f"Parent: n={n_best}, score={best_score:.10f}")
    else:
        n_best = 8192
        best_a = np.ones(n_best) / n_best
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_best}, score={best_score:.10f}")

    # === Phase 1: Aggressive L-BFGS + Adam hybrid on parent ===
    print(f"\n[P1] L-BFGS+Adam hybrid (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600.0, t_start + 400.0)
    a1 = _adam_lbfgs_hybrid(best_a, p1_deadline, n_best, min_remaining=20.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [P1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc1:.10f} vs {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Direct peak reduction ===
    print(f"\n[P2] Direct peak reduction (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 450.0, t_start + 600.0)
    a2, sc2 = _direct_peak_reduction(best_a, p2_deadline, n_reductions=300)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [P2] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc2:.10f} vs {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak shaving with smart redistribution ===
    print(f"\n[P3] Smart peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300.0, t_start + 800.0)
    a3, sc3 = _peak_shave_adaptive(best_a, p3_deadline, n_trials=400)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [P3] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc3:.10f} vs {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Re-optimize if peak shaving helped
    if sc3 < best_score + 0.002 and time.monotonic() < deadline - 250:
        p3b_deadline = min(deadline - 250.0, t_start + 900.0)
        a3b = _adam_lbfgs_hybrid(best_a, p3b_deadline, n_best, min_remaining=20.0)
        sc3b = _score(a3b)
        if sc3b < best_score:
            best_a = a3b.copy()
            best_score = sc3b
            print(f"  NEW BEST: {best_score:.10f} [P3b] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Try different n values ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_options = [n_best + 2048, n_best - 2048, n_best + 4096]
    n_options = [n for n in n_options if n >= 4096 and n <= 24576]

    for n_try in n_options:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_best:
            continue
        a_init = _resample_linear(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        p4_deadline = min(deadline - 100.0, time.monotonic() + 120.0)
        a_opt = _adam_lbfgs_hybrid(a_init, p4_deadline, n_try, min_remaining=15.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            n_best = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P4] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  n={n_try}: best={sc_opt:.10f} (no improvement)")
        # Quick direct peak reduction on this n
        if time.monotonic() < deadline - 60:
            a_pr, sc_pr = _direct_peak_reduction(a_opt, min(deadline - 50, time.monotonic() + 40), 100)
            if sc_pr < best_score:
                best_a = a_pr.copy()
                best_score = sc_pr
                n_best = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P4_pr] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Diverse restarts ===
    print(f"\n[P5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        n_use = n_best
        inits = [
            ("noise", _make_diverse_init(n_use, "noise", rng)),
            ("sine", _make_diverse_init(n_use, "sine", rng)),
            ("quadratic", _make_diverse_init(n_use, "quadratic", rng)),
        ]
        # Perturbed best
        z_pert = np.log(np.maximum(best_a, 1e-30)) + 0.03 * rng.standard_normal(n_use)
        inits.append(("perturbed", _softmax(z_pert)))

        for label, a_r in inits:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic() - 30.0
            if remaining < 20:
                break
            a_opt = _adam_lbfgs_hybrid(a_r, deadline - 30.0, n_use, min_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Direct peak reduction with many small steps
        a_fin, sc_fin = _direct_peak_reduction(best_a, deadline - 3.0, n_reductions=500)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P6_direct] t={time.monotonic()-t_start:.1f}s")

        # One more peak shave
        if time.monotonic() < deadline - 3:
            a_ps, sc_ps = _peak_shave_adaptive(best_a, deadline - 2.0, n_trials=100)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  NEW BEST: {best_score:.10f} [P6_shave] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
