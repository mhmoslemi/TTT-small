**Gerchberg-Saxton (GS) method for autocorrelation flattening:**

The idea:
1. We have `a` (length n), zero-padded to length N.
2. Forward: `A = FFT(a, N)`, compute `c = IFFT(|A|^2)[:2n-1]`
3. Time-domain modification: Modify `c` to reduce peaks. E.g., `c_mod[k] = min(c[k], T)` where T is a target level.
4. Backward: Compute `|A_new|^2 = FFT(c_mod_padded)`. Update `A_new = sqrt(max(|A_new|^2, 0)) * exp(i * angle(A_old))`.
5. Compute `a_new = real(IFFT(A_new))[:n]`.
6. Project: `a_new = max(a_new, 0)`, normalize.
7. Repeat.

The key parameter is the target level T. If T is too high, we don't reduce peaks enough. If T is too low, the IFFT might produce negative values (which we project to 0).

A better approach is to use a "soft" modification:
`c_mod[k] = c[k] - alpha * max(0, c[k] - T)`

"""
IRLS + GS + L-BFGS hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: Iteratively Reweighted peak focusing with FFT-based gradients,
Gerchberg-Saxton-like spectral flattening, adaptive beta L-BFGS, and multi-phase
polish. Combines time-domain gradient methods with frequency-domain shaping.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a.copy()
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _irls_gradient_fft(a, w, n, N_fft):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a using FFT.
    grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)."""
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    # Cross-correlation: IFFT(FFT(w) * conj(FFT(a)))
    # With rfft, conj(A) means we conjugate the complex array
    corr = np.fft.irfft(W * np.conj(A), N_fft)[:n]
    return 2.0 * corr

def _gs_step(a, N, target_frac, alpha):
    """One GS iteration: flatten autocorrelation in frequency domain."""
    A = np.fft.rfft(a, N)
    power = np.abs(A) ** 2
    c = np.fft.irfft(power, N)[:2 * len(a) - 1]

    c_max = float(c.max())
    if c_max < 1e-20:
        return a

    target = c_max * target_frac
    excess = np.maximum(0.0, c - target)
    c_mod = c - alpha * excess
    c_mod = np.maximum(c_mod, 0.0)

    # Enforce symmetry
    c_mod = 0.5 * (c_mod + c_mod[::-1])

    c_padded = np.zeros(N)
    c_padded[:2 * len(a) - 1] = c_mod
    power_new = np.fft.irfft(np.fft.rfft(c_padded, N), N)
    # power_new should be real since c_mod is symmetric
    # Actually irfft of rfft gives back the original, so power_new = c_padded
    # We need FFT(c_mod_padded) as a power spectrum
    power_new = np.fft.rfft(c_mod, N)
    # Wait, c_mod has length 2n-1, we need to zero-pad to N and take FFT
    c_full = np.zeros(N)
    c_full[:2 * len(a) - 1] = c_mod
    power_new = np.fft.rfft(c_full, N).real  # Should be real if c_mod is symmetric
    power_new = np.maximum(power_new, 0.0)

    phase = np.angle(A)
    amp_new = np.sqrt(power_new)
    A_new = amp_new * (np.cos(phase) + 1j * np.sin(phase))

    a_new = np.fft.irfft(A_new, N)[:len(a)]
    return a_new

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    return _softmax(z)

        else:
            print(f"  No improvement: {sc1:.10f} vs {best_score:.10f}")

    # === Phase 2: IRLS ===
    print(f"\n[P2] IRLS optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 1000:
        a2, sc2 = _irls_optimize(best_seq, deadline, time_budget=450)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [irls] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement: {sc2:.10f} vs {best_score:.10f}")

    # === Phase 3: GS spectral ===
    print(f"\n[P3] GS spectral flattening (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 700:
        a3, sc3 = _gs_optimize(best_seq, deadline, time_budget=250)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [gs] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  GS no improvement: {sc3:.10f} vs {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 (refine) ===
    print(f"\n[P4] L-BFGS refine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a4, sc4 = _run_lbfgs_adaptive(best_seq, deadline, time_budget=250)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Active-set polish ===
    print(f"\n[P5] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a5, sc5 = _active_set_polish(best_seq, deadline, time_budget=300,
                                     tol_start=0.003, tol_end=0.00002, lr_init=0.025)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: IRLS round 2 ===
    print(f"\n[P6] IRLS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a6, sc6 = _irls_optimize(best_seq, deadline, time_budget=150)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [irls2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [24576, 16384, 12288, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")

        # Quick L-BFGS
        a_opt, sc_opt = _run_lbfgs_adaptive(a_init, deadline, time_budget=min(120, rem - 40))
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

        # Quick polish
        if time.monotonic() < deadline - 60:
            a_pol, sc_pol = _active_set_polish(a_opt, deadline, time_budget=min(60, deadline - 60 - time.monotonic()),
                                               tol_start=0.002, tol_end=0.00005, lr_init=0.015)
            if sc_pol < best_score:
                best_seq = a_pol.copy()
                best_score = sc_pol
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Random restart perturbation ===
    print(f"\n[P8] Perturbation restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        for attempt in range(6):
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            sigma = 0.02 * (1.0 + 0.5 * attempt)
            a_pert = _perturb(best_seq, sigma, rng)
            a_new, sc_new = _run_lbfgs_adaptive(a_pert, deadline, time_budget=min(100, rem))
            if sc_new < best_score:
                best_seq = a_new.copy()
                best_score = sc_new
                print(f"  NEW BEST: {best_score:.10f} [perturb_{attempt}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        rem = deadline - time.monotonic() - 3
        a_fin, sc_fin = _active_set_polish(best_seq, deadline, time_budget=max(20, rem),
                                           tol_start=0.001, tol_end=0.000005, lr_init=0.008)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS at the end
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.0001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(1000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 40})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
