import numpy as np
import time
from scipy.optimize import minimize


# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# LSE objective and gradient
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

    c_safe = np.maximum(c, 1e-30)

    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):

def _active_grad_structured(a, active, n):
    """Gradient of sum_{k in active} c[k] w.r.t. a, using convolution structure."""
    active_f = active.astype(np.float64)
    a_rev = a[::-1]
    wc = _full_convolve(a_rev, active_f)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _hess_matvec_structured(v, active_f, n):
    """Compute H*v where H[j,l] = 2 if j+l in active set."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve_structured(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / max(pH, 1e-30)
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

            dx_norm = float(np.linalg.norm(dx))
            if dx_norm > 1e-15:
                break
            mu *= 10.0

        if dx_norm < 1e-15:
            break

        # Backtracking line search
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try a small step in negative gradient direction
            a_try = a - 0.01 * grad_t / (gnorm + 1e-30)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a_new = a_try
                    improved = True
            if not improved:
                break

        a = a_new
        sc = _score(a)
        c_max_new = float(_full_convolve(a, a).max())
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            best_c_max = c_max_new

    return best_a, best_sc

# L-BFGS with LSE (improved beta schedule)
# ============================================================

def _lbfgs_lse_v2(a_init, deadline, min_remaining=60.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    # Adaptive beta based on spread of c values
    c_sorted = np.sort(c_cur)[::-1]
    n_tail = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[n_tail]
    base_beta = max(gap * 0.2, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # More stages, smoother schedule
    n_stages = 16
    ratios = np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                       0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.001, 0.0005])
    betas = base_beta * ratios[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, reduce beta more aggressively
                break
        except Exception:
            continue

    return current_a, current_sc

# Variance-based L-BFGS (smooth landscape optimizer)

def _lbfgs_variance(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    remaining = deadline - min_remaining - time.monotonic()
    if remaining < 3:
        return current_a, current_sc
    est_time = 0.002 * max(1.0, n / 8192.0)
    max_iter = min(5000, max(200, int(remaining / (est_time * 8))))

    try:
        result = minimize(
            _variance_objective, z, args=(n,),
            jac=_variance_gradient, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-16,
                     'gtol': 1e-13, 'maxcor': 60, 'maxls': 100}
        )
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if sc_new < current_sc:
            return a_new, sc_new
        else:
            # Variance reduction might increase peak; check
            return a_new, sc_new
    except Exception:
        return current_a, current_sc

# Subgradient polish (improved)
# ============================================================

def _subgrad_polish_v2(a_init, deadline, n_steps=8000, lr_init=0.012, tol_frac_init=0.003):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Adaptive tolerance
        tol_frac = max(tol_frac_init * 0.99 ** (i // 100), 0.0002)
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 15:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


# ============================================================
# KKT residual-based update (novel)
# ============================================================

def _kkt_residual_update(a_init, deadline, n_steps=2000):
    """Directly target KKT conditions: sum_{k in A} a[k-j]/|A| should be constant."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.05

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.002 * c_max, 1e-12)
        active = c >= (c_max - tol)
        n_active = max(1, active.sum())

        # Residual: (sum_{k in A} a[k-j]) / n_active should be constant
        active_f = active.astype(np.float64) / n_active
        a_rev = a[::-1]
        wc = _full_convolve(a_rev, active_f)
        residual = wc[n - 1:2 * n - 1]
        residual = residual - np.mean(residual)

        a_new = a - lr * residual
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            lr = min(lr * 1.01, 0.1)
        else:
            a = a  # revert
            lr *= 0.9
        a = a_new if sc_new < _score(a) else a

    return best_a, best_sc

def _perturb_fourier_phase(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.08):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 1e-10)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_mirror(a, rng):
    """Create a perturbation that flips part of the sequence."""
    n = len(a)
    split = int(rng.integers(n // 4, 3 * n // 4))
    a_new = a.copy()
    a_new[split:2 * n - split] = a_new[2 * n - split - 1:split - 1:-1]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_mix_v2(a, rng):
    a1 = _perturb_log(a, 0.02, rng)
    a2 = _perturb_fourier_phase(a1, rng, max_amp=0.15)
    return a2

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "arch":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10 * np.abs(t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def _power_sum_lbfgs(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for p in [4, 16, 64, 256]:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 10))))
        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            continue
    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase structured optimizer combining variance smoothing,
    LSE L-BFGS, structured active-set Newton, KKT residual updates,
    subgradient polish, and diverse spectral perturbations.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init")

    # === Phase 1: Quick diverse profiles (20s) ===
    print(f"\n[Phase 1] Quick profile exploration")
    p1_deadline = min(deadline, t_start + 20)
    for kind in ["uniform", "tapered", "cos2", "gauss", "bimodal", "ramp", "arch"]:
        if time.monotonic() > p1_deadline:
            break
        a_try = _make_profile(n_parent, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"prof_{kind}")

    # === Phase 2: Variance smoothing (80s) ===
    print(f"\n[Phase 2] Variance smoothing")
    if time.monotonic() < deadline - 500:
        a_var, sc_var = _lbfgs_variance(_best_seq.copy(), deadline, min_remaining=500)
        upd(a_var, sc_var, "variance_smooth")
        # Also try on a perturbed version
        if time.monotonic() < deadline - 480:
            a_pert = _perturb_log(_best_seq, 0.01, rng)
            a_var2, sc_var2 = _lbfgs_variance(a_pert, deadline, min_remaining=480)
            upd(a_var2, sc_var2, "variance_pert")

    # === Phase 3: Main LSE L-BFGS (400s) ===
    print(f"\n[Phase 3] Main LSE L-BFGS")
    if time.monotonic() < deadline - 400:
        a_lse, sc_lse = _lbfgs_lse_v2(_best_seq.copy(), deadline, min_remaining=400)
        upd(a_lse, sc_lse, "lse_main")

    # === Phase 4: Power-sum L-BFGS (150s) ===
    print(f"\n[Phase 4] Power-sum L-BFGS")
    if time.monotonic() < deadline - 250:
        a_ps, sc_ps = _power_sum_lbfgs(_best_seq.copy(), deadline, min_remaining=250)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 5: Structured Newton (200s) ===
    print(f"\n[Phase 5] Structured Newton")
    if time.monotonic() < deadline - 200:
        nw_dead = min(deadline - 180, time.monotonic() + 200)
        a_nw, sc_nw = _newton_active_set_v2(_best_seq, nw_dead, max_iter=400,
                                            tol_frac_init=0.01, tol_frac_min=0.0003)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 6: KKT residual update (100s) ===
    print(f"\n[Phase 6] KKT residual")
    if time.monotonic() < deadline - 150:

        upd(a_kkt, sc_kkt, "kkt_residual")

    # === Phase 7: Subgradient polish (100s) ===
    print(f"\n[Phase 7] Subgradient polish")
    if time.monotonic() < deadline - 80:

        a_sg, sc_sg = _subgrad_polish_v2(_best_seq, sg_dead, n_steps=6000, lr_init=0.01,
                                         tol_frac_init=0.003)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 8: Newton + KKT + SG alternating (200s) ===
    print(f"\n[Phase 8] Alternating refinement")
    if time.monotonic() < deadline - 150:
        a_cur = _best_seq.copy()
        for round_i in range(4):

                                              tol_frac_init=0.005, tol_frac_min=0.001)
            if sc_n < _score(a_cur):
                a_cur = a_n
            upd(a_cur, _score(a_cur), f"alt_nw_{round_i}")
            # KKT
            kkt_dead = min(deadline - 100, time.monotonic() + 30)
            a_k, sc_k = _kkt_residual_update(a_cur, kkt_dead, n_steps=1000)
            if sc_k < _score(a_cur):
                a_cur = a_k
            upd(a_cur, _score(a_cur), f"alt_kkt_{round_i}")
            # SG

            a_s, sc_s = _subgrad_polish_v2(a_cur, sg_dead, n_steps=2000, lr_init=0.005,
                                           tol_frac_init=0.002)
            if sc_s < _score(a_cur):
                a_cur = a_s
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 9: Multi-n exploration (200s) ===
    print(f"\n[Phase 9] Multi-n exploration")
    n_candidates = [4096, 16384]
    if n_parent not in n_candidates and 2048 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)
        a_opt_n, sc_opt_n = _lbfgs_lse_v2(a_init, deadline, min_remaining=60)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")
        if _score(a_opt_n) < _best_score * 1.003 and time.monotonic() < deadline - 50:
            nw_dead = min(deadline - 40, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set_v2(a_opt_n, nw_dead, max_iter=80,
                                                        tol_frac_init=0.005, tol_frac_min=0.001)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_nw")

    # === Phase 10: Diverse restarts (300s) ===
    print(f"\n[Phase 10] Diverse restarts")
    if time.monotonic() < deadline - 80:
        perturbations = [
            ("log_003", lambda: _perturb_log(_best_seq, 0.003, rng)),
            ("log_01", lambda: _perturb_log(_best_seq, 0.01, rng)),
            ("fourier", lambda: _perturb_fourier_phase(_best_seq, rng, max_amp=0.2)),
            ("amp", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.05)),
            ("block", lambda: _perturb_block_swap(_best_seq, rng, n_swaps=3)),
            ("mirror", lambda: _perturb_mirror(_best_seq, rng)),
            ("mix", lambda: _perturb_mix_v2(_best_seq, rng)),
            ("log_02", lambda: _perturb_log(_best_seq, 0.02, rng)),
        ]
        for label, gen_func in perturbations:
            if time.monotonic() > deadline - 40:
                break
            p_start = gen_func()
            sc_start = _score(p_start)
            if sc_start < _best_score:
                upd(p_start, sc_start, f"restart_{label}")
            # Quick LSE
            a_r, sc_r = _lbfgs_lse_v2(p_start, deadline, min_remaining=30)
            upd(a_r, sc_r, f"restart_{label}_lse")
            # Quick Newton
            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 25:
                nw_dead = min(deadline - 20, time.monotonic() + 25)
                a_r2, sc_r2 = _newton_active_set_v2(a_r, nw_dead, max_iter=60,
                                                    tol_frac_init=0.005, tol_frac_min=0.001)
                upd(a_r2, sc_r2, f"restart_{label}_nw")

    # === Phase 11: Final ultra-fine polish (100s) ===
    print(f"\n[Phase 11] Final ultra-fine polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            # Ultra-low beta LSE
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-17,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass
        # Final Newton
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_active_set_v2(_best_seq, deadline - 3, max_iter=100,
                                                    tol_frac_init=0.003, tol_frac_min=0.0001)
            upd(a_fin2, sc_fin2, "final_newton")
        # Final SG
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _subgrad_polish_v2(_best_seq, deadline - 1, n_steps=5000,
                                                 lr_init=0.004, tol_frac_init=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")
        # Final KKT
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _kkt_residual_update(_best_seq, deadline - 1, n_steps=3000)
            upd(a_fin4, sc_fin4, "final_kkt")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
