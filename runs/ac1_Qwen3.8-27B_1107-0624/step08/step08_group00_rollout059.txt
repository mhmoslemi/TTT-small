def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    a_rev = a[::-1]

    h = wc[n - 1:2 * n - 1]

    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()

def _fista_phase(a_init, deadline, min_time_remaining=30.0):
    """FISTA on the smoothed objective with direct (non-softmax) parameterization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    base_beta = max(c_max * 0.005, 1e-10)
    
    # FISTA with momentum
    t = 1.0
    a_prev = a.copy()
    
    # Adaptive step size
    step = 1.0 / (4.0 * n * c_max + 1e-15)  # rough Lipschitz bound
    
    for beta_mult in [1.0, 0.5, 0.25, 0.1]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        beta = base_beta * beta_mult
        
        # Reset FISTA state
        t = 1.0
        a_prev = a.copy()
        
        # Run FISTA iterations
        max_iter = 5000
        for it in range(max_iter):
            if time.monotonic() + min_time_remaining > deadline:
                break
            
            # Compute gradient of smoothed objective
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_exp = e.sum()
            if s_exp < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_exp
            
            # Gradient: 4*n*a*(h - W) where h = conv(w, reverse(a))[n-1:2n-1]
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            
            # Normalize out the mean component (since we're on the simplex)
            grad = grad - np.mean(grad)
            
            # Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + (t - 1.0) / t_new * (a - a_prev)
            y = np.maximum(y, 0.0)
            
            # Gradient step
            a_new = y - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            
            # Backtracking line search (simplified: just accept)
            a_prev = a
            a = a_new
            t = t_new
            
            # Check improvement
            sc = float(2.0 * n * _full_convolve(a, a).max())
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
            # Reduce step if no improvement for a while
            elif it % 100 == 0:
                step *= 0.5
            elif it % 50 == 0:
                # Restart momentum
                t = 1.0
                a_prev = a.copy()
    
    return best_a, _score(best_a)
