def _run_lbfgs_topk(a_init, deadline, min_time_remaining=30.0, k_frac=0.05):
    """L-BFGS with top-k peak-focused log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _conv(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Choose top-k
    k = max(3, int(k_frac * len(c_cur)))
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[min(20, len(c_sorted)-1)] - c_sorted[min(100, len(c_sorted)-1)] if len(c_sorted) > 100 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)
    if base_beta < max_c * 0.001:
        base_beta = max_c * 0.01

    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                                   0.08, 0.05, 0.03, 0.02, 0.01, 0.006,
                                   0.003, 0.0015, 0.0008, 0.0004, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Estimate iterations
        if n <= 4096:
            est_per_iter = 0.004
        elif n <= 8192:
            est_per_iter = 0.008
        elif n <= 16384:
            est_per_iter = 0.02
        else:
            est_per_iter = 0.04 * (n / 32768.0)

        max_iter = min(3000, max(20, int(remaining / (est_per_iter * 15))))

        try:
            def obj_fn(z_local):
                return _lsobj_topk(z_local, beta, n, k)
            
            def grad_fn(z_local):
                return _lsgrad_topk(z_local, beta, n, k)

            result = minimize(
                obj_fn,
                z,
                jac=grad_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            # Even if not improved, update z to continue from new point
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
        except Exception as e:
            print(f"  LBFGS error at beta={beta:.4f}: {e}")
            pass

    return current_a


def _peak_refine(a_init, deadline, max_rounds=80):
    """Peak-targeted refinement: directly reduce the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _conv(a, a)
        c_max = float(c.max())
        
        # Gradually increase the number of peaks we target
        n_peaks = min(5 + rnd // 3, 50, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        
        # Weight peaks by how much they exceed the median
        median_c = np.median(c)
        weights = np.zeros(len(c))
        for k in top_peaks:
            excess = c[k] - median_c
            weights[k] = max(excess, 0.01 * c_max)
        weights = weights / weights.sum()

        a_rev = a[::-1]
        wc = _conv(weights, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(weights, c))
        grad = 2.0 * a * (h - W)

        # Adaptive step size
        f_old = c_max
        step = 0.5 / (1.0 + 0.05 * rnd)
        improved = False
        for ls in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _conv(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old * (1 - 1e-10):
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try a smaller fixed step
            step = 0.01 / (1.0 + 0.1 * rnd)
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _conv(a_try, a_try)
            if float(c_try.max()) < f_old:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
            else:
                if rnd > 5:
                    break

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.03, k=10):
    """Mirror descent in log-space with top-k peak targeting."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_lr = lr_init

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _conv(a, a)
        c_max = float(c.max())

        k_use = min(k + i // 100, len(c) // 4)
        top_k = np.argsort(c)[::-1][:k_use]
        w = np.zeros_like(c)
        w[top_k] = 1.0 / k_use

        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = max(lr_init * 0.3, 0.001)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old + 1)[:-1]
    x_new = np.linspace(0, 1, n_new + 1)[:-1]
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.25) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 0.3 + 2.0 * t ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_structured(a, freq, amp, rng):
    """Apply structured perturbation (low-frequency modulation)."""
    n = len(a)
    t = np.linspace(0, 1, n)
    mod = 1.0 + amp * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

"""
Optimized step function coefficient minimizer.
Strategy:
1. Load parent, normalize to unit sum
2. Multi-n exploration with interpolation
3. Top-k focused log-sum-exp L-BFGS with adaptive beta schedule
4. Peak-targeted gradient refinement (adaptive peak count)
5. Active-set Newton with multiple tolerances
6. Mirror descent polishing
7. Structured low-frequency perturbations for diversity
All helper functions are top-level; no lambdas or closures.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    z = np.clip(z, -700, 700)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n, k):
    a = _softmax(z)
    c = _conv(a, a)
    if k < len(c):
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[k - 1]
        c_eff = np.where(c >= threshold, c, threshold)
    else:
        c_eff = c
    c_max = float(c.max())
    shifted = (c_eff - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n, k):
    a = _softmax(z)
    c = _conv(a, a)
    if k < len(c):
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[k - 1]
        c_eff = np.where(c >= threshold, c, threshold)
    else:
        c_eff = c
    c_max = float(c.max())
    shifted = (c_eff - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, k_frac=0.03):
    n = len(a_init)
    c_cur = _conv(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    k = max(3, int(k_frac * (2 * n - 1)))

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[10] - c_sorted[50]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.4, max_c * 0.015)
    if base_beta < max_c * 0.005:
        base_beta = max_c * 0.01

    betas = base_beta * np.array([
        1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.055, 0.035, 0.022, 0.014, 0.009,
        0.0055, 0.0035, 0.002, 0.0012, 0.0007
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        if n <= 4096:
            est_per_iter = 0.004
        elif n <= 8192:
            est_per_iter = 0.008
        elif n <= 16384:
            est_per_iter = 0.018
        elif n <= 32768:
            est_per_iter = 0.035
        else:
            est_per_iter = 0.06

        max_iter = min(4000, max(20, int(remaining / (est_per_iter * 12))))

        try:
            def obj_fn(z_local):
                return _lsobj(z_local, beta, n, k)

            def grad_fn(z_local):
                return _lsgrad(z_local, beta, n, k)

            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_refine(a_init, deadline, max_rounds=100):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _conv(a, a)
        c_max = float(c.max())

        n_peaks = min(3 + rnd // 2, 80, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]

        median_c = np.median(c)
        weights = np.zeros(len(c))
        for k in top_peaks:
            weights[k] = max(c[k] - median_c, 0.005 * c_max)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum

        a_rev = a[::-1]
        wc = _conv(weights, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(weights, c))
        grad = 2.0 * a * (h - W)

        f_old = c_max
        step = 1.0 / (2.0 + 0.03 * rnd)
        improved = False
        for ls in range(50):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _conv(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old * (1.0 - 1e-11):
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            step = 0.005 / (1.0 + 0.05 * rnd)
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _conv(a_try, a_try)
            if float(c_try.max()) < f_old:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
            elif rnd > 10:
                break

    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _conv(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        max_active = 3 * n
        if n_active > max_active:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:max_active]
            active[top_idx] = True
            n_active = max_active

        A_vec = active.astype(np.float64)
        m = n_active

        flip_a = a[::-1]
        out_len = n + len(A_vec) - 1
        N = _next_pow2(out_len)
        fa = np.fft.rfft(flip_a, N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fa * fA, N)[:out_len]
        h = result[n - 1:2 * n - 1]

        g = (2.0 / m) * h
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-10 * c_max / m + 1e-14

        N2 = _next_pow2(n + len(A_vec) - 1)
        fA2 = np.fft.rfft(A_vec, N2)

        def hess_matvec(v):
            fv = np.fft.rfft(v, N2)
            res2 = np.fft.irfft(fv * fA2, N2)[:out_len]
            return (4.0 / m) * res2[n - 1:2 * n - 1] + mu * v

        rhs = g_t.copy()
        x = np.zeros(n)
        r = rhs - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(150):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        u = x

        ones = np.ones(n)
        x_w = np.zeros(n)
        r_w = ones - hess_matvec(x_w)
        p_w = r_w.copy()
        rs_w = np.dot(r_w, r_w)
        for cg_it in range(150):
            Ap_w = hess_matvec(p_w)
            pAp_w = np.dot(p_w, Ap_w)
            if pAp_w < 1e-30:
                break
            alpha_w = rs_w / pAp_w
            x_w += alpha_w * p_w
            r_new_w = r_w - alpha_w * Ap_w
            rs_new_w = np.dot(r_new_w, r_new_w)
            if np.sqrt(rs_new_w) < 1e-13:
                break
            p_w = r_new_w + (rs_new_w / rs_w) * p_w
            r_w = r_new_w
            rs_w = rs_new_w
        w_vec = x_w

        sum_w = float(w_vec.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_vec
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _conv(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-11:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(25):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _conv(a_try, a_try)
                if float(c_try.max()) < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _conv(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.025, k_start=5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _conv(a, a)
        c_max = float(c.max())

        k_use = min(k_start + i // 80, max(10, len(c) // 5))
        top_k = np.argsort(c)[::-1][:k_use]
        w = np.zeros_like(c)
        w[top_k] = 1.0 / k_use

        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.0004 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 40 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 15:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = max(lr_init * 0.4, 0.002)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old + 1)[:-1]
    x_new = np.linspace(0, 1, n_new + 1)[:-1]
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.3) ** 2) / 0.03) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.03)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_structured(a, freq, amp, rng):
    n = len(a)
    t = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    mod = 1.0 + amp * np.sin(2 * np.pi * freq * t + phase)
    a_new = a * mod
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"d={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        update_best(parent, parent_sc, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS on parent (main optimization)
    print(f"\n[1] L-BFGS top-k n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = deadline - 400
    if time.monotonic() < p1_deadline:
        a1 = _run_lbfgs(parent, p1_deadline, min_time_remaining=300.0, k_frac=0.025)
        update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Multi-n exploration
    print(f"\n[2] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_tries = [8192, 12288, 20480, 24576, 32768]
    n_tries = [x for x in n_tries if x != n_parent]
    for n_try in n_tries[:3]:
        if time.monotonic() > deadline - 150:
            break
        a_try = _interpolate_to_n(parent, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try}: init={sc_try:.10f}")
        a_opt = _run_lbfgs(a_try, min(deadline - 100, time.monotonic() + 80),
                           min_time_remaining=40.0, k_frac=0.03)
        update_best(a_opt, _score(a_opt), f"n{n_try}")

    # Phase 3: Peak refinement
    print(f"\n[3] Peak refine (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_pr, sc_pr = _peak_refine(best_seq, min(deadline - 50, time.monotonic() + 60), max_rounds=80)
        update_best(a_pr, sc_pr, "peak_ref")

    # Phase 4: Active-set Newton (multiple tolerances)
    print(f"\n[4] AS Newton (t={time.monotonic()-t_start:.1f}s)")
    as_tols = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    for tol in as_tols:
        if time.monotonic() > deadline - 30:
            break
        t_as = min(deadline - 15, time.monotonic() + 30)
        a_as, sc_as = _active_set_newton(best_seq, t_as, max_iter=100, tol_active=tol)
        update_best(a_as, sc_as, f"as_{tol}")

    # Phase 5: Mirror descent
    print(f"\n[5] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_md, sc_md = _mirror_descent(best_seq, min(deadline - 10, time.monotonic() + 30),
                                      max_steps=3000, lr_init=0.02, k_start=5)
        update_best(a_md, sc_md, "mirror")

    # Phase 6: Structured perturbations
    print(f"\n[6] Perturbations (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        configs = [(1, 0.1), (2, 0.08), (3, 0.15), (1, 0.25), (4, 0.12)]
        for freq, amp in configs:
            if time.monotonic() > deadline - 8:
                break
            p_st = _perturb_structured(best_seq, freq, amp, rng)
            a_r = _run_lbfgs(p_st, min(deadline - 4, time.monotonic() + 15),
                             min_time_remaining=3.0, k_frac=0.05)
            update_best(a_r, _score(a_r), f"p_{freq}_{amp}")
            if time.monotonic() < deadline - 4:
                a_r2, sc_r2 = _peak_refine(a_r, min(deadline - 2, time.monotonic() + 8), max_rounds=15)
                update_best(a_r2, sc_r2, f"p_{freq}_{amp}_peak")

    # Phase 7: Final tight AS Newton + peak refine
    print(f"\n[7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_f1, sc_f1 = _peak_refine(best_seq, min(deadline - 1, time.monotonic() + 10), max_rounds=30)
        update_best(a_f1, sc_f1, "final_peak")
        if time.monotonic() < deadline - 1:
            a_f2, sc_f2 = _active_set_newton(best_seq, deadline - 0.5, max_iter=50, tol_active=0.0001)
            update_best(a_f2, sc_f2, "final_as")

    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: {sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: {_score(fallback):.10f}")
    return [float(x) for x in fallback]
