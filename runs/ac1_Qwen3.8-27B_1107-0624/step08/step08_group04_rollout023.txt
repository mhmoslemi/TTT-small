import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H_{i,j} = 2*W[i+j], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _active_grad(a, w, n):
    """Compute gradient of (1/m)*sum_{k in A} c_k using FFT."""
    a_rev = a[::-1]
    V = _full_convolve(a_rev, w)
    return 2.0 * V[n - 1:2 * n - 1]


def _cg_solve_Hankel(W, n, rhs, max_iter=2000, tol=1e-16):
    """Solve (2*Hankel(W)) * x = rhs using CG with residual monitoring."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    if rs < 1e-40:
        return x
    rs0 = rs
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x += alpha * p
        r -= alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs_new > rs * 1.5:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _active_set_loop(a, deadline, start_tol=0.01, min_tol=1e-6, rounds_per_tol=8,

def _smoobj(z, beta, n):

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        max_iter = min(8000, max(100, int(remaining / (0.003 * max(1, n / 4096)))))
        if max_iter < 10:
            break

        try:
            result = minimize(
                _make_smoobj_closure(beta, n),
                z,
                jac=_make_smojac_closure(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_smoobj_closure(beta, n):
    def obj(z):
        return _smoobj(z, beta, n)
    return obj


def _make_smojac_closure(beta, n):
    def jac(z):
        return _smograd(z, beta, n)
    return jac


def _peak_equalize(a, deadline, max_iters=200, min_time_remaining=30.0):
    """
    Direct peak equalization: iteratively push down the highest convolution peaks
    by adjusting the sequence in the direction that reduces them.
    """
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_act = int(active.sum())
        if n_act == 0:
            active[int(np.argmax(c))] = True
            n_act = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act

        grad = _active_grad(a, w, n)
        grad_t = grad - np.mean(grad)

        # Use steepest descent direction with adaptive step
        gn = float(np.linalg.norm(grad_t))
        if gn < 1e-15:
            break

        # Try to find good step size
        best_alpha = 0.0
        best_cmax = c_max
        alpha = 0.01 / gn
        for ls in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try_max = float(_full_convolve(a_try, a_try).max())
            if c_try_max < best_cmax:
                best_cmax = c_try_max
                best_alpha = alpha
            alpha *= 0.7
            if alpha * gn < 1e-10:
                break

        if best_alpha <= 0:
            break

        a = a - best_alpha * grad_t
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_left":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.05
        return a / a.sum()
    elif kind == "asym_right":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (1.0 - t)) + 0.05
        return a / a.sum()
    elif kind == "asym_center_left":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.3) ** 2) + 0.03
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * t) + 0.2
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(12345)
        a = rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(16) / 16
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
