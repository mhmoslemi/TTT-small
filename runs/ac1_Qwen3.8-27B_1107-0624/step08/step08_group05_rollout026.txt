    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-12, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _newton_active_set(a_init, deadline, max_iter=300, tol_frac=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_c_active = 0.0
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = max(tol_frac * c_max, 1e-14)
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    best_c_active = float(c[active].sum())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-12, max_iter=300)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            if alpha < 1e-14:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _newton_wide(a_init, deadline, max_iter=200):

            grad = _hess_matvec(a, active_f, n)
            grad_t = grad - np.mean(grad)
            gnorm = float(np.linalg.norm(grad_t))
            if gnorm < 1e-14:

            mu = 1e-9

            def _hv2(v):
                return _hess_matvec(v, active_f, n) + mu * v

            dx = _cg_solve(_hv2, -grad_t, tol=1e-12, max_iter=300)

            a = a_new

def _adaptive_lbfgs_lse(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    c_cur = _full_convolve(current_a, current_a)
    c_sorted = np.sort(c_cur)[::-1]
    c_max = float(c_sorted[0])
    if c_max < 1e-15:
        return current_a, current_sc

    gap = c_max - c_sorted[min(50, len(c_sorted) - 1)]
    n_active = int(np.sum(c_cur >= c_max - 0.01 * c_max))
    base_beta = max(gap * 0.2, c_max * 0.002)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    n_stages = 20
    for stage in range(n_stages):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        frac = stage / max(1, n_stages - 1)
        beta = base_beta * (0.0001) ** (frac ** 0.7)
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 10))))
        max_iter = min(max_iter, max(100, int(120.0 / (est_time * (n_stages - stage)))))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                jac=_lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _peak_shaving(a_init, deadline, n_rounds=500):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.002

    for i in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        target = np.zeros(2 * n - 1)
        target[peak_idx] = 1.0
        a_rev = a[::-1]
        tc = _full_convolve(target, a_rev)
        h = tc[n - 1:2 * n - 1]
        W = float(np.dot(target, c))
        grad = 2.0 * n * (h - W)

        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        grad_z = _softmax(z) * (grad - np.dot(_softmax(z), grad))
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.005)
        else:
            lr *= 0.9
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-30))
            z = z - z.mean()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_spectral_balance(a, rng):
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    amp = np.abs(A)
    phase = np.angle(A)
    target = np.full_like(amp, amp.sum() / len(amp))
    mix = 0.1 + 0.2 * rng.random()
    amp_new = (1 - mix) * amp + mix * target
    amp_new = np.maximum(amp_new, 0)
    A_new = amp_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "ramp2":
        t = np.linspace(0, 1, n)
        a = 0.3 + t * t
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "s_curve":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-8.0 * (t - 0.5)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (2 * t - 1) ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-stage optimizer: adaptive LSE, wide Newton, peak shaving, spectral balancing,
    diverse restarts, and multi-n exploration. Targets score < 1.5027.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "ramp")
        n_parent = 16384
        upd(parent, _score(parent), "init_ramp")

    # === Phase 1: Quick diverse profiles (30s) ===
    print(f"\n[Phase 1] Diverse profile exploration")
    for kind in ["ramp", "ramp2", "s_curve", "cos2", "quadratic", "tapered"]:
        if time.monotonic() > deadline - 1500:
            break
        a_try = _make_profile(n_parent, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"prof_{kind}")

    # Quick LSE on best profile
    if _best_seq is not None and time.monotonic() < deadline - 1400:
        a_q, sc_q = _adaptive_lbfgs_lse(_best_seq.copy(), min(deadline - 1400, time.monotonic() + 40), min_remaining=5)
        upd(a_q, sc_q, "quick_lbfgs")

    # === Phase 2: Main adaptive LSE (500s) ===
    print(f"\n[Phase 2] Main adaptive LSE")
    if time.monotonic() < deadline - 500:
        a_main, sc_main = _adaptive_lbfgs_lse(_best_seq.copy(), deadline, min_remaining=450)
        upd(a_main, sc_main, "main_lbfgs")

    # === Phase 3: Power-sum LSE (100s) ===
    print(f"\n[Phase 3] Power-sum refinement")
    if time.monotonic() < deadline - 250:
        n = len(_best_seq)
        z = np.log(np.maximum(_best_seq, 1e-30))
        z = z - z.mean()
        for p in [8, 16, 32, 64]:
            if time.monotonic() > deadline - 250:
                break
            remaining = deadline - 250 - time.monotonic()
            if remaining < 3:
                break
            est_time = 0.003 * max(1.0, n / 4096.0)
            max_iter = min(2000, max(50, int(remaining / (est_time * 5))))
            try:
                result = minimize(
                    _power_sum_obj, z, args=(float(p), n),
                    jac=_power_sum_grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-14, 'maxcor': 60}
                )
                a_ps = _softmax(result.x)
                sc_ps = _score(a_ps)
                if sc_ps < _best_score:
                    upd(a_ps, sc_ps, f"power_p{p}")
                    z = np.log(np.maximum(a_ps, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass

    # === Phase 4: Wide Newton active-set (200s) ===
    print(f"\n[Phase 4] Wide Newton")
    if time.monotonic() < deadline - 200:
        nw_dead = min(deadline - 150, time.monotonic() + 200)
        a_nw, sc_nw = _newton_wide(_best_seq, nw_dead, max_iter=200)
        upd(a_nw, sc_nw, "newton_wide")

    # === Phase 5: Peak shaving (100s) ===
    print(f"\n[Phase 5] Peak shaving")
    if time.monotonic() < deadline - 120:
        ps_dead = min(deadline - 100, time.monotonic() + 100)
        a_ps2, sc_ps2 = _peak_shaving(_best_seq, ps_dead, n_rounds=5000)
        upd(a_ps2, sc_ps2, "peak_shave")

    # === Phase 6: Tight Newton + SG alternating (150s) ===
    print(f"\n[Phase 6] Newton + SG alternating")
    if time.monotonic() < deadline - 120:
        a_cur = _best_seq.copy()
        for rnd in range(4):
            if time.monotonic() > deadline - 80:
                break
            # Tight Newton
            nw_dl = min(deadline - 80, time.monotonic() + 40)
            a_n3, sc_n3 = _newton_active_set(a_cur, nw_dl, max_iter=100, tol_frac=0.0005)
            if sc_n3 < _score(a_cur):
                a_cur = a_n3
            upd(a_cur, _score(a_cur), f"alt_nw_{rnd}")
            # SG polish
            sg_dl = min(deadline - 80, time.monotonic() + 30)
            a_s3, sc_s3 = _subgrad_polish(a_cur, sg_dl, n_steps=3000, lr_init=0.005, tol_frac=0.0005)
            if sc_s3 < _score(a_cur):
                a_cur = a_s3
            upd(a_cur, _score(a_cur), f"alt_sg_{rnd}")

    # === Phase 7: Multi-n exploration (200s) ===
    print(f"\n[Phase 7] Multi-n exploration")
    n_candidates = sorted(set([4096, 8192, 12288, 16384, 20480]))
    if n_parent not in n_candidates and 2048 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        a_opt_n, sc_opt_n = _adaptive_lbfgs_lse(a_init, deadline, min_remaining=40)
        upd(a_opt_n, sc_opt_n, f"n{n_try}_lse")

        if _score(a_opt_n) < _best_score * 1.002 and time.monotonic() < deadline - 40:
            nw_dl2 = min(deadline - 35, time.monotonic() + 40)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dl2, max_iter=80, tol_frac=0.001)
            upd(a_opt_n2, sc_opt_n2, f"n{n_try}_nw")

    # === Phase 8: Diverse restarts (150s) ===
    print(f"\n[Phase 8] Diverse restarts")
    if time.monotonic() < deadline - 50:
        restarts = [
            ("log03", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("fourier", lambda: _perturb_fourier_phase(_best_seq, rng, 8, 0.3)),
            ("spectral", lambda: _perturb_spectral_balance(_best_seq, rng)),
            ("log05", lambda: _perturb_log(_best_seq, 0.05, rng)),
            ("mix", lambda: _perturb_log(_perturb_fourier_phase(_best_seq, rng, 5, 0.2), 0.02, rng)),
        ]
        for name, gen in restarts:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen()
            a_r, sc_r = _adaptive_lbfgs_lse(p_start, deadline, min_remaining=20)
            upd(a_r, sc_r, f"restart_{name}")
            if sc_r < _best_score * 1.003 and time.monotonic() < deadline - 15:
                nw_dl3 = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dl3, max_iter=50, tol_frac=0.002)
                upd(a_r2, sc_r2, f"restart_{name}_nw")

    # === Phase 9: Final ultra-fine polish (60s) ===
    print(f"\n[Phase 9] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining * 0.5 / 0.005)))
            try:
                result = minimize(
                    _lse_obj, z_fin, args=(beta_final, n_fin),
                    jac=_lse_grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_lse")
            except Exception:
                pass

        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=100, tol_frac=0.0003)
            upd(a_fin2, sc_fin2, "final_nw")

        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=5000, lr_init=0.003, tol_frac=0.0003)
            upd(a_fin3, sc_fin3, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
