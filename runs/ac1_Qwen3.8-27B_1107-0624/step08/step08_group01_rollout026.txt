**Improved Active-Set GD:**

The objective (with sum(a)=1, a>=0) is to minimize `max_k c[k]`.

At the optimum, the KKT conditions say:
- There exists a set of active lags A and a multiplier M such that:
  - `c[k] = M` for k in A
  - `c[k] <= M` for k not in A
  - The gradient of `sum_{k in A} (c[k] - M)` is in the span of the constraint gradients (which are: sum(a)=1 gives gradient 1, and a>=0 gives gradient e_i for active i)

So at the optimum, `sum_{k in A} ∂c[k]/∂a[j] = lambda + mu[j]` for all j, where lambda is the multiplier for the sum constraint and mu[j] >= 0 is the multiplier for the non-negativity constraint (mu[j] = 0 if a[j] > 0).

This gives us a system of equations that we can solve! The active set A determines the problem, and we solve for a and M.

The approach:
1. Guess the active set A (e.g., the top-k lags where c[k] is close to max)
2. Solve the KKT system for a and M
3. Check if the solution is feasible (a >= 0, c[k] <= M for k not in A)
4. If not, update the active set and repeat

This is similar to the "active set method" in constrained optimization.

For step 2, the KKT system is:
- `c[k] = M` for k in A (|A| equations)
- `sum_j ∂c[k]/∂a[j] * v[j] = 0` for k in A, v in the null space of the active constraint gradients...

"""
Step-function optimizer minimizing 2n*max(conv(a,a))/sum(a)^2.
Multi-phase approach: adaptive LSE with accurate gradients, QP active-set
refinement using FFT-accelerated structured matrix products, improved
active-set GD with curvature-aware step sizes, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_grad_full(a, w):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a, accounting for softmax.
    Returns the gradient w.r.t. z (the log-parameterization).
    dL/dz[j] = a[j]*(dL/da[j] - sum_i dL/da[i]*a[i])
    where dL/da[i] = 2*(h[i] + a[i]*S[i])
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    # S[i] = sum_{k=i}^{i+n-1} w[k]
    C = np.zeros(2 * n)
    C[1:] = np.cumsum(w)
    S = C[n:2 * n] - C[0:n]
    dlda = 2.0 * (h + a * S)
    mean_term = float(np.dot(dlda, a))
    return a * (dlda - mean_term)


def _conv_grad_approx(a, w):
    """Approximate gradient (faster, ignores S terms). Good for large n."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, _full_convolve(a, a)))
    return a * (h - W)


class _LseAccurate:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        return 2.0 * self.n * _conv_grad_full(a, w)

def _run_lbfgs_adaptive(a_init, deadline, use_accurate=False):
    """L-BFGS with adaptive beta based on convolution flatness."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    # Estimate flatness: how close are top values to max?
    top_100 = c_sorted[:min(100, len(c_sorted))]
    flatness = 1.0 - (top_100[-1] / max_c) if max_c > 0 else 1.0
    gap = max_c * max(0.001, flatness * 0.5)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Adaptive beta schedule: start large, decrease based on convergence
    n_stage = 20
    betas = []
    beta = gap
    for i in range(n_stage):
        betas.append(beta)
        # Decrease more aggressively if convergence is fast
        decay = 0.55 if i > n_stage // 2 else 0.65
        beta *= decay
    # Ensure we go very small at the end
    for i in range(5):
        betas.append(max_c * 1e-7 * (0.3 ** i))

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5.0:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.0015 * scale
        remaining = deadline - time.monotonic() - 5.0
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 12))))
        if max_iter < 20:
            break

        if use_accurate:
            prob = _LseAccurate(beta, n)
        else:
            prob = _LseApprox(beta, n)

        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _qp_active_set(a_init, deadline, n_inner=5000):
    """QP-based active set: iterate (compute weights, solve weighted QP via L-BFGS on simplex)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Beta schedule for the QP iterations
    c0 = _full_convolve(a, a)
    max_c0 = float(c0.max())
    c_sorted = np.sort(c0)[::-1]
    gap = max_c0 * 0.01

    n_out = 12
    for outer in range(n_out):
        if time.monotonic() > deadline - 3.0:
            break

        beta = gap * (0.5 ** outer)
        if beta < max_c0 * 1e-8:
            beta = max_c0 * 1e-8

        # Compute current weights
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_exp = e.sum()
        if s_exp < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_exp

        # Solve weighted QP: minimize a^T W_w a subject to sum(a)=1, a>=0
        # Using L-BFGS on softmax parameterization
        class _QpProb:
            def __call__(self, z_q):
                a_q = _softmax(z_q)
                # Objective: sum_k w[k]*c[k] = a^T W_w a
                # Compute via: sum_k w[k] * sum_i a[i]*a[k-i]
                grad_vec = _conv_grad_approx(a_q, w)
                # The actual objective value:
                c_q = _full_convolve(a_q, a_q)
                return float(np.dot(w, c_q))

            def grad(self, z_q):
                a_q = _softmax(z_q)
                return 2.0 * _conv_grad_full(a_q, w)

        prob = _QpProb()
        remaining = deadline - time.monotonic() - 3.0
        scale = max(1.0, n / 8192.0)
        max_iter = min(n_inner, max(50, int(remaining / (0.002 * scale * 15))))

        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 80, 'maxls': 30}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, _score(best_a)


def _active_set_gd_v3(a_init, deadline, tol_start=0.003, tol_end=0.000005,
                      lr_start=0.02, top_k=10, use_curvature=True):
    """Improved active-set GD with curvature-aware step size and continuous weights."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall_count = 0
    max_stall = 200
    curvature_est = None

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Continuous weights: exponential decay from max
        tol_val = tol * c_max
        shifted = (c - c_max) / max(tol_val, c_max * 1e-10)
        w = np.exp(np.clip(shifted, -50, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        # S[i] = sum_{k=i}^{i+n-1} w[k]
        C = np.zeros(2 * n)
        C[1:] = np.cumsum(w)
        S = C[n:2 * n] - C[0:n]
        grad = 2.0 * (h + a * S)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Curvature estimation for adaptive step size
        if use_curvature:
            alpha = lr
            if curvature_est is not None and curvature_est > 1e-20:
                alpha = min(alpha, gnorm / (curvature_est * 2.0))
            alpha = max(alpha, 1e-10)
        else:
            alpha = min(lr, 1.0 / gnorm * 0.5)

        # Backtracking line search
        base_sc = current_sc
        improved = False
        old_a = a.copy()
        old_sc = base_sc

        for it in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-15:
                a = a_try
                improved = True
                # Update curvature estimate
                if use_curvature and sc_try < old_sc:
                    delta_a = a - old_a
                    delta_sc = sc_try - old_sc
                    da_norm = np.dot(delta_a, delta_a)
                    if da_norm > 1e-20:
                        curv = abs(delta_sc) / (da_norm * max(gnorm, 1e-15))
                        if curvature_est is None:
                            curvature_est = curv
                        else:
                            curvature_est = 0.8 * curvature_est + 0.2 * curv
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            tol = max(tol_end, tol * 0.85)
            lr *= 0.7
            curvature_est = None
            if no_improve >= max_stall:
                a = best_a.copy()
                z = np.log(np.maximum(a, 1e-30))
                z = z + 0.01 * np.random.default_rng(no_improve).standard_normal(n)
                z = z - z.mean()
                a = _softmax(z)
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.5
                no_improve = 0
                stall_count += 1
                if stall_count > 15:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.005)

    return best_a, _score(best_a)


def _peak_shave_v2(a, deadline, n_trials=500):
    """Direct peak shaving with smarter redistribution."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Contribution: a[i]*a[k_star-i] for valid i
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            k_lo, k_hi = 0, n - 1

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * j_range.copy()
        # Actually contrib[i] = a[i] * a[k_star - i]
        contrib = a[i_range] * a[k_star - i_range]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            continue

        # Strategy: reduce top contributors, distribute to low-contrib positions
        n_reduce = max(1, int(len(contrib) * 0.15))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        # Also identify positions that don't contribute to any top peak
        top_peaks = np.argsort(c)[::-1][:5]
        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.08
            a_try[idx] -= red
            reduction += red

        # Find positions far from all top peaks
        inj_mask = np.zeros(n, dtype=bool)
        for tp in top_peaks:
            lo = max(0, tp - n + 1)
            hi = min(n - 1, tp)
            inj_mask[lo:hi + 1] = True

        # Reduce injectivity mask: keep only positions that contribute to top peaks
        free_mask = ~inj_mask
        if free_mask.sum() < n // 10:
            # If too few free positions, use all non-top-contrib positions
            free_mask = np.ones(n, dtype=bool)
            for idx in top_idx:
                free_mask[idx] = False
                jv = k_star - idx
                if 0 <= jv < n:
                    free_mask[jv] = False

        if free_mask.sum() == 0:
            free_mask = np.ones(n, dtype=bool)
            for idx in top_idx:
                free_mask[idx] = False
            if free_mask.sum() == 0:
                continue

        a_try[free_mask] += reduction / free_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_boost":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "center_dip":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.exp(-20.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "atria":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t) + 0.15 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer targeting score < 1.5027."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS adaptive (approximate grad for speed) ===
    print(f"\n[P1] L-BFGS adaptive approx (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200.0, t_start + 250.0)
    a1 = _run_lbfgs_adaptive(best_a, p1_deadline, use_accurate=False)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_p1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: QP active set (accurate grad) ===
    print(f"\n[P2] QP active set (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 900.0, t_start + 450.0)
    a2, sc2 = _qp_active_set(best_a, p2_deadline)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [qp_p2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set GD v3 ===
    print(f"\n[P3] Active-set GD v3 (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 600.0, t_start + 650.0)
    a3, sc3 = _active_set_gd_v3(best_a, p3_deadline, tol_start=0.002, tol_end=0.000003,
                                 lr_start=0.015, top_k=10)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [gsd_p3] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3.5: L-BFGS accurate (short burst) ===
    if time.monotonic() < deadline - 500.0:
        print(f"\n[P3.5] L-BFGS accurate (t={time.monotonic()-t_start:.1f}s)")
        p35_deadline = min(deadline - 450.0, t_start + 750.0)
        a35 = _run_lbfgs_adaptive(best_a, p35_deadline, use_accurate=True)
        sc35 = _score(a35)
        if sc35 < best_score:
            best_a = a35.copy()
            best_score = sc35
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_acc] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak shaving ===
    print(f"\n[P4] Peak shaving v2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 400.0, t_start + 850.0)
    a4, sc4 = _peak_shave_v2(best_a, p4_deadline, n_trials=400)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [peak_p4] t={time.monotonic()-t_start:.1f}s")

    # Re-optimize after peak shaving
    if time.monotonic() < deadline - 350.0:
        p4b_deadline = min(deadline - 300.0, t_start + 950.0)
        a4b = _run_lbfgs_adaptive(best_a, p4b_deadline, use_accurate=False)
        sc4b = _score(a4b)
        if sc4b < best_score:
            best_a = a4b.copy()
            best_score = sc4b
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_post4] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_targets = [16384, 20480, 9216, 14336]
    if n_parent not in n_targets and n_parent != n_best:
        n_targets.append(n_parent)

        if time.monotonic() > deadline - 200.0:
            break
        if n_try == n_best:
            continue
        remaining = deadline - time.monotonic() - 100.0
        if remaining < 50:
            break

        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}, init={sc_init:.10f}")

        p5_deadline = min(deadline - 100.0, time.monotonic() + remaining * 0.4)
        a_opt = _run_lbfgs_adaptive(a_init, p5_deadline, use_accurate=False)

            best_a = a_opt.copy()
            best_score = sc_opt
            n_best = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

                                               tol_start=0.002, tol_end=0.00001,
                                               lr_start=0.01)
            if sc_pol < best_score:
                best_a = a_pol.copy()
                best_score = sc_pol
                n_best = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80.0:
        n_best = len(best_a)
        restarts = [
            ("pert_002", _perturb(best_a, 0.02, rng)),
            ("pert_005", _perturb(best_a, 0.05, rng)),
            ("edge_boost", _make_profile(n_best, "edge_boost")),
            ("center_dip", _make_profile(n_best, "center_dip")),

            ("cos_mod", _make_profile(n_best, "cosine_mod")),
            ("tapered", _make_profile(n_best, "tapered")),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 40.0:
                break
            remaining = deadline - time.monotonic() - 30.0
            if remaining < 20.0:
                break
            a_opt = _run_lbfgs_adaptive(a_r, deadline - 30.0, use_accurate=False)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine polishing ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        a_fin, sc_fin = _active_set_gd_v3(best_a, deadline - 3.0,
                                           tol_start=0.0003, tol_end=0.000001,
                                           lr_start=0.005, top_k=5)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_gs] t={time.monotonic()-t_start:.1f}s")

        # Final peak shave
        if time.monotonic() < deadline - 2.0:
            a_ps, sc_ps = _peak_shave_v2(best_a, deadline - 1.0, n_trials=100)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  NEW BEST: {best_score:.10f} [final_ps] t={time.monotonic()-t_start:.1f}s")

        # Final tiny L-BFGS with very small beta
        remaining = deadline - 1.0 - time.monotonic()
        if remaining > 5.0:
            n = len(best_a)
            c_best = _full_convolve(best_a, best_a)
            mb = float(c_best.max())
            beta_fin = mb * 5e-6
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
            prob = _LseApprox(beta_fin, n)
            max_iter = min(500, max(10, int(remaining / 0.05)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 50})

                if s_fin2 > 1e-10:

                        best_score = sc_fin2

    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
