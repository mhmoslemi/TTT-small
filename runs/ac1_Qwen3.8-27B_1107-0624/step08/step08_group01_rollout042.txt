```
class _LSEProblem:
    def __init__(self, beta, n):
       self.beta=float(beta)
       self.n=int(n)
       self.N=_next_pow2(2*n-1)
       freqs=np.arange(self.N//2+1, dtype=np.float64)
       self.phase=np.exp(-2j*np.pi*freqs*(n-1)/self.N)
       self._zc=None; self._a=None; self._Fa=None; self._C=None; self._w=None; self._val=None
    def _state(self,z):
       zc=np.asarray(z,dtype=np.float64)
       zc=zc - float(np.mean(zc))
       e=np.exp(zc - float(np.max(zc)))
       s=e.sum()
       if s<=1e-300 or not np.isfinite(s):
          a=np.ones(self.n)/self.n
       else:
          a=e/s

       self._Fa=Fa
```

```
def _random_fourier_z(n, rng):
    n=max(2,int(n))
    Q=int(rng.integers(3, min(16,n//2)+1)) if n//2>=3 else 1
    freqs=rng.choice(np.arange(1, n//2+1), size=min(Q,n//2), replace=False)
    phases=rng.uniform(0.0, 2.0*np.pi, size=len(freqs))
    amps=rng.uniform(0.2, 1.0, size=len(freqs))
    idx=np.arange(n, dtype=np.float64)

        z += amps[ii] * np.cos(2.0*np.pi*freqs[ii]*idx/n + phases[ii])
    return z * float(rng.uniform(0.2, 1.0))
```

```python
"""
Search for a nonnegative step-function coefficient sequence minimizing
    2*n*max(conv(a,a)) / (sum(a)**2).

The program combines several complementary ideas:
  * L-BFGS minimization of a smoothed max-constraint objective in log-space.
  * Temperature annealing over different soft-max temperatures.
  * Quick random / Fourier / chirp starts and spectral water-filling projections.
  * Pooling of near-optimal basins and convex mixing to reduce peak overlap.
  * Multi-length exploration and final polishing.

The required function is propose_candidate(seed, budget_s).
"""

import time
import numpy as np
from scipy.optimize import minimize


def _pool_score_key(item):
    return item[1]


def _next_pow2(n):
    k = 1
    n = int(n)
    while k < n:
        k <<= 1
    return max(2, k)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-30:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _convolve(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    if N is None:
        N = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, N)
    C = np.fft.irfft(Fa * Fa, N)
    return C[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.inf
    n = a.size
    C = _convolve(a)
    m = float(C.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.zeros(1, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.zeros_like(z)
    z = z - float(np.mean(z))
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def _resample(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    new_n = int(new_n)
    if new_n <= 0:
        return np.ones(1, dtype=np.float64)
    if a.size == new_n:
        return a.copy()
    if a.size == 0:
        return np.ones(new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    return _normalize(b)


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-300))
    z = z + float(sigma) * rng.standard_normal(a.size)
    return _softmax(z)


def _blend_log(a, z_rand, frac):
    za = np.log(np.maximum(a, 1e-300))
    z_rand = np.asarray(z_rand, dtype=np.float64)
    if z_rand.size != za.size:
        z_rand = _resample(z_rand, za.size)
    z = (1.0 - float(frac)) * za + float(frac) * z_rand
    return _softmax(z)


def _random_fourier_z(n, rng):
    n = int(n)
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    maxf = max(1, n // 2)
    Q = int(rng.integers(1, min(16, maxf) + 1))
    if Q > maxf:
        Q = maxf
    freqs = rng.choice(np.arange(1, maxf + 1), size=Q, replace=False)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=Q)
    amps = rng.uniform(0.2, 1.0, size=Q)
    idx = np.arange(n, dtype=np.float64)
    mat = 2.0 * np.pi * (idx[:, None] * freqs[None, :] / float(n)) + phases[None, :]
    z = np.cos(mat).dot(amps)
    z = z * float(rng.uniform(0.2, 1.0))
    return z


def _fast_random_search(a_init, deadline, rng):
    a = _normalize(a_init)
    n = a.size
    if n == 0:
        a = np.ones(256, dtype=np.float64) / 256.0
        n = a.size
    best = a.copy()
    best_sc = _score(a)
    while time.monotonic() < deadline:
        r = int(rng.integers(0, 3))
        if r == 0:
            scale = float(rng.uniform(0.2, 0.8))
            z = rng.normal(0.0, scale, n)
        elif r == 1:
            z = _random_fourier_z(n, rng)
        else:
            t = np.arange(n, dtype=np.float64) / float(n)
            alpha = float(rng.uniform(1.0, 12.0))
            beta = float(rng.uniform(0.0, 2.0 * np.pi))
            gamma = float(rng.uniform(0.0, 2.0 * np.pi))
            z = np.cos(2.0 * np.pi * alpha * t * t + beta * t + gamma)
            z = z * float(rng.uniform(0.2, 1.0))
            if rng.random() < 0.5:
                z = z + 0.2 * rng.normal(0.0, 1.0, n)
        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best = a_new.copy()
            best_sc = sc_new
    return best, best_sc


def _water_fill_threshold(C, total):
    C = np.asarray(C, dtype=np.float64)
    lo = float(C.min())
    hi = float(C.max())
    if hi <= lo:
        return hi
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.minimum(C, mid))) < float(total):
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def _spectral_search(a_init, deadline, rng):
    a = _normalize(a_init)
    n = a.size
    if n == 0:
        a = np.ones(256, dtype=np.float64) / 256.0
        n = a.size
    N = _next_pow2(2 * n - 1)
    best = a.copy()
    best_sc = _score(a)

    while time.monotonic() < deadline:
        C = _convolve(a, N)
        csum = float(C.sum())
        cmax = float(C.max())
        if csum <= 1e-15 or cmax <= 0.0:
            break
        kind = int(rng.integers(0, 3))
        if kind == 0:
            tau = float(rng.uniform(0.5, 1.0)) * cmax
            target = np.minimum(C, tau)
        elif kind == 1:
            tau = _water_fill_threshold(C, csum)
            target = np.minimum(C, tau)
        else:
            tau = float(rng.uniform(0.5, 0.95)) * cmax
            target = np.minimum(C, tau)
            target = 0.5 * target + 0.5 * C
        ts = float(target.sum())
        if ts <= 1e-15:
            continue
        target = target * (csum / ts)
        G = np.fft.rfft(target, N)
        F = np.sqrt(G)
        cand = np.fft.irfft(F, N)[:n]
        cand = np.maximum(cand, 0.0)
        cand = _normalize(cand)
        sc = _score(cand)
        if sc < best_sc - 1e-12:
            best = cand.copy()
            best_sc = sc
    return best, best_sc


class _LSEProblem:
    """Smoothed auto-convolution peak objective in log-amplitude parameters."""

    def __init__(self, beta, n):
        self.beta = float(beta)
        self.n = int(n)
        self.N = _next_pow2(2 * n - 1)
        freqs = np.arange(self.N // 2 + 1, dtype=np.float64)
        self.phase = np.exp(-2j * np.pi * freqs * (n - 1) / self.N)
        self._zc = None
        self._a = None
        self._Fa = None
        self._C = None
        self._w = None
        self._val = np.inf

    def _state(self, z):
        z = np.asarray(z, dtype=np.float64).copy()
        if not np.all(np.isfinite(z)):
            z = np.zeros_like(z)
        z = z - float(np.mean(z))

        zmax = float(np.max(z))
        e = np.exp(z - zmax)
        s = float(e.sum())
        if not np.isfinite(s) or s < 1e-300:
            a = np.ones(self.n, dtype=np.float64) / float(self.n)
        else:
            a = e / s

        self._a = a
        Fa = np.fft.rfft(a, self.N)
        self._Fa = Fa
        C = np.fft.irfft(Fa * Fa, self.N)[:2 * self.n - 1]
        C = np.maximum(C, 0.0)
        self._C = C

        cmax = float(C.max())
        beta = self.beta
        if cmax <= 0.0 or beta <= 0.0:
            w = np.zeros(C.size, dtype=np.float64)
            w[int(np.argmax(C))] = 1.0
            soft = cmax
        else:
            shifted = (C - cmax) / beta
            ee = np.exp(shifted)
            ss = float(ee.sum())
            if not np.isfinite(ss) or ss < 1e-300:
                w = np.zeros(C.size, dtype=np.float64)
                w[int(np.argmax(C))] = 1.0
                soft = cmax
            else:
                w = ee / ss
                soft = cmax + beta * float(np.log(ss))
        self._w = w
        val = 2.0 * self.n * soft
        if not np.isfinite(val):
            val = np.inf
        self._val = float(val)
        self._zc = z
        return self._val

    def fun(self, z):
        return self._state(z)

    def grad(self, z):
        z = np.asarray(z, dtype=np.float64).ravel().copy()
        if not np.all(np.isfinite(z)):
            z = np.zeros_like(z)
        z = z - float(np.mean(z))

        if self._zc is not None and z.shape == self._zc.shape and np.array_equal(z, self._zc):
            a = self._a
            w = self._w
            C = self._C
            Fa = self._Fa
        else:
            self._state(z)
            a = self._a
            w = self._w
            C = self._C
            Fa = self._Fa

        Fw = np.fft.rfft(w, self.N)
        H = np.fft.irfft(Fw * (self.phase * np.conj(Fa)), self.N)
        h = H[self.n - 1:2 * self.n - 1]
        W = float(np.dot(w, C))
        g = 4.0 * self.n * a * (h - W)
        g = g - float(np.mean(g))
        if not np.all(np.isfinite(g)):
            g = np.zeros(self.n, dtype=np.float64)
        return g


def _optimize_lse(a_init, deadline, total_budget, mode="anneal", tag=""):
    """
    Run annealed L-BFGS on the smoothed LSE objective.

    mode:
      'score'   -- move only when exact score improves (or almost neutral + LSE improves)
      'anneal'  -- allow very small score regression when LSE clearly improves
      'lse'     -- more exploratory, move whenever the LSE objective at this beta improves
    """
    a = _normalize(a_init)
    n = a.size
    if n == 0:
        a = np.ones(256, dtype=np.float64) / 256.0
        n = a.size

    best = a.copy()
    best_sc = _score(a)
    if tag:
        print(f"  [{tag}] start n={n} score={best_sc:.10f}")

    if total_budget <= 1.0 or time.monotonic() >= deadline:
        return best, best_sc

    z = np.log(np.maximum(a, 1e-300))
    z = z - float(np.mean(z))

    C = _convolve(a)
    cmax = float(C.max())
    if cmax <= 0.0:
        return best, best_sc

    L = C.size
    c_sorted = np.sort(C)[::-1]
    gap = 0.0
    for idx in (5, 20, 100, 500):
        if idx < L:
            g = float(cmax - c_sorted[idx])
            if g > gap:
                gap = g
    if gap <= cmax * 1e-3:
        gap = cmax * 1e-3

    start_beta = max(cmax * 0.05, 3.0 * gap, cmax * 1e-4)
    min_beta = max(cmax * 1e-6, 1e-35)

    betas = []
    b = start_beta
    while b > min_beta:
        betas.append(b)
        if b < 2.0 * min_beta:
            break
        b *= 0.7
    betas.append(min_beta)

    if len(betas) > 32:
        step = max(1, len(betas) // 32)
        betas = betas[::step]

    weights = np.linspace(0.4, 2.0, len(betas), dtype=np.float64)
    stage_budget = total_budget * weights
    stage_budget = stage_budget / max(1e-30, float(stage_budget.sum()))

    score_tol = 1e-6 if mode == "score" else (1e-4 if mode == "anneal" else 1e-2)

    for i, beta in enumerate(betas):
        tnow = time.monotonic()
        if tnow >= deadline - 1.5:
            break
        sb = float(stage_budget[i])
        stage_end = min(deadline - 1.5, tnow + max(2.0, sb))
        if stage_end - tnow < 1.5:
            continue

        prob = _LSEProblem(float(beta), n)
        val0 = prob.fun(z)
        if not np.isfinite(val0):
            continue

        secs = stage_end - time.monotonic()
        scale = max(1.0, float(prob.N) / 32768.0)
        maxfun = int(min(300000, max(300, secs / (0.002 * scale))))
        maxiter = int(min(20000, max(30, maxfun // 20)))

        try:
            res = minimize(
                prob,
                z,
                jac=prob.grad,
                method="L-BFGS-B",
                options={
                    "maxiter": maxiter,
                    "maxfun": maxfun,
                    "ftol": 1e-16,
                    "gtol": 1e-12,
                    "maxcor": 40,
                    "maxls": 50,
                    "disp": False,
                },
            )
            z_new = np.asarray(res.x, dtype=np.float64)
            a_new = _softmax(z_new)
            a_new = np.maximum(a_new, 0.0)
            s_new = float(a_new.sum())
            if s_new <= 1e-30:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            lse_new = float(res.fun) if hasattr(res, "fun") and np.isfinite(res.fun) else np.inf

            if sc_new < best_sc - 1e-13:
                best = a_new.copy()
                best_sc = sc_new
                z = z_new.copy()
                if tag:
                    print(
                        f"  [{tag}] #beta={float(beta):.3e} NEW score={best_sc:.10f} "
                        f"t={time.monotonic():.1f}"
                    )
            elif mode == "lse" and lse_new < val0 - 1e-12:
                z = z_new.copy()
            elif sc_new < best_sc + score_tol and lse_new < val0 - 1e-10:
                z = z_new.copy()
        except Exception:
            pass

    return best, best_sc


def _add_pool(pool, a, sc, ref_n, tol=5e-4, max_pool=6):
    """Add a sequence to the pool if it is competitive and sufficiently distinct."""
    if pool is None:
        return pool
    if a is None or not np.isfinite(sc):
        return pool

    ar = _resample(a, ref_n) if a.size != ref_n else a.copy()
    sc_r = _score(ar)
    if not np.isfinite(sc_r):
        return pool

    if len(pool) == 0:
        pool.append((ar, sc_r))
        return pool

    ref_best = min(item[1] for item in pool)
    if sc_r >= ref_best + tol:
        return pool

    replaced = False
    for k in range(len(pool)):
        b, sb = pool[k]
        br = _resample(b, ref_n) if b.size != ref_n else b
        d = float(np.linalg.norm(ar - br))
        if d < 0.05:
            if sc_r < sb - 1e-12:
                pool[k] = (ar, sc_r)
            replaced = True
            break

    if not replaced:
        pool.append((ar, sc_r))
        if len(pool) > max_pool:
            worst_idx = 0
            worst_sc = -np.inf
            for idx in range(len(pool)):
                if pool[idx][1] > worst_sc:
                    worst_sc = pool[idx][1]
                    worst_idx = idx
            pool.pop(worst_idx)

    pool.sort(key=_pool_score_key)
    return pool


def _mix_pool(pool, n_ref, deadline):
    if pool is None or len(pool) == 0:
        return None, np.inf

    resampled = []
    for a, sc in pool:
        ar = _resample(a, n_ref) if a.size != n_ref else a.copy()
        sc_r = _score(ar)
        resampled.append((ar, sc_r))

    best_a = None
    best_sc = np.inf
    for ar, sc in resampled:
        if sc < best_sc:
            best_a = ar.copy()
            best_sc = sc

    alphas = (0.1, 0.25, 0.5, 0.75, 0.9)
    m = len(resampled)
    for i in range(m):
        for j in range(i + 1, m):
            if time.monotonic() > deadline:
                return best_a, best_sc
            a1, _ = resampled[i]
            a2, _ = resampled[j]
            for alpha in alphas:
                cand = float(alpha) * a1 + (1.0 - float(alpha)) * a2
                cand = _normalize(cand)
                sc = _score(cand)
                if sc < best_sc - 1e-13:
                    best_a = cand.copy()
                    best_sc = sc
    return best_a, best_sc


def _consider_candidate(pool, best, best_sc, pool_n, a_new, sc_new, add_tol=5e-4, tag="", verbose=True):
    if a_new is None or sc_new is None or not np.isfinite(sc_new):
        return best, best_sc, pool, pool_n

    a_new = np.asarray(a_new, dtype=np.float64)
    a_new = np.where(np.isfinite(a_new), a_new, 0.0)
    a_new = np.maximum(a_new, 0.0)

    if sc_new < best_sc - 1e-12:
        best = _normalize(a_new)
        best_sc = _score(best)
        pool_n = int(best.size)
        pool = _add_pool(pool, best, best_sc, pool_n, 1e-8)
        if verbose and tag:
            print(f"  [{tag}] NEW BEST score={best_sc:.10f} n={pool_n}")
    else:
        if pool_n is None or pool_n <= 0:
            pool_n = int(best.size) if best is not None else 256
        pool = _add_pool(pool, a_new, sc_new, pool_n, add_tol)

    return best, best_sc, pool, pool_n


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Uses budget_s seconds of wall-clock time and returns
    a list of nonnegative floats.
    """
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget - 8.0
    scale_time = max(0.15, budget / 1800.0)

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent if available.
    parent = None
    try:
        p_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p_raw.size > 0:
            p_raw = np.where(np.isfinite(p_raw), p_raw, 0.0)
            p_raw = np.maximum(p_raw, 0.0)
            s = float(p_raw.sum())
            if s > 0.01:
                parent = p_raw / s
    except Exception:
        parent = None

    if parent is not None:
        best = parent.copy()
        best_sc = _score(best)
        n_best = int(best.size)
        print(f"Parent loaded: n={n_best} score={best_sc:.10f}")
    else:
        n_default = 8192 if budget > 600 else 2048
        best = np.ones(n_default, dtype=np.float64) / float(n_default)
        best_sc = _score(best)
        n_best = int(best.size)
        print(f"No parent. Initial uniform n={n_best} score={best_sc:.10f}")

    pool = []
    pool_n = n_best
    pool = _add_pool(pool, best, best_sc, pool_n, 1e-8)

    # ------------------------------------------------------------
    # 1. Quick random/Fourier/chirp starts.
    # ------------------------------------------------------------
    if budget > 250:
        end = min(deadline, time.monotonic() + 25.0 * scale_time)
        if end > time.monotonic() + 3.0:
            a_rand, sc_rand = _fast_random_search(best, end, rng)
            best, best_sc, pool, pool_n = _consider_candidate(
                pool, best, best_sc, pool_n, a_rand, sc_rand, 2e-4, tag="random"
            )

    # ------------------------------------------------------------
    # 2. Spectral water-filling / deconvolution projection starts.
    # ------------------------------------------------------------
    if budget > 250:
        end = min(deadline, time.monotonic() + 20.0 * scale_time)
        if end > time.monotonic() + 3.0:
            a_spec, sc_spec = _spectral_search(best, end, rng)
            best, best_sc, pool, pool_n = _consider_candidate(
                pool, best, best_sc, pool_n, a_spec, sc_spec, 2e-4, tag="spectral"
            )

    # ------------------------------------------------------------
    # 3. Main annealed L-BFGS on the current best sequence.
    # ------------------------------------------------------------
    tnow = time.monotonic()
    remaining = max(0.0, deadline - tnow)
    main_alloc = 600.0 * scale_time
    main_alloc = min(main_alloc, max(0.0, remaining - 350.0 * scale_time))
    main_alloc = max(main_alloc, min(60.0, remaining * 0.4))

    if main_alloc > 10.0:
        end = min(deadline, time.monotonic() + main_alloc)
        a_main, sc_main = _optimize_lse(best, end, main_alloc, mode="anneal", tag="main")
        best, best_sc, pool, pool_n = _consider_candidate(
            pool, best, best_sc, pool_n, a_main, sc_main, 5e-5, tag="main"
        )

    # ------------------------------------------------------------
    # 4. Multi-length exploration.
    # ------------------------------------------------------------
    tnow = time.monotonic()
    remaining = max(0.0, deadline - tnow)
    if remaining > 120.0:
        lengths = [n_best, 8192, 12288, 16384, 20480]
        clean = []
        seen = set()
        for x in lengths:
            x = int(x)
            if 1024 <= x <= 20480 and x not in seen:
                seen.add(x)
                clean.append(x)

        quick = []
        for n in clean:
            ar = _resample(best, n)
            sc = _score(ar)
            quick.append((sc, n, ar))
        quick.sort(key=lambda item: item[0])

        per_try = min(160.0 * scale_time, max(20.0, remaining / max(1, len(quick))))
        tried = 0
        for sc_try, n_try, a_try in quick:
            tnow = time.monotonic()
            rem = max(0.0, deadline - tnow)
            if rem < 30.0 or tried >= 3:
                break
            if abs(sc_try - best_sc) > 0.05 and n_try == n_best:
                continue
            alloc = min(per_try, rem - 20.0)
            if alloc < 15.0:
                break
            end = min(deadline, tnow + alloc)
            a_opt, sc_opt = _optimize_lse(a_try, end, alloc, mode="anneal", tag=f"len{n_try}")
            if sc_opt < best_sc + 1e-4:
                best, best_sc, pool, pool_n = _consider_candidate(
                    pool, best, best_sc, pool_n, a_opt, sc_opt, 5e-4, tag=f"len{n_try}"
                )
            tried += 1

    # ------------------------------------------------------------
    # 5. Diverse restarts from the best and pool.
    # ------------------------------------------------------------
    tnow = time.monotonic()
    remaining = max(0.0, deadline - tnow)
    if remaining > 120.0:
        restarts = []

        for sigma in (0.03, 0.08, 0.15, 0.30):
            restarts.append(_perturb_log(best, sigma, rng))

        z_rand = _random_fourier_z(best.size, rng)
        restarts.append(_blend_log(best, z_rand, 0.25))
        restarts.append(_blend_log(best, z_rand, 0.50))

        # Add a few pool candidates (resampled to current best length).
        for a, sc in pool:
            if a.size != best.size:
                a = _resample(a, best.size)
            restarts.append(a)
            if len(restarts) >= 8:
                break

        # De-duplicate a little by norm.
        uniq = []
        for a in restarts:
            d = 0.0
            keep = True
            for b in uniq:
                if a.size == b.size:
                    d = max(d, float(np.linalg.norm(a - b)))
            if len(uniq) == 0 or d > 0.08:
                uniq.append(a)
        restarts = uniq[:5]

        per_try = min(80.0 * scale_time, max(15.0, remaining / max(1, len(restarts))))
        for i, a_r in enumerate(restarts):
            tnow = time.monotonic()
            rem = max(0.0, deadline - tnow)
            if rem < 25.0:
                break
            alloc = min(per_try, rem - 15.0)
            if alloc < 10.0:
                break
            end = min(deadline, tnow + alloc)
            a_opt, sc_opt = _optimize_lse(a_r, end, alloc, mode="lse", tag=f"restart{i}")
            if sc_opt < best_sc + 5e-4:
                best, best_sc, pool, pool_n = _consider_candidate(
                    pool, best, best_sc, pool_n, a_opt, sc_opt, 5e-4, tag=f"restart{i}"
                )

    # ------------------------------------------------------------
    # 6. Pool mixing after exploration.
    # ------------------------------------------------------------
    tnow = time.monotonic()
    deadline_mix = min(deadline, tnow + max(5.0, 30.0 * scale_time))
    a_mix, sc_mix = _mix_pool(pool, int(best.size), deadline_mix)
    if a_mix is not None and np.isfinite(sc_mix):
        best, best_sc, pool, pool_n = _consider_candidate(
            pool, best, best_sc, pool_n, a_mix, sc_mix, 5e-5, tag="mix"
        )

    # ------------------------------------------------------------
    # 7. Final ultra-polishing.
    # ------------------------------------------------------------
    tnow = time.monotonic()
    final_alloc = max(0.0, deadline - 2.0 - tnow)
    if final_alloc > 10.0:
        end = deadline - 2.0
        a_fin, sc_fin = _optimize_lse(best, end, final_alloc, mode="score", tag="final")
        if np.isfinite(sc_fin) and sc_fin < best_sc + 1e-8:
            best, best_sc, pool, pool_n = _consider_candidate(
                pool, best, best_sc, pool_n, a_fin, sc_fin, 1e-8, tag="final"
            )

    # ------------------------------------------------------------
    # Final sanity check and return.
    # ------------------------------------------------------------
    final = np.asarray(best, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    final = np.minimum(final, 1000.0)

    s = float(final.sum())
    if final.size == 0 or not np.isfinite(s) or s < 0.01:
        final = np.ones(256, dtype=np.float64) / 256.0

    # Optionally verify with the official evaluator if it is available.
    try:
        ev = evaluate_sequence([float(x) for x in final])
        if np.isfinite(ev):
            print(f"Final validated by evaluate_sequence: {ev:.10f}")
    except Exception:
        pass

    print(f"FINAL internal score={float(_score(final)):.10f} n={int(final.size)} "
          f"time={time.monotonic() - t0:.2f}s")

    return [float(x) for x in final]
```
