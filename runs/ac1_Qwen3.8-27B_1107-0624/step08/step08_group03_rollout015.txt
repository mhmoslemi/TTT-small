def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _build_reduced_conv_matrix(a, top_idx):
    """Build sparse constraint matrix for top K autocorrelation peaks."""
    n = len(a)
    K = len(top_idx)
    rows = []
    cols = []
    vals = []
    for kk in range(K):
        k = int(top_idx[kk])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        if length > 0:
            rows.extend([kk] * length)
            cols.extend(range(i_start, i_end + 1))
            vals.extend(a[k - i_start:k - i_start + length].tolist())
    if len(vals) == 0:
        return csr_matrix((K, n))
    A = coo_matrix((np.array(vals, dtype=np.float64),
                    (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                   shape=(K, n)).tocsr()
    return A


def _lp_direction(a, top_k=300):
    """Solve reduced LP to find descent direction. Returns b or None."""
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-15:
        return None
    top_idx = np.argsort(c)[::-1][:top_k]
    A = _build_reduced_conv_matrix(a, top_idx)
    c_obj = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    return None


def _lp_phase(a_init, deadline, max_iters=20, top_k=300):
    """Iterative LP-based optimization phase."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    print(f"    LP phase start: n={n}, score={best_sc:.10f}")

    for it in range(max_iters):
        if time.monotonic() > deadline - 30:
            break
        b = _lp_direction(a, top_k=top_k)
        if b is None:
            print(f"    LP phase: iteration {it}, LP failed, stopping")
            break

        # Line search: a_new = (1-t)*a + t*b
        f_old = float(_full_convolve(a, a).max())
        accepted = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            f_try = float(_full_convolve(a_try, a_try).max())
            sc_try = _score(a_try)
            if f_try < f_old - 1e-12 or t == 0.1:
                a = a_try
                accepted = True
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    print(f"    LP iter {it}: t={t}, score={best_sc:.10f}, "
                          f"peak={f_try:.6e}, improve={f_old - f_try:.2e}")
                break
        if not accepted:
            print(f"    LP phase: iteration {it}, no improvement, stopping")
            break

        # Check convergence
        c = _full_convolve(a, a)
        f_new = float(c.max())
        if f_new < f_old * (1 - 1e-6):
            pass
        else:
            if abs(f_new - f_old) < 1e-10 * f_old:
                print(f"    LP phase: converged at iteration {it}")
                break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a


def _active_set_newton(a_init, deadline, max_iter=500, tol_active=0.001):
    """Active-set Newton method with KKT convergence check."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            top_idx = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 4 * n

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient: g_i = (2/m) * sum_{k in active} a_{k-i} (valid range)
        flip_a = a[::-1]
        out_len = n + m - 1  # Actually n + n_active - 1 but we use full length
        N = _next_pow2(n + n - 1)
        fa = np.fft.rfft(flip_a, N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fa * fA, N)[:n + n - 1]
        g = (2.0 / m) * result[n - 1:2 * n - 1]

        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-5 * (c_max / max(m, 1)) + 1e-8

        # Hessian matrix-vector product: H*v = (2/m) * conv(v, A_vec) restricted + mu*v
        def hess_matvec(v):
            N2 = _next_pow2(2 * n - 1)
            fv = np.fft.rfft(v, N2)
            fA2 = np.fft.rfft(A_vec, N2)
            res2 = np.fft.irfft(fv * fA2, N2)[:2 * n - 1]
            return (2.0 / m) * res2[n - 1:2 * n - 1] + mu * v

        # CG solve for Newton direction
        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old_stage = c_max
        accepted = False
        t = min(1.0, t_max)
        for _ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old_stage - 1e-13 * t * gnorm or _ls == 49:
                if f_try <= f_old_stage + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Gradient fallback
            t = min(0.1, t_max)
            for _ls in range(25):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old_stage - 1e-13 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-11 * c_max:
            break

    return best_a, best_sc


def _block_coordinate_descent(a_init, deadline, block_size=256, max_rounds=200):
    """Block coordinate descent: optimize blocks of coordinates at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    n_blocks = (n + block_size - 1) // block_size

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Process blocks in random order
        block_order = list(range(n_blocks))
        np.random.shuffle(block_order)

        improved = False
        for blk in block_order:
            if time.monotonic() > deadline:
                break
            start = blk * block_size
            end = min(n, start + block_size)
            bs = end - start

            # Compute gradient for this block
            # g_i = 2 * sum_{k in active} w_k * a_{k-i}
            # Use top peaks
            tol = 0.002 * c_max
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            w = np.zeros_like(c)
            w[active] = 1.0 / active.sum()

            # For block positions i in [start, end):
            # g_i = 2 * sum_k w_k * a_{k-i} where 0 <= k-i < n
            # = 2 * sum_k w_k * a_{k-i}
            # For a fixed i, k-i ranges over [0, n-1], so k ranges over [i, i+n-1]
            # We need a_{k-i} for the block, which involves a at positions k-i
            # This is essentially a correlation

            # Simplified: compute gradient for the block using FFT
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad_all = 2.0 * a * (h - W)

            # For the block, do a small Newton step
            g_blk = grad_all[start:end].copy()
            g_blk_mean = np.mean(g_blk)
            g_blk_t = g_blk - g_blk_mean

            # Simple gradient step with line search
            step = 0.05
            f_old = c_max
            for _ in range(20):
                a_try = a.copy()
                a_try[start:end] = np.maximum(a[start:end] - step * g_blk_t, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    step *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-14:
                    a = a_try
                    improved = True
                    break
                step *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                      0.015, 0.0075, 0.0037, 0.00185, 0.0009,
                                      0.00045, 0.00022, 0.00011, 0.00005,
                                      0.000025, 0.0000125, 0.000006])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.002
        elif n <= 8192:
            est_time_per_iter = 0.004
        elif n <= 16384:
            est_time_per_iter = 0.01
        else:
            est_time_per_iter = 0.02

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 0.5 + 2.0 * t ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: LP-based on n=8192 (fast, robust)
    Phase 2: Interpolate to n=16384, L-BFGS smoothing
    Phase 3: Active-set Newton fine-tuning
    Phase 4: Block coordinate descent
    Phase 5: Subgradient polish and final Newton
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    np.random.seed(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if
