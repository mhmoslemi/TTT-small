def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):

    out_n = n + m - 1

"""
Multi-resolution optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
1. Multi-resolution initialization: solve at small n, upsample to refine
2. L-BFGS with adaptive log-sum-exp smoothing for main optimization
3. Newton's method with careful active set management and dual tracking for final polish
4. Diversification via perturbations, different n values, and restarts
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_precond(W, n, rhs, max_iter=600, tol=1e-14):
    """Preconditioned CG with diagonal preconditioner for Hankel system."""
    n_int = n
    # Diagonal of H: H_ii = 2 * W[2i] (if 2i < len(W)), else 0
    diag_idx = 2 * np.arange(n_int)
    diag_val = np.where(diag_idx < len(W), 2.0 * W[diag_idx], 0.5)
    d_inv = 1.0 / np.maximum(diag_val, 1e-30)

    x = np.zeros(n_int)
    r = rhs.copy()
    z = r * d_inv
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x

    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n_int)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        z_new = r_new * d_inv
        rz_new = float(r_new @ z_new)
        if not np.isfinite(rz_new) or rz_new < -1e-20:
            # Loss of positive definiteness, stop
            break
        if rz_new < tol * tol * max(rz0, 1e-30):
            x = x_new
            break
        beta = rz_new / max(rz, 1e-30)
        p = z_new + beta * p
        x = x_new
        r = r_new
        z = z_new
        rz = rz_new
        # Periodic restart
        if i % 50 == 49:
            p = z_new.copy()
    return x


def _newton_active_polish(a_init, deadline, max_rounds=200, verbose=False):
    """Newton's method with active set identification and dual tracking."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_range = c_max - c.min()
        if c_range < 1e-14 * c_max:
            break

        # Active set: lags within tight tolerance of max
        # Use adaptive tolerance: start tight, relax if too few active
        best_active = None
        for tol_frac in [0.000005, 0.00001, 0.00003, 0.0001, 0.0003, 0.001, 0.005, 0.01]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if 1 <= n_active <= min(n, 5000):
                best_active = active
                break
        if best_active is None:
            best_active = c >= (c_max - 0.01 * c_max)
        if best_active.sum() == 0:
            best_active[int(np.argmax(c))] = True

        n_active = int(best_active.sum())
        # If active set is too large, restrict to most extreme
        if n_active > 3000:
            thresholds = np.sort(c[best_active])[::-1][:3000]
            best_active = c >= thresholds[-1]
            n_active = int(best_active.sum())

        W = np.zeros(2 * n - 1)
        W[best_active] = 1.0 / n_active

        # Compute gradient: g_i = 2 * sum_k W[k] * a_{k-i} (clipped)
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))

        # Check KKT residual
        if gn < 1e-12 * c_max:
            break

        # Solve H * dx = rhs where rhs = -g + lambda*1
        # Use CG for H * x = g and H * x = 1
        Hg = _cg_solve_precond(W, n, g, max_iter=500)
        H1 = _cg_solve_precond(W, n, np.ones(n), max_iter=500)

        H1_sum = float(np.ones(n) @ H1)
        if abs(H1_sum) < 1e-20:
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / H1_sum
            dx_raw = -Hg + lam * H1
            dx = dx_raw - np.mean(dx_raw)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Check dual feasibility: for a_i = 0, need g_i >= lambda
        # (i.e., 2*sum W[k]*a_{k-i} >= lambda for inactive i)
        # We handle this by clamping: only step in positive direction where a_i = 0

        # Line search with cubic interpolation
        alpha = 1.0
        step_done = False
        best_alpha = 0.0
        best_c_try = c_max

        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())

            if c_max_try < best_c_try:
                best_c_try = c_max_try
                best_alpha = alpha

            # Sufficient decrease condition
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-15:
                a = a_try
                step_done = True
                break

            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done and best_alpha > 0 and best_c_try < c_max - 1e-15:
            # Take the best step found
            a_try = a + best_alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try > 1e-15:
                a = a_try / s_try
                step_done = True
            else:
                best_alpha = 0.0

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and rnd % 20 == 0:
            print(f"    Newton round {rnd}: score={sc:.10f}, c_max={c_max:.10f}, "
                  f"n_active={n_active}, gn={gn:.2e}, step={step_done}")

        if not step_done and no_improve > 8:
            break
        if no_improve > 30:
            break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_phase(a_init, deadline, min_remaining=5.0, max_beta_iters=8):
    """Adaptive L-BFGS with automatic beta reduction."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.3, max_c * 0.002)
    min_beta = max_c * 1e-10

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    prev_best = current_sc

    for _ in range(max_beta_iters):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 8192.0)
        est_per_iter = 0.002 * scale
        max_iter = min(5000, max(20, int(remaining / (est_per_iter * 30))))

        try:
            result = minimize(
                _SmoObjAdapter(beta, n), z, jac=_SmoGradAdapter(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adapt beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_s_new = np.sort(c_new)[::-1]
        new_gap = c_s_new[0] - c_s_new[min(20, len(c_s_new) - 1)]

        sc_improved = prev_best - current_sc
        prev_best = current_sc

        if new_gap < 1e-9 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        elif sc_improved < 1e-13:
            beta *= 0.3
        else:
            beta *= 0.35

        beta = min(beta, c_max_new * 0.0003)
        if beta <= min_beta:
            break

    return current_a


def _interpolate_up(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a, period=None)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_initial(n, kind="sine_taper"):
    """Generate initial profile."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sine_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_taper_15":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangular":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = (1.0 - (2.0 * t - 1.0) ** 2) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_sine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 1.2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()


def _perturb(a, sigma, rng):
    """Perturb by adding noise in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _multi_resolution_init(n_target, deadline, min_remaining=100.0, verbose=True):
    """Build good initialization by solving at increasing n."""
    levels = [64, 128, 256, 512, 1024, 2048, 4096, 8192]
    # Add target if not already there
    if n_target not in levels:
        levels.append(n_target)
        levels.sort()
    # Keep only levels up to n_target
    levels = [l for l in levels if l <= n_target]
    if not levels or levels[-1] < n_target:
        levels.append(n_target)

    if verbose:
        print(f"  Multi-res levels: {levels}")

    current_a = _make_initial(levels[0], "sine_taper")
    for i, n_lvl in enumerate(levels):
        if time.monotonic() + min_remaining > deadline:
            break
        if n_lvl > levels[0]:
            current_a = _interpolate_up(current_a, n_lvl)

        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        # Brief L-BFGS at this level
        time_alloc = min(remaining * 0.3, 30.0)
        if time_alloc > 2:
            current_a = _run_lbfgs_phase(current_a, time.monotonic() + time_alloc,
                                          min_remaining=1.0, max_beta_iters=3)

        # Brief Newton
        if time.monotonic() + min_remaining < deadline:
            current_a, _ = _newton_active_polish(current_a,
                                                  min(time.monotonic() + 5, deadline - min_remaining),
                                                  max_rounds=10)

    if len(current_a) != n_target:
        current_a = _interpolate_up(current_a, n_target)

    return current_a

    # Phase 7: Try different n values

    print(f"Phase 7: Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if best_seq is not None:
        n_tries = [n_parent * 2, n_parent // 2, n_parent + 1000, n_parent - 1000]
        n_tries = [n for n in n_tries if 512 <= n <= 65536]
        for n_try in n_tries:
            if time.monotonic() > deadline - 30:
                break
            a_resample = _interpolate_up(best_seq, n_try)
            a_opt = _run_lbfgs_phase(a_resample, deadline, min_remaining=15.0, max_beta_iters=4)
            update_best(a_opt, _score(a_opt), f"n_{n_try}")
            if time.monotonic() < deadline - 10:
                a_opt_np, _ = _newton_active_polish(a_opt, min(deadline - 5, time.monotonic() + 10),
                                                     max_rounds=30)
                update_best(a_opt_np, _score(a_opt_np), f"n_{n_try}_np")

    # Phase 8: Try different initial shapes at target n
    print(f"\n{'='*60}")
    print(f"Phase 8: Shape exploration (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 30:
        shapes = ["sine_taper_15", "quartic", "gaussian", "asym_sine", "triangular"]
        for shape in shapes:
            if time.monotonic() > deadline - 20:
                break
            a_shape = _make_initial(n_parent, shape)
            # Blend with best
            a_blend = 0.5 * a_shape + 0.5 * best_seq
            a_blend = a_blend / a_blend.sum()
            sc_blend = _score(a_blend)
            if sc_blend < best_score:
                update_best(a_blend, sc_blend, f"blend_{shape}")

            if time.monotonic() < deadline - 15:
                a_shape_opt = _run_lbfgs_phase(a_blend, deadline, min_remaining=10.0, max_beta_iters=3)
                update_best(a_shape_opt, _score(a_shape_opt), f"shape_lbfgs_{shape}")

    # Phase 9: Final intensive Newton
    print(f"\n{'='*60}")
    print(f"Phase 9: Final Newton (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 5:
        a_final_np, sc_final_np = _newton_active_polish(best_seq, deadline - 1, max_rounds=200)
        update_best(a_final_np, sc_final_np, "final_newton")

    # Phase 10: Ultra-fine L-BFGS
    if best_seq is not None and time.monotonic() < deadline - 2:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 1e-11
        remaining = deadline - 2 - time.monotonic()
        if remaining > 2:
            max_iter = min(3000, max(20, int(remaining / 0.005)))
            try:
                result = minimize(
                    _SmoObjAdapter(beta_fin, n), z, jac=_SmoGradAdapter(beta_fin, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxcor': 100, 'maxls': 200}
                )
                a_ultra = _softmax(result.x)
                update_best(a_ultra, _score(a_ultra), "ultra_lbfgs")
            except Exception:
                pass

    # Final Newton after everything
    if best_seq is not None and time.monotonic() < deadline - 0.5:
        a_very_final, _ = _newton_active_polish(best_seq, deadline - 0.1, max_rounds=50)
        update_best(a_very_final, _score(a_very_final), "very_final")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256, "sine_taper")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {ev_score:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    # Fallback
    fallback = _make_initial(8192, "sine_taper")
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}, n=8192")
    return [float(x) for x in fallback]
