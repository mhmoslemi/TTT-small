Optimized step-function search using Adam-based LSE optimization with
adaptive smoothing, spectral initialization, multi-scale approach,
and KKT-based active set polishing.
Targets improvement below 1.5027.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj_grad(z, beta):
    """LSE-smoothed objective and gradient in log-space."""
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))

def _init_profile(n, kind="raised_cosine"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = (np.arange(n) + 0.5) / n
        a = 0.5 * (1 - np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "parabolic":
        t = (np.arange(n) + 0.5) / n
        a = 4 * t * (1 - t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "misc_log":
        # Based on the "miscible log" or "optimal window" structure
        t = (np.arange(n) + 0.5) / n
        a = np.log(n * np.sin(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "prolate_approx":
        # Approximate prolate spheroidal: product of sinc-like functions
        t = (np.arange(n) + 0.5) / n - 0.5
        a = np.sinc(3 * t) ** 2 + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponentials":

        a = np.maximum(a, 1e-10)

    else:
        return np.ones(n) / n

def _upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _downsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:

    return a_new

def _next_pow2(n):
    k = 1

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    """Refined peak shaving: reduce top contributors, redistribute to low ones."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)
    rng = np.random.default_rng(777)

    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo >= k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = best_a[i_range] * best_a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            continue

        # Reduce fraction of top contributors
        frac = rng.uniform(0.05, 0.15)
        n_red = max(1, int(len(contrib) * frac))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_red]]

        a_try = best_a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.08
            a_try[idx] -= red
            reduction += red

        # Redistribute to non-contributing elements
        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
            jv = k_star - idx
            if 0 <= jv < n:
                low_mask[jv] = False

        if low_mask.sum() == 0:
            low_mask[:] = True
            for idx in top_idx:
                low_mask[idx] = False
            if low_mask.sum() == 0:
                continue

        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Multi-scale warm start (150s)
    P2: Main Adam+L-BFGS hybrid (480s)
    P3: KKT active-set polish (250s)
    P4: Diversification with varied inits (350s)
    P5: Multi-n exploration (250s)
    P6: Final polish (150s)
    Total ~1630s + buffer.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_p = float(parent_raw.sum())
            if s_p > 0.01:
                parent = parent_raw / s_p
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_target = len(parent)
        print(f"Parent: n={n_target}, score={best_score:.10f}")
    else:
        n_target = 12288
        best_a = _init_profile(n_target, "raised_cosine")
        best_score = _score(best_a)
        print(f"No parent: n={n_target}, raised_cosine, score={best_score:.10f}")

    # === P1: Multi-scale warm start ===
    print(f"\n[P1] Multi-scale warm start (t={time.monotonic()-t_start:.1f}s)")
    p1_end = t_start + 150.0
    scales = [max(256, n_target // 16), max(512, n_target // 8),
              max(1024, n_target // 4), max(2048, n_target // 2), n_target]
    seen = set()
    scales_f = []
    for sc in scales:
        if sc not in seen:
            seen.add(sc)
            scales_f.append(sc)

    for n_sc in scales_f:
        remaining_p1 = p1_end - time.monotonic() - 5.0
        if remaining_p1 < 15:
            break
        if len(best_a) != n_sc:
            a_sc = _downsample(best_a, n_sc)
        else:
            a_sc = best_a.copy()
        a_opt, sc_opt = _adam_optimize(a_sc, time.monotonic() + min(remaining_p1, 40.0),
                                        min_time=3.0, use_lbfgs=True, lbfgs_interval=200)
        if n_sc < n_target:
            a_up = _upsample(a_opt, n_target)
        else:
            a_up = a_opt
        sc_up = _score(a_up)
        print(f"  n={n_sc}: opt={sc_opt:.10f}, upsampled={sc_up:.10f}")
        if sc_up < best_score:
            best_a = a_up.copy()
            best_score = sc_up
            print(f"  NEW BEST: {best_score:.10f} [P1 multi_scale]")

    # === P2: Main hybrid optimization ===
    print(f"\n[P2] Main Adam+L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_end = t_start + 630.0
    a2, sc2 = _adam_optimize(best_a, p2_end, min_time=30.0, use_lbfgs=True, lbfgs_interval=300)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [P2]")
    else:
        print(f"  No improvement (score={sc2:.10f} vs best={best_score:.10f})")

    # === P3: KKT active-set polish ===
    print(f"\n[P3] KKT active-set (t={time.monotonic()-t_start:.1f}s)")
    p3_end = t_start + 880.0
    a3, sc3 = _kkt_active_set(best_a, p3_end, top_k=8)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [P3 KKT]")

    # Post-KKT refinement with short Adam
    if time.monotonic() < p3_end - 20:
        a3b, sc3b = _adam_optimize(best_a, p3_end - 5.0, min_time=5.0, use_lbfgs=True, lbfgs_interval=200)
        if sc3b < best_score:
            best_a = a3b.copy()
            best_score = sc3b
            print(f"  NEW BEST: {best_score:.10f} [P3b post-KKT adam]")

    # === P4: Diversification ===
    print(f"\n[P4] Diversification (t={time.monotonic()-t_start:.1f}s)")
    p4_end = t_start + 1230.0
    n_best = len(best_a)

    init_list = [
        ("parent_copy", best_a.copy()),
        ("raised_cos", _init_profile(n_best, "raised_cosine")),
        ("cos_blend", _init_profile(n_best, "cosine_blend")),
        ("misc_log", _init_profile(n_best, "misc_log")),
        ("power_taper", _init_profile(n_best, "power_taper")),
        ("perturb_01", _softmax(np.log(np.maximum(best_a, 1e-30)) + 0.01 * rng.standard_normal(n_best))),
        ("perturb_03", _softmax(np.log(np.maximum(best_a, 1e-30)) + 0.03 * rng.standard_normal(n_best))),
        ("perturb_08", _softmax(np.log(np.maximum(best_a, 1e-30)) + 0.08 * rng.standard_normal(n_best))),
    ]

    for label, a_init in init_list:
        remaining = p4_end - time.monotonic() - 10.0
        if remaining < 25:
            break
        allocate = min(100.0, remaining / max(1, len(init_list) - init_list.index((label, a_init))))
        a_opt, sc_opt = _adam_optimize(a_init, time.monotonic() + allocate,
                                        min_time=5.0, use_lbfgs=True, lbfgs_interval=250)
        sc_final = sc_opt
        # Quick KKT polish if improved
        if sc_opt < best_score and time.monotonic() < p4_end - 15:
            a_pol, sc_pol = _kkt_active_set(a_opt, p4_end - 10.0, top_k=5)
            if sc_pol < sc_opt:
                sc_final = sc_pol
                sc_opt = sc_pol
                a_opt = a_pol
        if sc_final < best_score:
            best_a = a_opt.copy()
            best_score = sc_final
            print(f"  NEW BEST: {best_score:.10f} [P4 {label}]")
        else:
            print(f"  {label}: {sc_final:.10f} (no improvement)")

    # === P5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    p5_end = deadline - 120.0
    n_candidates = [int(n_best * 1.2), int(n_best * 0.8), n_best + 4096, n_best - 4096]
    n_candidates = [n for n in n_candidates if 2048 <= n <= 40960 and n != n_best]
    seen_n = set()
    n_candidates_f = []
    for nc in n_candidates:
        if nc not in seen_n:
            seen_n.add(nc)
            n_candidates_f.append(nc)

    for n_try in n_candidates_f[:3]:
        remaining = p5_end - time.monotonic() - 10.0
        if remaining < 40:
            break
        a_res = _upsample(best_a, n_try)
        allocate = min(150.0, remaining - 10.0)
        a_opt, sc_opt = _adam_optimize(a_res, time.monotonic() + allocate,
                                        min_time=10.0, use_lbfgs=True, lbfgs_interval=300)
        if sc_opt < best_score and time.monotonic() < p5_end - 15:
            a_pol, sc_pol = _kkt_active_set(a_opt, p5_end - 10.0, top_k=5)
            if sc_pol < sc_opt:
                sc_opt = sc_pol
                a_opt = a_pol
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P5]")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no improvement)")

    # === P6: Final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    # Peak shave
    a_ps, sc_ps = _peak_shave_refined(best_a, min(deadline - 30.0, time.monotonic() + 60.0), n_trials=100)
    if sc_ps < best_score:
        best_a = a_ps.copy()
        best_score = sc_ps
        print(f"  NEW BEST: {best_score:.10f} [P6 peak_shave]")

    # Final KKT
    a_fk, sc_fk = _kkt_active_set(best_a, min(deadline - 5.0, time.monotonic() + 50.0), top_k=5)
    if sc_fk < best_score:
        best_a = a_fk.copy()
        best_score = sc_fk
        print(f"  NEW BEST: {best_score:.10f} [P6 final_KKT]")

    # Final short Adam
    if time.monotonic() < deadline - 3.0:
        a_fa, sc_fa = _adam_optimize(best_a, deadline - 2.0, min_time=2.0, use_lbfgs=False)
        if sc_fa < best_score:
            best_a = a_fa.copy()
            best_score = sc_fa
            print(f"  NEW BEST: {best_score:.10f} [P6 final_adam]")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) / 256.0
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
