def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    """Find optimal alpha for step a + alpha*d to minimize max of active peaks."""
    n = len(a)

    #           = c_k + alpha * sum_i (d_i*a_{k-i} + a_i*d_{k-i}) + alpha^2 * sum_i d_i*d_{k-i}

    alpha_lo = 0.0

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)

    if max_c < 1e-15:

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.0003:
        gap = max_c * 0.0003
    base_beta = max(gap, max_c * 0.001)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
         0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
         0.0000005, 0.0000002, 0.0000001]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 20))))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 30,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_exact(a_init, deadline, tol_start=0.003, tol_end=0.000005,
                      lr_start=0.02, top_k=8):
    """Active-set polish with optimal step size computation."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0
    max_stall = 200

    while time.monotonic() < deadline - 1.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        k_eff = min(top_k, 2 * n - 1)
        tol_val = tol * c_max
        active_mask = c >= (c_max - tol_val)
        n_active = int(active_mask.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[idx] = True
            n_active = 1
        if n_active > k_eff * 5:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active_mask] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.4
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.4

        if not improved:
            no_improve += 1
            if no_improve % 10 == 0:
                tol = max(tol_end, tol * 0.6)
                lr *= 0.5
            if no_improve >= max_stall:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.4)
                lr = lr_start * 0.3
                no_improve = 0
                stall += 1
                if stall > 15:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.97)
            lr = min(lr_start, lr * 1.02)

    return best_a, _score(best_a)


def _peak_shave_v2(a, deadline, n_trials=400):
    """Targeted peak shaving with optimal redistribution."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)

        if k_lo > k_hi:
            idx = int(np.argmax(a))
            a_try = a.copy()
            a_try[idx] *= 0.9
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            continue

        # Try multiple reduction fractions
        for frac in [0.03, 0.05, 0.08, 0.12, 0.15]:
            n_reduce = max(1, int(len(contrib) * frac))
            top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

            a_try = a.copy()
            reduction = 0.0
            for idx in top_idx:
                red = a_try[idx] * 0.15
                a_try[idx] -= red
                reduction += red

            if reduction < 1e-15:
                continue

            # Distribute to low-contributing positions
            low_mask = np.ones(n, dtype=bool)
            for idx in top_idx:
                low_mask[idx] = False
                j_val = k_star - idx
                if 0 <= j_val < n:
                    low_mask[j_val] = False

            if low_mask.sum() == 0:
                low_mask = np.ones(n, dtype=bool)
                for idx in top_idx:
                    low_mask[idx] = False
                if low_mask.sum() == 0:
                    break

            a_try[low_mask] += reduction / low_mask.sum()
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * (t - 0.5))
        return a / a.sum()
    elif kind == "asym_ramp2":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.3))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _quick_lbfgs(a_init, deadline, max_betas=8):
    """Short L-BFGS run for restarts."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    )[:max_betas]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + 5.0 > deadline:
            break
        remaining = deadline - 5.0 - time.monotonic()
        if remaining < 3.0:
            break
        scale = max(1.0, n / 8192.0)
        max_iter = min(3000, max(20, int(remaining / (0.003 * scale * 15))))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-13,
                         'maxcor': 60, 'maxls': 25}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a
