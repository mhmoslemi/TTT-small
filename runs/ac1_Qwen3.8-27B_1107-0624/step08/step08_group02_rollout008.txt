    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _soft_newton(a_init, deadline, min_time_remaining=60.0):
    """Smooth Newton method with adaptive soft-max weights."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Initial beta: scale relative to the spread of convolution values
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.5, c_max * 0.01)
    min_beta = c_max * 1e-12
    no_improve_count = 0

    while beta > min_beta and time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Compute soft-max weights
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        # Compute gradient (KKT residual)
        g = _hankel_matvec(a, w, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))

        if gn < 1e-14 * n:
            break

        # Solve Hessian system
        Hg = _cg_solve(w, n, g_t, max_iter=600)
        H1 = _cg_solve(w, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        c_max_current = c_max
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Sufficient decrease
            if c_max_try < c_max_current * (1.0 - 0.01 * alpha) - 1e-15:
                a = a_try
                c_max = c_max_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Try smaller fixed steps
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a + trial * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max_current - 1e-15:
                    a = a_try
                    c_max = float(c_try.max())
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if not step_done and no_improve_count > 8:
            break

        # Adaptive beta reduction
        c_new = _full_convolve(a, a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]

        if new_gap < 1e-7 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.001)

        if no_improve_count > 20:
            break

    return best_a


def _newton_polish(a_init, deadline, max_rounds=200):
    """Hard active-set Newton polish with multiple tolerance levels."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Try multiple active set tolerances, pick best
        best_dx = None
        best_alpha = 0
        best_c_max = c_max
        for tol_frac in [1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active < 1:
                active[int(np.argmax(c))] = True
                n_active = 1

            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / n_active

            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-13 * n:
                if best_dx is None:
                    best_dx = np.zeros(n)
                continue

            Hg = _cg_solve(W, n, g, max_iter=500)
            H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
            H1_norm = float(np.linalg.norm(H1))
            if H1_norm < 1e-20 or not np.isfinite(H1_norm):
                dx = -g_t / max(gn, 1e-30)
            else:
                lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
                dx = -Hg + lam * H1
                dx = dx - np.mean(dx)
                dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    dx = -g_t / max(gn, 1e-30)

            # Try this direction
            alpha = 1.0
            for ls in range(30):
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-16:
                    if c_max_try < best_c_max:
                        best_c_max = c_max_try
                        best_dx = dx
                        best_alpha = alpha
                    break
                alpha *= 0.5
                if alpha < 1e-14:
                    break

        if best_dx is None or best_alpha == 0:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        a = a + best_alpha * best_dx
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.0 + 0.01
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fejer_like":
        t = np.linspace(0, 1, n)
        a = (1.0 - np.abs(2.0 * t - 1.0)) ** 1.5 + 0.001
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _smooth_newton_from_z(z, beta_init, deadline, min_time_remaining=30.0):
    """Newton in log-space to handle non-negativity more naturally."""
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(beta_init, gap * 0.3, c_max * 0.005)
    min_beta = c_max * 1e-11

    best_a = a.copy()
    best_sc = _score(a)

    while beta > min_beta and time.monotonic() < deadline - min_time_remaining:
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Soft-max weights
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        # Gradient in z-space: grad_z = a * (grad_a - mean(grad_a))
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad_a = 4.0 * n * a * (h - W_val)
        grad_z = a * (grad_a - np.mean(grad_a))
        gnorm = np.linalg.norm(grad_z)

        if gnorm < 1e-14 * n:
            break

        # Newton step in a-space, then map back to z-space
        g = _hankel_matvec(a, w, n)
        g_t = g - np.mean(g)
        gn_a = float(np.linalg.norm(g_t))

        if gn_a < 1e-13 * n:
            # Use gradient step in z-space
            step = 0.1 / (gnorm + 1e-30)
            z = z - step * grad_z
            z = z - z.mean()
            continue

        Hg = _cg_solve(w, n, g_t, max_iter=500)
        H1 = _cg_solve(w, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx_a = -g_t / max(gn_a, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx_a = -Hg + lam * H1
            dx_a = dx_a - np.mean(dx_a)

        # Map to z-space: dz ≈ dx_a / a
        dz = dx_a / np.maximum(a, 1e-20)
        dz = dz - np.mean(dz)
        dz_norm = np.linalg.norm(dz)

        # Line search in z-space
        alpha = 1.0
        step_done = False
        for ls in range(40):
            z_try = z + alpha * dz
            z_try = np.clip(z_try, -50, 50)
            a_try = _softmax(z_try)
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.005 * alpha) - 1e-15:
                z = z_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient fallback in z-space
            for trial in [0.1, 0.05, 0.02, 0.01]:
                z_try = z - trial * grad_z / max(gnorm, 1e-30)
                z_try = np.clip(z_try, -50, 50)
                a_try = _softmax(z_try)
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    z = z_try
                    step_done = True
                    break

        a = _softmax(z)
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Adaptive beta
        c_check = _full_convolve(a, a)
        c_max_check = float(c_check.max())
        c_sorted_check = np.sort(c_check)[::-1]
        new_gap = c_sorted_check[0] - c_sorted_check[min(20, len(c_sorted_check) - 1)]

        if new_gap < 1e-8 * c_max_check:
            beta *= 0.15
        elif new_gap < 1e-5 * c_max_check:
            beta *= 0.3
        else:
            beta *= 0.5
        beta = min(beta, c_max_check * 0.0005)

    return best_a


def _multi_n_optimize(a_start, n_target, deadline, min_time_remaining=30.0):
    """Optimize at a specific n by resampling and running Newton."""
    n = n_target
    if len(a_start) != n:
        a_init = _resample(a_start, n)
    else:
        a_init = a_start.copy()

    # Quick score check
    sc = _score(a_init)

    # Soft Newton
    a_opt = _soft_newton(a_init, deadline, min_time_remaining)
    sc_opt = _score(a_opt)

    if sc_opt < sc:
        a_init = a_opt
        sc = sc_opt

    # Newton polish
    a_pol, sc_pol = _newton_polish(a_init, deadline, max_rounds=80)
    if sc_pol < sc:
        sc = sc_pol
        a_init = a_pol

    return a_init, sc


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization function. Minimizes 2*n*max(conv(a,a))/sum(a)^2."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick profile comparison at parent length
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["tapered", "super_tapered", "fejer_like", "asym_taper", "cos", "exp_decay"]
    best_init = parent.copy()
    best_init_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 800:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_init_sc:
            best_init_sc = sc_p
            best_init = a_p
            print(f"  Better profile: '{prof_kind}' score={sc_p:.10f}")

    # Also try perturbed parent
    for sigma in [0.03, 0.1]:
        if time.monotonic() > deadline - 700:
            break
        a_pert = _perturb(parent, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_init_sc:
            best_init_sc = sc_pert
            best_init = a_pert

    # Phase 2: Soft Newton from best initialization
    print(f"\n[Phase 2] Soft Newton main (t={time.monotonic() - t_start:.1f}s)")
    a_soft = _soft_newton(best_init, min(deadline - 400, time.monotonic() + 600), min_time_remaining=300)
    update_best(a_soft, _score(a_soft), "soft_newton_main")

    # Phase 3: Soft Newton in z-space (different parameterization)
    print(f"\n[Phase 3] Z-space Newton (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        z = np.log(np.maximum(best_seq if best_seq is not None else best_init, 1e-30))
        z = z - z.mean()
        c_tmp = _full_convolve(best_seq if best_seq is not None else best_init,
                               best_seq if best_seq is not None else best_init)
        beta_z = float(c_tmp.max()) * 0.01
        a_z = _smooth_newton_from_z(z, beta_z, min(deadline - 250, time.monotonic() + 300),
                                     min_time_remaining=200)
        update_best(a_z, _score(a_z), "z_newton")

    # Phase 4: Hard active-set Newton polish
    print(f"\n[Phase 4] Newton polish 1 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_np1, sc_np1 = _newton_polish(best_seq, min(deadline - 150, time.monotonic() + 200),
                                        max_rounds=200)
        update_best(a_np1, sc_np1, "newton_polish_1")

    # Phase 5: More polish with fresh soft Newton from polished result
    print(f"\n[Phase 5] Cycles (t={time.monotonic() - t_start:.1f}s)")
    remaining_for_cycles = deadline - 100 - time.monotonic()
    n_cycles = max(1, int(remaining_for_cycles // 120))
    for cyc in range(min(n_cycles, 5)):
        if time.monotonic() > deadline - 60:
            break
        # Soft Newton short burst
        if time.monotonic() < deadline - 80:
            a_c = _soft_newton(best_seq, min(deadline - 60, time.monotonic() + 80),
                                min_time_remaining=50)
            update_best(a_c, _score(a_c), f"cycle_{cyc}_soft")
        # Newton polish
        if best_seq is not None and time.monotonic() < deadline - 30:
            a_np, sc_np = _newton_polish(best_seq, min(deadline - 20, time.monotonic() + 40),
                                          max_rounds=50)
            update_best(a_np, sc_np, f"cycle_{cyc}_newton")

    # Phase 6: Try different n values
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if n_try == n_parent:
            continue
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        # Quick optimize
        a_opt = _soft_newton(a_r, min(deadline - 30, time.monotonic() + 60), min_time_remaining=25)
        update_best(a_opt, _score(a_opt), f"n_{n_try}_soft")
        if best_seq is not None and time.monotonic() < deadline - 15:
            a_np, sc_np = _newton_polish(best_seq if len(best_seq) == n_try else a_opt,
                                          deadline - 5, max_rounds=30)
            update_best(a_np, sc_np, f"n_{n_try}_polish")

    # Phase 7: Final ultra-fine polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 1, max_rounds=300)
        update_best(a_fin, sc_fin, "final_polish")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
