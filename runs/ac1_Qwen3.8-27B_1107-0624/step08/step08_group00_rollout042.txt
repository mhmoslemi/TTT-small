def _next_pow2(n):
    k = 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break

        if not improved:
            lr *= 0.95
            v *= 0.5

        # Adapt learning rate based on progress
        if improved and stall == 0:
            lr = min(0.02, lr * 1.01)

        # Periodic spectral perturbation to escape local minima
        if stall > stall_limit:
            # Add random frequency perturbation
            noise_amp = lr * 0.5 * (stall / stall_limit)
            z = np.fft.rfft(a)
            mask = np.ones(len(z), dtype=bool)
            mask[0] = False  # preserve DC component
            z[mask] += noise_amp * a.max() * np.random.default_rng(iter_count).standard_normal(len(z) - 1)
            a_pert = np.fft.irfft(z, 2 * n - 1)[:n]
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                sc_pert = _score(a_pert)
                if sc_pert < best_sc:
                    a = a_pert
                    best_a = a.copy()
                    best_sc = sc_pert
            v = np.zeros(n)
            mu = 0.8
            lr = 0.005
            stall = 0
            stall_limit = int(stall_limit * 1.3)
            iter_count += 1000

        iter_count += 1
        if iter_count % 2000 == 0:
            print(f"    MD iter {iter_count}: best={best_sc:.10f}, beta={beta:.1f}, lr={lr:.5f}, stall={stall}")

    return best_a, best_sc

def _active_set_polish(a_init, deadline, min_remaining=30.0, top_k_frac=0.01):
    """Directly equalize the top peaks by adjusting contributing elements."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.02
    top_k = max(3, int(n * top_k_frac))
    stall = 0
    stall_limit = 300

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]

        # Compute combined gradient from top peaks
        equal_weights = np.zeros_like(c)
        equal_weights[top_idx] = 1.0 / top_k
        grad = _gradient_weighted(a, equal_weights)
        grad = grad - np.mean(grad)

        # Take step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = lr * grad / gnorm

        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = a - alpha * step
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break

        if not improved:
            lr *= 0.9
            top_k = max(2, int(top_k * 0.8))
            if stall > stall_limit:
                a = best_a.copy()
                lr = 0.01
                top_k = max(3, int(n * top_k_frac * 0.5))
                stall = 0
                stall_limit = int(stall_limit * 1.5)
                if stall_limit > 2000:
                    break
        else:
            # Gradually increase top_k to smooth
            top_k = min(int(n * 0.2), top_k + max(1, int(n * 0.001)))

    return best_a, best_sc

def _lbfgs_smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lbfgs_smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of smoothed objective w.r.t. a
    grad_a = _gradient_weighted(a, w)
    # Chain rule: d/dz = a * (I - outer(a,a)) * d/da
    # softmax gradient: d a_i / d z_j = a_i*(delta_ij - a_j)
    grad_z = a * (grad_a - np.sum(a * grad_a))
    return 4.0 * n * grad_z  # factor of 4 from the 2n * d(smoothed_max)/da * 2 (chain)

def _run_lbfgs(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Try multiple beta values
    betas = [max_c * f for f in [0.05, 0.02, 0.01, 0.004, 0.001, 0.0004, 0.0001]]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 15))))

        def obj(z_in):
            return _lbfgs_smooth_obj(z_in, beta, n)

        def grad(z_in):
            return _lbfgs_smooth_grad(z_in, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 50, 'maxls': 30}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc

def _spectral_perturb(a, strength, rng):
    """Add perturbation in frequency domain to escape local minima."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    # Add random phase perturbation to non-DC components
    perm = rng.permutation(len(A))
    n_perturb = max(1, len(A) // 10)
    freqs = perm[1:1 + n_perturb]  # skip DC
    A[freqs] *= (1.0 + strength * rng.standard_normal(n_perturb))
    A = A.real  # keep real for real output (approximate)
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def propose_candidate(seed=42, budget_s=1800):
    """Main search function combining multiple optimization strategies."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 0: Parametric initialization exploration ===
    print(f"\n[P0] Parametric init (t={time.monotonic()-t_start:.1f}s)")
    p0_deadline = min(deadline - 1500, time.monotonic() + 40)
    kinds = ["sin1", "sin2", "sin3", "tapered", "gauss_center", "multi_sin"]
    for kind in kinds:
        if time.monotonic() > p0_deadline:
            break
        a_try = _parametric_init(n_main, kind, rng)
        sc_try = _score(a_try)
        if sc_try < best_score:
            best_seq = a_try.copy()
            best_score = sc_try
            print(f"  {kind}: score={sc_try:.10f} (NEW BEST)")

    # === Phase 1: Momentum gradient descent ===
    print(f"\n[P1] Momentum GD (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 900, time.monotonic() + 600)
    a_p1 = _momentum_gd(best_seq, p1_deadline, min_remaining=30.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [momentum_gd] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set polish ===
    print(f"\n[P2] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 600, time.monotonic() + 400)
    a_p2 = _active_set_polish(best_seq, p2_deadline, min_remaining=30.0, top_k_frac=0.005)
    sc_p2 = _score(a_p2)
    if sc_p2 < best_score:
        best_seq = a_p2.copy()
        best_score = sc_p2
        print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS polish ===
    print(f"\n[P3] L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, time.monotonic() + 300)
        a_p3 = _run_lbfgs(best_seq, p3_deadline, min_remaining=30.0)
        sc_p3 = _score(a_p3)
        if sc_p3 < best_score:
            best_seq = a_p3.copy()
            best_score = sc_p3
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second momentum GD round with different params ===
    print(f"\n[P4] Momentum GD round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p4_deadline = min(deadline - 250, time.monotonic() + 150)
        a_p4 = _momentum_gd(best_seq, p4_deadline, min_remaining=20.0)
        sc_p4 = _score(a_p4)
        if sc_p4 < best_score:
            best_seq = a_p4.copy()
            best_score = sc_p4
            print(f"  NEW BEST: {best_score:.10f} [md2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 32768, 8192]
    if n_main not in n_candidates:
        n_candidates.insert(0, n_main)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # Quick momentum GD
        if time.monotonic() < deadline - 80:
            a_opt = _momentum_gd(a_init, min(deadline - 80, time.monotonic() + 80), min_remaining=20.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

        # Quick active-set
        if time.monotonic() < deadline - 40:
            a_opt2 = _active_set_polish(a_opt if sc_opt < best_score else a_init,
                                         min(deadline - 40, time.monotonic() + 30),
                                         min_remaining=10.0, top_k_frac=0.003)
            sc_opt2 = _score(a_opt2)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, min(deadline - 80, time.monotonic() + 30))
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 40:
                a_shr2 = _active_set_polish(best_seq, min(deadline - 40, time.monotonic() + 25),
                                             min_remaining=10.0, top_k_frac=0.002)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Spectral perturbation + re-optimize ===
    print(f"\n[P7] Spectral exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        for _ in range(3):
            if time.monotonic() > deadline - 50:
                break
            a_pert = _spectral_perturb(best_seq, 0.05, rng)
            sc_pert = _score(a_pert)
            if sc_pert < best_score:
                # Re-optimize from perturbed point
                a_opt = _momentum_gd(a_pert, min(deadline - 30, time.monotonic() + 40), min_remaining=15.0)
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final L-BFGS + polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin = _run_lbfgs(best_seq, deadline - 8, min_remaining=5.0)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < deadline - 5:
            a_fin2 = _active_set_polish(best_seq, deadline - 3, min_remaining=2.0, top_k_frac=0.002)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
