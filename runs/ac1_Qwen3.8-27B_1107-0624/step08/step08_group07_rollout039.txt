def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

Optimized step function search using multi-phase convex/nonsmooth optimization:
1. FFT-based log-sum-exp smooth optimization (Adam in log-space)
2. Active-set LP direction finding via cutting planes
3. Subgradient polish with active-set equalization
4. Multiple restarts for escaping local minima
5. Final fine-tuning
Target: minimize 2n * max_k (p*p)[k] / (sum p)^2 for non-negative p
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _convolution_matrix_row(f, k, n):
    """Get the k-th row of the convolution matrix A[k, j] = f[k-j]."""
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        row[lo:hi+1] = f[k-hi:k-lo+1][::-1]
    return row

                          max_iter, eval_interval=50, restart_threshold=200):

    Optimize using log-sum-exp smooth approximation with Adam in log-space.
    a_init: initial probability vector (sums to 1)
    Returns: (best_a, best_score_norm)
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    last_improve = 0
    lr_mult = 1.0
    
    # Precompute p_rev FFT for gradient computation
    p = a_init.copy()
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        
        # Temperature: exponential cooling
        tau = tau_start * (tau_end / tau_start) ** frac
        # Learning rate: polynomial decay
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.7) * lr_mult
        
        p = _softmax_stable(z)
        c = _fft_convolve(p, p)
        
        # Compute smoothed gradient using FFT
        c_shifted = (c - c.max()) / tau
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / s

            grad = np.zeros(n)

        W = float(np.dot(p, grad))
        z_grad = p * (grad - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)

        z = z - z.mean()
        
        # Evaluate periodically

            p_cur = _softmax_stable(z)
            sc = _score_norm(p_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = p_cur.copy()
                patience = 0
                last_improve = t
            else:
                patience += 1
            
            if patience > restart_threshold:
                lr_mult *= 0.6
                patience = 0
                if lr_mult < 0.05:
                    # Restart from best
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.2
    
    p_final = _softmax_stable(z)
    sc_final = _score_norm(p_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = p_final.copy()
    
    return best_a, best_score


def _active_set_lp_direction(f, M, max_active=30, time_limit=5.0):
    """
    Find direction b >= 0 maximizing sum(b) subject to (f*b)[k] <= M for active k.
    Uses cutting-plane to build active set.
    Returns: b (normalized to sum=1) or None if infeasible.
    """
    n = len(f)
    t_start = time.monotonic()
    
    # Start with the top peak
    # The constraint (f*b)[k] = sum_j f[k-j] b[j] <= M
    # Initial active set: just k* = argmax of "potential violation"
    # Actually, let's start with k* where f has its peak contribution
    
    # Heuristic: start with k* = n-1 (the "center" of the convolution)
    active = [n - 1]
    
    for iteration in range(max_active):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Build constraint matrix for active set
        A_rows = []
        for k in active:
            row = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                row[lo:hi+1] = f[k-hi:k-lo+1][::-1]
            A_rows.append(row)
        
        A = np.array(A_rows)  # (m, n)
        b_rhs = np.full(len(active), M)
        
        # LP: maximize 1^T b subject to A @ b <= b_rhs, b >= 0
        # In standard form for linprog: minimize -1^T b
        c_obj = -np.ones(n)
        
        # Add bound on sum to prevent unboundedness
        # Actually, the constraints should bound it. Let's add a safety:
        # sum(b) <= n*M / min(f) + 1 (rough bound)
        # Or just let linprog handle it (it will report unbounded if so)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_rhs, 
                         bounds=[(0, None)] * n, 
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
            
            if not res.success:
                if res.status == 2:  # unbounded
                    # Add all constraints
                    break
                if res.status == 3:  # infeasible
                    return None
                if res.status == 4:  # time limit
                    break
                return None
            
            b = res.x
            
            # Check if all constraints are satisfied
            # (f*b)[k] for all k
            fb = _fft_convolve(f, b)
            violations = fb - M
            if violations.max() <= 1e-8:
                # Feasible! Return b
                s = b.sum()
                if s < 1e-15:
                    return None
                return b / s
            
            # Add the most violated constraint
            k_viol = int(np.argmax(violations))
            if k_viol not in active:
                active.append(k_viol)
            else:
                # Already in active set but violated? Shouldn't happen.
                break
                
        except Exception:
            break
    
    # If we couldn't find a feasible direction with the active set,
    # try with top few peaks
    c_f = _fft_convolve(f, f)
    top_k = np.argsort(c_f)[::-1][:10]
    active = list(top_k)
    
    A_rows = []
    for k in active:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            row[lo:hi+1] = f[k-hi:k-lo+1][::-1]
        A_rows.append(row)
    
    A = np.array(A_rows)
    b_rhs = np.full(len(active), M)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_rhs,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'time_limit': time_limit * 2})
        if res.success:
            b = res.x
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    
    return None


def _lp_direction_optimize(a_init, deadline, max_rounds=200, lp_time=3.0):
    """
    Iteratively find LP directions and update.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        
        b_dir = _active_set_lp_direction(a, M, max_active=20, time_limit=lp_time)
        if b_dir is None:
            break
        
        # Try different mixing parameters
        improved = False
        for t_mix in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_cand = (1 - t_mix) * a + t_mix * b_dir
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for t_mix in [0.01, 0.005, 0.002]:
                a_cand = (1 - t_mix) * a + t_mix * b_dir
                a_cand = np.maximum(0, a_cand)
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                sc = _score_norm(a_cand)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_score


def _subgradient_optimize(a_init, deadline, max_iter=50000, top_k=5, 
                           lr_start=0.01, lr_end=0.0001):
    """
    Subgradient method: at each step, compute the subgradient of max(c_k)
    as the average of gradients of the top-k peaks, and take a projected step.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    lr = lr_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Get top-k peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]
        
        # Only include peaks close to max
        active = [k for k in active if c[k] > c_max * 0.99]
        if len(active) == 0:
            active = [int(sorted_idx[0])]
        
        # Compute subgradient (average of active peak gradients)
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        grad /= len(active)
        
        # Project to tangent space of simplex (remove component along 1)
        # The projection of g onto {x: sum(x) = 0} is g - mean(g)
        # But we also need to respect p >= 0, so use a simpler approach:
        # Subtract the weighted mean to preserve the sum
        g_mean = float(np.dot(a, grad))
        grad_tangent = grad - g_mean
        
        g_norm = np.linalg.norm(grad_tangent)
        if g_norm < 1e-15:
            break
        
        # Step
        a_new = a - lr * grad_tangent / g_norm
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr = max(lr * 0.999, lr_end)
        else:
            lr = max(lr * 0.95, lr_end)
            if lr <= lr_end * 1.01:
                break
    
    return best_a, best_score


def _coordinate_polish(a_init, deadline, max_rounds=100):
    """
    Coordinate descent: try scaling individual elements to reduce the max peak.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        improved = False
        # Try modifying elements that contribute to the peak
        indices = list(range(lo, hi + 1))
        np.random.shuffle(indices)
        
        for j in indices:
            if best_a[j] < 1e-12:
                continue
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a_cand = best_a.copy()
                a_cand[j] *= scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(2.0 * n * c_cand.max())
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try a broader search: random coordinates
            for _ in range(50):
                j = np.random.randint(n)
                if best_a[j] < 1e-12:
                    continue
                for scale in [0.3, 0.6, 1.4, 2.0]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(2.0 * n * c_cand.max())
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return best_a, best_score


def _multi_tau_adam(a_init, deadline, n_phases=4):
    """
    Run Adam optimization at multiple temperature levels.
    """
    n = len(a_init)
    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    # Temperature schedule: start high, end low
    tau_values = [max_c * f for f in [0.02, 0.01, 0.003, 0.001, 0.0003, 0.0001]]
    lr_values = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
    
    n_actual = min(n_phases, len(tau_values))
    time_per = max(20.0, (deadline - time.monotonic()) / n_actual)
    
    for i in range(n_actual):
        if time.monotonic() > deadline:
            break
        sub_dl = min(deadline, time.monotonic() + time_per)
        
        a_opt, sc = _lse_smooth_optimize(
            a_init=best_a,
            deadline=sub_dl,
            tau_start=tau_values[i],
            tau_end=tau_values[i] * 0.1,
            lr_start=lr_values[i],
            lr_end=lr_values[i] * 0.1,
            max_iter=200000,
            eval_interval=30
        )
        
        full_sc = _score_norm(a_opt)
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()
        print(f"  Adam T={tau_values[i]:.6f} lr={lr_values[i]:.4f}: "
              f"score={full_sc:.10f} t={time.monotonic()-t_start_global:.0f}s")
    
    return best_a, best_score
