    """
    Peeling method for active set identification.
    Start with a small active set, expand it until KKT conditions are satisfied.
    """
    a = a_init.copy() / a_init.sum()
    A = [int(np.argmax(_full_convolve(a, a)))]  # Start with the argmax lag
    
    for round in range(max_rounds):
        # Solve KKT for current active set
        a_new = solve_kkt_for_active_set(a, A, n)
        if a_new is None:
            break
        
        # Check if solution is valid
        c = _full_convolve(a_new, a_new)
        t = max(c[k] for k in A)
        max_other = max(c[k] for k in range(len(c)) if k not in A)

        cmax_best = c_max
        for ls_i in range(25):
            alpha = (1.0 if ls_i == 0 else alpha_best * 0.6)
            if ls_i > 0 and alpha_best == 0.0:
                alpha = 0.5 / (ls_i + 1)
            a_try = np.maximum(a + alpha * dx, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            cm_try = float(c_try.max())
            if cm_try < c_max - 1e-16:
                a = a_try
                c_max = cm_try
                alpha_best = alpha
                cmax_best = cm_try
                break
            elif alpha_best == 0.0:
                alpha_best = alpha
                cmax_best = cm_try
            if alpha < 1e-15:
                break

        # If no improvement, try subgradient step
        if cmax_best >= c_max - 1e-16:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = np.maximum(a - trial * g_t / (gn + 1e-30), 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                cm_try = float(c_try.max())
                # Only the middle part matters (the HANKEL product captures lags n-1 to 2n-2)
                # But we need full check
                if cm_try < c_max - 1e-16:
                    a = a_try
                    c_max = cm_try
                    cmax_best = cm_try
                    break

        # Update best
        sc = 2.0 * n * c_max  # (since sum(a) = 1, score = 2n * cmax)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            best_cmax = c_max
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            break

    # Final verification
    final_sc = _score(best_a)
    return best_a, final_sc


def _run_adaptive_lbfgs_v2(a_init, deadline, min_time_remaining=30.0):
    """Enhanced adaptive L-BFGS with more aggressive beta scheduling."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    stagnation_count = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        iter_time_est = 0.004 * scale
        max_iter = min(12000, max(50, int(remaining / (iter_time_est * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnation_count = 0
            else:
                stagnation_count += 1
                if stagnation_count > 3:
                    beta *= 0.1
                    stagnation_count = 0
                    continue
        except Exception:
            beta *= 0.3
            continue

        # Adaptive beta reduction based on convergence
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        rel_gap = new_gap / max(c_max_new, 1e-30)

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if rel_gap < 1e-9 and sc_improved > 1e-14:
            beta *= 0.1
        elif rel_gap < 1e-7:
            beta *= 0.15
        elif rel_gap < 1e-5:
            beta *= 0.25
        elif rel_gap < 1e-3:
            beta *= 0.4
        else:
            beta *= 0.5

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.4) ** 0.6
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 3.0 + 0.005
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 3.0 + 0.005
        return a / a.sum()
    elif kind == "step_like":
        a = np.ones(n)
        third = n // 3
        a[:third] = 1.0
        a[third:2*third] = 0.5
        a[2*third:] = 0.2
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(-3, 3, n)
        a = np.exp(-t**2 / 4.0)
        return a / a.sum()
    elif kind == "two_step":
        a = np.ones(n)
        half = n // 2
        a[:half] = 2.0
        a[half:] = 0.4
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()


def _grueneisen_iteration(a_init, deadline, max_iter=500):
    """
    Grunbaum-type iterative method:
    minimize max_k c[k] s.t. sum(a)=1, a>=0
    using subgradient with Polyak step size.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_cmax = float(_full_convolve(a, a).max())
    # Lower bound: average of c over all lags = (sum a)^2 / (2n-1) = 1/(2n-1)
    lb = 1.0 / (2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        if c_max < best_cmax - 1e-15:
            best_cmax = c_max
            best_a = a.copy()

        # Polyak step size
        gap = c_max - lb
        if gap < 1e-18:
            break
        step = gap / (2.0 * n * c_max + 1e-30)  # conservative

        # Subgradient: weighted average of active gradients
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W = np.zeros(len(c))
        W[active] = 1.0 / active.sum()

        g = _hankel_matvec(a, W, n) / 2.0  # gradient of (1/2)*sum W_k c[k]
        g = g - np.mean(g)  # project onto tangent space

        a_try = a - step * g
        a_try = _project_simplex(a_try)

        # Check improvement
        c_try = _full_convolve(a_try, a_try)
        cm_try = float(c_try.max())
        if cm_try < c_max:
            a = a_try
        else:
            # Try smaller step
            step *= 0.5
            a_try = _project_simplex(a - step * g)
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max:
                a = a_try
            else:
                break

    return best_a, 2.0 * n * best_cmax


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization function."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            improvement = prev - best_score if np.isfinite(prev) else 0
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"delta={improvement:.2e} t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Profile exploration (120s) ===
    print(f"\n[Phase 1] Profile exploration")
    profiles = ["tapered", "asym_taper", "left_heavy", "right_heavy", "exp_decay",
                "bimodal", "step_like", "two_step", "gaussian_center", "cos"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 1500:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': {sc_p:.8f}")

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 1400:
        a_quick = _run_adaptive_lbfgs_v2(best_profile_a, min(deadline - 1400, time.monotonic() + 120),
                                          min_time_remaining=1300.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_lbfgs")
        if sc_quick < best_profile_sc:
            best_profile_a = a_quick
            best_profile_sc = sc_quick

    # === Phase 2: Main L-BFGS (600s) ===
    print(f"\n[Phase 2] Main L-BFGS")
    start_a = best_seq if best_seq is not None else best_profile_a
    a_main = _run_adaptive_lbfgs_v2(start_a, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # === Phase 3: Newton polish (300s) ===
    print(f"\n[Phase 3] Newton polish")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_np, sc_np = _newton_polish_v2(best_seq, deadline - 300, max_rounds=150)
        update_best(a_np, sc_np, "newton_1")

    # === Phase 4: Grunbaum iteration (200s) ===
    print(f"\n[Phase 4] Grunbaum iteration")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_gr, sc_gr = _grueneisen_iteration(best_seq, deadline - 200, max_iter=1000)
        update_best(a_gr, sc_gr, "grunbaum")
        if sc_gr < best_score + 0.002:
            a_gr_np, sc_gr_np = _newton_polish_v2(a_gr, min(deadline - 150, time.monotonic() + 60), max_rounds=50)
            update_best(a_gr_np, sc_gr_np, "grunbaum_np")

    # === Phase 5: Multi-n exploration (300s) ===
    print(f"\n[Phase 5] Multi-n exploration")
    n_candidates = [8192, 32768, 4096]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        print(f"  Trying n={n_try}...")
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _run_adaptive_lbfgs_v2(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_np2, sc_np2 = _newton_polish_v2(a_opt, min(deadline - 30, time.monotonic() + 40), max_rounds=50)
            update_best(a_np2, sc_np2, f"n_{n_try}_np")

    # === Phase 6: Diverse restarts (150s) ===
    print(f"\n[Phase 6] Diverse restarts")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        restart_configs = [
            ("perturb_0.03", _perturb(best_seq, 0.03, rng)),
            ("perturb_0.1", _perturb(best_seq, 0.1, rng)),
            ("asym_restart", _make_profile(n_best, "asym_taper")),
            ("left_restart", _make_profile(n_best, "left_heavy")),
        ]
        for label, a_r_start in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            # Quick L-BFGS
            a_r = _run_adaptive_lbfgs_v2(a_r_start, min(deadline - 20, time.monotonic() + 40),
                                          min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, label)

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _newton_polish_v2(a_r, deadline - 5, max_rounds=40)
                update_best(a_rp, sc_rp, label + "_np")

    # === Phase 7: Final ultra-fine polish (remaining time) ===
    print(f"\n[Phase 7] Final polish")
    if best_seq is not None and time.monotonic() < deadline - 2:
        # Multiple rounds of Newton with decreasing tolerance
        a_cur = best_seq.copy()
        for polish_round in range(4):
            if time.monotonic() > deadline - 1:
                break
            a_fin, sc_fin = _newton_polish_v2(a_cur, deadline - 1, max_rounds=80)
            update_best(a_fin, sc_fin, f"polish_{polish_round}")
            a_cur = a_fin

        # Final L-BFGS at tiny beta
        if time.monotonic() < deadline - 1.5 and best_seq is not None:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-11
            remaining = deadline - 1.5 - time.monotonic()
            if remaining > 2:
                max_iter = min(3000, max(50, int(remaining / 0.005)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-17,
                                 'maxcor': 100, 'maxls': 500}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_vf, sc_vf = _newton_polish_v2(best_seq, deadline - 0.1, max_rounds=30)
            update_best(a_vf, sc_vf, "very_final")

    # Final validation and output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = float(final.sum())
        # Scale to be in valid range
        if s > 1000:
            final = final * (1000.0 / s)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, sum={s:.4f}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
