- Gradient: `grad[m] = 4 * conv(c, a_rev)[n-1-m]` (computable in O(n log n) via FFT)

This should be quite efficient!

But wait, the variance objective might not be as directly related to the "max" objective as I'd like. Minimizing variance reduces the "spread" of the convolution, but the max might still be high if there's one very high peak. In practice, for this problem, minimizing variance should correlate well with minimizing the max.

Let me code up the full solution now. I'll combine:
1. FFT equalization (new)
2. Variance minimization with L-BFGS (new)
3. Log-sum-exp L-BFGS (from previous, kept)
4. Active-set Newton (from previous, kept)
5. Multi-scale approach (new)
6. Diverse restarts (from previous, kept)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous:
1. FFT-based convolution equalization (alternating linearization)
2. Variance minimization objective (smooth, O(n log n) gradient via FFT)
3. Multi-scale optimization (small n → large n)
4. Improved blending of strategies
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# FFT-based equalization
def _fft_equalize(a, n):
    """Solve conv(a, x) ≈ target * 1 in least-squares sense using FFT."""
    L = _next_pow2(2 * n - 1)
    c = _full_convolve(a, a)
    target = c.sum() / (2 * n - 1)
    
    A = np.fft.rfft(a, L)
    # FFT of target * ones(2n-1) padded to L
    rhs = np.zeros(L // 2 + 1, dtype=np.complex128)
    # target * ones(2n-1): its DFT
    # Actually, let me just compute it directly
    rhs_signal = np.zeros(L)
    rhs_signal[:2*n-1] = target
    RHS = np.fft.rfft(rhs_signal)
    
    # Regularization
    eps = 1e-10 * (np.abs(A).max() + 1e-30)
    X = RHS / (A + eps)
    
    x_full = np.fft.irfft(X, L)
    x = x_full[:n]
    return x

def _fft_equalize_iter(a_init, max_iters=100, t=0.5, deadline=None):
    """Iterate the FFT equalization with damping."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        x = _fft_equalize(a, n)
        x = np.maximum(x, 0)
        s_x = x.sum()
        if s_x < 1e-15:
            break
        x = x / s_x
        
        a_new = (1 - t) * a + t * x
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        if abs(sc_new - best_sc) < 1e-12:
            break
        a = a_new
    
    return best_a

# Variance objective
def _variance_obj(z, n):
    """Minimize sum_k conv(a,a)[k]^2 where a = softmax(z)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.dot(c, c))

def _variance_grad(z, n):
    """Gradient of sum_k c[k]^2 w.r.t. z, where a = softmax(z), c = conv(a,a)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    # grad_a[m] = 4 * sum_k c[k] * a[k-m] (with boundary handling)
    # = 4 * conv(c, a_rev)[n-1-m]
    corr = _full_convolve(c, a_rev)
    grad_a = 4.0 * corr[n-1:2*n-1][::-1]
    # Convert to grad_z: grad_z = a * (grad_a - mean(a * grad_a))
    a_grad = a * grad_a
    grad_z = a * (grad_a - np.mean(a_grad))
    return grad_z

# Log-sum-exp objective (from previous)
def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

# Active set Newton (simplified from previous)
def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            top_idx = np.argsort(c)[::-1][:2*n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 2 * n

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient: (2/m) * conv(a, A_vec)[n-1:2n-1] ... need to be careful
        # grad[m] = (2/m) * sum_{k in active} a[k-m] (with boundary)
        # = (2/m) * conv(a_reversed, A_vec)[??]
        flip_a = a[::-1]
        out_len = n + len(A_vec) - 1
        N = _next_pow2(out_len)
        fa = np.fft.rfft(flip_a, N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fa * fA, N)[:out_len]
        g = (2.0 / m) * result[n - 1:2 * n - 1]
        
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        # Hessian-vector product: (2/m) * conv(v, A_vec) [appropriate slice] + mu * v
        # The Hessian of the active constraint is (2/m) * T where T is the "correlation" matrix
        # T[v][i] = (2/m) * sum_{k in active} a[k-i]... 
        # Actually, the Hessian of (1/m) * sum_{k in active} c[k] w.r.t. a is:
        # H[i,j] = (2/m) * (delta condition)
        # This is complicated. Let me use the same approach as before.
        
        def hess_matvec(v):
            flip_v = v[::-1]
            out_len2 = n + len(A_vec) - 1
            N2 = _next_pow2(out_len2)
            fv = np.fft.rfft(flip_v, N2)
            fA2 = np.fft.rfft(A_vec, N2)
            res2 = np.fft.irfft(fv * fA2, N2)[:out_len2]
            return (2.0 / m) * res2[n - 1:2 * n - 1] + mu * v

        # CG solve
        def cg_solve(rhs, max_cg=100):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u - np.mean(-u)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(30):
            a_try = np.maximum(a + t * dx, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-10:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-12:
            break

    return best_a, best_sc

# L-BFGS runner
def _run_lbfgs_ls(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001, 0.00005, 0.00002,
                                      0.00001, 0.000005])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.006
        elif n <= 16384:
            est_time_per_iter = 0.015
        else:
            est_time_per_iter = 0.03 * (n / 32768.0)

        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _lsobj,
                z,
                jac=_lsgrad,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _run_lbfgs_var(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on variance objective."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 5:
        return current_a

    if n <= 4096:
        est_time_per_iter = 0.004
    elif n <= 8192:
        est_time_per_iter = 0.008
    elif n <= 16384:
        est_time_per_iter = 0.02
    else:
        est_time_per_iter = 0.04 * (n / 32768.0)

    max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 10))))

    try:
        result = minimize(
            _variance_obj,
            z,
            jac=_variance_grad,
            args=(n,),
            method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-15,
                'gtol': 1e-11,
                'maxcor': 60,
                'maxls': 100,
            }
        )
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-15:
            current_a = a_new
    except Exception:
        pass

    return current_a

# Triangle solve (for smaller n)
def _triangle_solve(a, max_iter=50, t=0.3):
    """Use the triangular structure of conv(a,x) for x < n to solve step by step."""
    n = len(a)
    if a[0] < 1e-15:
        return a.copy()
    
    c = _full_convolve(a, a)
    target = c.sum() / (2 * n - 1)
    
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_sc = _score(a_curr)
    
    for it in range(max_iter):
        # Forward substitution: x[0] = target/a[0], x[k] = (target - sum_{i=1}^k a[i]*x[k-i]) / a[0]
        x = np.zeros(n)
        x[0] = target / a_curr[0]
        for k in range(1, n):
            s = 0.0
            for i in range(1, k+1):
                s += a_curr[i] * x[k-i]
            x[k] = (target - s) / a_curr[0]
        
        x = np.maximum(x, 0)
        s_x = x.sum()
        if s_x < 1e-15:
            break
        x = x / s_x
        
        a_new = (1 - t) * a_curr + t * x
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a_curr = a_new
    
    return best_a

def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
