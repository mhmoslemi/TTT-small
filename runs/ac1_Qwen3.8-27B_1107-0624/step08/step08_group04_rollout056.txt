"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) L-BFGS with log-sum-exp beta-continuation, (2) active-set QP with
adaptive smooth weights and preconditioned CG, (3) coordinate descent polish,
(4) multi-n exploration, (5) diverse restarts with structure-aware perturbations.
Key improvements over prior: smooth weight scheduling (c/cmax)^gamma, diagonal
preconditioning for CG, line-search safeguarding, and targeted coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_precon(W, n, rhs, max_iter=800, tol=1e-13):
    x = np.zeros(n)
    r = rhs.copy()
    z = r.copy()
    # Simple diagonal preconditioner: use row sum of H approximated by W[0]*2 + W[1]
    # Actually for Hankel H_{j,l} = W[j+l], the "diagonal" is W[j+j] which varies.
    # Use a constant preconditioner: sum of W
    prescale = max(float(np.sum(W)) / n, 1e-20)
    p = (r / prescale).copy()
    rz = float(r @ p)
    if rz < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r_new = r - alpha * Hp
        rz_new = float(r @ r_new)
        if not np.isfinite(rz_new) or rz_new > float(r @ r) * 1.5:
            break
        r = r_new
        p = r / prescale + (rz_new / rz) * p
        rz = rz_new
        if rz < tol * tol * float(rhs @ rhs):
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_dot_c = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_dot_c)


def _lbfgs_run(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.4, max_c * 0.005)

    if aggressive:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.00375, 0.0019, 0.00095, 0.0005, 0.00025, 0.00012,
                  0.00006, 0.00003, 0.000015, 0.000007, 0.0000035, 0.000002]
    else:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.004, 0.002, 0.001, 0.0005, 0.00025, 0.00012, 0.00006,
                  0.00003, 0.000015, 0.000007, 0.000003]

    betas = [base_beta * r for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(100, int(remaining / (est_time * 5))))

        try:
            obj_func = lambda z_local: _smoobj(z_local, beta, n)
            jac_func = lambda z_local: _smograd(z_local, beta, n)
            result = minimize(
                obj_func, z, jac=jac_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_qp_run(a_init, deadline, max_rounds=200, gamma_start=2.0, gamma_end=50.0):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive gamma: start low (broad), increase (focused)
        frac = rnd / max(1, max_rounds - 1)
        gamma = gamma_start + (gamma_end - gamma_start) * frac ** 0.7

        # Smooth weights: w_k = (c_k / c_max) ^ gamma
        ratio = c / c_max
        w = ratio ** gamma
        w = w / w.sum()

        # Solve QP: minimize a^T H a s.t. sum(a)=1, a>=0
        # H is Hankel with H_{j,l} = w_{j+l}
        # Use CG: H * y = 1, then a_new = y / sum(y), clip to nonneg
        y = _cg_solve_precon(w, n, np.ones(n), max_iter=600, tol=1e-12)
        sum_y = float(np.sum(y))
        if abs(sum_y) < 1e-20 or not np.isfinite(sum_y):
            # Fallback: projected gradient
            grad = _hankel_matvec(a, w, n)
            lam = float(np.mean(grad))
            step = 0.5 / (float(np.max(np.abs(grad))) + 1e-15)
            step = min(step, 0.3)
            a_new = a - step * (grad - lam)
        else:
            a_new = y / sum_y

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Line search: try blending old and new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try damped step
            found = False
            for t in [0.7, 0.5, 0.3, 0.1]:
                a_try = (1 - t) * a + t * a_new
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    found = True
                    break
            if not found:
                # No improvement, break
                if rnd > 20:
                    break

    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _subgradient_nesterov(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.05
    momentum = 0.9
    tol_frac = 0.001

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Smooth weights
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = z_prev + momentum * (z_new - z) + (1 - momentum) * (-step * grad)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    patience = 0
                    lr *= 0.6

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

        _update_best(a5, sc5, "active_qp_2")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue
        if best_seq is not None:
            a_init_n = _resample(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "tapered")
        t_n_end = min(deadline - 60, time.monotonic() + 120)
        a_opt = _lbfgs_run(a_init_n, t_n_end, min_time_remaining=30.0, aggressive=True)
        sc_opt = _score(a_opt)
        _update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if time.monotonic() < deadline - 40:
            a_qp, sc_qp = _active_qp_run(a_opt, min(deadline - 30, time.monotonic() + 60),
                                          max_rounds=50, gamma_start=3.0, gamma_end=50.0)
            _update_best(a_qp, sc_qp, f"n_{n_try}_qp")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_strategies = [
            ("perturb_small", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_med", lambda: _perturb(best_seq, 0.1, rng)),
            ("tapered", lambda: _make_profile(len(best_seq), "tapered")),
            ("bimodal", lambda: _make_profile(len(best_seq), "bimodal")),
        ]
        for label, gen_fn in restart_strategies[:3]:
            if time.monotonic() > deadline - 15:
                break
            p_start = gen_fn()
            t_r_end = min(deadline - 10, time.monotonic() + 40)
            a_r = _lbfgs_run(p_start, t_r_end, min_time_remaining=5.0)
            sc_r = _score(a_r)
            _update_best(a_r, sc_r, f"restart_{label}")
            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 8:
                a_rq, sc_rq = _active_qp_run(a_r, deadline - 3, max_rounds=30,
                                              gamma_start=5.0, gamma_end=60.0)
                _update_best(a_rq, sc_rq, f"restart_{label}_qp")

    # Phase 8: Final ultra-tight polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_qp_run(best_seq, deadline - 1, max_rounds=100,
                                        gamma_start=20.0, gamma_end=200.0)
        _update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 2:
            a_fd, sc_fd = _coord_descent_polish(best_seq, deadline - 1, max_sweeps=10)
            _update_best(a_fd, sc_fd, "final_coord")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
