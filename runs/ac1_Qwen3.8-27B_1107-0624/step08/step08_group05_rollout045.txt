def _lp_iteration(f, M, solver='CLP'):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    import cvxpy as cp
    
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_j f[k-j] * g[j] for valid j
    constraints = []
    for k in range(2 * n - 1):
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        # (f*g)_k = sum_{j=j_min}^{j_max} f[k-j] * g[j]
        terms = [f[k-j] * g[j] for j in range(j_min, j_max + 1)]
        constraints.append(cp.sum(terms) <= M)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    # Solve
    try:
        prob.solve(solver=solver)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return g.value
    except Exception:
        pass
    
    return None

            alpha *= 0.5

        if not improved:
            mu *= 10.0
            if mu > 1.0:
                break
            stall += 1
            if stall > 5:
                break
            continue
        else:
            stall = 0
            mu = max(mu * 0.9, 1e-12)

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _lbfgs_lse_multi(a_init, deadline, min_remaining=30.0):
    """L-BFGS with LSE smoothing, multi-stage beta schedule."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                                   0.015625, 0.0078125, 0.004, 0.002,
                                   0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective_top, z, args=(beta, n),
                jac=_lse_gradient_top, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lse_objective_top(z, beta, n):
    return _lse_obj(z, beta, n)


def _lse_gradient_top(z, beta, n):
    return _lse_grad(z, beta, n)


def _subgrad_polish(a_init, deadline, n_steps=20000, lr_init=0.008, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    momentum = np.zeros(n)
    beta_m = 0.9

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _convolve(a, a)
        grad_a, _ = _active_grad_vec(a, c, n, tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        momentum = beta_m * momentum + (1 - beta_m) * grad_z
        step = lr / (1.0 + 0.0001 * i)
        z = z - step * momentum
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                beta_m = min(beta_m * 1.01, 0.95)
            else:
                patience += 1
                lr *= 0.92
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, max_amp=0.3):
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if len(a) % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, len(a))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s
