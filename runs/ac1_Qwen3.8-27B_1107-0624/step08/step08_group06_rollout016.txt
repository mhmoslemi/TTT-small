    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()

    lr = 0.1
    decay = 0.999

            break

        c_max = float(c.max())

        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if not active.any():

        w = np.zeros_like(c)

        g = 2.0 * 2.0 * n * (h - W)  # gradient of 2*n*max_c

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _corr_a_A(a, A_vec, n):

    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)

    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                                  0.0156, 0.0078, 0.0039, 0.002, 0.001,
                                  0.0005, 0.00025, 0.0001, 0.00005, 0.000025])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:

def _mirror_descent(a_init, deadline, max_iter=50000):
    n = len(a_init)
    a = a_init.copy()

    a = a / a.sum()
    best_a = a.copy()

        w = np.zeros_like(c)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        g = 4.0 * n * (h - W)
        g = g - np.mean(g)

        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-15)

                lr = 0.05
            if it % 1000 == 0 and lr < 1e-6:

        else:
            a_small = best_a.copy()
            for k_pos in top_positions[:15]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_small[i] * a_small[k_pos - i]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    bigger = max_i if a_small[max_i] >= a_small[j] else j
                    a_small[bigger] *= 0.999
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-12:
                    best_sc = sc_small
                    best_a = a_small.copy()
                else:
                    break
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    elif kind == "block_asym":
        a = np.zeros(n)
        b1 = n // 5
        b2 = 2 * n // 5
        a[:b1] = 1.5
        a[b1:b2] = 1.0
        a[b2:] = 0.7
        s = a.sum()
        return a / s
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining active-set Newton, L-BFGS with
    log-sum-exp smoothing, mirror descent, deflation, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 800, t_start + 300)
    a_main = _run_lbfgs(parent, lbfgs_dl, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_1")

    # Phase 2: Active-set Newton (broad)
    print(f"\n[Phase 2] Active-set Newton broad (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        as_dl = min(deadline - 600, t_start + 450)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=400,
                                         tol_list=[0.02, 0.01, 0.005, 0.002, 0.001])
        update_best(a_as, sc_as, "as_newton_broad")

    # Phase 3: Mirror descent
    print(f"\n[Phase 3] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        md_dl = min(deadline - 400, t_start + 550)
        a_md, sc_md = _mirror_descent(best_seq, md_dl, max_iter=30000)
        update_best(a_md, sc_md, "mirror_desc")

    # Phase 4: L-BFGS round 2 (from best so far)
    print(f"\n[Phase 4] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        lbfgs_dl2 = min(deadline - 350, t_start + 650)
        a_l2 = _run_lbfgs(best_seq, lbfgs_dl2, min_time_remaining=300.0)
        sc_l2 = _score(a_l2)
        update_best(a_l2, sc_l2, "lbfgs_2")

    # Phase 5: Active-set Newton (tight)
    print(f"\n[Phase 5] Active-set Newton tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        as_dl2 = min(deadline - 250, t_start + 750)
        a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=400,
                                           tol_list=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        update_best(a_as2, sc_as2, "as_newton_tight")

    # Phase 6: Deflation + L-BFGS
    print(f"\n[Phase 6] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        def_dl = min(deadline - 200, t_start + 820)
        a_def, sc_def = _deflation_step(best_seq, def_dl, n_attempts=40)
        update_best(a_def, sc_def, "deflation")
        if sc_def < best_score + 0.001 and time.monotonic() < deadline - 100:
            a_dl3 = min(deadline - 100, t_start + 900)
            a_l3 = _run_lbfgs(a_def, a_dl3, min_time_remaining=80.0)
            update_best(a_l3, _score(a_l3), "lbfgs_post_def")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent in n_candidates:
        n_candidates.remove(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try > 32768:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
            sc_init = _score(a_init)
            a_init2 = _make_profile(n_try, "block_asym")
            sc_init2 = _score(a_init2)
            if sc_init2 < sc_init:
                a_init = a_init2
                sc_init = sc_init2
        else:
            a_init = _make_profile(n_try, "uniform")

        print(f"  Trying n={n_try}, init_score={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 30, t_start + 1000),
                                                  max_iter=150, tol_list=[0.002, 0.001, 0.0005, 0.0002])
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 8: Diversified restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        perturbations = [
            ("small", 0.02),
            ("medium", 0.05),
            ("large", 0.1),
            ("anti", 0.15),
        ]
        for label, sigma in perturbations:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, t_start + 1200),
                                                  max_iter=80, tol_list=[0.001, 0.0005, 0.0002])
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 9: Final intensive polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-tight active-set Newton
        as_final_dl = min(deadline - 2, t_start + 1300)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final_dl, max_iter=300,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        update_best(a_fin, sc_fin, "final_as_tight")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00003
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 80, 'maxls': 150}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # Final deflation
        if time.monotonic() < deadline - 2:
            a_def3, sc_def3 = _deflation_step(best_seq, min(deadline - 1, t_start + 1350),
                                              n_attempts=20)
            update_best(a_def3, sc_def3, "final_def")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
