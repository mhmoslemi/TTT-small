"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements:
1. LP-based best response via multiplicative weight updates
2. Multi-resolution warm start (optimize at smaller n, interpolate up)
3. Adaptive active-set Newton with geometrically tightening tolerance
4. Multi-n exploration to find optimal dimension
5. Aggressive beta-continuation L-BFGS
6. Structured initializations targeting asymmetric profiles

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _best_response_iter(f, M, deadline, max_iter=2000, alpha_init=0.05):
    """LP-based best response via multiplicative weight updates.
    Finds g >= 0, sum(g)=1, that maximizes sum(g) subject to (f*g)[k] <= M.
    Uses exponential weight updates to project onto the feasible set.
    """
    n = len(f)
    g = f.copy()
    s = g.sum()
    if s < 1e-15:
        return f.copy()
    g = g / s

    alpha = alpha_init
    best_g = g.copy()
    best_violation = np.inf
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(f, g)
        violations = np.maximum(c - M, 0)
        total_viol = float(violations.sum())

        if total_viol < 1e-14:
            if total_viol < best_violation:
                best_violation = total_viol
                best_g = g.copy()
            patience += 1
            if patience > 50:
                break
        else:
            best_violation = min(best_violation, total_viol)
            best_g = g.copy()
            patience = 0

        # Compute gradient contribution: for each i, how much does g[i] contribute to violations?
        # d(c[k])/d(g[i]) = f[k - i + n - 1] (for valid indices)
        # We want to reduce g[i] proportional to sum_k violations[k] * f[|k-i+n-1|]
        v_clipped = np.clip(violations, 0, None)
        a_rev = f[::-1]
        wc = _full_convolve(v_clipped, a_rev)
        h = wc[n - 1:2 * n - 1]

        # Multiplicative update: reduce g where h is large
        step = alpha * h / max(float(h.max()), 1e-30)
        g = g * np.exp(-step)

        # Ensure g doesn't vanish completely
        g = np.maximum(g, 1e-30)
        s = g.sum()
        if s < 1e-15:
            g = best_g.copy()
            break
        g = g / s

        # Adaptive learning rate
        if it % 50 == 0 and it > 0:
            alpha *= 0.97

    return g


def _active_set_newton(a_init, deadline, max_iter=500, tol_start=0.01, tol_end=0.0001):
    """Active-set Newton with geometrically tightening tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    n_iters_tol = max(10, max_iter // 5)
    tol_schedule = np.geomspace(tol_start, tol_end, n_iters_tol)

    tol_idx = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Update tolerance
        tol_val = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        if it % n_iters_tol == 0 and it > 0:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
            tol_val = tol_schedule[tol_idx]

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_val * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            n_active = 4 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha_cg = rs / pAp
                x += alpha_cg * p
                r_new = r - alpha_cg * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / max(rs, 1e-30)) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback: gradient step
            t = min(0.05, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with log-sum-exp beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[50]
    else:
        gap = c_sorted[0] / 2
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                      0.015, 0.0075, 0.0037, 0.00185, 0.0009,
                                      0.00045, 0.0002, 0.0001, 0.00005,
                                      0.00002, 0.00001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        if n <= 4096:
            est_iter_time = 0.004
        elif n <= 8192:
            est_iter_time = 0.008
        elif n <= 16384:
            est_iter_time = 0.02
        else:
            est_iter_time = 0.04 * (n / 32768.0)

        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 15))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_refine(a_init, deadline, max_rounds=100):
    """Peak-targeted refinement using top-k weighted gradient."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top peaks with exponential weighting
        n_peaks = min(20, len(c))
        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        weights = np.ones(n_peaks)
        # Emphasize the top peak more
        weights[0] = 2.0
        weights = weights / weights.sum()

        w = np.zeros(2 * n - 1)
        w[sorted_idx] = weights

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        grad = 2.0 * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        step = 0.08 / (1.0 + 0.02 * rnd)
        f_old = c_max
        for ls in range(30):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + t
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * (1 - t))
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 0.2 + 0.8 * t
        return a / a.sum()
    elif kind == "sine_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t + 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * t)
        return a / a.sum()
    elif kind == "bimodal_edge":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.1) ** 2) / 0.01) + np.exp(-((t - 0.9) ** 2) / 0.01) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)

        if i % 50 == 0:
            a_check = _softmax(z)

            else:

                lr *= 0.92

                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        par_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={par_sc:.10f}")
        update_best(parent, par_sc, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick multi-n exploration (first 90s) ===
    print(f"\n[Phase 1] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    phase1_end = t_start + 90
    n_candidates = [4096, 8192, 16384, 32768]
    quick_bests = {}

    for n_try in n_candidates:
        if time.monotonic() > phase1_end:
            break
        # Interpolate parent to this n
        if parent is not None:
            a_try = _interpolate_to_n(parent, n_try)
        else:
            a_try = _make_profile(n_try, "uniform")
        sc_try = _score(a_try)
        quick_bests[n_try] = (a_try, sc_try)
        print(f"  n={n_try}: score={sc_try:.10f}")

    # Pick the best n for main optimization
    if quick_bests:
        best_n = min(quick_bests, key=lambda k: quick_bests[k][1])
    else:
        best_n = n_parent
    print(f"  Best n for main optimization: {best_n}")

    main_a = quick_bests.get(best_n, (parent, np.inf))[0].copy()

    # === Phase 2: Multi-resolution warm start (up to 200s) ===
    print(f"\n[Phase 2] Multi-resolution warm start (t={time.monotonic() - t_start:.1f}s)")
    phase2_end = min(t_start + 250, deadline - 800)

    # Optimize at smaller n, then interpolate up
    small_ns = [1024, 2048, 4096]
    current_small = None
    for n_small in small_ns:
        if time.monotonic() > phase2_end:
            break
        if current_small is not None:
            a_small = _interpolate_to_n(current_small, n_small)
        else:
            a_small = _interpolate_to_n(parent, n_small)
        # Quick optimization at this n
        a_small_opt = _run_lbfgs(a_small, min(phase2_end, time.monotonic() + 30),
                                  min_time_remaining=5, aggressive=True)
        sc_small = _score(a_small_opt)
        print(f"  n={n_small}: score={sc_small:.10f}")
        current_small = a_small_opt

    if current_small is not None:
        a_interp = _interpolate_to_n(current_small, best_n)
        sc_interp = _score(a_interp)
        print(f"  Interpolated to n={best_n}: score={sc_interp:.10f}")
        if sc_interp < _score(main_a):
            main_a = a_interp

    # === Phase 3: Aggressive L-BFGS (large portion of time) ===
    print(f"\n[Phase 3] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase3_end = min(deadline - 400, t_start + 250 + 600)
    a_main = _run_lbfgs(main_a, phase3_end, min_time_remaining=200, aggressive=True)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 4: Best response iteration ===
    print(f"\n[Phase 4] Best response (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_br_start = best_seq.copy()
        s = a_br_start.sum()
        a_br_start = a_br_start / s
        c_br = _full_convolve(a_br_start, a_br_start)
        M_br = float(c_br.max())
        phase4_end = min(deadline - 250, time.monotonic() + 120)
        a_br = _best_response_iter(a_br_start, M_br, phase4_end, max_iter=3000, alpha_init=0.03)
        # Convex combination with original
        for t_mix in [0.3, 0.5, 0.7, 1.0]:
            a_mix = (1 - t_mix) * a_br_start + t_mix * a_br
            a_mix = np.maximum(a_mix, 0)
            s_mix = a_mix.sum()
            if s_mix > 1e-15:
                a_mix = a_mix / s_mix
                update_best(a_mix, _score(a_mix), f"bestresp_t{t_mix}")

    # === Phase 5: Active-set Newton with tightening tolerance ===
    print(f"\n[Phase 5] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        phase5_end = min(deadline - 120, time.monotonic() + 200)
        a_as, sc_as = _active_set_newton(best_seq, phase5_end, max_iter=500,
                                          tol_start=0.005, tol_end=0.00005)
        update_best(a_as, sc_as, "as_newton_main")

    # === Phase 6: Peak refinement ===
    print(f"\n[Phase 6] Peak refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        phase6_end = min(deadline - 80, time.monotonic() + 80)
        a_pt, sc_pt = _peak_refine(best_seq, phase6_end, max_rounds=80)
        update_best(a_pt, sc_pt, "peak_refine")

    # === Phase 7: Second active-set pass (tighter) ===
    print(f"\n[Phase 7] Second AS pass (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        phase7_end = min(deadline - 40, time.monotonic() + 50)
        a_as2, sc_as2 = _active_set_newton(best_seq, phase7_end, max_iter=300,
                                            tol_start=0.002, tol_end=0.00001)
        update_best(a_as2, sc_as2, "as_newton_tight")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma in [0.01, 0.03, 0.06]:
            if time.monotonic() > deadline - 20:
                break
            z_pert = np.log(np.maximum(best_seq, 1e-30)) + sigma * rng.standard_normal(len(best_seq))
            z_pert = z_pert - z_pert.mean()
            a_pert = _softmax(z_pert)
            a_opt = _run_lbfgs(a_pert, min(deadline - 15, time.monotonic() + 20),
                                min_time_remaining=5)
            update_best(a_opt, _score(a_opt), f"restart_s{sigma}")

    # === Phase 9: Final subgradient polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 3, n_max_steps=3000, lr_init=0.005)
        update_best(a_fin, sc_fin, "final_sg")

        # One more AS pass
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _active_set_newton(best_seq, deadline - 1, max_iter=100,
                                                  tol_start=0.001, tol_end=0.00001)
            update_best(a_fin2, sc_fin2, "final_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
