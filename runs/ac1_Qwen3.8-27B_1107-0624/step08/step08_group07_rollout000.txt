"""
Optimized step function search using accelerated subgradient methods with
adaptive active-set tracking, diverse initializations, and flattening phases.
Targets n=8192-16384 with careful hyperparameter tuning.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n * max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_subgradient(a, c, lags):
    """Compute subgradient of max(c[k] for k in lags) w.r.t. a.
    g[j] = (1/|lags|) * sum_{k in lags} 2*a[k-j] for valid (k-j)."""
    n = len(a)
    g = np.zeros(n)
    K = len(lags)
    for k in lags:
        k = int(k)
        # c[k] = sum_i a[i] * a[k-i] for valid i
        # d/dc w.r.t a[j] = 2*a[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        # a[k-j] for j in [lo, hi]: when j=lo, k-j = k-lo; when j=hi, k-j = k-hi
        # So a[k-hi : k-lo+1] reversed gives a[k-j] for j=lo,...,hi
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= K
    return g

def _subgradient_step(a, c, lags, lr):
    """Take a subgradient step and project onto simplex."""
    g = _compute_subgradient(a, c, lags)
    # Project gradient to tangent space (preserve sum)
    g = g - np.mean(g)
    a_new = a - lr * g
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), 0.0
    a_new /= s
    return a_new, 1.0

def _polyak_step_size(f_curr, f_ref, g):
    """Polyak step size: theta = (f_curr - f_ref) / ||g||^2"""
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return 0.0
    return (f_curr - f_ref) / (gn * gn)

def _optimize_subgradient(a_init, deadline, max_iter=100000, 
                           active_tol=0.99, init_lr=0.01, min_lr=1e-6):
    """Subgradient method with adaptive step size and active set tracking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_f = _score_norm(a)
    f_ref = best_f  # Reference for Polyak step
    
    lr = init_lr
    no_improve = 0
    max_no_improve = 2000
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        f_curr = 2.0 * n * c_max
        
        if f_curr < best_f - 1e-12:
            best_f = f_curr
            best_a = a.copy()
            no_improve = 0
            f_ref = best_f - 1.0  # Keep Polyak step reasonable
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Find active set
        tol = c_max * active_tol
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Limit active set size
        if len(active) > 20:
            idx = np.argsort(c[active])[::-1][:20]
            active = active[idx]
        
        # Compute subgradient
        g = _compute_subgradient(a, c, active)
        g = g - np.mean(g)  # Preserve sum
        
        # Adaptive step size
        if it % 100 == 0:
            pn = np.linalg.norm(g)
            if pn > 1e-15:
                lr_polyak = (f_curr - f_ref) / (pn * pn)
                lr = max(min_lr, min(0.1, 0.5 * lr + 0.5 * lr_polyak))
            # Decay if stuck
            if no_improve > 500:
                lr *= 0.95
            elif no_improve > 100:
                lr *= 0.99
        
        a_new = a - lr * g
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new /= s_new
        a = a_new
    
    # Final check
    c_final = _fft_convolve(best_a, best_a)
    f_final = 2.0 * n * c_final.max()
    if f_final < best_f:
        best_f = f_final
    return best_a, best_f

def _fista_optimize(a_init, deadline, max_iter=50000, init_lr=0.05):
    """FISTA-style accelerated gradient for the max-of-quadratics problem."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    y = a.copy()  # FISTA momentum variable
    t_k = 1.0
    best_a = a.copy()
    best_f = _score_norm(a)
    lr = init_lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        f_y = 2.0 * n * c_max
        
        if f_y < best_f - 1e-12:
            best_f = f_y
            best_a = y.copy()
        
        # Active set
        tol = c_max * 0.995
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 15:
            idx = np.argsort(c[active])[::-1][:15]
            active = active[idx]
        
        # Subgradient at y
        g = _compute_subgradient(y, c, active)
        g = g - np.mean(g)
        
        # Gradient step
        x_new = y - lr * g
        x_new = np.maximum(0, x_new)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = a.copy()
        else:
            x_new /= s_new
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + (t_k - 1.0) / t_new * (x_new - a)
        y = np.maximum(0, y)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x_new.copy()
        else:
            y = y / s_y
        a = x_new
        t_k = t_new
        
        # Decay learning rate
        lr *= 0.9995
        if lr < 1e-5:
            lr = 1e-5
    
    c_final = _fft_convolve(best_a, best_a)
    f_final = 2.0 * n * c_final.max()
    if f_final < best_f:
        best_f = f_final
    return best_a, best_f
