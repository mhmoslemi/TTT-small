        lr = lr_schedule(t)
        
        grad = _lsgrad(z, beta, n)

        m = beta1 * m + (1 - beta1) * grad

        v_hat = v / (1 - beta2**(t+1))

        # Optional: periodically check score

            a = _softmax(z)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

The key question is: what should `lr` be? Since the gradient has magnitude roughly `4n * O(1)` (where O(1) depends on the scale of a), and we want to take steps of size O(1/n) in z-space (to keep a in a reasonable range), the learning rate should be roughly `O(1/(4n))`.

Actually, let me think about this differently. The z-space is the log of a (approximately). A change of δz in z corresponds to a relative change of δz in a. So if we want a to change by a factor of, say, 0.1 per step, we need δz ≈ 0.1. The gradient in z-space has magnitude ≈ 4n * |a| * |h - W| ≈ 4n * (1/n) * O(1/n) = O(1/n). So lr should be ≈ 0.1 / (1/n) = 0.1n. That seems too large.

Let me just empirically set lr and see. I'll start with `lr = 0.01 * beta` and adjust.

Actually, a better approach: use the gradient norm to normalize. `lr = step_size / (||grad|| + eps)` where `step_size` is a target step size in z-space.

Or even better: use the natural gradient (Fisher information metric) which is automatically adapted to the softmax parameterization. The natural gradient of the log-sum-exp objective with respect to z is just `a * (h - W)` (the centered gradient), and the step is `lr * a * (h - W)` which automatically has the right scale.

Actually, that IS what we already have: `grad_z = 4n * a * (h - W)`. The factor of `4n` is just a scaling of the objective. So the "natural" step would be `lr * a * (h - W)` where lr absorbs the 4n factor.

Let me use: `effective_lr = base_lr / (4n)` so that the update is `base_lr * a * (h - W)`.

With `base_lr = 0.1` (a typical value for multiplicative updates), this should work well.

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam optimizer with adaptive beta and cosine LR schedule on log-sum-exp objective
2. L-BFGS polish with very small beta
3. Active-set Newton refinement
4. Multi-n exploration and perturbation-based restarts
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cb = c / beta
    cb_max = np.max(cb)
    e = np.exp(cb - cb_max)
    s = e / e.sum()
    a_rev = a[::-1]
    sc = _full_convolve(s, a_rev)
    h = sc[n - 1:2 * n - 1]
    W = float(np.dot(s, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize(a_init, deadline, beta_start, beta_end, total_steps, lr_start, lr_end):
    """Adam optimization on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    for t in range(total_steps):
        if time.monotonic() > deadline:
            break
        
        # Beta schedule (exponential decay)
        frac = t / max(1, total_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac if beta_start > 0 else beta_end
        
        # Learning rate schedule (cosine decay)
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * frac))
        
        grad = _lsgrad(z, beta, n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic score check
        if t % 200 == 0 or t == total_steps - 1:
            a = _softmax(z)
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"    Adam step {t}: score={sc:.10f}, beta={beta:.6f}")
    
    # Final check
    a = _softmax(z)
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(4000, max(20, int(remaining / (est_time_per_iter * 5))))
        
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception:
            pass
    
    return current_a


def _active_set_newton(a_init, deadline, max_iter=200):
    """Active-set Newton method."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())
        if m < 2 or m > 4 * n:
            if m > 4 * n:
                top_idx = np.argsort(c)[::-1][:4 * n]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                m = int(active.sum())
            else:
                active[np.argmax(c)] = True
                active[np.argsort(c)[::-1][1]] = True
                m = 2
        
        A_vec = active.astype(np.float64)
        
        # Compute gradient: (2/m) * cross-correlation of a with A_vec
        a_rev = a[::-1]
        conv_result = _full_convolve(A_vec, a_rev)
        g = (2.0 / m) * conv_result[n - 1:2 * n - 1]
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break
        
        # Newton direction via conjugate gradient
        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8
        
        def _hess_vp(v):
            v_rev = v[::-1]
            cr = _full_convolve(A_vec, v_rev)
            return (2.0 / m) * cr[n - 1:2 * n - 1] + mu * v
        
        # CG solve
        x = np.zeros(n)
        r = g_t - _hess_vp(x)
        p = r.copy()
        rs = np.dot(r, r)
        for _ in range(200):
            Ap = _hess_vp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x = x + alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r + (rs_new / rs) * p
            rs = rs_new
        
        # Find w = H^{-1} * 1
        x2 = np.zeros(n)
        r2 = np.ones(n) - _hess_vp(x2)
        p2 = r2.copy()
        rs2 = np.dot(r2, r2)
        for _ in range(200):
            Ap2 = _hess_vp(p2)
            pAp2 = np.dot(p2, Ap2)
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            x2 = x2 + alpha2 * p2
            r2 = r2 - alpha2 * Ap2
            rs2_new = np.dot(r2, r2)
            if np.sqrt(rs2_new) < 1e-12:
                break
            p2 = r2 + (rs2_new / rs2) * p2
            rs2 = rs2_new
        
        w_sum = float(x2.sum())
        if abs(w_sum) < 1e-15:
            dx = -x
        else:
            lam = float(x.sum()) / w_sum
            dx = -x + lam * x2
        
        dx = dx - np.mean(dx)
        
        # Line search
        neg = dx < -1e-12
        if neg.any():
            t_max = np.min(a[neg] / (-dx[neg])) * 0.9
        else:
            t_max = 1.0
        t = min(1.0, t_max)
        t = max(t, 1e-10)
        
        accepted = False
        f_old = c_max
        for ls in range(50):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old - 1e-14 * max(1, t) or ls >= 40:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5
        
        if not accepted:
            break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break
    
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
