Multi-scale gradient optimization for minimizing max autocorrelation.
Uses FFT-based convolution, adaptive temperature smoothing, Nesterov momentum,
coarse-to-fine upsampling, and periodic diversification.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Taper edges
        edge = max(1, n // 10)
        for i in range(edge):
            a[i] = 0.5 + 0.5 * np.sin(np.pi * i / (2 * edge))
            a[n-1-i] = a[i]
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:

    return c_max + beta * np.log(s)

def _make_profile(n, kind="asym_taper"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sym_taper":
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        # Asymmetric: rises quickly, falls slowly
        a = np.sin(np.pi * t ** 0.7)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper2":

        a = np.sin(np.pi * t ** 1.3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_boost":
        # Boost edges relative to center
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t) + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_bump":
        # Single bump offset from center
        a = np.exp(-8 * (t - 0.4) ** 2) + 0.1
        return a / a.sum()
    elif kind == "plateau":
        a = np.ones(n)
        edge = max(1, n // 8)
        for i in range(edge):

            a[n-1-i] = a[i]
        return a / a.sum()
    elif kind == "sqrt_taper":
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:

def _resample_fn(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

Multi-scale gradient optimization for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses L-BFGS with adaptive temperature smoothing, coarse-to-fine upsampling,
asymmetric profile exploration, direct convolution flattening, and
Nesterov momentum gradient descent for polishing.
Key insight: optimal sequences are typically non-symmetric, and the
convolution maximum is not necessarily at the center.

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="asym_taper"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sym_taper":
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        a = np.sin(np.pi * t ** 0.65)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper2":
        a = np.sin(np.pi * t ** 1.4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_boost":
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_bump":
        a = np.exp(-6.0 * (t - 0.35) ** 2) + 0.08
        return a / a.sum()
    elif kind == "plateau":
        a = np.ones(n)
        edge = max(1, n // 6)
        for i in range(edge):
            a[i] = 0.5 + 0.5 * np.sin(np.pi * i / (2.0 * edge))
            a[n - 1 - i] = a[i]
        return a / a.sum()
    elif kind == "sqrt_taper":
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_bump":
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.65) ** 2) + 0.04
        return a / a.sum()
    elif kind == "linear_taper":
        a = 1.0 - 0.5 * np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample_fn(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _smoobj_func(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return 0.0
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_func(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad_a = 2.0 * wc[n - 1:2 * n - 1]
    grad_z = a * (grad_a - float(np.dot(a, grad_a)))
    return grad_z


def _lbfgs_pass(a_init, deadline, min_remaining=20.0, aggressive=True):
    """L-BFGS with adaptive beta continuation in log-space."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    sorted_c = np.sort(c)[::-1]
    gap = sorted_c[0] - sorted_c[min(100, len(sorted_c) - 1)]
    base_beta = max(gap * 0.2, c_max * 0.008)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
                           0.0075, 0.004, 0.002, 0.001, 0.0005, 0.00025,
                           0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                           0.000002, 0.000001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
                           0.0075, 0.004, 0.002, 0.001, 0.0005, 0.00025,
                           0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for ratio in ratios:
        if time.monotonic() + min_remaining > deadline:
            break
        beta = base_beta * ratio
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(50, int(remaining / (est_time_per_iter * 8))))

        def _obj(z_local):
            return _smoobj_func(z_local, beta, n)

        def _jac(z_local):
            return _smograd_func(z_local, beta, n)

        try:
            result = minimize(
                _obj,
                z,
                jac=_jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new < current_sc - 1e-1

5:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            break

    return current_a, current_sc


def _flatten_polish(a_init, deadline, max_iters=3000):
    """Direct iterative convolution flattening with backtracking."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(np.sum(c) / len(c))

        if c_max <= c_mean * 1.001:
            break

        # Target: between mean and max
        frac = 0.3 + 0.5 * (it / max_iters)
        target = c_mean + frac * (c_max - c_mean)

        excess = np.maximum(c - target, 0)
        if float(excess.max()) < 1e-18 * c_max:
            break

        # Gradient: 2 * cross-correlation of excess with a
        a_rev = a[::-1]
        ec = _full_convolve(excess, a_rev)
        grad = 2.0 * ec[n - 1:2 * n - 1]

        # Adaptive step with backtracking
        base_step = 0.05 / (1.0 + 0.005 * it)
        step = base_step
        improved = False
        for _ in range(30):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if c_try.max() < c_max - 1e-16:
                a = a_try
                improved = True
                sc_try = _score(a)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a.copy()
                    patience = 0
                break
            step *= 0.5

        if not improved:
            patience += 1
            if patience > 200:
                # Add small perturbation to escape
                rng = np.random.default_rng(it)
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                patience = 0

    return best_a, best_sc


def _momentum_gd(a_init, deadline, max_steps=10000):
    """Nesterov momentum gradient descent in log-space."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    sorted_c = np.sort(c)[::-1]
    gap = sorted_c[0] - sorted_c[min(100, len(sorted_c) - 1)]
    base_beta = max(gap * 0.1, c_max * 0.01)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    momentum = 0.92
    lr = 0.08
    patience = 0

    n_betas = 30
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        beta_idx = min(int(i / (max_steps / n_betas)), n_betas - 1)
        beta = base_beta * (1.0 - beta_idx / n_betas) ** 2 + base_beta * 0.001

        a = _softmax_fn(z)
        grad_z = _smograd_func(z, beta, n)

        v = momentum * v + (1.0 - momentum) * grad_z
        z = z - lr * v
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if i % 100 == 0:
            a_new = _softmax_fn(z)
            sc = _score(a_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    lr *= 0.8
                    patience = 0

    a_final = _softmax_fn(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _active_set_polish(a_init, deadline, max_rounds=200):
    """Progressive active-set QP style polish."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                    0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                    0.000002, 0.000001, 0.0000005]
    tol_idx = 0
    rounds_per_tol = 8

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Compute gradient under active weights
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        grad_centered = grad - float(np.mean(grad))

        # Newton-like step: solve approximately
        # For simplicity, use scaled gradient with backtracking
        gn = float(np.linalg.norm(grad_centered))
        if gn < 1e-15:
            break

        step = 0.5 / (gn + 1e-10)
        improved = False
        for _ in range(30):
            a_try = a - step * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if c_try.max() < c_max:
                a = a_try
                sc_try = _score(a)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try smaller perturbation
            a_try = a - 0.01 * grad_centered / (gn + 1e-10)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                a = a_try

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _perturb_fn(a, sigma, seed):
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_fn(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with coarse-to-fine strategy, diverse initializations,
    L-BFGS with adaptive temperature, active-set polish, and flattening.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"dep={prev - best_score:.2e} t={time.monotonic()

 - t_start:.1f}s")

    # Load parent if available
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    # === Phase 1: Diverse coarse optimization (n=512) ===
    print(f"\n[Phase 1] Coarse diverse starts at n=512 (t={time.monotonic() - t_start:.1f}s)")
    n_coarse = 512
    coarse_profiles = ["asym_taper", "asym_taper2", "sym_taper", "sqrt_taper",
                       "edge_boost", "plateau", "cos_bump", "double_bump"]
    coarse_best = None
    coarse_best_sc = np.inf
    phase1_budget = min(120.0, (deadline - t_start) * 0.08)

    for prof in coarse_profiles:
        if time.monotonic() > t_start + phase1_budget:
            break
        a_init = _make_profile(n_coarse, prof)
        if parent is not None:
            a_init = 0.5 * a_init + 0.5 * _resample_fn(parent, n_coarse)
            a_init = a_init / a_init.sum()
        a_opt, sc_opt = _lbfgs_pass(a_init, t_start + phase1_budget,
                                     min_remaining=5.0, aggressive=False)
        if sc_opt < coarse_best_sc:
            coarse_best_sc = sc_opt
            coarse_best = a_opt.copy()
        print(f"    {prof}: {sc_opt:.8f}")

    if coarse_best is not None:
        _update_best(coarse_best, coarse_best_sc, "coarse_best")
    else:
        coarse_best = _make_profile(n_coarse, "asym_taper")
        coarse_best_sc = _score(coarse_best)
        if parent is not None:
            _update_best(_resample_fn(parent, n_coarse), None, "parent_coarse")
            coarse_best = _resample_fn(parent, n_coarse)
            coarse_best_sc = _score(coarse_best)

    # === Phase 2: Coarse-to-fine upsampling ===
    print(f"\n[Phase 2] Coarse-to-fine (t={time.monotonic() - t_start:.1f}s)")
    n_target = max(len(parent) if parent is not None else 16384, 4096)
    n_target = min(n_target, 32768)
    scales = []
    n_cur = n_coarse
    while n_cur < n_target:
        n_cur = min(n_cur * 2, n_target)
        scales.append(n_cur)
    if not scales:
        scales = [n_target]

    current_a = coarse_best
    current_sc = coarse_best_sc

    phase2_budget = min(400.0, (deadline - t_start) * 0.25)
    for i, n_s in enumerate(scales):
        if time.monotonic() > t_start + phase2_budget:
            break
        if n_s <= len(current_a):
            continue
        print(f"  Upsampling to n={n_s} (t={time.monotonic() - t_start:.1f}s)")
        a_up = _resample_fn(current_a, n_s)
        # Also try parent resampled
        if parent is not None:
            a_parent_up = _resample_fn(parent, n_s)
            # Blend
            a_up = 0.6 * a_up + 0.4 * a_parent_up
            a_up = np.maximum(a_up, 0)
            a_up = a_up / a_up.sum()

        alloc = min(60.0, (t_start + phase2_budget - time.monotonic()) * 0.4)
        a_opt, sc_opt = _lbfgs_pass(a_up, t_start + phase2_budget,
                                     min_remaining=30.0, aggressive=True)
        if sc_opt < current_sc:
            current_sc = sc_opt
            current_a = a_opt
        _update_best(a_opt, sc_opt, f"upsample_{n_s}")

        # Quick flatten
        if _score(a_opt) < best_score + 0.01 and time.monotonic() < t_start + phase2_budget - 10:
            a_fl, sc_fl = _flatten_polish(a_opt, t_start + phase2_budget - 5, max_iters=500)
            _update_best(a_fl, sc_fl, f"flatten_{n_s}")
            if sc_fl < current_sc:
                current_sc = sc_fl
                current_a = a_fl

    # === Phase 3: Intensive L-BFGS at target scale ===
    print(f"\n[Phase 3] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a3, sc3 = _lbfgs_pass(current_a, min(deadline - 300, t_start + 600),
                               min_remaining=100.0, aggressive=True)
        _update_best(a3, sc3, "lbfgs_intensive")
        if sc3 < current_sc:
            current_sc = sc3
            current_a = a3

    # === Phase 4: Active-set polish ===
    print(f"\n[Phase 4] Active-set polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a4, sc4 = _active_set_polish(current_a, min(deadline - 200, t_start + 800),
                                     max_rounds=300)
        _update_best(a4, sc4, "active_set")
        if sc4 < current_sc:
            current_sc = sc4
            current_a = a4

    # === Phase 5: Flatten polish ===
    print(f"\n[Phase 5] Flatten polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a5, sc5 = _flatten_polish(current_a, min(deadline - 150, t_start + 1000),
                                  max_iters=5000)
        _update_best(a5, sc5, "flatten_main")
        if sc5 < current_sc:
            current_sc = sc5
            current_a = a5

    # === Phase 6: Momentum GD ===
    print(f"\n[Phase 6] Momentum GD (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a6, sc6 = _momentum_gd(current_a, min(deadline - 100, t_start + 1200),
                               max_steps=20000)
        _update_best(a6, sc6, "momentum_gd")
        if sc6 < current_sc:
            current_sc = sc6
            current_a = a6

    # === Phase 7: Another L-BFGS round with tighter params ===
    print(f"\n[Phase 7] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a7, sc7 = _lbfgs_pass(current_a, min(deadline - 80, t_start + 1400),
                               min_remaining=30.0, aggressive=True)
        _update_best(a7, sc7, "lbfgs_r2")
        if sc7 < current_sc:
            current_sc = sc7
            current_a = a7

    # === Phase 8: Final active-set + flatten ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        a8, sc8 = _active_set_polish(current_a, min(deadline - 40, t_start + 1500),
                                     max_rounds=200)
        _update_best(a8, sc8, "active_final")
        if sc8 < current_sc:
            current_sc = sc8
            current_a = a8

    if time.monotonic() < deadline - 20:
        a9, sc9 = _flatten_polish(current_a, min(deadline - 20, t_start + 1600),
                                  max_iters=3000)
        _update_best(a9, sc9, "flatten_final")
        if sc9 < current_sc:
            current_sc = sc9
            current_a = a9

    # === Phase 9: Diversified restarts with time remaining ===
    print(f"\n[Phase 9] Diversified restarts (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 40 and best_seq is not None:
        n_div = len(best_seq)
        restart_types = [
            ("perturb_small", 0.02),
            ("perturb_med", 0.05),
            ("asym_blend", None),
        ]
        for name, sigma in restart_types:
            if time.monotonic() > deadline - 25:
                break
            if sigma is not None:
                a_r = _perturb_fn(best_seq, sigma, int(time.monotonic() * 1000) % 100000)
            else:
                a_r = 0.5 * best_seq + 0.5 * _make_profile(n_div, "asym_taper")
                a_r = a_r / a_r.sum()

            a_ro, sc_ro = _lbfgs_pass(a_r, deadline - 20, min_remaining=5.0,
                                       aggressive=False)
            _update_best(a_ro, sc_ro, f"restart_{name}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    final_seq = None
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(4096, "asym_taper")
        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, "
              f"time={time.monotonic() - t_start:.1f}s")

    if final_seq is None:
        final_seq = _make_profile(8192, "asym_taper")
        sc_final = _score(final_seq)
        print(f"FALLBACK: score={sc_final:.10f}, n={len(final_seq)}")

    return [float(x) for x in final_seq]
