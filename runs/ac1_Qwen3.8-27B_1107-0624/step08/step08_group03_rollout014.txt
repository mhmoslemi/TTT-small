import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):

    c_cur = _full_convolve(a_init, a_init)

    if max_c < 1e-15:

    c_sorted = np.sort(c_cur)[::-1]

        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    # More aggressive continuation
    betas = base_beta
    n_betas = 25
    beta_ratios = np.exp(np.linspace(0, np.log(1e-6), n_betas))

    current_sc = _score(current_a)

def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) aggressive log-sum-exp L-BFGS with 30-step beta continuation,
(2) Nesterov momentum peak shaving, (3) multi-tolerance active-set Newton,
(4) crossover between diverse solutions, (5) larger n exploration,
(6) final subgradient polish with adaptive step.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_betas=30):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    beta_ratios = np.exp(np.linspace(0.0, np.log(1e-7), n_betas))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i in range(n_betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * beta_ratios[i]

        if n <= 4096:
            est_tpi = 0.002
        elif n <= 8192:
            est_tpi = 0.004
        elif n <= 16384:
            est_tpi = 0.01
        elif n <= 32768:
            est_tpi = 0.04
        else:
            est_tpi = 0.08

        max_iter = min(4000, max(30, int(remaining / (est_tpi * 12))))

        try:
            def obj_func(z_local):
                return _lsobj(z_local, beta, n)

            def grad_func(z_local):
                return _lsgrad(z_local, beta, n)

            result = minimize(
                obj_func, z, jacobian=grad_func,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_shave_nesterov(a_init, deadline, max_steps=8000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    v = np.zeros(n)
    lr = lr_init
    patience = 0
    decay = 0.997

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Exponentially weighted peaks (top 30 with geometric decay)
        k = min(30, max(5, len(c) // 100))
        top_k = np.argsort(c)[::-1][:k]
        weights = np.exp(-np.arange(k) * 0.25)
        w = np.zeros_like(c)
        for j, idx in enumerate(top_k):
            w[idx] = weights[j]
        w = w / w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Nesterov momentum
        v = 0.9 * v - lr * grad
        z_new = z + v
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new
        lr *= decay

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = lr_init * 0.7

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001):

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:n]

            active[top_idx] = True
            n_active = n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        mu = 1e-8 * c_max + 1e-12

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(max_cg):

                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13:
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w

        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for _ in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:

            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)

                accepted = True

def _crossover(a1, a2, t):
    n1, n2 = len(a1), len(a2)
    if n1 != n2:
        a2 = _interpolate_to_n(a2, n1)
    z1 = np.log(np.maximum(a1, 1e-30))
    z2 = np.log(np.maximum(a2, 1e-30))
    z = (1 - t) * z1 + t * z2
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 0.3 + 1.5 * t**2
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    solutions_pool = []

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")
        if len(solutions_pool) < 20:
            solutions_pool.append(p.copy())

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS on parent
    print(f"\n[Phase 1] L-BFGS n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    t1_deadline = min(deadline - 600, time.monotonic() + 400)
    a1 = _run_lbfgs(parent, t1_deadline, min_time_remaining=200.0, n_betas=30)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Nesterov peak shaving
    print(f"\n[Phase 2] Nesterov peak shave (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        t2_deadline = min(deadline - 400, time.monotonic() + 120)
        a2, sc2 = _peak_shave_nesterov(best_seq, t2_deadline, max_steps=5000, lr_init=0.03)
        update_best(a2, sc2, "nesterov")

    # Phase 3: Active-set Newton (multiple tolerances)
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        t3a = min(deadline - 200, time.monotonic() + 40)
        a3a, sc3a = _active_set_newton(best_seq, t3a, max_iter=150, tol_active=0.005)
        update_best(a3a, sc3a, "as_loose")

        t3b = min(deadline - 150, time.monotonic() + 40)
        a3b, sc3b = _active_set_newton(best_seq, t3b, max_iter=150, tol_active=0.001)
        update_best(a3b, sc3b, "as_mid")

        t3c = min(deadline - 100, time.monotonic() + 40)
        a3c, sc3c = _active_set_newton(best_seq, t3c, max_iter=150, tol_active=0.0003)
        update_best(a3c, sc3c, "as_tight")

    # Phase 4: Try larger n
    print(f"\n[Phase 4] Larger n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_large = 32768
    if time.monotonic() < deadline - 200 and n_parent < n_large:
        a_large_init = _interpolate_to_n(parent, n_large)
        sc_large_init = _score(a_large_init)
        print(f"  n={n_large} init: {sc_large_init:.10f}")
        t4_deadline = min(deadline - 100, time.monotonic() + 100)
        a_large = _run_lbfgs(a_large_init, t4_deadline, min_time_remaining=50.0, n_betas=20)
        sc_large = _score(a_large)
        print(f"  n={n_large} after L-BFGS: {sc_large:.10f}")
        if sc_large < best_score:
            update_best(a_large, sc_large, f"lbfgs_n{n_large}")
        elif sc_large < best_score + 0.01:
            t4b = min(deadline - 50, time.monotonic() + 40)
            a_large2, sc_large2 = _peak_shave_nesterov(a_large, t4b, max_steps=3000, lr_init=0.02)
            update_best(a_large2, sc_large2, f"nesterov_n{n_large}")

    # Phase 5: Crossover
    print(f"\n[Phase 5] Crossover (t={time.monotonic()-t_start:.1f}s)")
    if len(solutions_pool) >= 2 and time.monotonic() < deadline - 60:
        # Sort by score
        scored = []
        for sol in solutions_pool:
            scored.append((_score(sol), sol))
        scored.sort(key=lambda x: x[0])
        scored = scored[:6]  # top 6

        for i in range(min(4, len(scored))):
            if time.monotonic() > deadline - 30:
                break
            for j in range(i + 1, min(i + 2, len(scored))):
                for t in [0.3, 0.5, 0.7]:
                    if time.monotonic() > deadline - 20:
                        break
                    a_cross = _crossover(scored[i][1], scored[j][1], t)
                    if len(a_cross) == len(best_seq):
                        sc_cross = _score(a_cross)
                        if sc_cross < best_score + 0.005:
                            tc_deadline = min(deadline, time.monotonic() + 15)
                            a_cross_opt = _run_lbfgs(a_cross, tc_deadline, min_time_remaining=5.0, n_betas=10)
                            update_best(a_cross_opt, _score(a_cross_opt), f"cross_{i}{j}_t{t}")
                        else:
                            update_best(a_cross, sc_cross, f"cross_{i}{j}_t{t}")

    # Phase 6: Structured restarts
    print(f"\n[Phase 6] Structured restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        for kind in ["ramp", "cos_asym", "quadratic"]:
            if time.monotonic() > deadline - 20:
                break
            a_struct = _make_profile(n_parent, kind)
            sc_struct = _score(a_struct)
            if sc_struct < best_score + 0.3:
                # Interpolate with best
                a_mix = _crossover(best_seq, a_struct, 0.7)
                a_mix_opt = _run_lbfgs(a_mix, deadline, min_time_remaining=10.0, n_betas=15)
                update_best(a_mix_opt, _score(a_mix_opt), f"restart_{kind}")

    # Phase 7: Final L-BFGS polish
    print(f"\n[Phase 7] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a7 = _run_lbfgs(best_seq, deadline - 5, min_time_remaining=5.0, n_betas=15)
        update_best(a7, _score(a7), "final_lbfgs")

    # Phase 8: Final active-set with very tight tolerance
    print(f"\n[Phase 8] Final AS Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a8, sc8 = _active_set_newton(best_seq, deadline - 3, max_iter=100, tol_active=0.0001)
        update_best(a8, sc8, "final_as_ultra")

    # Phase 9: Final subgradient
    print(f"\n[Phase 9] Final subgrad (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a9, sc9 = _subgrad_polish(best_seq, deadline - 1, n_max_steps=5000, lr_init=0.003)
        update_best(a9, sc9, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
