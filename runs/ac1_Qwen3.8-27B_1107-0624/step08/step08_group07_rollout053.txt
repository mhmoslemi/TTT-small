                   eval_interval=50, top_k=10):
    """Adam optimizer in log-space with smoothed max objective."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        frac = min(1.0, t / max_iter)
        T = T_init * (T_final / T_init) ** frac
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _fft_convolve(a, a)
        
        # Compute smooth max and weights
        smooth_val, w = _smooth_max_and_weights(c, T)
        
        # But for efficiency, only use top (top_k + 5) lags
        c_sorted = np.argsort(c)[::-1]
        active = c_sorted[:top_k + 5]
        w_active = w[active]
        w_active = w_active / w_active.sum()
        w_masked = np.zeros_like(c)
        w_masked[active] = w_active
        
        # Gradient
        g = _gradient_smooth_max(a, c, w_masked)
        
        # Project to tangent space (sum constraint in log-space)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        lr_t = lr * (1.0 - 0.5 * frac)  # linear decay
        update = lr_t * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > 20:
                # Reset from best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0
    
    return best_a, best_score

def _newton_active(a, deadline, max_iter=100, tol=0.005):
    """Newton's method to equalize active peaks."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks (within tol of max)
        active = np.where(c >= c_max * (1 - tol))[0]
        if len(active) < 2:
            break
        if len(active) > 20:
            # Keep only the top 20
            active = active[np.argsort(c[active])[::-1][:20]]
        
        K = len(active)
        
        # Build Jacobian J[i,j] = 2*a[active[i] - j] (with bounds)
        J = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                J[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # We want to find d such that:
        # J*d = -(c[active] - t) where t = mean(c[active]) (to equalize)
        # sum(d) = 0
        # Minimum norm solution
        
        target = np.zeros(K)  # we want J*d = 0 (equalize to current mean)
        # Actually, we want the active peaks to be equal. 
        # c[active] + J*d should be all equal to their mean.
        c_act = c[active]
        c_mean = c_act.mean()
        rhs_J = c_mean - c_act  # we want c_act + J*d = c_mean, so J*d = c_mean - c_act
        
        # System: [J; ones] * d = [rhs_J; 0]
        M = np.vstack([J, np.ones(n)])
        rhs = np.concatenate([rhs_J, [0.0]])
        
        # Minimum norm: d = M^T * pinv(M*M^T) * rhs
        G = M @ M.T  # (K+1) x (K+1)
        try:
            # Add small regularization
            G_reg = G + 1e-10 * np.eye(G.shape[0])
            d = M.T @ np.linalg.solve(G_reg, rhs)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M, rhs, rcond=None)[0]
        
        # Line search
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        improved = False
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score:
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score

def _local_search(a, deadline, max_rounds=10000):
    """Local search: try pairwise mass transfers and single-coordinate scaling."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    
    # Identify "important" coordinates (those that contribute to peaks)
    peak_lags = np.argsort(c)[::-1][:20]
    important = set()
    for k in peak_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        important.update(range(lo, hi + 1))
    important = list(important)
    if len(important) > 200:
        # Prioritize coordinates with larger values
        important.sort(key=lambda j: -best_a[j])
        important = important[:200]
    
    rng = np.random.default_rng(12345)
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Choose two coordinates
        if rng.random() < 0.7:
            # Pairwise transfer
            i = important[rng.integers(len(important))]
            j = important[rng.integers(len(important))]
            if i == j:
                continue
            epsilon = best_a[i] * 0.1 * (1 if rng.random() < 0.5 else -1)
            if best_a[j] + epsilon < 0:
                epsilon = -best_a[j] * 0.5
            a_cand = best_a.copy()
            a_cand[i] -= epsilon
            a_cand[j] += epsilon
        else:
            # Single coordinate scale
            i = important[rng.integers(len(important))]
            if best_a[i] < 1e-15:
                continue
            scale = 1.0 + 0.2 * (rng.random() - 0.5)
            a_cand = best_a.copy()
            a_cand[i] *= scale
        
        a_cand = np.maximum(0, a_cand)
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_cand /= s
        
        c_cand = _fft_convolve(a_cand, a_cand)
        sc = float(2.0 * n * c_cand.max())
        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = a_cand.copy()
            c = c_cand
    
    return best_a, best_score

def _spectral_flatten(a, deadline, n_iters=200):
    """Flatten the power spectrum to reduce convolution peaks."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    N = next_fast_len(2 * n - 1)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        
        # Check if spectrum is already flat
        P_mean = P.mean()
        P_std = P.std()
        if P_std / P_mean < 0.01:
            break
        
        # Modify spectrum: reduce peaks, boost valleys
        # But maintain the DC component (which determines the sum)
        P_new = P.copy()
        P_dc = P_new[0]  # preserve DC
        P_ac = P_new[1:]
        
        # Soft clipping of peaks
        threshold = P_ac.mean() * 2.0
        excess = np.maximum(P_ac - threshold, 0)
        deficit = np.maximum(threshold - P_ac, 0)
        
        # Move some excess to deficit
        transfer = 0.1 * excess.sum()
        if deficit.sum() > 0:
            P_ac += transfer * deficit / deficit.sum() - 0.1 * excess
        P_new[1:] = P_ac
        P_new[0] = P_dc  # restore DC
        
        # Take square root and inverse FFT
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    
    return best_a, best_score
