        # {d : sum(d) = 0, d[j] >= 0 where a[j] = 0}
        # Approximate: just subtract the weighted mean
        g_tan = g - float(np.dot(a, g))
        
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break
        
        # Step with projection
        a_new = a - step * g_tan / g_norm
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            step = max(1e-8, step * 0.9999)
        else:
            step = max(1e-10, step * 0.95)
            if step < 1e-10:
                break
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Adam in log-space (multiple temperatures)
    Phase 2: Gauss-Newton equalization
    Phase 3: Subgradient polish
    Phase 4: Final fine-tuning
    Tries multiple n values and keeps the best.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

        if parent.sum() > 0.01:
            parent = parent / parent.sum()
        else:
            parent = None
    except Exception:
        pass
    
    # Decide which n values to try
    n_candidates = [8192]
    if parent is not None:
        n_candidates.append(len(parent))

        if n_try not in n_candidates and len(n_candidates) < 3:

    n_candidates = sorted(set(n_candidates))
    
    time_per_n = (deadline - t_start) / len(n_candidates)
    
    for n in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        
        n_deadline = min(deadline - 30, t_start + time_per_n * n_candidates.index(n) + time_per_n)

            a = parent.copy()
        elif parent is not None:

                parent
            )
            a = a_new / a_new.sum()
        else:
            # Use a structured initialization
            # Cosine-tapered uniform
            j = np.arange(n)
            a = 1.0 + 0.5 * np.cos(2 * np.pi * j / n)
