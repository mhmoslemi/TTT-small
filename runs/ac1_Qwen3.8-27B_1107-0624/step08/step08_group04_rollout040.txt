```python
def _project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not cond.any():
        return np.ones(n) / n
    k = np.where(cond)[0][-1]
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0)
```

The QP is:
minimize (1/2) a^T H a
subject to 1^T a = 1

where H = 2 * sum_k lambda_k H_k (the block-Hankel matrix).

The KKT conditions (for the case where a > 0 everywhere):

If any a_i <= 0, we need to handle the inequality constraint. The standard approach is:
1. Solve the equality-only problem
2. If any a_i < 0, add i to the "inactive" set (fix a_i = 0)
3. Restrict H to the remaining indices and re-solve
4. Repeat

This is the active-set method. For efficiency, we only need to solve the system on the "active" (non-zero) indices.

But for large n, even the equality-only solve requires many CG iterations if H is ill-conditioned. Let me think about conditioning.

H = 2 * sum_k lambda_k H_k. If lambda is spread out (like uniform), H is a "sum of many anti-diagonals" which is well-conditioned. If lambda is concentrated on few lags, H might be ill-conditioned.

In the early iterations, lambda is spread out, so CG converges fast. In later iterations, lambda concentrates on peaks, and H might become ill-conditioned. This is where preconditioning helps.

diag(H)_i = 2 * sum_k lambda_k * [1 if 0 <= k-i <= n-1 else 0] = 2 * sum_{k: max(0,i) <= k <= min(2n-2, i+n-1)} lambda_k

This is just a "sliding window sum" of lambda, which can be computed in O(n) using a cumulative sum.

Preconditioned CG should converge much faster with this.

```python
Uses the dual (simplex-constrained) to guide active-set selection in the primal QP,
combined with L-BFGS polishing and peak-specific repair.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
```

```python
def _pcg_solve(lam, n, rhs, x0=None, max_iter=500, tol=1e-12):
    """Preconditioned CG to solve H_lambda x = rhs."""
    n_local = n
    if x0 is None:
        x = np.zeros(n_local)
    else:
        x = x0.copy()
    
    r = rhs - _hankel_matvec(x, lam, n_local)
    z = _precond(r, lam, n_local)
    p = z.copy()
    rz = float(r @ z)
    
    for i in range(max_iter):
        Hp = _hankel_matvec(p, lam, n_local)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30:
            break
        alpha = rz / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rz_new = float(r_new @ _precond(r_new, lam, n_local))
        if abs(rz_new) < tol * tol * float(rhs @ rhs):
            x = x_new
            break
        p_new = _precond(r_new, lam, n_local) + (rz_new / rz) * p
        x = x_new
        r = r_new
        z = _precond(r, lam, n_local)
        p = p_new
        rz = rz_new
    
    return x

def _precond(r, lam, n):
    """Jacobi preconditioner: divide by diagonal of H_lambda."""
    # diag(H)_i = 2 * sum_{k=i}^{i+n-1} lam_k (truncated)
    # = 2 * (cumsum(lam)[i+n-1] - cumsum(lam)[i-1]) (with boundary handling)
    cs = np.cumsum(lam)
    diag = np.zeros(n)
    for i in range(n):
        lo = max(0, i)
        hi = min(2*n-2, i+n-1)
        if hi >= lo:
            diag[i] = 2.0 * (cs[hi] - (cs[lo-1] if lo > 0 else 0))
        else:
            diag[i] = 1e-10
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_matvec(v, lam, n):
    """Compute (H*v)[i] = 2 * sum_{m=0}^{n-1} lam[i+m] * v[m] for i=0..n-1."""
    V = v[::-1]
    out_len = n + len(lam) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_lam = np.fft.rfft(lam, N)
    conv = np.fft.irfft(F_V * F_lam, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _compute_hankel_diag(lam, n):
    """Compute diagonal of H_lambda (size n)."""
    L = len(lam)
    cs = np.concatenate([[0.0], np.cumsum(lam)])
    upper = np.minimum(L, np.arange(n) + n)
    lower = np.arange(n)
    diag = 2.0 * (cs[upper] - cs[lower])
    return np.maximum(diag, 1e-15)


def _pcg_solve(lam, n, rhs, x0=None, max_iter=300, tol=1e-13):
    """Preconditioned CG to solve H_lambda * x = rhs."""
    if x0 is None:
        x = np.zeros(n)
    else:
        x = x0.copy()
    diag = _compute_hankel_diag(lam, n)

    r = rhs - _hankel_matvec(x, lam, n)
    z = r / diag
    p = z.copy()
    rz = float(r @ z)
    rhs_norm = float(rhs @ rhs)
    if rhs_norm < 1e-30:
        return x
    target = tol * tol * rhs_norm

    for _ in range(max_iter):
        Hp = _hankel_matvec(p, lam, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        z_new = r_new / diag
        rz_new = float(r_new @ z_new)
        if not np.isfinite(rz_new) or abs(rz_new) > abs(rz) * 2:
            break
        x = x_new
        r = r_new
        if abs(rz_new) < target:
            break
        p = z_new + (rz_new / rz) * p
        rz = rz_new
    return x


def _cg_solve(lam, n, rhs, x0=None, max_iter=300, tol=1e-13):
    """Plain CG to solve H_lambda * x = rhs."""
    if x0 is None:
        x = np.zeros(n)
    else:
        x = x0.copy()
    r = rhs - _hankel_matvec(x, lam, n)
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    target = tol * tol * rs0

    for _ in range(max_iter):
        Hp = _hankel_matvec(p, lam, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        x = x_new
        r = r_new
        if rs_new < target:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not cond.any():
        return np.ones(n) / n
    k = np.where(cond)[0][-1]
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0.0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _primal_dual_phase(a_init, deadline, min_remaining=300.0):
    """Primal-dual iterations for peak autocorrelation minimization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    lam = np.ones(2 * n - 1) / (2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    # Adaptive parameters
    momentum = 0.3
    dual_power = 1.0

    max_iters = 800
    for it in range(max_iters):
        if time.monotonic() + min_remaining > deadline:
            break

        # Primal step: solve H_lambda x = 1
        x = _pcg_solve(lam, n, np.ones(n), x0=None, max_iter=250, tol=1e-12)
        a_pos = np.maximum(x, 0)
        s = a_pos.sum()
        if s < 1e-15:
            # Fallback: use previous a
            a_new = a.copy()
        else:
            a_new = a_pos / s

        # Compute autocorrelation for dual update
        c = _full_convolve(a_new, a_new)
        c_sum = c.sum()
        if c_sum < 1e-30:
            break

        # Dual update: tempered + momentum
        lam_target = (c / c_sum) ** dual_power
        lam_target = lam_target / lam_target.sum()
        lam = (1.0 - momentum) * lam + momentum * lam_target
        lam = lam / lam.sum()

        # Track best
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # Adapt parameters
        if it < 100:
            momentum = 0.3 + 0.3 * (it / 100.0)
            dual_power = 1.0 + 0.5 * (it / 100.0)
        elif it < 300:
            momentum = 0.6 + 0.2 * ((it - 100) / 200.0)
            dual_power = 1.5 + 1.0 * ((it - 100) / 200.0)
        else:
            momentum = 0.8
            dual_power = 2.5 + 0.5 * ((it - 300) / max(1, max_iters - 300))

        if it % 100 == 0:
            print(f"    PD iter {it}: score={sc:.10f}, momentum={momentum:.2f}, power={dual_power:.2f}")

    return best_a, best_sc


def _active_set_qp(a_init, deadline, min_remaining=100.0, max_rounds=60,
                   tol_schedule=None):
    """Active-set QP with progressive tolerance tightening and PCG."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                        0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                        0.000001, 0.0000005, 0.0000002, 0.0000001,
                        0.00000005, 0.00000002, 0.00000001]

    rounds_per_tol = 3
    tol_idx = 0

    for rnd in range(max_rounds):
        if time.monotonic() + min_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        lam = np.zeros(2 * n - 1)
        lam[active] = 1.0 / n_active

        # Solve QP: min a^T H a / 2 s.t. sum(a)=1, a>=0
        # KKT: H a = mu * 1 (for full support case)
        x = _pcg_solve(lam, n, np.ones(n), x0=a, max_iter=500, tol=1e-14)
        a_qp = np.maximum(x, 0)
        s_qp = a_qp.sum()
        if s_qp < 1e-15:
            a_qp = a.copy()
        else:
            a_qp = a_qp / s_qp

        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-15:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Damped update
            a_try = 0.5 * a + 0.5 * a_qp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _peak_repair(a_init, deadline, min_remaining=30.0):
    """Targeted repair of top peaks via local gradient steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(200):
        if time.monotonic() + min_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-3 peaks with soft weighting
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_act = max(1, int(active.sum()))
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act

        # Gradient of weighted sum of active peaks
        # d/d a[i] [sum_k w_k (a*a)[k]] = 2 * sum_k w_k * (H_k a)_i
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 2.0 * n * a * (h - W_val)

        # Normalized gradient step
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Adaptive step size with backtracking
        alpha = 0.1 / (1.0 + 0.05 * it)
        for _ in range(20):
            a_try = a - alpha * grad / gn
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n

    a_pd, sc_pd = _primal_dual_phase(parent, pd_deadline, min_remaining=60.0)
    update_best(a_pd, sc_pd, "primal_dual")

    # Phase 2: L-BFGS polish from primal-dual result
    print(f"\n[Phase 2] L-BFGS Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        lbfgs_deadline = t_start + pd_time_budget + min(350.0, (deadline - t_start) * 0.25)
        a_lbfgs = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=200.0, aggressive=True)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_1")

    # Phase 3: Active-set QP with tight tolerances
    print(f"\n[Phase 3] Active-Set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        qp_deadline = t_start + pd_time_budget + min(700.0, (deadline - t_start) * 0.5)
        a_qp, sc_qp = _active_set_qp(best_seq, qp_deadline, min_remaining=150.0,
                                     max_rounds=80)
        update_best(a_qp, sc_qp, "active_qp_1")

    # Phase 4: Peak repair
    print(f"\n[Phase 4] Peak Repair (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        repair_deadline = t_start + pd_time_budget + min(800.0, (deadline - t_start) * 0.6)
        a_rep, sc_rep = _peak_repair(best_seq, repair_deadline, min_remaining=100.0)
        update_best(a_rep, sc_rep, "peak_repair")

    # Phase 5: Second L-BFGS from QP result
    print(f"\n[Phase 5] L-BFGS Round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a_lb2 = _run_lbfgs(best_seq, min(deadline - 80, t_start + 1400.0),
                           min_time_remaining=60.0, aggressive=True)
        update_best(a_lb2, _score(a_lb2), "lbfgs_2")

    # Phase 6: Tight active-set QP round 2
    print(f"\n[Phase 6] Active-Set QP Round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        tight_tols = [0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                      0.000002, 0.000001, 0.0000005, 0.0000002,
                      0.0000001, 0.00000005, 0.00000002, 0.00000001]
        a_qp2, sc_qp2 = _active_set_qp(best_seq, min(deadline - 30, t_start + 1600.0),
                                       min_remaining=30.0, max_rounds=60,
                                       tol_schedule=tight_tols)
        update_best(a_qp2, sc_qp2, "active_qp_2")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 12288, 20480]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 40:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, min(deadline - 20, t_start + 1700.0),
                          min_time_remaining=20.0, aggressive=False)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 8: Final ultra-tight polish
    print(f"\n[Phase 8] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        ultra_tols = [0.000001, 0.0000005, 0.0000002, 0.0000001,
                      0.00000005, 0.00000002, 0.00000001, 0.000000005]
        a_fin, sc_fin = _active_set_qp(best_seq, deadline - 2,
                                       min_remaining=2.0, max_rounds=40,
                                       tol_schedule=ultra_tols)
        update_best(a_fin, sc_fin, "final_qp")

    # Phase 9: Final peak repair
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_frep, sc_frep = _peak_repair(best_seq, deadline - 1, min_remaining=1.0)
        update_best(a_frep, sc_frep, "final_repair")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
