    n, m = len(a), len(b)
    out_len = n + m - 1

    fa = np.fft.rfft(a, N)

    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

$\frac{\partial a_i}{\partial z_j} = a_i (\delta_{ij} - a_j)$
