import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    out_n = n_a + n_b - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs_single_beta(a_current, z, beta, n, max_iter, deadline, min_remaining=5.0):

        def __init__(self, beta_val, n_val):
            self.beta = beta_val
            self.n = n_val
        def __call__(self, z_local):
            return _lsobj(z_local, self.beta, self.n)

    class ObjGrad:
        def __init__(self, beta_val, n_val):
            self.beta = beta_val
            self.n = n_val
        def __call__(self, z_local):
            return _lsgrad(z_local, self.beta, self.n)

    grad_func = ObjGrad(beta, n)

                'ftol': 1e-18,
                'gtol': 1e-14,
                'maxcor': 50,
                'maxls': 50,
            }

        return result.x, True
    except Exception:
        return z, False


def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=30.0, n_betas=40):

    L-BFGS with adaptive beta continuation.
    Uses a geometric beta schedule from large to very small.
    """

    beta_start = max_c * 0.05
    beta_end = max_c * 0.00002

    betas = beta_start * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    t_func = time.monotonic()

        if remaining < 10:

def _multiplicative_update(a, deadline, n_iter=200):
    """
    Multiplicative update (power method variant) to flatten autocorrelation.
    a[i] <- a[i] / (load[i])^{eta} where load[i] measures contribution to peaks.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Weight by proximity to max
        tol = 0.05
        w = np.exp(np.clip((c - c_max) / (c_max * tol), -100, 0))
        w = w / w.sum()

        # Compute load: for each i, sum of w[k] * a[k-i] over valid k
        a_rev = a[::-1]
        load_full = _full_convolve(w, a_rev)
        load = load_full[n - 1:2 * n - 1]

        # Normalized load
        load_mean = load.mean()
        if load_mean < 1e-30:
            break

        # Multiplicative update with damping
        eta = 0.3
        ratio = (load / load_mean) ** eta
        a_new = a / ratio
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller eta
            eta2 = 0.1
            ratio2 = (load / load_mean) ** eta2
            a_try = a / ratio2
            a_try = np.maximum(a_try, 1e-30)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    """Perturb in log-space (preserves non-negativity)."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

                    print(f"    final beta={beta_f:.2e}, score={best_score:.10f}")
            except Exception:
                pass

        # One more multiplicative pass
        if time.monotonic() < deadline - 10:
            a_final_mul, sc_final_mul = _multiplicative_update(best_seq, deadline - 5, n_iter=100)
            update_best(a_final_mul, sc_final_mul, "final_multiplicative")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
