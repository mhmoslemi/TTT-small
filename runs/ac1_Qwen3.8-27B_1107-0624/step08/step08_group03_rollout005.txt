"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: parametric family search + improved log-sum-exp L-BFGS + active-set Newton
with monotonicity exploitation + diverse n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _parametric_exp_power(n, alpha, p):
    t = np.linspace(0, 1, n)
    a = np.exp(-alpha * t**p)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _parametric_exp_neg_power(n, alpha, p):
    t = np.linspace(0, 1, n)
    a = (1.0 - t * 0.999)**p
    a = a * np.exp(-alpha * t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _parametric_linear(n, alpha):
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * t
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _parametric_quad(n, alpha, beta):
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * t + beta * t * t
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _parametric_bump(n, center, width, height):
    t = np.linspace(0, 1, n)
    a = 1.0 + height * np.exp(-((t - center) / width) ** 2)
    return a / a.sum()


def _score_parametric_exp_power(n, params):
    alpha, p = params
    if alpha < 0 or p < 0.01:
        return 1e10
    a = _parametric_exp_power(n, alpha, p)
    return _score(a)


def _parametric_search(n, deadline):
    """Search over parametric families to find good starting points."""
    best_a = None
    best_sc = np.inf

    # Grid search over alpha and p for exp(-alpha * t^p)
    alphas = [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0]
    ps = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]

    for alpha in alphas:
        if time.monotonic() > deadline:
            break
        for p in ps:
            if time.monotonic() > deadline:
                break
            a = _parametric_exp_power(n, alpha, p)
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # Refine with scipy
    if best_a is not None and time.monotonic() < deadline - 5:
        # Find which params gave best
        for alpha in alphas:
            for p in ps:
                a = _parametric_exp_power(n, alpha, p)
                if np.allclose(a, best_a, atol=1e-6):
                    res = minimize_scalar(
                        lambda x: _score_parametric_exp_power(n, (x, p)),
                        bounds=(max(0.01, alpha * 0.5), alpha * 2.0),
                        method='bounded', options={'maxiter': 100}
                    )
                    a_ref = _parametric_exp_power(n, res.x, p)
                    sc_ref = _score(a_ref)
                    if sc_ref < best_sc:
                        best_sc = sc_ref
                        best_a = a_ref.copy()
                    res2 = minimize_scalar(
                        lambda x: _score_parametric_exp_power(n, (alpha, x)),
                        bounds=(max(0.05, p * 0.5), p * 2.0),
                        method='bounded', options={'maxiter': 100}
                    )
                    a_ref2 = _parametric_exp_power(n, alpha, res2.x)
                    sc_ref2 = _score(a_ref2)
                    if sc_ref2 < best_sc:
                        best_sc = sc_ref2
                        best_a = a_ref2.copy()
                    break
            else:
                continue
            break

    # Also try linear and quadratic families
    for alpha in np.linspace(-0.5, 5.0, 20):
        if time.monotonic() > deadline:
            break
        a = _parametric_linear(n, alpha)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    for alpha in np.linspace(-2.0, 5.0, 15):
        for beta in np.linspace(-3.0, 5.0, 15):
            if time.monotonic() > deadline:
                break
            a = _parametric_quad(n, alpha, beta)
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # Try bump functions (non-monotone)
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.05, 0.1, 0.2, 0.3]:
            for height in [0.5, 1.0, 2.0, 5.0]:
                if time.monotonic() > deadline:
                    break
                a = _parametric_bump(n, center, width, height)
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

    return best_a, best_sc


def _run_lbfgs_improved(a_init, deadline, min_time_remaining=30.0):
    """Improved log-sum-exp L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Estimate the "effective" number of near-peaks
    threshold = max_c * 0.99
    n_peaks = int((c_cur >= threshold).sum())
    gap = max_c - c_cur[min(n_peaks, len(c_cur) - 1)] if n_peaks < len(c_cur) else max_c * 0.01

    base_beta = max(gap * 0.3, max_c * 0.005)

    # More aggressive schedule with many more steps
    betas = base_beta * np.array([
        1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.025,
        0.015, 0.009, 0.005, 0.003, 0.0018, 0.001, 0.0006,
        0.00035, 0.0002, 0.00012, 0.00007, 0.00004
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        if n <= 4096:
            est_tpi = 0.003
        elif n <= 8192:
            est_tpi = 0.006
        elif n <= 16384:
            est_tpi = 0.012
        elif n <= 32768:
            est_tpi = 0.025
        else:
            est_tpi = 0.05 * (n / 65536.0)

        max_iter = min(3000, max(50, int(remaining / (est_tpi * 8))))

        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, try with half the beta
                pass
        except Exception:
            pass

    return current_a


def _active_set_newton_v2(a_init, deadline, max_iter=200, tol_active=0.002):
    """Improved active-set Newton with better handling."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size
        max_active = min(3 * n, len(c))
        if n_active > max_active:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:max_active]
            active[top_idx] = True
            n_active = max_active

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of the average active quadratic form
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = float(np.linalg.norm(g_t))
        if gnorm < 1e-15:
            break

        # Hessian with regularization
        mu = 1e-5 * (c_max / max(m, 1)) + 1e-10

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve for the Newton step
        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha_cg = rs / pAp
                x += alpha_cg * p
                r_new = r - alpha_cg * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search with backtracking
        f_old = float(c.max())
        accepted = False
        t_ls = min(1.0, 0.99 * np.min(a[a < -dx + 1e-15] / (-dx[a < -dx + 1e-15])) if np.any(a < -dx + 1e-15) else 1.0)
        t_ls = max(t_ls, 1e-10)

        for ls in range(50):
            a_try = a + t_ls * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t_ls *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t_ls * gnorm or ls == 49:
                a = a_try
                accepted = True
                break
            t_ls *= 0.5

        if not accepted:
            # Fallback: steepest descent step
            t_ls = 0.05
            for ls in range(30):
                a_try = a - t_ls * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t_ls *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _subgrad_momentum(a_init, deadline, n_steps=5000, lr_init=0.02):
    """Subgradient method with momentum for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-k weighted subgradient
        k = min(30, len(c) // 5)
        top_k = np.argsort(c)[::-1][:k]
        # Use exponential weighting
        weights = np.exp(-(c[top_k] - c_max) / (c_max * 0.01 + 1e-20))
        weights = weights / weights.sum()

        w = np.zeros(2 * n - 1)
        w[top_k] = weights

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.0005 * i)
        v = momentum * v - step * grad
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                if patience < 5:
                    lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr *= 0.8
                    momentum *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _peak_flatten(a_init, deadline, max_rounds=100):
    """Target the top peaks specifically using weighted subgradient."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Focus on the top peaks with increasing aggressiveness
        n_peaks = min(5 + rnd // 10, 20)
        top_peaks = np.argsort(c)[::-1][:n_peaks]

        # Exponential weighting: focus more on the highest peaks
        peak_vals = c[top_peaks]
        weights = np.exp(-(peak_vals - c_max) / (c_max * 0.005 + 1e-20))
        weights = weights / weights.sum()

        w = np.zeros(2 * n - 1)
        w[top_peaks] = weights

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)

        # Adaptive step size
        step = 0.15 / (1.0 + 0.005 * rnd)
        f_old = c_max

        for ls in range(20):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        # Convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _monotone_project(a):
    """Project onto the set of non-increasing sequences (isotonic regression, decreasing)."""
    # We want a[0] >= a[1] >= ... >= a[n-1]
    # Use pool-adjacent-violators algorithm on the negated array
    neg_a = -a
    n = len(neg_a)
    result = neg_a.copy()
    blocks = [0]
    weights = [1]
    for i in range(1, n):
        blocks.append(i)
        weights.append(1)
        # Merge while constraint violated
        while len(blocks) > 1 and result[blocks[-2]] < result[blocks[-1]]:
            # Merge last two blocks
            b1, b2 = blocks[-2], blocks[-1]
            w1, w2 = weights[-2], weights[-1]
            new_val = (result[b1] * w1 + result[b2] * w2) / (w1 + w2)
            result[b1] = new_val
            result[b2] = new_val
            if len(blocks) >= 3:
                if result[blocks[-3]] > new_val + 1e-15:
                    break
            blocks.pop()
            weights.pop()
            blocks[-1] = b1
            weights[-1] = w1 + w2
    return -result


def _monotone_param_search(n, deadline):
    """Search over monotone parametric families."""
    best_a = None
    best_sc = np.inf

    # Power-law decreasing: a[i] = (1 - i/(n-1))^p
    for p in np.linspace(0.2, 10.0, 50):
        if time.monotonic() > deadline:
            break
        t = np.linspace(0, 1, n)
        a = (1.0 - t * 0.9999)**p
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Exponential: a[i] = exp(-alpha * i/(n-1))
    for alpha in np.linspace(0.1, 20.0, 60):
        if time.monotonic() > deadline:
            break
        t = np.linspace(0, 1, n)
        a = np.exp(-alpha * t)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Generalized: a[i] = exp(-alpha * (i/(n-1))^p)
    for alpha in [0.5, 1.0, 2.0, 4.0, 8.0, 15.0]:
        for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
            if time.monotonic() > deadline:
                break
            t = np.linspace(0, 1, n)
            a = np.exp(-alpha * t**p)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Parametric search across multiple n values
    2. Log-sum-exp L-BFGS refinement
    3. Active-set Newton
    4. Peak-flattening subgradient
    5. Diverse restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None

        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 16384
        parent = _parametric_exp_power(n_parent, 2.0, 1.0)
        print(f"No parent, using parametric init n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Multi-n parametric search
    print(f"\n[Phase 1] Multi-n parametric search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192, 16384, 32768]
    param_deadline = deadline - 800

    for n_try in n_candidates:
        if time.monotonic() > param_deadline:
            break

        a_param, sc_param = _monotone_param_search(n_try, min(param_deadline, time.monotonic() + 30))
        if a_param is not None:
            update_best(a_param, sc_param, f"param_n{n_try}")

        # General parametric search
        a_gen, sc_gen = _parametric_search(n_try, min(param_deadline, time.monotonic() + 20))
        if a_gen is not None:
            update_best(a_gen, sc_gen, f"genparam_n{n_try}")

    # Phase 2: L-BFGS from parent and best parametric
    print(f"\n[Phase 2] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    n_main = max(n_parent, 8192)

    # From parent
    if time.monotonic() < deadline - 500:
        a_lbfgs = _run_lbfgs_improved(parent.copy(), deadline - 400, min_time_remaining=300)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_parent")

    # From best parametric (if different n, interpolate)
    if best_seq is not None and len(best_seq) != n_parent:
        a_interp = _interpolate_to_n(best_seq, n_parent)
        if time.monotonic() < deadline - 400:
            a_lbfgs2 = _run_lbfgs_improved(a_interp, deadline - 300, min_time_remaining=250)
            update_best(a_lbfgs2, _score(a_lbfgs2), "lbfgs_interp")

    # Phase 3: Active-set Newton
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_as, sc_as = _active_set_newton_v2(best_seq.copy(), min(deadline - 100, time.monotonic() + 120), max_iter=150, tol_active=0.003)
        update_best(a_as, sc_as, "as_newton_wide")

    if best_seq is not None and time.monotonic() < deadline - 150:
        a_as2, sc_as2 = _active_set_newton_v2(best_seq.copy(), min(deadline - 80, time.monotonic() + 80), max_iter=100, tol_active=0.001)
        update_best(a_as2, sc_as2, "as_newton_tight")

    # Phase 4: Peak flattening
    print(f"\n[Phase 4] Peak flattening (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_pf, sc_pf = _peak_flatten(best_seq.copy(), min(deadline - 50, time.monotonic() + 80), max_rounds=80)
        update_best(a_pf, sc_pf, "peak_flatten")

    # Phase 5: Subgradient with momentum
    print(f"\n[Phase 5] Subgradient momentum (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_sg, sc_sg = _subgrad_momentum(best_seq.copy(), min(deadline - 40, time.monotonic() + 60), n_steps=3000, lr_init=0.015)
        update_best(a_sg, sc_sg, "subgrad_mom")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restarts = [
            ("small_pert", 0.01),
            ("med_pert", 0.03),
            ("large_pert", 0.08),
        ]
        for label, sigma in restarts:
            if time.monotonic() > deadline - 20:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(len(z))
            z = z - z.mean()
            a_pert = _softmax(z)
            a_opt = _run_lbfgs_improved(a_pert, deadline - 10, min_time_remaining=5)
            update_best(a_opt, _score(a_opt), f"restart_{label}")

            # Quick Newton polish
            if time.monotonic() < deadline - 10:
                a_n, sc_n = _active_set_newton_v2(a_opt, min(deadline - 5, time.monotonic() + 20), max_iter=50, tol_active=0.002)
                update_best(a_n, sc_n, f"restart_{label}_newton")

        # Try monotone projection of best
        a_mono = _monotone_project(best_seq.copy())
        s_mono = a_mono.sum()
        if s_mono > 0.01:
            a_mono = a_mono / s_mono
            a_mono_opt = _run_lbfgs_improved(a_mono, deadline - 10, min_time_remaining=5)
            update_best(a_mono_opt, _score(a_mono_opt), "monotone_proj")

    # Phase 7: Try different n for the best shape
    print(f"\n[Phase 7] Multi-n final check (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for n_try in [4096, 8192, 16384, 32768, 65536]:
            if time.monotonic() > deadline - 10:
                break
            if n_try == len(best_seq):
                continue
            a_resized = _interpolate_to_n(best_seq, n_try)
            sc_resized = _score(a_resized)
            if sc_resized < best_score:
                update_best(a_resized, sc_resized, f"resized_n{n_try}")
            elif time.monotonic() < deadline - 10:
                # Quick optimize at this n
                a_opt_n = _run_lbfgs_improved(a_resized, min(deadline - 5, time.monotonic() + 20), min_time_remaining=3)
                update_best(a_opt_n, _score(a_opt_n), f"opt_n{n_try}")

    # Phase 8: Final active-set with very tight tolerance
    print(f"\n[Phase 8] Final tight Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton_v2(best_seq.copy(), deadline - 2, max_iter=80, tol_active=0.0005)
        update_best(a_fin, sc_fin, "final_newton_tight")

    # Phase 9: Final peak flatten
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_pf2, sc_pf2 = _peak_flatten(best_seq.copy(), deadline - 1, max_rounds=30)
        update_best(a_pf2, sc_pf2, "final_peak")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
