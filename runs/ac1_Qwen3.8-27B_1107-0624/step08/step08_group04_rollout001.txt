"""
Multi-scale adaptive gradient optimizer with Fourier-domain diagnostics,
Nesterov-accelerated log-sum-exp smoothing, CG-based active-set QP,
and multi-resolution refinement. Targets score < 1.5027 by combining
aggressive temperature annealing, adaptive step sizes, and n=32768 exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=2000, tol=1e-16):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * (1.0 + 1e-6):
            break
        x = x_new
        r = r_new
        p = r + (rs_new / max(rs, 1e-30)) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap * 0.2, max_c * 0.001)

    if aggressive:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625,
                  0.0078125, 0.00390625, 0.002, 0.001, 0.0005, 0.00025,
                  0.000125, 0.0000625, 0.00003, 0.000015, 0.0000075,
                  0.000004, 0.000002, 0.000001, 0.0000005, 0.0000002,
                  0.0000001, 0.00000005, 0.00000002]
    else:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625,
                  0.0078125, 0.004, 0.002, 0.001, 0.0005, 0.00025,
                  0.000125, 0.0000625, 0.00003125, 0.000015, 0.0000075,
                  0.000004, 0.000002, 0.000001]
    betas = [base_beta * r for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 8))))

        try:
            result = minimize(
                lambda z_local, b=beta, nn=n: _smoobj(z_local, b, nn),
                z,
                jac=lambda z_local, b=beta, nn=n: _smograd(z_local, b, nn),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _adaptive_gradient(a_init, deadline, max_steps=500000, label=""):
    """
    Nesterov-accelerated gradient with adaptive temperature and backtracking.
    Works in log-space for automatic positivity.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_mom = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Adaptive temperature schedule
    c_init = _full_convolve(a_init, a_init)
    c_max_init = float(c_init.max())
    c_sorted = np.sort(c_init)[::-1]
    gap_init = c_max_init - c_sorted[min(200, len(c_sorted) - 1)]
    tau_start = max(gap_init * 0.1, c_max_init * 0.0005)
    tau_end = c_max_init * 1e-7

    lr_base = 0.05
    momentum = 0.9
    decay_rate = 0.99997

    patience = 0
    max_patience = 5000
    step = 0

    while time.monotonic() < deadline and step < max_steps:
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive temperature
        progress = step / max(1, min(max_steps, 100000))
        progress = min(progress, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress
        tau = max(tau, c_max * 1e-8)

        # Softmax weights on autocorrelation
        shifted = (c - c_max) / tau
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        # Gradient: d/max_c_smo / dz_i = 2*n*a_i*(h_i - W)
        # where h = (w_convolved_with_a_rev)[n-1:2n-1], W = w@c
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad_z = 2.0 * n * a * (h - W_val)

        # Center gradient to maintain normalization
        grad_z = grad_z - a * np.dot(grad_z / np.maximum(a, 1e-30), a)

        # Backtracking line search
        lr = lr_base * (decay_rate ** step)
        improved = False
        for ls in range(20):
            z_trial = z_mom - lr * grad_z
            z_trial = z_trial - z_trial.mean()
            a_trial = _softmax(z_trial)
            sc_trial = _score(a_trial)
            if sc_trial < best_sc - 1e-15:
                improved = True
                break
            lr *= 0.5
            if lr < 1e-12:
                break

        if improved:
            z_new = z_mom - lr * grad_z
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50.0, 50.0)
            z_mom = z + momentum * (z_new - z)
            z_mom = z_mom - z_mom.mean()
            z = z_new
            patience = 0
        else:
            z_mom = z + momentum * (z - z_mom) * 0.5
            z_mom = z_mom - z_mom.mean()
            patience += 1
            if patience > max_patience:
                # Restart from best
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                z_mom = z.copy()
                lr_base *= 0.8
                patience = 0

        # Periodic scoring (expensive)
        if step % 500 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-14:
                best_sc = sc_check
                best_a = a_check.copy()

        step += 1

    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _direct_equalization(a_init, deadline, max_rounds=500):
    """
    Direct peak equalization: iteratively solve small linear systems
    to equalize the top autocorrelation values.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find active set: top autocorrelation values
        sort_idx = np.argsort(c)[::-1]
        # Adaptive threshold: include values within relative tolerance
        tol = 0.001 * c_max
        active_mask = c >= (c_max - tol)
        n_active = int(active_mask.sum())

        if n_active < 2:
            active_mask[sort_idx[:5]] = True
            n_active = 5
        if n_active > 500:
            active_mask = np.zeros_like(c)
            active_mask[sort_idx[:500]] = True
            n_active = 500

        # Compute gradient direction to "flatten" active peaks
        # g_j = sum_{k in active} w_k * 2 * a_{k-j}
        w = np.zeros_like(c)
        w[active_mask] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Center for sum preservation
        grad = grad - grad.mean()

        # Newton-like step: solve H*dx = -grad where H is the "active Hessian"
        # H = sum_{k in active} w_k * (S_k^T S_k) -- this is the Hessian of sum w_k * c_k
        # Use CG to solve
        dx = _cg_solve(w, n, -grad, max_iter=500, tol=1e-14)

        # Line search
        alpha = 1.0
        c_max_cur = c_max
        for ls in range(30):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_cur * (1 - 0.001):
                a = a_try
                c_max_cur = c_max_try
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _active_qp_polish(a_init, deadline, max_rounds=60, tol_schedule=None):
    """CG-Newton active-set QP with tighter tolerances."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                        0.00002, 0.00001, 0.000005, 0.000002,
                        0.000001, 0.0000005, 0.0000002, 0.0000001,
                        0.00000005, 0.00000002, 0.00000001]

    tol_idx = 0
    rounds_per_tol = 4

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2000:
            sort_idx = np.argsort(c)[::-1]
            active = np.zeros_like(c)
            active[sort_idx[:2000]] = True
            n_active = 2000

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve: min a^T H_W a s.t. sum(a)=1
        # KKT: H_W a = lambda * 1, 1^T a = 1
        # => a = H_W^{-1} 1 / (1^T H_W^{-1} 1)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=1500, tol=1e-16)
        oneH1 = float(np.sum(H1))
        if abs(oneH1) < 1e-20 or not np.isfinite(oneH1):
            tol_idx += 1
            continue

        a_new = H1 / oneH1
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            tol_idx += 1
            continue
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Damped update
            mix = 0.5
            a_mix = (1 - mix) * a + mix * a_new
            a_mix = np.maximum(a_mix, 0)
            s_mix = a_mix.sum()
            if s_mix > 1e-15:
                a_mix = a_mix / s_mix
                sc_mix = _score(a_mix)
                if sc_mix < best_sc:
                    best_sc = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
            tol_idx += 1

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _fourier_guided_init(n, reference=None):
    """
    Create an initial sequence guided by Fourier-domain considerations.
    Aim for a flat power spectrum with non-negative time-domain signal.
    """
    if reference is not None:
        x_old = np.linspace(0, 1, len(reference))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, reference)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
        return a

    # Tapered with slight asymmetry
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 0.7
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.8) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS from parent ===
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 300.0
    a1 = _run_lbfgs(parent, min(p1_deadline, deadline - 600),
                    min_time_remaining=60.0, aggressive=True)
    _update_best(a1, _score(a1), "lbfgs_main")

    # === Phase 2: Adaptive gradient (Nesterov) ===
    print(f"\n[Phase 2] Adaptive gradient (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = t_start + 600.0
    if best_seq is not None:
        a2, sc2 = _adaptive_gradient(best_seq, min(p2_deadline, deadline - 500),
                                     max_steps=200000, label="grad_1")
        _update_best(a2, sc2, "adaptive_grad_1")

    # === Phase 3: Direct equalization ===
    print(f"\n[Phase 3] Direct equalization (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = t_start + 780.0
    if best_seq is not None:
        a3, sc3 = _direct_equalization(best_seq, min(p3_deadline, deadline - 400),
                                       max_rounds=300)
        _update_best(a3, sc3, "direct_eq_1")

    # === Phase 4: Active-set QP ===
    print(f"\n[Phase 4] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = t_start + 960.0
    if best_seq is not None:
        a4, sc4 = _active_qp_polish(best_seq, min(p4_deadline, deadline - 350),
                                    max_rounds=40)
        _update_best(a4, sc4, "active_qp_1")

    # === Phase 5: Resample to n=32768 and optimize ===
    print(f"\n[Phase 5] n=32768 exploration (t={time.monotonic() - t_start:.1f}s)")
    n_large = 32768
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_large = _resample(best_seq, n_large)
        p5a_deadline = min(t_start + 1260, deadline - 200)
        a5 = _run_lbfgs(a_large, p5a_deadline, min_time_remaining=60.0, aggressive=True)
        _update_best(a5, _score(a5), "n32768_lbfgs")

        # Gradient polish on large
        if time.monotonic() < deadline - 200:
            p5b_deadline = min(t_start + 1560, deadline - 100)
            a5b, sc5b = _adaptive_gradient(best_seq if len(best_seq) == n_large else a_large,
                                            p5b_deadline, max_steps=100000)
            _update_best(a5b, sc5b, "n32768_grad")

        # QP on large
        if best_seq is not None and len(best_seq) == n_large and time.monotonic() < deadline - 100:
            p5c_deadline = min(t_start + 1700, deadline - 30)
            a5c, sc5c = _active_qp_polish(best_seq, p5c_deadline, max_rounds=20)
            _update_best(a5c, sc5c, "n32768_qp")

    # === Phase 6: Final gradient polish on best ===
    print(f"\n[Phase 6] Final gradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        p6_deadline = deadline - 5
        a6, sc6 = _adaptive_gradient(best_seq, p6_deadline, max_steps=500000)
        _update_best(a6, sc6, "final_grad")

    # === Phase 7: Quick restart with different profile ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_r = len(best_seq)
        # Asymmetric taper restart
        a_asym = _make_profile(n_r, "asym_taper")
        # Blend with best
        a_blend = 0.3 * a_asym + 0.7 * best_seq
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()

        p7_deadline = deadline - 5
        a7 = _run_lbfgs(a_blend, p7_deadline, min_time_remaining=5.0)
        _update_best(a7, _score(a7), "restart_asym")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
