"""
Multi-phase optimizer for minimum peak autocorrelation sequences.
Uses adaptive log-sum-exp smoothing with L-BFGS, momentum-based peak shaving,
and coordinate descent for final polish.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# Smoother and gradient for L-BFGS
def _soft_max_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _soft_max_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _soft_max_obj(z, self.beta, self.n)
    def grad(self, z):
        return _soft_max_grad(z, self.beta, self.n)

def _run_lbfgs_adaptive(a_init, deadline, min_remaining=100.0):
    """L-BFGS with continuously decreasing beta."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_sc = _score(a)
    best_a = a.copy()
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < 1e-15:
        gap = c_max * 0.001
    beta = max(gap, c_max * 0.005)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    total_time = deadline - time.monotonic()
    n_phases = 8
    phase_time = total_time / n_phases
    
    for phase in range(n_phases):
        if time.monotonic() + min_remaining > deadline:
            break
        
        phase_deadline = min(deadline - min_remaining, time.monotonic() + phase_time)
        prob = _SmoProb(beta, n)
        
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        # Decrease beta
        beta *= 0.4
    
    return best_a, best_sc

    n = len(a_init)

    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    velocity = np.zeros(n)
    momentum = 0.85
    K = 3
    lr = 0.02
    no_improve = 0
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        no_improve += 1
        
        # Adaptive K
        K = min(100, max(2, int(K * (1.02 if no_improve % 20 == 0 else 1.0))))
        
        # Top K peaks
        top_idx = np.argpartition(c, -K)[-K:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / K
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        g = 2.0 * h
        g -= g.mean()
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive lr
        lr_adapt = lr / (gnorm + 1e-10)

        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            velocity = np.zeros(n)
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc - 1e-15:
            a = a_new
        else:

            if no_improve % 50 == 0:
                lr *= 0.8
                if lr < 1e-6:
                    velocity = np.zeros(n)
                    lr = 0.02
                    momentum = 0.85
                    K = 3
    
    return best_a, best_sc

def _coordinate_descent(a_init, deadline, min_remaining=30.0):

"""
Multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Strategy: adaptive L-BFGS with log-sum-exp smoothing, Nesterov momentum peak shaving,
incremental coordinate descent, and multiple restarts. Targets fine improvement
over the parent (1.505662) to beat 1.5027.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remaining=100.0, n_phases=10, beta_decay=0.35):
    """L-BFGS with geometrically decreasing beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_sc = _score(a)
    best_a = a.copy()

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = c_max * 0.002
    beta = max(gap, c_max * 0.004)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    total_time = max(1.0, deadline - min_remaining - time.monotonic())
    phase_time = total_time / n_phases

    for phase in range(n_phases):
        if time.monotonic() + min_remaining + 2.0 > deadline:
            break

        prob = _SmoProb(beta, n)
        est_per_iter = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(phase_time / (est_per_iter * 15))))

        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-16:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        beta *= beta_decay

    return best_a, best_sc

                continue

            denom_sq = (1.0 + delta) ** 2
            if denom_sq < 1e-10:
                continue

            # Affected range: k in [idx, idx+n-1]
            k_lo = idx
            k_hi = min(conv_len - 1, idx + n - 1)
            length = k_hi - k_lo + 1
            a_vals = a[:length]

            # New values in affected range
            new_vals = (c[k_lo:k_hi + 1] + 2.0 * delta * a_vals) / denom_sq

            # Add delta^2 term at k=2*idx if in range
            if 2 * idx >= k_lo and 2 * idx <= k_hi:
                new_vals[2 * idx - k_lo] += delta * delta / denom_sq

            new_max_inside = float(new_vals.max())

            # Outside max: c[k]/denom_sq for k outside [k_lo, k_hi]
            if k_lo == 0 and k_hi == conv_len - 1:
                new_max = new_max_inside
            else:
                if k_lo > 0:
                    outside_left = c[:k_lo].max()
                else:
                    outside_left = 0.0
                if k_hi < conv_len - 1:
                    outside_right = c[k_hi + 1:].max()
                else:
                    outside_right = 0.0
                new_max = max(new_max_inside, outside_left / denom_sq, outside_right / denom_sq)

            if new_max < best_new_max - 1e-18:
                best_new_max = new_max
                best_delta = delta

        if best_delta != 0.0:
            a[idx] = a_idx + best_delta
            s_new = a.sum()
            a = a / s_new
            # Recompute convolution (expensive but necessary for correctness after normalization)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            sc = 2.0 * n * c_max
            if sc < best_sc - 1e-16:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 20000:
            break

    return best_a, best_sc


def _active_set_fine(a_init, deadline, min_remaining=30.0):
    """Fine active-set optimization: directly minimize peak by adjusting
    elements that contribute to it, using a smoothed objective over top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.01
    no_improve = 0
    tol_frac = 0.002

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        no_improve += 1

        # Active set: peaks within tol_frac of max
        threshold = c_max * (1.0 - tol_frac)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c)
            active[idx] = 1.0
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h
        g -= g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(30):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if no_improve % 20 == 0:
                tol_frac = max(0.0001, tol_frac * 0.8)
                lr *= 0.85
            if no_improve > 300:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        a = np.abs(np.random.default_rng(99).standard_normal(n)) ** 0.5
        return a / a.sum()
    return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: L-BFGS with adaptive smoothing (400s)
    P2: Nesterov peak shaving (350s)
    P3: Fine active-set polish (250s)
    P4: L-BFGS round 2 (200s)
    P5: Peak shaving round 2 (150s)
    P6: Coordinate descent (300s)
    P7: Final micro-optimization (100s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Default uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warm-up (400s) ===
    p1_deadline = t_start + 400
    print(f"\n[P1] L-BFGS (deadline={p1_deadline - t_start:.0f}s)")
    a1, sc1 = _run_lbfgs(best_seq, p1_deadline, min_remaining=800.0, n_phases=12, beta_decay=0.3)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  P1: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P1: no improvement (still {best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Nesterov peak shaving (350s) ===
    p2_deadline = t_start + 750
    print(f"\n[P2] Peak shaving Nesterov (deadline={p2_deadline - t_start:.0f}s)")
    a2, sc2 = _peak_shave_nesterov(best_seq, p2_deadline, min_remaining=500.0)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  P2: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P2: no improvement (still {best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Fine active-set (250s) ===
    p3_deadline = t_start + 1000
    print(f"\n[P3] Active-set fine (deadline={p3_deadline - t_start:.0f}s)")
    a3, sc3 = _active_set_fine(best_seq, p3_deadline, min_remaining=300.0)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  P3: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P3: no improvement (still {best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS round 2 (200s) ===
    p4_deadline = t_start + 1200
    print(f"\n[P4] L-BFGS round 2 (deadline={p4_deadline - t_start:.0f}s)")
    a4, sc4 = _run_lbfgs(best_seq, p4_deadline, min_remaining=200.0, n_phases=8, beta_decay=0.25)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  P4: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P4: no improvement (still {best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak shaving round 2 (150s) ===
    p5_deadline = t_start + 1350
    print(f"\n[P5] Peak shaving round 2 (deadline={p5_deadline - t_start:.0f}s)")
    a5, sc5 = _peak_shave_nesterov(best_seq, p5_deadline, min_remaining=150.0)
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  P5: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P5: no improvement (still {best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Coordinate descent (300s) ===
    p6_deadline = t_start + 1650
    print(f"\n[P6] Coordinate descent (deadline={p6_deadline - t_start:.0f}s)")
    a6, sc6 = _coord_descent(best_seq, p6_deadline, min_remaining=50.0)
    if sc6 < best_score:
        best_seq = a6.copy()
        best_score = sc6
        print(f"  P6: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P6: no improvement (still {best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final micro-optimization (100s) ===
    p7_deadline = deadline - 5
    print(f"\n[P7] Final micro-opt (deadline={p7_deadline - t_start:.0f}s)")

    # Quick active-set
    a7a, sc7a = _active_set_fine(best_seq, p7_deadline, min_remaining=20.0)
    if sc7a < best_score:
        best_seq = a7a.copy()
        best_score = sc7a
        print(f"  P7a: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Tiny L-BFGS with very small beta
    remaining = p7_deadline - 3 - time.monotonic()
    if remaining > 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.00003
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _SmoProb(beta_fin, n)
        max_iter = min(3000, max(20, int(remaining / 0.03)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 100})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  P7b: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final coordinate descent micro
    remaining = p7_deadline - 3 - time.monotonic()
    if remaining > 3:
        a_fin2, sc_fin2 = _coord_descent(best_seq, p7_deadline - 2, min_remaining=1.0)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  P7c: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
