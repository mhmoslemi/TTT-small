```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Start from parent, use multi-phase optimization:
1. Aggressive log-sum-exp L-BFGS with adaptive beta continuation
2. Peak balancing (equalize top peaks)
3. Active-set Newton with careful KKT solve
4. Coordinate descent polish
5. n-exploration with interpolation
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _convolve_a_with_weights(w, a, n):
    """Compute conv(w, a[::-1])[n-1:2n-1], which gives the gradient contributions."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n-1:2*n-1]

def _lsobj(z, beta, n, c=None, a=None):
    if a is None:
        a = _softmax(z)
    if c is None:
        c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n, c=None, a=None):
    if a is None:
        a = _softmax(z)
    if c is None:
        c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _convolve_a_with_weights(w, a, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)
```

Actually, let me think about this differently. The problem "minimize max(c[k_1], ..., c[k_m]) subject to sum(a)=1, a>=0" where k_1,...,k_m are the top m peaks can be formulated as:

minimize t
subject to: a^T Q_{k_i} a <= t for i=1..m
           sum(a) = 1
           a >= 0

where Q_{k_i} is the matrix representing the quadratic form c[k_i] = a^T Q_{k_i} a.

This is a QCQP (quadratically constrained quadratic program). It's non-convex in general.

But if we fix the active set (which of the a[j] are zero), the problem becomes:
minimize t
subject to: a^T Q_{k_i} a <= t for i in active constraints
           a_j > 0 for j in free set
           sum(a) = 1

This is still non-convex (quadratic constraints).

An alternative: use the KKT conditions directly. At the optimum:
- There exist multipliers λ_i >= 0 for each c[k_i] <= t
- ∇_a [t - sum(a) + sum_i λ_i (c[k_i] - t)] = 0 for a_j > 0
- This gives: -1 + sum_i λ_i * ∇_a c[k_i] = 0 for a_j > 0

So: sum_i λ_i * (∇c[k_i])_j = 1 for all j with a_j > 0.

This is a system of linear equations in λ (if the free set is known):
Let G be the matrix with G[j, i] = (∇c[k_i])_j
Then: G * λ = 1 (restricted to free coordinates)
With constraints: λ_i >= 0, and c[k_i] = t for active i, c[k_i] < t for inactive i.

This is essentially a linear programming problem in λ (once the active set is known), which can be solved efficiently.

But we also need to find a such that:
- sum(a) = 1
- a >= 0
- c[k_i] = t for active i
- a_j = 0 for j not in free set

The last two conditions are nonlinear (they involve the quadratic form c[k_i]).

So the overall problem is: find (a, t, λ) satisfying the KKT conditions. This is a nonlinear system that can be solved by Newton's method.

This is essentially what the active-set Newton in the previous code does. Let me improve it.

**Another idea: Use the fact that the sum of the convolution is fixed**

Since `sum(c) = (sum a)^2 = 1` (after normalization), if we reduce one peak, we must "push" the mass to other peaks. The question is where to push it.

The "best" place to push the mass is to the positions that are currently below the average. This is a water-filling argument.

**Practical approach: "Diffusion" of peaks**

Think of the convolution `c` as a "landscape" and we want to "erase" the peaks by "spreading" the mass to the valleys. The spreading should be done by adjusting `a`.

The direction that spreads mass from peak k to valley k' is:
d[k] direction: reduce a at positions that contribute to c[k]
d[k'] direction: increase a at positions that contribute to c[k']

This can be done by:
d[j] = -sum_{k in peaks} w_k * a[k-j] + sum_{k' in valleys} w_{k'} * a[k'-j]

where the weights are chosen to balance the reduction and increase.

f(a) = sum_{peaks} c[k]^2 - sum_{valleys} c[k]^2

The gradient of this w.r.t. a is:
∇f[j] = 2 * (sum_{peaks} c[k]*a[k-j] - sum_{valleys} c[k]*a[k-j])

This is a well-defined direction that should help flatten the convolution.

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Multi-phase strategy:
1. Log-sum-exp L-BFGS with aggressive beta continuation
2. Peak-contrast gradient (peaks vs valleys balancing)
3. Active-set Newton with top-k peak focus
4. Coordinate descent polish
5. n-exploration with interpolation
6. Final active-set polish with tight tolerance
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _convolve_weights_a(w, a, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _convolve_weights_a(w, a, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _peak_contrast_grad(a, n, n_peaks=10, n_valleys=10):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_min = float(c.min())

    top_idx = np.argsort(c)[::-1][:n_peaks]
    bot_idx = np.argsort(c)[:n_valleys]

    # Gradient of c[k] w.r.t. a[j] = 2*a[k-j] if 0 <= k-j < n
    # Weighted sum for peaks (negative) and valleys (positive)
    peak_weights = c[top_idx] - c_min
    peak_weights = peak_weights / (peak_weights.sum() + 1e-30)
    valley_weights = c_max - c[bot_idx]
    valley_weights = valley_weights / (valley_weights.sum() + 1e-30)

    d = np.zeros(n)
    for p in range(n_peaks):
        k = top_idx[p]
        w = peak_weights[p]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            d[j] -= 2.0 * w * a[k - j]
    for p in range(n_valleys):
        k = bot_idx[p]
        w = valley_weights[p]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            d[j] += 2.0 * w * a[k - j]

    return d, c


def _lbfgs_phase(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else 0
    base_beta = max(gap * 0.3, max_c * 0.005)

    betas = base_beta * np.concatenate([
        np.ones(3),
        0.5 * np.ones(3),
        0.25 * np.ones(3),
        0.125 * np.ones(3),
        0.0625 * np.ones(3),
        0.03 * np.ones(3),
        0.015 * np.ones(3),
        0.007 * np.ones(3),
        0.003 * np.ones(3),
        0.001 * np.ones(3),
        0.0004 * np.ones(3),
        0.00015 * np.ones(3),
        0.00005 * np.ones(2),
        0.00002 * np.ones(2),
        0.000008 * np.ones(2),
        0.000003 * np.ones(1),
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        if n <= 4096:
            est_tpi = 0.003
        elif n <= 8192:
            est_tpi = 0.007
        elif n <= 16384:
            est_tpi = 0.016
        else:
            est_tpi = 0.03

        max_iter = min(8000, max(50, int(remaining / (est_tpi * 8))))

        def obj_fn(z_local):
            return _lsobj(z_local, beta, n)

        def grad_fn(z_local):
            return _lsgrad(z_local, beta, n)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_contrast_phase(a_init, deadline, max_iter=2000, lr_init=0.02):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    no_improve = 0

    c = _full_convolve(a, a)
    c_max = float(c.max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        n_peaks = min(15, n // 4)
        n_valleys = min(15, n // 4)
        d, c = _peak_contrast_grad(a, n, n_peaks, n_valleys)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break

        d = d - np.mean(d)
        d = d / (np.linalg.norm(d) + 1e-30)

        step = lr / (1.0 + 0.0002 * it)
        t = step
        accepted = False
        f_old = c_max
        for ls in range(25):
            a_try = a + t * d
            if np.min(a_try) < -1e-12:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                c_max = f_try
                accepted = True
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    no_improve = 0
                break
            t *= 0.5

        if not accepted:
            no_improve += 1
            lr *= 0.9
            if no_improve > 50:
                a = best_a.copy()
                c = _full_convolve(a, a)
                c_max = float(c.max())
                no_improve = 0
                lr = lr_init * 0.5
            if no_improve > 200:
                break
        else:
            if no_improve > 0:
                no_improve = max(0, no_improve - 1)

    return best_a


def _active_set_newton(a_init, deadline, max_iter=200, m_peaks=7):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        top_idx = np.argsort(c)[::-1][:m_peaks]

        # Build gradient matrix: G[p, j] = dc[top_idx[p]]/da[j] = 2*a[top_idx[p]-j]
        G = np.zeros((m_peaks, n))
        for p in range(m_peaks):
            k = top_idx[p]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            indices = np.arange(lo, hi)
            G[p, indices] = 2.0 * a[k - indices]

        # Weights: excess above median of top peaks
        peak_vals = c[top_idx]
        med = np.median(peak_vals)
        w_peaks = np.maximum(peak_vals - med, 0.0)
        w_sum = w_peaks.sum()
        if w_sum < 1e-15:
            w_peaks = np.ones(m_peaks) / m_peaks
        else:
            w_peaks = w_peaks / w_sum

        # Direction: weighted sum of gradients
        d = G.T @ w_peaks
        d = d - np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break

        d = d / d_norm

        # Line search (backtracking)
        f_old = c_max
        accepted = False
        t = 0.15
        for ls in range(35):
            a_try = a + t * d
            if np.min(a_try) < -1e-12:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t:
                a = a_try
                accepted = True
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            t *= 0.5

        if not accepted:
            # Try steepest descent on the smoothed objective
            beta = (c_max - float(c[top_idx[-1]])) * 0.5 + c_max * 0.005
            z = np.log(np.maximum(a, 1e-30))
            z = z - z.mean()
            grad = _lsgrad(z, beta, n)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            t = 0.05
            for ls2 in range(20):
                a_try = a - t * grad / gn
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < f_old - 1e-14:
                    a = a_try
                    sc = _score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    break
                t *= 0.5
            else:
                break

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=30):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        top_idx = np.argsort(c)[::-1][:10]
        c = _full_convolve(a, a)
        c_max = float(c.max())

        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[j]
            # Try small perturbations
            for delta in [-0.001, -0.0005, 0.0005, 0.001]:
                a_try = a.copy()
                a_try[j] = max(0, orig + delta)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    c = c_try
                    c_max = float(c.max())
                    improved = True
                    break
            a[j] = orig  # restore before next j

        if improved:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        else:
            break

    return best_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
```
