    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Build sparse convolution matrix
    # C[k, j] = a[j] if max(0, k-n+1) <= j <= min(k, n-1)

"""
Optimizer combining L-BFGS log-sum-exp, LP-based alternating response,
active-set Newton, and multi-scale optimization to minimize
2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
import scipy.sparse
from scipy.optimize import minimize
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr_a_A(a, A_vec, n):
    """Compute (a_rev * A_vec) cropped to [n-1, 2n-2] which is the gradient-related term"""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)

def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()

    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)

def _build_conv_matrix(a, n):
    n_rows = 2 * n - 1
    indptr = np.zeros(n_rows + 1, dtype=np.int64)
    indices = np.zeros(n * n, dtype=np.int64)
    data = np.zeros(n * n, dtype=np.float64)
    ptr = 0
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        count = j_hi - j_lo + 1
        indptr[k + 1] = indptr[k] + count
        indices[ptr:ptr + count] = np.arange(j_lo, j_hi + 1)
        data[ptr:ptr + count] = a[j_lo:j_hi + 1]
        ptr += count
    return scipy.sparse.csr_matrix((data, indices, indptr), shape=(n_rows, n))


def _lp_best_response(a, M, solver='ECOS'):
    n = len(a)
    C = _build_conv_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [C @ b_var <= M, cp.sum(b_var) <= 1.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            return None
    b_val = b_var.value
    if b_val is None or not np.all(np.isfinite(b_val)):
        return None
    return np.maximum(b_val, 0)

def _spectral_init(n, mode='flat', rng=None):
    """Initialize based on spectral design principles."""
    if rng is None:
        rng = np.random.default_rng(0)

    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'cosine':
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == 'raised_cosine':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == 'two_peak':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.2) / 0.15) ** 2) + np.exp(-0.5 * ((t - 0.8) / 0.15) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == 'random_smooth':
        k = max(2, n // 50)
        coeffs = rng.standard_normal(k)
        freqs = np.fft.rfftfreq(n)
        A = np.zeros(n // 2 + 1, dtype=complex)
        for i in range(min(k, n // 2 + 1)):
            A[i] = coeffs[i % len(coeffs)] / (1 + 0.1 * i)
        a = np.fft.irfft(A * n, n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
