"""
Optimizer combining spectral IAA initialization, adaptive log-sum-exp L-BFGS,
and active-set Newton refinement for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _corr_a_A(a, A_vec, n):
    """Compute (a * reverse(A_vec))[n-1:2n-1] using FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0, verbose=False):

    """

        gap = c_sorted[0] - c_sorted[min(5, len(c_sorted)-1)]

    base_beta = max(gap, max_c * 0.02)
    
    # Create adaptive schedule
    betas = []
    beta = base_beta

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    if s < 1e-300 or not np.isfinite(s):

def _iaa_spectral(n, n_iters=150, alpha=0.4):

    elif len(c_sorted) > 5:
        gap = c_sorted[0] - c_sorted[-1]
    else:
        gap = max_c * 0.01
    base_beta = max(gap * 0.5, max_c * 0.02)

    betas = []
    beta = base_beta
    min_beta = max_c * 0.00003
    while beta > min_beta:
        betas.append(beta)
        beta *= 0.55
    betas.append(min_beta)
    betas.append(min_beta * 0.3)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

        if time.monotonic() + min_time_remaining > deadline:

        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 10))))

        if verbose and i % 3 == 0:
            print(f"    L-BFGS beta={beta:.2e}, max_iter={max_iter}, t={time.monotonic()-deadline+deadline:.0f}s")

        try:
            res = minimize(
                lambda z_local: _lsobj(z_local, beta, n),

            dx = dx - np.mean(dx)

            neg = dx < -1e-12
            if neg.any():
                t_max = float(np.min(a[neg] / (-dx[neg]))) * 0.99
            else:
                t_max = 1.0
            t = min(1.0, max(t_max, 1e-10))

            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)

            if sc_try < best_dir_score:
                best_dir_score = sc_try
                best_dir = dx
                best_t = t

        if best_dir is None:
            break

        f_old = c_max
        accepted = False
        t = best_t
        for ls in range(60):
            a_try = a + t * best_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_conv(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old - 1e-13 * max(1.0, t) or ls >= 50:
                if f_try <= f_old + 1e-11:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _fft_conv(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _deflation(a, deadline, n_attempts=20):
    """Targeted deflation of top convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a, a)
        c_max = float(c.max())

        top_k = np.argsort(c)[::-1]
        n_top = max(3, min(len(c), int(0.03 * len(c))))
        top_positions = top_k[:n_top]

        a_new = a.copy()
        for k_pos in top_positions[:min(n_top, 20)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a[i] * a[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a[max_i] >= a[j]:
                    a_new[max_i] *= 0.996
                else:
                    a_new[j] *= 0.996

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            a_small = a.copy()
            for k_pos in top_positions[:8]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0.0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_small[i] * a_small[k_pos - i]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    if a_small[max_i] >= a_small[j]:
                        a_small[max_i] *= 0.9995
                    else:
                        a_small[j] *= 0.9995
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-12:
                    best_sc = sc_small
                    best_a = a_small.copy()
                else:
                    break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t) + 0.1 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block = n // 4
        a[:block] = 1.0
        a[3 * block:] = 1.0
        s = a.sum()
        a = np.maximum(a, 0)
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = n // 6
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        s = a.sum()
        a = np.maximum(a, 0)
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _evaluate_candidate(seq, label, best_tracker):
    """Evaluate and update best if improved."""
    p_arr = np.asarray(seq, dtype=np.float64).ravel()
    p_arr = np.clip(p_arr, 0.0, 1000.0)
    s = float(p_arr.sum())
    if s < 0.01:
        return None
    sc = _score(p_arr)
    if np.isfinite(sc) and sc < best_tracker[0]:
        best_tracker[0] = float(sc)
        best_tracker[1] = p_arr.copy()
        print(f"  NEW BEST: {best_tracker[0]:.10f} (n={len(p_arr)}) [{label}]")
    return sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: spectral IAA init + adaptive L-BFGS +
    active-set Newton + spectral projection + multi-n + restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_tracker = [np.inf, None]

    def update(p, label):
        _evaluate_candidate(p, label, best_tracker)

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update(parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update(parent, "init_uniform")

    # === Phase 1: Spectral IAA initializations ===
    print(f"\n[Phase 1] Spectral IAA (t={time.monotonic()-t_start:.1f}s)")
    n_main = n_parent if n_parent > 0 else 16384

    iaa_inits = {}
    if time.monotonic() < deadline - 600:
        iaa1 = _iaa_spectral(n_main, n_iters=100, alpha=0.4)
        iaa_inits['iaa1'] = iaa1
        update(iaa1, "iaa1")

        iaa2 = _iaa_v2(n_main, n_iters=150, alpha_start=0.5, alpha_end=0.1)
        iaa_inits['iaa2'] = iaa2
        update(iaa2, "iaa2")

    # === Phase 2: Main L-BFGS from best initialization ===
    print(f"\n[Phase 2] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    if best_tracker[1] is not None:
        a_main = best_tracker[1].copy()
        a_main = a_main / a_main.sum()
        a_main, sc_main = _run_lbfgs_adaptive(a_main, deadline, min_time_remaining=400.0, verbose=True)
        update(a_main, "lbfgs_main")
    else:
        a_main = parent

    # === Phase 3: Active-set Newton ===
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_tracker[1] is not None and time.monotonic() < deadline - 300:
        as_dl = min(deadline - 300, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_tracker[1].copy(), as_dl, max_iter=300,
                                         tol_list=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        update(a_as, "as_newton_1")

        if time.monotonic() < deadline - 250:
            as_dl2 = min(deadline - 250, time.monotonic() + 60)
            a_as2, sc_as2 = _active_set_newton(best_tracker[1].copy(), as_dl2, max_iter=200,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            update(a_as2, "as_newton_2")

    # === Phase 4: Spectral projection post-processing ===
    print(f"\n[Phase 4] Spectral projection (t={time.monotonic()-t_start:.1f}s)")
    if best_tracker[1] is not None and time.monotonic() < deadline - 200:
        a_spec = _spectral_project(best_tracker[1].copy(), n_iters=80, alpha=0.3)
        update(a_spec, "spectral_proj")
        if best_tracker[0] < 1.51:
            a_spec2 = _spectral_project(best_tracker[1].copy(), n_iters=40, alpha=0.15)
            update(a_spec2, "spectral_proj_fine")

    # === Phase 5: Deflation ===
    print(f"\n[Phase 5] Deflation (t={time.monotonic()-t_start:.1f}s)")
    if best_tracker[1] is not None and time.monotonic() < deadline - 150:
        a_def, sc_def = _deflation(best_tracker[1].copy(),
                                   min(deadline - 150, time.monotonic() + 45), n_attempts=25)
        update(a_def, "deflation")
        if sc_def < best_tracker[0]:
            a_def2, _ = _run_lbfgs_adaptive(a_def, deadline, min_time_remaining=100.0)
            update(a_def2, "lbfgs_post_def")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(len(n_candidates) // 2, n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent and best_tracker[1] is not None:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        # Also try IAA init for this n
        if time.monotonic() < deadline - 120:
            a_iaa = _iaa_v2(n_try, n_iters=80, alpha_start=0.4, alpha_end=0.1)
            sc_iaa = _score(a_iaa)
            sc_init = _score(a_init)
            if sc_iaa < sc_init:
                a_init = a_iaa

        a_opt, sc_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=40.0, verbose=False)
        update(a_opt, f"n_{n_try}")

        if sc_opt < best_tracker[0] + 0.01 and time.monotonic() < deadline - 30:
            a_as3, sc_as3 = _active_set_newton(a_opt,
                                                min(deadline - 20, time.monotonic() + 30),
                                                max_iter=100,
                                                tol_list=[0.002, 0.001, 0.0005])
            update(a_as3, f"n_{n_try}_as")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_tracker[1] is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("perturb_002", 0.02),
            ("perturb_005", 0.05),
            ("perturb_010", 0.10),
            ("perturb_020", 0.20),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_tracker[1].copy(), sigma, rng)
            a_r, _ = _run_lbfgs_adaptive(p_start, deadline, min_time_remaining=10.0, verbose=False)
            update(a_r, f"restart_{label}")

    # === Phase 8: Final polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_tracker[1] is not None and time.monotonic() < deadline - 3:
        # Ultra-tight active set Newton
        as_final_dl = min(deadline - 1, time.monotonic() + 45)
        a_fin, sc_fin = _active_set_newton(best_tracker[1].copy(), as_final_dl,
                                           max_iter=300,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update(a_fin, "final_as_tight")

        # Spectral projection on final
        if time.monotonic() < deadline - 2:
            a_fin_sp = _spectral_project(best_tracker[1].copy(), n_iters=30, alpha=0.1)
            update(a_fin_sp, "final_spectral")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 4:
            n = len(best_tracker[1])
            c_best = _fft_conv(best_tracker[1], best_tracker[1])
            mb = float(c_best.max())
            z = np.log(np.maximum(best_tracker[1], 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(100, int(remaining / 0.03)))
                try:
                    res = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                    )
                    a_fin3 = _softmax(res.x)
                    update(a_fin3, "final_lbfgs_tiny")
                except Exception:
                    pass

        # One more active set after tiny beta
        if time.monotonic() < deadline - 2:
            a_fin4, _ = _active_set_newton(best_tracker[1].copy(),
                                           min(deadline - 1, time.monotonic() + 15),
                                           max_iter=100,
                                           tol_list=[0.0001, 0.00005, 0.00002])
            update(a_fin4, "final_as_last")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_tracker[1] is not None:
        final = np.clip(best_tracker[1], 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
