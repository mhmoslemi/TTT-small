- sum(a) = 1
- (a*a)[k_j] = t for active j
- λ_j >= 0, ν_i >= 0, a_i >= 0

This is a system of nonlinear equations. We can solve it using Newton's method or fixed-point iteration.

The key equation for positive entries: `2 sum_j λ_j (T_{k_j} a)_i = -μ` (a constant for all i with a_i > 0).

This means: `sum_j λ_j (T_{k_j} a) = constant` (on the support of a).

If the support is all of {0,...,n-1}, then: `A λ = c * 1` where A is the matrix with A[i,j] = (T_{k_j} a)_i and c = -μ/2.

This gives us: `A λ` is a constant vector. Combined with the complementarity conditions a_i >= 0 and the constraints (a*a)[k_j] = t, we have a system we can iteratively solve.

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) L-BFGS with adaptive log-sum-exp smoothing, 
(2) KKT-based active-set iteration, 
(3) targeted peak shaving, 
(4) multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    betas = []
    b = base_beta
    while b > max_c * 1e-5:
        betas.append(b)
        b *= 0.4
    betas.append(max_c * 1e-5)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(10, int(remaining / (est_time_per_iter * 15))))
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 80, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a

def _active_set_kkt(a_init, deadline, min_time_remaining=60.0):
    """KKT-based active set iteration.
    At optimum: for support S of a, sum_j lambda_j * (T_{k_j} a)_i = const for i in S.
    We iterate: (1) find active peaks, (2) solve for lambda, (3) update a.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    M = max(10, min(200, int(n * 0.01)))  # number of active peaks
    lr = 0.1
    no_improve = 0
    max_stall = 200
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_stall:
                break
        
        # Find active peaks (top M)
        top_M = np.argsort(c)[::-1][:M]
        
        # Compute gradients for active peaks
        # grad_j[i] = 2 * (T_{k_j} a)_i = 2 * sum over valid (i, k_j) pairs
        # (T_k a)_i = a[i+k] if i+k < n else ... 
        # Actually (T_k a)_i = a_{i+k} for the autocorrelation gradient
        # (a*a)_k = sum_i a_i a_{i+k}
        # d/d a_j (a*a)_k = a_{j+k} + a_{j-k} (with appropriate bounds)
        
        # Build gradient matrix G of shape (n, M)
        # G[i, j] = d(a*a)[k_j]/d a_i = a[(i + k_j) % n] + a[|i - k_j|] ... 
        # No, for linear convolution:
        # (a*a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
        # d/d a_j (a*a)[k] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n and j != k-j... wait)
        # Actually: (a*a)[k] = sum_i a_i a_{k-i}
        # d/d a_j = a_{k-j} + a_{j-k} if both are in range, but that double counts when j = k/2
        # More carefully: d/d a_j sum_i a_i a_{k-i} = a_{k-j} (from the a_i term when i=j) + a_{j-k} (from the a_{k-i} term when k-i=j, i.e., i=k-j)
        # If j != k-j (i.e., j != k/2), both terms contribute: a_{k-j} + a_{k-j} = 2*a_{k-j}
        # Wait no: when i=j, the term is a_j * a_{k-j}, derivative w.r.t. a_j gives a_{k-j}.
        # When k-i=j, i.e., i=k-j, the term is a_{k-j} * a_j, derivative w.r.t. a_j gives a_{k-j}.
        # So total: if j != k-j (j != k/2): 2 * a_{k-j}
        # If j == k/2: a_{k-j} (only one term)
        # But we also need bounds: 0 <= k-j < n and 0 <= j < n.
        
        # For efficiency, let's compute this using the convolution approach:
        # G[:, j] = 2 * convolve(delta_at_k_j, a) restricted to [0, n-1]
        # Actually, the gradient of (a*a)[k] w.r.t. a is:
        # g[i] = 2 * a[k-i] for the valid range, with a factor of 1 at i = k/2 if k is even
        # 
        # Simpler: g = 2 * a[::-1] convolved with delta at position k, then take [n-1 : 2n-1]
        # Or: g[i] = 2 * a_rev[k-i] where a_rev = a[::-1]
        # 
        # Let me just use the full convolution approach:
        # grad of (a*a)[k] w.r.t. a = 2 * (a shifted) ... 
        
        # Actually the cleanest way: the gradient of (a*a)[k] w.r.t. a is
        # (2 * circ_shift(a, k)) restricted to valid indices
        # where circ_shift(a, k)[i] = a[(i+k) % n] for circular, but for linear:
        # grad[i] = a[i+k] (if i+k < n) + a[i-k] (if i >= k)
        # = (a shifted right by k) + (a shifted left by k)
        
        # For the active set, build the matrix G (n x M)
        G = np.zeros((n, M))
        for j_idx, k in enumerate(top_M):
            # grad[i] = a[i+k] if i+k < n, plus a[i-k] if i >= k
            # Shift right by k: a[k:] goes to positions 0 to n-1-k
            # Shift left by k: a[:n-k] goes to positions k to n-1
            r = min(n - k, n)  # when k < n
            l = n - k
            if k < n:
                G[:n-k, j_idx] = a[k:]  # a[i+k] for i = 0 to n-1-k
                G[k:, j_idx] += a[:n-k]  # a[i-k] for i = k to n-1
            else:
                # k >= n, this shouldn't happen for valid indices
                pass
        
        # KKT condition: G @ lambda = mu * 1 for i in support of a
        # Simplified: find lambda >= 0 such that G @ lambda is as constant as possible on support
        # Use least squares: minimize ||G @ lambda - mean(G @ lambda on support)||^2
        # Or simpler: minimize ||G @ lambda||^2 subject to lambda >= 0 (this drives all gradients to zero)
        
        # Actually, let's use a simpler iterative approach:
        # The direction that reduces all active peaks is:
        # d = -G @ lambda where lambda = (G^T G + eps I)^{-1} 1
        # This minimizes ||G d + 1||^2 (reduces all peaks by at least 1 in gradient sense)
        
        # For efficiency with large n, use conjugate gradient or just a few power iterations
        # Let's use the simple approach: d = -G @ (G^T G + eps*I)^{-1} @ 1
        
        # Since G is n x M with M <= 200, G^T G is M x M (small!)
        GTG = G.T @ G  # M x M
        GTG += 1e-8 * np.eye(M)
        ones_M = np.ones(M)
        try:
            lam = np.linalg.solve(GTG, ones_M)
        except:
            lam = np.linalg.lstsq(GTG, ones_M, rcond=None)[0]
        lam = np.maximum(lam, 0)  # lambda >= 0
        
        d = -G @ lam
        # Enforce sum(d) = 0
        d = d - d.mean()
        
        # Line search
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            no_improve += 5
            continue
        
        d = d / d_norm  # normalize direction
        alpha = lr / (np.max(np.abs(d)) + 1e-15)
        
        improved = False
        for _ in range(40):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.7
            no_improve += 1
            if no_improve >= max_stall:
                break
        else:
            lr = min(0.5, lr * 1.05)
    
    return best_a, best_sc

def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Targeted single-peak reduction with careful gradient computation."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = 0.02
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Find the single largest peak
        k_star = int(np.argmax(c))
        
        # Gradient of (a*a)[k_star] w.r.t. a
        # g[i] = a[i + k_star] (if i + k_star < n) + a[i - k_star] (if i >= k_star)
        g = np.zeros(n)
        if k_star < n:
            g[:n - k_star] = a[k_star:]
            g[k_star:] += a[:n - k_star]
        # Note: if k_star == 0, g = 2*a (they overlap completely)
        # Actually for k=0: (a*a)[0] = sum a_i^2, grad = 2*a
        if k_star == 0:
            g = 2.0 * a
        else:
            # The terms above are correct for k > 0
            # g[i] = a[i+k] for i < n-k, and g[i] += a[i-k] for i >= k
            # When i < k: only a[i+k] contributes (a[i-k] is out of bounds)
            # When k <= i < n-k: both a[i+k] and a[i-k] contribute
            # When i >= n-k: only a[i-k] contributes
            # This is already handled by the vectorized operations above
            pass
        
        # Project gradient onto the tangent space of the simplex
        g = g - g.mean()
        
        # Step in the negative gradient direction
        d = -g
        d_norm = np.max(np.abs(d))
        if d_norm < 1e-15:
            break
        
        alpha = lr / d_norm
        improved = False
        for _ in range(50):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.8
            if lr < 1e-8:
                break
            no_improve += 1
        else:
            lr = min(0.1, lr * 1.03)
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "pswf_approx":
        # Approximate PSWF: use DCT-like window
        t = np.arange(n)
        a = np.cos(np.pi * t / (n + 1)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _spectral_init(n, rng):
    """Generate a non-negative sequence via spectral method."""
    # Random phase, fixed amplitude
    N = _next_pow2(n)
    # Create a spectrum with some structure
    freqs = np.fft.rfftfreq(N)
    # Low-pass filter
    cutoff = 0.4
    amplitude = np.exp(-5.0 * (freqs / cutoff) ** 2)
    # Add some random modulation
    amplitude *= 1.0 + 0.3 * rng.standard_normal(len(amplitude))
    amplitude = np.maximum(amplitude, 0)
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, len(amplitude))
    spectrum = amplitude * np.exp(1j * phase)
    # Inverse FFT
    a_full = np.fft.irfft(spectrum, N)[:n]
    a = np.maximum(a_full, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def _multiplicative_weights(a_init, deadline, min_time_remaining=30.0):
    """Mirror descent / multiplicative weights update.
    a_i <- a_i * exp(-eta * grad_i), then normalize.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    eta = 0.1
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        # Soft peak selection (log-sum-exp weights)
        beta = c_max * 0.01
        shifted = (c - c_max) / beta
        w = np.exp(np.clip(shifted, -700, 0))
        w = w / w.sum()
        
        # Compute "smooth gradient": weighted average of peak gradients
        # grad_smooth[i] = sum_k w[k] * 2 * (T_k a)[i]
        # Use FFT: this is equivalent to convolving w (reversed) with a
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * a * (h - W)  # This is the gradient of the smooth objective
        
        # Multiplicative update
        z = np.log(np.maximum(a, 1e-30))
        z -= eta * grad
        z -= z.mean()
        a_new = _softmax(z)
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        if sc_new < current_sc - 1e-14:
            a = a_new
            eta = min(0.5, eta * 1.02)
        else:
            eta *= 0.7
            if eta < 1e-8:
                break
            no_improve += 1
    
    return best_a, best_sc

def _try_shrinkage(a, deadline):
    """Try zeroing out small elements to potentially improve the score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    a_sorted = np.argsort(a)
    for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
1. More aggressive L-BFGS with log-sum-exp (smaller beta range, more iterations)
2. "Deep subgradient" method with explicit active-set tracking

4. Multiplicative weights (mirror descent) for natural non-negativity
5. Multi-n exploration with different initializations
6. Targeted peak shaving with adaptive step sizes

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    # More aggressive beta schedule: go much smaller
    betas = []
    b = base_beta
    min_beta = max_c * 1e-6

        betas.append(b)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

def _deep_subgradient(a_init, deadline, min_time_remaining=30.0):

        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # grad = sum_{k in active} w_k * 2*(T_k a) where w_k = 1/n_active
        w = np.zeros_like(c)

def _kkt_active_set(a_init, deadline, min_time_remaining=60.0):
    """KKT-based active set method. Solves for the direction that simultaneously
    reduces all active peaks using the Gram matrix of peak gradients."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 0.1
    no_improve = 0
    max_stall = 200
    M_base = max(20, min(500, int(n * 0.005)))

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_stall:
                break

        # Adaptive active set size
        M = M_base + int(no_improve * 0.5)
        M = min(M, max(20, int(n * 0.02)))

        top_M = np.argsort(c)[::-1][:M]

        # Build gradient matrix G (n x M)
        # G[i, j] = d(a*a)[k_j]/d a_i
        G = np.zeros((n, M))
        for j_idx in range(M):
            k = top_M[j_idx]
            if k < n:
                r = n - k
                G[:r, j_idx] = a[k:]
                G[k:, j_idx] += a[:r]
            elif k >= n:
                # For k >= n (shouldn't happen for valid autocorrelation indices)
                # (a*a)[k] involves a[n-1-(k-n+1)] to a[n-1] dotted with a[0] to a[k-n]
                # Actually for linear conv, k ranges from 0 to 2n-2
                # (a*a)[k] = sum_{i=k-n+1}^{n-1} a[i] * a[k-i]
                # d/d a_j = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and j != k-j)
                # For j in [k-n+1, n-1]: a[k-j] is in [1, n-1] valid
                # For j in [0, k-1] with k-j in [0,n-1]: a[k-j] valid
                # This is getting complex, let's use a simpler approach for k >= n
                # By symmetry, (a*a)[k] = (a*a)[2n-2-k]
                k_sym = 2 * n - 2 - k
                if k_sym < n:
                    r = n - k_sym
                    G[:r, j_idx] = a[k_sym:]
                    G[k_sym:, j_idx] += a[:r]

        # Compute Gram matrix: GTG = G^T G (M x M)
        # But this is expensive for large n. Use the fact that G = [T_{k_1} a, ..., T_{k_M} a]
        # GTG[j, l] = G[:,j] . G[:,l] = (T_{k_j} a) . (T_{k_l} a)
        # This can be computed via convolution: it's (a*a) at lag |k_j - k_l|
        # Actually: (T_k a) . (T_l a) = sum_i a[i+k] a[i+l] = (a*a)[|k-l|] (approximately)
        # More precisely: sum_i a[i+k] a[i+l] where both indices are valid
        # This equals (a*a)[l-k] if l>=k, or (a*a)[k-l] if k>=l
        GTG = np.zeros((M, M))
        for j_idx in range(M):
            for l_idx in range(M):
                diff = abs(top_M[j_idx] - top_M[l_idx])
                if diff < len(c):
                    GTG[j_idx, l_idx] = c[diff]
                else:
                    GTG[j_idx, l_idx] = 0.0

        # Add regularization
        GTG += 1e-10 * np.eye(M)

        # Solve: find lambda >= 0 minimizing (G lambda - mu*1)^T (G lambda - mu*1)
        # Simplified: find lambda that makes G lambda as constant as possible
        # Use: lambda = (GTG + eps I)^{-1} * (G^T 1) ... hmm not quite
        # Actually we want: minimize ||G d||^2 s.t. ... 
        # Let's use the simpler approach: d = -G @ (GTG + eps I)^{-1} @ ones
        # This minimizes ||G d + ones||^2, i.e., reduces all peaks by approximately 1 in gradient
        ones_M = np.ones(M)
        try:
            lam = np.linalg.solve(GTG, ones_M)
        except np.linalg.LinAlgError:
            lam = np.linalg.lstsq(GTG, ones_M, rcond=None)[0]
        lam = np.maximum(lam, 0)

        d = -G @ lam
        d = d - d.mean()  # enforce sum(d) = 0

        d_max = np.max(np.abs(d))
        if d_max < 1e-15:
            no_improve += 5
            continue

        alpha = lr / d_max
        improved = False
        for _ in range(40):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.7
        else:
            lr = min(0.5, lr * 1.04)

    return best_a, best_sc

        # Adaptive beta: tighten over time
        beta = c_max * max(0.001, beta_frac * 0.99)
        beta_frac = max(0.0005, beta_frac * 0.995)

        # Compute log-sum-exp weights
        shifted = (c - c_max) / beta
        w = np.exp(np.clip(shifted, -700, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Smooth gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * a * (h - W)

        # Multiplicative update in log-space
        z = np.log(np.maximum(a, 1e-30))
        z -= eta * grad
        z -= z.mean()
        a_new = _softmax(z)

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        if sc_new < current_sc - 1e-14:
            a = a_new
            eta = min(0.3, eta * 1.02)
        else:
            eta *= 0.8
            if eta < 1e-9:
                break

    return best_a, best_sc


def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Aggressive single-peak shaving with Newton-like step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 0.03
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break

        # Find top few peaks and use their average gradient
        n_top = max(3, min(10, int(n * 0.001)))
        top_idx = np.argsort(c)[::-1][:n_top]
        # Check if they're all similar
        if c[top_idx[0]] - c[top_idx[-1]] > c_max * 0.05:
            n_top = 1  # Just use the single peak

        # Average gradient over top peaks
        g = np.zeros(n)
        for k in top_idx[:n_top]:
            if k < n:
                r = n - k
                g[:r] += a[k:]
                g[k:] += a[:r]
            else:
                k_sym = 2 * n - 2 - k
                if k_sym < n:
                    r = n - k_sym
                    g[:r] += a[k_sym:]
                    g[k_sym:] += a[:r]
        g /= n_top

        # Project onto simplex tangent
        g = g - g.mean()
        gnorm = np.max(np.abs(g))
        if gnorm < 1e-15:
            break

        alpha = lr / gnorm
        improved = False
        for _ in range(50):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.85
            if lr < 1e-10:
                break
        else:
            lr = min(0.1, lr * 1.02)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "pswf":
        # Prolate spheroidal wave function approximation
        t = np.arange(n)
        W = 0.25  # bandwidth parameter
        # Use the fact that PSWFs are eigenvectors of the prolate matrix
        # Approximate with a modulated sinc
        a = np.sinc(2 * W * t / n * np.pi - W * np.pi)
        a = a ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _try_shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted = np.argsort(a)
    for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc


def _spectral_fft_optimize(a_init, deadline, min_time_remaining=30.0):
    """Optimize in log-frequency domain. Exploit the relation:
    autocorrelation peak relates to time-domain concentration of |FFT(a)|^2.
    Use iterative Fourier-domain peak reduction (similar to PAPR reduction)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    N = _next_pow2(2 * n - 1)
    threshold_frac = 1.05  # threshold as multiple of mean autocorrelation
    no_improve = 0

    for iteration in range(2000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        # Threshold on autocorrelation
        threshold = c_max * threshold_frac - 0.0
        # Actually, we want to reduce peaks above threshold
        # In frequency domain: if c_k > threshold, we want to reduce it
        # c = ifft(|A|^2) approximately (for circular conv)
        # Reduce: set |A|^2[f] *= exp(-alpha * real(ifft_selective))
        # This is the "selective mapping" approach from PAPR literature

        # Compute FFT of a (padded)
        a_pad = np.zeros(N, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        power = np.abs(A) ** 2

        # Inverse FFT of power gives circular autocorrelation
        circ_c = np.fft.ifft(power).real
        # Take the relevant part
        circ_c = np.concatenate([circ_c[N // 2:], circ_c[1:N // 2 + 1]])[:2 * n - 1]

        # Find peaks in circular autocorrelation that are too high
        peak_mask = circ_c > threshold
        if not peak_mask.any():
            threshold_frac = max(1.01, threshold_frac * 0.98)
            continue

        # Create a "correction" in frequency domain
        # Reduce power at frequencies that contribute to the peaks
        # correction_time = ifft(peak_mask * power) ... this is circular
        # Simpler: multiply power by exp(-beta * circ_c) where circ_c > threshold
        beta = 0.1
        correction = np.exp(-beta * np.maximum(circ_c - threshold, 0))
        # Convert back to frequency domain correction
        # Actually, let's work differently:
        # The peak in time domain corresponds to constructive interference in frequency
        # We want to reduce the "peaky" part of the time signal
        # Selective mapping: a_new = a - ifft(selective_mask * A)
        # where selective_mask zeros out the "peaky" frequencies

        # Simpler approach: just do gradient step in time domain
        # (fall back to time-domain if this doesn't converge)
        g = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        # Map circular indices to linear
        for k in peak_indices[:50]:  # Limit to top 50 peaks
            if k < n:
                r = n - k
                g[:r] += a[k:]
                g[k:] += a[:r]
            else:
                k_sym = 2 * n - 2 - k
                if k_sym < n:
                    r = n - k_sym
                    g[:r] += a[k_sym:]
                    g[k_sym:] += a[:r]

        g = g / max(1, len(peak_indices[:50]))
        g = g - g.mean()
        gnorm = np.max(np.abs(g))
        if gnorm < 1e-15:
            threshold_frac = max(1.01, threshold_frac * 0.98)
            continue

        alpha = 0.05 / gnorm
        for _ in range(30):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                threshold_frac = max(1.01, threshold_frac * 0.995)
                break
            alpha *= 0.5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS log-sum-exp, deep subgradient, KKT active-set,
    multiplicative weights, peak shaving, spectral method, multi-n, shrinkage.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS with log-sum-exp (aggressive) ===
    print(f"\n[P1] L-BFGS log-sum-exp (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 900, t_start + 400)
    a1 = _run_lbfgs(parent, p1_deadline, min_time_remaining=600.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc1:.10f} vs {best_score:.10f}")

    # === Phase 2: Deep subgradient ===
    print(f"\n[P2] Deep subgradient (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 700, t_start + 550)
    a2, sc2 = _deep_subgradient(best_seq, p2_deadline, min_time_remaining=400.0)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [deep_subgrad] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: KKT active-set ===
    print(f"\n[P3] KKT active-set (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 500, t_start + 700)
    a3, sc3 = _kkt_active_set(best_seq, p3_deadline, min_time_remaining=300.0)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [kkt] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multiplicative weights ===
    print(f"\n[P4] Multiplicative weights (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 400, t_start + 850)
    a4, sc4 = _multiplicative_weights(best_seq, p4_deadline, min_time_remaining=200.0)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [mw] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak shaving ===
    print(f"\n[P5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 300, t_start + 950)
    a5, sc5 = _peak_shave(best_seq, p5_deadline, min_time_remaining=150.0)
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peak_shave] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Second L-BFGS round (on improved sequence) ===
    print(f"\n[P6] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p6_deadline = min(deadline - 250, time.monotonic() + 150)
        a6 = _run_lbfgs(best_seq, p6_deadline, min_time_remaining=100.0)
        sc6 = _score(a6)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 20480, 32768, 24576]
    n_candidates = [x for x in n_candidates if x != n_parent]

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 80:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < deadline - 30:
            a_ps, sc_ps = _peak_shave(best_seq if sc_opt >= best_score else a_opt,
                                      min(deadline - 25, time.monotonic() + 25), min_time_remaining=10.0)
            if sc_ps < best_score:
                best_seq = a_ps.copy()
                best_score = sc_ps
                print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [multi_n_ps] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Shrinkage ===
    print(f"\n[P8] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        a_shr, sc_shr = _try_shrinkage(best_seq, deadline - 30)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 20:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=10.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _deep_subgradient(best_seq, deadline - 2, min_time_remaining=1.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_subgrad] t={time.monotonic()-t_start:.1f}s")

        a_fin2, sc_fin2 = _peak_shave(best_seq, deadline - 1, min_time_remaining=0.5)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  NEW BEST: {best_score:.10f} [final_shave] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
