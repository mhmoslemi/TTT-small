"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Aggressive log-sum-exp L-BFGS with deep beta continuation
2. Improved semi-smooth Newton with adaptive active set
3. Peak-shaving via coordinate descent on top convolution peaks
4. Multi-n exploration (8192, 16384, 32768)
5. Spectral-aware initialization and perturbation restarts
6. Final ultra-tight polish with micro-Newton steps

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _psobj(z, p, n):
    """Power-sum softening: (sum c[k]^p)^(1/p)"""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    cp = c ** p
    s = cp.sum()
    if s < 1e-300:
        return 2.0 * n * c.max()
    return 2.0 * n * s ** (1.0 / p)


def _psgrad(z, p, n):
    """Gradient of power-sum objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    cp = c ** (p - 1)
    s = (c ** p).sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = cp / (s ** (1.0 / p))
        w = w / w.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=500, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _hessian_vp(v, A_vec, m, mu, n):
    """Hessian-vector product for the active-set subproblem."""
    return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

def _peak_shave(a_init, deadline, n_rounds=100):

        a_new = a.copy()

            if best_i >= 0 and best_prod > 0:

def _nesterov_grad(a_init, deadline, n_steps=50000, top_k_frac=0.01):
    """Nesterov-accelerated gradient with top-K peak selection."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    best_cmax = float(_full_convolve(a, a).max())

    # Estimate Lipschitz constant
    K_est = 2.0 * np.linalg.norm(a, 2)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Select top-K peaks
        K = max(3, int(len(c) * top_k_frac))
        top_idx = np.argpartition(c, -K)[-K:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / K

        # Gradient
        grad = 2.0 * _corr_a_A(y, w, n)
        g_t = grad - np.mean(grad)

        # Step size (inverse Lipschitz)
        alpha = 1.0 / (K_est * (1.0 + 0.01 * step))

        a_new = y - alpha * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / t_new
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t_mom = t_new

        if step % 500 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()
                best_cmax = float(_full_convolve(a, a).max())

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, best_sc


def _run_lbfgs_v2(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with more aggressive beta continuation and both softening types."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Compute initial beta based on peak structure
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    else:
        gap = c_sorted[0] * 0.1
    base_beta = max(gap, max_c * 0.02)

    # More aggressive beta schedule
    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
         0.015625, 0.0078125, 0.00390625, 0.001953125,
         0.000976563, 0.000488281, 0.000244141, 0.00012207,
         0.000061035, 0.000030518, 0.000015259, 0.00000763]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale iterations based on remaining time
        est_time_per_iter = 0.005 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Also try power-sum softening at intermediate stages
        if i in [3, 6, 9, 12, 15] and remaining > 30:
            p = 10 + i * 5
            try:
                result2 = minimize(
                    lambda z_local: _psobj(z_local, p, n),
                    z,
                    jac=lambda z_local: _psgrad(z_local, p, n),
                    method='L-BFGS-B',
                    options={
                        'maxiter': min(2000, max_iter // 3),
                        'ftol': 1e-17,
                        'gtol': 1e-13,
                        'maxcor': 60,
                        'maxls': 100,
                    }
                )
                a_ps = _softmax(result2.x)
                sc_ps = _score(a_ps)
                if sc_ps < current_sc - 1e-15:
                    current_a = a_ps
                    current_sc = sc_ps
                    z = np.log(np.maximum(a_ps, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 4
        a[:block] = 1.0
        a[3 * block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "block3":
        a = np.zeros(n)
        block = n // 6
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "bump_edge":
        t = np.linspace(0, 1, n)
        a = np.exp(-50 * t ** 2) + np.exp(-50 * (t - 1) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_structural(a, rng, n):
    """Structural perturbation: add a random smooth component."""
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.linspace(0, 1, n)
    smooth = 0.1 * np.sin(2 * np.pi * freq * t + phase)
    a_new = a * (1.0 + smooth)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _multi_start_optimize(a_init, deadline, min_remaining, label_prefix, best_score_tracker):
    """Run a full optimization pipeline on a single initialization."""
    n = len(a_init)
    results = []

    # Step 1: L-BFGS
    a1 = _run_lbfgs_v2(a_init, deadline, min_time_remaining=min_remaining)
    sc1 = _score(a1)
    results.append((a1, sc1, label_prefix + "_lbfgs"))

    # Step 2: Active-set Newton
    if time.monotonic() < deadline - min_remaining * 0.5:
        as_dl = min(deadline - min_remaining * 0.4, time.monotonic() + 120)
        a2, sc2 = _active_set_newton_v2(a1, as_dl, max_iter=400,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        results.append((a2, sc2, label_prefix + "_newton"))
        a1 = a2
        sc1 = sc2

    # Step 3: Peak shaving
    if time.monotonic() < deadline - min_remaining * 0.3:
        ps_dl = min(deadline - min_remaining * 0.25, time.monotonic() + 60)
        a3, sc3 = _peak_shave(a1, ps_dl, n_rounds=200)
        results.append((a3, sc3, label_prefix + "_shave"))
        if sc3 < sc1:
            a1 = a3
            sc1 = sc3

    # Step 4: Nesterov gradient
    if time.monotonic() < deadline - min_remaining * 0.2:
        ng_dl = min(deadline - min_remaining * 0.15, time.monotonic() + 40)
        a4, sc4 = _nesterov_grad(a1, ng_dl, n_steps=30000, top_k_frac=0.005)
        results.append((a4, sc4, label_prefix + "_nesterov"))
        if sc4 < sc1:
            a1 = a4
            sc1 = sc4

    # Step 5: Final L-BFGS polish
    if time.monotonic() < deadline - min_remaining * 0.1:
        pol_dl = min(deadline - min_remaining * 0.05, time.monotonic() + 30)
        a5 = _run_lbfgs_v2(a1, pol_dl, min_time_remaining=5.0)
        sc5 = _score(a5)
        results.append((a5, sc5, label_prefix + "_polish"))

    # Return best result
    best_a = results[0][0]
    best_sc = results[0][1]
    for a_r, sc_r, lab in results:
        if sc_r < best_sc:
            best_sc = sc_r
            best_a = a_r
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer with aggressive L-BFGS, semi-smooth Newton,
    peak shaving, Nesterov gradient, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print("  NEW BEST: {:.10f} (n={}) [{}] improve={:.2e} t={:.1f}s".format(
                best_score, len(p_arr), label, prev - best_score, time.monotonic() - t_start))

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print("Parent loaded: n={}, score={:.10f}".format(n_parent, sc_parent))
        _update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print("No parent, using uniform n={}".format(n_parent))
        _update_best(parent, _score(parent), "init")

    # === Phase 1: Main optimization on parent n ===
    print("\n[Phase 1] Main optimization n={} (t={:.1f}s)".format(n_parent, time.monotonic() - t_start))
    if time.monotonic() < deadline - 400:
        a_main, sc_main = _multi_start_optimize(
            parent, deadline, min_remaining=400,
            label_prefix="main", best_score_tracker=None)
        _update_best(a_main, sc_main, "main_pipeline")

    # === Phase 2: Active-set Newton with very tight tolerances ===
    print("\n[Phase 2] Tight Newton (t={:.1f}s)".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 300:
        tw = min(deadline - 300, time.monotonic() + 150)
        a_tw, sc_tw = _active_set_newton_v2(
            best_seq, tw, max_iter=500,
            tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            cg_max=600)
        _update_best(a_tw, sc_tw, "tight_newton")

    # === Phase 3: Peak shaving on current best ===
    print("\n[Phase 3] Peak shaving (t={:.1f}s)".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 200:
        ps_dl = min(deadline - 200, time.monotonic() + 80)
        a_ps, sc_ps = _peak_shave(best_seq, ps_dl, n_rounds=300)
        _update_best(a_ps, sc_ps, "peak_shave")

        if sc_ps < best_score and time.monotonic() < deadline - 150:
            a_ps2 = _run_lbfgs_v2(a_ps, min(deadline - 100, time.monotonic() + 60), min_time_remaining=50.0)
            _update_best(a_ps2, _score(a_ps2), "post_shave_lbfgs")

    # === Phase 4: Nesterov gradient ===
    print("\n[Phase 4] Nesterov gradient (t={:.1f}s)".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 150:
        ng_dl = min(deadline - 150, time.monotonic() + 80)
        a_ng, sc_ng = _nesterov_grad(best_seq, ng_dl, n_steps=50000, top_k_frac=0.003)
        _update_best(a_ng, sc_ng, "nesterov")

    # === Phase 5: Multi-n exploration ===
    print("\n[Phase 5] Multi-n (t={:.1f}s)".format(time.monotonic() - t_start))
    n_candidates = [8192, 16384, 32768]
    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent:
            continue

        # Interpolate parent
        a_interp = _interpolate_to_n(parent, n_try)
        sc_interp = _score(a_interp)

        # Also try alternative profiles
        a_alt = _make_profile(n_try, "tapered")
        sc_alt = _score(a_alt)
        a_start = a_interp if sc_interp < sc_alt else a_alt
        sc_start = min(sc_interp, sc_alt)

        print("  n={}: init score={:.10f} (t={:.1f}s)".format(n_try, sc_start, time.monotonic() - t_start))
        _update_best(a_start, sc_start, "n{}_init".format(n_try))

        if time.monotonic() < deadline - 40:
            a_opt, sc_opt = _multi_start_optimize(
                a_start, deadline, min_remaining=40,
                label_prefix="n{}".format(n_try), best_score_tracker=None)
            _update_best(a_opt, sc_opt, "n{}_opt".format(n_try))

    # === Phase 6: Perturbation restarts ===
    print("\n[Phase 6] Restarts (t={:.1f}s)".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 30:
        pert_sigs = [0.01, 0.02, 0.05, 0.1, 0.2]
        for sig in pert_sigs:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sig, rng)
            a_r = _run_lbfgs_v2(p_start, min(deadline - 10, time.monotonic() + 25), min_time_remaining=5.0)
            sc_r = _score(a_r)
            _update_best(a_r, sc_r, "pert_{:.2f}".format(sig))

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton_v2(
                    a_r, min(deadline - 5, time.monotonic() + 20),
                    max_iter=100, tol_list=[0.001, 0.0005, 0.0002])
                _update_best(a_r2, sc_r2, "pert_{:.2f}_newton".format(sig))

    # === Phase 7: Structural perturbations ===
    print("\n[Phase 7] Structural perturbations (t={:.1f}s)".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_bp = len(best_seq)
        for sp in range(3):
            if time.monotonic() > deadline - 10:
                break
            p_struct = _perturb_structural(best_seq, rng, n_bp)
            a_sr = _run_lbfgs_v2(p_struct, min(deadline - 5, time.monotonic() + 15), min_time_remaining=3.0)
            _update_best(a_sr, _score(a_sr), "struct_{}".format(sp))

    # === Phase 8: Final ultra-tight polish ===
    print("\n[Phase 8] Final polish (t={:.1f}s)".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very tight Newton
        fn_dl = min(deadline - 2, time.monotonic() + 30)
        a_fn, sc_fn = _active_set_newton_v2(
            best_seq, fn_dl, max_iter=300,
            tol_list=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
            cg_max=800)
        _update_best(a_fn, sc_fn, "final_newton")

        # Final peak shave
        if time.monotonic() < deadline - 2:
            a_fs, sc_fs = _peak_shave(best_seq, min(deadline - 1, time.monotonic() + 15), n_rounds=500)
            _update_best(a_fs, sc_fs, "final_shave")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100, 'maxls': 300}
                    )
                    a_fin = _softmax(result.x)
                    _update_best(a_fin, _score(a_fin), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print("\n" + "=" * 60)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print("FINAL: score={:.10f}, n={}, time={:.1f}s".format(sc_final, len(final), time.monotonic() - t_start))
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print("FALLBACK: score={:.10f}".format(_score(fallback)))
    return [float(x) for x in fallback]
