"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam optimizer in log-domain with log-sum-exp continuation
2. Very fine continuation schedule (many small beta values)
3. Active-set Newton refinement

5. Structured restarts with Fourier-based perturbations

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the correlation: for each i, sum_k A_vec[k] * a[i+k] (with appropriate bounds)"""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_ls(z_init, beta, n, max_iter, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for log-sum-exp objective."""
    z = z_init.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_z = z.copy()
    best_val = _lsobj(z, beta, n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        grad = _lsgrad(z, beta, n)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 50 == 0:
            val = _lsobj(z, beta, n)
            if val < best_val:
                best_val = val
                best_z = z.copy()
    
    return best_z


def _cg_solve(matvec, rhs, max_cg=200, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _lbfgs_ls(z_init, beta, n, deadline, max_iter=3000):
    """L-BFGS for log-sum-exp objective."""
    remaining = deadline - time.monotonic()
    if remaining < 5:
        return z_init
    
    est_time = 0.003 * max(1, n / 2048.0)
    max_iter = min(max_iter, max(50, int(remaining / (est_time * 5))))
    
    try:
        def obj(z_local):
            return _lsobj(z_local, beta, n)
        
        def grad(z_local):
            return _lsgrad(z_local, beta, n)
        
        result = minimize(
            obj,
            z_init,
            jac=grad,
            method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-17,
                'gtol': 1e-13,
                'maxcor': 50,
                'maxls': 80,
            }
        )
        return result.x
    except Exception:
        return z_init


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = max(1, n // 4)
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = max(1, n // 6)
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_fourier(a, rng, n_modes=10):
    """Add low-frequency Fourier perturbations."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    # Add random low-frequency components
    freqs = np.arange(n)
    for _ in range(n_modes):
        f = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.1)
        z = z + amp * np.cos(2 * np.pi * f * freqs / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _continuation_optimize(a_init, deadline, n_betas=20, min_time_remaining=120.0):
    """Multi-scale continuation with Adam + L-BFGS."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)] if len(c_sorted) > 50 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    # Geometric continuation
    betas = base_beta * (0.7 ** np.arange(n_betas))
    betas = np.append(betas, [max_c * 1e-5, max_c * 5e-6, max_c * 1e-6])
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 30:
            break
        
        # Adam phase
        adam_iter = min(5000, max(100, int(remaining * 0.4 / (0.001 * max(1, n/2048.0)))))
        z = _adam_ls(z, beta, n, adam_iter, time.monotonic() + remaining * 0.4)
        
        # L-BFGS phase
        remaining2 = deadline - min_time_remaining - time.monotonic()
        if remaining2 > 20:
            z = _lbfgs_ls(z, beta, n, time.monotonic() + remaining2 * 0.6, max_iter=3000)
        
        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < current_sc:
            current_a = a_new
            current_sc = sc_new
    
    return current_a

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam + L-BFGS continuation in log-domain with log-sum-exp smoothing
2. Very fine geometric continuation schedule (30+ beta values)
3. Active-set Newton refinement with adaptive tolerances
4. Multi-n exploration (8192, 16384, 32768)
5. Fourier-based structured restarts
6. Gradient-clipped Adam for stability at small beta
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LSObjFunc:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lsobj(z, self.beta, self.n)


class _LSGradFunc:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lsgrad(z, self.beta, self.n)


class _HessMatvec:
    def __init__(self, A_vec, m, mu, n):
        self.A_vec = A_vec
        self.m = m
        self.mu = mu
        self.n = n

    def __call__(self, v):
        return (2.0 / self.m) * _corr_a_A(v, self.A_vec, self.n) + self.mu * v


def _adam_ls(z_init, beta, n, max_iter, deadline, lr=0.005, clip_norm=100.0):
    z = z_init.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_z = z.copy()
    best_val = _lsobj(z, beta, n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        grad = _lsgrad(z, beta, n)
        gnorm = np.linalg.norm(grad)
        if gnorm > clip_norm:
            grad = grad * (clip_norm / gnorm)

        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100 == 0:
            val = _lsobj(z, beta, n)
            if val < best_val:
                best_val = val
                best_z = z.copy()

    return best_z


def _lbfgs_ls(z_init, beta, n, deadline, max_iter=3000):
    remaining = deadline - time.monotonic()
    if remaining < 5:
        return z_init

    est_time = 0.004 * max(1, n / 2048.0)
    max_iter = min(max_iter, max(50, int(remaining / (est_time * 4))))

    try:
        obj_fn = _LSObjFunc(beta, n)
        grad_fn = _LSGradFunc(beta, n)
        result = minimize(
            obj_fn,
            z_init.copy(),
            jac=grad_fn,
            method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-18,
                'gtol': 1e-14,
                'maxcor': 50,
                'maxls': 60,
            }
        )
        return result.x
    except Exception:
        return z_init


def _cg_solve(matvec, rhs, max_cg=200, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = float(np.dot(r, r))
    rhs_norm = float(np.linalg.norm(rhs))
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = float(np.dot(p, Ap))
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = max(1, n // 4)
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = max(1, n // 6)
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_fourier(a, rng, n_modes=10, amp_range=(0.01, 0.08)):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    freqs = np.arange(n)
    for _ in range(n_modes):
        f = int(rng.integers(1, max(2, n // 8)))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(amp_range[0], amp_range[1]))
        z = z + amp * np.cos(2.0 * np.pi * f * freqs / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _continuation_optimize(a_init, deadline, n_betas=25, min_time_remaining=120.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)] if len(c_sorted) > 100 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.05)

    decay = 0.72
    betas = base_beta * (decay ** np.arange(n_betas))
    final_betas = [max_c * 3e-5, max_c * 1e-5, max_c * 3e-6, max_c * 1e-6, max_c * 3e-7]
    betas = np.concatenate([betas, np.array(final_betas)])

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    current_a = a_init.copy()
    current_sc = _score(current_a)

    total_betas = len(betas)
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        frac = i / max(1, total_betas - 1)

        if frac < 0.5:
            adam_iter = min(4000, max(200, int(remaining * 0.5 / max(0.001, 0.0008 * n / 2048.0))))
            adam_lr = 0.008 * (1.0 - frac) + 0.002
            z = _adam_ls(z, beta, n, adam_iter, time.monotonic() + remaining * 0.5, lr=adam_lr)
        else:
            a_new_check = _softmax(z)
            sc_check = _score(a_new_check)
            if sc_check < current_sc:
                current_a = a_new_check
                current_sc = sc_check

        remaining2 = deadline - min_time_remaining - time.monotonic()
        if remaining2 > 15:
            lbfgs_iter = min(5000, max(100, int(remaining2 * 0.6 / max(0.002, 0.002 * n / 2048.0))))

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-15:

            current_sc = sc_new

                z = np.log(np.maximum(a_polish, 1e-30))
                z = z - z.mean()

    # Phase 4: Active-set Newton (round 2, tighter)
    print(f"\n[Phase 4] AS Newton round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        as_dl2 = min(deadline - 150, time.monotonic() + 100)
        a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=300,
                                           tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    remaining_multin = deadline - 100 - time.monotonic()
    if remaining_multin > 60:
        time_per_n = max(50, remaining_multin / max(1, len(n_candidates)))
        for n_try in n_candidates:
            if time.monotonic() > deadline - 60:
                break
            if n_try == n_parent:
                continue
            if n_try > 32768:
                continue

            if parent is not None:
                a_init = _interpolate_to_n(parent, n_try)
                profiles = ["block", "three_block", "tapered", "cos"]
                best_init = a_init
                best_init_sc = _score(a_init)
                for prof in profiles:
                    a_p = _make_profile(n_try, prof)
                    sc_p = _score(a_p)
                    if sc_p < best_init_sc:
                        best_init = a_p
                        best_init_sc = sc_p
                a_init = best_init
            else:
                a_init = _make_profile(n_try, "uniform")

            a_opt = _short_continuation(a_init, min(deadline - 50, time.monotonic() + time_per_n),
                                        n_betas=15, min_time_remaining=20.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")

            if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
                a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 30, time.monotonic() + 30),
                                                      max_iter=80, tol_list=[0.002, 0.001, 0.0005])
                update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 6: Structured restarts
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for ri in range(3):
            if time.monotonic() > deadline - 20:
                break
            if ri == 0:
                p_start = _perturb_fourier(best_seq, rng, n_modes=8, amp_range=(0.005, 0.04))
            elif ri == 1:
                z_p = np.log(np.maximum(best_seq, 1e-30))
                z_p = z_p + 0.03 * rng.standard_normal(len(z_p))
                z_p = z_p - z_p.mean()
                p_start = _softmax(z_p)
            else:
                p_start = _perturb_fourier(best_seq, rng, n_modes=15, amp_range=(0.01, 0.06))

            a_r = _short_continuation(p_start, min(deadline - 20, time.monotonic() + 60),
                                      n_betas=10, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{ri}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 15),
                                                  max_iter=50, tol_list=[0.001, 0.0005])
                update_best(a_r2, sc_r2, f"restart_{ri}_as")

    # Phase 7: Final ultra-tight polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Final AS Newton
        as_final = min(deadline - 3, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=200,
                                           tol_list=[0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_as")

        # Final tiny-beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            remaining = deadline - 6 - time.monotonic()
            if remaining > 5:
                for bf in [mb * 5e-7, mb * 2e-7]:
                    if time.monotonic() > deadline - 5:
                        break
                    rem = deadline - 6 - time.monotonic()
                    if rem < 3:
                        break
                    z = _lbfgs_ls(z, bf, n, time.monotonic() + rem * 0.8, max_iter=2000)
                    a_fin3 = _softmax(z)
                    sc_fin3 = _score(a_fin3)
                    if sc_fin3 < best_score:
                        update_best(a_fin3, sc_fin3, "final_ultra")
                        best_seq = a_fin3.copy()
                        z = np.log(np.maximum(a_fin3, 1e-30))
                        z = z - z.mean()

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}")
    return [float(x) for x in fallback]
