import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def _lse_obj_and_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _run_adaptive_lbfgs(a_init, deadline, min_remaining=30.0):
    """L-BFGS with adaptive temperature and restart-on-stagnation."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(current_a, current_a)
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c_sorted[0] * 0.001:
        gap = c_sorted[0] * 0.001
    base_beta = max(gap, c_sorted[0] * 0.005)

    # Adaptive beta schedule with restarts
    beta_schedule = base_beta * np.array(
        [1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
         0.055, 0.04, 0.028, 0.02, 0.014, 0.01, 0.007,
         0.005, 0.0035, 0.0025, 0.0015, 0.001, 0.0007,
         0.0005, 0.00035, 0.00025, 0.00015, 0.0001,
         0.00007, 0.00005, 0.00003, 0.00002, 0.00001,
         0.000007, 0.000005, 0.000003, 0.000002, 0.000001]
    )

    scale = max(1.0, n / 4096.0)
    no_improve_count = 0
    max_no_improve = 3

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3.0:
            break

        est_iter_time = 0.003 * scale
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 20))))

        def func(zv):
            obj, _ = _lse_obj_and_grad(zv, beta, n)
            return obj

        def grad(zv):
            _, g = _lse_obj_and_grad(zv, beta, n)
            return g

        try:
            result = minimize(
                func, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 60,
                    'maxls': 40,
                    'iprint': -1,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)

                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

                no_improve_count = 0
            else:
                no_improve_count += 1

                    z = z + 0.01 * rng.standard_normal(n)
                    z = z - z.mean()
                    no_improve_count = 0
        except Exception:
            continue

    return current_a


def _active_set_nesterov(a_init, deadline, top_k_start=5, top_k_end=30,
                         max_stall=50):
    """Active-set subgradient with Nesterov acceleration and BB step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    y = np.zeros(n)  # Nesterov momentum position
    v = np.zeros(n)  # Velocity
    gamma = 2.0 / 3.0  # Nesterov parameter
    alpha_prev = 0.01
    g_prev = None
    prev_a = a.copy()
    no_improve = 0
    top_k = top_k_start

    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())
    tol = 0.01

    while time.monotonic() < deadline - 2.0:
        # Nesterov extrapolation
        y = a + gamma * (a - prev_a)
        y = _project_simplex(y)

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Select active set
        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 20:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Compute gradient
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * y * (h - W)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Barzilai-Borwein step size
        if g_prev is not None:
            s_bb = a - prev_a
            y_bb = grad - g_prev
            s_norm = float(np.dot(s_bb, s_bb))
            y_norm = float(np.dot(y_bb, y_bb))
            s_dot_y = float(np.dot(s_bb, y_bb))
            if s_norm > 1e-20 and y_norm > 1e-20:
                alpha_bb1 = s_norm / abs(s_dot_y) if abs(s_dot_y) > 1e-20 else alpha_prev
                alpha_bb2 = abs(s_dot_y) / y_norm
                # Alternate between alpha1 and alpha2

                    alpha = np.clip(alpha_bb1, 1e-8, 0.5)
                else:
                    alpha = np.clip(alpha_bb2, 1e-8, 0.5)
                alpha = min(alpha, alpha_prev * 2.0)
                alpha = max(alpha, alpha_prev * 0.1)
            else:
                alpha = alpha_prev
        else:
            alpha = min(0.01, 1.0 / (gnorm + 1e-15))

        # Backtracking line search
        base_a = a
        base_sc = _score(base_a)
        improved = False
        alpha_try = alpha

        for _ in range(50):
            a_try = base_a - alpha_try * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha_try *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < base_sc - 1e-15:
                prev_a = base_a
                a = a_try
                alpha_prev = alpha_try
                improved = True
                break
            alpha_try *= 0.5

        if improved:
            sc_cur = _score(a)
            if sc_cur < best_sc - 1e-14:
                best_sc = sc_cur
                best_a = a.copy()
                no_improve = 0
                # Adapt top_k
                if top_k < top_k_end:
                    top_k = min(top_k_end, top_k + 1)
                # Tighten tolerance periodically
                if no_improve == 0 and (int((best_sc * 1e10)) % 7 == 0):
                    tol = max(0.0005, tol * 0.95)
            g_prev = grad.copy()
        else:
            no_improve += 1
            alpha_prev *= 0.5
            if no_improve % 10 == 0:
                tol = max(0.0005, tol * 0.7)
            if no_improve >= max_stall:
                a = best_a.copy()
                prev_a = best_a.copy()
                tol = max(0.0002, tol * 0.5)
                top_k = max(top_k_start, top_k // 2)
                no_improve = 0
                # Add small perturbation to escape
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                if _score(a) > best_sc * 1.01:
                    a = best_a.copy()
                    prev_a = best_a.copy()

    return best_a, _score(best_a)


def _coordinate_refinement(a_init, deadline, n_steps=5000):
    """Coordinate-wise refinement: find elements that most reduce the peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(42)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    k_star = int(np.argmax(c))

    step = 0
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline - 1.0 and step < n_steps:
        step += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find elements contributing to the peak
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * j_range.clip(0, n - 1)
        # Actually j_range should be k_star - i_range, and we need 0 <= j < n
        valid = (j_range >= 0) & (j_range < n)
        if not valid.any():
            continue
        i_valid = i_range[valid]
        j_valid = j_range[valid]
        contrib_valid = a[i_valid] * a[j_valid]

        if contrib_valid.sum() < 1e-20:
            continue

        # Try moving mass from top contributor to a random other element
        top_idx = int(i_valid[np.argmax(contrib_valid)])
        # Choose a target element (one that doesn't contribute to the peak)
        peak_contrib_set = set(int(x) for x in i_valid)
        other_indices = np.array([i for i in range(n) if i not in peak_contrib_set])
        if len(other_indices) == 0:
            other_indices = np.arange(n)

        # Weighted random selection from non-peak elements
        weights = a[other_indices] + 1e-10
        weights = weights / weights.sum()
        target_idx = int(other_indices[rng.choice(len(other_indices), p=weights)])

        # Try different transfer amounts
        best_local_sc = best_sc
        best_local_frac = 0
        for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:

            if transfer < 1e-15:
                continue
            a_try = a.copy()
            a_try[top_idx] -= transfer
            a_try[target_idx] += transfer
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_local_sc:

                best_local_frac = frac
                best_a_try = a_try.copy()

            a = best_a_try
            best_a = a.copy()
            best_sc = best_local_sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, np.append(a, a[0]))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + np.abs(t - 0.5)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    a2, sc2 = _active_set_nesterov(best_a, p2_deadline, top_k_start=5, top_k_end=40, max_stall=60)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [nesterov_p2] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc2:.10f} vs {best_score:.10f}")

    # === Phase 3: Coordinate refinement (allocate ~10% of time) ===
    p3_budget = 0.10 * (budget_s - 15.0)
    p3_deadline = t_start + p1_budget + p2_budget + p3_budget
    print(f"\n[P3] Coordinate refinement (budget={p3_budget:.0f}s)")
    a3, sc3 = _coordinate_refinement(best_a, p3_deadline, n_steps=10000)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [coord_p3] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc3:.10f} vs {best_score:.10f}")

    # Re-run L-BFGS briefly after coordinate refinement if it helped
    if sc3 < best_score + 0.001 and time.monotonic() < deadline - 200:
        p3b_deadline = min(p3_deadline + 60.0, deadline - 150.0)
        a3b = _run_adaptive_lbfgs(best_a, p3b_deadline, min_remaining=30.0)
        sc3b = _score(a3b)
        if sc3b < best_score:
            best_a = a3b.copy()
            best_score = sc3b
            n_best = len(best_a)
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [lbfgs_post_coord] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-n exploration (allocate ~15% of time) ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    remaining_for_n = max(0, deadline - time.monotonic() - 120.0)
    if remaining_for_n > 60:
        # Try nearby n values
        n_candidates = []
        for delta in [0, 512, -512, 1024, -1024, 2048, -2048]:
            n_try = n_best + delta
            if 1024 <= n_try <= 30000 and n_try not in n_candidates:
                n_candidates.append(n_try)

        n_candidates.sort(key=lambda x: abs(x - n_best))
        time_per_n = remaining_for_n / max(1, len(n_candidates))

        for n_try in n_candidates[:5]:
            if time.monotonic() > deadline - 100.0:
                break
            if n_try == n_best:
                continue

            a_init = _resample(best_a, n_try)
            sc_init = _score(a_init)
            print(f"  n={n_try}: init score={sc_init:.10f}")

            n_deadline = min(time.monotonic() + time_per_n * 1.5, deadline - 80.0)
            if n_deadline - time.monotonic() < 20:
                continue

            # Quick L-BFGS
            a_opt = _run_adaptive_lbfgs(a_init, n_deadline, min_remaining=10.0)
            sc_opt = _score(a_opt)

            # Quick active-set
            if time.monotonic() < n_deadline - 5:
                a_opt2, sc_opt2 = _active_set_nesterov(a_opt, n_deadline, top_k_start=3, top_k_end=20, max_stall=30)
                if sc_opt2 < sc_opt:
                    a_opt, sc_opt = a_opt2, sc_opt2

            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                n_best = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  n={n_try}: best={sc_opt:.10f} (no improvement)")

    # === Phase 5: Diverse restarts (allocate ~5% of time) ===
    print(f"\n[P5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50.0:
        n_best = len(best_a)
        restarts = []
        # Small perturbation of best
        z_best = np.log(np.maximum(best_a, 1e-30))
        z_best = z_best - z_best.mean()
        z_pert = z_best + 0.03 * rng.standard_normal(n_best)
        z_pert = z_pert - z_pert.mean()
        restarts.append(("perturbed", _softmax(z_pert)))
        # Different profiles
        restarts.append(("tapered", _make_profile(n_best, "tapered")))
        restarts.append(("wedge", _make_profile(n_best, "wedge")))
        restarts.append(("cos_mod", _make_profile(n_best, "cosine_mod")))

        time_per_restart = max(10.0, (deadline - 30.0 - time.monotonic()) / len(restarts))
        for label, a_r in restarts:
            if time.monotonic() > deadline - 25.0:
                break
            r_deadline = min(time.monotonic() + time_per_restart, deadline - 20.0)
            if r_deadline - time.monotonic() < 15:
                break
            a_opt = _run_adaptive_lbfgs(a_r, r_deadline, min_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        # Active-set with very tight tolerance
        a_fin, sc_fin = _active_set_nesterov(best_a, deadline - 3.0, top_k_start=2, top_k_end=10, max_stall=30)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_nesterov] t={time.monotonic()-t_start:.1f}s")

        # Coordinate refinement
        if time.monotonic() < deadline - 2.0:
            a_cr, sc_cr = _coordinate_refinement(best_a, deadline - 1.0, n_steps=5000)
            if sc_cr < best_score:
                best_a = a_cr.copy()
                best_score = sc_cr
                print(f"  NEW BEST: {best_score:.10f} [final_coord] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
