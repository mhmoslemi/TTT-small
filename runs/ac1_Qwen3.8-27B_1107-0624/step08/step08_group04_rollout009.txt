a_i <- a_i * exp(-eta * 2*(W*a)_i) / Z

This automatically maintains a >= 0 and sum(a) = 1 (after normalization).

Let me implement this carefully.

For the step size eta in the exponential weights:
- The QP is min a^T W a s.t. sum(a)=1, a>=0
- The gradient is 2*W*a
- For mirror descent with the negative entropy mirror, the update is:
  a_i <- a_i * exp(-eta * 2*(W*a)_i) / Z
- The optimal eta depends on the "radius" of the feasible set and the "diameter" of the sublevel set
- A safe choice is eta = 1 / (2 * ||W||) where ||W|| is the operator norm
- We can estimate ||W|| using power iteration or just use a conservative value

For adaptive eta:
- Start with a moderate eta
- If the objective increases, decrease eta
- If the objective decreases steadily, increase eta slightly

Let me now write the complete code:

```python
"""
Multi-strategy optimizer using soft-active-set QP with exponential weights.
Key improvements: (1) smooth active set via softmax weights, (2) mirror descent
QP solver with adaptive step size, (3) Gerchberg-Saxton spectral init,
(4) multi-n exploration, (5) CG-Newton polish, (6) diverse restarts.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    """Compute H*v where H_{j,m} = W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]

def _exp_weights_qp(W, n, a_init, max_iter=10000, eta_init=None, tol=1e-15):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using mirror descent (exponential weights).
    W is a vector of length 2n-1 (Hankel entries).
    Returns the optimal a (sum=1, a>=0).
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    
    # Initialize z (log-space)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Estimate initial step size
    if eta_init is None:
        Wa = _hankel_matvec(a, W, n)
        eta = 1.0 / (float(np.max(Wa)) + 1e-15)
        eta = min(eta, 2.0)
    else:
        eta = eta_init
    
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n))
    
    for i in range(max_iter):
        a_cur = _softmax(z)
        Wa = _hankel_matvec(a_cur, W, n)
        
        # Mirror descent step
        z = z - eta * Wa
        z = z - z.mean()
        z = np.clip(z, -200, 200)
        
        # Check every 50 iterations
        if (i + 1) % 50 == 0:
            a_check = _softmax(z)
            val = float(a_check @ _hankel_matvec(a_check, W, n))
            if val < best_val - 1e-16:
                best_val = val
                best_a = a_check.copy()
            
            # Adaptive step size
            if val > best_val + 1e-12:
                eta *= 0.7
            elif i > 100 and (i + 1) % 500 == 0:
                eta = min(eta * 1.05, 5.0)
            
            # Convergence check
            if best_val < tol:
                break
    
    a_final = _softmax(z)
    val_final = float(a_final @ _hankel_matvec(a_final, W, n))
    if val_final < best_val:
        best_a = a_final
    return best_a

def _soft_active_optimize(a_init, deadline, max_rounds=200, beta_start=0.05, beta_end=0.001):
    """
    Soft active-set optimization:
    - Compute convolution weights using softmax
    - Solve QP with exponential weights
    - Decrease beta (sharpen the weights)
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Beta schedule (logarithmic)
    log_beta_start = np.log(beta_start)
    log_beta_end = np.log(beta_end)
    
    eta = 1.0  # Initial step size for QP
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if rnd < max_rounds * 0.3:
            frac = rnd / (max_rounds * 0.3)
            log_beta = log_beta_start + frac * (log_beta_end - log_beta_start)
        else:
            frac = (rnd - max_rounds * 0.3) / (max_rounds * 0.7)
            log_beta = log_beta_end - 0.5 * frac * log_beta_end  # Continue decreasing
        beta = max(np.exp(log_beta), 1e-10)
        
        # Compute softmax weights
        w = _softmax(c / beta)
        
        # Solve QP
        a_qp = _exp_weights_qp(w, n, a, max_iter=2000, eta_init=eta)
        sc_qp = _score(a_qp)
        
        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Damped update
            t = max(0.1, 1.0 - 0.5 * rnd / max_rounds)
            a = (1 - t) * a + t * a_qp
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        
        # Occasionally update eta
        if rnd % 20 == 0:
            Wa = _hankel_matvec(a, w, n)
            eta = min(eta * 1.1, 3.0 / (float(np.max(Wa)) + 1e-15))
    
    return best_a, best_sc

def _gerchberg_saxton(n, max_iter=500, target_magnitude=None):
    """
    GS algorithm to find a non-negative sequence with approximately flat spectrum.
    """
    if target_magnitude is None:
        target_magnitude = np.ones(n)  # Flat magnitude
    
    # Initialize with random phases
    rng = np.random.default_rng(42)
    phases = rng.uniform(0, 2 * np.pi, n)
    # Make DC component real and positive
    phases[0] = 0
    
    spectrum = target_magnitude * np.exp(1j * phases)
    # Ensure conjugate symmetry for real signal
    if n % 2 == 0:
        # Even length: no Nyquist issue
        for k in range(1, n // 2 + 1):
            spectrum[n - k] = np.conj(spectrum[k])
    else:
        # Odd length
        for k in range(1, (n - 1) // 2 + 1):
            spectrum[n - k] = np.conj(spectrum[k])
    
    a = np.real(np.fft.ifft(spectrum))
    a = np.maximum(a, 0)
    a = a / (a.sum() + 1e-30)
    
    for it in range(max_iter):
        # Step 1: Enforce time-domain positivity
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-30:
            break
        a = a / s
        
        # Step 2: Enforce frequency-domain magnitude
        A = np.fft.fft(a)
        if n % 2 == 0:
            for k in range(1, n // 2 + 1):
                phase_k = np.angle(A[k])
                A[k] = target_magnitude[k] * np.exp(1j * phase_k)
                A[n - k] = target_magnitude[k] * np.exp(1j * (-phase_k))
        else:
            for k in range(1, (n - 1) // 2 + 1):
                phase_k = np.angle(A[k])
                A[k] = target_magnitude[k] * np.exp(1j * phase_k)
                A[n - k] = target_magnitude[k] * np.exp(1j * (-phase_k))
        
        a = np.real(np.fft.ifft(A))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

def _cg_solve_hankel(W, n, rhs, max_iter=2000, tol=1e-14):
    """Solve H*x = rhs using CG with Hankel structure."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs_new < tol * tol * rs0:
            break
        factor = rs_new / rs
        p = r + factor * p
        rs = rs_new
    
    return x

def _newton_polish(a_init, deadline, max_rounds=50):
    """Newton-type polish using CG for the Hessian solve."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: top percentage of convolution values
        tol_frac = max(0.001, 0.01 * (1 - rnd / max_rounds))
        threshold = c_max * (1 - tol_frac)
        active = c >= threshold
        if not active.any():
            active[int(np.argmax(c))] = True
        
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()
        
        # Newton step: solve H*dx = -grad, project onto sum=0
        grad = _hankel_matvec(a, W, n)
        grad_centered = grad - grad.mean()
        
        dx = _cg_solve_hankel(W, n, -grad_centered, max_iter=500)
        dx = dx - dx.mean()
        
        # Line search
        alpha = 1.0
        for ls in range(30):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < _score(a) - 1e-15:
                a = a_try
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a.copy()
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
    
    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
