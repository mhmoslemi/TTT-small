def _next_pow2(n):

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=5000, tol=1e-16):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-40:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-40 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: start tight, loosen if no progress
        if no_improve < 5:
            tol_mult = 1e-9
        elif no_improve < 15:
            tol_mult = 1e-7
        elif no_improve < 30:
            tol_mult = 1e-5
        else:
            tol_mult = 1e-3

        active = c >= (c_max - tol_mult * c_max)
        if active.sum() < 1:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_c = g - np.mean(g)
        g_norm = float(np.linalg.norm(g_c))

        if g_norm < 1e-13 * n:
            break

        # Solve H * dx = -g_c using CG
        dx = _cg_solve(W, n, -g_c, max_iter=5000, tol=1e-16)
        dx = dx - np.mean(dx)
        dx_norm = float(np.linalg.norm(dx))

        if dx_norm < 1e-30 or not np.isfinite(dx_norm):
            dx = -g_c / max(g_norm, 1e-30)
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30:
                break

        # Normalize for line search
        dx_scaled = dx / max(dx_norm, 1e-30)

        step_done = False
        alpha = 1.0
        best_alpha = 0.0
        best_cmax_try = c_max

        for ls in range(80):
            a_try = a + alpha * dx_scaled
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.4
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())

            if c_max_try < best_cmax_try:
                best_cmax_try = c_max_try
                best_alpha = alpha

            if c_max_try < c_max - 1e-16 * c_max:
                a = a_try
                step_done = True
                break
            alpha *= 0.4
            if alpha < 1e-18:
                break

        if not step_done and best_alpha > 0:
            a_try = a + best_alpha * dx_scaled
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a = a_try / s_try
                step_done = True

        if not step_done:
            # Steepest descent fallback with multiple step sizes
            for trial in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
                a_try = a - trial * g_c / max(g_norm, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done:
            no_improve += 3

        if no_improve > 50:
            break

    return best_a, best_sc


def _adaptive_lbfgs_fine(a_init, deadline, min_time_remaining=30.0):
    """
    Multi-pass L-BFGS with a finer beta grid than before.
    Uses a geometric sequence of betas from gap-scale down to 1e-12 * c_max.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Finer beta schedule: more steps, slower reduction
    beta = base_beta
    min_beta = max_c * 1e-14
    prev_sc = current_sc
    stagnation_count = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations: be conservative for large n
        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.004 * scale
        max_iter = min(15000, max(50, int(remaining / (est_iter_time * 40))))

        try:
            obj_fn = _SmoAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-18,
                         'maxcor': 100, 'maxls': 500, 'max_fun': max_iter * 2}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnation_count = 0
            else:
                stagnation_count += 1
        except Exception:
            stagnation_count += 1

        # Adaptive beta reduction based on progress
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-10 * c_max_new and sc_improved > 1e-14:
            beta *= 0.1
        elif new_gap < 1e-7 * c_max_new:
            beta *= 0.15
        elif stagnation_count > 2:
            beta *= 0.25
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _fourier_perturb(a, sigma, rng):
    """Perturb in Fourier domain: modify phase randomly, keep magnitude roughly."""
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    # Add random phase perturbation
    dphase = sigma * rng.standard_normal(len(phase))
    new_phase = phase + dphase
    new_A = mag * np.exp(1j * new_phase)
    # Ensure real and non-negative
    a_new = np.fft.irfft(new_A, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        # Fallback: small Gaussian perturbation
        z = np.log(np.maximum(a, 1e-30))
        z = z + sigma * 0.1 * rng.standard_normal(n)
        return _softmax(z)
    return a_new / s


def _gaussian_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.0 + 0.01
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick profile exploration ===
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["tapered", "asym_taper", "gaussian", "cos", "left_heavy", "exp_decay"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 1200:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': score={sc_p:.10f} (new best profile)")

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 1100:
        a_quick = _adaptive_lbfgs_fine(best_profile_a, min(deadline - 1100, time.monotonic() + 150),
                                        min_time_remaining=900.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_profile")
        if sc_quick < best_profile_sc:
            best_profile_a = a_quick
            best_profile_sc = sc_quick

    # Also try parent with quick optimize
    if time.monotonic() < deadline - 1000:
        a_par = _adaptive_lbfgs_fine(parent, min(deadline - 1000, time.monotonic() + 120),
                                       min_time_remaining=900.0)
        sc_par = _score(a_par)
        update_best(a_par, sc_par, "parent_quick")

    # === Phase 2: Main adaptive L-BFGS ===
    print(f"\n[Phase 2] Main adaptive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_profile_a
    a_main = _adaptive_lbfgs_fine(start_a, deadline, min_time_remaining=600.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: Newton KKT polish ===
    print(f"\n[Phase 3] Newton KKT polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_np, sc_np = _newton_kkt_polish(best_seq, min(deadline - 400, time.monotonic() + 300),
                                          max_rounds=200)
        update_best(a_np, sc_np, "newton_1")

    # === Phase 4: Fourier-domain diverse restarts ===
    print(f"\n[Phase 4] Fourier restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        for i, (sigma, method) in enumerate([(0.3, "fourier"), (0.5, "fourier"),
                                              (0.1, "gauss"), (0.3, "gauss"),
                                              (0.8, "fourier"), (0.2, "gauss")]):
            if time.monotonic() > deadline - 250:
                break
            if method == "fourier":
                p_start = _fourier_perturb(best_seq, sigma, rng)
            else:
                p_start = _gaussian_perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)

            # Quick optimize the perturbation
            a_r = _adaptive_lbfgs_fine(p_start, min(deadline - 250, time.monotonic() + 80),
                                         min_time_remaining=240.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{method}_{i}")

            # Newton polish if promising
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 200:
                a_rp, sc_rp = _newton_kkt_polish(a_r, min(deadline - 200, time.monotonic() + 60),
                                                   max_rounds=80)
                update_best(a_rp, sc_rp, f"restart_{method}_{i}_np")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_base = len(best_seq) if best_seq is not None else n_parent
    n_candidates = []
    for factor in [0.5, 0.75, 1.0, 1.25, 2.0]:
        n_try = int(n_base * factor)
        n_try = max(2048, min(65536, n_try))
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    # Also try some fixed sizes
    for nf in [8192, 16384, 32768, 65536]:
        if nf not in n_candidates:
            n_candidates.append(nf)
    n_candidates.sort()

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if abs(n_try - n_base) < 100 and best_seq is not None:
            continue  # Skip sizes very close to current best
        a_init = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt = _adaptive_lbfgs_fine(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 60:
            a_np3, sc_np3 = _newton_kkt_polish(a_opt, min(deadline - 60, time.monotonic() + 40),
                                                 max_rounds=60)
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # === Phase 6: Extended Newton on best ===
    print(f"\n[Phase 6] Extended Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Multiple Newton passes with increasing tightness
        a_cur = best_seq.copy()
        for pass_num in range(3):
            if time.monotonic() > deadline - 10:
                break
            a_np, sc_np = _newton_kkt_polish(a_cur, min(deadline - 10, time.monotonic() + 50),
                                              max_rounds=100)
            update_best(a_np, sc_np, f"ext_newton_{pass_num}")
            if a_np is not a_cur:
                a_cur = a_np

    # === Phase 7: Final micro-optimization ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())

        # Ultra-fine L-BFGS
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 1e-12
        remaining = deadline - 2 - time.monotonic()
        if remaining > 3:
            max_iter = min(5000, max(50, int(remaining / 0.01)))
            try:
                obj_fn = _SmoAdapter(beta_fin, n)
                grad_fn = _SmoGradAdapter(beta_fin, n)
                result = minimize(
                    obj_fn, z, jac=grad_fn,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                             'maxcor': 100, 'maxls': 300}
                )
                a_flb = _softmax(result.x)
                update_best(a_flb, _score(a_flb), "final_lbfgs")
            except Exception:
                pass

        # Final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_veryfin, sc_veryfin = _newton_kkt_polish(best_seq, deadline - 0.1, max_rounds=100)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
