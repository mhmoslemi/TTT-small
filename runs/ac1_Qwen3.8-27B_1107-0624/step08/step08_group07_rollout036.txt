    # Simplest correct approach: use the FFT with the right signs
    # h[j] = sum_m a[m] * p[m+j]
    # Let's compute via: FFT approach where we zero-pad p to length N >= n + (2n-1)
    # h = IFFT(conj(FFT(a, N)) * FFT(p, N))[N - j] ... no this is getting too complex.
    
    # Let me just use the peak-based approximation which is exact for small T:
    # For small T, p is concentrated on the top few peaks.
    g = np.zeros(n)
    sorted_idx = np.argsort(b)[::-1]
    # Include all lags with non-negligible weight
    for idx in sorted_idx:
        if p[idx] < 1e-12:
            break
        k = idx
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    
    # Convert to z-gradient: grad_z = a * (g - dot(a, g))
    W = float(np.dot(a, g))
    grad_z = a * (g - W)
    return obj, grad_z


def _lrefs_optimize(z_init, T_schedule, max_iter_per_T=5000):
    """Run L-BFGS with continuation over T values."""
    n = len(z_init)
    best_z = z_init.copy()
    best_score = _score_full(_softmax(z_init))
    
    for T in T_schedule:
        if time.monotonic() > _deadline:
            break
        z = best_z.copy()
        
        def fun(z_eval):
            obj, _ = _objective_and_grad(z_eval, T)
            return obj
        
        def fprime(z_eval):
            _, grad = _objective_and_grad(z_eval, T)
            return grad
        
        res = minimize(fun, z, method='L-BFGS-B', jac=fprime,
                       options={'maxiter': max_iter_per_T, 'maxls': 20,
                                'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': max_iter_per_T * 3,
                                'iprint': -1})
        if res.x is not None and np.all(np.isfinite(res.x)):
            a_try = _softmax(res.x)
            sc = _score_full(a_try)
            if sc < best_score:
                best_score = sc
                best_z = res.x.copy()
        print(f"  T={T:.6f}: L-BFGS iters={res.nit}, score={best_score:.10f}")
    
    return best_z, best_score


def _active_set_polish(a, deadline, max_iter=3000):
    """Active-set equalization: find direction that equalizes top peaks."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_full(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks (within 0.5% of max)
        tol = c_max * 0.995
        active = np.where(c >= tol)[0]
        if len(active) < 2:
            active = np.array([np.argmax(c)])
        if len(active) > 12:
            top = active[np.argsort(c[active])[::-1][:12]]
            active = top
        
        K = len(active)
        # Build M matrix: M[i,j] = 2*a[k_i - j] if valid
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        
        # Add sum constraint row
        M_full = np.vstack([M, np.ones(n)])
        b_vec = np.zeros(K + 1)
        b_vec[:K] = -1.0  # want to decrease all active peaks
        
        # Minimum norm solution
        G = M_full @ M_full.T
        try:
            G_inv = np.linalg.inv(G + 1e-10 * np.eye(G.shape[0]))
            d = M_full.T @ (G_inv @ b_vec)
        except:
            d = np.linalg.lstsq(M_full, b_vec, rcond=None)[0]
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d /= d_norm
        
        # Try multiple step sizes
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_full(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=20000):
    """Direct subgradient method with adaptive step size."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_full(a)
    lr = 0.003
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: peaks within 0.1% of max
        active = np.where(c >= c_max * 0.999)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Subgradient (average over active)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)
        
        # Project to tangent space of simplex
        g_tan = g - a * np.dot(a, g)
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break
        
        step = lr * g_tan / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc = _score_full(a_new)
        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.01, 0.01)
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-10 or no_improve > 200:
                break
    
    return best_a, best_score


def _coordinate_polish(a, deadline, max_rounds=100):
    """Targeted coordinate descent on elements affecting the peak."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_full(a)
    c = _fft_convolve(best_a, best_a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c_max = float(c.max())
        k_max = int(np.argmax(c))
        
        # Elements contributing to the peak
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        improved = False
        # Try modifying contributing elements
        step = max(1, (hi - lo + 1) // 50)
        for j in range(lo, hi + 1, step):
            old = best_a[j]
            if old < 1e-15:
                continue
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(2.0 * n * c_cand.max())
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score
