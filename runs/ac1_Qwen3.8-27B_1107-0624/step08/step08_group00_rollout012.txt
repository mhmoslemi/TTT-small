"""
Optimized step function minimizer using FFT-based L-BFGS with adaptive
log-sum-exp smoothing, Nesterov-accelerated active-set gradient descent,
and iterative peak reduction. Key improvements over previous: adaptive beta

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 4:
                print(f"    L-BFGS beta={beta:.2e}: score={current_sc:.10f} t={time.monotonic()-t_start_ref:.1f}s")
        except Exception:
            pass

    return current_a


def _active_set_polish_momentum(a_init, deadline, min_time_remaining=30.0,
                                tol_frac_start=0.005, tol_frac_end=0.00002,
                                lr_init=0.04, momentum_init=0.85):
    """Active-set gradient descent with Nesterov-like momentum and adaptive step size."""
    global t_start_ref
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    momentum = momentum_init
    v = np.zeros(n)  # momentum accumulator
    no_improve = 0
    max_stall = 300
    stall_count = 0
    N = _next_pow2(2 * n)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Determine active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Compute gradient: g[j] = 2 * sum_i w[j+i] * a[i]
        W = np.fft.rfft(w, N)
        A = np.fft.rfft(a, N)
        cross = np.fft.irfft(W * np.conjugate(A), N)[:n]
        grad = 2.0 * cross
        grad_t = grad - np.mean(grad)  # center to preserve sum
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        # Nesterov momentum step
        v = momentum * v - lr * grad_t
        a_try = a + v

        # Line search with backtracking
        a_old_sc = current_sc
        improved = False
        alpha = 1.0
        for _ in range(50):
            a_cand = a + alpha * v
            a_cand = np.maximum(a_cand, 0.0)
            s_cand = a_cand.sum()
            if s_cand < 1e-10:
                alpha *= 0.5
                continue
            a_cand = a_cand / s_cand
            c_cand = _full_convolve(a_cand, a_cand)
            sc_cand = 2.0 * n * float(c_cand.max())
            if sc_cand < a_old_sc - 1e-13 * max(1.0, a_old_sc):
                a = a_cand
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            v *= 0.5  # damp momentum
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.8)
                lr *= 0.75
                momentum = max(0.5, momentum * 0.95)
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.4
                momentum = momentum_init * 0.8
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _nesterov_peak_reduction(a_init, deadline, min_time_remaining=30.0):
    """Direct peak reduction using Nesterov-accelerated gradient steps
    on the exact (non-smoothed) max-convolution objective."""
    global t_start_ref
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    N = _next_pow2(2 * n)
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    max_stall = 200

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Active set: top peaks within 0.1% of max
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = max(1, int(active.sum()))
        w = np.zeros_like(c)
        active_indices = np.where(active)[0]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])
        w[active_indices] = 1.0 / len(active_indices)

        # Gradient
        W = np.fft.rfft(w, N)
        A = np.fft.rfft(a, N)
        cross = np.fft.irfft(W * np.conjugate(A), N)[:n]
        grad = 2.0 * cross
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-16:
            break

        # Nesterov step
        v = momentum * v - lr * grad_t
        proposed = a + v

        # Line search
        a_old_sc = current_sc
        improved = False
        alpha = 1.0
        for _ in range(40):
            a_cand = a + alpha * v
            a_cand = np.maximum(a_cand, 0.0)
            s_cand = a_cand.sum()
            if s_cand < 1e-10:
                alpha *= 0.5
                continue
            a_cand = a_cand / s_cand
            c_cand = _full_convolve(a_cand, a_cand)
            sc_cand = 2.0 * n * float(c_cand.max())
            if sc_cand < a_old_sc - 1e-13:
                a = a_cand
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.8
            momentum = max(0.5, momentum * 0.97)
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.02
                momentum = 0.9
                no_improve = 0
                break
        else:
            if no_improve == 0 and lr < 0.02:
                lr = min(0.02, lr * 1.1)

    return best_a, _score(best_a)


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
