```python
def _project_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Optimized step-function coefficient search using aggressive LSE annealing,
momentum-based active-set polishing, LP-guided refinement, and multi-n exploration.
Improves upon previous 1.505665 by spending more time at small LSE temperatures,
using Nesterov momentum in active-set steps, and exploring larger sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_anneal(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(80, len(c_sorted) - 1)]
    if gap < max_c * 0.0003:
        gap = max_c * 0.0003
    base_beta = max(gap, max_c * 0.003)

    # 40 geometric steps from 1.0 down to ~0.0001
    ratio = 0.82
    betas = base_beta * (ratio ** np.arange(40, dtype=np.float64))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10.0:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.003 * scale
        # Spend more time at small beta
        if beta < base_beta * 0.1:
            max_iter = min(8000, max(100, int(remaining / (est_iter_time * 20))))
        elif beta < base_beta * 0.5:
            max_iter = min(5000, max(50, int(remaining / (est_iter_time * 15))))
        else:
            max_iter = min(3000, max(30, int(remaining / (est_iter_time * 12))))

        def obj_func(z_val):
            return _lse_obj(z_val, beta, n)

        def grad_func(z_val):
            return _lse_grad(z_val, beta, n)

        try:
            result = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_momentum(a_init, deadline, tol_start=0.004, tol_end=0.00001,
                         lr_start=0.03, top_k=7, momentum=0.7):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_start
    tol = tol_start
    mu = momentum
    v = np.zeros(n)
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > top_k * 8:
            top_indices = np.argsort(c)[::-1][:top_k]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = top_k

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Nesterov momentum update
        v = mu * v - lr * grad
        a_try = a + v
        a_try = _project_simplex(a_try)

        # Check improvement
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < current_sc - 1e-15:
            a = a_try
            iter_count += 1

            # Decay momentum and step
            if no_improve == 0 and iter_count % 100 == 0:
                mu = max(0.5, mu * 0.995)
            if tol > tol_end and iter_count % 50 == 0:
                tol = max(tol_end, tol * 0.99)
        else:
            a = a - v  # undo
            v = 0.0
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.65
                mu *= 0.9
                tol = max(tol_end, tol * 0.7)
            if no_improve >= 500:
                # Restart from best
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_start * 0.5
                mu = 0.6
                tol = max(tol_end, tol * 0.4)
                no_improve = 0

        if lr < 1e-8:
            break

    return best_a, _score(best_a)


def _active_set_direct(a_init, deadline, tol_start=0.003, tol_end=0.000005,

    n = len(a_init)
    a = a_init.copy()
    s = a.sum()

    stall = 0

        c = _full_convolve(a, a)
        c_max = float(c.max())

            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > top_k * 6:
            top_indices = np.argsort(c)[::-1][:top_k]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True

        w = np.zeros_like(c)

        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)

        if gnorm < 1e-17:
            break

        # Backtracking line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(60):

            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try

            if sc_try < base_sc - 1e-15:
                a = a_try
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve % 25 == 0:

                lr *= 0.5
            if no_improve >= max_stall:
                a = best_a.copy()

        else:

                tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.02)

    return best_a, _score(best_a)


def _lp_refine(a, deadline, n_rounds=20):
    """LP-guided direction finding: maximize sum(g) s.t. g>=0, (a*g)[k] <= M."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())
        s_a = a.sum()

        # Find nearly-active constraints
        tol_frac = 0.002
        active_constraints = np.where(c >= M * (1 - tol_frac))[0]
        # Limit to at most 60 constraints for LP speed
        if len(active_constraints) > 60:
            top_idx = np.argsort(c)[::-1][:60]
            active_constraints = top_idx

        m_con = len(active_constraints)
        # Build constraint matrix A_ub: (a*g)[k] <= M for k in active_constraints
        # (a*g)[k] = sum_j a[j] * g[k-j] (with bounds checking)
        A = np.zeros((m_con, n))
        for i, k in enumerate(active_constraints):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            A[i, j_lo:j_hi + 1] = a[j_lo:j_hi + 1]

        b_ub = np.full(m_con, M)
        # Objective: maximize sum(g) => minimize -sum(g)
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                          method='highs', options={'time_limit': 30})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0)
                s_g = g.sum()
                if s_g > 1e-10:
                    g = g / s_g  # normalize to sum 1
                    # Blend
                    t_val = 0.1
                    a_new = (1 - t_val) * (a / s_a) + t_val * g
                    a_new = np.maximum(a_new, 0)
                    s_new = a_new.sum()
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        sc_new = _score(a_new)
                        if sc_new < best_sc:
                            best_sc = sc_new
                            best_a = a_new.copy()
                            a = a_new
        except Exception:
            pass

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Slight taper at edges
        edge = max(1, n // 50)
        for i in range(edge):
            a[i] *= (i + 1) / edge
            a[n - 1 - i] *= (i + 1) / edge
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
```
