    """Project v onto the simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_c_at_lag(a, k):
    """Gradient of c[k] = sum_j a[j]*a[k-j] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    return g

    """Compute the smooth log-sum-exp objective and its gradient in z-space."""
    a = _softmax_stable(z)

    c_max = float(c.max())
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        f_val = c_max
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        f_val = c_max + np.log(s) / T
        p = e / s
    # Compute g = 2 * sum_k p[k] * a[k-j] for each j
    g = np.zeros(n)
    # Use top lags for efficiency
    threshold = p.max() * 1e-6

        hi = min(k, n - 1)

"""
Optimized step function search using smooth log-sum-exp surrogate with L-BFGS-B,
active-set subgradient refinement, and coordinate descent polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_c_at_lag(a, k):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _smooth_obj_grad(z, T, n):
    a = _softmax_stable(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        f_val = c_max
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        f_val = c_max + np.log(s) / T
        p = e / s
    g = np.zeros(n)
    threshold = p.max() * 1e-8
    active = np.where(p >= threshold)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += p[k] * 2.0 * a[k - hi:k - lo + 1][::-1]
    W = float(a @ g)
    grad_z = a * (g - W)
    return f_val, grad_z


def _lbgfsb_run(z_init, T, n, max_iter=300, deadline=None):
    """Run L-BFGS-B on the smooth surrogate."""
    def fun_and_grad(z):
        return _smooth_obj_grad(z, T, n)

    best_z = z_init.copy()
    best_f = _smooth_obj_grad(z_init, T, n)[0]
    best_exact = _score_norm(_softmax_stable(z_init))

    result = scipy_minimize(
        fun_and_grad, z_init, method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
    )
    if result.success or result.fun < best_f:
        best_z = result.x
        best_f = result.fun
    exact = _score_norm(_softmax_stable(best_z))
    if exact < best_exact:
        best_exact = exact
    return best_z, best_exact


def _subgradient_optimize(a, deadline, max_iter=100000, top_k=5,
                          initial_lr=0.01, use_soft=True):
    """Projected subgradient with Barzilai-Borwein step size."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    lr = initial_lr
    s_prev = None
    y_prev = None
    a_prev = None
    g_prev = None
    stall_count = 0

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        sorted_idx = np.argsort(c)[::-1]
        if use_soft:
            active_range = min(top_k + 3, len(sorted_idx))
            c_top = c[sorted_idx[:active_range]]
            T_soft = max(c_max - c_top[-1], c_max * 0.001) * 0.5
            w = np.exp((c_top - c_max) / T_soft)
            w = w / w.sum()
            active = sorted_idx[:active_range]
            weights = w
        else:
            active = sorted_idx[:top_k]
            w = np.ones(len(active)) / len(active)
            weights = w

        g = np.zeros(n)
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += weights[i] * 2.0 * a[k - hi:k - lo + 1][::-1]

        g_centered = g - float(a @ g) * np.ones(n)
        g_norm = np.linalg.norm(g_centered)
        if g_norm < 1e-18:
            break

        a_cand = a - lr * g_centered / g_norm
        a_cand = _project_simplex(a_cand)

        sc = _score_norm(a_cand)
        if sc < best_score - 1e-15:
            s_new = a_cand - a
            y_new = g_centered - (g_prev if g_prev is not None else 0.0)

            if s_prev is not None and y_prev is not None:
                sy = float(s_prev @ y_new)
                ss = float(s_prev @ s_new)
                if abs(sy) > 1e-15 and sy > 0:
                    lr_bb = ss / sy
                    lr_bb = np.clip(lr_bb, lr * 0.3, lr * 3.0)
                    lr = 0.7 * lr + 0.3 * lr_bb

            best_score = sc
            best_a = a_cand.copy()
            s_prev = s_new
            y_prev = y_new
            a = a_cand
            g_prev = g_centered
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                lr *= 0.8
                stall_count = 0
            if lr < 1e-12:
                break

    return best_a, best_score


def _active_set_equalize(a, deadline, max_iter=2000, tol=0.005):
    """Try to equalize top peak lags via Newton steps on active set."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        tol_val = c_max * (1.0 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 20:
            sorted_a = active[np.argsort(c[active])[::-1][:20]]
            active = sorted_a

        K = len(active)
        grads = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grads[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        G = grads @ grads.T
        g_sum = grads.sum(axis=0)
        g_proj = g_sum - float(a @ g_sum) * np.ones(n)

        G_reg = G + 1e-10 * np.eye(K)
        ones_K = np.ones(K)
        G_ext = np.block([[G_reg, ones_K[:, None]],
                          [ones_K[None, :], np.zeros((1, 1))]])
        b_ext = np.concatenate([c[active] - c_max, [0.0]])

        try:
            sol = np.linalg.solve(G_ext, b_ext)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(G_ext, b_ext, rcond=None)[0]

        d = -grads.T @ sol[:K]
        d_proj = d - float(np.sum(d)) * a
        d_norm = np.linalg.norm(d_proj)
        if d_norm < 1e-15:
            break
        d_dir = d_proj / d_norm

        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_cand = a + alpha * d_dir
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand = a_cand / s
            sc = _score_norm(a_cand)
            if sc < best_score * (1.0 - 1e-12):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            break

    return best_a, best_score

                  f"t={time.monotonic()-t_start:.0f}s")
            if full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()

    # === Phase 6: Final ultra-fine polish (1750 - deadline) ===
    if time.monotonic() < deadline - 10:
        print("Phase 6: Final polish")
        a_final, sc_final = _subgradient_optimize(
            best_a, deadline - 5, max_iter=50000, top_k=3,
            initial_lr=0.002, use_soft=False)
        full_sc_final = _score(a_final)
        if full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()
            print(f"  Final subgrad: score={full_sc_final:.10f}")

        if time.monotonic() < deadline - 3:
            a_cd_f, _ = _coordinate_polish(best_a, deadline - 2, max_rounds=100)
            full_sc_cd_f = _score(a_cd_f)
            if full_sc_cd_f < best_score:
                best_score = full_sc_cd_f
                best_a = a_cd_f.copy()
                print(f"  Final coord: score={full_sc_cd_f:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_check = _score(final)
    print(f"FINAL: score={sc_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
