Optimized step-function search using exact subgradient with Nesterov acceleration,
adaptive active-set L-BFGS, and structured multi-n exploration.
Key differences from previous approach:
- Uses exact subgradient (average of top-k peak gradients) instead of pure LSE smoothing
- Nesterov momentum for faster convergence in gradient phase
- Adaptive active-set size based on peak structure
- Symmetry-based initialization
- More focused time allocation to local refinement

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _subgradient_at_peaks(a, c, active_indices):
    """
    Compute the average subgradient over active peak indices.
    For a peak at k: grad_j = 4n * a[k-j] / S^2 - 4n * c[k] / S^3
    where a[k-j] = 0 if k-j is out of [0, n-1].
    """
    n = len(a)
    S = float(a.sum())
    k_active = len(active_indices)
    if k_active == 0:
        return np.zeros(n)

    grad = np.zeros(n)
    idx = np.arange(n)

    for k_star in active_indices:
        rev = k_star - idx
        valid = (rev >= 0) & (rev < n)
        a_rev = np.zeros(n)
        a_rev[valid] = a[rev[valid]]
        grad += 4.0 * n * a_rev / (S * S) - 4.0 * n * c[k_star] / (S * S * S)

    grad /= k_active
    return grad


def _find_active_set(c, tol_frac=0.01):
    """Find indices where convolution is within tol_frac of the maximum."""
    c_max = float(c.max())
    threshold = c_max * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    return active


def _nesterov_optimize(a_init, deadline, max_iters=200000,
                       tol_frac=0.005, momentum_start=0.9, momentum_end=0.999,
                       lr_start=0.05, lr_end=0.001):

    """

    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Nesterov momentum
    v = np.zeros(n)
    momentum = momentum_start
    lr = lr_start

    no_improve = 0
    stall_count = 0
    max_stall = 1000

    t_start = time.monotonic()
    total_iters = 0

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        active = _find_active_set(c, tol_frac=tol_eff)

        # Compute subgradient
        grad = _subgradient_at_peaks(a, c, active)

        # Project gradient: remove mean to maintain normalization
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive learning rate
        lr_eff = lr_start * (lr_end / lr_start) ** (min(1.0, total_iters / 50000.0))
        lr_eff = max(lr_eff, lr_end)

        # Nesterov step
        v = momentum * v - lr_eff * grad
        a_new = a + v

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a + 0.1 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0.0)

            if s_new < 1e-10:

            a_new = a_new / s_new
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Backtracking: if objective increased significantly, rollback
        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        if sc_new > current_sc * 1.01 + 1e-10:
            # Rollback and reduce momentum
            v = 0.5 * v
            a_new = a - 0.01 * grad / (gnorm + 1e-15)

            a = np.maximum(a, 0.0)

            v = np.zeros(n)
            momentum = momentum_start
            no_improve = 0

def _subgradient_at_z(active_indices, a, c, z):
    """Compute subgradient in z-space (log-space).
    Since a = softmax(z), da/dz = a * (I - a), so:
    d(score)/dz_j = d(score)/da_j * a_j * (1 - a_j) approx d(score)/da_j * a_j
    (the (1-a_j) factor is close to 1 for small a_j)
    For large n, a_j ~ 1/n, so (1-a_j) ~ 1. We use the approximation:
    d(score)/dz_j ≈ d(score)/da_j * a_j
    """
    n = len(a)
    S = float(a.sum())
    k_active = len(active_indices)
    if k_active == 0:
        return np.zeros(n)

    grad_a = np.zeros(n)
    idx = np.arange(n)

    for k_star in active_indices:
        rev = k_star - idx
        valid = (rev >= 0) & (rev < n)
        a_rev = np.zeros(n)
        a_rev[valid] = a[rev[valid]]
        grad_a += 4.0 * n * a_rev / (S * S) - 4.0 * n * c[k_star] / (S * S * S)

    grad_a /= k_active
    # Transform to z-space: grad_z = grad_a * a (approximation for large n)
    grad_z = grad_a * a
    return grad_z

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod2":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 4.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.arange(1, n + 1)
        a = 1.0 / np.sqrt(t.astype(float))
        return a / a.sum()
    elif kind == "symmetric_noise":
        base = np.ones(n)
        half = n // 2
        noise = np.random.default_rng(99).standard_normal(half) * 0.1
        base[:half] = 1.0 + noise[:n % 2 == 0 and half or half]
        # Make it properly symmetric
        if n % 2 == 0:
            base = np.concatenate([base[:half], base[:half][::-1]])
        else:
            base = np.concatenate([base[:half], [1.0], base[:half-1][::-1]])
        base = np.maximum(base, 1e-10)
        return base / base.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng_local):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng_local.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top contributor to the peak
        k_star = int(np.argmax(c))
        idx = np.arange(n)
        rev = k_star - idx
        valid = (rev >= 0) & (rev < n)

        # Contribution of each a_j to the peak: a_j * a[k*-j]
        contrib = np.zeros(n)
        contrib[valid] = a[valid] * a[rev[valid]]
        total = contrib.sum()
        if total < 1e-15:
            break

        # Compute gradient direction: reduce high-contributing, increase low-contributing
        # grad_j = 4n * a[k*-j] / S^2 - 4n * M / S^3
        S = float(a.sum())
        a_rev = np.zeros(n)
        a_rev[valid] = a[rev[valid]]
        grad = 4.0 * n * a_rev / (S * S) - 4.0 * n * c_max / (S * S * S)
        grad = grad - np.mean(grad)

        # Take a small step in the negative gradient direction
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Try multiple step sizes
        best_step = None
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
            a_try = a - alpha * grad / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if best_step is None or sc_try < best_step[1]:
                best_step = (a_try, sc_try)

        if best_step is not None and best_step[1] < best_sc - 1e-14:
            best_sc = best_step[1]
            best_a = best_step[0].copy()
            a = best_a

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS with LSE on parent (establish good baseline)
    Phase 2: Nesterov accelerated subgradient (exploit local structure)
    Phase 3: Active-set L-BFGS (refine with exact gradient)
    Phase 4: Peak shaving (targeted peak reduction)
    Phase 5: Multi-n exploration (find better n)
    Phase 6: Final polish
    """
    global rng
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS with LSE (quick, to get a good starting point) ===
    print(f"\n[P1] L-BFGS LSE on parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200.0, t_start + 300.0)
    a1 = _lbfgs_lse_standard(best_a, p1_deadline, min_remaining=30.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  P1 improved: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P1 no improvement: {sc1:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Nesterov accelerated subgradient ===
    print(f"\n[P2] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 800.0, t_start + 700.0)
    a2, sc2 = _nesterov_optimize(best_a, p2_deadline, max_iters=300000,
                                  tol_frac=0.003, momentum_start=0.9,
                                  momentum_end=0.999, lr_start=0.04, lr_end=0.0005)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  P2 improved: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P2 no improvement: {sc2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set L-BFGS (exact gradient refinement) ===
    print(f"\n[P3] Active-set L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 500.0, t_start + 1000.0)
    a3, sc3 = _active_set_lbfgs(best_a, p3_deadline, tol_start=0.003,
                                 tol_end=0.0002, top_k_max=8)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  P3 improved: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P3 no improvement: {sc3:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak shaving ===
    print(f"\n[P4] Peak shaving gradient (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 300.0, t_start + 1200.0)
    a4, sc4 = _peak_shave_gradient(best_a, p4_deadline, n_trials=500)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  P4 improved: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")

    # If peak shaving helped, do another L-BFGS pass
    if sc4 < best_score + 0.0005 and time.monotonic() < deadline - 200:
        p4b_deadline = min(deadline - 200.0, t_start + 1300.0)
        a4b = _lbfgs_lse_standard(best_a, p4b_deadline, min_remaining=20.0)
        sc4b = _score(a4b)
        if sc4b < best_score:
            best_a = a4b.copy()
            best_score = sc4b
            print(f"  P4b improved: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_candidates = set()
    # Generate candidates around n_best
    for delta in [-2048, -1024, -512, 512, 1024, 2048, 4096]:
        n_try = n_best + delta
        if n_try >= 1024 and n_try <= 65536:
            n_candidates.add(n_try)
    # Add some fixed candidates
    for n_fix in [8192, 12288, 16384, 20480, 10240, 14336]:
        if n_fix >= 1024:
            n_candidates.add(n_fix)

    n_candidates = sorted(n_candidates)
    # Limit to top 4 most promising (closest to n_best first)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_tries = n_candidates[:4]

    for n_try in n_tries:
        if time.monotonic() > deadline - 100.0:
            break
        if n_try == n_best:
            continue
        a_init = _resample(best_a, n_try)
        remaining = deadline - time.monotonic() - 50.0
        if remaining < 60:
            break
        # Quick L-BFGS
        a_opt = _lbfgs_lse_standard(a_init, deadline - 30.0, min_remaining=15.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            n_best = n_try
            print(f"  P5 improved: {best_score:.10f} (n={n_try}) t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  P5 n={n_try}: {sc_opt:.10f} (no improvement)")

    # === Phase 6: Final polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        # Nesterov with very small steps
        a_fin, sc_fin = _nesterov_optimize(best_a, deadline - 3.0, max_iters=100000,
                                            tol_frac=0.001, momentum_start=0.8,
                                            momentum_end=0.99, lr_start=0.01, lr_end=0.0001)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  P6 Nesterov improved: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

        # Peak shave
        if time.monotonic() < deadline - 2.0:
            a_ps, sc_ps = _peak_shave_gradient(best_a, deadline - 1.0, n_trials=100)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  P6 peak shave improved: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * (np.sqrt(0.01) / 256)
    else:
        # Ensure sum >= 0.01 by scaling if needed
        if s < 0.01:
            final = final * (0.01 / s)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
