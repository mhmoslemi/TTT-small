def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())

    c = _full_convolve(a, a)

def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)

def _cg_solve(hess_func, b, tol=1e-12, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha < 0 or not np.isfinite(alpha):
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

    best_a = a_init.copy()

    for tol_frac in [0.005, 0.003, 0.002, 0.001, 0.0005]:
        if time.monotonic() > deadline:
            break
        nw_dead = min(deadline, time.monotonic() + 30)
        a_nw, sc_nw = _newton_active_set(a_init, nw_dead, max_iter=100,
                                          tol_frac=tol_frac, mu_base=1e-8)
        if sc_nw < best_sc:
            best_sc = sc_nw
            best_a = a_nw.copy()
            a_init = a_nw.copy()

    return best_a, best_sc


def _spectral_phase_opt(a_init, deadline, n_iters=200):
    """
    Spectral phase optimization: modify the phase of the Fourier transform
    to reduce the autocorrelation peak while maintaining non-negativity.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    phase = np.angle(A)

    lr = 0.1
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Weight spectrum: more weight on frequencies contributing to peaks
        # Use the gradient in spectral domain
        # d(max_c)/d(phase_m) relates to imaginary part of A^2 * exp(i*omega*k_peak)

        # Simpler approach: use subgradient in spatial domain, update phase
        shifted = (c - c_max) / (c_max * 0.01)
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e > 1e-300:
            w = e / s_e
        else:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)

        # Update in spectral domain: modify phase to reduce gradient
        grad_F = np.fft.rfft(grad_a)
        # phase gradient: d|A|^2/d(phi_m) = 2 * Re(A * conj(dA/dphi)) ... 
        # Simpler: just do spatial step and recompute spectrum
        a_new = a - lr * grad_a
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.3)
        else:
            lr *= 0.9
            a = a_new  # still move, just track best
            # Project back
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                break
            a = a / s

    return best_a, best_sc


def _diversify_and_refine(a_best, deadline, rng, n_restarts=4):
    """Generate diverse candidates and refine the best ones."""
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score(a_best)

    for i in range(n_restarts):
        if time.monotonic() > deadline - 20:
            break

        # Different perturbation strategies
        if i == 0:
            a_pert = _perturb_log(a_best, 0.02, rng)
        elif i == 1:
            a_pert = _perturb_fourier_phase(a_best, rng, n_modes=8, max_amp=0.3)
        elif i == 2:
            a_pert = _perturb_structured(a_best, rng)
        elif i == 3:
            a_pert = _perturb_amplitude(a_best, rng, sigma=0.05)
        else:
            a_pert = _perturb_log(a_best, 0.04, rng)

        # Quick LSE refinement
        c_p = _full_convolve(a_pert, a_pert)
        beta_p = float(c_p.max()) * 0.02
        z_p = np.log(np.maximum(a_pert, 1e-30))
        z_p = z_p - z_p.mean()
        refine_dead = min(deadline - 15, time.monotonic() + 30)
        try:
            result = minimize(
                _lse_objective, z_p, args=(float(beta_p), n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-16,
                         'gtol': 1e-13, 'maxcor': 40, 'iprint': -1}
            )
            a_ref = _softmax(result.x)
            sc_ref = _score(a_ref)
            if sc_ref < best_sc:
                best_sc = sc_ref
                best_a = a_ref.copy()
        except Exception:
            pass

        # Newton polish on best
        if time.monotonic() < deadline - 10:
            nw_dead = min(deadline - 8, time.monotonic() + 15)
            a_nw, sc_nw = _newton_active_set(best_a, nw_dead, max_iter=50,
                                              tol_frac=0.002)
            if sc_nw < best_sc:
                best_sc = sc_nw
                best_a = a_nw.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining adaptive LSE with Nesterov acceleration,
    preconditioned Newton active-set, coarse-to-fine, and spectral diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    _best_seq = None
    _best_score_val = np.inf

    def upd(p, sc=None, label=""):
        nonlocal _best_seq, _best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score_val:
            _best_seq = p.copy()
            _best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init")

    n_main = n_parent if n_parent >= 4096 else 16384

    # === Phase 1: Coarse-to-fine (60s) ===
    print(f"\n[Phase 1] Coarse-to-fine (target n={n_main})")
    if time.monotonic() < deadline - 300:
        n_coarse = max(2048, n_main // 4)
        if n_coarse < len(_best_seq):
            a_up, sc_up = _coarse_to_fine(_best_seq, n_coarse, n_main,
                                          min(deadline - 250, t_start + 60))
            upd(a_up, sc_up, "coarse_fine")

    # === Phase 2: Main adaptive LSE (400s) ===
    print(f"\n[Phase 2] Main adaptive LSE")
    if time.monotonic() < deadline - 250:
        a_lse, sc_lse = _adaptive_lbfgs_lse(_best_seq.copy(), deadline, min_remaining=200)
        upd(a_lse, sc_lse, "lse_main")

    # === Phase 3: Nesterov LSE (150s) ===
    print(f"\n[Phase 3] Nesterov-accelerated LSE")
    if time.monotonic() < deadline - 150:
        nest_dead = min(deadline - 130, time.monotonic() + 150)
        a_ne, sc_ne = _nesterov_lse(_best_seq, nest_dead, max_steps=5000)
        upd(a_ne, sc_ne, "nesterov")

    # === Phase 4: Newton with various tolerances (150s) ===
    print(f"\n[Phase 4] Newton various tolerances")
    if time.monotonic() < deadline - 120:
        nw_dead = min(deadline - 100, time.monotonic() + 150)
        a_nw, sc_nw = _newton_various_tols(_best_seq, nw_dead)
        upd(a_nw, sc_nw, "newton_tols")

    # === Phase 5: Power-sum LSE (100s) ===
    print(f"\n[Phase 5] Power-sum LSE")
    if time.monotonic() < deadline - 80:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=60)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 6: Subgradient polish (100s) ===
    print(f"\n[Phase 6] Subgradient polish")
    if time.monotonic() < deadline - 70:
        sg_dead = min(deadline - 50, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=8000,
                                       lr_init=0.004, tol_frac=0.001)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 7: Spectral phase optimization (80s) ===
    print(f"\n[Phase 7] Spectral phase optimization")
    if time.monotonic() < deadline - 50:
        sp_dead = min(deadline - 40, time.monotonic() + 80)
        a_sp, sc_sp = _spectral_phase_opt(_best_seq, sp_dead, n_iters=300)
        upd(a_sp, sc_sp, "spectral")

    # === Phase 8: Newton + LSE alternating (100s) ===
    print(f"\n[Phase 8] Newton + LSE alternating")
    if time.monotonic() < deadline - 40:
        a_cur = _best_seq.copy()
        for round_i in range(4):
            if time.monotonic() > deadline - 30:
                break
            # Newton
            nw_d = min(deadline - 30, time.monotonic() + 20)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_d, max_iter=80,
                                                tol_frac=0.001, mu_base=1e-9)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")
            # Quick LSE
            if time.monotonic() < deadline - 15:
                a_lse2, sc_lse2 = _adaptive_lbfgs_lse(a_cur, deadline, min_remaining=10)
                if sc_lse2 < _score(a_cur):
                    a_cur = a_lse2
                upd(a_cur, _score(a_cur), f"alt_lse_{round_i}")

    # === Phase 9: Diversification (60s) ===
    print(f"\n[Phase 9] Diversification and refinement")
    if time.monotonic() < deadline - 20:
        a_div, sc_div = _diversify_and_refine(_best_seq, deadline - 10, rng,
                                              n_restarts=4)
        upd(a_div, sc_div, "diversify")

    # === Phase 10: Multi-n exploration (60s) ===
    print(f"\n[Phase 10] Multi-n exploration")
    n_try_list = [8192, 16384, 32768]
    if n_parent not in n_try_list and 4096 <= n_parent <= 65536:
        n_try_list.append(n_parent)
    n_try_list = sorted(set(n_try_list))

    for n_try in n_try_list:
        if time.monotonic() > deadline - 10:
            break
        if len(_best_seq) == n_try:
            a_init_n = _best_seq.copy()
        else:
            a_init_n = _resize_sequence(_best_seq, n_try)

        # Quick LSE
        if time.monotonic() < deadline - 15:
            a_opt, sc_opt = _adaptive_lbfgs_lse(a_init_n, deadline, min_remaining=8)
            upd(a_opt, sc_opt, f"n_{n_try}")

    # === Phase 11: Final aggressive polish (20s) ===
    print(f"\n[Phase 11] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 3:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        beta_final = mb * 0.0001
        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(float(beta_final), n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 80, 'maxls': 100,
                             'iprint': -1}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 1,
                                                  max_iter=100, tol_frac=0.0005,
                                                  mu_base=1e-10)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient
        if time.monotonic() < deadline - 1:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 0.5,
                                               n_steps=3000, lr_init=0.003,
                                               tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        # Verify with evaluate_sequence if available
        try:
            sc_verify = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_verify):
                sc_final = sc_verify
                print(f"VERIFIED score: {sc_final:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
