        # CG for H u = g_t
        x = np.zeros(n)
        r = g_t - _hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = _hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        # CG for H w = 1
        x2 = np.zeros(n)
        r2 = np.ones(n) - _hess_matvec(x2)
        p2 = r2.copy()
        rs2 = np.dot(r2, r2)
        for cg_it in range(200):
            Ap2 = _hess_matvec(p2)
            pAp2 = np.dot(p2, Ap2)
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            x2 += alpha2 * p2
            r2_new = r2 - alpha2 * Ap2
            rs2_new = np.dot(r2_new, r2_new)
            if np.sqrt(rs2_new) < 1e-12:
                break
            p2 = r2_new + (rs2_new / rs2) * p2
            r2 = r2_new
            rs2 = rs2_new

        sum_w = float(x2.sum())
        if abs(sum_w) < 1e-15:
            dx = -x
            dx = dx - np.mean(dx)
        else:
            lam = float(x.sum()) / sum_w
            dx = -x + lam * x2
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * t * gnorm or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-12:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = max(c_sorted[0] - c_sorted[30], max_c * 0.001)
    else:
        gap = max_c * 0.01
    base_beta = max(gap * 0.3, max_c * 0.003)

    if aggressive:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.0037, 0.00185, 0.0009, 0.00045, 0.00022, 0.00011,
                  0.00005, 0.000025, 0.00001, 0.000005]
    else:
        ratios = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007,
                  0.003, 0.001, 0.0005, 0.0002, 0.0001]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time = 0.002
        elif n <= 8192:
            est_time = 0.004
        elif n <= 16384:
            est_time = 0.008
        elif n <= 32768:
            est_time = 0.016
        else:
            est_time = 0.04

        max_iter = min(30000, max(50, int(remaining / (est_time * 6))))
        beta = base_beta * ratio

        prob = _LSProblem(beta, n)

        try:
            result = minimize(
                prob.fun,
                z,
                jac=prob.grad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_shaving(a_init, deadline, max_rounds=100):
    """Directly target top peaks with projected gradient steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        n_peaks = min(20, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]

        w = np.zeros(2 * n - 1)
        for k in top_peaks:
            w[k] = 1.0 / n_peaks

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)
        grad = grad - np.mean(grad)

        f_old = c_max
        step = 0.5 / (1.0 + 0.05 * rnd)
        improved = False
        for ls in range(30):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            break

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + t
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "exp_ramp_strong":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t)
        return a / a.sum()
    elif kind == "quadratic_ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + t ** 2
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau_asym":
        a = np.ones(n)
        a[n // 2:] *= 1.5
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimizer: coarse-to-fine L-BFGS with log-sum-exp,
    active-set Newton, peak shaving, and subgradient polish.
    Explores n from 1024 up to 65536.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "exp_ramp")
        update_best(parent, _score(parent), "init_exp_ramp")
        print(f"No parent, using exp_ramp n={n_parent}")

    # Phase 1: Coarse-to-fine multi-scale optimization
    print(f"\n[Phase 1] Multi-scale coarse-to-fine (t={time.monotonic()-t_start:.1f}s)")

    # Start small and go up
    n_coarse = min(n_parent, 2048)
    a_scale = _interpolate_to_n(parent, n_coarse)
    sc_scale = _score(a_scale)
    print(f"  n={n_coarse} init: {sc_scale:.10f}")

    # Optimize at coarse scale
    if time.monotonic() < deadline - 600:
        a_opt = _run_lbfgs(a_scale, deadline - 500, min_time_remaining=400.0, aggressive=True)
        sc_opt = _score(a_opt)
        print(f"  n={n_coarse} optimized: {sc_opt:.10f}")
        update_best(a_opt, sc_opt, f"coarse_{n_coarse}")

    # Scale up gradually
    scale_ns = [4096, 8192, n_parent]
    if n_parent < 32768:
        scale_ns.append(32768)
    if time.monotonic() < deadline - 200:
        scale_ns.append(65536)

    current_scale_a = a_opt
    for n_target in scale_ns:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - 120 - time.monotonic()
        if remaining < 30:
            break

        a_interp = _interpolate_to_n(current_scale_a, n_target)
        # Add a small perturbation to avoid getting stuck
        a_init = _perturb(a_interp, 0.01, rng)
        sc_init = _score(a_init)
        print(f"  Scaling to n={n_target}, init score: {sc_init:.10f}")

        remaining_for_opt = remaining - 30
        a_out = _run_lbfgs(a_init, t_start + t_start - t_start + deadline - 60,
                            min_time_remaining=max(30, remaining_for_opt * 0.3),
                            aggressive=True)
        sc_out = _score(a_out)
        print(f"  n={n_target} optimized: {sc_out:.10f}")
        update_best(a_out, sc_out, f"multiscale_{n_target}")

        # Quick peak shaving
        if time.monotonic() < deadline - 80 and sc_out < best_score + 0.01:
            a_ps, sc_ps = _peak_shaving(a_out, min(deadline - 60, time.monotonic() + 40), max_rounds=50)
            update_best(a_ps, sc_ps, f"peakshave_{n_target}")
            current_scale_a = a_ps if sc_ps < sc_out else a_out
        else:
            current_scale_a = a_out

    # Phase 2: Direct optimization from parent at parent n
    print(f"\n[Phase 2] Direct L-BFGS from parent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_direct = _run_lbfgs(parent, deadline - 200, min_time_remaining=200.0, aggressive=True)
        update_best(a_direct, _score(a_direct), "direct_parent")

    # Phase 3: Diverse initializations
    print(f"\n[Phase 3] Diverse initializations (t={time.monotonic()-t_start:.1f}s)")
    init_kinds = ["exp_ramp", "exp_ramp_strong", "quadratic_ramp", "ramp", "plateau_asym"]
    for kind in init_kinds:
        if time.monotonic() > deadline - 100:
            break
        a_init = _make_profile(n_parent, kind, rng)
        sc_init = _score(a_init)
        print(f"  {kind}: {sc_init:.10f}")
        if sc_init < best_score + 0.1:
            a_out = _run_lbfgs(a_init, deadline, min_time_remaining=50.0, aggressive=True)
            update_best(a_out, _score(a_out), f"init_{kind}")

    # Phase 4: Active-set Newton on best
    print(f"\n[Phase 4] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 30, time.monotonic() + 60),
                                         max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton")

        if time.monotonic() < deadline - 20:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 10, time.monotonic() + 30),
                                               max_iter=100, tol_active=0.0003)
            update_best(a_as2, sc_as2, "as_newton_tight")

    # Phase 5: Peak shaving on best
    print(f"\n[Phase 5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_ps, sc_ps = _peak_shaving(best_seq, min(deadline - 20, time.monotonic() + 40), max_rounds=80)
        update_best(a_ps, sc_ps, "peak_shave")

    # Phase 6: Subgradient polish
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 5, time.monotonic() + 30),
                                      n_max_steps=3000, lr_init=0.02)
        update_best(a_sg, sc_sg, "subgrad")

    # Phase 7: Perturbation restarts
    print(f"\n[Phase 7] Perturbation restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma in [0.03, 0.08, 0.15]:
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=5.0, aggressive=False)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"perturb_{sigma}")
            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 2, time.monotonic() + 10),
                                                 max_iter=80, tol_active=0.001)
                update_best(a_r2, sc_r2, f"perturb_{sigma}_as")

    # Phase 8: Final tight Newton
    print(f"\n[Phase 8] Final Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 2,
                                           max_iter=150, tol_active=0.0001)
        update_best(a_fin, sc_fin, "final_as")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
