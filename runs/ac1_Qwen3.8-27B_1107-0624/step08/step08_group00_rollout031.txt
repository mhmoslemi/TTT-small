def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_with_rev(w, a):
    """Compute h[j] = sum_i w[i]*a[i-j] for j=0..n-1 via FFT."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _conv_with_rev(w, a)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    current_a = a_init.copy()
    s = current_a.sum()
    if s < 1e-15:
        return current_a
    current_a = current_a / s
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001])

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale_factor = max(1.0, n / 4096.0)
        est_time_per_iter = 0.002 * scale_factor
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                    'maxfun': max_iter * 5,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _hankel_fixed_point(a_init, deadline, min_time_remaining=60.0):
    """Fixed-point iteration exploiting Hankel structure of KKT conditions.
    At optimum: conv_with_rev(p, a) is constant on support, where p is peak weights."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    eta = 0.05
    no_improve = 0
    max_no_improve = 300
    it = 0

    while time.monotonic() + min_time_remaining < deadline:
        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            eta = max(1e-5, eta * 0.5)
            if eta < 1e-6:
                break
            no_improve = 0
            if it > 20000:
                break

        # Compute peak weights (smoothed)
        beta_p = c_max * 0.02
        shifted = (c - c_max) / beta_p
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = e / s_e

        # Compute h = conv_with_rev(p, a)
        h = _conv_with_rev(p, a)
        h_mean = float(h.mean())
        residual = h - h_mean

        # Gradient of sum(residual^2) w.r.t. a:
        # d(h[j])/d(a[m]) = p[j+m] (for valid indices)
        # grad[m] = 2 * sum_j residual[j] * p[j+m]
        # This is 2 * conv_with_rev(residual, p) [with appropriate indexing]
        # Actually: sum_j residual[j] * p[j+m] = (residual_flipped * p)[m + n - 1]
        # Let's compute it via FFT
        r_padded = np.zeros(2 * n - 1)
        r_padded[:n] = residual
        # We want grad[m] = 2 * sum_{j=0}^{n-1} residual[j] * p[j+m]
        # = 2 * (cross-correlation of residual and p at lag m)
        # Using FFT: ifft(conj(FFT(residual)) * FFT(p))[n-1+m] ... 
        # Simpler: grad[m] = 2 * _conv_with_rev(residual, p)[:n] but p has length 2n-1
        # Let's just compute directly using convolution
        # sum_j residual[j] * p[j+m] for m=0..n-1
        # This is: conv(residual_flipped, p)[n-1+m] where residual_flipped = residual[::-1] padded
        # Actually let's use: it's the cross-correlation
        N_fft = _next_pow2(3 * n)
        R = np.fft.rfft(residual, N_fft)
        P = np.fft.rfft(p, N_fft)
        # cross-correlation: ifft(conj(R) * P)
        # (conj(R) * P)[k] corresponds to sum_j residual[j] * p[j+k] (with wrapping)
        # We need to handle the linear (non-circular) case with enough zero padding
        corr = np.fft.irfft(np.conj(R) * P, N_fft)
        # corr[m] = sum_j residual[j] * p[j+m] for m >= 0 (with zero padding)
        grad = 2.0 * corr[:n]

        # Update: a -= eta * grad, then project
        a_try = a - eta * grad
        a_try = np.maximum(a_try, 1e-15)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < current_sc - 1e-14:
            a = a_try
        else:
            # Try smaller step
            for mult in [0.5, 0.25, 0.1]:
                a_try2 = a - eta * mult * grad
                a_try2 = np.maximum(a_try2, 1e-15)
                s_try2 = a_try2.sum()
                if s_try2 < 1e-10:
                    continue
                a_try2 = a_try2 / s_try2
                sc_try2 = _score(a_try2)
                if sc_try2 < current_sc - 1e-14:
                    a = a_try2
                    break

        # Adapt eta
        if it % 100 == 0:
            c_now = _full_convolve(a, a)
            c_max_now = float(c_now.max())
            beta_p = c_max_now * 0.02
            if beta_p < c_max * 0.98:
                eta = min(0.1, eta * 1.1)

    return best_a, _score(best_a)


def _mirror_descent(a_init, deadline, min_time_remaining=60.0):
    """Nesterov-accelerated mirror descent in log-space (multiplicative updates)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Nesterov parameters
    t = 1.0
    a_prev = a.copy()
    a_y = a.copy()

    eta = 0.08
    eta_min = 1e-7
    no_improve = 0
    stall_count = 0
    max_stall = 400

    while time.monotonic() + min_time_remaining < deadline:
        # Evaluate at extrapolated point
        a_y = np.maximum(a_y, 1e-30)
        a_y = a_y / a_y.sum()
        c_y = _full_convolve(a_y, a_y)
        c_max = float(c_y.max())

        if time.monotonic() % 50 > 49:
            pass  # just for clarity

        # Smoothed subgradient with top-K peaks
        K = max(3, int(n * 0.002))
        idx_sorted = np.argsort(c_y)[::-1][:K]
        c_top = c_y[idx_sorted]
        c_peak = c_top[0]
        beta_w = max(c_peak * 0.005, 1e-10)
        weights = np.exp((c_top - c_peak) / beta_w)
        weights = weights / weights.sum()

        w = np.zeros(len(c_y))
        w[idx_sorted] = weights
        h = _conv_with_rev(w, a_y)
        W = float(np.dot(w, c_y))
        g = 4.0 * n * (h - W)

        # Center for softmax parameterization
        centered = g - np.dot(a_y, g) / a_y.sum()

        # Multiplicative update
        log_update = -eta * centered
        log_update = np.clip(log_update, -30, 30)
        a_new = a_y * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                eta = max(eta_min, eta * 0.85)
            if no_improve > max_stall:
                stall_count += 1
                if stall_count > 10:
                    break
                # Restart with perturbation
                noise = 0.01 * np.random.default_rng(no_improve).standard_normal(n)
                a = _softmax(np.log(np.maximum(a, 1e-30)) + noise)
                a_y = a.copy()
                a_prev = a.copy()
                t = 1.0
                eta = 0.08
                no_improve = 0
                continue

        a = a_new

        # Nesterov extrapolation in log-space
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        a_prev = a_y.copy()
        # Extrapolate in log-space
        log_a_new = np.log(np.maximum(a, 1e-30))
        log_a_prev = np.log(np.maximum(a_prev, 1e-30))
        log_a_y = log_a_new + theta * (log_a_new - log_a_prev)
        a_y = _softmax(log_a_y)
        t = t_new

    return best_a, best_sc


def _coord_descent(a_init, deadline, min_time_remaining=30.0):
    """Greedy coordinate descent: update coordinates that most reduce the peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    it = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        S = a.sum()

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break

        k_star = int(np.argmax(c))

        # Find top contributing coordinates to the peak
        contrib = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            kj = k_star - j
            if 0 <= kj < n:
                contrib[j] = a[kj]

        # Also consider second and third peaks
        top_peak_idx = np.argsort(c)[::-1][:3]
        for kp in top_peak_idx[1:]:
            j_lo2 = max(0, kp - (n - 1))
            j_hi2 = min(n - 1, kp)
            for j in range(j_lo2, j_hi2 + 1):
                kj2 = kp - j
                if 0 <= kj2 < n:
                    contrib[j] += 0.3 * a[kj2]

        n_update = max(10, int(n * 0.005))
        top_idx = np.argsort(contrib)[::-1][:n_update]

        improved = False
        for j in top_idx:
            if time.monotonic() + min_time_remaining > deadline:
                break
            aj = a[j]
            if aj < 1e-12:
                continue
            S_other = S - aj

            # Try different values of t
            best_t = aj
            best_f = current_sc

            # The optimal t balances the reduction in c[k_star] with the increase in other peaks
            # c_new[k_star] = c[k_star] + 2*(t-aj)*a[k_star-j] (if k_star != 2j)
            # We want c_new[k_star] to drop, so t < aj when a
