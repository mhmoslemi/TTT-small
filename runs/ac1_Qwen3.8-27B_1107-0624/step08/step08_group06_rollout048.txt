Maintain a weight vector $w$ over the convolution positions. The objective to minimize is:
$F(a) = 2n \cdot \max_k (a*a)_k$

Use a weighted version: $F_w(a) = 2n \cdot \sum_k w_k (a*a)_k$ where $w$ is concentrated on the peaks.

The gradient of $F_w$ is: $\nabla F_w = 4n \sum_k w_k P_k a = 4n P_w a$ where $P_w = \sum_k w_k P_k$.

If we use the "subgradient" of the max (i.e., $w$ is a point in the subdifferential), then we're doing subgradient descent. But we can do better:

This is like the "mirror descent" method with the negative entropy as the mirror.

"""
Optimizer using accelerated multiplicative gradient descent with
adaptive log-sum-exp smoothing, multi-n exploration, and peak
clustering for final polish.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute the gradient-related correlation: sum_k A_vec[k] * a[k-j] for valid k-j."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax_vec(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        h = n // 2
        a[:h//2] = 1.0
        a[3*h//2:] = 1.0

        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif kind == "block3":
        a = np.zeros(n)
        b = n // 6
        a[:b] = 1.0
        a[2*b:3*b] = 1.0
        a[5*b:] = 1.0

    elif kind == "random":

        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    return a_new / s


def _multiplicative_grad(a, beta, n, eta=1.0):
    """One step of multiplicative gradient descent on log-sum-exp smoothed objective."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Gradient of the smoothed max: (n/beta) * sum_k w_k * 2*a[k-j]
    g = (2.0 * n / beta) * _corr_a_A(a, w, n)
    
    # Multiplicative update
    g_max = np.max(np.abs(g))
    if g_max < 1e-30:
        return a
    log_a = np.log(np.maximum(a, 1e-300))
    log_a = log_a - eta * g / g_max
    a_new = np.exp(np.clip(log_a, -700, 700))
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new / s_new

            break

def _lsobj(z, beta, n):
    a = _softmax_vec(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax_vec(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _peak_shaving(a, deadline, n_rounds=50):
    """Iteratively shave the highest peaks by adjusting contributing elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        n_peaks = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        a_new = a.copy()
        for k_pos in top_idx:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            # Find the pair with largest product
            prods = np.array([a_new[i] * a_new[k_pos - i] for i in range(i_lo, i_hi + 1)])
            max_i = i_lo + int(np.argmax(prods))
            j = k_pos - max_i
            if a_new[max_i] >= a_new[j]:
                a_new[max_i] *= (1.0 - 0.01 / (rnd + 1))
            else:
                a_new[j] *= (1.0 - 0.01 / (rnd + 1))
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            break
    
    return best_a

def _cg_solve(matvec, rhs, max_cg=200, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- Accelerated projected gradient for coarse optimization
- L-BFGS with adaptive log-sum-exp smoothing for fine optimization
- Active-set Newton for peak-specific polish
- Multi-n exploration with diverse profiles
- Final very-small-beta L-BFGS polish

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax_vec(z):

    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos":
        t = np.linspace(0, 1, n)

        a = np.maximum(a, 1e-10)

    elif kind == "edge_heavy":

    elif kind == "random":
