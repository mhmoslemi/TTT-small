def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)

def _softmax(z):

    if s < 1e-300:

def _activate_weights(c, n_active_target=50, tol_frac=0.001):
    """Compute weights for active set lags based on proximity to max."""

def _hankel_matvec(v, W, n):
    """Compute H*v where H is the Hankel/Toeplitz matrix defined by W.
    Uses the structure: (H*v)[i] = sum_k W[k] * (v[i-(k-(n-1))] + v[i+(k-(n-1))])
    which is equivalent to 2 * (reversed conv) in the middle.
    """
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Conjugate gradient solve for H*x = rhs using FFT-based matvec."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            # Damp the step
            alpha *= 0.5
            x_new = x + alpha * p
            r_new = r - alpha * Hp
            rs_new = float(r_new @ r_new)
            if not np.isfinite(rs_new):
                break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_set(a_init, deadline, max_rounds=200, verbose=False):
    """Active-set Newton method with dynamic active set and LM regularization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set: start tight, expand if needed
        active_expanded = False
        for tol_frac in [0.0001, 0.0005, 0.001, 0.005, 0.01, 0.05]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 2:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Limit active set size for efficiency
        active_idx = np.where(active)[0]
        n_act = len(active_idx)
        max_active = min(200, n_act)
        if n_act > max_active:
            c_active = c[active_idx]
            thresh = np.sort(c_active)[::-1][max_active - 1]
            keep = c_active >= thresh - 1e-30
            active_idx = active_idx[keep]
            active = np.zeros(len(c), dtype=bool)
            active[active_idx] = True
        
        # Compute weights (uniform for active set)
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()
        
        # Gradient: g[j] = 2 * sum_k W[k] * (a[j-(k-n+1)] + a[j+(k-n+1)])
        # This is _hankel_matvec(a, W, n)
        g = _hankel_matvec(a, W, n)
        
        # Project to tangent space of simplex (sum dx = 0)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break
        
        # Newton step: solve H * dx = -g_t (with LM regularization)
        # First try without regularization
        Hg = _cg_solve(W, n, -g_t, max_iter=500, tol=1e-14)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500, tol=1e-14)
        H1_norm = float(np.linalg.norm(H1))
        
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            # Hessian is singular, use gradient step
            dx = g_t / gn  # positive step direction (minimize -g_t)
            # Actually we want dx = -g_t / ||g_t|| for descent
            dx = -g_t / gn
        else:
            # Enforce sum(dx) = 0
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = Hg - lam * H1
            dx = dx - np.mean(dx)
            
            # Check if Newton step is a descent direction
            if float(dx @ g_t) > 0:
                # Not descent, use gradient direction
                dx = -g_t / gn
        
        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-30 or not np.isfinite(dx_norm):
            break
        
        # Line search with Wolfe conditions
        alpha = 1.0
        step_taken = False
        c1 = 1e-4  # sufficient decrease
        c2 = 0.9    # curvature
        
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            
            # Check sufficient decrease
            if c_max_try <= c_max - c1 * alpha * float(np.abs(g_t @ -g_t) / gn) - 1e-15:
                step_taken = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break
        
        if not step_taken:
            # Try smaller fixed steps
            for trial_alpha in [0.1, 0.05, 0.01, 0.005, 0.001]:
                a_try = a + trial_alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_taken = True
                    break
        
        if not step_taken:
            # Gradient step fallback
            grad_dir = -g_t / max(gn, 1e-30)
            for trial_alpha in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a + trial_alpha * grad_dir
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_taken = True
                    break
        
        if step_taken:
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 10:
            break
        
        # Convergence check
        if abs(c_max - _full_convolve(a, a).max()) < 1e-16:
            break
    
    return best_a, best_sc


def _smooth_obj(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    """Gradient of the smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: 4*n*a*(h - W_val) where h = conv(w, a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_smoothed_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS on the log-sum-exp smoothed objective with adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    
    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations based on remaining time
        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.002 * scale
        max_iter = min(10000, max(50, int(remaining / max(est_time_per_iter * 20, 0.1))))
        
        try:
            def obj_fn(x):
                return _smooth_obj(x, beta, n)
            
            def grad_fn(x):
                return _smooth_grad(x, beta, n)
            
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]
        
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc
        
        if new_gap < 1e-7 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.15
        else:
            beta *= 0.3
        
        beta = min(beta, c_max_new * 0.001)
    
    return current_a


def _run_cycle(a_init, deadline, min_time_remaining=100.0):
    """Cycle between L-BFGS and Newton phases."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    n_cycles = 8
    cycle_time = max(10.0, (deadline - min_time_remaining - time.monotonic()) / n_cycles)
    
    for cyc in range(n_cycles):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # L-BFGS phase
        lbfgs_end = time.monotonic() + cycle_time * 0.5
        if lbfgs_end < deadline - min_time_remaining:
            a_lb = _run_smoothed_lbfgs(current_a, min(lbfgs_end, deadline - min_time_remaining),
                                        min_time_remaining=50.0)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb
        
        # Newton phase
        newton_end = time.monotonic() + cycle_time * 0.5
        if newton_end < deadline - min_time_remaining:
            a_np, sc_np = _newton_active_set(
                current_a, min(newton_end, deadline - min_time_remaining),
                max_rounds=100
            )
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np
        
        # Gradually increase cycle length
        cycle_time *= 1.3
        if cycle_time > 180:
            cycle_time = 180
    
    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.9
        a = np.maximum(a, 1e-10)
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.05
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 1.5 + 0.05
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.01
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
    elif kind == "asym_shift":
        # Mass shifted to one side - exploits non-symmetry
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.25) ** 2) + 0.02
    elif kind == "asym_shift2":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.35) ** 2) + 0.02
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _coordinate_descent_polish(a_init, deadline, max_sweeps=50):
    """Coordinate descent on individual elements for final polish."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Try small perturbations on random subsets
        rng_local = np.random.default_rng(sweep * 7919)
        n_try = min(n, 1000)
        idx = rng_local.choice(n, size=n_try, replace=False)
        
        for i in idx:
            if time.monotonic() > deadline:
                break
            # Try a few values for this coordinate
            orig = a[i]
            for factor in [0.5, 0.8, 1.2, 1.5, 0.7, 1.3]:
                a[i] = orig * factor
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = orig
                    continue
                a_norm = a / s_new
                sc = _score(a_norm)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_norm.copy()
                    improved = True
                    break
            a[i] = orig
        
        if not improved and sweep > 5:
            break
        a = best_a.copy()
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search function: finds best sequence minimizing 2n*max(conv)/sum^2."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, "parent")
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, "init")
    
    # Phase 1: Quick diverse initialization + short L-BFGS (budget: 200s)
    print(f"\n[Phase 1] Initialization & quick optimization (t={time.monotonic()-t_start:.1f}s)")
    phase1_end = t_start + 200.0
    
    profiles = ["tapered", "asym", "left_heavy", "right_heavy", "bimodal", 
                "asym_shift", "asym_shift2", "exp_decay"]
    best_init = parent.copy()
    best_init_sc = _score(parent)
    
    for prof in profiles:
        if time.monotonic() > phase1_end - 100:
            break
        a_p = _make_profile(n_parent, prof)
        sc_p = _score(a_p)
        if sc_p < best_init_sc:
            best_init_sc = sc_p
            best_init = a_p.copy()
            print(f"  Profile '{prof}': {sc_p:.8f}")
    
    # Perturbed parent
    for sigma in [0.03, 0.1]:
        if time.monotonic() > phase1_end - 80:
            break
        a_p = _perturb(parent, sigma, rng)
        sc_p = _score(a_p)
        if sc_p < best_init_sc:
            best_init_sc = sc_p
            best_init = a_p.copy()
    
    # Quick L-BFGS on best init
    if time.monotonic() < phase1_end - 30:
        a_q = _run_smoothed_lbfgs(best_init, phase1_end - 20, min_time_remaining=200)
        update_best(a_q, "quick_lbfgs")
    
    # Phase 2: Main cycle optimization (budget: 800s)
    print(f"\n[Phase 2] Main cycle optimization (t={time.monotonic()-t_start:.1f}s)")
    phase2_end = t_start + 1000.0
    
    start_a = best_seq if best_seq is not None else best_init
    a_main = _run_cycle(start_a, phase2_end, min_time_remaining=500.0)
    update_best(a_main, "cycle_main")
    
    # Phase 3: Additional Newton polish iterations (budget: 300s)
    print(f"\n[Phase 3] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    phase3_end = t_start + 1300.0
    
    if best_seq is not None:
        a_np, sc_np = _newton_active_set(best_seq, phase3_end, max_rounds=300)
        update_best(a_np, "newton_deep")
        
        # Another cycle from the Newton result
        if time.monotonic() < phase3_end - 30:
            a_c2 = _run_cycle(best_seq, phase3_end, min_time_remaining=300)
            update_best(a_c2, "cycle2")
    
    # Phase 4: Multi-n exploration (budget: 300s)
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    phase4_end = deadline - 50.0
    
    n_list = [n_parent * 2, n_parent // 2, n_parent * 4]
    n_list = [n for n in n_list if 1024 <= n <= 81920]
    # Also try the original n with a different init
    n_list = [n_parent] + [n for n in n_list if n != n_parent]
    
    for n_try in n_list:
        if time.monotonic() > phase4_end - 20:
            break
        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        
        # Short L-BFGS
        a_opt_n = _run_smoothed_lbfgs(a_init_n, min(phase4_end - 10, time.monotonic() + 80),
                                       min_time_remaining=50)
        update_best(a_opt_n, f"n_{n.try}")
        
        # Newton polish if close
        sc_n = _score(a_opt_n)
        if sc_n < best_score + 0.002 and time.monotonic() < phase4_end - 10:
            a_np_n, sc_np_n = _newton_active_set(a_opt_n, min(phase4_end, time.monotonic() + 40),
                                                  max_rounds=50)
            update_best(a_np_n, f"n_{n_try}_np")
    
    # Phase 5: Final ultra-fine polish (last 50s)
    print(f"\n[Phase 5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Deep Newton
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 2, max_rounds=200)
        update_best(a_fin, "final_newton")
        
        # One more short L-BFGS at very small beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-11
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                try:
                    def obj_fin(x):
                        return _smooth_obj(x, beta_fin, n)
                    def grad_fin(x):
                        return _smooth_grad(x, beta_fin, n)
                    result = minimize(
                        obj_fin, z, jac=grad_fin,
                        method='L-BFGS-B',
                        options={'maxiter': min(3000, int(remaining / 0.01)),
                                 'ftol': 1e-20, 'gtol': 1e-18,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, "final_lbfgs")
                except Exception:
                    pass
        
        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 1:
            a_vf, sc_vf = _newton_active_set(best_seq, deadline - 0.5, max_rounds=50)
            update_best(a_vf, "very_final")
    
    # Final validation and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Scale to a reasonable magnitude (avoid very small or very large values)
        scale = 1.0 / s
        final = final * scale
        # Actually keep original scale since score is scale-invariant
        return [float(x) for x in final]
    
    # Fallback
    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
