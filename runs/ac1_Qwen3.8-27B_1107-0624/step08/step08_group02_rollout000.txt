"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
- Golden-angle and Chebyshev node structured initializations
- Binary splitting refinement to escape local minima
- Zernike-like orthogonal polynomial blending

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0.0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_soft_max(x, beta):
    """Compute log-sum-exp based smooth max: max(x) + beta*log(sum(exp((x-max)/beta)))"""
    x_max = float(x.max())
    shifted = (x - x_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return x_max
    return x_max + beta * np.log(s)


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=120):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Multi-resolution active set selection
        best_active = None
        best_n_active = 0
        for tol_frac in [0.000005, 0.00002, 0.0001, 0.0003, 0.001, 0.005]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 1:
                best_active = active
                best_n_active = n_active
                break
        if best_active is None:
            best_active = np.zeros(2 * n - 1, dtype=bool)
            best_active[int(np.argmax(c))] = True
            best_n_active = 1

        W = np.zeros(2 * n - 1)
        W[best_active] = 1.0 / best_n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / max(float(np.ones(n) @ H1), 1e-30)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Backtracking line search with Armijo condition
        alpha = 1.0
        step_done = False
        c_slope = c_max
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_slope - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            # Try a random perturbation escape
            rng = np.random.default_rng(rnd)
            a_pert = a + 0.01 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a_pert = a_pert / s_pert
                sc_pert = _score(a_pert)
                if sc_pert < best_sc:
                    a = a_pert
                    best_a = a.copy()
                    best_sc = sc_pert
                    no_improve = 0
                else:
                    break

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap * 0.15, max_c * 0.0005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    stagnation = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 4096.0)
        est_time_per_iter = 0.002 * scale
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 3:
                    beta *= 0.05
                    continue
        except Exception:
            beta *= 0.3
            continue

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        idx2 = min(30, len(c_sorted_new) - 1)
        new_gap = c_sorted_new[0] - c_sorted_new[idx2]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4

        beta = min(beta, max(c_max_new * 1e-6, 1e-15))

    return current_a


def _split_refine(a_init, deadline, min_time_remaining=60.0):
    """Split the sequence into two halves, optimize each, then recombine."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    if n < 64:
        return current_a

    # Binary splitting: try splitting at different ratios
    split_ratios = [0.5, 0.4, 0.6, 0.3, 0.7]
    best_a = current_a.copy()
    best_sc = current_sc

    for ratio in split_ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n1 = int(n * ratio)
        n2 = n - n1
        if n1 < 16 or n2 < 16:
            continue

        # Extract halves
        h1 = current_a[:n1].copy()
        h2 = current_a[n1:].copy()

        # Optimize each half independently (with full convolution of each)
        time_per_half = min(60.0, (deadline - min_time_remaining - time.monotonic()) / (len(split_ratios) * 2))
        if time_per_half < 5:
            continue

        h1_opt = _run_adaptive_lbfgs(h1, time.monotonic() + time_per_half, min_time_remaining=5.0)
        sc1 = _score(h1_opt)
        if sc1 < _score(h1):
            h1 = h1_opt

        time_per_half2 = min(60.0, (deadline - min_time_remaining - time.monotonic()) / (len(split_ratios)))
        if time_per_half2 < 5:
            continue

        h2_opt = _run_adaptive_lbfgs(h2, time.monotonic() + time_per_half2, min_time_remaining=5.0)
        sc2 = _score(h2_opt)
        if sc2 < _score(h2):
            h2 = h2_opt

        # Recombine with different blending strategies
        for blend in [0.5, 0.3, 0.7]:
            combined = np.concatenate([h1, h2 * blend])
            combined = np.maximum(combined, 0)
            s = combined.sum()
            if s < 1e-15:
                continue
            combined = combined / s
            sc_comb = _score(combined)
            if sc_comb < best_sc:
                best_sc = sc_comb
                best_a = combined.copy()

    # Final full re-optimization
    if best_a is not current_a and time.monotonic() + min_time_remaining < deadline:
        final = _run_adaptive_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
        sc_final = _score(final)
        if sc_final < best_sc:
            best_a = final
            best_sc = sc_final

    return best_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.5) ** 0.7
        a = np.maximum(a, 1e-10)
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.01
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.02
    elif kind == "triamodal":
        t = np.linspace(0, 1, n)
        a = (np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.5) ** 2) +
             np.exp(-20.0 * (t - 0.75) ** 2) + 0.01)
    elif kind == "gaussian_center":
        t = np.linspace(-1, 1, n)
        a = np.exp(-4.0 * t ** 2) + 0.005
    elif kind == "chebyshev":
        k = np.arange(n)
        t = np.cos(np.pi * (2 * k + 1) / (2 * n))
        a = np.sqrt(1 - t ** 2) + 0.01
    elif kind == "golden_angle":
        k = np.arange(n)
        angle = 2.399963229728653 * k
        t = k / max(n - 1, 1)
        a = 0.5 + 0.5 * np.cos(angle) * np.sin(np.pi * t) + 0.1
        a = np.maximum(a, 1e-8)
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.02
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 1.5 + 0.02
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where((t > 0.2) & (t < 0.8), 1.0, 0.3)
        a = a + 0.05 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 1e-8)
    else:
        a = np.ones(n)
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _orthogonal_blend(n, coeffs, base_profile="tapered"):
    """Create a profile by blending orthogonal polynomial modes."""
    t = np.linspace(0, 1, n)
    a = _make_profile(n, base_profile).copy()

    # Add Legendre polynomial modes mapped to [0,1]
    x = 2 * t - 1
    for k, c in enumerate(coeffs):
        if k == 0:
            a += c * np.ones(n)
        elif k == 1:
            a += c * x
        elif k == 2:
            a += c * (0.5 * (3 * x ** 2 - 1))
        elif k == 3:
            a += c * (0.5 * (5 * x ** 3 - 3 * x))
        elif k == 4:
            a += c * (0.125 * (35 * x ** 4 - 30 * x ** 2 + 3))
        else:
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return _make_profile(n, base_profile)
    return a / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.0003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.05)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.015

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()
    return best_a, best_sc


def _projective_phase(a_init, deadline, min_time_remaining=30.0):
    """Optimize in projective space using a different parameterization: a_i = x_i^2 / sum(x_j^2)."""
    n = len(a_init)
    x_init = np.sqrt(np.maximum(a_init, 1e-30))
    x_init = x_init / x_init.sum()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    def proj_obj(x):
        x = np.abs(x)
        s2 = np.dot(x, x)
        if s2 < 1e-30:
            return 2.0 * n
        a = x * x / s2
        c = _full_convolve(a, a)
        return 2.0 * n * c.max()

    # Use Adam-like manual gradient descent
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01

    for i in range(2000):
        if time.monotonic() > deadline - min_time_remaining:
            break
        x = np.abs(x_init)
        s2 = np.dot(x, x)
        if s2 < 1e-30:
            break
        a = x * x / s2
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use active set gradient
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient of 2n*max(c) w.r.t. x_i
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))

        # da/dx_i = (2*x_i*s2 - x_i^2*2*x_i) / s2^2 = 2*x_i*(s2 - x_i^2)/s2^2
        da_dx = 2.0 * x * (s2 - x * x) / (s2 * s2)
        grad_x = 4.0 * n * da_dx * (h - W_val)

        grad_norm = float(np.linalg.norm(grad_x))
        if grad_norm < 1e-15:
            break

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x * grad_x
        m_hat = m / (1 - beta1 ** (i + 1))
        v_hat = v / (1 - beta2 ** (i + 1))
        x_init = x_init - lr * m_hat / (np.sqrt(v_hat) + eps)

        if i % 25 == 24:
            a_check = _softmax(np.log(np.maximum(x_init ** 2, 1e-30)))
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.1, 0.05)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Diverse initialization search ===
    print(f"\n[Phase 1] Diverse initialization (t={time.monotonic() - t_start:.1f}s)")
    n_try = n_parent

    profiles = ["tapered", "asym_taper", "gaussian_center", "chebyshev", "golden_angle",
                "bimodal", "triamodal", "plateau", "left_heavy", "right_heavy", "cos"]

    best_init = parent.copy()
    best_init_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 1500:
            break
        a_p = _make_profile(n_try, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_init_sc:
            best_init_sc = sc_p
            best_init = a_p
            print(f"  Profile '{prof_kind}': score={sc_p:.10f}")

    # Try orthogonal blends
    blend_configs = [
        [0.1, 0.05, 0.02],
        [0.1, -0.05, 0.03],
        [0.05, 0.1, -0.02],
        [-0.05, 0.08, 0.04],
        [0.08, 0.06, -0.03, 0.02],
    ]
    for cfg in blend_configs:
        if time.monotonic() > deadline - 1400:
            break
        a_blend = _orthogonal_blend(n_try, cfg, "tapered")
        sc_blend = _score(a_blend)
        if sc_blend < best_init_sc:
            best_init_sc = sc_blend
            best_init = a_blend
            print(f"  Blend {cfg}: score={sc_blend:.10f}")

    # Perturb parent
    for sigma in [0.03, 0.08, 0.15, 0.25]:
        if time.monotonic() > deadline - 1300:
            break
        a_pert = _perturb(parent, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_init_sc:
            best_init_sc = sc_pert
            best_init = a_pert

    update_best(best_init, best_init_sc, "best_init_phase1")

    # === Phase 2: Quick L-BFGS on top candidates ===
    print(f"\n[Phase 2] Quick L-BFGS (t={time.monotonic() - t_start:.1f}s}")
    candidates = [best_init, parent]
    # Add a few diverse perturbations
    for sigma in [0.05, 0.12]:
        candidates.append(_perturb(best_init, sigma, rng))

    for i, cand in enumerate(candidates[:4]):
        if time.monotonic() > deadline - 1100:
            break
        t_end = min(deadline - 1100, time.monotonic() + 250)
        if t_end - time.monotonic() < 30:
            break
        a_quick = _run_adaptive_lbfgs(cand, t_end, min_time_remaining=100.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, f"lbfgs_quick_{i}")

    # === Phase 3: Main deep L-BFGS optimization ===
    print(f"\n[Phase 3] Main deep L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_main = _run_adaptive_lbfgs(best_seq, deadline, min_time_remaining=500.0)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 4: Newton active set polishing ===
    print(f"\n[Phase 4] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 200),
                                             max_rounds=100)
        update_best(a_np, sc_np, "newton_1")

    # === Phase 5: Projective phase optimization ===
    print(f"\n[Phase 5] Projective phase (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_proj, sc_proj = _projective_phase(best_seq, deadline, min_time_remaining=350.0)
        update_best(a_proj, sc_proj, "projective")
        if sc_proj < best_score + 0.001:
            a_np2, sc_np2 = _newton_active_polish(a_proj, min(deadline - 320, time.monotonic() + 60),
                                                   max_rounds=50)
            update_best(a_np2, sc_np2, "projective_newton")

    # === Phase 6: Split and refine ===
    print(f"\n[Phase 6] Split refine (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_split = _split_refine(best_seq, deadline, min_time_remaining=250.0)
        update_best(a_split, _score(a_split), "split_refine")

    # === Phase 7: Subgradient diversification ===
    print(f"\n[Phase 7] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 150, time.monotonic() + 100))
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.002:
            a_np3, sc_np3 = _newton_active_polish(a_sg, min(deadline - 120, time.monotonic() + 50),
                                                   max_rounds=60)
            update_best(a_np3, sc_np3, "subgrad_newton")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_options = [16384, 24576, 32768, 8192]
    # Filter out the current n
    current_n = len(best_seq) if best_seq is not None else n_parent
    n_options = [x for x in n_options if x != current_n]

    for n_new in n_options:
        if time.monotonic() > deadline - 60:
            break
        if time.monotonic() < deadline - 40:
            a_res = _resample(best_seq, n_new)
            a_opt = _run_adaptive_lbfgs(a_res, deadline, min_time_remaining=40.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_new}")
            if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 20:
                a_np4, sc_np4 = _newton_active_polish(a_opt, deadline - 10, max_rounds=40)
                update_best(a_np4, sc_np4, f"n_{n_new}_np")

    # === Phase 9: Diverse restarts ===
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_kinds = ["gaussian_center", "chebyshev", "triamodal", "plateau"]
        for i, rk in enumerate(restart_kinds):
            if time.monotonic() > deadline - 20:
                break
            if i < 2:
                p_start = _perturb(best_seq, 0.04 + 0.04 * i, rng)
            else:
                p_start = _make_profile(len(best_seq), rk)

            t_end = min(deadline - 15, time.monotonic() + 40)
            if t_end - time.monotonic() < 10:
                break
            a_r = _run_adaptive_lbfgs(p_start, t_end, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{rk}")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        a_final, sc_final = _newton_active_polish(best_seq, deadline - 1, max_rounds=200)
        update_best(a_final, sc_final, "final_newton")

        if time.monotonic() < deadline - 2:
            c_best = _full_convolve(a_final, a_final)
            mb = float(c_best.max())
            z = np.log(np.maximum(a_final, 1e-30))
            z = z - z.mean()
            beta_fin = max(mb * 1e-9, 1e-15)
            remaining = max(deadline - 1 - time.monotonic(), 0.5)
            if remaining > 1:
                max_iter = min(3000, max(30, int(remaining / 0.02)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_ultra = _softmax(result.x)
                    sc_ultra = _score(a_ultra)
                    update_best(a_ultra, sc_ultra, "final_ultra_lbfgs")
                except Exception:
                    pass

        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_veryfin, sc_veryfin = _newton_active_polish(best_seq, deadline - 0.1, max_rounds=80)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
