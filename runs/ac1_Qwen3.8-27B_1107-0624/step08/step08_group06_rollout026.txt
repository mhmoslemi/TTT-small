"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing and adaptive beta continuation,
followed by active-set Newton polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec_fn, rhs, max_cg=200, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

    if s < 1e-15:

    a = a / s

    for it in range(max_iter):

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS with log-sum-exp smoothing and aggressive adaptive beta continuation
2. Active-set Newton with conjugate gradient Hessian solve
3. Deflation steps targeting top convolution peaks
4. Multi-n exploration (8192, 16384, 32768)
5. Final ultra-fine beta polish

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _cg_solve_hess(v, A_vec, m, mu, n):
    return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v


def _cg_solve(rhs, A_vec, m, mu, n, max_cg=200, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - _cg_solve_hess(x, A_vec, m, mu, n)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = _cg_solve_hess(p, A_vec, m, mu, n)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _active_set_newton(a_init, deadline, max_iter=300, tol_list=None):
    if tol_list is None:
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None
        best_dir_score = np.inf
        best_t = 1.0

        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2:
                continue
            if m > 4 * n:
                top_idx = np.argsort(c)[::-1][:4 * n]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                m = int(active.sum())

            A_vec = active.astype(np.float64)
            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            mu = 1e-7 * (c_max / max(m, 1)) + 1e-9

            u = _cg_solve(g_t, A_vec, m, mu, n, max_cg=200)
            w = _cg_solve(np.ones(n), A_vec, m, mu, n, max_cg=200)
            w_sum = float(w.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w

            dx = dx - np.mean(dx)

            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)
            t = min(1.0, t_max)

            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)

            if sc_try < best_dir_score:
                best_dir_score = sc_try
                best_dir = dx.copy()
                best_t = t

        if best_dir is None:
            active = c >= (c_max - 0.01 * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            A_vec = active.astype(np.float64)
            g = _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            t = 0.05
            accepted = False
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-13:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break
        else:
            f_old = c_max
            accepted = False
            t = best_t
            for ls in range(50):
                a_try = a + t * best_dir
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try <= f_old - 1e-12 * t or ls >= 40:
                    if f_try <= f_old + 1e-10:
                        a = a_try
                        accepted = True
                        break
                t *= 0.5
            if not accepted:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _deflation(a_init, deadline, n_attempts=30):
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())

        n_top = max(5, min(len(c), int(0.03 * len(c))))
        top_positions = np.argsort(c)[::-1][:n_top]

        a_new = best_a.copy()
        for k_pos in top_positions:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = -1.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= 0.997
                else:
                    a_new[j] *= 0.997

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, label=""):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.05)

    ratios = [1.0, 0.6, 0.4, 0.25, 0.15, 0.09, 0.05, 0.03,
              0.018, 0.01, 0.006, 0.0035, 0.002, 0.0012,
              0.0007, 0.0004, 0.0002, 0.0001, 0.00005, 0.00002]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        weight = 1.0 / (1.0 + i * 0.2)
        est_time_per_iter = 0.006 * max(1.0, n / 8192.0)
        max_iter = int(remaining * weight / (est_time_per_iter * 10))
        max_iter = max(50, min(max_iter, 12000))

        try:
            result = minimize(
                _lsobj, z,
                jac=_lsgrad,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if (i + 1) % 5 == 0:
                    print(f"    [{label}] beta_idx={i}, sc={current_sc:.10f}, t={time.monotonic()-t_global:.1f}s")
        except Exception:
            pass

    return current_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    else:
        return np.ones(n) / n

                        args=(beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # One more deflation
        if time.monotonic() < deadline - 2:
            a_def3, sc_def3 = _deflation(best_seq, min(deadline - 1, time.monotonic() + 15), n_attempts=15)
            update_best(a_def3, sc_def3, "final_def")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
