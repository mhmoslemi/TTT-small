    # h[j] = sum_{k: 0<=k-j<n} W[k] * a[k-j]
    # = sum_{i: 0<=i<n, 0<=k<n where k=j+i} W[j+i] * a[i]

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)

    result = np.fft.irfft(fa * fA, N)[:out_len]

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, max_beta_skip=0):
    """L-BFGS with adaptive beta continuation. Returns best (a, score)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

                                  0.012, 0.008, 0.005, 0.003, 0.002,
                                  0.0012, 0.0007, 0.0004, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    best_a = current_a.copy()
    best_sc = current_sc

    for i, beta in enumerate(betas):
        if i < max_beta_skip:

                _lsobj_factory(z, beta, n),
                z,

                    'maxcor': max_cor,

                z = np.log(np.maximum(a_new, 1e-30))

                best_a = a_new.copy()
                print(f"    LBFGS beta={beta:.2e}: score={sc_new:.10f}")

def _peak_shaving(a, deadline, n_iters=500):
    """Targeted peak shaving: reduce weights at positions contributing to top peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active < 1:
            active[np.argmax(c)] = True
            n_active = 1

        # Compute gradient direction: reduce contribution to active peaks
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        grad = _corr_a_A(a, w, n)

        # Also add a small uniform component to maintain sum
        grad_centered = grad - np.mean(grad)

        # Line search
        t = 0.01
        improved = False
        for ls in range(30):
            a_try = a - t * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-14:
                a = a_try
                c_max = f_try
                improved = True
                break
            t *= 0.5

        if not improved:
            # Try smaller step or different active set
            t = 0.001
            for ls in range(20):
                a_try = a - t * grad_centered
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max:
                    a = a_try
                    c_max = f_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=300, tol_list=None):
    """Active-set Newton method with adaptive tolerances."""
    if tol_list is None:
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None
        best_dir_score = np.inf
        best_t = 0.0

        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[int(np.argmax(c))] = True
            m = int(active.sum())
            if m < 2:
                continue
            if m > 4 * n:
                top_idx = np.argsort(c)[::-1][:4 * n]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                m = int(active.sum())

            A_vec = active.astype(np.float64)
            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:

            mu = 1e-6 * (c_max / max(m, 1)) + 1e-9

            def hess_vp(v, A_vec=A_vec, m=m, mu=mu, n=n):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

            # CG solve
            u = _cg_solve_simple(hess_vp, g_t, max_cg=150)
            w = _cg_solve_simple(hess_vp, np.ones(n), max_cg=150)
            w_sum = float(w.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w

            dx = dx - np.mean(dx)

            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)
            t = min(1.0, t_max)

            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)

            if sc_try < best_dir_score:
                best_dir_score = sc_try
                best_dir = dx
                best_t = t

        if best_dir is None:
            break

        # Line search
        f_old = _score(a)
        accepted = False
        t = best_t
        for ls in range(40):
            a_try = a + t * best_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try <= f_old - 1e-14:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _cg_solve_simple(matvec, rhs, max_cg=200):
    """Simple CG solver."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < 1e-12 * (1 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _basin_hopping_step(a, rng, deadline, min_time_remaining=20):
    """Perturb and re-optimize."""
    n = len(a)
    # Choose perturbation type
    r = rng.random()
    if r < 0.3:
        sigma = 0.01 + 0.1 * rng.random()
        a_pert = _perturb_random(a, sigma, rng)
    elif r < 0.5:
        kind = rng.choice(["sin1", "sin2", "sin3", "cos1", "asym", "asym2"])
        scale = 0.005 + 0.05 * rng.random()
        a_pert = _perturb_structured(a, kind, scale, rng)
    else:
        # Mix of random and structured
        a_pert = _perturb_random(a, 0.02, rng)
        a_pert = _perturb_structured(a_pert, "sin1", 0.02, rng)

    a_opt, sc_opt = _run_lbfgs(a_pert, deadline, min_time_remaining)
    return a_opt, sc_opt


def propose_candidate(budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with log-sum-exp smoothing,
    active-set Newton, peak shaving, and basin hopping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # ============ Phase 1: Main L-BFGS from parent ============
    print(f"\n[Phase 1] Main L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = min(deadline, t_start + 400)
    a_main, sc_main = _run_lbfgs(parent, phase1_deadline, min_time_remaining=600)
    update_best(a_main, sc_main, "lbfgs_main")

    # ============ Phase 2: Active-set Newton polish ============
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        as_dl = min(deadline - 400, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=300,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        update_best(a_as, sc_as, "as_newton_1")

        if best_seq is not None and time.monotonic() < deadline - 350:
            as_dl2 = min(deadline - 350, time.monotonic() + 120)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=200,
                                               tol_list=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            update_best(a_as2, sc_as2, "as_newton_2")

    # ============ Phase 3: Peak shaving ============
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        ps_dl = min(deadline - 300, time.monotonic() + 120)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_dl, n_iters=400)
        update_best(a_ps, sc_ps, "peak_shave")

        if sc_ps < best_score:
            # Re-optimize from peak-shaved solution
            if time.monotonic() < deadline - 250:
                a_ps2, sc_ps2 = _run_lbfgs(a_ps, deadline, min_time_remaining=250)
                update_best(a_ps2, sc_ps2, "lbfgs_post_ps")

    # ============ Phase 4: Diversified L-BFGS from parent ============
    print(f"\n[Phase 4] Diversified starts (t={time.monotonic() - t_start:.1f}s)")
    diverse_inits = [
        ("tapered", "tapered"),
        ("cos", "cos"),
        ("bimodal", "bimodal"),
        ("triangular", "triangular"),
    ]
    for label, kind in diverse_inits:
        if time.monotonic() > deadline - 200:
            break
        if best_seq is not None:
            # Blend parent with structured profile
            a_struct = _make_profile(n_parent, kind)
            for alpha in [0.3, 0.5]:
                if time.monotonic() > deadline - 200:
                    break
                a_mix = (1 - alpha) * best_seq + alpha * a_struct
                a_mix = np.maximum(a_mix, 0)
                a_mix = a_mix / a_mix.sum()
                a_d, sc_d = _run_lbfgs(a_mix, deadline, min_time_remaining=200, max_beta_skip=3)
                update_best(a_d, sc_d, f"div_{label}_{alpha}")
        else:
            a_struct = _make_profile(n_parent, kind)
            a_d, sc_d = _run_lbfgs(a_struct, deadline, min_time_remaining=200)
            update_best(a_d, sc_d, f"div_{label}")

    # ============ Phase 5: Basin hopping ============
    print(f"\n[Phase 5] Basin hopping (t={time.monotonic() - t_start:.1f}s)")
    n_hops = 0
    while best_seq is not None and time.monotonic() < deadline - 100:
        n_hops += 1
        a_hop, sc_hop = _basin_hopping_step(best_seq, rng, deadline, min_time_remaining=80)
        update_best(a_hop, sc_hop, f"hop_{n_hops}")

    # ============ Phase 6: Multi-n exploration ============
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        a_opt, sc_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60, max_beta_skip=5)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # ============ Phase 7: Final intensive polish ============
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight L-BFGS with smallest betas
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        final_betas = [mb * 0.00001, mb * 0.000005, mb * 0.000002, mb * 0.000001]
        for fb in final_betas:
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - 8 - time.monotonic()
            if remaining < 3:
                break
            max_iter = min(3000, max(100, int(remaining / 0.02)))
            try:
                result = minimize(
                    _lsobj_factory(z, fb, n),
                    z,
                    jac=_lsgrad_factory(z, fb, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-19, 'gtol': 1e-15,
                             'maxcor': 80, 'maxls': 100}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    z = np.log(np.maximum(a_fin, 1e-30))
                    z = z - z.mean()
                    update_best(a_fin, sc_fin, f"final_b{fb:.0e}")
            except Exception:
                pass

        # Final active-set Newton
        if time.monotonic() < deadline - 3:
            as_final = min(deadline - 2, time.monotonic() + 30)
            a_fin2, sc_fin2 = _active_set_newton(best_seq, as_final, max_iter=200,
                                                 tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
            update_best(a_fin2, sc_fin2, "final_as")

        # Final peak shaving
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _peak_shaving(best_seq, deadline - 1, n_iters=200)
            update_best(a_fin3, sc_fin3, "final_ps")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
