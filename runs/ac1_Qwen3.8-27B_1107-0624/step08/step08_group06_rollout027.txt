def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0, palindromic_flag=False):

def _muvwu_update(a_init, deadline, n_steps=5000, palindromic_flag=False):
    """Multiplicative Weight Update for escaping local minima."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Adaptive learning rate
    eta = 0.1
    patience = 0
    max_patience = 200

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Soft active set with temperature
        temp = 0.02 * c_max
        w = np.exp((c - c_max) / temp)
        w = w / w.sum()

        # Compute pressure: p[i] = sum_k w[k] * a[k-i]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        p = wc[n - 1:2 * n - 1]

        # Normalize pressure
        p_max = p.max()
        if p_max < 1e-15:
            break
        p_norm = p / p_max

        # Multiplicative update: reduce elements with high pressure
        a_new = a * np.exp(-eta * p_norm)
        a_new = np.maximum(a_new, 1e-30)

        if palindromic_flag:
            a_new = _palindromic(a_new)

        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            patience = 0
            # Increase eta slightly on improvement
            eta = min(eta * 1.01, 0.5)
        else:
            patience += 1
            if patience >= max_patience:
                # Decrease eta to be more conservative
                eta *= 0.7
                patience = 0
                if eta < 0.001:
                    break

        a = a_new

        if step % 500 == 0:
            print(f"    MWU step {step}: score={sc_new:.10f}, eta={eta:.4f}")

    return best_a

        return a_init
    a = a / s

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    beta = max_c * 0.01
    step_size = 1.0 / (4.0 * n * max_c * 2.0)
    t = 1.0
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_y = a - t * v
        a_y = np.maximum(a_y, 1e-30)
        a_y = a_y / a_y.sum()

        # Compute gradient at lookahead point
        c = _full_convolve(a_y, a_y)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        a_rev = a_y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a_y * (h - W)

        # Project gradient (remove mean for sum constraint)
        grad_t = grad - np.mean(grad)

        # Update
        v_new = 0.5 * (v + grad_t)
        a_new = a - step_size * t * v_new
        a_new = np.maximum(a_new, 0)
        if palindromic_flag:
            a_new = _palindromic(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        v = v_new
        a = a_new
        t = t_new

        # Adaptive step size
        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
            # Adjust step size
            if step > 1000 and step % 1000 == 0:
                step_size *= 0.5
                beta *= 0.5

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a
