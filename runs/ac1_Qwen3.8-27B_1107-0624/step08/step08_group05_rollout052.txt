# Core utilities (same as before but refined)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# LSE objective and gradient (same as before)

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

def _nesterov_lse(a_init, deadline, betas, n_steps_each=2000, lr_init=0.05):

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

        if time.monotonic() > deadline:
            break
        # Initialize Nesterov momentum
        y = z.copy()
        v = np.zeros_like(z)
        momentum = 0.9
        
        for i in range(n_steps_each):
            if time.monotonic() > deadline:
                break

            grad = _lse_gradient(y, beta, n)
            
            # Adaptive learning rate

def _fourier_flat_projection(a, n_iters=100, deadline=None):

    s = a_curr.sum()
    if s < 1e-15:
        return a.copy()

    best_a = a_curr.copy()
    best_sc = _score(a_curr)

        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a_curr, 2 * n)
        
        # Target: flat |A|^2 (but keep DC component fixed = sum)
        # |A[0]| = sum(a) = 1 (normalized)
        # We want |A[k]|^2 to be as equal as possible for k >= 1
        # The constraint is that sum(|A[k]|^2) = sum(a^2) * (2n) by Parseval
        # Actually, sum(|A[k]|^2) for k=0..n-1 (rfft) relates to sum(a^2)
        
        # Let's target: |A[k]| = target_amp for k >= 1
        # where target_amp is chosen so that Parseval's theorem is satisfied
        # sum_k |A[k]|^2 = (2n) * sum_j a[j]^2 (for full FFT of length 2n)
        # With rfft: |A[0]|^2 + 2*sum_{k=1}^{n-1} |A[k]|^2 + |A[n]|^2 (if 2n even) = (2n)*sum(a^2)
        
        # Actually let's just try to flatten |A| while keeping |A[0]| fixed
        A0 = A[0].copy()
        amp = np.abs(A)
        # Target amplitude: geometric mean of non-DC amplitudes? Or just flatten?
        # Simple: target = mean of non-DC amplitudes
        non_dc = amp[1:]
        target = np.mean(non_dc)
        target = max(target, 1e-10)
        
        # New spectrum with target amplitude and old phase
        phase = np.angle(A)
        amp_new = np.full_like(amp, target)
        amp_new[0] = np.abs(A0)
        A_new = amp_new * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_curr = a_new
        
        if it % 10 == 0:
            sc = _score(a_curr)
            if sc < best_sc:
                best_sc = sc
                best_a = a_curr.copy()
    
    sc_final = _score(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_sc

def _upsample(a, n_target):
    """Upsample sequence a to length n_target using linear interpolation."""
    n = len(a)
    if n == n_target:
        return a.copy()
    if n == 1:
        return np.full(n_target, a[0] / n_target)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_new / s

def _hess_matvec(v, active, n):

    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha <= 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_log(a, sigma, rng):

    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):

    s = float(a.sum())

    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))

    c = _full_convolve(a, a)

def _upsample(a, n_target):
    n = len(a)
    if n == n_target:
        return a.copy()
    if n == 1:
        return np.full(n_target, a[0] / n_target)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_new / s


def _downsample(a):

    s = down.sum()
    if s < 1e-15:
        return np.ones(len(down)) / len(down)
    return down / s

        momentum = 0.9
        lr = 0.05

        # Adaptive number of steps based on remaining time
        est_time_per_step = 0.0005 * (n_curr / 512.0)
        time_available = max(1.0, deadline - time.monotonic() - 10)
        n_steps = min(5000, max(200, int(time_available / max(est_time_per_step, 0.001))))

        for i in range(n_steps):
            if time.monotonic() > deadline:
                break
            y = z + momentum * v
            grad = _lse_gradient(y, beta, n_curr)
            lr_step = lr / (1.0 + 0.003 * i)
            v = momentum * v - lr_step * grad
            z = z + v
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()

        a_opt = _softmax(z)
        sc = _score(a_opt)

        if level > 0:
            target_len = len(pyramid[level - 1])
            a_up = _upsample(a_opt, target_len)
            pyramid[level - 1] = a_up
        else:
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
            a_full = a_opt

    return best_a, best_sc

# Newton active-set with top-k
# ============================================================

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _hess_matvec_closed(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve_func(hess_v, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_v(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30 or pH <= 0:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_cg_hv(v, active_f, n, mu):
    return _hess_matvec(v, active_f, n) + mu * v


def _newton_active_set(a_init, deadline, max_iter=200, top_k=10):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-k active set
        if top_k >= len(c):
            active = np.ones_like(c, dtype=bool)
        else:
            sorted_idx = np.argsort(c)[::-1]
            thresh = c[sorted_idx[min(top_k, len(c) - 1)]]
            active = c >= thresh

        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)

        # Gradient: H*a where H[i,j] = active[i+j] (for valid indices)
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        # CG solve
        dx = _cg_solve_cg(a, active_f, n, mu, -grad_t)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _cg_solve_cg(a_curr, active_f, n, mu, b_vec):
    """CG solve using precomputed parameters."""
    n2 = len(b_vec)
    x = np.zeros(n2)
    r = b_vec.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b_vec))
    if b_norm < 1e-30:
        return x
    for _ in range(200):
        Hp = _hess_matvec(p, active_f, n) + mu * p
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30 or pH <= 0:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < 1e-10 * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


# ============================================================
# L-BFGS with LSE
# ============================================================

def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=14):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.6, 0.35, 0.2, 0.12, 0.07,
                                   0.04, 0.02, 0.01, 0.005, 0.002,
                                   0.001, 0.0005, 0.0002])
    betas = betas[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-13, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


# ============================================================
# Subgradient polish
# ============================================================

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

# Perturbation methods
# ============================================================

def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_blocks))
    for _ in range(n_blocks):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

# Profile generators
# ============================================================

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


# ============================================================
# Power-sum smoothing (higher p for sharper peaks)
# ============================================================

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p - 1)
    sum_cp = np.sum(cp)
    w = cp / sum_cp if sum_cp > 1e-300 else np.zeros_like(c)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (np.sum(np.power(c_safe, p))) ** (1.0 / p - 1.0)
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer: Nesterov LSE, Fourier flat-spectrum,
    multi-resolution, adaptive Newton (top-k), L-BFGS, and subgradient polish.
    """
    global _g_best_seq, _g_best_score, _g_t_start
    _g_best_seq = None
    _g_best_score = np.inf

    _g_t_start = time.monotonic()
    deadline = _g_t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        _update_best(parent, _score(parent), "init_uniform")

    print(f"Starting optimization, deadline in {deadline - _g_t_start:.0f}s")

    # === Phase 1: Multi-resolution warm-up (60s) ===
    print(f"\n[Phase 1] Multi-resolution warm-up")
    mr_dead = min(deadline, _g_t_start + 60)
    if len(_g_best_seq) > 1024:
        a_mr, sc_mr = _multires_optimize(_g_best_seq.copy(), mr_dead)
        _update_best(a_mr, sc_mr, "multires_warmup")

    # === Phase 2: Nesterov LSE - high beta (400s) ===
    print(f"\n[Phase 2] Nesterov LSE high-beta")
    n = len(_g_best_seq)
    c = _full_convolve(_g_best_seq, _g_best_seq)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, c_max * 0.005)

    nesterov_betas_high = base_beta * np.array([1.0, 0.5, 0.25])
    nest_dead = min(deadline, _g_t_start + 460)
    a_nest, sc_nest = _nesterov_lse(
        _g_best_seq.copy(), nest_dead,
        betas=nesterov_betas_high, n_steps_each=3000,
        lr_init=0.08, momentum_init=0.9
    )
    _update_best(a_nest, sc_nest, "nesterov_high")

    # === Phase 3: L-BFGS LSE (350s) ===
    print(f"\n[Phase 3] L-BFGS LSE")
    if time.monotonic() < deadline - 350:
        a_lb, sc_lb = _lbfgs_lse(_g_best_seq.copy(), deadline, min_remaining=330, n_stages=12)
        _update_best(a_lb, sc_lb, "lbfgs_lse")

    # === Phase 4: Fourier flat-spectrum (200s) ===
    print(f"\n[Phase 4] Fourier flat-spectrum")
    if time.monotonic() < deadline - 200:
        four_dead = min(deadline - 180, time.monotonic() + 180)
        a_four, sc_four = _fourier_flat_projection(_g_best_seq, four_dead, n_iters=1000)
        _update_best(a_four, sc_four, "fourier_flat")
        # Follow up with LSE from Fourier result
        if sc_four < _g_best_score + 0.05:
            a_f2, sc_f2 = _lbfgs_lse(a_four, deadline, min_remaining=160)
            _update_best(a_f2, sc_f2, "fourier_lse")

    # === Phase 5: Nesterov LSE - low beta (250s) ===
    print(f"\n[Phase 5] Nesterov LSE low-beta")
    if time.monotonic() < deadline - 250:
        nesterov_betas_low = base_beta * np.array([0.12, 0.06, 0.03, 0.01, 0.005])
        nest_dead2 = min(deadline - 230, time.monotonic() + 230)
        a_nest2, sc_nest2 = _nesterov_lse(
            _g_best_seq.copy(), nest_dead2,
            betas=nesterov_betas_low, n_steps_each=3000,
            lr_init=0.03, momentum_init=0.92
        )

    # === Phase 6: Newton active-set with varying top-k (200s) ===
    print(f"\n[Phase 6] Newton active-set")
    if time.monotonic() < deadline - 180:
        for top_k in [3, 7, 15, 30, 60]:
            if time.monotonic() > deadline - 160:

    print(f"\n[Phase 7] Subgradient polish")
    if time.monotonic() < deadline - 100:
        sg_dead = min(deadline - 80, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(_g_best_seq, sg_dead, n_steps=8000, lr_init=0.006, tol_frac=0.0008)
        _update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 8: More Newton after SG (80s) ===
    print(f"\n[Phase 8] Post-SG Newton")
    if time.monotonic() < deadline - 60:
        a_nw2, sc_nw2 = _newton_active_set(_g_best_seq, deadline - 50, max_iter=100, top_k=10)
        _update_best(a_nw2, sc_nw2, "post_sg_newton")

    # === Phase 9: Diverse restarts (150s) ===
    print(f"\n[Phase 9] Diverse restarts")
    if time.monotonic() < deadline - 120:
        restarts = [
            ("log_perturb", _perturb_log(_g_best_seq, 0.04, rng)),
            ("fourier_phase", _perturb_fourier_phase(_g_best_seq, rng, n_modes=10, max_amp=0.4)),
            ("amplitude", _perturb_amplitude(_g_best_seq, rng, sigma=0.1)),
            ("block", _perturb_block(_g_best_seq, rng, n_blocks=6)),
            ("log_large", _perturb_log(_g_best_seq, 0.08, rng)),
        ]
        for i, (name, p_start) in enumerate(restarts):
            if time.monotonic() > deadline - 40:
                break
            # Quick LSE from perturbed start
            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=30, n_stages=8)
            _update_best(a_r, sc_r, f"restart_{name}")

            # Newton follow-up if promising
            if sc_r < _g_best_score * 1.01 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _newton_active_set(a_r, deadline - 15, max_iter=50, top_k=7)
                _update_best(a_r2, sc_r2, f"restart_{name}_nw")

    # === Phase 10: Final aggressive polish (60s) ===
    print(f"\n[Phase 10] Final aggressive polish")
    if _g_best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-low beta Nesterov
        n_fin = len(_g_best_seq)
        c_fin = _full_convolve(_g_best_seq, _g_best_seq)
        mb_fin = float(c_fin.max())
        z_fin = np.log(np.maximum(_g_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()

        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            beta_final = mb_fin * 0.001
            a_fin, sc_fin = _nesterov_lse(
                _g_best_seq.copy(), deadline - 3,
                betas=[beta_final], n_steps_each=5000,
                lr_init=0.01, momentum_init=0.95
            )
            _update_best(a_fin, sc_fin, "final_nesterov")

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_g_best_seq, deadline - 2, max_iter=80, top_k=5)
            _update_best(a_fin2, sc_fin2, "final_newton")

        # Final subgradient with very tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_g_best_seq, deadline - 1, n_steps=3000, lr_init=0.003, tol_frac=0.0003)
            _update_best(a_fin3, sc_fin3, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")

        s = float(final.sum())

            print("FALLBACK: uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - _g_t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
