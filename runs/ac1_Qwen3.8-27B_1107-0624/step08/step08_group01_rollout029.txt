```python
def _lp_step(f, M, K=50):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for top K lags."""
    n = len(f)
    b_conv = np.convolve(f, f)
    # Get top K lags
    top_k = np.argsort(b_conv)[::-1][:K]
    
    # Build constraint matrix A (K x n) where A[j, i] = f[i] if 0 <= j-i < n
    # (f*b)[j] = sum_i f[i] * b[j-i]
    # For constraint j: sum_i f[i] * b[j-i] <= M
    # Let k = j - i, so i = j - k. Constraint: sum_k f[j-k] * b[k] <= M (where 0 <= j-k < n and 0 <= k < n)
    
    from scipy.optimize import linprog
    from scipy.sparse import csc_matrix
    
    # A will be K x n, sparse
    rows = []
    cols = []
    vals = []
    for jj in range(K):
        k_val = top_k[jj]
        # (f*b)[k_val] = sum_i f[i] * b[k_val - i] for 0 <= i < n, 0 <= k_val - i < n
        # = sum_i f[i] * b[k_val - i]
        # Let index of b be m = k_val - i, so i = k_val - m
        # m ranges from max(0, k_val - (n-1)) to min(k_val, n-1)
        m_lo = max(0, k_val - n + 1)
        m_hi = min(k_val, n - 1)
        for m in range(m_lo, m_hi + 1):
            i = k_val - m
            # A entry: coefficient of b[m] in constraint jj is f[i]
            rows.append(jj)
            cols.append(m)
            vals.append(f[i])
    
    A = csc_matrix((vals, (rows, cols)), shape=(K, n))
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    result = linprog(c_obj, A_ub=A, b_ub=np.full(K, M), 
                     bounds=[(0, None)] * n, method='highs')
    
    if result.success:
        return result.x
    return None
```

    """Solve LP: max sum(b) s.t. b>=0, (f*b)[top_k lags] <= M. Returns normalized b."""
    n = len(f)

    top_k = np.argsort(c_conv)[::-1][:K]

    all_rows = []
    all_cols = []
    all_vals = []

        m_lo = max(0, k_val - n + 1)
        m_hi = min(k_val, n - 1)
        m_range = np.arange(m_lo, m_hi + 1)
        i_range = k_val - m_range
        all_rows.extend([jj] * len(m_range))
        all_cols.extend(m_range.tolist())
        all_vals.extend(f[i_range].tolist())

"""
Optimized step-function search using LP-based active set direction finding combined
with L-BFGS polishing, growing active sets, multi-n exploration, and diverse restarts.
Key new idea: Linear programming to find directions that provably reduce the maximum
autocorrelation at active lags, with adaptive active set management.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csc_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_sparse_constraint(f, top_k):
    """Build sparse constraint matrix for LP: rows are lags, cols are b indices."""
    n = len(f)
    K = len(top_k)
    row_parts = []
    col_parts = []
    val_parts = []
    for jj in range(K):
        k_val = int(top_k[jj])
        m_lo = max(0, k_val - n + 1)
        m_hi = min(k_val, n - 1)
        m_range = np.arange(m_lo, m_hi + 1)
        i_range = k_val - m_range
        row_parts.append(np.full(len(m_range), jj, dtype=np.int32))
        col_parts.append(m_range.astype(np.int32))
        val_parts.append(f[i_range])
    if not row_parts:
        return csc_matrix((K, n))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    return csc_matrix((vals, (rows, cols)), shape=(K, n))


def _lp_direction(f, M, top_k, timeout=8.0):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for k in top_k."""
    n = len(f)
    K = len(top_k)
    if K == 0:
        return None
    A = _build_sparse_constraint(f, top_k)
    c_obj = -np.ones(n)
    try:
        result = linprog(c_obj, A_ub=A, b_ub=np.full(K, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': timeout,
                                  'pfeasibility_tolerance': 1e-9,
                                  'dfeasibility_tolerance': 1e-9,
                                  'mip_rel_gap': 0.01})
        if result.status == 0 and result.x is not None:
            g = np.maximum(result.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def _lp_solve_dual(f, M, top_k, timeout=8.0):
    """Solve dual LP: min M*sum(y) s.t. A^T y >= 1, y >= 0. Returns primal via complementary slackness."""
    n = len(f)
    K = len(top_k)
    if K == 0:
        return None
    A = _build_sparse_constraint(f, top_k)
    AT = A.T.tocsc()
    # Dual: min M * 1^T y s.t. A^T y >= 1, y >= 0
    # Convert to standard form: min M * 1^T y s.t. -A^T y <= -1, y >= 0
    c_dual = M * np.ones(K)
    A_ub_dual = -AT
    b_ub_dual = -np.ones(n)
    try:
        result = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual,
                         bounds=[(0, None)] * K, method='highs',
                         options={'time_limit': timeout,
                                  'pfeasibility_tolerance': 1e-9,
                                  'dfeasibility_tolerance': 1e-9})
        if result.status == 0 and result.x is not None:
            y = np.maximum(result.x, 0.0)
            # Get primal from dual slack: b[m] > 0 only where (A^T y)[m] = 1
            Ay = A.T @ y
            tol = 1e-6
            active_primal = Ay >= (1.0 - tol)
            n_active = int(active_primal.sum())
            if n_active == 0:
                return None
            if n_active >= n:
                # All active, uniform
                return np.ones(n) / n
            # Solve smaller LP on active set
            sub_A = A[:, active_primal].tocsc()
            c_sub = -np.ones(n_active)
            try:
                sub_result = linprog(c_sub, A_ub=sub_A, b_ub=np.full(K, M),
                                     bounds=[(0, None)] * n_active, method='highs',
                                     options={'time_limit': 2.0})
                if sub_result.status == 0 and sub_result.x is not None:
                    g = np.zeros(n)
                    g[active_primal] = np.maximum(sub_result.x, 0.0)
                    s = g.sum()
                    if s > 1e-15:
                        return g / s
            except Exception:
                pass
            # Fallback: uniform on active set
            g = np.zeros(n)
            g[active_primal] = 1.0 / n_active
            return g
    except Exception:
        pass
    return None


def _lp_iteration(f, deadline, K_start=50, K_max=500, K_growth=20, max_no_improve=8):
    """LP-based active set iterations with growing K."""
    n = len(f)
    f_cur = f.copy()
    s = f_cur.sum()
    if s < 1e-15:
        return f, _score(f)
    f_cur = f_cur / s
    best_f = f_cur.copy()
    best_sc = _score(f_cur)
    print(f"  LP iteration start: score={best_sc:.10f}, n={n}")

    K = K_start
    no_improve = 0
    step_count = 0

    while time.monotonic() < deadline:
        c_conv = _full_convolve(f_cur, f_cur)
        M = float(c_conv.max())
        current_sc = 2.0 * n * M

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_f = f_cur.copy()

        # Get top K lags
        top_k = np.argsort(c_conv)[::-1][:K]

        # Try primal LP first, then dual
        g = _lp_direction(f_cur, M, top_k, timeout=5.0)
        if g is None:
            g = _lp_solve_dual(f_cur, M, top_k, timeout=5.0)
        if g is None:
            no_improve += 1
            if no_improve >= 3:
                break
            K = min(K_max, K + K_growth)
            continue

        # Line search
        best_t = 0.0
        best_new_sc = current_sc
        for t in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_try = (1.0 - t) * f_cur + t * g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try / s_try
            sc_try = _score(f_try)
            if sc_try < best_new_sc - 1e-14:
                best_new_sc = sc_try
                best_t = t

        step_count += 1
        if best_t > 0:
            f_cur = (1.0 - best_t) * f_cur + best_t * g
            f_cur = np.maximum(f_cur, 0.0)
            f_cur = f_cur / f_cur.sum()
            if best_new_sc < best_sc - 1e-13:
                no_improve = 0
                if step_count <= 20 or step_count % 10 == 0:
                    print(f"    LP step {step_count}: t={best_t:.3f}, K={K}, score={best_new_sc:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            K = min(K_max, K + K_growth)

        if no_improve >= max_no_improve:
            break

    print(f"  LP iteration done: best_score={best_sc:.10f}, steps={step_count}, K_final={K}")
    return best_f, best_sc


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def projgrad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remaining=20.0):
    """L-BFGS with beta annealing for polishing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
         0.00005, 0.00002, 0.00001, 0.000005]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(5000, max(30, int(remaining / (est_iter_time * 15))))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.projgrad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_start=0.005, tol_end=0.00001,
                       lr_start=0.03, top_k=5):
    """Active-set gradient descent with top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0
    max_stall = 200

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 10:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.4
                no_improve = 0
                stall += 1
                if stall > 10:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.985)
            lr = min(lr_start, lr * 1.01)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)

def _peak_shave(a, deadline, n_trials=100):
    """Direct peak shaving: reduce top contributors to the max lag."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(99999)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))

        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue

        frac = rng.uniform(0.02, 0.10)
        n_reduce = max(1, int(len(contrib) * frac))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.15
            a_try[idx] -= red
            reduction += red

        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
            jv = k_star - idx
            if 0 <= jv < n:
                low_mask[jv] = False
        if low_mask.sum() < 5:
            low_mask = np.ones(n, dtype=bool)
            for idx in top_idx:
                low_mask[idx] = False

        a_try[low_mask] += reduction / max(1, low_mask.sum())
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc

    if time.monotonic() < deadline - 400.0:
        a4b = _run_lbfgs(best_a, min(deadline - 350.0, t_start + 1000.0), min_remaining=20.0)
        sc4b = _score(a4b)
        if sc4b < best_score - 1e-12:
            best_a = a4b.copy()
            best_score = sc4b
            print(f"  IMPROVED: {best_score:.10f} [lbfgs_p4b]")

    # === Phase 5: Peak shaving ===
    print(f"\n[P5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 300.0, t_start + 1100.0)
    if time.monotonic() < p5_deadline:
        a5, sc5 = _peak_shave(best_a, p5_deadline, n_trials=150)
        if sc5 < best_score - 1e-12:
            best_a = a5.copy()
            best_score = sc5
            print(f"  IMPROVED: {best_score:.10f} [peak_p5]")
        # Quick L-BFGS after peak shave
        if time.monotonic() < deadline - 250.0:
            a5b = _run_lbfgs(best_a, min(deadline - 200.0, t_start + 1200.0), min_remaining=15.0)
            sc5b = _score(a5b)
            if sc5b < best_score - 1e-12:
                best_a = a5b.copy()
                best_score = sc5b
                print(f"  IMPROVED: {best_score:.10f} [lbfgs_p5b]")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [16384, 24576]
    if len(best_a) not in n_targets:
        n_targets.append(len(best_a))

    for n_try in sorted(n_targets, key=lambda x: abs(x - len(best_a))):
        if time.monotonic() > deadline - 150.0:
            break
        if n_try == len(best_a):
            continue
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  Resampled to n={n_try}, init score={sc_init:.10f}")
        n_deadline = min(deadline - 100.0, time.monotonic() + 200.0)
        if n_deadline - time.monotonic() < 30.0:
            continue
        a_opt = _run_lbfgs(a_init, n_deadline, min_remaining=20.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score - 1e-12:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  IMPROVED: {best_score:.10f} (n={n_try}) [multi_n]")
        # Quick polish
        if time.monotonic() < deadline - 80.0:
            a_pol, sc_pol = _active_set_polish(a_opt, min(deadline - 60.0, time.monotonic() + 60.0),
                                                tol_start=0.002, tol_end=0.0001, lr_start=0.01)
            if sc_pol < best_score - 1e-12:
                best_a = a_pol.copy()
                best_score = sc_pol
                print(f"  IMPROVED: {best_score:.10f} (n={n_try}) [multi_n_polish]")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80.0:
        n_best = len(best_a)
        restarts = [
            ("pert_small", _perturb(best_a, 0.01, rng)),
            ("pert_med", _perturb(best_a, 0.03, rng)),
            ("tapered", _make_profile(n_best, "tapered")),
            ("cos_mod", _make_profile(n_best, "cos_mod")),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 40.0:
                break
            remaining = deadline - time.monotonic() - 30.0
            if remaining < 20.0:
                break
            a_opt = _run_lbfgs(a_r, deadline - 30.0, min_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score - 1e-12:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  IMPROVED: {best_score:.10f} (n={n_best}) [restart_{label}]")

    # === Phase 8: Final ultra-fine polishing ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 3.0,
                                           tol_start=0.0003, tol_end=0.000003,
                                           lr_start=0.005, top_k=3)
        if sc_fin < best_score - 1e-12:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  IMPROVED: {best_score:.10f} [final_polish]")

        # Tiny LP at the end
        if time.monotonic() < deadline - 10.0:
            a_lp, sc_lp = _lp_iteration(best_a, deadline - 5.0, K_start=200, K_max=400,
                                        K_growth=50, max_no_improve=3)
            if sc_lp < best_score - 1e-12:
                best_a = a_lp.copy()
                best_score = sc_lp
                print(f"  IMPROVED: {best_score:.10f} [final_lp]")

        # Peak shave at end
        if time.monotonic() < deadline - 5.0:
            a_ps, sc_ps = _peak_shave(best_a, deadline - 3.0, n_trials=30)
            if sc_ps < best_score - 1e-12:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  IMPROVED: {best_score:.10f} [final_peak]")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
