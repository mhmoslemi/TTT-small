"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp L-BFGS, active-set top-k peak shaving,
multiplicative mirror descent, and multi-scale n exploration.
Key innovations: adaptive temperature based on peak gap, direct top-k
active-set gradient, and targeted peak cancellation with redistribution.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * h
    grad_z = a * (grad_a - 4.0 * n * W)
    return grad_z


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta based on peak gap."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    c_max = float(c_cur.max())
    if c_max < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    k = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[k]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap * 0.5, c_max * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    n_betas = 14
    beta_multipliers = np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18,
                                 0.12, 0.08, 0.05, 0.03, 0.02,
                                 0.012, 0.007, 0.004])

    for i, mult in enumerate(beta_multipliers):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * mult
        # Recompute adaptive beta based on current peak structure
        c_now = _full_convolve(current_a, current_a)
        c_now_max = float(c_now.max())
        c_now_sorted = np.sort(c_now)[::-1]
        kk = min(30, len(c_now_sorted) - 1)
        current_gap = c_now_sorted[0] - c_now_sorted[kk]
        if current_gap > 0:
            adaptive_beta = max(current_gap * 0.1, c_now_max * 0.001)
            beta = min(beta, adaptive_beta * 3)
            beta = max(beta, c_now_max * 0.0003)

        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:

        except Exception:

    return current_a


def _top_k_peak_shaving(a_init, deadline, min_time_remaining=30.0):
    """Direct active-set top-k peak shaving with multiplicative updates."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    k_top = max(5, int(n * 0.003))
    eta = 0.02
    patience = 0
    max_patience = 200

    while time.monotonic() + min_time_remaining < deadline and patience < max_patience:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
            eta = min(0.05, eta * 1.05)
        else:
            patience += 1
            if patience % 20 == 0:
                eta = max(0.001, eta * 0.8)

        # Find top-k peaks
        top_idx = np.argsort(c)[::-1][:k_top]
        top_vals = c[top_idx]
        target = c_max * 0.95

        # Compute gradient from active set
        # grad_a[j] = 2 * sum_{k in top_idx} a[k-j] / k_top (for valid k-j)
        a_rev = a[::-1]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / k_top
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Multiplicative update
        a_try = a * np.exp(-eta * 2.0 * (h - W))
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < current_sc - 1e-15:
            a = a_try
        else:
            # Try smaller step
            a_try = a * np.exp(-eta * 0.5 * 2.0 * (h - W))
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
            else:
                eta = max(0.0005, eta * 0.7)

        # Adapt k_top
        if patience % 50 == 0 and patience > 0:
            k_top = max(3, int(k_top * 0.7))

    return best_a, best_sc


def _mirror_descent_polish(a_init, deadline, min_time_remaining=20.0):
    """Mirror descent with adaptive step size for fine-tuning."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    eta = 0.005
    patience = 0
    max_patience = 500

    while time.monotonic() + min_time_remaining < deadline and patience < max_patience:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.85
            if patience > max_patience:
                break

        # Find all peaks above target
        target = c_max * 0.99
        active = c >= target
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Gradient direction (want to decrease peaks)
        grad = 2.0 * (h - W)
        gnorm = np.linalg.norm(grad * a)
        if gnorm < 1e-18:
            break

        # Multiplicative step
        a_try = a * np.exp(-eta * grad / (gnorm + 1e-18))
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < current_sc - 1e-16:
            a = a_try
        else:
            eta *= 0.8

    return best_a, best_sc


def _targeted_peak_reduction(a_init, deadline, min_time_remaining=20.0):
    """Directly target the single worst peak and reduce it."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()

        # Focus on the single worst peak
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]

        # Compute contributions to this peak
        # b[k_peak] = sum_i a[i] * a[k_peak - i]
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)

        # Find which pairs contribute most
        contributions = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            contributions[i] += a[j]
            if j != i:
                contributions[j] += a[i]

        # Reduce top contributors, redistribute to bottom contributors
        top_frac = 0.05
        n_top = max(3, int(n * top_frac))
        top_idx = np.argsort(contributions)[::-1][:n_top]
        bot_idx = np.argsort(contributions)[:n_top]

        # How much to reduce
        reduction_factor = 0.005
        total_red = 0.0
        for idx in top_idx:
            red = a[idx] * reduction_factor
            a[idx] -= red
            total_red += red

        # Redistribute to bottom contributors
        if total_red > 0:
            for idx in bot_idx:
                a[idx] += total_red / n_top

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc_new = _score(a)
        if sc_new > current_sc * 1.01:
            # Revert if it got much worse
            a = best_a.copy()
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_bell":
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.cos(np.pi * (t - center) / n) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_center":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin
                    best_score = sc_fin
                    print(f"  Final LBFGS: {best_score:.10f}")
            except Exception:
                pass

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"Verified with evaluate_sequence: {ev_score:.10f}")
            if ev_score < best_score:
                best_score = ev_score
        except Exception:
            pass
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
