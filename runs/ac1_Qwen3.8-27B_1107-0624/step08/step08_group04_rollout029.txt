"""
Active-set banded QP optimizer for minimizing peak autocorrelation.
Uses: (1) FFT-convolution smoothed L-BFGS for initialization, (2) exact banded
linear-system solves for active-set QP subproblems (much more accurate than CG),
(3) adaptive active-set identification via softmax weighting, (4) multi-n exploration,
(5) equioscillation-based convergence checking, and (6) targeted coordinate polish.
"""
import numpy as np
import time
import scipy.linalg
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    """Compute centered autocorrelation: R[m] = sum_i a[i]*a[i+m] for m=0..n-1."""
    n = len(a)
    c = _full_convolve(a, a)
    # c has length 2n-1, c[n-1] = R[0], c[n-1±m] relates to R[m]
    # R[m] = c[n-1+m] for m >= 0 (right half)
    # Actually: c[k] = sum_i a[i]*a[k-i], and R[m] = sum_i a[i]*a[i+m]
    # c[n-1+m] = sum_i a[i]*a[n-1+m-i]. Let j = n-1-i, then = sum_j a[n-1-j]*a[m+j]
    # This equals R[m] only if a is symmetric. In general, c[n-1+m] != R[m].
    # For non-symmetric a, the max of c IS the max autocorrelation (just at a different lag).
    # So we just use c directly and find its max.
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _lbfgs_optimize(a_init, deadline, time_limit=300):
    """L-BFGS with adaptive beta continuation on log-parametrization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.001)

    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                       0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                       0.00002, 0.00001, 0.000005, 0.000002, 0.000001, 0.0000005,
                       0.0000002, 0.0000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    end_time = time.monotonic() + time_limit
    end_time = min(end_time, deadline)

    for i, beta in enumerate(betas):
        if time.monotonic() > end_time - 5:
            break
        remaining = end_time - time.monotonic()
        if remaining < 5:
            break
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / max(est_iter_time, 0.001))))

        try:
            result = minimize(
                _make_smoobj_wrapper(beta, n),
                z,
                jac=_make_smograd_wrapper(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_smoobj_wrapper(beta, n):
    def obj(z):
        return _smoobj(z, beta, n)
    return obj


def _make_smograd_wrapper(beta, n):
    def grad(z):
        return _smograd(z, beta, n)
    return grad


def _build_banded_H(active_lags, weights, n, max_bandwidth=300):
    """Build banded matrix H = sum_k w_k * C_k as scipy banded format.
    active_lags: list of lags m >= 0
    weights: corresponding weights
    Returns (band_M, band_U, ab) where ab is the (2*K+1, n) banded array.
    """
    max_k = min(max(active_lags), max_bandwidth) if active_lags else 0
    K = int(max_k)
    ab = np.zeros((2 * K + 1, n))
    for m, w in zip(active_lags, weights):
        if m == 0:
            ab[K, :] += w  # diagonal
        elif m <= K:
            # m-th superdiagonal and subdiagonal
            ab[K + m, m:] = w  # superdiagonal: ab[K+m, j] = H[j-m, j] for j >= m
            ab[K - m, :-m] = w  # subdiagonal: ab[K-m, j] = H[j+m, j] for j < n-m
    return K, ab


def _solve_banded_qp(active_lags, weights, n, a_init, max_active_constraints=50):

    H is banded with bandwidth = max(active_lags).
    """
    max_k = min(max(active_lags), 300) if active_lags else 0
    K = int(max_k)
    K, ab = _build_banded_H(active_lags, weights, n, max_bandwidth=300)

    # Initial support: where a_init > threshold
    threshold = 1e-12
    support = a_init > threshold
    if not support.any():
        support[:] = True

    a = np.zeros(n)
    a[support] = a_init[support] / a_init[support].sum() if a_init[support].sum() > 1e-15 else 1.0 / n
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = float(a @ _hankel_mv(a, active_lags, weights, n))

    for it in range(50):
        # Solve H_TT x = 1 on current support
        T_idx = np.where(support)[0]
        n_T = len(T_idx)

        if n_T == n:
            # Full support: solve H x = 1 directly
            try:
                x = scipy.linalg.solve_banded((K, K), ab, np.ones(n), overwrite_ab=False)
            except Exception:
                break
            if x.min() < -1e-10:
                # Some entries negative: reduce support
                neg_idx = np.where(x < 0)[0]
                # Remove the most negative entries
                n_remove = min(len(neg_idx), max(1, len(neg_idx) // 2))
                worst = neg_idx[np.argsort(x[neg_idx])[:n_remove]]
                support[worst] = False
                continue

            a_new = x / x.sum() if x.sum() > 1e-15 else a
        else:

                v_full = np.zeros(n)
                v_full[T_idx] = v
                return _hankel_mv(v_full, active_lags, weights, n)[T_idx]

            try:

                                                    tol=1e-12, maxiter=500)

            if x_T.min() < -1e-10:
                neg_local = np.where(x_T < 0)[0]
                n_remove = min(len(neg_local), max(1, len(neg_local) // 2))
                worst_local = neg_local[np.argsort(x_T[neg_local])[:n_remove]]
                support[T_idx[worst_local]] = False
                continue

            a_new = np.zeros(n)
            a_new[T_idx] = np.maximum(x_T, 0)
            a_new = a_new / a_new.sum() if a_new.sum() > 1e-15 else a

        # Check objective
        obj = float(a_new @ _hankel_mv(a_new, active_lags, weights, n))
        if obj < best_obj - 1e-16:
            best_obj = obj
            best_a = a_new.copy()

        # Check KKT conditions for inactive indices
        grad = _hankel_mv(a_new, active_lags, weights, n)
        lam = grad[support].mean() if support.any() else 0
        inactive = ~support
        if inactive.any():
            violation = np.max(lam - grad[inactive]) if np.any(lam - grad[inactive] > 0) else 0
            if violation > 1e-10 * (np.max(np.abs(grad)) + 1e-15):
                # Add most violating inactive indices to support
                viol_idx = np.where(inactive)[0]
                viol_scores = lam - grad[viol_idx]
                n_add = min(10, max(1, int(0.01 * n)))
                top_add = viol_idx[np.argsort(viol_scores)[::-1][:n_add]]
                support[top_add] = True
                a_new[top_add] = 0
                a_new = a_new / a_new.sum()
                a = a_new
                continue

        # Check convergence
        if support.any():
            resid = np.max(np.abs(grad[support] - lam))
        else:
            resid = np.inf
        if resid < 1e-10 * (np.max(np.abs(grad)) + 1e-15):
            break

        a = a_new

    return best_a


def _hankel_mv(a, active_lags, weights, n):
    """Compute H*a where H = sum_k w_k * C_k using FFT.
    (H*a)_i = sum_k w_k * (a[i+k] + a[i-k]) for k>0, plus w_0 * a[i] for k=0.
    """
    out = np.zeros(n)
    for m, w in zip(active_lags, weights):
        if w < 1e-15:
            continue
        if m == 0:
            out += w * a
        else:
            # a[i+m] for i < n-m
            if n - m > 0:
                out[:n - m] += w * a[m:]
            # a[i-m] for i >= m
            if m < n:
                out[m:] += w * a[:n - m]
    return out


def _active_set_iteration(a_init, deadline, time_limit=300, tol_schedule=None):
    """Iterative active-set method with banded QP solves.
    At each iteration:
    1. Compute autocorrelation
    2. Identify active lags (those close to max)
    3. Solve QP on active set
    4. Update
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.004,
                        0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0005, 0.0004,
                        0.0003, 0.0002, 0.00015, 0.0001, 0.00008, 0.00005,
                        0.00004, 0.00003, 0.00002, 0.00001]

    end_time = min(deadline, time.monotonic() + time_limit)
    tol_idx = 0
    rounds_per_tol = 3
    round_count = 0

    for rnd in range(200):
        if time.monotonic() > end_time - 2:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]

        # Identify active lags: convolution indices where c[k] >= c_max * (1 - tol)
        active_mask = c >= c_max * (1.0 - tol)
        n_active = int(active_mask.sum())
        if n_active == 0:
            active_mask[int(np.argmax(c))] = True
            n_active = 1

        # Convert active convolution indices to autocorrelation lags
        center = n - 1
        active_indices = np.where(active_mask)[0]
        active_lags = np.abs(active_indices - center)
        active_lags = np.sort(np.unique(active_lags))

        # Limit bandwidth for efficiency
        max_band = min(int(active_lags.max()), 200)
        active_lags = active_lags[active_lags <= max_band]

        # Compute weights: softmax over active convolution values
        active_vals = c[active_mask]
        beta_w = max((active_vals.max() - active_vals.min()) * 0.3, active_vals.max() * 0.01, 1e-10)
        w_vals = np.exp((active_vals - active_vals.max()) / beta_w)
        w_vals = w_vals / w_vals.sum()

        # Map to lag -> weight (sum weights for same lag)
        from collections import defaultdict
        lag_weight = defaultdict(float)
        for idx, w in zip(active_indices, w_vals):
            m = int(abs(idx - center))
            if m <= max_band:
                lag_weight[m] += w

        active_lags_list = sorted(lag_weight.keys())
        weights_list = [lag_weight[m] for m in active_lags_list]
        # Normalize weights
        w_sum = sum(weights_list)
        if w_sum > 1e-15:
            weights_list = [w / w_sum for w in weights_list]

        # Solve QP
        a_qp = _solve_banded_qp(active_lags_list, weights_list, n, a)
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Damped update
            a_mix = 0.3 * a + 0.7 * a_qp
            a_mix = np.maximum(a_mix, 0)
            a_mix = a_mix / a_mix.sum()
            sc_mix = _score(a_mix)
            if sc_mix < best_sc - 1e-14:
                best_sc = sc_mix
                best_a = a_mix.copy()
                a = a_mix.copy()

        # Advance tolerance
        round_count += 1
        if round_count % rounds_per_tol == 0:
            tol_idx += 1

        # Convergence check
        c = _full_convolve(a, a)
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > n_active:
            gap = c_sorted[n_active - 1] - c_sorted[min(n_active, len(c_sorted) - 1)]
            if c_max > 0 and gap / c_max < 1e-8:
                break

    return best_a, best_sc


def _coordinate_polish(a_init, deadline, time_limit=60):
    """Targeted 1D coordinate descent on the most impactful variables."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    end_time = min(deadline, time.monotonic() + time_limit)

    # Identify most impactful coordinates (those that contribute most to peak)
    c = _full_convolve(a, a)
    peak_idx = int(np.argmax(c))
    center = n - 1
    peak_lag = abs(peak_idx - center)

    # The coordinates most affecting the peak are those near the center
    # and those at distance peak_lag from center
    importance = np.zeros(n)
    for i in range(n):
        # Contribution to peak autocorrelation
        j1 = i + peak_lag
        j2 = i - peak_lag
        contrib = 0
        if 0 <= j1 < n:
            contrib += a[j1]
        if 0 <= j2 < n:
            contrib += a[j2]
        importance[i] = contrib * a[i]

    top_coords = np.argsort(importance)[::-1][:min(200, n // 4)]

    for sweep in range(20):
        if time.monotonic() > end_time - 2:
            break
        for i in top_coords:
            if time.monotonic() > end_time - 1:
                break
            # Optimize a[i] by scanning
            best_val = a[i]
            best_local_sc = best_sc
            # Try scaling a[i]
            for factor in [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5, 2.0]:
                a[i] = best_val * factor
                s = a.sum()
                if s < 1e-15:
                    a[i] = best_val
                    continue
                sc = _score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = a[i]
            a[i] = best_val
            if best_local_sc < best_sc - 1e-15:
                best_sc = best_local_sc
                best_a = a.copy()

    return best_a, best_sc


def _resample_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_tapered(n, power=1.0):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cos_taper(n):
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _perturb_seq(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS initialization from parent
    Phase 2: Active-set banded QP (main optimization)
    Phase 3: L-BFGS polish
    Phase 4: Second active-set pass (tighter)
    Phase 5: Coordinate descent polish
    Phase 6: Multi-n exploration
    Phase 7: Final tight polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_tapered(n_parent, 0.7)
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS from parent ===
    print(f"\n[Phase 1] L-BFGS init (t={time.monotonic()-t_start:.1f}s)")
    a1 = _lbfgs_optimize(parent, deadline, time_limit=200)
    update_best(a1, _score(a1), "lbfgs_init")

    # === Phase 2: Active-set banded QP (main) ===
    print(f"\n[Phase 2] Active-set BQ (t={time.monotonic()-t_start:.1f}s)")
    a2, sc2 = _active_set_iteration(best_seq, deadline, time_limit=400)
    update_best(a2, sc2, "bq_main")

    # === Phase 3: L-BFGS polish ===
    print(f"\n[Phase 3] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    a3 = _lbfgs_optimize(best_seq, deadline, time_limit=150)
    update_best(a3, _score(a3), "lbfgs_polish_1")

    # === Phase 4: Second active-set pass (tighter) ===
    print(f"\n[Phase 4] Active-set BQ tight (t={time.monotonic()-t_start:.1f}s)")
    a4, sc4 = _active_set_iteration(best_seq, deadline, time_limit=300)
    update_best(a4, sc4, "bq_tight")

    # === Phase 5: Coordinate polish ===
    print(f"\n[Phase 5] Coordinate polish (t={time.monotonic()-t_start:.1f}s)")
    a5, sc5 = _coordinate_polish(best_seq, deadline, time_limit=60)
    update_best(a5, sc5, "coord")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    for n_try in [32768, 8192]:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_try = _resample_seq(best_seq, n_try)
        a_try = _lbfgs_optimize(a_try, deadline, time_limit=80)
        sc_try = _score(a_try)
        update_best(a_try, sc_try, f"n_{n_try}_lbfgs")

        if sc_try < best_score + 0.005:
            a_try2, sc_try2 = _active_set_iteration(a_try, deadline, time_limit=100)
            update_best(a_try2, sc_try2, f"n_{n_try}_bq")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_r = len(best_seq)
        # Restart 1: perturbed best
        p1 = _perturb_seq(best_seq, 0.03, rng)
        a_r1 = _lbfgs_optimize(p1, deadline, time_limit=40)
        sc_r1 = _score(a_r1)
        update_best(a_r1, sc_r1, "restart_perturb")
        if sc_r1 < best_score + 0.003:
            a_r1b, sc_r1b = _active_set_iteration(a_r1, deadline, time_limit=40)
            update_best(a_r1b, sc_r1b, "restart_perturb_bq")

        # Restart 2: different profile
        if time.monotonic() < deadline - 40:
            p2 = _make_cos_taper(n_r)
            a_r2 = _lbfgs_optimize(p2, deadline, time_limit=30)
            sc_r2 = _score(a_r2)
            update_best(a_r2, sc_r2, "restart_cos")

    # === Phase 8: Final ultra-tight polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin, sc_fin = _active_set_iteration(
            best_seq, deadline - 2, time_limit=60,
            tol_schedule=[0.0005, 0.0003, 0.0002, 0.0001, 0.00005,
                          0.00003, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001])
        update_best(a_fin, sc_fin, "final_bq")

        if time.monotonic() < deadline - 5:
            a_fin2 = _lbfgs_optimize(best_seq, deadline - 2, time_limit=30)
            update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered(256, 1.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_tapered(8192, 1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
