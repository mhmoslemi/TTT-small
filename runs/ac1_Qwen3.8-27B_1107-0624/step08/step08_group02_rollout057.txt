```python
"""
Optimizer that starts from asymmetric (ramp/power-law) profiles to minimize
2*n*max(conv(a,a))/sum(a)^2. Key insight: asymmetric sequences have much
lower scores than symmetric ones because the convolution peak is shifted
away from the center where it would be largest.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _ramp_profile(n, p=1.0):
    """Power-law ramp: a[i] = (i+1)^p, normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a = a / a.sum()
    return a


def _reverse_ramp_profile(n, p=1.0):
    """Reverse power-law ramp."""
    return _ramp_profile(n, p)[::-1]


def _asym_bump_profile(n, center=0.3, width=0.4, p=2.0):
    """Asymmetric bump."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - center) / width) ** p)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
