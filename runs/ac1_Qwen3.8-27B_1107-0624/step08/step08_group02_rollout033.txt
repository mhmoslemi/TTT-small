"""
Optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses log-sum-exp continuation with adaptive scheduling, active-set Newton polish,
and multi-restart perturbation to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,j] = 2*W[i+j] (Hankel matrix)."""
    v_rev = v[::-1]
    W_rev = W[::-1]
    conv = _fft_convolve(v, W_rev)
    # conv has length n + (2n-1) - 1 = 3n-2
    # We want indices 2n-2, 2n-3, ..., n-1 (reversed)
    return 2.0 * conv[n-1:2*n-1][::-1]


def _hankel_matvec_ones(W, n):
    """H * ones(n) = 2*sum_j W[i+j] for each i."""
    v = np.ones(n)
    return _hankel_matvec(v, W, n)


def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-12):
    """Solve H*x = rhs using CG with restarts, where H is the Hankel matrix."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad_c = _hankel_matvec(a, w, n)  # This is 2*H_a where H is defined by w
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (grad_c / 2.0 - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=200, tight_tol=1e-6):
    """Active-set Newton method for the non-smooth min-max problem."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set tolerance
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
        tol_frac = max(1e-8 * c_max, gap * 0.1)
        active = c >= (c_max - tol_frac)
        n_active = int(active.sum())
        if n_active < 1:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-12 * n:
            break

        # Solve Newton system with constraint sum(dx) = 0
        Hg = _cg_solve_hankel(W, n, g, max_iter=500)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _run_lbfgs_continuation(a_init, deadline, min_remaining=30.0):
    """Log-sum-exp continuation with adaptive beta."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _fft_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    beta = max(gap * 0.5, max_c * 0.01)
    min_beta = max_c * 1e-9

    prev_sc = current_sc
    best_a = current_a.copy()
    best_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 16384.0)
        est_time_per_iter = 0.002 * scale
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if current_sc < best_sc:
                best_sc = current_sc
                best_a = current_a.copy()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _fft_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(50, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-6 * c_max_new and sc_improved > 1e-10:
            beta *= 0.12
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.35

        beta = min(beta, c_max_new * 0.001)

    return best_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.55) ** 0.9
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "antisympert":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bilin_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * (1.0 - 2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quad_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * (1.0 - 3.0 * (t - 0.5) ** 2 * 4.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind =
