    N = _next_pow2(out_n)

    fb = np.fft.rfft(b, N)

    """Compute (H @ v) where H[i,j] = w[j+i] (Hankel matrix from w)."""
    v_rev = v[::-1]
    out_len = len(w) + n - 1
    N = _next_pow2(out_len)
    Fw = np.fft.rfft(w, N)
    Fv = np.fft.rfft(v_rev, N)
    conv = np.fft.irfft(Fw * Fv, N)[:out_len]
    return conv[n-1:2*n-1]

    """Conjugate gradient solve for H @ x = rhs where H is Hankel from w."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x

        Hp = _hankel_matvec(w, p, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break

        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs_new < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new

The objective is F(a) = max_k c[k] where c = conv(a, a).

= a_i * (dF/da_i - sum_j (dF/da_j) * a_j)

Where dF/da = 2 * P_{k*} a (for the subgradient at the argmax lag k*).

So the gradient in z-space is:

This is similar to what the previous program computes in `_smograd`.

    N = _next_pow2(out_len)

def _cg_solve(w, n, rhs, max_iter=300, tol=1e-12):

    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(w, p, n)
        pHp = float(p @ Hp)

        r = r - alpha * Hp

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
    elif kind == "asym_taper2":

    elif kind == "exp_decay":

def _resample(a, n_new):

    x_old = np.linspace(0, 1, n_old)

    s = a_new.sum()

def _subgrad_grad_z(a, c, n, top_k=50):
    c_max = float(c.max())
    threshold = c_max * (1.0 - 0.002)
    active = c >= threshold
    if active.sum() < top_k:

        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    w = np.zeros(2*n-1)
    w[active] = 1.0 / active.sum()
    H_a = _hankel_matvec(w, a, n)

    grad = 2.0 * a * (H_a - W_val)

def _run_subgradient(a_init, deadline, max_iter=5000):

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    H_a = _hankel_matvec(w, a, n)
    W_val = float(np.dot(w, c))
    return 2.0 * n * a * (H_a - W_val)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        scale = max(1.0, n / 8192.0)
        est_iter = 0.002 * scale
        max_iter = min(5000, max(20, int(remaining / (est_iter * 20))))
        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc
        if new_gap < 1e-7 * c_max_new and sc_improved > 1e-13:
            beta *= 0.12
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.3
        beta = min(beta, c_max_new * 0.0003)
    return current_a

    a = a_init.copy()

    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 1:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        for tol_frac in [5e-6, 2e-5, 1e-4, 5e-4, 2e-3]:
            tol = tol_frac * c_max

            n_active = int(active.sum())
            if n_active >= 1:

            n_active = 1
        w = np.zeros(2*n-1)
        w[active] = 1.0 / n_active
        H_a = _hankel_matvec(w, a, n)
        W_val = float(np.dot(w, c))
        g = 2.0 * a * (H_a - W_val)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break
        Hg = _cg_solve(w, n, g, max_iter=300)
        H1 = _cg_solve(w, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-14:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break
        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    step_done = True
                    break
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not step_done and no_improve > 4:
            break
    return best_a, best_sc

def _lp_direction(a, n, M, deadline, min_time_remaining=30.0):
    if n > 12000:
        return None
    try:
        rows = []
        cols = []
        vals = []
        for i in range(n):
            for k in range(max(0, -i), min(n, 2*n-1-i)):
                row_idx = k + i
                if 0 <= row_idx < 2*n-1:
                    rows.append(row_idx)
                    cols.append(i)
                    vals.append(a[i])
        if len(rows) == 0:
            return None
        H_lp = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        c_obj = -np.ones(n)
        res = linprog(
            c_obj,
            A_ub=H_lp, b_ub=M * np.ones(2*n-1),
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': max(1, deadline - min_time_remaining - time.monotonic())}
        )
        if res.status == 0:
            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
        return None
    except Exception:
        return None

def _run_lp_iteration(a_init, deadline, min_time_remaining=60.0, max_iters=5):
    n = len(a_init)
    if n > 12000:
        return a_init
    current_a = a_init.copy()
    current_sc = _score(current_a)
    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(current_a, current_a)
        M = float(c.max())
        g = _lp_direction(current_a, n, M, deadline, min_time_remaining)
        if g is None:
            break
        best_a = current_a.copy()
        best_local_sc = current_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * current_a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_a = a_try.copy()
        if best_local_sc < current_sc - 1e-12:
            current_a = best_a
            current_sc = best_local_sc
        else:
            break
    return current_a

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    print(f"\n[Phase 1] Profile exploration")
    profiles = ["tapered", "asym_taper", "asym_taper2", "left_heavy", "right_heavy",
                "gaussian_off", "bimodal", "exp_decay", "step_asym", "cos"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)
    for prof_kind in profiles:
        if time.monotonic() > deadline - 600:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': {sc_p:.8f}")
    for sigma in [0.1, 0.3, 0.5]:
        if time.monotonic() > deadline - 500:
            break
        a_pert = _perturb(best_profile_a, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_profile_sc:
            best_profile_sc = sc_pert
            best_profile_a = a_pert
    if time.monotonic() < deadline - 400:
        a_quick = _run_lbfgs(best_profile_a, min(deadline - 400, time.monotonic() + 80), min_time_remaining=300.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_profile")
        if sc_quick < best_profile_sc:
            best_profile_a = a_quick
            best_profile_sc = sc_quick

    start_a = best_seq if best_seq is not None else best_profile_a

    print(f"\n[Phase 2] Subgradient descent")
    if time.monotonic() < deadline - 400:
        a_sg, sc_sg = _run_subgradient(start_a, min(deadline - 400, time.monotonic() + 200), max_iter=3000)
        update_best(a_sg, sc_sg, "subgrad")

    print(f"\n[Phase 3] Main L-BFGS")
    if time.monotonic() < deadline - 300:
        a_main = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0)
        update_best(a_main, _score(a_main), "lbfgs_main")

    print(f"\n[Phase 4] Newton polish")
    if time.monotonic() < deadline - 200:
        a_np, sc_np = _newton_polish(best_seq, min(deadline - 200, time.monotonic() + 120), max_rounds=60)
        update_best(a_np, sc_np, "newton_1")

    print(f"\n[Phase 5] LP iteration")
    if best_seq is not None and len(best_seq) <= 12000 and time.monotonic() < deadline - 150:
        a_lp = _run_lp_iteration(best_seq, min(deadline - 150, time.monotonic() + 120), max_iters=3)
        update_best(a_lp, _score(a_lp), "lp")
        if _score(a_lp) < best_score + 0.001 and time.monotonic() < deadline - 100:
            a_lp_np, sc_lp_np = _newton_polish(a_lp, min(deadline - 100, time.monotonic() + 60), max_rounds=30)
            update_best(a_lp_np, sc_lp_np, "lp_newton")

    print(f"\n[Phase 6] Cycle optimization")
    cycle_time = 40.0
    for cycle in range(8):
        if time.monotonic() + 100 > deadline:
            break
        t_end = min(time.monotonic() + cycle_time, deadline - 80)
        if time.monotonic() < t_end - 10:
            a_c = _run_lbfgs(best_seq, t_end - 10, min_time_remaining=20.0)
            update_best(a_c, _score(a_c), f"cycle_lbfgs_{cycle}")
        if time.monotonic() < t_end:
            a_c2, sc_c2 = _newton_polish(best_seq, t_end, max_rounds=20)
            update_best(a_c2, sc_c2, f"cycle_newton_{cycle}")
        cycle_time = min(cycle_time * 1.5, 100.0)

    print(f"\n[Phase 7] Multi-n exploration")
    n_candidates = sorted(set([8192, 16384, 4096, 32768, 65536]),
                          key=lambda x: abs(x - len(best_seq)))
    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 60:
            break
        a_init_n = _resample(best_seq, n_try)
        a_opt_n = _run_lbfgs(a_init_n, deadline, min_time_remaining=40.0)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")
        if sc_opt_n < best_score + 0.002 and time.monotonic() < deadline - 20:
            a_np_n, sc_np_n = _newton_polish(a_opt_n, min(deadline - 20, time.monotonic() + 20), max_rounds=20)
            update_best(a_np_n, sc_np_n, f"n_{n_try}_np")

    print(f"\n[Phase 8] Diverse restarts")
    if time.monotonic() < deadline - 30:
        for i, rp in enumerate(["asym_taper2", "gaussian_off", "right_heavy"]):
            if time.monotonic() > deadline - 15:
                break
            if i < 2:
                p_start = _perturb(best_seq, 0.03 + 0.03 * i, rng)
            else:
                p_start = _make_profile(len(best_seq), rp)
            if time.monotonic() > deadline - 10:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=5.0)
            update_best(a_r, _score(a_r), f"restart_{i}")
            if _score(a_r) < best_score + 0.001 and time.monotonic() < deadline - 5:
                a_rp, sc_rp = _newton_polish(a_r, deadline - 2, max_rounds=20)
                update_best(a_rp, sc_rp, f"restart_{i}_np")

    print(f"\n[Phase 9] Final ultra-fine polish")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 1, max_rounds=100)
        update_best(a_fin, sc_fin, "final_newton")
        if time.monotonic() < deadline - 1:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-11
            remaining = deadline - 1.5 - time.monotonic()
            if remaining > 1:
                max_iter = min(1000, max(10, int(remaining / 0.02)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 100, 'maxls': 150}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_vf, sc_vf = _newton_polish(best_seq, deadline - 0.1, max_rounds=30)
            update_best(a_vf, sc_vf, "very_final")

    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
