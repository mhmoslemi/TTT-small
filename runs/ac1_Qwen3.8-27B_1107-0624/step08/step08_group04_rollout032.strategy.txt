        candidate_peaks_opt[:max_num_peaks//2] = 1.0
        candidate_peaks_opt[0:max_num_peaks, :] = peak_amplitudes

        best_candidate[0:max_num_peaks, :] = candidate_peaks_opt
        best_candidate[clear_window:max_num_peaks] = best_candidate[clear_window] * (1 - 1e-8) ** (len(range(clear_window, max_num_peaks)))
    else:
        best_candidate = final_candidate

    return best_candidate.tolist()
