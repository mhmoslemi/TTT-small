def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    def __call__(self, z):

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)

def _active_set_polish_v2(a_init, deadline, tol_start=0.005, tol_end=0.00001,

def _peak_shave(a, deadline, n_trials=200):
    """Direct peak shaving: identify top contributors and reduce them."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find the elements contributing most to the peak
        # conv(a,a)[k] = sum_i a[i] * a[k-i]
        # Gradient wrt a[i] is a[k-i] (approximately)
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo >= k_hi:
            idx = int(np.argmax(a))
            contrib = a.copy() * a.copy()
            # Reduce top contributor
            a_try = a.copy()
            a_try[idx] *= 0.95
            # Distribute the reduction to others
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            continue

        # Contribution vector: a[i] * a[k_star - i]
        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            continue

        # Reduce the top contributors
        frac = rng.uniform(0.01, 0.08)
        n_reduce = max(1, int(len(contrib) * frac))
        top_contrib_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_contrib_idx:
            red = a_try[idx] * 0.1
            a_try[idx] -= red
            reduction += red

        # Distribute reduction to low-contribution elements
        low_mask = np.zeros(n, dtype=bool)
        low_mask[:] = True
        for idx in top_contrib_idx:
            low_mask[idx] = False
        j_set = set(int(j) for j in j_range)
        for idx in top_contrib_idx:
            j_val = k_star - idx
            if 0 <= j_val < n:
                low_mask[j_val] = False

        if low_mask.sum() == 0:
            low_mask = np.ones(n, dtype=bool)
            for idx in top_contrib_idx:
                low_mask[idx] = False
            if low_mask.sum() == 0:
                continue

        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + np.exp(-20.0 * (t - 0.85) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.arange(1, n + 1)
        a = 1.0 / np.sqrt(t.astype(float))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
