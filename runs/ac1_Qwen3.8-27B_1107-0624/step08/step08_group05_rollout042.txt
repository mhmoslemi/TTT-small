```python
def gs_flatten(a, n_iters=1000, target='flat'):
    n = len(a)
    N = next_power_of_2(2*n - 1)
    a = a.copy()
    for it in range(n_iters):
        # Compute DFT with zero padding
        A = np.fft.fft(a, N)
        # Modify amplitude to flatten spectrum
        amp = np.abs(A)
        phase = np.angle(A)
        # Target: flat amplitude
        if target == 'flat':
            target_amp = np.ones(N) * amp.mean()
        else:
            target_amp = amp  # no change
        # Blend
        A_new = target_amp * np.exp(1j * phase)
        # Back to time domain
        a_new = np.fft.ifft(A_new, N)[:n].real
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    return a
```

```python
    idx = np.argsort(c)[::-1][:k_active]
    w = np.zeros(len(c))

    # Gradient: grad[i] = sum_j w[j] * a[j-i] (for valid j-i)
```

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(5000, max(200, int(remaining / (est_iter_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _hess_matvec(v, active_arr, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_arr, N)
    r = np.fft.irfft(fa * fb, N)
    return 2.0 * r[n - 1:2 * n - 1]


def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha < 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[np.argmax(c)] = True

        active_f = active.astype(np.float64)
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=300)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        accepted = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            if alpha < 1e-14:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _peak_shave(a_init, deadline, n_steps=20000, k_active=20, lr_init=0.005):
    """Iteratively reduce the peak of the autocorrelation via subgradient steps."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_act = min(k_active, len(c))
        top_idx = np.argsort(c)[::-1][:k_act]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k_act

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = wc[n - 1:2 * n - 1]
        grad = grad - grad.mean()

        step = lr / (1.0 + 0.0001 * i)
        a = a - step * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if i % 500 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                patience = 0
                best_patience_sc = sc
            else:
                patience += 1
                if patience > 15:
                    a = best_a.copy()
                    lr *= 0.8
                    patience = 0
                    if lr < 1e-6:
                        break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _spectral_shape(a_init, deadline, n_iters=5000, blend=0.3):
    """
    Iteratively shape the power spectrum to reduce autocorrelation peak.
    Alternates between time-domain projection (non-negativity) and
    frequency-domain amplitude modification.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    # Target: the amplitude spectrum of a "flat autocorrelation"
    # Flat autocorrelation of length 2n-1 -> DFT is Dirichlet kernel
    target_c = np.ones(2 * n - 1) / (2 * n - 1)
    target_c_padded = np.zeros(N)
    target_c_padded[:2 * n - 1] = target_c
    target_spectrum = np.fft.rfft(target_c_padded)
    target_amp = np.abs(target_spectrum)
    target_amp[target_amp < 1e-10] = 1e-10

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        # Time to frequency
        A = np.fft.rfft(a, N)
        amp = np.abs(A)
        phase = np.angle(A)

        # Blend current amplitude toward target
        # Use geometric mean for smooth blending
        amp_new = amp ** (1 - blend) * target_amp ** blend
        # Ensure DC component preserves the sum
        amp_new[0] = amp[0]
        if N % 2 == 0:
            amp_new[-1] = amp[-1]

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new

        if it % 500 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _spectral_shape_v2(a_init, deadline, n_iters=10000):
    """
    Spectral shaping with adaptive target based on current peak structure.
    The target spectrum is modified to specifically counteract the peak.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Target: make c more flat
        # Current c has peak at c_max. We want c[k] closer to s^2/(2n-1).
        # In frequency domain, c = IDFT(|A|^2).
        # To reduce the peak, we want |A|^2 to be more concentrated at DC.
        A = np.fft.rfft(a, N)
        amp = np.abs(A)
        phase = np.angle(A)

        # Compress the spectrum: reduce high-frequency components
        # Use a soft high-frequency cutoff
        freqs = np.arange(len(amp))
        # Gaussian window in frequency: more weight at low freqs
        sigma_f = max(5, N // (4 * n))
        weight = np.exp(-(freqs ** 2) / (2 * sigma_f ** 2))
        # Blend: amp * (alpha + (1-alpha) * weight)
        alpha = 0.3 + 0.4 * (it / max(1, n_iters))  # gradually increase flatness
        amp_mod = amp * (alpha + (1 - alpha) * weight)
        amp_mod[0] = amp[0]  # preserve sum
        if N % 2 == 0 and len(amp_mod) > 1:
            amp_mod[-1] = amp_mod[0]

        # Also try to equalize: reduce peaks in |A|^2
        amp_sq = amp_mod ** 2
        mean_amp_sq = np.mean(amp_sq[1:]) if len(amp_sq) > 1 else 1e-10
        # Soft clipping of amplitude
        clip_factor = np.minimum(1.0, np.sqrt(mean_amp_sq * 3.0) / np.maximum(amp_mod, 1e-30))
        amp_mod = amp_mod * (0.7 + 0.3 * clip_factor)
        amp_mod[0] = amp[0]

        A_new = amp_mod * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new

        if it % 1000 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_steps=10000, lr_init=0.008, tol_frac=0.002):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_for_restart(a, rng, mode=0):
    n = len(a)
    if mode == 0:
        z = np.log(np.maximum(a, 1e-30))
        z = z + 0.03 * rng.standard_normal(n)
        z = z - z.mean()
        return _softmax(z)
    elif mode == 1:
        # Fourier phase perturbation
        A = np.fft.rfft(a)
        phase_pert = rng.uniform(-0.3, 0.3, len(A))
        phase_pert[0] = 0.0
        if n % 2 == 0:
            phase_pert[-1] = 0.0
        A_new = A * np.exp(1j * phase_pert)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        return a_new / s if s > 1e-10 else a.copy()
    elif mode == 2:
        # Random block swap
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        n_swaps = 4
        block_size = max(8, n // (4 * n_swaps))
        for _ in range(n_swaps):
            i = int(rng.integers(0, n - 2 * block_size))
            j = int(rng.integers(0, n - 2 * block_size))
            if abs(i - j) < block_size:
                continue
            tmp = z[i:i + block_size].copy()
            z[i:i + block_size] = z[j:j + block_size]
            z[j:j + block_size] = tmp
        return _softmax(z)
    elif mode == 3:
        # Amplitude perturbation
        A = np.fft.rfft(a)
        amp = np.abs(A)
        amp_pert = amp * (1.0 + 0.05 * rng.standard_normal(len(amp)))
        amp_pert = np.maximum(amp_pert, 0)
        phase = np.angle(A)
        A_new = amp_pert * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        return a_new / s if s > 1e-10 else a.copy()
    else:
        return a.copy()


def _alternating_optimize(a_init, deadline, n_rounds=10):
    """Alternate between LSE, peak shaving, and Newton."""
    current_a = a_init.copy()
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # LSE step
        lse_dead = min(deadline, time.monotonic() + 60)
        a_lse, sc_lse = _lbfgs_lse(current_a, lse_dead, min_remaining=5)
        if sc_lse < current_sc:
            current_a = a_lse
            current_sc = sc_lse

        # Peak shave step
        ps_dead = min(deadline, time.monotonic() + 30)
        a_ps, sc_ps = _peak_shave(current_a, ps_dead, n_steps=3000, k_active=30, lr_init=0.003)
        if sc_ps < current_sc:
            current_a = a_ps
            current_sc = sc_ps

        # Newton step
        nw_dead = min(deadline, time.monotonic() + 20)
        a_nw, sc_nw = _newton_active_set(current_a, nw_dead, max_iter=100, tol_frac=0.002)
        if sc_nw < current_sc:
            current_a = a_nw
            current_sc = sc_nw

        if current_sc < best_sc:
            best_sc = current_sc
            best_a = current_a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer: spectral shaping + LSE L-BFGS + peak shaving + Newton.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init")

    # === Phase 1: Quick diverse exploration (40s) ===
    print("\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 40)
    n_expl = max(2048, n_parent // 4) if n_parent > 4096 else 2048
    for kind in ["uniform", "tapered", "asym_ramp", "asym_ramp2", "bump_off_center",
                 "chirp_rect", "double_bump"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 2: Spectral shaping on best (80s) ===
    print("\n[Phase 2] Spectral shaping")
    if _best_seq is not None and time.monotonic() < deadline - 600:
        a_spec, sc_spec = _spectral_shape(_best_seq, min(deadline, t_start + 120), n_iters=3000, blend=0.2)
        upd(a_spec, sc_spec, "spectral_v1")

        a_spec2, sc_spec2 = _spectral_shape_v2(_best_seq, min(deadline, t_start + 200), n_iters=5000)
        upd(a_spec2, sc_spec2, "spectral_v2")

    # === Phase 3: Main LSE optimization (400s) ===
    print("\n[Phase 3] Main LSE L-BFGS")
    if time.monotonic() < deadline - 400:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=350)
        upd(a_main, sc_main, "lbfgs_main")

    # === Phase 4: Peak shaving (200s) ===
    print("\n[Phase 4] Peak shaving")
    if time.monotonic() < deadline - 200:
        ps_dead = min(deadline - 150, time.monotonic() + 200)
        a_ps, sc_ps = _peak_shave(_best_seq, ps_dead, n_steps=30000, k_active=50, lr_init=0.004)
        upd(a_ps, sc_ps, "peak_shave")

    # === Phase 5: Newton refinement (150s) ===
    print("\n[Phase 5] Newton refinement")
    if time.monotonic() < deadline - 150:
        nw_dead = min(deadline - 120, time.monotonic() + 150)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=500, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 6: Subgradient polish (150s) ===
    print("\n[Phase 6] Subgradient polish")
    if time.monotonic() < deadline - 120:
        sg_dead = min(deadline - 100, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=15000, lr_init=0.006, tol_frac=0.001)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 7: Alternating optimization (200s) ===
    print("\n[Phase 7] Alternating optimization")
    if time.monotonic() < deadline - 150:
        alt_dead = min(deadline - 100, time.monotonic() + 200)
        a_alt, sc_alt = _alternating_optimize(_best_seq, alt_dead, n_rounds=6)
        upd(a_alt, sc_alt, "alternating")

    # === Phase 8: Multi-n exploration (300s) ===
    print("\n[Phase 8] Multi-n exploration")
    n_candidates = [4096, 8192, 12288, 16384, 20480, 24576]
    if n_parent not in n_candidates and 2048 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if _best_seq is not None and len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        elif _best_seq is not None:
            a_init = _resize_sequence(_best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        # LSE
        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, deadline, min_remaining=40)
        upd(a_opt_n, sc_opt_n, f"n{n_try}_lse")

        # Peak shave
        if time.monotonic() < deadline - 30:
            ps_d = min(deadline - 20, time.monotonic() + 60)
            a_ps_n, sc_ps_n = _peak_shave(a_opt_n, ps_d, n_steps=5000, k_active=30, lr_init=0.003)
            upd(a_ps_n, sc_ps_n, f"n{n_try}_ps")

        # Newton
        if time.monotonic() < deadline - 20:
            nw_d = min(deadline - 15, time.monotonic() + 40)
            a_nw_n, sc_nw_n = _newton_active_set(a_opt_n, nw_d, max_iter=100, tol_frac=0.002)
            upd(a_nw_n, sc_nw_n, f"n{n_try}_nw")

    # === Phase 9: Diverse restarts (150s) ===
    print("\n[Phase 9] Diverse restarts")
    if _best_seq is not None and time.monotonic
