"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses SMA+L-BFGS for coarse optimization, Newton active-set for refinement,
multi-scale exploration, and diverse restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs_new < tol * tol * rs0:
            break
        p = r + (rs_new / max(rs, 1e-30)) * p
        rs = rs_new
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_polish(a_init, deadline, max_rounds=200):
    """Newton active-set method with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: start tight, widen if stuck
        tol_fracs = [0.000005, 0.00002, 0.0001, 0.0005, 0.002]
        if no_improve > 10:
            tol_fracs = [0.001, 0.005, 0.01, 0.05]

        active = None
        for tol_frac in tol_fracs:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 1:
                break
        if active is None or active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True

        n_active = int(active.sum())
        W = np.zeros(len(c))
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with sufficient decrease
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-18:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Try subgradient steps
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-18:
                    a = a_try
                    step_done = True
                    break
            if not step_done:
                # Try damped Newton
                for damp in [0.5, 0.25, 0.1]:
                    a_try = a + damp * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max - 1e-18:
                        a = a_try
                        step_done = True
                        break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 15:
            break

    return best_a, best_sc

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_a  # placeholder
    prev_sc = current_sc

        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:

        scale = max(1.0, n / 8192.0)

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.9
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0.001, 0.999, n)
        a = t ** 0.5 * (1 - t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "dpss_like":
        # Approximate DPSS using a carefully shaped window
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8 * (1 + 0.1 * np.cos(4 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = np.sqrt(np.sin(np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_nesterov(a_init, deadline, max_steps=5000):
    """Nesterov-accelerated subgradient method for diversification."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()  # Nesterov lookahead
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr_base = 0.02
    momentum = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(y)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr_base / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()

        # Nesterov update
        y = z_new + momentum * (z_new - z)
        z = z_new
        y = y - y.mean()

        if i % 50 == 49:
            a_check = _softmax(y)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr_base = min(lr_base * 1.03, 0.05)
                momentum = min(momentum + 0.005, 0.95)
            else:
                patience += 1
                lr_base *= 0.8
                if patience > 80:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    y = z.copy()
                    patience = 0
                    lr_base = 0.015
                    momentum = 0.9

    a_final = _softmax(y)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_polish(a_init, deadline, max_rounds=50):
    """Coordinate descent on the most impactful variables."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_sc = _score(a)
    best_a = a.copy()

    c = _full_convolve(a, a)
    c_max = float(c.max())
    peak_idx = int(np.argmax(c))

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Find the peak and determine which a[i] contribute most
        # c[k] = sum_i a[i]*a[k-i], so a[i] contributes to c[k] via a[i]*a[k-i]
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Try adjusting the top contributors
        top_k = min(20, n)
        top_idx = np.argsort(contrib)[::-1][:top_k]

        improved = False
        for idx in top_idx:
            # Try small multiplicative adjustment
            for factor in [0.95, 0.9, 1.1, 1.15]:
                a_try = a.copy()
                a_try[idx] *= factor
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-16:
                    a = a_try
                    best_sc = sc_try
                    best_a = a_try.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick profile exploration ===
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    best_init = parent.copy()
    best_init_sc = _score(parent)

    profiles = ["tapered", "gaussian", "dpss_like", "wedge", "power_law", "asym_taper", "cos"]
    for prof_kind in profiles:
        if time.monotonic() > deadline - 1000:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        print(f"  Profile '{prof_kind}': score={sc_p:.10f}")
        if sc_p < best_init_sc:
            best_init_sc = sc_p
            best_init = a_p

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 900:
        t_phase2_end = min(deadline - 800, time.monotonic() + 150)
        a_quick = _run_lbfgs(best_init, t_phase2_end, min_time_remaining=700.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_lbfgs")
        if sc_quick < best_init_sc:
            best_init = a_quick
            best_init_sc = sc_quick

    # === Phase 2: Main L-BFGS optimization ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_init
    t_phase2 = min(deadline - 500, time.monotonic() + 400)
    a_main = _run_lbfgs(start_a, t_phase2, min_time_remaining=400.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # === Phase 3: Newton polish ===
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        t_np = min(deadline - 350, time.monotonic() + 200)
        a_np, sc_np = _newton_polish(best_seq, t_np, max_rounds=200)
        update_best(a_np, sc_np, "newton_1")

    # === Phase 4: Specialized Newton with different tolerances ===
    print(f"\n[Phase 4] Multi-tolerance Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        for trial in range(3):
            if time.monotonic() > deadline - 200:
                break
            a_pert = _perturb(best_seq, 0.01 * (trial + 1), rng)
            t_np2 = min(deadline - 180, time.monotonic() + 80)
            a_np2, sc_np2 = _newton_polish(a_pert, t_np2, max_rounds=100)
            update_best(a_np2, sc_np2, f"newton_pert_{trial}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192, 4096]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        base = best_seq if best_seq is not None else parent
        a_resampled = _resample(base, n_try)
        sc_res = _score(a_resampled)
        print(f"  n={n_try}: initial score={sc_res:.10f}")

        # L-BFGS on resampled
        t_lb = min(deadline - 150, time.monotonic() + 120)
        a_opt = _run_lbfgs(a_resampled, t_lb, min_time_remaining=100.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        # Newton on the best at this n
        if time.monotonic() < deadline - 80:
            best_at_n = a_opt if _score(a_opt) < (best_score if best_seq is None else _score(_resample(best_seq, n_try))) else a_resampled
            t_np3 = min(deadline - 60, time.monotonic() + 60)
            a_np3, sc_np3 = _newton_polish(best_at_n, t_np3, max_rounds=80)
            update_best(a_np3, sc_np3, f"n_{n_try}_newton")

    # === Phase 6: Diverse restarts ===
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        for i in range(5):
            if time.monotonic() > deadline - 50:
                break
            sigma = 0.03 + 0.04 * i
            a_r = _perturb(best_seq, sigma, rng)
            # Quick L-BFGS
            t_lr = min(deadline - 30, time.monotonic() + 50)
            a_opt_r = _run_lbfgs(a_r, t_lr, min_time_remaining=20.0)
            sc_r = _score(a_opt_r)
            if sc_r < best_score + 0.005:
                update_best(a_opt_r, sc_r, f"restart_{i}")
                # Newton if promising
                if sc_r < best_score + 0.002 and time.monotonic() < deadline - 15:
                    a_nr, sc_nr = _newton_polish(a_opt_r, deadline - 5, max_rounds=50)
                    update_best(a_nr, sc_nr, f"restart_{i}_np")

    # === Phase 7: Coordinate polish ===
    print(f"\n[Phase 7] Coordinate polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_cp, sc_cp = _coordinate_polish(best_seq, min(deadline - 20, time.monotonic() + 30), max_rounds=30)
        update_best(a_cp, sc_cp, "coord")

    # === Phase 8: Final aggressive Newton ===
    print(f"\n[Phase 8] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 2, max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

        # One more L-BFGS at ultra-small beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = max(mb * 1e-12, 1e-15)
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                try:
                    max_iter = min(3000, max(20, int(remaining / 0.02)))
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-17,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 1:
            a_vf, sc_vf = _newton_polish(best_seq, deadline - 0.1, max_rounds=80)
            update_best(a_vf, sc_vf, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    return [float(x) for x in fallback]
