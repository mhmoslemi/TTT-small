        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    # Also try optimizing a 3-harmonic profile with scipy
    if time.monotonic() < deadline - 5:
        def _prof_obj(params):
            c1, c2, c3, p = params
            a = _make_harmonic_profile(n, [c1, c2, c3], p)
            return _score(a)
        
        # Try a few starting points
        starts = [
            [0.2, 0.05, 0.01, 1.0],
            [0.3, -0.1, 0.05, 1.0],
            [0.5, 0.1, -0.05, 1.0],
            [0.1, 0.1, 0.05, 0.7],
        ]
        for x0 in starts:
            if time.monotonic() > deadline:
                break
            try:
                res = minimize(_prof_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-5, 'fatol': 1e-7})
                if res.fun < best_sc:
                    best_sc = res.fun
                    c1, c2, c3, p = res.x
                    best_a = _make_harmonic_profile(n, [c1, c2, c3], p)
            except Exception:
                pass
    
    print(f"  Profile search: best={best_sc:.10f}")
    return best_a, best_sc


def _adaptive_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """Adaptive L-BFGS continuation with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Compute gap between top peaks for initial beta
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 2, max_c * 0.05)
    
    # Start with large beta, decrease geometrically
    beta = base_beta
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    min_beta = max_c * 0.00003
    beta_ratio = 0.4  # geometric ratio
    
    while time.monotonic() + min_time_remaining < deadline:
        if beta < min_beta:
            beta = min_beta
            ratio = 0.5  # slow down near the end
        else:
            ratio = beta_ratio
        
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        z_old = z.copy()
        
        try:
            result = minimize(
                _lsobj, z, args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            
            # Check if solution changed significantly
            change = np.linalg.norm(a_new - current_a)
            
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # No improvement, try smaller beta anyway
                z = z_old  # revert
        
        except Exception:
            pass
        
        # Adapt beta: if change was small, decrease faster
        if change < 1e-4:
            beta *= 0.3
        elif change < 1e-3:
            beta *= 0.4
        else:
            beta *= ratio
        
        if time.monotonic() % 60 < est_time_per_iter * max_iter:
            print(f"    L-BFGS: beta={beta:.2e}, score={current_sc:.10f}, "
                  f"change={change:.2e}, t={time.monotonic()-t_start_global:.1f}s")
    
    return current_a


def _dual_flow(a_init, deadline, n_iter=5000):
    """
    Dual-flow peak balancing: iteratively redistribute mass from
    high-pressure to low-pressure regions using the convolution structure.
    Different from simple deflation - uses the full convolution gradient.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()
    
    # Adaptive step size
    eta = 0.05
    n_no_improve = 0
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Compute the "pressure" field: gradient of max(c) w.r.t. a
        # Using top-K peaks for robustness
        K = max(5, min(len(c), n // 4))
        top_idx = np.argsort(c)[::-1][:K]
        
        # Weight the top peaks
        c_top = c[top_idx]
        c_max_local = c_top[0]
        # Soft weights: favor higher peaks
        w = np.exp((c_top - c_max_local) / (0.01 * c_max_local + 1e-15))
        w = w / w.sum()
        
        # Compute gradient: for each peak k, dc_k/da_i = 2*a_{k-i}
        # Weighted sum: grad_i = 2 * sum_k w_k * a_{k-i}
        # This is: 2 * (w_ext * a_rev) where w_ext is the extended weight vector
        w_ext = np.zeros(2 * n - 1)
        w_ext[top_idx] = w
        a_rev = a[::-1]
        grad_full = _full_convolve(w_ext, a_rev)
        grad = grad_full[n - 1:2 * n - 1] * 2.0
        
        # Project gradient onto the tangent space (sum = 0)
        grad_t = grad - np.mean(grad)
        
        # Step: move opposite to gradient (reduce peaks)
        # With non-negativity constraint
        a_new = a - eta * grad_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            n_no_improve = 0
            # Slightly increase step if making good progress
            eta = min(eta * 1.02, 0.3)
        else:
            n_no_improve += 1
            if n_no_improve > 20:
                eta *= 0.7
                n_no_improve = 0
            if eta < 1e-6:
                break
        
        if it % 500 == 0:
            print(f"    Dual-flow it={it}, score={sc_new:.10f}, eta={eta:.4f}")
    
    return best_a


def _peak_equalization(a_init, deadline):
    """
    Iteratively equalize peaks by solving a small QP at each step.
    Focus on the top few peaks and find the optimal redistribution.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(50):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find the top peaks (within 5% of max)
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # For each active peak k, the contribution to a_i is through a_{k-i}
        # We want to find da (sum=0, da>=-a) that minimizes max_k (c_k + 2*(a*da)_k)
        
        # Approximate: minimize the weighted sum of changes at active peaks
        # The change at peak k is: 2 * sum_i da_i * a_{k-i}
        # So we want: minimize sum_k w_k * 2 * sum_i da_i * a_{k-i} = minimize 2 * sum_i da_i * (sum_k w_k * a_{k-i})
        # Subject to: sum da_i = 0, da_i >= -a_i
        
        # The optimal da_i is proportional to max(0, threshold - p_i) / a_i type solution
        # where p_i = sum_k w_k * a_{k-i}
        
        # Equal weights for active peaks
        w = np.ones(len(active)) / len(active)
        
        # Compute pressure: p_i = sum_k w_k * a_{k-i}
        w_ext = np.zeros(2 * n - 1)
        w_ext[active] = w
        a_rev = a[::-1]
        p = _full_convolve(w_ext, a_rev)[n - 1:2 * n - 1]
        
        # Water-filling: da_i = -eta * (p_i - p_median) clipped
        p_med = np.median(p)
        da = -0.1 * (p - p_med)
        
        # Clip for non-negativity
        neg_mask = da < -a
        if neg_mask.any():
            da[neg_mask] = -a[neg_mask]
        
        # Fix sum
        da_sum = da.sum()
        if abs(da_sum) > 1e-15:
            # Distribute the correction evenly where possible
            free = da > -a + 1e-10
            if free.any():
                da[free] -= da_sum / free.sum()
            else:
                da -= da_sum / n
        
        a_new = a + da
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        
        if outer % 10 == 0:
            print(f"    Peak-eq iter={outer}, score={sc_new:.10f}, n_active={len(active)}")
    
    return best_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _active_set_gd(a_init, deadline, max_iter=2000):
    """
    Gradient descent with adaptive active set selection.
    At each step, choose the active set based on current peaks,
    compute the exact Newton direction for that set, and take a line search step.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    def _corr(v, A_vec):
        """Correlation of v with A_vec, extracted to length n."""
        out_len = n + len(A_vec) - 1
        N = _next_pow2(out_len)
        fv = np.fft.rfft(v[::-1], N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fv * fA, N)[:out_len]
        return result[n - 1:2 * n - 1]
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance
        tol = max(0.001 * c_max, (c_max - np.median(c)) * 0.1)
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())
        
        if m < 2:
            active = np.zeros_like(c, dtype=bool)
            top_k = np.argsort(c)[::-1][:max(5, n // 100)]
            active[top_k] = True
            m = len(top_k)
        
        A_vec = active.astype(np.float64)
        
        # Gradient of the active-set average
        g = (2.0 / m) * _corr(a, A_vec)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        
        if gnorm < 1e-14:
            break
        
        # Simple step with backtracking line search
        # Direction: -g_t / ||g_t|| (normalized gradient descent)
        d = -g_t / gnorm
        
        t = 0.1
        for ls in range(30):
            a_try = a + t * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            
            if f_try <= c_max - 1e-12:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            t *= 0.5
        else:
            # Try smaller fixed step
            a_try = a + 0.01 * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
        
        if it % 200 == 0:
            sc_cur = _score(a)
            print(f"    AS-GD it={it}, score={sc_cur:.10f}, c_max={c_max:.6e}, m={m}")
    
    return best_a

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Low-dim profile search for good initialization
    2. Adaptive L-BFGS continuation
    3. Dual-flow peak balancing
    4. Active-set gradient descent
    5. Peak equalization
    6. Final L-BFGS polish
    7. Multi-n exploration and restarts
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start_global:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # Phase 1: Low-dim profile search (quick)
    print(f"\n[Phase 1] Profile search (t={time.monotonic() - t_start_global:.1f}s)")
    n_main = n_parent
    profile_dl = min(deadline, t_start_global + 60)
    prof_a, prof_sc = _profile_search(n_main, profile_dl)
    update_best(prof_a, prof_sc, "profile")
    
    # Choose the best starting point
    a_start = best_seq.copy() if best_seq is not None else parent
    
    # Phase 2: Adaptive L-BFGS continuation
    print(f"\n[Phase 2] Adaptive L-BFGS (t={time.monotonic() - t_start_global:.1f}s)")
    lbfgs_dl = deadline - 800  # leave 800s for other phases
    a_lbfgs = _adaptive_lbfgs(a_start, lbfgs_dl, min_time_remaining=200.0)
    sc_lbfgs = _score(a_lbfgs)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs")
    
    # Phase 3: Dual-flow peak balancing
    print(f"\n[Phase 3] Dual-flow (t={time.monotonic() - t_start_global:.1f}s)")
    if time.monotonic() < deadline - 400:
        df_dl = min(deadline - 400, t_start_global + time.monotonic() - t_start_global + 120)
        a_df = _dual_flow(a_lbfgs, df_dl, n_iter=8000)
        sc_df = _score(a_df)
        update_best(a_df, sc_df, "dual_flow")
        if sc_df < best_score:
            a_start = a_df
        else:
            a_start = a_lbfgs
    
    # Phase 4: Active-set gradient descent
    print(f"\n[Phase 4] AS-GD (t={time.monotonic() - t_start_global:.1f}s)")
    if time.monotonic() < deadline - 300:
        asgd_dl = min(deadline - 300, time.monotonic() + 100)
        a_asgd = _active_set_gd(a_start, asgd_dl, max_iter=3000)
        sc_asgd = _score(a_asgd)
        update_best(a_asgd, sc_asgd, "as_gd")
        if sc_asgd < best_score:
            a_start = a_asgd
    
    # Phase 5: Peak equalization
    print(f"\n[Phase 5] Peak equalization (t={time.monotonic() - t_start_global:.1f}s)")
    if time.monotonic() < deadline - 250:
        peq_dl = min(deadline - 250, time.monotonic() + 60)
        a_peq = _peak_equalization(a_start, peq_dl)
        sc_peq = _score(a_peq)
        update_best(a_peq, sc_peq, "peak_eq")
        if sc_peq < best_score:
            a_start = a_peq
    
    # Phase 6: Final L-BFGS polish
    print(f"\n[Phase 6] Final L-BFGS polish (t={time.monotonic() - t_start_global:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_final = _adaptive_lbfgs(a_start, deadline, min_time_remaining=30.0)
        sc_final = _score(a_final)
        update_best(a_final, sc_final, "final_lbfgs")
    
    # Phase 7: Restart from perturbed best
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _adaptive_lbfgs(p_start, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_s{sigma}")
    
    # Phase 8: Try larger n (32768) if time permits
    print(f"\n[Phase 8] Larger n (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_large = 32768
        if n_large != n_parent:
            a_large_interp = _interpolate_to_n(best_seq, n_large)
            sc_init_large = _score(a_large_interp)
            print(f"  Interpolated to n={n_large}: score={sc_init_large:.10f}")
            
            if sc_init_large < 1.52:  # Only worth trying if interpolation is good
                a_large = _adaptive_lbfgs(a_large_interp, deadline, min_time_remaining=30.0)
                sc_large = _score(a_large)
                update_best(a_large, sc_large, f"n_{n_large}")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start_global:.1f}s")
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
