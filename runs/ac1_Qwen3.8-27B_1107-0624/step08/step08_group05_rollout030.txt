            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _conv_full(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a
        
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def _nesterov_sg(a_init, deadline, n_steps=10000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    momentum = 0.9
    patience = 0
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(y)
        c = _conv_full(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _conv_full(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        t = (i + 1) / (n_steps + 1)
        momentum_t = (1 + 1 / (t + 1)) / 2  # Nesterov momentum schedule
        lr = lr_init / (1.0 + 0.001 * i)
        
        y_new = y - lr * grad_z
        z_new = y_new + momentum_t * (y_new - y)
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        
        z = z_new
        y = y_new
        
        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    y = z.copy()
                    patience = 0
                    lr_init *= 0.5
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc
