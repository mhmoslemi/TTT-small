"""
Improved multi-strategy optimizer for minimizing step function autocorrelation peak.
Focuses on Newton active-set with adaptive tolerance, aggressive LSE cooling,
convolution-aware local search, and improved time management.

import numpy as np
import time
from scipy.optimize import minimize

_best_seq = None
_best_score = np.inf


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:

    a_rev = a[::-1]

    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0)
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = max(tol_frac * c_max, 1e-30)

    W = float(np.dot(w, c))

def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if not np.isfinite(alpha) or alpha <= 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if not np.isfinite(r_new):
            break
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

        c_max = float(c.max())

        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)

            return _hess_matvec(v, active_f, n) + 1e-6 * v

        grad_full = _hess_matvec(a, active_f, n)
        grad_t = grad_full - np.mean(grad_full)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=300)
        dx = dx - np.mean(dx)

        if dx_norm < 1e-15:
            break

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128, 256, 512]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(200, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _subgrad_polish(a_init, deadline, n_steps=10000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    avg_a = np.zeros(n)
    count = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        a_iter = _softmax(z)
        avg_a += a_iter
        count += 1

        if i % 100 == 0:
            a_avg = avg_a / count
            sc_avg = _score(a_avg)
            a_check = a_iter
            sc = _score(a_check)
            best_local = min(sc, sc_avg)
            if best_local < best_sc - 1e-14:
                best_sc = best_local
                best_a = a_avg if sc_avg < sc else a_check
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    lr *= 0.9
                    if patience > 60:
                        z = np.log(np.maximum(best_a, 1e-30))
                        z = z - z.mean()
                        patience = 0
                        lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    avg_final = avg_a / max(1, count)
    sc_avg_final = _score(avg_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final
    if sc_avg_final < best_sc:
        best_sc = sc_avg_final
        best_a = avg_final

    return best_a, best_sc


def _flatten_iteration(a, n_hot=5, n_steps=5000, deadline=None):
    """Direct iteration targeting the hottest convolution lags."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.005

    for i in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Pick the hottest lags
        sorted_idx = np.argsort(c)[::-1]
        hot_lags = sorted_idx[:n_hot]

        # Weight for hot lags
        w = np.zeros_like(c)
        for k in hot_lags:
            w[k] = 1.0
        w = w / float(n_hot)

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)

        # Normalize gradient and take step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_norm = grad / gnorm

        a_new = a - lr * grad_norm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < _score(a):
            a = a_new
            lr = min(lr * 1.02, 0.02)
        else:
            lr *= 0.85

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _local_perturb_search(a_init, deadline, n_trials=200):
    """Try small random perturbations and keep improvements."""
    n = len(a_init)
    rng = np.random.default_rng(12345)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_perturb = max(2, n // 512)

    for i in range(n_trials):
        if time.monotonic() > deadline:
            break
        z = np.log(np.maximum(a, 1e-30))
        idx = rng.choice(n, size=n_perturb, replace=False)
        sigma = 0.01 * (1 + rng.random())
        z[idx] += sigma * rng.standard_normal(n_perturb)
        z = z - z.mean()
        z = np.clip(z, -80, 80)
        a_try = _softmax(z)
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.12 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.01) + np.exp(-((t - 0.7) ** 2) / 0.01)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * t
        return a / a.sum()
    elif kind == "chirp_abs":
        t = np.linspace(0, 1, n)
        a = np.abs(np.sin(2 * np.pi * 3 * t + 0.5))
        a = a + 0.1
        return a / a.sum()
    elif kind == "sinc":
        t = np.linspace(-5, 5, n)
        a = np.sinc(t / 5.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining adaptive Newton active-set, LSE smoothing,
    power-sum optimization, convolution flattening, local perturbations, and
    multi-n exploration to minimize the autocorrelation peak score.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
        main_n = n_parent
    else:
        main_n = 8192
        parent = _make_profile(main_n, "tapered")
        upd(parent, _score(parent), "init")

    # === Phase 1: Quick exploration (45s) ===
    print(f"\n[Phase 1] Quick exploration (target: 45s)")
    expl_deadline = min(deadline, t_start + 45)
    for kind in ["tapered", "cos2", "gauss", "chirp_abs", "sinc"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(main_n, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # Quick Newton on best
    if _best_seq is not None and time.monotonic() < deadline - 100:
        nw_dead = min(deadline - 100, time.monotonic() + 20)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=50, tol_init=0.01, tol_min=0.002)
        upd(a_nw, sc_nw, "expl_newton")

    # === Phase 2: Main LSE optimization (300s) ===
    print(f"\n[Phase 2] Main LSE (target: 300s)")
    lse_dead = min(deadline, t_start + 350)
    a_lse, sc_lse = _lbfgs_lse(_best_seq.copy(), lse_dead, min_remaining=200, n_stages=14)
    upd(a_lse, sc_lse, "lse_main")

    # Newton polish after LSE
    if time.monotonic() < deadline - 80:
        nw_dead = min(deadline - 80, time.monotonic() + 60)
        a_nw2, sc_nw2 = _newton_active_set(_best_seq, nw_dead, max_iter=100, tol_init=0.003, tol_min=0.0005)
        upd(a_nw2, sc_nw2, "lse_newton")

    # === Phase 3: Extended Newton (500s) ===
    print(f"\n[Phase 3] Extended Newton (target: 500s)")
    nw_dead = min(deadline, t_start + 850)
    a_nw3, sc_nw3 = _newton_active_set(_best_seq, nw_dead, max_iter=2000, tol_init=0.003, tol_min=0.0001)
    upd(a_nw3, sc_nw3, "newton_ext")

    # === Phase 4: Power sum (150s) ===
    print(f"\n[Phase 4] Power sum (target: 150s)")
    if time.monotonic() < deadline - 150:
        ps_dead = min(deadline, t_start + 1000)
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), ps_dead, min_remaining=100)
        upd(a_ps, sc_ps, "power_sum")
        # Newton after power sum
        if time.monotonic() < deadline - 50:
            nw_dead = min(deadline - 50, time.monotonic() + 40)
            a_nw4, sc_nw4 = _newton_active_set(_best_seq, nw_dead, max_iter=100, tol_init=0.002, tol_min=0.0003)
            upd(a_nw4, sc_nw4, "ps_newton")

    # === Phase 5: Subgradient + flatten (200s) ===
    print(f"\n[Phase 5] Subgradient + flatten (target: 200s)")
    if time.monotonic() < deadline - 200:
        sg_dead = min(deadline, t_start + 1200)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=8000, lr_init=0.008, tol_frac=0.001)
        upd(a_sg, sc_sg, "sg")
        # Newton after SG
        if time.monotonic() < deadline - 30:
            nw_dead = min(deadline - 30, time.monotonic() + 30)
            a_nw5, sc_nw5 = _newton_active_set(_best_seq, nw_dead, max_iter=80, tol_init=0.002, tol_min=0.0002)
            upd(a_nw5, sc_nw5, "sg_newton")

    # Flatten
    if time.monotonic() < deadline - 40:
        fl_dead = min(deadline - 40, time.monotonic() + 80)
        a_fl, sc_fl = _flatten_iteration(_best_seq, n_hot=8, n_steps=5000, deadline=fl_dead)

    print(f"\n[Phase 6] Multi-n exploration (target: 200s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if main_n not in n_candidates:
        n_candidates.append(main_n)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:

        else:
            a_init = _resize_sequence(_best_seq, n_try)

        sc_init = _score(a_init)
        if sc_init < _best_score * 1.001:
            upd(a_init, sc_init, f"n_{n_try}_init")

        # Quick LSE
        if time.monotonic() < deadline - 30:
            lse_dead_n = min(deadline, time.monotonic() + 60)
            a_opt_n, sc_opt_n = _lbfgs_lse(a_init, lse_dead_n, min_remaining=20, n_stages=6)
            upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

            # Quick Newton
            if time.monotonic() < deadline - 20:
                nw_dead_n = min(deadline - 20, time.monotonic() + 30)
                a_opt_n2, sc_opt_n2 = _newton_active_set(_best_seq, nw_dead_n, max_iter=50, tol_init=0.002, tol_min=0.0005)
                upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_nw")

    # === Phase 7: Diverse restarts (150s) ===
    print(f"\n[Phase 7] Diverse restarts (target: 150s)")
    if time.monotonic() < deadline - 60:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=6, max_amp=0.2)
            elif i == 2:
                p_start = _perturb_log(_best_seq, 0.04, rng)
            elif i == 3:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=12, max_amp=0.15)
            else:
                p_start = _perturb_log(_best_seq, 0.01, rng)

            # Short LSE
            lse_dead_r = min(deadline, time.monotonic() + 30)
            a_r, sc_r = _lbfgs_lse(p_start, lse_dead_r, min_remaining=10, n_stages=5)
            upd(a_r, sc_r, f"restart_{i}")

            # Short Newton
            if time.monotonic() < deadline - 10:
                nw_dead_r = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(_best_seq, nw_dead_r, max_iter=40, tol_init=0.002, tol_min=0.001)
                upd(a_r2, sc_r2, f"restart_{i}_nw")

    # === Phase 8: Final intensive Newton (100s) ===
    print(f"\n[Phase 8] Final intensive Newton (target: 100s)")
    if time.monotonic() < deadline - 20:
        nw_final_dead = min(deadline - 5, time.monotonic() + 100)
        a_fin, sc_fin = _newton_active_set(_best_seq, nw_final_dead, max_iter=3000, tol_init=0.001, tol_min=0.00005)

    if _best_seq is not None and time.monotonic() < deadline - 3:

    return [float(x) for x in fallback]
