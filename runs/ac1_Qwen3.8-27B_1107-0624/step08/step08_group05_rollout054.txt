# Core utilities (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# LSE objective and gradient
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = float(c.max())

    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)

    a = _softmax(z)

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

    a = a / s
    best_a = a.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if gap_adaptive:

            c_sorted = np.sort(c)[::-1]

            else:
                tol = c_max * 0.002
        else:
            tol = max(tol_frac * c_max, 1e-14)
        
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

            threshold = c_sorted_idx[-1]
            active = c >= threshold

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))

            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)

def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=18):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # More aggressive annealing schedule
    betas = base_beta * np.array([1.0, 0.7, 0.45, 0.3, 0.2, 0.14,
                                   0.1, 0.07, 0.05, 0.035,
                                   0.025, 0.018, 0.012, 0.008,
                                   0.005, 0.003, 0.002, 0.001])
    betas = betas[:n_stages]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.002 * max(1.0, n / 8192.0)
        # Later stages get more iterations (they're more important)
        stage_weight = 1.0 + i * 0.3
        max_iter = min(5000, max(100, int(remaining * stage_weight / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

        max_iter = min(4000, max(100, int(remaining / (est_time * 8))))

        try:

                _power_sum_obj, z, args=(float(p), n),

            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

# Iterative Peak Equalization (NEW - key innovation)
# ============================================================

def _peak_equalize(a_init, deadline, n_iters=2000, top_k=5, lr_decay=0.999):
    """
    Directly targets the top-K autocorrelation peaks by computing
    a direction that specifically reduces them.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.005

    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top-K peaks
        k = min(top_k, len(c))
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[k-1]
        active = c >= threshold
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        z = z - lr * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        lr *= lr_decay

        if i % 100 == 0:
            sc = _score(_softmax(z))
            if sc < best_sc:
                best_sc = sc
                best_a = _softmax(z).copy()

    sc_final = _score(_softmax(z))
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = _softmax(z).copy()
    return best_a, best_sc

def _spectral_init(n, target_flatness=0.3, rng=None):
    """
    Create a sequence whose power spectrum has a specific shape.
    Target: concentrate power at DC with flat remainder.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    # Create target power spectrum
    N = _next_pow2(n + n - 1)
    freqs = np.fft.rfftfreq(N)
    
    # DC component gets most of the power
    A_target = np.zeros(N // 2 + 1, dtype=complex)
    A_target[0] = 1.0  # DC
    
    # Spread remaining power with some randomness
    for k in range(1, N // 2 + 1):
        # Decay with frequency but add randomness
        A_target[k] = target_flatness * (1.0 / (1.0 + 0.1 * k)) * (1.0 + 0.5 * rng.standard_normal())
        A_target[k] = complex(np.maximum(np.abs(A_target[k]), 0), 
                             0.1 * rng.standard_normal())
    
    # Inverse FFT
    a = np.fft.irfft(A_target * np.conj(A_target) ** 0.5, N)
    # Actually we need |A|^2 to give us the circular autocorrelation
    # Let's just use the inverse of the power spectrum
    power = np.abs(A_target)**2
    circ_acorr = np.fft.irfft(power, N)
    
    # Extract the linear part and reshape
    a = np.maximum(circ_acorr[:n], 1e-10)
    a = a / a.sum()
    return a

def _spectral_init_v2(n, rng=None):
    """
    Better spectral initialization: create a sequence with
    controlled spectral flatness.
    """
    if rng is None:
        rng = np.random.default_rng(123)
    
    # Target: power spectrum that's mostly flat with a DC peak
    N = n  # Use circular for simplicity initially
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = np.sqrt(n)  # DC
    
    # Random phases for other frequencies (gives flat autocorrelation)
    magnitudes = 0.5 + 0.3 * rng.random(N // 2)
    phases = 2 * np.pi * rng.random(N // 2)
    A[1:] = magnitudes * np.exp(1j * phases)
    
    a = np.fft.irfft(A, N)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

# Profile generators
# ============================================================

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "parabolic":
        t = np.linspace(0, 1, n)
        a = t * (1 - t) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "butterfly":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.01) + np.exp(-((t - 0.7) ** 2) / 0.01) + 0.2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "superposition":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2*np.cos(2*np.pi*t) + 0.1*np.cos(4*np.pi*t) + 0.05*np.cos(6*np.pi*t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(77)
        a = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

def _perturb_mix(a, rng):
    a1 = _perturb_log(a, 0.02, rng)
    a2 = _perturb_fourier_phase(a1, rng, n_modes=5, max_amp=0.2)
    return a2

def _perturb_spectral_redesign(a, rng):
    """Completely redesign the spectrum while keeping non-negativity."""
    n = len(a)
    A = np.fft.rfft(a)
    # Keep DC, randomize phases of rest
    new_A = np.zeros_like(A)
    new_A[0] = A[0]
    mags = np.abs(A)
    new_phases = 2 * np.pi * rng.random(len(A) - 1)
    new_A[1:] = mags[1:] * np.exp(1j * new_phases)
    a_new = np.fft.irfft(new_A, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

# Multi-n interpolation

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

    n, m = len(a), len(b)

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)

        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


# ============================================================
# Peak-targeted L-BFGS
# ============================================================

def _lbfgs_peak(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())

    configs = [
        (10, max_c * 0.01),
        (20, max_c * 0.008),
        (50, max_c * 0.005),
        (10, max_c * 0.003),
        (20, max_c * 0.002),
        (50, max_c * 0.001),
        (5, max_c * 0.0005),
    ]

    for K, temp in configs:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _peak_obj, z, args=(K, n, temp),
                jac=_peak_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc
