def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    return _fft_convolve(a, a)

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)

    """
    Compute the minimum-norm direction d that equalizes the active lags
    while preserving the sum constraint.
    
    Solves: M d = b where
      M[i, :] = 2 * a[active[i] - j] (reversed) for i in range(len(active))
      M[-1, :] = 1 (sum constraint)
      b[i] = c_max - c[active[i]]
      b[-1] = 0

        hi = min(k, n - 1)
        if lo <= hi:
            M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        b[i] = c_max - c[k]

    b[m] = 0.0
    
    # Minimum norm solution
    G = M @ M.T
    try:
        G_inv_b = np.linalg.solve(G + 1e-12 * np.eye(m + 1), b)
        d = M.T @ G_inv_b
    except np.linalg.LinAlgError:
        d = np.linalg.lstsq(M, b, rcond=None)[0]
    
    return d
