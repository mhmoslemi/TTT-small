"""
Ramp-initialized optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: non-uniform initializations (linear ramps, power laws) start much closer
to the optimal solution than uniform, giving scores near 4/3 vs 2 for uniform.
Uses L-BFGS with multi-scale log-sum-exp smoothing, active-set polish, and
multi-n exploration from ramp starting points.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.00025, 0.0001,
                                  0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                    'maxfun': max_iter * 3,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i < 6:
                    print(f"    beta={beta:.6e}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.2
                no_improve = 0
                if tol_frac <= tol_frac_end * 1.5 and lr < lr_init * 0.1:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.01)

    return best_a, _score(best_a)


def _peak_directed_descent(a_init, deadline, min_time_remaining=30.0):
    """Direct peak reduction: repeatedly target the top peaks with
    gradient steps weighted by peak prominence."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.02
    n_peaks = max(3, int(n * 0.01))

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            lr = min(0.05, lr * 1.02)

        # Get top peaks with prominence weighting
        c_sorted_idx = np.argsort(c)[::-1]
        top_idx = c_sorted_idx[:n_peaks]
        top_vals = c[top_idx]

        # Weight by prominence (how much above the mean)
        c_mean = c.max() / (2.0 * n - 1)
        weights = np.maximum(top_vals - c_mean, 0)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        # Compute combined gradient
        a_rev = a[::-1]
        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            # grad_k[j] = 2*a[k-j] if valid
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            # a[k-j] for j in [j_lo, j_hi]
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += w * 2.0 * a[idx]

        grad_t = grad - grad.sum() * a / a.sum()
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.85
            n_peaks = max(1, int(n_peaks * 0.7))

    return best_a, _score(best_a)


def _make_ramp_increase(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _make_ramp_decrease(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_power_law(n, p):
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_sine_quarter(n):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t / 2.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bell(n):
    t = np.linspace(0, 1, n)
    a = np.exp(-2.0 * (t - 0.7) ** 2) + 0.01
    return a / a.sum()
