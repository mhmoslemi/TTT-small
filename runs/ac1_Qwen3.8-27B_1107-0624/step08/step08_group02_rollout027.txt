def _kkt_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve Hx = rhs using CG, where H is the Hankel matrix defined by W."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x, 0
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp < 0:  # Not positive definite, CG fails
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * (1 + 1e-10):
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if np.sqrt(rs) < tol * np.sqrt(rs0):
            return x, 1
    return x, 0

"""

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=800, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if pHp <= 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.01:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if np.sqrt(rs) < tol * np.sqrt(rs0):
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]

def _resample(a, n_new):

        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 10:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent loaded: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        n = 16384
        parent = _make_profile(n, "tapered")
        print(f"Using default profile n={n}")
        update_best(parent, _score(parent), "default")

    n = len(parent)

    # Phase 1: Short L-BFGS to ensure good basin (max 200s)
    print(f"\n[Phase 1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline, t_start + 200)
    a_p1 = _lbfgs_smoothed(parent, p1_deadline, min_remaining=100.0)
    sc_p1 = _score(a_p1)
    update_best(a_p1, sc_p1, "lbfgs_warmup")

    # Phase 2: Main Newton with dual feasibility continuation (bulk of time)
    print(f"\n[Phase 2] Newton continuation with dual check (t={time.monotonic()-t_start:.1f}s)")
    a_main = _newton_with_dual_check(a_p1, 0.05, 1e-8, min(deadline, t_start + 1500))
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "newton_continuation")

    # Phase 3: Fine Newton polish with multiple tolerance levels
    print(f"\n[Phase 3] Fine Newton polish (t={time.monotonic()-t_start:.1f}s}")
    if time.monotonic() < deadline - 50:
        for tol_frac in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - 30 - time.monotonic()
            if remaining < 5:
                break
            a_fine, sc_fine = _newton_polish(
                a_main, tol_frac,
                min(deadline - 30, time.monotonic() + min(60, remaining)),
                max_rounds=200
            )
            if sc_fine < best_score:
                a_main = a_fine
            update_best(a_fine, sc_fine, f"polish_tol{tol_frac}")

    # Phase 4: Ultra-fine final polish
    print(f"\n[Phase 4] Ultra-fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        for tol_frac in [2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]:
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - 5 - time.monotonic()
            if remaining < 3:
                break
            a_ultra, sc_ultra = _newton_polish(
                a_main, tol_frac,
                min(deadline - 5, time.monotonic() + min(30, remaining)),
                max_rounds=150
            )
            if sc_ultra < best_score:
                a_main = a_ultra
            update_best(a_ultra, sc_ultra, f"ultra_{tol_frac:.0e}")

    # Phase 5: Try n=32768
    print(f"\n[Phase 5] n=32768 exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        a_32k = _resample(a_main, 32768)
        a_32k_opt, sc_32k = _newton_with_dual_check(
            a_32k, 0.02, 1e-7, min(deadline - 5, time.monotonic() + 120)
        )
        update_best(a_32k_opt, sc_32k, "n32768")
        if sc_32k < best_score + 0.005 and time.monotonic() < deadline - 10:
            a_32k_fine, sc_32k_fine = _newton_polish(
                a_32k_opt, 0.0005, min(deadline - 5, time.monotonic() + 30), max_rounds=100
            )
            update_best(a_32k_fine, sc_32k_fine, "n32768_fine")

    # Phase 6: Try perturbation + Newton
    print(f"\n[Phase 6] Perturbation exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        for sigma in [0.02, 0.005]:
            if time.monotonic() > deadline - 10:
                break
            z = np.log(np.maximum(a_main, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
            a_pert_opt, sc_pert = _newton_polish(
                a_pert, 0.001, min(deadline - 10, time.monotonic() + 20), max_rounds=100
            )
            update_best(a_pert_opt, sc_pert, f"perturb_{sigma}")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
