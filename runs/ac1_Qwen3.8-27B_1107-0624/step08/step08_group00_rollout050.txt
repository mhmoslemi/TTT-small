Improved hybrid optimizer: multi-start initialization, Nesterov-accelerated 
active-set gradient, stochastic iterated local search, and L-BFGS with 
adaptive log-sum-exp annealing for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":

        a = 0.5 * (1 + np.cos(2 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _active_set_gradient(a, c, tol_frac):
    """Compute the active-set gradient for the L∞ autocorrelation."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    
    # Weight: use inverse of (c_max - c[k] + eps) for active peaks
    # This gives more weight to the highest peaks
    eps = 1e-10
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()
    
    # Gradient of c[k] wrt a[j] = 2 * a[k-j] (for valid k-j)
    # So grad[j] = 2 * sum_k w[k] * a[k-j]
    # This is 2 * conv(w_padded, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1 : 2*n-1]
    grad = 2.0 * h
    
    # Project onto tangent space (remove mean for sum=1 constraint)
    grad = grad - np.mean(grad)
    return grad, w, active

    no_improve = 0

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        grad, w, active = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad  # Actually, standard Nesterov is different
        # Let me use standard momentum: v = momentum * v - lr * grad
        # Then a = a + v
        
        # Backtracking line search with momentum
        v = momentum * v - lr * (grad / gnorm)  # normalized gradient step
        alpha = 1.0
        improved = False
        for _ in range(50):
            a_try = a + alpha * v
            a_try = _normalize(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            v = np.zeros(n)  # reset momentum
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.5
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.01)
    
    return best_a, _score(best_a)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1 : 2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def f(self, z):
        return _smoobj(z, self.beta, self.n)
    def g(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                                   0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                   0.00005, 0.00002, 0.00001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 15))))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _stochastic_ils(a_init, deadline, rng, min_time_remaining=60.0):
    """Stochastic Iterated Local Search: perturb + re-optimize."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    # Different perturbation strategies
    strategies = [
        ('global', 0.01),    # small global perturbation
        ('global', 0.03),    # medium global perturbation
        ('block', 0.05),     # block perturbation
        ('sparse', 0.1),     # sparse perturbation
        ('global', 0.005),   # very small
    ]
    
    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        strategy = strategies[attempt % len(strategies)]
        kind, sigma = strategy
        
        if kind == 'global':
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            # Perturb a random block
            block_size = max(10, n // 10)
            start = rng.integers(0, n - block_size)
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start+block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start+block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            # Perturb a random subset
            n_perturb = max(5, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        
        # Quick optimization from perturbed point
        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 10)
        a_opt = _run_lbfgs(a_perturbed, opt_deadline, min_time_remaining=5)
        sc_opt = _score(a_opt)
        
        # Additional polish
        if time.monotonic() + min_time_remaining < deadline:
            pol_deadline = min(deadline - min_time_remaining, time.monotonic() + 5)
            a_pol, sc_pol = _nesterov_active_polish(a_opt, pol_deadline,
                                                     tol_start=0.005, tol_end=0.0001,
                                                     lr_init=0.02)
            if sc_pol < sc_opt:
                a_opt = a_pol
                sc_opt = sc_pol
        
        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS improvement: {best_sc:.10f} (strategy={kind}, sigma={sigma})")
        elif sc_opt < current_sc - 1e-12:
            current_a = a_opt.copy()
        
        attempt += 1
    
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

"""
Multi-start optimizer with Nesterov-accelerated active-set gradient,
stochastic iterated local search, diverse profile initialization, and
L-BFGS with adaptive log-sum-exp annealing for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key innovations over previous approach:
- Diverse initializations (triangle, Hann, Gaussian, random) to avoid local minima
- Nesterov momentum in active-set gradient for faster convergence
- Stochastic iterated local search (ILS) with multiple perturbation strategies
- Adaptive peak weighting (inverse-gap) in gradient computation
- More aggressive multi-n exploration with non-power-of-2 sizes
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _active_set_gradient(a, c, tol_frac):
    """Compute weighted active-set gradient for L-infinity autocorrelation."""
    n = len(a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1

    # Inverse-gap weighting: more weight to higher peaks
    eps = 1e-12 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / (c_max - c[active] + eps)
    w = w / w.sum()

    # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h

    # Project onto tangent space of sum(a)=1
    grad = grad - np.mean(grad)
    return grad


def _nesterov_active_polish(a_init, deadline, tol_start=0.01, tol_end=0.00005,
                            lr_init=0.05, momentum=0.85):
    """Active-set gradient with heavy-ball momentum and adaptive step size."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol_frac = tol_start
    lr = lr_init
    no_improve = 0
    max_stall = 250

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        grad = _active_set_gradient(a, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Heavy ball momentum
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(50):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            v = np.zeros(n)
            no_improve += 1
            if no_improve % 25 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.4)
                lr = lr_init * 0.5
                no_improve = 0
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.985)
            lr = min(lr_init, lr * 1.008)

    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def f(self, z):
        return _smoobj(z, self.beta, self.n)

    def g(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on log-softmax parameterization with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.004)

    betas = base_beta * np.array([1.0, 0.6, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01,
                                  0.005, 0.003, 0.0015, 0.0008, 0.0004,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(40, int(remaining / (est_time * 18))))

        prob = _LbfgsProblem(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _quick_optimize(a_init, deadline, min_time_remaining=20.0):
    """Quick combined L-BFGS + polish optimization."""
    a = _run_lbfgs(a_init, deadline, min_time_remaining=min_time_remaining + 10)
    if time.monotonic() < deadline - min_time_remaining:
        a2, sc2 = _nesterov_active_polish(
            a, min(deadline - min_time_remaining, time.monotonic() + 15),
            tol_start=0.005, tol_end=0.0005, lr_init=0.02, momentum=0.8)
        if sc2 < _score(a):
            a = a2
    return a


def _stochastic_ils(a_init, deadline, rng, min_time_remaining=60.0):
    """Stochastic Iterated Local Search with diverse perturbation strategies."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    strategies = [
        ('global_small', 0.005),
        ('global_med', 0.02),
        ('global_large', 0.05),
        ('block', 0.04),
        ('sparse', 0.08),
        ('mirror', 0.03),
        ('global_small', 0.003),
        ('block', 0.06),
    ]

    attempt = 0
    while time.monotonic() + min_time_remaining < deadline:
        kind, sigma = strategies[attempt % len(strategies)]

        if kind == 'global_small' or kind == 'global_med' or kind == 'global_large':
            z = np.log(np.maximum(current_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        elif kind == 'block':
            block_size = max(20, n // 8)
            start = int(rng.integers(0, n - block_size))
            a_perturbed = current_a.copy()
            z = np.log(np.maximum(a_perturbed[start:start + block_size], 1e-30))
            z = z + sigma * rng.standard_normal(block_size)
            a_perturbed[start:start + block_size] = np.exp(z - np.max(z))
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'sparse':
            n_perturb = max(10, n // 15)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_perturbed = current_a.copy()
            factors = np.exp(sigma * rng.standard_normal(n_perturb))
            a_perturbed[idx] *= factors
            a_perturbed = _normalize(a_perturbed)
        elif kind == 'mirror':
            # Mirror perturbation: flip a random segment
            seg_len = max(10, n // 20)
            start = int(rng.integers(0, n - seg_len))
            a_perturbed = current_a.copy()
            seg = a_perturbed[start:start + seg_len].copy()
            a_perturbed[start:start + seg_len] = seg[::-1]
            # Add small noise
            z = np.log(np.maximum(a_perturbed, 1e-30))
            z = z + 0.005 * rng.standard_normal(n)
            z = z - z.mean()
            a_perturbed = _softmax(z)
        else:
            a_perturbed = current_a.copy()

        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 8)
        a_opt = _quick_optimize(a_perturbed, opt_deadline, min_time_remaining=3)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            print(f"    ILS NEW BEST: {best_sc:.10f} ({kind}, sigma={sigma}) t={time.monotonic()-t_start:.0f}s")
        elif sc_opt < current_sc - 1e-12:
            current_a = a_opt.copy()

        attempt += 1

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-start optimizer combining diverse initialization, L-BFGS with
    log-sum-exp annealing, Nesterov momentum active-set gradient, and
    stochastic iterated local search.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent if n_parent > 0 else 16384

    # === Phase 1: Multi-start quick optimization (300s) ===
    print(f"\n[P1] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    ms_deadline = t_start + 300

    starts = []
    starts.append(('parent', parent.copy()))
    for kind in ['triangle', 'hann', 'gaussian', 'raised_cos2', 'random']:
        starts.append((kind, _make_profile(n_main, kind)))

    for name, a_init in starts:
        if time.monotonic() > ms_deadline - 40:
            break
        sc_init = _score(a_init)
        a_opt = _quick_optimize(a_init, ms_deadline, min_time_remaining=50)
        sc_opt = _score(a_opt)
        print(f"  {name}: init={sc_init:.10f}, opt={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [ms/{name}] t={time.monotonic() - t_start:.1f}s")

    # === Phase 2: Intensive L-BFGS on best (250s) ===
    print(f"\n[P2] Intensive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        lbfgs_deadline = min(deadline - 500, time.monotonic() + 250)
        a_main = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=60)
        sc = _score(a_main)
        if sc < best_score:
            best_seq = a_main.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_int] t={time.monotonic() - t_start:.1f}s")

    # === Phase 3: Nesterov polish (120s) ===
    print(f"\n[P3] Nesterov polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        pol_deadline = min(deadline - 350, time.monotonic() + 120)
        a_pol, sc_pol = _nesterov_active_polish(best_seq, pol_deadline,
                                                 tol_start=0.003, tol_end=0.00002,
                                                 lr_init=0.02, momentum=0.85)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic() - t_start:.1f}s")

    # === Phase 4: Stochastic ILS (250s) ===
    print(f"\n[P4] Stochastic ILS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        ils_deadline = min(deadline - 250, time.monotonic() + 250)
        a_ils, sc_ils = _stochastic_ils(best_seq, ils_deadline, rng, min_time_remaining=60)
        if sc_ils < best_score:
            best_seq = a_ils.copy()
            best_score = sc_ils
            print(f"  NEW BEST: {best_score:.10f} [ils] t={time.monotonic() - t_start:.1f}s")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main, 24576, 32768, 12288, 20480, 8192]
    # Remove duplicates
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen and nn != len(best_seq):
            seen.add(nn)
            n_unique.append(nn)

    for n_try in n_unique:
        if time.monotonic() > deadline - 120:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")

        a_opt = _quick_optimize(a_init, deadline, min_time_remaining=40)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic() - t_start:.1f}s")

    # === Phase 6: Second ILS round on best (100s) ===
    print(f"\n[P6] ILS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        ils2_deadline = min(deadline - 80, time.monotonic() + 100)
        rng2 = np.random.default_rng(int(seed) + 999)
        a_ils2, sc_ils2 = _stochastic_ils(best_seq, ils2_deadline, rng2, min_time_remaining=30)
        if sc_ils2 < best_score:
            best_seq = a_ils2.copy()
            best_score = sc_ils2
            print(f"  NEW BEST: {best_score:.10f} [ils2] t={time.monotonic() - t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish (50s) ===
    print(f"\n[P7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _nesterov_active_polish(best_seq, deadline - 3,
                                                 tol_start=0.001, tol_end=0.000003,
                                                 lr_init=0.005, momentum=0.8)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic() - t_start:.1f}s")

        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProblem(beta_fin, n)
            max_iter = min(4000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(prob.f, z, jac=prob.g, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic() - t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
