```python
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _lp_descent_step(f, active_lags, epsilon=1e-10):
    """
    Solve the LP to find a descent direction.
    f: current sequence (normalized, sum=1)
    active_lags: list of lag indices to constrain
    Returns: delta (descent direction) or None if infeasible
    """
    n = len(f)
    c = _fft_convolve(f, f)
    
    # LP: minimize t
    # subject to: for k in active_lags: sum_j 2*f[k-j]*delta[j] - t <= -(c[k])
    #             sum_j delta[j] = 0
    #             delta[j] >= -f[j] + epsilon for all j
    
    K = len(active_lags)
    # Number of variables: n (for delta) + 1 (for t) = n+1
    # c vector: [0, ..., 0, 1] (minimize t)
    c_lp = np.zeros(n + 1)
    c_lp[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag k: sum_j 2*f[k-j]*delta[j] - t <= -c[k]
    A_ub = np.zeros((K + n, n + 1))
    b_ub = np.zeros(K + n)
    
    row = 0
    for k in active_lags:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[row, j] = 2.0 * f[idx]
        A_ub[row, n] = -1.0  # coefficient of t
        b_ub[row] = -c[k]
        row += 1
    
    # Non-negativity: delta[j] >= -f[j] + epsilon → -delta[j] <= f[j] - epsilon
    for j in range(n):
        A_ub[row, j] = -1.0
        b_ub[row] = f[j] - epsilon
        row += 1
    
    # Equality constraint: sum_j delta[j] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Solve
    result = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     method='highs', options={'presolve': True})
    
    if result.status != 0:
        return None
    
    delta = result.x[:n]
    return delta
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _lp_step(f, active_lags, epsilon=1e-8, small_thresh=0.005):
    """
    Solve LP to find optimal descent direction.
    Returns delta or None.
    """
    n = len(f)
    c = _fft_convolve(f, f)
    K = len(active_lags)
    
    # Find indices where f[j] is small (need non-negativity constraint)
    small_idx = np.where(f < small_thresh)[0]
    m = len(small_idx)
    
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Constraints:
    # 1) For each active lag k: sum_j 2*f[max(0,k-j).. ] * delta[j] - t <= -c[k]
    # 2) For each small j: -delta[j] <= f[j] - epsilon
    
    A_ub = np.zeros((K + m, n + 1))
    b_ub = np.zeros(K + m)
    
    for i, k in enumerate(active_lags):
        # (f*f_delta)[k] = sum_j f[j] * delta[k-j] + delta[j] * f[k-j]
        # = 2 * sum_j f[k-j] * delta[j] (for valid k-j)
        # Actually: (f*(f+δ))[k] - (f*f)[k] = (f*δ)[k] + (δ*f)[k] = 2(f*δ)[k] by symmetry
        # (f*δ)[k] = sum_j f[j] * δ[k-j] where 0≤j<n and 0≤k-j<n
        # = sum_j f[j] * δ[k-j] for j in [max(0,k-n+1), min(k, n-1)]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # delta index = k - j, so as j goes from lo to hi, delta index goes from k-lo to k-hi
        # A_ub[i, delta_idx] = 2 * f[j] where delta_idx = k - j
        for j in range(lo, hi + 1):
            delta_idx = k - j
            A_ub[i, delta_idx] = 2.0 * f[j]
        A_ub[i, n] = -1.0  # -t
        b_ub[i] = -c[k]
    
    # Non-negativity for small entries
    for i, j in enumerate(small_idx):
        A_ub[K + i, j] = -1.0
        b_ub[K + i] = f[j] - epsilon
    
    # Equality: sum delta = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     method='highs', options={'presolve': True, 'time_limit': 5})
    
    if result.status != 0:
        return None, None
    
    delta = result.x[:n]
    t_val = result.x[n]
    return delta, t_val
```

```python
"""
LP-based optimization for step function score minimization.
Uses linear programming to find optimal descent directions, combined with
subgradient methods and random restarts. Explores multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _get_active_lags(c, top_fraction=0.01, max_lags=50, min_lags=5):
    """Get the active lags (those near the maximum)."""
    c_max = float(c.max())
    tol = c_max * (1.0 - top_fraction)
    active = np.where(c >= tol)[0]
    if len(active) > max_lags:
        # Keep the top max_lags
        top_idx = active[np.argsort(c[active])[::-1][:max_lags]]
        active = top_idx
    if len(active) < min_lags:
        # Add next closest lags
        remaining = np.setdiff1d(np.arange(len(c)), active)
        remaining = remaining[np.argsort(c[remaining])[::-1]][:min_lags - len(active)]
        active = np.concatenate([active, remaining])
    return active

def _lp_descent(f, active_lags, epsilon=1e-10, small_thresh=0.002):
    """
    Solve LP to find the optimal descent direction.
    Returns (delta, t_predicted) or (None, None) if infeasible.
    """
    n = len(f)
    c = _fft_convolve(f, f)
    K = len(active_lags)
    
    # Find indices where f[j] is small
    small_mask = f < small_thresh
    small_idx = np.where(small_mask)[0]
    m = len(small_idx)
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    
    # Active lag constraints
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            delta_idx = k - j
            rows.append(i)
            cols.append(delta_idx)
            vals.append(2.0 * f[j])
        # -t term
        rows.append(i)
        cols.append(n)
        vals.append(-1.0)
    
    # Non-negativity for small entries
    for i, j in enumerate(small_idx):
        rows.append(K + i)
        cols.append(j)
        vals.append(-1.0)
    
    b_ub = np.array([-c[k] for k in active_lags])
    b_ub = np.concatenate([b_ub, f[small_idx] - epsilon])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K + m, n + 1))
    
    # Equality: sum delta = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     method='highs', options={'presolve': True})
    
    if result.status != 0:
        return None, None
    
    delta = result.x[:n]
    t_val = result.x[n]
    return delta, t_val

def _subgradient_direction(f, active_lags):
    """Compute average subgradient at active lags."""
    n = len(f)
    g = np.zeros(n)
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        g[lo:hi+1] += 2.0 * f[k-hi:k-lo+1][::-1]
    g /= len(active_lags)
    return g

def _project_tangent(g, f):
    """Project gradient to tangent space of {x : sum(x) = 1}."""
    # The tangent space is {d : sum(d) = 0}
    # Projection: d - (sum(d)/n) * ones
    # But for non-negativity, we need a projection onto the simplex tangent
    # Simple approach: subtract the mean
    return g - g.mean()

def _optimize_lp(f, deadline, max_iter=5000, start_k=5, end_k=40, 
                 epsilon=1e-10, small_thresh=0.002):
    """Main LP-based optimization loop."""
    n = len(f)
    f = f.copy() / f.sum()
    best_f = f.copy()
    best_score = _score_norm(f)
    c = _fft_convolve(f, f)
    c_max = float(c.max())
    
    no_improve = 0
    step_scale = 1.0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Get active lags
        frac = it / max_iter
        k_lags = int(start_k + (end_k - start_k) * frac)
        active = _get_active_lags(c, top_fraction=0.01, max_lags=k_lags, min_lags=3)
        
        # Try LP
        delta, t_pred = _lp_descent(f, active, epsilon, small_thresh)
        
        if delta is not None and np.linalg.norm(delta) > 1e-15:
            # Line search
            Improved = False
            for alpha in [1.0, 0.5, 0.3, 0.1, 0.05]:
                f_new = f + alpha * step_scale * delta
                f_new = np.maximum(f_new, 0.0)
                s = f_new.sum()
                if s < 1e-15:
                    continue
                f_new /= s
                c_new = _fft_convolve(f_new, f_new)
                c_max_new = float(c_new.max())
                if c_max_new < c_max * (1 - 1e-12):
                    f = f_new
                    c = c_new
                    c_max = c_max_new
                    Improved = True
                    break
            if Improved:
                sc = _score_norm(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                step_scale = min(1.0, step_scale * 1.01)
            else:
                no_improve += 1
                step_scale *= 0.8
        else:
            # Fallback: subgradient
            g = _subgradient_direction(f, active)
            g_t = _project_tangent(g, f)
            g_norm = np.linalg.norm(g_t)
            if g_norm < 1e-15:

            f = f + sigma * np.random.randn(n)

            f /= f.sum()
            c = _fft_convolve(f, f)
            c_max = float(c.max())
            no_improve = 0
            step_scale = 1.0
        
        if it % 200 == 0:
            print(f"  LP iter {it}: c_max={c_max:.10f}, score={_score_norm(f):.10f}, "
                  f"step_scale={step_scale:.4f}, no_improve={no_improve}")
    
    # Final check
    sc = _score_norm(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    
    return best_f, best_score
```

```python
"""
LP-based optimization for step function score minimization.
Uses linear programming to find optimal descent directions with adaptive active sets,
combined with subgradient methods, Adam in log-space, and random restarts.
Explores multiple sequence lengths for better global minimum.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _get_active_lags(c, top_frac=0.01, max_lags=50, min_lags=5):
    c_max = float(c.max())
    if c_max < 1e-15:
        return np.array([0])
    tol = c_max * (1.0 - top_frac)
    active = np.where(c >= tol)[0]
    if len(active) > max_lags:
        top_idx = active[np.argsort(c[active])[::-1][:max_lags]]
        active = top_idx
    if len(active) < min_lags:
        all_idx = np.arange(len(c))
        remaining = np.setdiff1d(all_idx, active)
        if len(remaining) > 0:
            add = remaining[np.argsort(c[remaining])[::-1]][:min_lags - len(active)]
            active = np.concatenate([active, add])
    return active


def _lp_descent(f, active_lags, epsilon=1e-10, small_thresh=0.001):
    n = len(f)
    c = _fft_convolve(f, f)
    K = len(active_lags)

    small_mask = f < small_thresh
    small_idx = np.where(small_mask)[0]
    m = len(small_idx)

    rows = []
    cols = []
    vals = []

    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for m_idx in range(lo, hi + 1):
            rows.append(i)
            cols.append(m_idx)
            vals.append(2.0 * f[k - m_idx])
        rows.append(i)
        cols.append(n)
        vals.append(-1.0)

    for i, j in enumerate(small_idx):
        rows.append(K + i)
        cols.append(j)
        vals.append(-1.0)

    b_ub = np.array([-c[k] for k in active_lags])
    b_ub = np.concatenate([b_ub, f[small_idx] - epsilon])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(K + m, n + 1))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         method='highs', options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None, None

    if result.status != 0:
        return None, None

    delta = result.x[:n]
    t_val = result.x[n]
    return delta, t_val


def _subgradient_direction(f, active_lags):
    n = len(f)
    g = np.zeros(n)
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        g[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1][::-1]
    g /= len(active_lags)
    return g


def _project_tangent(g):
    return g - g.mean()

            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            g[lo:hi + 1] += w[i] * 2.0 * f[k - hi:k - lo + 1][::-1]

        W = float(np.dot(f, g))
        z_grad = f * (g - W)

        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * 0.3 / update_norm
        z = z - update
        z = z - z.mean()
        z = np.clip(z, -80, 80)

        f = np.exp(z)
        f = f / f.sum()

        if t % 100 == 0:
            c = _fft_convolve(f, f)
            c_max = float(c.max())
            sc = _score_norm(f)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 80:
                z = np.log(np.maximum(best_f, 1e-15))
                z = z - z.mean()
                z += 0.005 * np.random.randn(n)
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                no_improve = 0

    return best_f, best_score


def _subgradient_polish(f, deadline, max_iter=50000, top_k=5, initial_lr=0.003):
    n = len(f)
    f = f.copy() / f.sum()
    best_f = f.copy()
    best_score = _score_norm(f)
    c = _fft_convolve(f, f)
    c_max = float(c.max())
    lr = initial_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.9995:
                active.append(idx)

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1][::-1]
        g /= len(active)

        g_t = _project_tangent(g)
        g_norm = np.linalg.norm(g_t)
        if g_norm < 1e-15:
            break

        f_new = f - lr * g_t / g_norm
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s

        c_new = _fft_convolve(f_new, f_new)
        c_max_new = float(c_new.max())

        if c_max_new < c_max * (1 - 1e-14):
            best_score = _score_norm(f_new)
            best_f = f_new.copy()
            f = f_new
            c = c_new
            c_max = c_max_new
            lr *= 0.9998
        else:
            lr *= 0.95
            if lr < 1e-10:
                break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase LP-based optimizer:
    Phase 1: Multi-n exploration with quick LP
    Phase 2: Intensive LP with growing active set
    Phase 3: Adam log-space refinement
    Phase 4: More LP with larger active set
    Phase 5: Random restarts
    Phase 6: Final subgradient polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    np.random.seed(int(seed) % (2**31 - 1))

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
        else:
            raise ValueError
    except Exception:
        parent = None

    best_f = None
    best_score = np.inf
    best_n = 0

    # Phase 1: Multi-n exploration (0-250s)
    print("Phase 1: Multi-n exploration")
    n_values = [4096, 8192, 16384]
    for n_trial in n_values:
        if time.monotonic() > deadline - 400:
            break
        if parent is not None and len(parent) == n_trial:
            f = parent.copy()
        elif parent is not None:
            f = np.interp(np.linspace(0, 1, n_trial), np.linspace(0, 1, len(parent)), parent)
            f = np.maximum(f, 0)
            f /= f.sum()
        else:
            f = np.ones(n_trial) / n_trial

        sub_dl = min(deadline - 350, time.monotonic() + 70)
        f_opt, sc = _optimize_lp(f, sub_dl, max_iter=3000, start_k=5, end_k=25)
        full_sc = _score(f_opt)
        print(f"  n={n_trial}: score={full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc
            best_f = f_opt.copy()
            best_n = n_trial

    print(f"Phase 1 best: n={best_n}, score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Intensive LP (250-800s)
    print(f"Phase 2: Intensive LP, n={best_n}")
    sub_dl = min(deadline - 300, t_start + 800)
    if time.monotonic() < sub_dl:
        f_opt, sc = _optimize_lp(best_f, sub_dl, max_iter=100000,
                                 start_k=10, end_k=50, small_thresh=0.001)
        full_sc = _score(f_opt)
        print(f"Phase 2: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_f = f_opt.copy()

    # Phase 3: Adam log-space (800-1100s)
    print("Phase 3: Adam log-space")
    sub_dl = min(deadline - 250, t_start + 1100)
    if time.monotonic() < sub_dl:
        f_opt, sc = _adam_log_space(best_f, sub_dl, lr_start=0.04)
        full_sc = _score(f_opt)
        print(f"Phase 3: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_f = f_opt.copy()

    # Phase 4: More LP with larger active set (1100-1450s)
    print("Phase 4: LP with larger active set")
    sub_dl = min(deadline - 200, t_start + 1450)
    if time.monotonic() < sub_dl:
        f_opt, sc = _optimize_lp(best_f, sub_dl, max_iter=200000,
                                 start_k=20, end_k=80, small_thresh=0.0005, top_frac=0.02)
        full_sc = _score(f_opt)
        print(f"Phase 4: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_f = f_opt.copy()

    # Phase 5: Random restarts (1450-1720s)
    print("Phase 5: Random restarts")
    n_restarts = 10
    sub_dl = t_start + 1720
    time_per = max(15, (sub_dl - time.monotonic()) / n_restarts)
    n_curr = len(best_f)

    for r in range(n_restarts):
        if time.monotonic() > sub_dl - time_per * 0.6:
            break
        sigma = 0.002 + 0.003 * r / max(1, n_restarts)
        f_pert = best_f + sigma * np.random.randn(n_curr)
        f_pert = np.maximum(f_pert, 0.0)
        f_pert /= f_pert.sum()

        rd = min(sub_dl - 5, time.monotonic() + time_per * 0.6)
        f_pert_opt, sc_pert = _optimize_lp(f_pert, rd, max_iter=5000, start_k=10, end_k=40)
        full_sc_pert = _score(f_pert_opt)
        if full_sc_pert < best_score:
            print(f"  Restart[{r}] sigma={sigma:.4f}: NEW BEST score={full_sc_pert:.10f}")
            best_score = full_sc_pert
            best_f = f_pert_opt.copy()
        else:
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_sc_pert:.10f}")

    # Phase 6: Final subgradient + LP polish (1720-deadline)
    if time.monotonic() < deadline - 5:
        print("Phase 6: Final polish")
        sub_dl = deadline - 3
        f_opt, sc = _subgradient_polish(best_f, sub_dl, max_iter=50000, top_k=5, initial_lr=0.001)
        full_sc = _score(f_opt)
        if full_sc < best_score:
            best_score = full_sc
            best_f = f_opt.copy()
            print(f"Phase 6 SG: score={full_sc:.10f}")

        if time.monotonic() < deadline - 3:
            f_opt2, sc2 = _optimize_lp(best_f, deadline - 1, max_iter=50000,
                                       start_k=30, end_k=100, small_thresh=0.0001)
            full_sc2 = _score(f_opt2)
            if full_sc2 < best_score:
                best_score = full_sc2
                best_f = f_opt2.copy()
                print(f"Phase 6 LP: score={full_sc2:.10f}")

    final = np.clip(best_f, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
