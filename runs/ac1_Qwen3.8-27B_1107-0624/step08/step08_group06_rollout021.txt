"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Improved L-BFGS with adaptive log-sum-exp smoothing
2. Enhanced active-set Newton with graduated tolerances and adaptive set management
3. Principled deflation targeting top peaks
4. Multi-n exploration with interpolation
5. Spectral-based initialization for diversity

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute M_A * a where M_A[i,j] = A_vec[i+j] (Hankel matrix)."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=500, tol=1e-14):
    """Conjugate gradient with diagonal preconditioner."""
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    # Diagonal preconditioner
    v_test = np.zeros_like(rhs)
    v_test[::max(1, len(rhs)//10)] = 1.0
    d_est = np.abs(matvec(v_test)) + 1e-15
    d_est = np.maximum(d_est, 1e-15)
    
    def precond(r):
        return r / d_est
    
    z = precond(r)
    p = z.copy()
    rz = np.dot(r, z)
    rhs_norm = np.linalg.norm(rhs)
    
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rz / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rz_new = np.dot(r, precond(r))
        if np.sqrt(np.dot(r, r)) < tol * (1 + rhs_norm):
            break
        if abs(rz_new) < 1e-30:
            break
        p = precond(r) + (rz_new / rz) * p
        rz = rz_new
    return x


def _active_set_newton_v2(a_init, deadline, max_iter=500, tol_schedule=None):
    """
    Improved active-set Newton with:
    - Graduated tolerance schedule (wide to tight)
    - Adaptive active set (add/remove based on proximity)
    - Preconditioned CG for Newton system
    - Armijo line search on true objective
    - Proper handling of non-negativity via active variable set
    """
    if tol_schedule is None:
        tol_schedule = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00002, 0.00001]
    
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_cmax = float(_full_convolve(a, a).max())
    
    # Track progress
    stagnation_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Determine tolerance based on iteration progress
        progress = min(1.0, it / (max_iter * 0.8))
        idx = min(len(tol_schedule) - 1, int(progress * len(tol_schedule)))
        tol_a = tol_schedule[idx]
        
        # Active set: coefficients within tol_a * c_max of the maximum
        active = c >= (c_max - tol_a * c_max)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())
        
        # Cap active set size
        if m > 8 * n:
            top_idx = np.argsort(c)[::-1][:8 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = int(active.sum())
        
        if m < 2:
            active[np.argmax(c)] = True
            second = np.argsort(c)[::-1][1]
            active[second] = True
            m = 2
        
        A_vec = active.astype(np.float64)
        
        # Gradient of c_S = (1/m) sum_{k in S} c[k]
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        
        # Project gradient onto tangent space (sum=0)
        g_mean = np.mean(g)
        g_t = g - g_mean
        
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break
        
        # Hessian: H = 2/m * M_S, so H*v = (2/m) * _corr_a_A(v, A_vec, n)
        # Add regularization for numerical stability
        mu = 1e-8 * (c_max / max(m, 1)) + 1e-10 * c_max
        
        def hess_vp(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v
        
        # Solve Newton system: H * da = -g_t, with sum(da) = 0
        # First solve H * u = -g_t
        u = _cg_solve(hess_vp, -g_t, max_cg=300, tol=1e-13)
        # Then solve H * w = 1 (for the Lagrange multiplier correction)
        w = _cg_solve(hess_vp, np.ones(n), max_cg=300, tol=1e-13)
        w_sum = float(w.sum())
        
        if abs(w_sum) < 1e-15:
            da = u
        else:
            lam = float(u.sum()) / w_sum
            da = u - lam * w
        
        # Ensure sum(da) = 0 (numerical cleanup)
        da = da - np.mean(da)
        
        # Line search with Armijo condition on true objective (max c[k])
        f_old = c_max
        dir_norm = np.linalg.norm(da)
        if dir_norm < 1e-18:
            break
        
        best_t = 0.0
        best_f = np.inf
        t = 1.0
        
        for ls in range(60):
            a_try = a + t * da
            neg_mask = a_try < -1e-15
            if neg_mask.any():
                # Find max step before hitting zero
                t_max = np.min(a[a_try < 0] / (-da[a_try < 0])) * 0.999
                t = max(t_max, 1e-10)
        
            a_try = np.maximum(a + t * da, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            
            # Armijo condition
            if f_try <= f_old + 1e-12 * t * dir_norm * gnorm or f_try < best_f:
                best_f = f_try
                best_t = t
            
            if f_try <= f_old - 1e-7 * t * dir_norm * gnorm:
                break
            
            t *= 0.5
        
        if best_t > 0 and best_f < f_old + 1e-14:
            a_new = np.maximum(a + best_t * da, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                c_new_max = float(_full_convolve(a_new, a_new).max())
                
                if sc_new < best_sc - 1e-15:
                    best_sc = sc_new
                    best_a = a_new.copy()
                
                if c_new_max < best_cmax - 1e-15:
                    best_cmax = c_new_max
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                
                a = a_new
            else:
                break
        else:
            # Try a simpler gradient step
            t = 0.05
            accepted = False
            for ls in range(30):
                a_try = np.maximum(a - t * g_t, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < _score(a) - 1e-15:
                    a = a_try
                    accepted = True
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                    break
                t *= 0.5
            if not accepted:
                break
        
        # Check convergence
        if it > 10:
            c_check = _full_convolve(a, a)
            if abs(float(c_check.max()) - c_max) < 1e-14:
                stagnation_count += 2
            else:
                if stagnation_count > 5:
                    # Try to escape: slightly perturb
                    break
        
        if it % 50 == 0:
            print(f"    AS-Newton iter {it}: cmax={c_max:.10e}, tol={tol_a:.6f}, m={m}, score={best_sc:.10f}")
    
    return best_a, best_sc


def _deflation_v2(a, deadline, n_rounds=50):
    """
    Principled deflation: iteratively reduce elements that contribute
    most to the top convolution peaks, with small adaptive steps.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        n_top = max(5, min(len(c), int(0.02 * len(c))))
        top_positions = np.argsort(c)[::-1][:n_top]
        
        # For each top peak, find the dominant contributing pair and reduce it
        a_new = a.copy()
        any_reduction = False
        
        for k_pos in top_positions[:min(n_top, 30)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            if i_lo > i_hi:
                continue
            
            # Find the dominant pair
            products = a_new[i_lo:i_hi+1] * a_new[k_pos - i_hi:k_pos - i_lo][::-1]
            max_idx = i_lo + int(np.argmax(products))
            max_prod = products[int(np.argmax(products))]
            
            if max_prod > 0:
                j = k_pos - max_idx
                # Reduce the larger element
                if a_new[max_idx] >= a_new[j]:
                    a_new[max_idx] *= (1.0 - 0.01)
                else:
                    a_new[j] *= (1.0 - 0.01)
                any_reduction = True
        
        if not any_reduction:
            break
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller reduction
            a_small = a.copy()
            for k_pos in top_positions[:10]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                if i_lo > i_hi:
                    continue
                products = a_small[i_lo:i_hi+1] * a_small[k_pos - i_hi:k_pos - i_lo][::-1]
                max_idx = i_lo + int(np.argmax(products))
                if products[int(np.argmax(products))] > 0:
                    j = k_pos - max_idx
                    if a_small[max_idx] >= a_small[j]:
                        a_small[max_idx] *= (1.0 - 0.003)
                    else:
                        a_small[j] *= (1.0 - 0.003)
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-15:
                    best_sc = sc_small
                    best_a = a_small.copy()
                    a = a_small
                else:
                    break
    
    return best_a, best_sc


def _lbfgs_ls(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with log-sum-exp and adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Estimate initial beta from the gap between top coefficients
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.05)
    
    # Adaptive beta schedule: start larger, decrease geometrically
    n_betas = 25
    decay = 0.7
    betas = [base_beta * (decay ** i) for i in range(n_betas)]
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # No improvement, try smaller beta
                continue
        except Exception:
            continue
    
    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        center = n // 2
        a = np.exp(-4.0 * ((np.arange(n) - center * 0.25) / (n * 0.2)) ** 2) + \
            np.exp(-4.0 * ((np.arange(n) - center * 1.75) / (n * 0.2)) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        # Flat in the middle, tapered at edges
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = max(1, n // 10)
        for i in range(edge):
            a[i] = 0.5 * (1 + np.cos(np.pi * i / edge))
            a[n - 1 - i] = 0.5 * (1 + np.cos(np.pi * i / edge))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _spectral_init(n, rng, mode="flat"):
    """
    Generate a non-negative sequence with a desired spectral shape.
    mode='flat': try to make power spectrum flat (minimize variance)
    mode='lowpass': low-pass filtered (smooth sequence)
    mode='highpass': more oscillatory
    """
    if mode == "flat":
        # Start with random phase, take absolute value of IFFT
        A = np.exp(1j * rng.standard_normal(n) * 2 * np.pi)
        A[0] = 1.0  # DC component
        a = np.real(np.fft.ifft(A))
        a = np.abs(a)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == "lowpass":
        t = np.linspace(0, 1, n)
        a = np.exp(-2 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == "highpass":
        # Sequence with more high-frequency content
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4 * np.pi * t) + 0.3 * np.cos(8 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _dual_descent(a_init, deadline, n_iter=2000):
    """
    Mirror descent / dual averaging approach:
    1. Maintain weights w over convolution positions
    2. Solve QP: min w^T c(a) s.t. sum(a)=1, a>=0 (approximately)
    3. Update w = softmax(c/temperature)
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Temperature schedule
    temp_start = 0.05
    temp_end = 0.001
    n_eff = n_iter
    
    for it in range(n_eff):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Temperature (annealing)
        progress = it / max(1, n_eff - 1)
        temp = temp_start * (temp_end / temp_start) ** progress
        
        # Update weights
        w = np.exp(np.clip((c - c_max) / max(temp * c_max, 1e-15), -700, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient of w^T c(a) = 2 * M_w * a
        grad = 2.0 * _corr_a_A(a, w, n)
        grad_t = grad - np.mean(grad)
        
        # Step size (Polyak-style)
        step = temp / (1.0 + 0.1 * it)
        
        a_new = a - step * grad_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        
        if it % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. L-BFGS warm-up with adaptive log-sum-exp
    2. Enhanced active-set Newton (main workhorse)
    3. Principled deflation
    4. Dual descent for diversity
    5. Multi-n exploration
    6. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # Phase 1: L-BFGS warm-up from parent
    print(f"\n[Phase 1] L-BFGS warm-up (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = min(deadline - 800, t_start + 300)
    a_main = _lbfgs_ls(parent, p1_deadline, min_time_remaining=30)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_warmup")
    
    # Phase 2: Enhanced active-set Newton (main phase)
    print(f"\n[Phase 2] Active-Set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_dl = min(deadline - 500, time.monotonic() + 400)
        a_as, sc_as = _active_set_newton_v2(best_seq, as_dl, max_iter=600)
        update_best(a_as, sc_as, "as_newton_main")
        
        # Second pass with tighter schedule
        if time.monotonic() < deadline - 300:
            as_dl2 = min(deadline - 300, time.monotonic() + 200)
            tol_tight = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00002]
            a_as2, sc_as2 = _active_set_newton_v2(best_seq, as_dl2, max_iter=500, tol_schedule=tol_tight)
            update_best(a_as2, sc_as2, "as_newton_tight1")
    
    # Phase 3: Deflation
    print(f"\n[Phase 3] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        def_dl = min(deadline - 300, time.monotonic() + 80)
        a_def, sc_def = _deflation_v2(best_seq, def_dl, n_rounds=60)
        update_best(a_def, sc_def, "deflation")
        
        if sc_def < best_score and time.monotonic() < deadline - 200:
            # Follow up with Newton
            as_dl3 = min(deadline - 200, time.monotonic() + 100)
            a_def2, sc_def2 = _active_set_newton_v2(a_def, as_dl3, max_iter=300)
            update_best(a_def2, sc_def2, "post_deflation_as")
    
    # Phase 4: Dual descent for diversity
    print(f"\n[Phase 4] Dual Descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        dd_dl = min(deadline - 200, time.monotonic() + 60)
        a_dd = _dual_descent(best_seq, dd_dl, n_iter=3000)
        sc_dd = _score(a_dd)
        update_best(a_dd, sc_dd, "dual_descent")
        
        if sc_dd < best_score + 0.005 and time.monotonic() < deadline - 100:
            as_dl4 = min(deadline - 100, time.monotonic() + 60)
            a_dd2, sc_dd2 = _active_set_newton_v2(a_dd, as_dl4, max_iter=200)
            update_best(a_dd2, sc_dd2, "post_dd_as")
    
    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n Exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    
    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        if n_try > 32768:
            continue
        
        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
            # Also try a spectral init
            a_spec = _spectral_init(n_try, rng, "flat")
            sc_init = _score(a_init)
            sc_spec = _score(a_spec)
            if sc_spec < sc_init:
                a_init = a_spec
            a_init2 = _make_profile(n_try, "raised_cosine")
            sc_init2 = _score(a_init2)
            if sc_init2 < min(sc_init, sc_spec):
                a_init = a_init2
        else:
            a_init = _make_profile(n_try, "uniform")
        
        rem = deadline - 80 - time.monotonic()
        if rem < 20:
            break
        
        # Quick L-BFGS
        lb_dl = min(deadline - 80, time.monotonic() + max(30, rem * 0.4))
        a_opt = _lbfgs_ls(a_init, lb_dl, min_time_remaining=10)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")
        
        # Quick Newton if promising
        if sc_opt < best_score + 0.02 and time.monotonic() < deadline - 30:
            as_dl5 = min(deadline - 30, time.monotonic() + max(20, (deadline - 80 - time.monotonic()) * 0.5))
            a_opt2, sc_opt2 = _active_set_newton_v2(a_opt, as_dl5, max_iter=200)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")
    
    # Phase 6: Perturbation restarts
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("perturb_002", 0.02),
            ("perturb_005", 0.05),
            ("perturb_010", 0.10),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(len(z))
            z = z - z.mean()
            p_start = _softmax(z)
            
            rem = deadline - 20 - time.monotonic()
            if rem < 10:
                break
            
            # Quick Newton from perturbed start
            as_dl6 = min(deadline - 20, time.monotonic() + max(10, rem * 0.7))
            a_r, sc_r = _active_set_newton_v2(p_start, as_dl6, max_iter=150)
            update_best(a_r, sc_r, f"restart_{label}")
    
    # Phase 7: Final polish
    print(f"\n[Phase 7] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active-set Newton
        as_final = min(deadline - 3, time.monotonic() + 50)
        tol_very_tight = [0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00002, 0.00001, 0.000005]
        a_fin, sc_fin = _active_set_newton_v2(best_seq, as_final, max_iter=300, tol_schedule=tol_very_tight)
        update_best(a_fin, sc_fin, "final_as_tight")
        
        # One more deflation + Newton
        if time.monotonic() < deadline - 5:
            a_def3, sc_def3 = _deflation_v2(best_seq, min(deadline - 3, time.monotonic() + 15), n_rounds=20)
            if sc_def3 < best_score:
                update_best(a_def3, sc_def3, "final_def")
                as_fin2 = min(deadline - 2, time.monotonic() + 10)
                a_fin2, sc_fin2 = _active_set_newton_v2(a_def3, as_fin2, max_iter=100, tol_schedule=tol_very_tight[-5:])
                update_best(a_fin2, sc_fin2, "final_def_as")
        
        # L-BFGS with very small beta
        if time.monotonic() < deadline - 4:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(1000, max(30, int(remaining / 0.04)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 80, 'maxls': 80}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        # Ensure we return the version with the best score
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
