Let me also add a key improvement: **the "dual" approach**. Instead of minimizing the smoothed max, we can use the dual formulation:

The primal: min_a max_k R(k) - t * (sum(a) - 1) subject to a >= 0
The dual: max_{p in simplex} min_a sum_k p_k * R(k) - t * (sum(a) - 1) subject to a >= 0

For a fixed p (distribution over lags), the inner problem is:
min_a sum_k p_k * R(k) - t * sum(a) subject to a >= 0

This is a QP: min_a (1/2) a^T (sum_k p_k H_k) a - t * 1^T a

The optimal a satisfies: (sum_k p_k H_k) a = t * (1 - mu) where mu_i >= 0 and mu_i * a_i = 0.

On the support S = {i: a_i > 0}: H_S a_S = t * 1_S where H = sum_k p_k H_k.

So a_S = t * H_S^{-1} 1_S, and we need a_S >= 0, which requires H_S^{-1} 1_S >= 0.

The step t is then determined by sum(a) = 1: t = 1 / (1_S^T H_S^{-1} 1_S).

Then the dual objective is: sum_k p_k * R(a) - t * (sum(a) - 1) = sum_k p_k * R(a) (since sum(a) = 1).

And we want to maximize this over p in the simplex.

The gradient of the dual objective w.r.t. p_k is: R(k) - R(a) (the difference between the k-th autocorrelation value and the "average" weighted by p).

So the dual update is: p_k += lr * (R(k) - sum_j p_j R(j)) = lr * (R(k) - R_bar)

This is just projecting the autocorrelation onto the simplex! (Apex-multiplicative weights.)

Wait, this is interesting. The dual problem is:
max_p sum_k p_k R(k) subject to p in simplex

The gradient is R - R_bar (where R_bar = sum p_k R(k)).
The multiplicative weight update is: p_k *= exp(lr * (R(k) - R_bar))

This is essentially the "softmax" of the autocorrelation values! And it's related to the log-sum-exp smoothing.

In fact, the dual of the smoothed primal (with beta) is exactly:
p_k = exp(R(k)/beta) / sum_j exp(R(j)/beta)

And the primal solution for this p is:
a = argmin sum_k p_k R(k) subject to a in simplex

So the primal-dual relationship is:
- Given p (dual variable), compute a (primal) by solving the QP
- Given a (primal), compute p (dual) by softmax of R(k)

This is an alternating minimization/maximization algorithm!

"""
Multi-phase optimizer for minimum peak autocorrelation sequences.
Uses primal-dual iteration (softmax dual + QP primal), L-BFGS with log-sum-exp
smoothing, projected gradient with momentum, and multi-n exploration.
Key insight: the problem is equivalent to flattening the linear autocorrelation,
and the primal-dual gap gives a certificate of optimality.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    """Project v onto the probability simplex {a: sum(a)=1, a>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _hvp_pv(v, p, n):
    """Compute H_p * v where H_p = 2 * sum_k p_k H_k.
    p has length 2n-1 (autocorrelation lags), v has length n.
    Uses FFT for the convolution-based Hessian-vector product.
    """
    b = v[::-1]  # reversal of v
    conv_len = len(b) + len(p) - 1
    N = _next_pow2(conv_len)
    Fb = np.fft.rfft(b, N)
    Fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(Fb * Fp, N)[:conv_len]
    # H_p * v[m] = 2 * conv[n-1+m] - correction
    result = 2.0 * conv[n - 1:2 * n - 1].copy()
    # Diagonal correction: for each k where 2m=k, subtract v[m]
    # (H_k * v)[m] should be v[k-m] not 2*v[k-m] when 2m=k
    # The correction is: for k=0,2,4,...,2(n-1): subtract v[k//2]
    for k in range(0, 2 * n - 1, 2):
        m = k // 2
        if m < n and k < len(p):
            result[m] -= v[m] * p[k]
    return result


def _grad_r_k(a, k, n):
    """Compute dR(k)/da where R(k) = (a*a)[k] is the k-th lag of linear autocorrelation."""
    grad = np.zeros(n)
    for m in range(n):
        j = k - m
        if 0 <= j < n:
            if j == m:
                grad[m] = a[m]
            else:
                grad[m] = 2.0 * a[j]
    return grad


def _grad_smoothed(a, p, n):
    """Compute gradient of sum_k p_k * R(k) w.r.t. a."""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                if j == m:
                    grad[m] += p[k] * a[m]
                else:
                    grad[m] += 2.0 * p[k] * a[j]
    return grad


def _grad_smoothed_fast(a, p, n):
    """Fast gradient computation using FFT-based Hessian-vector product trick.
    grad_k[m] = (H_k * a)[m] essentially.
    """
    # grad = H_p * a / 2 (since H_p = 2 * sum p_k H_k, and grad of sum p_k R(k) = H_p * a / 2 * 2 = H_p * a)
    # Wait: R(k) = a^T Q_k a, so dR(k)/da = (Q_k + Q_k^T) a = 2 Q_k a (since Q_k symmetric)
    # So d/d a [sum p_k R(k)] = sum p_k * 2 Q_k a = H_p * a (where H_p = 2 sum p_k Q_k)
    return _hvp_pv(a, p, n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS optimization in log-space with adaptive beta continuation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.001)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 5))))

        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _full_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c_local))
                return 4.0 * n * a_local * (h - W_val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _run_primal_dual(a_init, deadline, min_time_remaining=60.0):
    """Primal-dual iteration: alternate between QP primal solve and softmax dual update."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    best_a = a.copy()
    best_sc = _score(a)

    # Beta schedule for dual
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    base_beta = max_c * 0.01
    n_outer = 0
    max_outer = 500

    while time.monotonic() + min_time_remaining < deadline and n_outer < max_outer:
        n_outer += 1
        # Dual update: p = softmax(R / beta)
        c = _full_convolve(a, a)
        beta = base_beta * (0.8 ** min(n_outer, 50))
        p = _softmax(c / beta)

        # Primal step: minimize sum_k p_k R(k) s.t. a in simplex
        # Use projected gradient with a few iterations
        lr = 0.01
        for inner in range(20):
            grad = _grad_smoothed_fast(a, p, n)
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            lr = 0.05 / (gnorm + 1e-10)
            a_new = _proj_simplex(a - lr * grad)
            # Check improvement
            obj_cur = float(np.dot(p, _full_convolve(a, a)))
            a = a_new
            if time.monotonic() + min_time_remaining > deadline:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _run_proj_grad(a_init, deadline, min_time_remaining=60.0, use_momentum=True):
    """Projected gradient with adaptive step size and optional Nesterov momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta_m = 0.9
    patience = 0
    max_patience = 50

    for step in range(5000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        # Top-k active set
        k_active = min(5, len(c))
        top_k = np.argsort(c)[::-1][:k_active]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_active

        grad = _grad_smoothed_fast(a_curr, p, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad
        lr = 0.02 * (1.0 + 0.1 * step) / (np.linalg.norm(momentum) / grad_norm + 1e-10)
        z_new = z - lr * momentum
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if step % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    beta_m *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_fw(a_init, deadline, min_time_remaining=60.0):
    """Frank-Wolfe / Conditional Gradient method for the smoothed problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for k in range(3000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Top-k active set
        k_act = min(10, len(c))
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_act

        grad = _grad_smoothed_fast(a, p, n)
        # FW step: find s = argmin grad, move toward e_s
        s = int(np.argmin(grad))
        # LAR step size
        # t = (grad[s] - grad[a.argmax()]) / ||a - e_s||^2 approximately
        t_fw = 2.0 / (k + 2)
        a_new = (1 - t_fw) * a + t_fw * np.zeros(n)
        a_new[s] += t_fw

        # Check if this improves
        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        if c_max_new < c_max:
            a = a_new
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = (1 - t_fw * factor) * a + t_fw * factor * np.zeros(n)
                a_try[s] += t_fw * factor
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break

    return best_a, best_sc


def _active_set_qp(a_init, deadline, min_time_remaining=60.0, max_rounds=100):
    """Active-set QP with progressive tightening using projected gradient inner solve."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 5

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()

        # Solve QP: min sum_k p_k R(k) s.t. a in simplex
        # Using projected gradient with adaptive step
        lr = 0.01
        for inner in range(50):
            grad = _grad_smoothed_fast(a, p, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            lr = 0.1 / (gnorm * (1.0 + 0.01 * inner))
            a_try = _proj_simplex(a - lr * grad)
            a = a_try

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif rnd > 5:
            a = best_a.copy()

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _flatten_direct(a_init, deadline, min_time_remaining=60.0):
    """Direct flattening: iteratively push down the peak autocorrelation values."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find peaks
        peaks = c > c_max * 0.999
        if peaks.sum() == 0:
            peaks[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[peaks] = 1.0 / peaks.sum()

        # Gradient of the peak objective
        grad = _grad_smoothed_fast(a, p, n)
        # Step in negative gradient direction, project to simplex
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 0.05 / (gnorm + 1e-10)
        for ls in range(30):
            a_new = _proj_simplex(a - step * grad)
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max:
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5
            if step < 1e-12:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "ramp2":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.1
        return a / a.sum()
    elif kind == "bump2":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, primal-dual QP, projected gradient,
    Frank-Wolfe, and active-set methods. Explores multiple sequence lengths
    and diverse initial profiles to avoid local minima.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Multi-start L-BFGS (300s) ===
    print(f"\n[Phase 1] Multi-start L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 300
    candidates = []

    # Start from parent
    a1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=30.0)
    update_best(a1, _score(a1), "lbfgs_parent")
    candidates.append(a1)

    # Try other profiles at same n
    if time.monotonic() < phase1_deadline - 60:
        for prof in ["cos", "ramp", "bump"]:
            if time.monotonic() > phase1_deadline - 30:
                break
            a_init = _make_profile(n_parent, prof)
            a_p = _run_lbfgs(a_init, phase1_deadline, min_time_remaining=20.0)
            sc_p = _score(a_p)
            if sc_p < best_score + 0.01:
                update_best(a_p, sc_p, f"lbfgs_{prof}")
                candidates.append(a_p)

    # Try perturbation of parent
    if time.monotonic() < phase1_deadline - 30:
        a_pert = _perturb(parent, 0.03, rng)
        a_p = _run_lbfgs(a_pert, phase1_deadline, min_time_remaining=10.0)
        update_best(a_p, _score(a_p), "lbfgs_perturb")

    # === Phase 2: Primal-dual optimization (400s) ===
    print(f"\n[Phase 2] Primal-Dual (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 600, phase1_deadline + 400)
    if best_seq is not None and time.monotonic() < phase2_deadline:
        a_pd, sc_pd = _run_primal_dual(best_seq, phase2_deadline, min_time_remaining=30.0)
        update_best(a_pd, sc_pd, "primal_dual_1")

    # === Phase 3: Active-set QP (300s) ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = min(deadline - 400, phase2_deadline + 300)
    if best_seq is not None and time.monotonic() < phase3_deadline:
        a_qp, sc_qp = _active_set_qp(best_seq, phase3_deadline, min_time_remaining=30.0,
                                      max_rounds=80)
        update_best(a_qp, sc_qp, "active_qp_1")

    # === Phase 4: Projected gradient with momentum (200s) ===
    print(f"\n[Phase 4] Projected Gradient (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = min(deadline - 250, phase3_deadline + 200)
    if best_seq is not None and time.monotonic() < phase4_deadline:
        a_pg, sc_pg = _run_proj_grad(best_seq, phase4_deadline, min_time_remaining=20.0)
        update_best(a_pg, sc_pg, "proj_grad_1")

    # === Phase 5: Direct flattening (150s) ===
    print(f"\n[Phase 5] Direct Flatten (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = min(deadline - 200, phase4_deadline + 150)
    if best_seq is not None and time.monotonic() < phase5_deadline:
        a_fl, sc_fl = _flatten_direct(best_seq, phase5_deadline, min_time_remaining=20.0)
        update_best(a_fl, sc_fl, "flatten_1")

    # === Phase 6: Second round of combined optimization (200s) ===
    print(f"\n[Phase 6] Combined Round 2 (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = min(deadline - 150, phase5_deadline + 200)
    if best_seq is not None and time.monotonic() < phase6_deadline:
        # L-BFGS with very tight beta
        a_lb2 = _run_lbfgs(best_seq, phase6_deadline, min_time_remaining=30.0)
        update_best(a_lb2, _score(a_lb2), "lbfgs_r2")

        if time.monotonic() < phase6_deadline - 30:
            a_qp2, sc_qp2 = _active_set_qp(best_seq, phase6_deadline,
                                            min_time_remaining=20.0, max_rounds=60)
            update_best(a_qp2, sc_qp2, "active_qp_2")

        if time.monotonic() < phase6_deadline - 15:
            a_pg2, sc_pg2 = _run_proj_grad(best_seq, phase6_deadline, min_time_remaining=10.0)
            update_best(a_pg2, sc_pg2, "proj_grad_2")

    # === Phase 7: Multi-n exploration (200s) ===
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = min(deadline - 100, phase6_deadline + 200)
    n_tries = [32768, 24576, 8192]
    for n_try in n_tries:
        if time.monotonic() > phase7_deadline - 30:
            break
        if n_try == n_parent:
            continue
        a_init_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_n = _run_lbfgs(a_init_n, phase7_deadline, min_time_remaining=20.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

        if sc_n < best_score + 0.005 and time.monotonic() < phase7_deadline - 20:
            a_nq, sc_nq = _active_set_qp(a_n, phase7_deadline,
                                          min_time_remaining=10.0, max_rounds=30)
            update_best(a_nq, sc_nq, f"n_{n_try}_qp")

    # === Phase 8: FW exploration (100s) ===
    print(f"\n[Phase 8] Frank-Wolfe (t={time.monotonic() - t_start:.1f}s)")
    phase8_deadline = min(deadline - 60, phase7_deadline + 100)
    if best_seq is not None and time.monotonic() < phase8_deadline:
        a_fw, sc_fw = _run_fw(best_seq, phase8_deadline, min_time_remaining=10.0)
        update_best(a_fw, sc_fw, "fw")

    # === Phase 9: Final ultra-tight polish (60s) ===
    print(f"\n[Phase 9] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very tight active set
        a_fin, sc_fin = _active_set_qp(best_seq, deadline - 20,
                                        min_time_remaining=5.0, max_rounds=50)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 15:
            a_fl2, sc_fl2 = _flatten_direct(best_seq, deadline - 15, min_time_remaining=5.0)
            update_best(a_fl2, sc_fl2, "final_flatten")

        # Final L-BFGS with extremely tight beta
        if time.monotonic() < deadline - 10:
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta_fin = mb * 1e-8
            remaining = deadline - 10 - time.monotonic()
            if remaining > 3:
                max_iter = min(1000, max(50, int(remaining / 0.01)))

                def final_obj(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        return 2.0 * n * c_max_local
                    return 2.0 * n * (c_max_local + beta_fin * np.log(s_e))

                def final_grad(z_local):
                    a_local = _softmax(z_local)
                    c_local = _full_convolve(a_local, a_local)
                    c_max_local = float(c_local.max())
                    shifted = (c_local - c_max_local) / beta_fin
                    e = np.exp(np.clip(shifted, -700, 0))
                    s_e = e.sum()
                    if s_e < 1e-300:
                        w = np.zeros_like(c_local)
                        w[int(np.argmax(c_local))] = 1.0
                    else:
                        w = e / s_e
                    a_rev = a_local[::-1]
                    wc = _full_convolve(w, a_rev)
                    h = wc[n - 1:2 * n - 1]
                    W_val = float(np.dot(w, c_local))
                    return 4.0 * n * a_local * (h - W_val)

                try:
                    result = minimize(
                        final_obj, z,
                        jac=final_grad,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 40}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
