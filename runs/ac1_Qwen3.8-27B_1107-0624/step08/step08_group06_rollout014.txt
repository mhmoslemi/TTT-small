    """Full linear convolution using FFT"""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    c_max = float(c.max())
    return float(2.0 * n * c_max / (s * s))


def _softmax(z):

def _cg_solve(matvec, rhs, max_cg=200, tol=1e-10):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

        c = _full_convolve(y, y)
        c_max = float(c.max())

        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1: 2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * y * (h - W)
        g_t = grad - np.mean(grad)

        step_size = 0.02 / (1.0 + 0.02 * step)
        a_new = y - step_size * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = a_new + ((t_param - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t_param = t_new

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_chirp(n_parent)
        print(f"No parent, using chirp n={n_parent}")
        update_best(parent, _score(parent), "init_chirp")

    # Phase 1: L-BFGS with beta schedule from parent
    print(f"\n[Phase 1] L-BFGS beta schedule from parent (t={time.monotonic() - t_start:.1f}s)")
    min_rem_1 = 700.0
    a_main = _run_lbfgs_beta_schedule(parent, deadline, min_time_remaining=min_rem_1, n_phases=10)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_phase1")

    # Phase 2: Peak shaving
    print(f"\n[Phase 2] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        def_dl = min(deadline - 500, time.monotonic() + 60)
        a_def, sc_def = _peak_shaving(best_seq, def_dl, n_rounds=20)
        update_best(a_def, sc_def, "peak_shave_1")

        if sc_def < best_score:
            a_post = _run_lbfgs_beta_schedule(a_def, deadline, min_time_remaining=400.0, n_phases=6)
            update_best(a_post, _score(a_post), "lbfgs_post_shave")

    # Phase 3: Active set Newton
    print(f"\n[Phase 3] Active set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        as_dl = min(deadline - 400, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=250,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 300:
            as_dl2 = min(deadline - 300, time.monotonic() + 80)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=150,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 4: More peak shaving + LBFGS
    print(f"\n[Phase 4] Second peak shaving + LBFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        def_dl2 = min(deadline - 250, time.monotonic() + 40)
        a_def2, sc_def2 = _peak_shaving(best_seq, def_dl2, n_rounds=25)
        update_best(a_def2, sc_def2, "peak_shave_2")

        if sc_def2 < best_score + 0.001 and time.monotonic() < deadline - 180:
            a_post2 = _run_lbfgs_beta_schedule(a_def2, deadline, min_time_remaining=150.0, n_phases=5)
            update_best(a_post2, _score(a_post2), "lbfgs_post_shave2")

    # Phase 5: FISTA subgradient
    print(f"\n[Phase 5] FISTA subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        fista_dl = min(deadline - 150, time.monotonic() + 50)
        a_fista = _fista_subgradient(best_seq, fista_dl, n_steps=8000)
        sc_fista = _score(a_fista)
        update_best(a_fista, sc_fista, "fista")

    # Phase 6: Diverse initializations
    print(f"\n[Phase 6] Diverse initializations (t={time.monotonic() - t_start:.1f}s)")
    n_main = len(best_seq) if best_seq is not None else 16384

    diverse_inits = [
        ("chirp", _make_chirp(n_main)),
        ("asym_block", _make_asymmetric_block(n_main)),
        ("three_block", _make_three_block_asym(n_main)),
        ("rand_smooth_1", _make_random_smooth(n_main, 123)),
        ("rand_smooth_2", _make_random_smooth(n_main, 456)),
    ]

    # Also try interpolating parent to different n
    if parent is not None and n_parent != n_main:
        diverse_inits.append(("interp_parent", _interpolate_to_n(parent, n_main)))

    for label, a_init in diverse_inits:
        if time.monotonic() > deadline - 60:
            break
        sc_init = _score(a_init)
        if sc_init < best_score:
            update_best(a_init, sc_init, f"diverse_{label}")

        if sc_init < best_score + 0.02 and time.monotonic() < deadline - 30:
            a_r = _run_lbfgs_beta_schedule(a_init, deadline, min_time_remaining=25.0, n_phases=4)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"diverse_{label}_opt")

    # Phase 7: Structured perturbations of best
    print(f"\n[Phase 7] Perturbations (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for sigma in [0.03, 0.08, 0.15]:
            if time.monotonic() > deadline - 20:
                break
            p = _perturb_structured(best_seq, sigma, rng)
            a_r = _run_lbfgs_beta_schedule(p, deadline, min_time_remaining=15.0, n_phases=3)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"perturb_{sigma}")

    # Phase 8: Final polish with tight active set + peak shaving
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        as_final = min(deadline - 4, time.monotonic() + 50)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=200,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 3:
            a_def3, sc_def3 = _peak_shaving(best_seq, min(deadline - 2, time.monotonic() + 15), n_rounds=15)
            update_best(a_def3, sc_def3, "final_shave")

        if time.monotonic() < deadline - 5:
            n_fin = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z_fin = np.log(np.maximum(best_seq, 1e-30))
            z_fin = z_fin - z_fin.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 6 - time.monotonic()
            if remaining > 5:
                max_iter = min(3000, max(100, int(remaining / 0.04)))
                try:
                    result = minimize(
                        lambda z_local: _ls_objective_no_scale(z_local, beta_final, n_fin),
                        z_fin,
                        jac=lambda z_local: _ls_gradient_no_scale(z_local, beta_final, n_fin),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 100}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
