            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score_unit(a_new)
            if sc_new < best_sc - 1e-15:
                best_a = a_new
                best_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a


def _momentum_polish(a_init, deadline, n_iter_max=50000, eta_init=0.005, 
                     momentum=0.9, cool_rate=0.99995):
    """Momentum-based gradient descent on the LSE objective (in a-space directly)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score_unit(a)
    v = np.zeros(n)
    eta = eta_init

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    T = gap * 0.5

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        # Compute weights
        shifted = (c - c_max) / T
        e = np.exp(np.clip(shifted, -500, 0))
        w = e / e.sum()

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h - 2.0 * np.dot(w, c)  # centered gradient

        # Update with momentum
        v = momentum * v + grad
        a_new = a - eta * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            v *= 0.5
            continue
        a_new = a_new / s_new

        c = _full_convolve(a_new, a_new)
        c_max = float(c.max())
        sc_new = 2.0 * n * c_max

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()

        # Backtracking if we increased the score significantly
        if sc_new > best_sc * 1.01:
            eta *= 0.5
            v *= 0.9
            a = best_a.copy()
            c = _full_convolve(a, a)
            c_max = float(c.max())
        else:
            a = a_new

        # Adapt temperature
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
        if gap > c_max * 0.0001:
            T = max(gap * 0.3, c_max * 0.0005)
        else:
            T *= 0.98

        eta = max(1e-6, eta * cool_rate)

    return best_a, best_sc


def _peak_equalize(a_init, deadline, n_iters=10000):
    """Targeted peak equalization: reduce the highest peaks of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_unit(a)

    eta = 0.01
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks (within 1% of max)
        threshold = c_max * 0.99
        active = c >= threshold
        n_active = active.sum()
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Uniform weights over active set
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Gradient for this active set
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)

        # Normalized gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Try the step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            eta = min(0.05, eta * 1.02)
        else:
            eta *= 0.7
            if eta < 1e-6:
                break

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """Multi-n adaptive L-BFGS with momentum polish and peak equalization."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)  # Leave 15s buffer
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 16384
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Multi-n L-BFGS ===
    # Try parent n, and larger n values
    n_candidates = sorted(set([n_parent, 16384, 32768]))
    # Remove if too large (memory/time concerns)
    n_candidates = [n for n in n_candidates if n <= 65536]
    
    print(f"\n[P1] Multi-n L-BFGS: candidates = {n_candidates}")
    
    # Allocate time: more time for larger n
    time_per_n = {}
    total_time = deadline - t_start - 300  # Reserve 300s for later phases
    # Weight by n (larger n gets more time since it has more potential)
    weights = np.array([n ** 0.5 for n in n_candidates])
    weights = weights / weights.sum()
    for i, n_c in enumerate(n_candidates):
        time_per_n[n_c] = weights[i] * total_time
    
    for n_c in n_candidates:
        if time.monotonic() > deadline - 310:
            break
        t_phase_start = time.monotonic()
        allocated = time_per_n.get(n_c, 200.0)
        phase_deadline = min(t_phase_start + allocated, deadline - 310)
        
        if n_c == n_parent and parent is not None:
            a_init = parent.copy()
        else:
            a_init = _resample(best_a, n_c)
        
        sc_init = _score(a_init)
        print(f"  n={n_c}: init score={sc_init:.10f}, time allocated={allocated:.0f}s")
        
        # L-BFGS
        a_opt = _adapt_lbfgs(a_init, phase_deadline - allocated * 0.3, 
                             min_remaining=10.0, n_beta_stages=20)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_c}) [lbfgs] t={time.monotonic()-t_start:.1f}s")
        
        # Momentum polish
        remaining_mom = min(allocated * 0.3, deadline - 310 - time.monotonic())
        if remaining_mom > 20:
            a_mom, sc_mom = _momentum_polish(a_opt, time.monotonic() + remaining_mom)
            if sc_mom < best_score:
                best_a = a_mom.copy()
                best_score = sc_mom
                print(f"  NEW BEST: {best_score:.10f} (n={n_c}) [momentum] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Peak equalization on best ===
    print(f"\n[P2] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    pe_deadline = min(deadline - 100, time.monotonic() + 200)
    a_pe, sc_pe = _peak_equalize(best_a, pe_deadline)
    if sc_pe < best_score:
        best_a = a_pe.copy()
        best_score = sc_pe
        print(f"  NEW BEST: {best_score:.10f} [peak_eq] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Final L-BFGS polish ===
    print(f"\n[P3] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        a_fin = _adapt_lbfgs(best_a, deadline - 20, min_remaining=10, n_beta_stages=15)
        sc_fin = _score(a_fin)
