"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(ind[cond])
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0.0)

def _subgradient_bb(a, deadline, max_iter=50000, top_k=3):
    """Subgradient method with Barzilai-Borwein step size."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a, a)
    best_max = float(c.max())
    
    a_prev = a.copy()
    g_prev = None
    lr = 0.001
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Compute subgradient (average over top-k active peaks)
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]
        # Only use peaks within 0.999 of max
        active = [k for k in active if c[k] > c_max * 0.999]
        if not active:
            active = [int(sorted_idx[0])]
        
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project to tangent space (sum-preserving)
        g_tan = g - g.mean()
        
        # Barzilai-Borwein step size
        if g_prev is not None:
            s_vec = a - a_prev
            y_vec = g - g_prev
            s_norm2 = float(np.dot(s_vec, s_vec))
            y_norm2 = float(np.dot(y_vec, y_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm2 > 1e-20 and y_norm2 > 1e-20:
                # BB step: alpha = <s,s>/<s,y> or <s,y>/<y,y>
                if abs(sy) > 1e-20:
                    lr_bb1 = s_norm2 / abs(sy)
                    lr_bb2 = abs(sy) / y_norm2
                    lr = 0.5 * (lr_bb1 + lr_bb2)
                lr = np.clip(lr, 1e-6, 0.01)
        
        a_prev = a.copy()
        g_prev = g.copy()
        
        # Take step
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break
        step = lr * g_tan / g_norm
        a_new = a - step
        a_new = _project_simplex(a_new)
        
        c_new = _fft_convolve(a_new, a_new)
        max_new = float(c_new.max())
        
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack
            lr *= 0.8
            if lr < 1e-9:
                break
    
    return best_a, best_max

            for scale in [0.5, 0.7, 0.85, 0.9, 1.1, 1.3, 1.5]:
                a_cand = a.copy()
                a_cand[j] = old_val * scale
                s_cand = a_cand.sum()
                if s_cand < 1e-15:
                    continue
                a_cand /= s_cand
                c_cand = _fft_convolve(a_cand, a_cand)
                max_cand = float(c_cand.max())
                if max_cand < best_local_max:
                    best_local_max = max_cand
                    best_scale = scale
            
            if best_scale != 1.0 and best_local_max < best_max * (1 - 1e-12):
                a[j] = old_val * best_scale
                a /= a.sum()
                c = _fft_convolve(a, a)
                best_max = float(c.max())
                best_a = a.copy()
                improved_this_round = True
        
        if not improved_this_round:
            # Try all elements with smaller perturbations
            any_improve = False
            step_size = max(1, n // 400)
            for j in range(0, n, step_size):
                old_val = a[j]
                if old_val < 1e-14:
                    continue
                for scale in [0.9, 1.1]:
                    a_cand = a.copy()
                    a_cand[j] = old_val * scale
                    a_cand /= a_cand.sum()
                    c_cand = _fft_convolve(a_cand, a_cand)
                    max_cand = float(c_cand.max())
                    if max_cand < best_max * (1 - 1e-10):
                        best_max = max_cand
                        best_a = a_cand.copy()
                        a = a_cand
                        c = c_cand
                        any_improve = True
                        break
                if any_improve:
                    break
            if not any_improve:
                # Try with finer step
                step_size = max(1, n // 1000)
                for j in range(0, n, step_size):
                    old_val = a[j]
                    if old_val < 1e-14:
                        continue
                    for scale in [0.95, 1.05]:
                        a_cand = a.copy()
                        a_cand[j] = old_val * scale
                        a_cand /= a_cand.sum()
                        c_cand = _fft_convolve(a_cand, a_cand)
                        max_cand = float(c_cand.max())
                        if max_cand < best_max * (1 - 1e-10):
                            best_max = max_cand
                            best_a = a_cand.copy()
                            a = a_cand
                            c = c_cand
                            any_improve = True
                            break
                    if any_improve:
                        break
                if not any_improve:
                    break
    
    return best_a, best_max


def _simulated_annealing(a, deadline, max_iter=200000, T_start=0.001, T_end=1e-6):
    """Simulated annealing with single-element perturbations."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a, a)
    best_max = float(c.max())
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (it / max_iter)
        
        # Propose: move a small amount of mass from one element to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Amount to move (as fraction of a[i])
        frac = rng.uniform(0.05, 0.3)
        delta = a[i] * frac
        
        a[i] -= delta
        a[j] += delta
        if a[i] < 0 or a[j] < 0:
            a[i] += delta
            a[j] -= delta
            continue
        
        s = a.sum()
        a /= s
        
        c_new = _fft_convolve(a, a)
        max_new = float(c_new.max())
        delta_obj = max_new - c.max()
        
        # Metropolis criterion
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / (T * c.max() + 1e-15)):
            c = c_new
            if max_new < best_max:
                best_max = max_new
                best_a = a.copy()
        else:
            # Revert
            a[j] -= delta
            a[i] += delta
            a /= a.sum()
    
    return best_a, best_max


def _try_different_n(a_parent, n_target, deadline):
    """Try to construct a good sequence of a different length."""
    n_orig = len(a_parent)
    
    if n_target == n_orig:
        return a_parent
    
    # Interpolate or truncate
    if n_target < n_orig:
        # Keep the middle portion (where most energy typically is)
        # Or just take every other element
        idx = np.linspace(0, n_orig - 1, n_target).astype(int)
        a = a_parent[idx]
    else:
        # Interpolate
        x_old = np.linspace(0, 1, n_orig)
        x_new = np.linspace(0, 1, n_target)
        a = np.interp(x_new, x_old, a_parent)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _fast_gradient_step(a, deadline, max_iter=100000, lr_start=0.01, lr_end=0.0001):
    """
    Fast gradient descent with smooth approximation of max.
    Uses log-sum-exp for smoothness.
    """
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a, a)
    best_max = float(c.max())
    
    c_max_initial = c.max()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        frac = it / max_iter
        lr = lr_start * (lr_end / lr_start) ** frac
        # Temperature for log-sum-exp (adaptive)
        T = c_max_initial * 0.001 * (1 + 100 * frac)
        
        # Compute smooth max gradient
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            # Fall back to hard max
            k_star = int(np.argmax(c))
            g = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            if lo <= hi:
                g[lo:hi+1] = 2.0 * a[k_star-hi:k_star-lo+1][::-1]
        else:
            p = e / s
            # Gradient: g[j] = 2 * sum_k p[k] * a[k-j]
            # Use only top contributing lags for efficiency
            g = np.zeros(n)
            # Get top lags by p value
            top_lags = np.argsort(p)[::-1][:20]
            for k in top_lags:
                if p[k] < 1e-6:
                    break
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if lo <= hi:
                    g[lo:hi+1] += p[k] * 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Project gradient to preserve sum
        g_tan = g - g.mean()
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break
        
        # Step
        a_new = a - lr * g_tan / g_norm
        a_new = _project_simplex(a_new)
        
        c_new = _fft_convolve(a_new, a_new)
        max_new = float(c_new.max())
        
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
    
    return best_a, best_max


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining peak-shaving multiplicative updates,
    active-set Newton, subgradient with BB step, coordinate descent,
    and simulated annealing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")
    
    n = len(a)
    best_a = a.copy()
    c_best = _fft_convolve(best_a, best_a)
    best_score = _score_full(best_a)
    print(f"Initial: n={n}, score={best_score:.10f}, max_conv={c_best.max():.2e}, t={time.monotonic()-t_start:.1f}s")
    
    # ==================== Phase 1: Fast gradient (0 - 200s) ====================
    phase1_end = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: Fast gradient ---")
    a_opt, _ = _fast_gradient_step(best_a, phase1_end, max_iter=100000, lr_start=0.008, lr_end=0.0005)
