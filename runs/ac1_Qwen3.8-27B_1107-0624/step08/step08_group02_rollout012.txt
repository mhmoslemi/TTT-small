        #     = -H^{-1}*g_t + λ*H^{-1}*1
        #     = -(H^{-1}*g_t) + λ*(H^{-1}*1)
        
        # Let u = H^{-1}*g_t and v = H^{-1}*1 (both via CG)
        # λ = -(1^T*u) / (1^T*v) = -sum(u)/sum(v)
        # dx = -u + λ*v
        
        # Wait, I need to be more careful with the sign. The Newton equation is:
        # P*H*P*dx = -P*g = -g_t (since P*g = g - mean(g)*1 = g_t)
        # And we want sum(dx) = 0.
        # The equation P*H*P*dx = -g_t with sum(dx)=0 can be solved as:
        # H*dx = -g_t + β*1 (for some β, since P*H*P*dx = P*(H*dx) and we need P*(H*dx) = -g_t,
        # which means H*dx = -g_t + β*1 for some β)
        # Taking inner product with 1: 1^T*H*dx = -1^T*g_t + β*n = 0 + β*n => β = -(1^T*H*dx)/n
        # This is still circular. Let me use the augmented system approach.
        
        # Solve H*dx = -g_t + λ*1 with constraint 1^T*dx = 0.
        # From the first equation: dx = H^{-1}*(-g_t + λ*1) = -H^{-1}*g_t + λ*H^{-1}*1
        # Constraint: 1^T*(-H^{-1}*g_t + λ*H^{-1}*1) = 0
        # => -1^T*H^{-1}*g_t + λ*1^T*H^{-1}*1 = 0
        # => λ = (1^T*H^{-1}*g_t) / (1^T*H^{-1}*1) = sum(u) / sum(v)
        # where u = H^{-1}*g_t, v = H^{-1}*1
        
        # So: dx = -u + (sum(u)/sum(v)) * v
        
        # Let me compute u and v using CG.
        
        u = _cg_solve(w, n, g_t, max_iter=600)
        v = _cg_solve(w, n, np.ones(n), max_iter=600)
        
        sum_u = float(np.sum(u))
        sum_v = float(np.sum(v))
        
        if abs(sum_v) < 1e-20:
            # Fallback: steepest descent
            dx = -g_t / gn
        else:
            lam = sum_u / sum_v
            dx = -u + lam * v
        
        # Ensure sum(dx) ≈ 0
        dx = dx - np.mean(dx)
        
        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-30 or not np.isfinite(dx_norm):
            dx = -g_t / gn
            dx_norm = float(np.linalg.norm(dx))
        
        # Line search with cubic interpolation
        alpha = 1.0
        step_done = False
        a_old = a.copy()
        f_old = c_max  # current max
        g_old_dot_dx = float(np.dot(g_t, dx))
        
        if g_old_dot_dx >= 0:
            # Not a descent direction, use steepest descent
            dx = -g_t / gn
            g_old_dot_dx = -gn
        
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            
            if f_try < f_old + 1e-15:
                # Check sufficient decrease (Armijo)
                if f_try <= f_old + 0.0001 * alpha * g_old_dot_dx:
                    a = a_try
                    c_max = f_try
                    step_done = True
                    break
                
                # Even if not Armijo, accept if we make progress
                if f_try < f_old - 1e-14:
                    a = a_try
                    c_max = f_try
                    step_done = True
                    break
            
            alpha *= 0.5
            if alpha < 1e-18:
                break
        
        if not step_done:
            # Try smaller steepest descent steps
            for trial_alpha in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial_alpha * g_t / gn
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    c_max = float(c_try.max())
                    step_done = True
                    break
        
        # Update score
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt beta
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 10:
            gap = c_sorted[0] - c_sorted[10]
        else:
            gap = c_sorted[0] - c_sorted[-1]
        
        if no_improve > 3:
            beta *= 0.5
        elif gap < 1e-8 * c_max:
            beta *= 0.3
        else:
            beta *= 0.7
        
        beta = max(beta, beta_min)
        
        if no_improve > max_no_improve and beta <= beta_min * 1.1:
            break
    
    return best_a, best_sc
