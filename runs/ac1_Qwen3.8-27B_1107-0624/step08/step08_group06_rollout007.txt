def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute gradient of sum_k A_vec[k]*conv(a,a)[k] w.r.t. a."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s


def _lsobj(z, beta, n):

def _cg_solve(matvec, rhs, max_cg=500, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _local_swaps(a_init, deadline, n_rounds=100):
    """Coordinate descent: transfer small amounts between paired elements."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 1:
            break

        # Try several random pairs
        for _ in range(20):
            if time.monotonic() > deadline - 1:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Transfer from larger to smaller (or vice versa)
            if a[i] < a[j]:
                i, j = j, i
            # Try small transfer
            delta = a[i] * 0.01
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                a = a_try
                best_sc = sc_try
                best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform", seed=42):
    rng = np.random.default_rng(seed)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        center = 0.5
        width = 0.15
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_wide":
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / 0.3) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tent":
        t = np.linspace(0, 1, n)
        a = np.maximum(0, 1.0 - 2.0 * np.abs(t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tent_wide":
        t = np.linspace(0, 1, n)
        a = np.maximum(0, 1.0 - np.abs(t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = max(1, n // 4)
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        a = rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    # Phase 1: Quick diverse initializations to find good starting points
    print(f"\n[Phase 1] Diverse initializations (t={time.monotonic() - t_start:.1f}s)")
    localhost_profiles = ["gaussian", "gaussian_wide", "tent", "tent_wide", "cos", "block", "sinusoidal"]
    best_init = None
    best_init_sc = best_score

    for prof in localhost_profiles:
        if time.monotonic() > deadline - 1400:
            break
        a_try = _make_profile(n_parent, prof, seed=int(seed) + hash(prof) % 1000)
        sc_try = _score(a_try)
        print(f"  Profile '{prof}': score={sc_try:.6f}")
        if sc_try < best_init_sc:
            best_init_sc = sc_try
            best_init = a_try.copy()
        update_best(a_try, sc_try, f"init_{prof}")

    # Use the best of parent and diverse inits
    work_a = parent.copy()
    if best_init is not None and best_init_sc < _score(parent):
        work_a = best_init.copy()

    # Phase 2: L-BFGS from best starting point
    print(f"\n[Phase 2] L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 800, time.monotonic() + 600)
    a_main = _run_lbfgs(work_a, lbfgs_dl, min_time_remaining=800.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 3: Soft active-set Newton
    print(f"\n[Phase 3] Soft active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_dl = min(deadline - 500, time.monotonic() + 200)
        a_as, sc_as = _soft_active_set_newton(best_seq, as_dl, max_iter=500,
                                              tau_start=0.03, tau_end=0.0005)
        update_best(a_as, sc_as, "soft_as_1")

        if time.monotonic() < deadline - 400:
            as_dl2 = min(deadline - 400, time.monotonic() + 120)
            a_as2, sc_as2 = _soft_active_set_newton(best_seq, as_dl2, max_iter=300,
                                                    tau_start=0.01, tau_end=0.0001)
            update_best(a_as2, sc_as2, "soft_as_2")

    # Phase 4: Peak reduction
    print(f"\n[Phase 4] Peak reduction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        pr_dl = min(deadline - 300, time.monotonic() + 80)
        a_pr, sc_pr = _peak_reduction(best_seq, pr_dl, n_steps=1000)
        update_best(a_pr, sc_pr, "peak_red")

    # Phase 5: Spectral shaping
    print(f"\n[Phase 5] Spectral shaping (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        for alpha in [0.8, 0.7, 0.6]:
            if time.monotonic() > deadline - 200:
                break
            a_sp, sc_sp = _spectral_shape(best_seq, n_iters=30, alpha=alpha)
            update_best(a_sp, sc_sp, f"spectral_a{alpha}")
            # L-BFGS from spectral result
            if sc_sp < best_score + 0.01 and time.monotonic() < deadline - 100:
                a_sp2 = _run_lbfgs(a_sp, min(deadline - 80, time.monotonic() + 60),
                                   min_time_remaining=80.0)
                update_best(a_sp2, _score(a_sp2), f"spectral_a{alpha}_lbfgs")

    # Phase 6: FISTA
    print(f"\n[Phase 6] FISTA (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        fista_dl = min(deadline - 150, time.monotonic() + 60)
        a_fista = _fista_projected(best_seq, fista_dl, n_steps=15000)
        sc_fista = _score(a_fista)
        update_best(a_fista, sc_fista, "fista")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        a_opt = _run_lbfgs(a_init, min(deadline - 80, time.monotonic() + 120),
                           min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 60:
            a_opt2, sc_opt2 = _soft_active_set_newton(a_opt,
                                                       min(deadline - 40, time.monotonic() + 50),
                                                       max_iter=200, tau_start=0.02, tau_end=0.001)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 8: Local swaps + perturbation restarts
    print(f"\n[Phase 8] Local search & restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        # Local swaps
        ls_dl = min(deadline - 30, time.monotonic() + 30)
        a_ls, sc_ls = _local_swaps(best_seq, ls_dl, n_rounds=50)
        update_best(a_ls, sc_ls, "local_swaps")

        # Perturbation restarts
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, min(deadline - 15, time.monotonic() + 40),
                             min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_s{sigma}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _soft_active_set_newton(a_r,
                                                       min(deadline - 5, time.monotonic() + 20),
                                                       max_iter=100, tau_start=0.01, tau_end=0.0005)
                update_best(a_r2, sc_r2, f"restart_s{sigma}_as")

    # Phase 9: Final polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Tight soft active-set
        as_final = min(deadline - 3, time.monotonic() + 40)
        a_fin, sc_fin = _soft_active_set_newton(best_seq, as_final, max_iter=300,
                                                tau_start=0.005, tau_end=0.0001)
        update_best(a_fin, sc_fin, "final_as")

        # Peak reduction
        if time.monotonic() < deadline - 5:
            a_pr2, sc_pr2 = _peak_reduction(best_seq, min(deadline - 3, time.monotonic() + 20),
                                            n_steps=500)
            update_best(a_pr2, sc_pr2, "final_peak")

        # Final tiny-beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 6 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(100, int(remaining / 0.04)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 80, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
