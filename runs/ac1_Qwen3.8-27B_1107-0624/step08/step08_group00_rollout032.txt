Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam with adaptive log-sum-exp smoothing, direct peak shaving,
coordinate descent polishing, and multi-n exploration.
"""

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    return c_max + beta * np.log(s), e / s

class _AdamState:
    def __init__(self, n):

        self.v = np.zeros(n)

        self.b1 = 0.9
        self.b2 = 0.999

def _adam_step(state, grad, lr):
    state.t += 1
    t = state.t
    state.m = state.b1 * state.m + (1 - state.b1) * grad
    state.v = state.b2 * state.v + (1 - state.b2) * (grad ** 2)
    m_hat = state.m / (1 - state.b1 ** t)
    v_hat = state.v / (1 - state.b2 ** t)
    return -lr * m_hat / (np.sqrt(v_hat) + state.eps)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsmax_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    return c_max + beta * np.log(s), e / s

def _compute_h(w, a):
    """Compute h[j] = sum_k w[k]*a[k-j] for j=0..n-1 (cross-correlation)."""
    n = len(a)
    a_rev = a[::-1]
    N = _next_pow2(len(w) + n - 1)
    hw = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a_rev, N), N)
    return hw[n-1:2*n-1]

def _resample(a, new_n):

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    return c_max + beta * np.log(s), e / s

def _compute_h(w, a):
    """h[j] = sum_k w[k]*a[k-j] for j=0..n-1 (cross-correlation)."""
    n = len(a)
    a_rev = a[::-1]
    N = _next_pow2(len(w) + n - 1)
    hw = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a_rev, N), N)
    return hw[n - 1:2 * n - 1]

def _obj_grad(z, beta, n):
    """Compute smoothed objective and gradient w.r.t. log-parameter z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    val, w = _lsmax(c, beta)
    obj = 2.0 * n * val
    h = _compute_h(w, a)
    g = 4.0 * n * h
    H = float(np.dot(g, a))
    grad = a * (g - H)
    return obj, grad

def _adam_optimize(z_init, n, deadline, min_time_remaining=30.0):
    """Adam with adaptive decreasing beta schedule."""
    a = _softmax(z_init)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = c_max * 0.005
    base_beta = max(gap, c_max * 0.003)

    z = z_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    # Beta schedule: geometric decrease
    n_stages = 12
    betas = [base_beta * (0.45 ** i) for i in range(n_stages)]

    for si, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Time allocation: more time for coarser stages
        frac = (1.0 / (1.0 + 0.15 * si))
        total_frac = sum(1.0 / (1.0 + 0.15 * j) for j in range(si, n_stages))
        time_alloc = remaining * frac / total_frac

        est_time_per_iter = 0.0015 * max(1.0, n / 8192.0)
        n_iters = max(50, int(time_alloc / (est_time_per_iter * 3)))

        # Adam state
        m = np.zeros(n)
        vx = np.zeros(n)
        t = 0
        b1, b2, eps = 0.9, 0.999, 1e-8

        # Learning rate: anneal within each stage
        lr_start = 0.3 if si < 4 else (0.1 if si < 8 else 0.03)
        lr_end = lr_start * 0.1

        for it in range(n_iters):
            if time.monotonic() + min_time_remaining > deadline:
                break
            obj, grad = _obj_grad(z, beta, n)

            # Gradient clipping
            gnorm = np.linalg.norm(grad)
            if gnorm > 10.0:
                grad = grad * (10.0 / gnorm)

            t += 1
            m = b1 * m + (1 - b1) * grad
            vx = b2 * vx + (1 - b2) * (grad ** 2)
            m_hat = m / (1 - b1 ** t)
            v_hat = vx / (1 - b2 ** t)

            # Cosine-like annealing
            progress = it / max(1, n_iters)
            lr_t = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))

            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 200 == 0:
                a_cur = _softmax(z)
                sc_cur = _score(a_cur)
                if sc_cur < best_sc:
                    best_sc = sc_cur
                    best_a = a_cur.copy()

        # End of stage
        a_cur = _softmax(z)
        sc_cur = _score(a_cur)
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_a = a_cur.copy()
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
        else:
            # Reset z to best
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()

    return best_a, best_sc

def _lbfgs_refine(a_init, deadline, min_time_remaining=20.0):
    """L-BFGS with very fine log-sum-exp smoothing."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = c_max * 0.001
    beta = max(gap * 0.4, c_max * 0.0002)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    def obj_func(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        val, _ = _lsmax(c, beta)
        return 2.0 * n * val

    def grad_func(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        _, w = _lsmax(c, beta)
        h = _compute_h(w, a)
        g = 4.0 * n * h
        H = float(np.dot(g, a))
        return a * (g - H)

    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 5:
        return best_a, best_sc

    max_iter = min(5000, max(50, int(remaining / 0.002)))

    try:
        result = minimize(obj_func, z, jac=grad_func, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-18,
                                  'gtol': 1e-15, 'maxcor': 100, 'maxls': 50})
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_a = a_new
            best_sc = sc_new
    except Exception:
        pass

    return best_a, best_sc

def _peak_shave(a_init, deadline, min_time_remaining=10.0):
    """Iterative peak shaving with momentum in log-space."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    momentum = np.zeros(n)
    mu = 0.9
    lr = 0.15
    stall = 0

    for it in range(20000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            a = best_a.copy()
            momentum *= 0.5
            lr = max(0.01, lr * 0.7)
            stall = 0
            if lr < 0.01:
                break

        # Active set: peaks within tolerance of max
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = max(1, int(active.sum()))
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        h = _compute_h(w, a)
        g = 4.0 * n * h
        H = float(np.dot(g, a))
        grad = a * (g - H)

        # Momentum step in log-space
        momentum = mu * momentum - lr * grad
        z = np.log(np.maximum(a, 1e-30))
        z_new = z + momentum
        a_try = _softmax(z_new)

        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < sc - 1e-15:
            a = a_try
            lr = min(0.3, lr * 1.01)
        else:
            # Halve step
            z_new = z + momentum * 0.5
            a_try = _softmax(z_new)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-15:
                a = a_try
                momentum *= 0.5
            else:
                momentum *= 0.5
                lr *= 0.85
                if lr < 0.005:
                    break

    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _shrinkage(a, deadline):
    """Try zeroing small elements and re-evaluating."""
    n = len(a)
    a_try = a.copy()
    s = a_try.sum()
    a_try = a_try / s
    best_sc = _score(a_try)

    sorted_idx = np.argsort(a_try)
    for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        s = a_try.sum()
        a_try = a_try / s
        a_try[a_try.argsort()[:n_zero]] = 0.0
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            a = a_try.copy()

    return a, best_sc

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Adam with adaptive beta (coarse to fine) at full resolution
    2. L-BFGS refinement with very fine smoothing
    3. Peak shaving with momentum
    4. Shrinkage + re-optimization
    5. Alternative n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Uniform n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: Adam optimization ===
    print(f"\n[P1] Adam optimization (t={time.monotonic()-t_start:.1f}s)")
    z = np.log(np.maximum(best_seq, 1e-30))
    z = z - z.mean()

    adam_deadline = min(deadline - 300, t_start + 900)
    a_new, sc_new = _adam_optimize(z, n, adam_deadline, min_time_remaining=300.0)
    if sc_new < best_score:
        best_score = sc_new
        best_seq = a_new.copy()
        print(f"  Adam: score={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n[P2] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_new, sc_new = _lbfgs_refine(best_seq, deadline, min_time_remaining=200.0)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new.copy()
            print(f"  L-BFGS: score={best_score:.10f}")

    # === Phase 3: Peak shaving ===
    print(f"\n[P3] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_new, sc_new = _peak_shave(best_seq, deadline, min_time_remaining=100.0)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new.copy()
            print(f"  Peak shave: score={best_score:.10f}")

    # === Phase 4: Second L-BFGS + peak shave ===
    print(f"\n[P4] Second refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_new, sc_new = _lbfgs_refine(best_seq, deadline, min_time_remaining=60.0)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new.copy()
            print(f"  L-BFGS 2: score={best_score:.10f}")
        if time.monotonic() < deadline - 30:
            a_new, sc_new = _peak_shave(best_seq, deadline, min_time_remaining=30.0)
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new.copy()
                print(f"  Peak shave 2: score={best_score:.10f}")

    # === Phase 5: Shrinkage ===
    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 40)
        if sc_shr < best_score:
            best_score = sc_shr
            best_seq = a_shr.copy()
            print(f"  Shrinkage: score={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 30:
                z_shr = np.log(np.maximum(best_seq, 1e-30))
                z_shr = z_shr - z_shr.mean()
                a_new, sc_new = _adam_optimize(z_shr, len(best_seq), deadline - 15,
                                               min_time_remaining=15.0)
                if sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Post-shrink Adam: score={best_score:.10f}")

    # === Phase 6: Final peak shave ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_new, sc_new = _peak_shave(best_seq, deadline, min_time_remaining=5.0)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new.copy()
            print(f"  Final peak shave: score={best_score:.10f}")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = float(final.sum())
        final = final / s
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
