def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        rs = rs_new
        p = (rs / rs0 * 0 + rs) / (rs + 1e-30) * p + r if rs > 0 else r
        # Proper CG update
        p = (rs_new / rs) * p + r if rs > 0 else r
        if rs_new < tol * tol * rs0:

    return x

    for rnd in range(max_rounds):

            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Multi-scale active set: start loose, tighten iteratively

                best_n_active = n_active

            best_n_active = 1
            best_tol_frac = 0.0

    min_beta = max_c * 1e-8

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Run Adam for a batch of steps
        max_steps = min(2000, max(50, int(remaining / 0.05)))
        for _ in range(max_steps):
            if time.monotonic() + min_time_remaining > deadline:
                break
            step += 1
            grad = _smograd(z, beta, n)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-14:
                break

            # Adaptive learning rate
            lr_eff = lr / (1.0 + 0.001 * step)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            z = z - lr_eff * m_hat / (np.sqrt(v_hat) + 1e-8)
            z = z - z.mean()

        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc_new
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
            lr = min(lr * 1.1, 0.2)
        else:
            lr *= 0.7

        # Reduce beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]
        if new_gap < 1e-6 * c_max_new:
            beta *= 0.15
        else:
            beta *= 0.4
        beta = min(beta, c_max_new * 0.0005)

    return current_a


def _coordinate_descent(a_init, deadline, min_time_remaining=30.0):
    """Optimize random subsets of coordinates."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    subset_sizes = [50, 100, 200, 500, 1000]

    while time.monotonic() + min_time_remaining < deadline:
        subset_size = subset_sizes[rng.integers(len(subset_sizes))]
        subset_size = min(subset_size, n)
        indices = rng.choice(n, size=subset_size, replace=False)

        # Optimize the subset using 2D search
        a_sub = a[indices].copy()
        a_sub = a_sub / a_sub.sum()  # normalize subset
        outer_cost = a.sum() - a_sub.sum()

        # Simple local optimization: try random directions in the subset
        base_sc = _score(a)
        improved = False

        for _ in range(50):
            if time.monotonic() + min_time_remaining > deadline:
                break
            direction = rng.standard_normal(subset_size)
            # Center the direction to preserve sum
            direction = direction - direction.mean()
            a_try = a.copy()
            a_try[indices] = a[indices] + 0.01 * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            sc_try = _score(a_try)
            if sc_try < base_sc - 1e-14:
                a = a_try
                base_sc = sc_try
                improved = True
            else:
                # Try smaller step
                a_try2 = a.copy()
                a_try2[indices] = a[indices] + 0.003 * direction
                a_try2 = np.maximum(a_try2, 0)
                s_try2 = a_try2.sum()
                if s_try2 > 1e-15:
                    sc_try2 = _score(a_try2)
                    if sc_try2 < base_sc - 1e-14:
                        a = a_try2
                        base_sc = sc_try2
                        improved = True

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered1":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        for sigma in [0.3, 0.35, 0.4]:
            a = np.exp(-((t - 0.5) ** 2) / (2 * sigma * sigma))
            a = np.maximum(a, 1e-10)
            result = a / a.sum()
            return result
    elif kind == "gaussian_wide":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.35 * 0.35))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "lorentzian":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 20.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 1.5 + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_symmetric(a, sigma_asym, sigma_sym, rng):
    """Add antisymmetric and symmetric perturbations."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    # Antisymmetric noise
    d = rng.standard_normal(n) * sigma_asym
    d = d - d[::-1]  # make antisymmetric: d[i] = -d[n-1-i]
    z = z + 0.5 * d
    # Symmetric noise
    s = rng.standard_normal(n) * sigma_sym
    s = 0.5 * (s + s[::-1])  # make symmetric
    z = z + s
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, max_steps=5000):
    """Subgradient method with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top peaks as active set
        n_peaks = min(10, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:n_peaks]
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        if active.sum() > 50:
            active = np.zeros_like(c, dtype=bool)
            active[top_indices[:50]] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.05)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.008

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick diverse initialization
    print(f"\n[Phase 1] Diverse initialization (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["tapered", "tapered1", "tapered2", "gaussian_wide", "lorentzian",
                "asym_taper", "power_law", "cos"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 800:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': score={sc_p:.10f} (new best profile)")

    # Try antisymmetric perturbations of best profile
    for sigma_a in [0.02, 0.05, 0.1]:
        if time.monotonic() > deadline - 700:
            break
        a_pert = _perturb_symmetric(best_profile_a, sigma_a, 0.02, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_profile_sc:
            best_profile_sc = sc_pert
            best_profile_a = a_pert
            print(f"  Asym pert sigma_a={sigma_a}: score={sc_pert:.10f}")

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 600:
        a_quick = _run_lbfgs_adaptive(best_profile_a, min(deadline - 600, time.monotonic() + 120),
                                       min_time_remaining=400.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_profile")
        if sc_quick < best_profile_sc:
            best_profile_a = a_quick
            best_profile_sc = sc_quick

    # Phase 2: Main L-BFGS optimization
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_profile_a
    a_main = _run_lbfgs_adaptive(start_a, deadline, min_time_remaining=400.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 3: Adam optimization from best
    print(f"\n[Phase 3] Adam optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_adam = _run_adam_optimize(best_seq, deadline, min_time_remaining=250.0)
        sc_adam = _score(a_adam)
        update_best(a_adam, sc_adam, "adam")

    # Phase 4: Newton multi-scale polish
    print(f"\n[Phase 4] Newton multi-scale (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_np, sc_np = _newton_multiscale(best_seq, min(deadline - 200, time.monotonic() + 200),
                                         max_rounds=150)
        update_best(a_np, sc_np, "newton_ms")

    # Phase 5: Coordinate descent
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_cd = _coordinate_descent(best_seq, min(deadline - 150, time.monotonic() + 100),
                                   min_time_remaining=100.0)
        sc_cd = _score(a_cd)
        update_best(a_cd, sc_cd, "coord_descent")

    # Phase 6: Subgradient polish
    print(f"\n[Phase 6] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 100, time.monotonic() + 60))
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.002:
            a_np2, sc_np2 = _newton_multiscale(a_sg, min(deadline - 80, time.monotonic() + 40),
                                               max_rounds=60)
            update_best(a_np2, sc_np2, "subgrad_newton")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 24576, 32768]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        base = best_seq if best_seq is not None else parent
        a_init = _resample(base, n_try)
        a_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 40:
            a_np3, sc_np3 = _newton_multiscale(a_opt, min(deadline - 30, time.monotonic() + 30),
                                               max_rounds=50)
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # Phase 8: Antisymmetric boost + re-optimize
    print(f"\n[Phase 8] Antisymmetric boost (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        for sigma_a in [0.01, 0.03, 0.05]:
            if time.monotonic() > deadline - 20:
                break
            a_boost = _perturb_symmetric(best_seq, sigma_a, 0.01, rng)
            if time.monotonic() < deadline - 15:
                a_boost_opt = _run_lbfgs_adaptive(a_boost, deadline, min_time_remaining=10.0)
                sc_boost = _score(a_boost_opt)
                update_best(a_boost_opt, sc_boost, f"asym_boost_{sigma_a}")

    # Phase 9: Final Newton polish (most important for last bits)
    print(f"\n[Phase 9] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_multiscale(best_seq, deadline - 1, max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-fine L-BFGS
        if best_seq is not None and time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-10
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(3000, max(20, int(remaining / 0.008)))
                try:
                    class ObjFin:
                        def __init__(s, b, nn):
                            s.b = b
                            s.nn = nn
                        def __call__(s, zz):
                            return _smoobj(zz, s.b, s.nn)
                    class GradFin:
                        def __init__(s, b, nn):
                            s.b = b
                            s.nn = nn
                        def __call__(s, zz):
                            return _smograd(zz, s.b, s.nn)
                    result = minimize(
                        ObjFin(beta_fin, n), z, jac=GradFin(beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Ultra-final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_veryfin, sc_veryfin = _newton_multiscale(best_seq, deadline - 0.1, max_rounds=100)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
