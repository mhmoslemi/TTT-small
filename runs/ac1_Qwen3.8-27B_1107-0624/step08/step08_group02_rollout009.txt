    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    """Conjugate gradient for the Hankel system H_W @ x = rhs."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.2:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj_half(z, beta, n_full):
    """Smoothed objective for symmetric parameterization."""
    half = _softmax(z)
    a = _make_symmetric(half)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return n_full * c_max
    return n_full * (c_max + beta * np.log(s))


def _smograd_half(z, beta, n_full):
    """Gradient of smoothed objective w.r.t. z (log-half-parameter)."""
    half = _softmax(z)
    a = _make_symmetric(half)
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # Gradient w.r.t. a: 2n * (w * a[::-1])[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h_full = 2.0 * wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))

    # Gradient w.r.t. a (for the full objective n * smoothed_max):
    # dF/da = 2n * (w * a_rev)[n-1:2n-1] - 2n * W_val * 1
    # Actually the gradient of n*(cmax + beta*log(sum exp(...))) w.r.t. c is n*w
    # Then w.r.t. a: 2 * (n*w * a_rev)[n-1:2n-1] = 2n * (w * a_rev)[n-1:2n-1]
    # But we need to account for the fact that we're using the smoothed version.
    # The gradient of the smoothed max w.r.t. c is w (the softmax weights)
    # So dF/dc = n * w
    # dF/da = 2 * (n*w * a_rev)[n-1:2n-1] = 2n * (w * a_rev)[n-1:2n-1]
    grad_a = 2.0 * n * (wc[n - 1:2 * n - 1] * n / n)  # this is 2n*(w*a_rev)[n-1:2n-1]
    # Wait, let me redo. The objective is n * m_smooth where m_smooth depends on c.
    # d(m_smooth)/dc_k = w_k
    # dF/da_i = sum_k n*w_k * d(c_k)/da_i = 2n * sum_k w_k * a[k-i] = 2n*(w * a_rev)[n-1+i]
    # Hmm, actually: dF/da = 2n * (w * a[::-1])[n-1:2n-1]
    grad_a = 2.0 * n * wc[n - 1:2 * n - 1]

    # But we also need to subtract the sum-constraint effect through softmax.
    # dF/dz_j = sum_i dF/da_i * da_i/dz_j = a[j]*(G[j] - a . G)
    # where G = dF/da (gradient w.r.t. a for the full sequence)
    # For symmetric: dF/dhalf[j] = dF/da[j] + dF/da[n-1-j]
    # = G[j] + G[n-1-j]
    # Then dF/dz[j] = half[j] * (G_sym[j] - half . G_sym)
    # where G_sym = G[:m] + G[:m][::-1] ... no wait.
    # G_sym[j] = G[j] + G[n-1-j]
    
    G = grad_a
    m = len(half)
    G_sym = G[:m] + G[n-1:m-1:-1]  # G[j] + G[n-1-j] for j=0..m-1
    
    # dF/dz_j = half[j] * (G_sym[j] - half . G_sym)
    dot = float(np.dot(half, G_sym))
    grad_z = half * (G_sym - dot)
    return grad_z


class _SmoObjAdapter:
    def __init__(self, beta, n_full):
        self.beta = beta
        self.n_full = n_full

    def __call__(self, z):
        return _smoobj_half(z, self.beta, self.n_full)


class _SmoGradAdapter:
    def __init__(self, beta, n_full):
        self.beta = beta
        self.n_full = n_full

    def __call__(self, z):
        return _smograd_half(z, self.beta, self.n_full)


def _newton_active_polish(a_init, deadline, max_rounds=200):
    """Newton method with active set identification for the full sequence."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find active set with tight tolerance
        active = None
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 3e-4, 1e-3]:
            tol = tol_frac * c_max
            candidate = c >= (c_max - tol)
            if candidate.sum() >= 1 and candidate.sum() <= 3 * n:
                active = candidate
                break
        if active is None or active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True

        W = np.zeros(2 * n - 1)
        n_active = int(active.sum())
        W[active] = 1.0 / n_active

        # Gradient of the "active" max: g = 2*(W * a_rev)[n-1:2n-1]
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)  # centered for the sum constraint
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Newton step: solve H*dx = -g_t + lambda*1
        Hg = _cg_solve(W, n, g, max_iter=600)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))
        
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _run_adaptive_lbfgs_sym(half_init, n_full, deadline, min_time_remaining=60.0):
    """Adaptive L-BFGS in the symmetric parameterization."""
    n = n_full
    half = half_init.copy()
    a = _make_symmetric(half)
    current_sc = _score(a)
    z = np.log(np.maximum(half, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    best_half = half.copy()

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 16384.0)
        est_iter = 0.004 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            half_new = _softmax(result.x)
            a_new = _make_symmetric(half_new)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_sc = sc_new
                half = half_new
                best_half = half_new.copy()
                z = np.log(np.maximum(half_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        a_cur = _make_symmetric(half)
        c_new = _full_convolve(a_cur, a_cur)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-7 * c_max_new and sc_improved > 1e-14:
            beta *= 0.15
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.001)

    return best_half


def _run_lbfgs_full(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS without symmetry constraint (for when symmetry might hurt)."""
    n = len(a_init)
    a = a_init.copy()
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    best_a = a.copy()

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.003 * scale
        max_iter = min(8000, max(30, int(remaining / (est_iter * 25))))

        try:
            # Use the full-sequence gradient
            def obj_fn(z_in):
                a_in = _softmax(z_in)
                c_in = _full_convolve(a_in, a_in)
                cm = float(c_in.max())
                sh = (c_in - cm) / beta
                ev = np.exp(np.clip(sh, -700, 0))
                sv = ev.sum()
                if sv < 1e-300 or not np.isfinite(sv):
                    return 2.0 * n * cm
                return 2.0 * n * (cm + beta * np.log(sv))

            def grad_fn(z_in):
                a_in = _softmax(z_in)
                c_in = _full_convolve(a_in, a_in)
                cm = float(c_in.max())
                sh = (c_in - cm) / beta
                ev = np.exp(np.clip(sh, -700, 0))
                sv = ev.sum()
                if sv < 1e-300 or not np.isfinite(sv):
                    w_in = np.zeros_like(c_in)
                    w_in[int(np.argmax(c_in))] = 1.0
                else:
                    w_in = ev / sv
                a_rev = a_in[::-1]
                wc = _full_convolve(w_in, a_rev)
                h = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w_in, c_in))
                return 4.0 * n * a_in * (h - W_val)

            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        a_cur = best_a
        c_new = _full_convolve(a_cur, a_cur)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.35
        beta = min(beta, c_max_new * 0.001)

    return best_a


def _make_profile_sym(n_full, kind="tapered"):
    """Create a symmetric profile of length n_full, return its first half."""
    n = n_full
    m = n // 2
    t = np.linspace(0, 1, m)
    if kind == "uniform":
        half = np.ones(m)
    elif kind == "tapered":
        half = np.sin(np.pi * t) ** 0.7
    elif kind == "cos":
        half = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "smooth_taper":
        half = np.cos(np.pi * t / 2.0) ** 2.0
    elif kind == "exp_taper":
        half = np.exp(-2.0 * t)
    elif kind == "pow_taper":
        half = (1.0 - t) ** 1.5
    elif kind == "bump":
        half = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
    else:
        half = np.sin(np.pi * t) ** 0.7
    half = np.maximum(half, 1e-12)
    return half / half.sum()


def _resample_half(half_old, n_new_full):
    """Resample a half-sequence to a new length."""
    m_old = len(half_old)
    m_new = n_new_full // 2
    x_old = np.linspace(0, 1, m_old)
    x_new = np.linspace(0, 1, m_new)
    half_new = np.interp(x_new, x_old, half_old)
    half_new = np.maximum(half_new, 0)
    s = half_new.sum()
    if s < 1e-15:
        half_new = np.ones(m_new)
    else:
        half_new = half_new / s
    return half_new


def _perturb_half(half, sigma, rng):
    """Perturb a half-sequence in log space."""
    m = len(half)
    z = np.log(np.maximum(half, 1e-30))
    z = z + sigma * rng.standard_normal(m)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        m = n_parent // 2
        half = _make_profile_sym(n_parent, "tapered")
        parent = _make_symmetric(half)
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Get the half of the parent (symmetrize first)
    m_parent = n_parent // 2
    if n_parent % 2 == 0:
        half_parent = 0.5 * (parent[:m_parent] + parent[m_parent-1::-1])
    else:
        m_parent = (n_parent + 1) // 2
        half_parent = 0.5 * (parent[:m_parent] + parent[m_parent-2::-1])
    half_parent = np.maximum(half_parent, 0)
    half_parent = half_parent / half_parent.sum()

    # Phase 1: Try resampling to longer sequence and quick optimization
    print(f"\n[Phase 1] Exploration (t={time.monotonic()-t_start:.1f}s)")
    
    # Try n = 32768 (symmetric -> 16384 params)
    n_main = 32768
    m_main = n_main // 2
    
    # Resample parent to main size
    half_main = _resample_half(half_parent, n_main)
    a_main = _make_symmetric(half_main)
    sc_main = _score(a_main)
    print(f"  Resampled to n={n_main}: score={sc_main:.10f}")
    update_best(a_main, sc_main, "resampled_32k")

    # Also try a few other profiles at n=32768
    for prof_kind in ["tapered", "cos", "smooth_taper", "pow_taper"]:
        if time.monotonic() > deadline - 800:
            break
        h = _make_profile_sym(n_main, prof_kind)
        a_p = _make_symmetric(h)
        sc_p = _score(a_p)
        print(f"  Profile '{prof_kind}' n={n_main}: score={sc_p:.10f}")
        if sc_p < sc_main:
            sc_main = sc_p
            half_main = h
            update_best(a_p, sc_p, f"profile_{prof_kind}")

    # Quick L-BFGS on the best starting point
    if time.monotonic() < deadline - 700:
        h_quick = _run_adaptive_lbfgs_sym(half_main, n_main, 
                                           min(deadline - 500, time.monotonic() + 200),
                                           min_time_remaining=400)
        a_quick = _make_symmetric(h_quick)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_32k")
        if sc_quick < sc_main:
            half_main = h_quick
            sc_main = sc_quick

    # Phase 2: Main L-BFGS optimization (symmetric, n=32768)
    print(f"\n[Phase 2] Main L-BFGS n={n_main} (t={time.monotonic()-t_start:.1f}s)")
    h_main = _run_adaptive_lbfgs_sym(half_main, n_main, deadline, 
                                       min_time_remaining=400)
    a_main = _make_symmetric(h_main)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_sym_32k")

    # Phase 3: Newton polish on the best
    print(f"\n[Phase 3] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 150, time.monotonic() + 200))
        update_best(a_np, sc_np, "newton_1")

    # Phase 4: Try without symmetry constraint on the best found
    print(f"\n[Phase 4] Full L-BFGS (no symmetry) (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a_full = _run_lbfgs_full(best_seq, min(deadline - 80, time.monotonic() + 180),
                                   min_time_remaining=60)
        update_best(a_full, _score(a_full), "lbfgs_full")

    # Phase 5: Diversification - perturb and re-optimize
    print(f"\n[Phase 5] Diversification (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        for sigma in [0.03, 0.08, 0.15]:
            if time.monotonic() > deadline - 80:
                break
            n_cur = len(best_seq)
            m_cur = n_cur // 2
            if n_cur % 2 == 0:
                h_cur = 0.5 * (best_seq[:m_cur] + best_seq[m_cur-1::-1])
            else:
                m_cur = (n_cur + 1) // 2
                h_cur = 0.5 * (best_seq[:m_cur] + best_seq[m_cur-2::-1])
            h_cur = np.maximum(h_cur, 0)
            h_cur = h_cur / max(h_cur.sum(), 1e-15)
            
            h_pert = _perturb_half(h_cur, sigma, rng)
            a_pert = _make_symmetric(h_pert)
            
            # Quick L-BFGS from perturbed start
            h_opt = _run_adaptive_lbfgs_sym(h_pert, n_cur, 
                                             min(deadline - 50, time.monotonic() + 120),
                                             min_time_remaining=40)
            a_opt = _make_symmetric(h_opt)
            update_best(a_opt, _score(a_opt), f"perturb_{sigma}")

    # Phase 6: Newton + full L-BFGS cycle
    print(f"\n[Phase 6] Final cycles (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        # Newton polish
        a_np2, _ = _newton_active_polish(best_seq, min(deadline - 20, time.monotonic() + 100))
        update_best(a_np2, _score(a_np2), "newton_2")
        
        if best_seq is not None:
            # Short full L-BFGS
            if time.monotonic() < deadline - 15:
                a_f3 = _run_lbfgs_full(best_seq, deadline - 5, min_time_remaining=10)
                update_best(a_f3, _score(a_f3), "lbfgs_final")
            
            # Final Newton
            if best_seq is not None and time.monotonic() < deadline - 3:
                a_fin, _ = _newton_active_polish(best_seq, deadline - 1, max_rounds=100)
                update_best(a_fin, _score(a_fin), "final_newton")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(8192) / 8192
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(8192) / 8192
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
