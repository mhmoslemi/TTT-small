Combines smooth optimization with direct max-minimization for sub-1.5027 scores.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj_and_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad

def _spectral_shave(a, deadline, n_rounds=100):
    """Identify frequency components causing autocorrelation peaks and suppress them."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)
    
    for round_idx in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # The peak at k_star is caused by correlations at certain frequencies.
        # In the frequency domain, c = IDFT(|A(f)|^2).
        # To reduce c[k_star], we want to reduce |A(f)|^2 at frequencies
        # where the phase aligns to constructively interfere at lag k_star.
        
        # Alternative: directly compute which a[i] contribute to c[k_star]
        i_range = np.arange(n)
        j_range = k_star - i_range
        valid = (j_range >= 0) & (j_range < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i_range[valid]] * a[j_range[valid]]
        
        # Reduce top contributors, boost bottom ones
        top_k = max(1, n // 100)
        bottom_k = max(1, n // 50)
        
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bottom_idx = np.argsort(contrib)[:bottom_k]
        
        mass_to_move = 0.0
        for idx in top_idx:
            mass_to_move += a[idx] * 0.05
            a[idx] *= 0.95
        
        if mass_to_move > 1e-15:
            for idx in bottom_idx:
                a[idx] += mass_to_move / len(bottom_idx)
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _refined_peak_shave(a, deadline, n_trials=200):
    """Refined peak shaving with contribution analysis."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # Compute contributions
        i_range = np.arange(n)
        j_range = k_star - i_range
        valid = (j_range >= 0) & (j_range < n)
        contrib = np.zeros(n)
        idx_valid = i_range[valid]
        contrib[idx_valid] = a[idx_valid] * a[j_range[valid]]
        
        total = contrib.sum()
        if total < 1e-15:
            continue
        
        # Select elements to reduce (high contribution) and elements to boost (low contribution)
        # Use a ratio-based approach
        threshold_high = np.percentile(contrib[contrib > 0], 90)
        threshold_low = np.percentile(contrib, 25)
        
        reduce_mask = contrib >= threshold_high
        boost_mask = contrib <= threshold_low & (contrib >= 0)
        # Don't boost the same elements we're reducing
        boost_mask[reduce_mask] = False
        
        if reduce_mask.sum() == 0 or boost_mask.sum() == 0:
            continue
        
        frac = rng.uniform(0.02, 0.08)
        mass_moved = 0.0
        for idx in np.where(reduce_mask)[0]:
            red = a[idx] * frac
            a[idx] -= red
            mass_moved += red
        
        if mass_moved > 1e-15:
            boost_idx = np.where(boost_mask)[0]
            a[boost_idx] += mass_moved / len(boost_idx)
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + np.exp(-20.0 * (t - 0.85) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        # Generate a smooth random profile
        n_coarse = max(8, n // 1024)
        coarse = np.random.default_rng(42).uniform(0.5, 1.5, n_coarse)
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), coarse)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

The solution is `a[i] = max(0, v[i] - tau)` where `tau` is chosen so that `sum a = 1`.

`tau` can be found by bisection or by sorting.

Let me implement this:

```python
def _project_simplex(v):
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find the largest rho such that sum_{i<=rho} (u[i] - 1/rho) > 0
    cumsum = np.cumsum(u)
    rho = np.where(u > (cumsum - 1) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1] + 1
    tau = (cumsum[rho-1] - 1) / rho
    return np.maximum(v - tau, 0)
```

"""
Multi-phase optimizer for minimizing step-function autocorrelation peak.
Combines adaptive LSE, proximal subgradient with simplex projection,
spectral-aware peak shaving, multi-resolution search, and perturbation restarts.
Targets score below 1.5027.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj_and_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj_and_grad(z, self.beta, self.n)[0]

    def grad(self, z):
        return _lse_obj_and_grad(z, self.beta, self.n)[1]


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    rho_candidates = np.where(u > (cumsum - 1.0) / np.arange(1, n + 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1] + 1
    tau = (cumsum[rho - 1] - 1.0) / rho
    return np.maximum(v - tau, 0.0)

def _proximal_subgradient(a_init, deadline, lr_init=0.02, momentum=0.85):
    """Proximal subgradient method with momentum and simplex projection."""
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    velocity = np.zeros(n)
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(4242)

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        k_star = int(np.argmax(c))

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(lr_init * 1.5, lr * 1.03)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.75
            if no_improve > 400:
                a = _project_simplex(best_a + 0.03 * rng.standard_normal(n))
                velocity = np.zeros(n)
                lr = lr_init
                no_improve = 0

        j = np.arange(n)
        idx = k_star - j
        valid = (idx >= 0) & (idx < n)
        g = np.zeros(n)
        g[valid] = 2.0 * a[idx[valid]]
        g_centered = g - g.mean()
        g_norm = np.linalg.norm(g_centered)

        if g_norm < 1e-18:
            break

        velocity = momentum * velocity - lr * g_centered / g_norm
        a_temp = a + velocity
        a_new = _project_simplex(a_temp)

        sc_new = _score(a_new)
        if sc_new < current_sc + 0.01:
            a = a_new
        else:
            lr *= 0.5
            velocity *= 0.5

    return best_a, best_sc


def _peak_shave_v2(a_init, deadline, n_trials=200):
    """Refined peak shaving with percentile-based contribution selection."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(7777)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        j = np.arange(n)
        idx = k_star - j
        valid = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        iv = j[valid]
        contrib[iv] = a[iv] * a[idx[valid]]

        total = contrib.sum()
        if total < 1e-15:
            continue

        p_high = np.percentile(contrib[contrib > 0], 88)
        p_low = np.percentile(contrib, 20)

        reduce_mask = contrib >= p_high
        boost_mask = (contrib <= p_low) & (~reduce_mask)

        if reduce_mask.sum() == 0 or boost_mask.sum() == 0:
            continue

        frac = rng.uniform(0.015, 0.06)
        mass = 0.0
        ridx = np.where(reduce_mask)[0]
        for i in ridx:
            mass += a[i] * frac
            a[i] *= (1.0 - frac)

        if mass > 1e-15:
            bidx = np.where(boost_mask)[0]
            a[bidx] += mass / len(bidx)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = _project_simplex(best_a)
            continue
        a = a / s

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.2) ** 2) + np.exp(-15.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "random_smooth":
        n_coarse = max(8, n // 512)
        coarse = np.random.default_rng(42).uniform(0.3, 1.7, n_coarse)
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), coarse)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.arange(n) / n
        a = 1.0 + 0.2 * np.sin(2.0 * np.pi * (5.0 * t + 2.0 * t * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
