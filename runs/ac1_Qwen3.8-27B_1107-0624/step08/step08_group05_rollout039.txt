def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_softmax_grad_correction(a, grad_a):
    return a * (grad_a - np.dot(a, grad_a))

# LSE smoothing

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return _log_softmax_grad_correction(a, grad_a)

def _power_sum_obj(z, p, n):

    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return _log_softmax_grad_correction(a, grad_a)

def _active_set_weights(c, tol_frac):
    c_max = float(c.max())
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[0] = 1.0
        return w, c_max
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w, c_max


def _active_gradient(a, c, n, tol_frac=0.002):
    w, c_max = _active_set_weights(c, tol_frac)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _nesterov_subgrad(a_init, deadline, n_steps=20000, lr_init=0.01, tol_frac=0.001, momentum=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    y = z.copy()  # Nesterov lookahead
    v = np.zeros_like(z)  # momentum
    mu = momentum
    f_min = best_sc

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(y)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = _log_softmax_grad_correction(a, grad_a)

        # Adaptive learning rate (Polyak-like)
        f_cur = float(c.max()) * (2.0 * n)
        gap = f_cur - f_min
        if gap > 0:
            polyak_lr = gap / max(float(np.dot(grad_z, grad_z)), 1e-30) * 0.5
        else:
            polyak_lr = lr
        step = min(lr, polyak_lr) if i > 100 else lr

        # Nesterov update
        v = mu * v + step * grad_z
        y = z - v
        y = np.clip(y, -60.0, 60.0)
        y = y - y.mean()

        # Update base point periodically
        if i % 50 == 0:
            z = y.copy()

        # Evaluate and update best
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                f_min = sc
                patience = 0
            else:
                patience += 1
                lr *= 0.95
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    y = z.copy()
                    v = np.zeros_like(z)
                    patience = 0
                    lr = lr_init * 0.5
            mu = min(0.999, momentum + 0.001 * i)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))

            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.12 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.2) ** 2) / 0.015) + np.exp(-((t - 0.8) ** 2) / 0.015)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "randn_clipped":
        rng = np.random.default_rng(12345)
        a = rng.standard_normal(n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _coarse_to_fine(a_coarse, n_fine, deadline):
    """Optimize at coarse resolution, then upscale and refine at fine resolution."""
    n_coarse = len(a_coarse)
    n_fine_target = n_fine

    # Optimize at coarse
    a_c, sc_c = _lbfgs_lse_adaptive(a_coarse, deadline, min_remaining=30, n_stages=8)

    # Upscale
    a_up = _resize_sequence(a_c, n_fine_target)

    # Refine at fine
    a_f, sc_f = _lbfgs_lse_adaptive(a_up, deadline, min_remaining=15, n_stages=10)
    return a_f, sc_f


# ============================================================
# Pairwise coordinate refinement
# ============================================================

def _pairwise_refine(a, deadline, n_rounds=10):
    """Try pairwise swaps to reduce the peak autocorrelation."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Find the peak lag(s)
    peak_lags = np.where(np.abs(c - c_max) < 0.01 * c_max)[0]

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Try swapping pairs around the peak lag
        for k in peak_lags[:3]:
            # The peak is at lag k, so b[k] = sum_i a[i]*a[k-i]
            # Try small adjustments
            for _ in range(20):
                i = int(np.random.randint(0, n))
                j = int(np.random.randint(0, n))
                if i == j or abs(i - j) < n // 10:
                    continue
                # Try small transfer from i to j
                delta = 0.01 * min(a[i], a[j])
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                s = a_try.sum()
                a_try = a_try / s
                c_try = _full_convolve(a_try, a_try)
                sc_try = float(c_try.max())
                if sc_try < c_max - 1e-12:
                    a = a_try
                    c = c_try
                    c_max = sc_try
                    peak_lags = np.where(np.abs(c - c_max) < 0.01 * c_max)[0]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining adaptive LSE L-BFGS, Nesterov subgradient,
    trust-region Newton, multi-resolution, and diverse restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (45s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 45)
    n_expl = max(1024, n_parent // 4) if n_parent > 2048 else 2048
    kinds = ["uniform", "tapered", "cos2", "wheatstone", "gauss", "bimodal", "ramp", "randn_clipped"]
    for kind in kinds:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # Quick L-BFGS on best exploration result
    if _best_seq is not None and time.monotonic() < deadline - 600:
        a_up = _resize_sequence(_best_seq, n_parent) if len(_best_seq) != n_parent else _best_seq.copy()
        a_opt, sc_opt = _lbfgs_lse_adaptive(a_up, deadline, min_remaining=500, n_stages=6)
        upd(a_opt, sc_opt, "expl_lbfgs")

    # === Phase 2: Multi-resolution warm start (80s) ===
    print(f"\n[Phase 2] Multi-resolution")
    if time.monotonic() < deadline - 500:
        n_coarse = max(512, n_parent // 8)
        a_coarse = _resize_sequence(_best_seq, n_coarse) if len(_best_seq) != n_coarse else _best_seq[:n_coarse].copy()
        a_coarse = a_coarse / a_coarse.sum()
        a_fine, sc_fine = _coarse_to_fine(a_coarse, n_parent, deadline)
        upd(a_fine, sc_fine, "coarse_to_fine")

    # === Phase 3: Main L-BFGS with adaptive LSE (400s) ===
    print(f"\n[Phase 3] Main L-BFGS LSE (adaptive)")
    if time.monotonic() < deadline - 400:
        a_main, sc_main = _lbfgs_lse_adaptive(_best_seq.copy(), deadline, min_remaining=350, n_stages=18)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 4: Power-sum L-BFGS (150s) ===
    print(f"\n[Phase 4] Power-sum L-BFGS")
    if time.monotonic() < deadline - 150:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=130)
        upd(a_ps, sc_ps, "lbfgs_power_sum")

    # === Phase 5: Newton active-set (120s) ===
    print(f"\n[Phase 5] Newton active-set")
    if time.monotonic() < deadline - 120:
        n_dead = min(deadline - 100, time.monotonic() + 120)
        a_nw, sc_nw = _newton_active_set(_best_seq, n_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 6: Nesterov subgradient (100s) ===
    print(f"\n[Phase 6] Nesterov subgradient")
    if time.monotonic() < deadline - 100:
        sg_dead = min(deadline - 80, time.monotonic() + 100)
        a_sg, sc_sg = _nesterov_subgrad(_best_seq, sg_dead, n_steps=30000, lr_init=0.008, tol_frac=0.001)
        upd(a_sg, sc_sg, "nesterov_sg")

    # === Phase 7: Newton + Nesterov alternating (100s) ===
    print(f"\n[Phase 7] Newton + Nesterov alternating")
    if time.monotonic() < deadline - 100:
        a_cur = _best_seq.copy()
        for round_i in range(4):
            if time.monotonic() > deadline - 80:
                break
            nw_dead = min(deadline - 80, time.monotonic() + 25)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.002)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")
            sg_dead2 = min(deadline - 80, time.monotonic() + 20)
            a_sg2, sc_sg2 = _nesterov_subgrad(a_cur, sg_dead2, n_steps=15000, lr_init=0.005, tol_frac=0.001)
            if sc_sg2 < _score(a_cur):
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_nesterov_{round_i}")

    # === Phase 8: Multi-n exploration (200s) ===
    print(f"\n[Phase 8] Multi-n exploration")
    n_candidates = [2048, 4096, 8192, 16384, 24576, 32768]
    if n_parent not in n_candidates and n_parent >= 1024:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))
    # Sort by proximity to current best n first
    n_candidates.sort(key=lambda x: abs(x - len(_best_seq)))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        a_opt_n, sc_opt_n = _lbfgs_lse_adaptive(a_init, deadline, min_remaining=50, n_stages=12)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        if _score(a_opt_n) < _best_score * 1.005 and time.monotonic() < deadline - 40:
            nw_dead = min(deadline - 30, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=100, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 9: Diverse restarts (150s) ===
    print(f"\n[Phase 9] Diverse restarts")
    if time.monotonic() < deadline - 50:
        restart_strategies = [
            ("log_perturb_03", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("fft_phase", lambda: _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)),
            ("block_swap", lambda: _perturb_block_swap(_best_seq, rng, n_swaps=4)),
            ("amp_perturb", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.08)),
            ("spectral_flat", lambda: _perturb_spectral_flat(_best_seq, rng)),
            ("mix", lambda: _perturb_mix(_best_seq, rng)),
            ("log_perturb_05", lambda: _perturb_log(_best_seq, 0.05, rng)),
        ]
        for name, gen_fn in restart_strategies:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            a_r, sc_r = _lbfgs_lse_adaptive(p_start, deadline, min_remaining=20, n_stages=8)
            upd(a_r, sc_r, f"restart_{name}")

            if sc_r < _best_score * 1.004 and time.monotonic() < deadline - 15:
                nw_dead = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=60, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{name}_newton")

    # === Phase 10: Pairwise coordinate refinement (30s) ===
    print(f"\n[Phase 10] Pairwise refinement")
    if time.monotonic() < deadline - 15:
        pw_dead = min(deadline - 10, time.monotonic() + 30)
        a_pw, sc_pw = _pairwise_refine(_best_seq, pw_dead, n_rounds=5)
        upd(a_pw, sc_pw, "pairwise")

    # === Phase 11: Final aggressive polish (30s) ===
    print(f"\n[Phase 11] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=80, tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final Nesterov
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _nesterov_subgrad(_best_seq, deadline - 1, n_steps=20000, lr_init=0.004, tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_nesterov")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
