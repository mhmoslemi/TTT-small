```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_matvec(h, v):
    """Compute H @ v where H[i,j] = h[i+j] for i,j in range(len(v)).
    h has length 2*len(v)-1."""
    n = len(v)
    # We need (Hv)_i = sum_j h[i+j] * v[j]
    # This is cross-correlation of h and v
    # Using FFT: IFFT(FFT(h) * conj(FFT(v)))
    N = _next_pow2(len(h) + n - 1)
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(v, N)
    result = np.fft.irfft(H * np.conj(V), N)[:n]
    return result

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
