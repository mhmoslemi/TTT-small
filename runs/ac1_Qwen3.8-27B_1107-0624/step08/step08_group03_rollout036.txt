"""
Successive Linear Programming (SLP) + L-BFGS + Active-set Newton optimizer
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
1. SLP with adaptive active set: Linearizes the peak constraints and solves
   LPs to take large, well-informed steps that gradient methods cannot match.
2. Spectral pre-conditioning via modified Gerchberg-Saxton iterations.
3. Combined pipeline: SLP → L-BFGS (log-sum-exp) → Newton → restarts.
4. Diversity through multiple initializations and perturbation restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _solve_slp_lp(a_star, b_cur, active_indices, n):
    """
    Solve the SLP linearized LP:
        min t
        s.t. 2 * (Q_k a*|^T) a <= t + b_k(a*)  for k in active_indices
             sum(a) = 1
             a >= 0
    Returns the LP solution a_new (normalized, non-negative) or None.
    """
    m = len(active_indices)
    if m == 0:
        return None

    # Build constraint matrix efficiently
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx, k in enumerate(active_indices):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # A_ub[idx, i] = 2 * a_star[k-i] for i in [i_min, i_max]
        segment = a_star[k - i_min:k - i_max + 1][::-1]
        A_ub[idx, i_min:i_max + 1] = 2.0 * segment
        A_ub[idx, n] = -1.0
        b_ub[idx] = b_cur[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            a_new = res.x[:n].copy()
            a_new = np.maximum(a_new, 0.0)
            s = float(a_new.sum())
            if s < 1e-15:
                return None
            return a_new / s
        return None
    except Exception:
        return None


def _slp_optimize(a_init, deadline, max_iters=300, tol_frac=0.005, max_active=300):
    """
    Successive Linear Programming optimizer.
    Repeatedly linearizes peak constraints at current point and solves LP.
    Uses line search to ensure descent on the true objective.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    cur_sc = best_sc

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = _full_convolve(a, a)
        b_max = float(b.max())

        # Determine active set: peaks within tol_frac of max
        threshold = b_max * (1.0 - tol_frac)
        active_mask = b >= threshold
        n_active = int(active_mask.sum())

        # Cap active set size
        if n_active > max_active:
            top_idx = np.argsort(b)[::-1][:max_active]
            active_mask = np.zeros_like(b, dtype=bool)
            active_mask[top_idx] = True
            n_active = max_active
        elif n_active == 0:
            active_mask = np.zeros_like(b, dtype=bool)
            active_mask[int(np.argmax(b))] = True
            n_active = 1

        active_indices = np.where(active_mask)[0]

        # Solve LP
        a_lp = _solve_slp_lp(a, b, active_indices, n)
        if a_lp is None:
            break

        # Line search along direction from a to a_lp
        direction = a_lp - a
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break

        f_old = b_max
        accepted = False
        for alpha in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            a_try = a + alpha * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            b_try = _full_convolve(a_try, a_try)
            f_try = float(b_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                accepted = True
                break

        if not accepted:
            # Try a small step in the LP direction
            for alpha in [0.02, 0.01, 0.005]:
                a_try = a + alpha * direction
                a_try = np.maximum(a_try, 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                b_try = _full_convolve(a_try, a_try)
                f_try = float(b_try.max())
                if f_try < f_old - 1e-14:
                    a = a_try
                    accepted = True
                    break

        if not accepted:
            # No improvement; try relaxing the tolerance
            tol_frac *= 2.0
            if tol_frac > 0.1:
                break
            continue

        # Update best
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        b_new = _full_convolve(a, a)
        if abs(float(b_new.max()) - f_old) < 1e-13 * f_old:
            break

        # Adaptively tighten tolerance
        tol_frac = max(0.001, tol_frac * 0.95)

    return best_a, best_sc


def _gs_precondition(a_init, n_iters=30, deadline=None):
    """
    Modified Gerchberg-Saxton spectral pre-conditioning.
    Iterates between flattening the spectrum and enforcing non-negativity.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)
    target_mag = 1.0 / np.sqrt(N)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Soft flattening: blend between current and target magnitude
        alpha = 0.3 + 0.7 * (it / max(n_iters, 1))
        new_mag = mag * (1.0 - alpha) + target_mag * alpha

        A_new = new_mag * np.exp(1j * phase)
        a_real = np.fft.ifft(A_new).real[:n]
        a_real = np.maximum(a_real, 0.0)
        s_new = float(a_real.sum())
        if s_new < 1e-15:
            break
        a_new = a_real / s_new

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()

        # Relaxation to avoid cycling
        a = 0.7 * a + 0.3 * a_new

    return best_a


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with log-sum-exp smoothing (beta continuation)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                      0.015, 0.0075, 0.0037, 0.00185, 0.0009,
                                      0.00045, 0.00022, 0.00011, 0.00005,
                                      0.000025, 0.00001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.006
        elif n <= 16384:
            est_time_per_iter = 0.015
        else:
            est_time_per_iter = 0.03 * (n / 32768.0)

        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    """Active-set Newton method for the peak minimization problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            n_active = 2 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha_cg = rs / pAp
                x += alpha_cg * p
                r_new = r - alpha_cg * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * t * gnorm or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-12 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-12:
            break

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "beta_family":
        t = np.linspace(0, 1, n)
        a = (t + 0.01) ** 2 * (1 - t + 0.01)
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.005):
    """Subgradient method in log-space with top-k peak targeting."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        k = min(30, max(5, len(c) // 50))
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c)
        w[top_k] = 1.0 / k

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining SLP, L-BFGS, Active-set Newton,
    and spectral pre-conditioning to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score_val:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: GS pre-conditioning to find a different basin
    print(f"\n[Phase 1] GS pre-conditioning (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 1200:
        gs_deadline = min(deadline - 1200, t_start + 60)
        a_gs = _gs_precondition(parent, n_iters=40, deadline=gs_deadline)
        sc_gs = _score(a_gs)
        print(f"  GS score: {sc_gs:.10f}")
        update_best(a_gs, sc_gs, "gs")

    # Phase 2: SLP from parent
    print(f"\n[Phase 2] SLP from parent (t={time.monotonic() - t_start:.1f}s)")
    slp_deadline = min(deadline - 600, t_start + 600)
    a_slp, sc_slp = _slp_optimize(parent, slp_deadline, max_iters=200, tol_frac=0.005, max_active=300)
    update_best(a_slp, sc_slp, "slp_parent")

    # Phase 3: SLP from GS result
    print(f"\n[Phase 3] SLP from GS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        slp_deadline2 = min(deadline - 400, t_start + 900)
        a_slp2, sc_slp2 = _slp_optimize(a_gs, slp_deadline2, max_iters=200, tol_frac=0.005, max_active=300)
        update_best(a_slp2, sc_slp2, "slp_gs")

    # Phase 4: L-BFGS polish on best SLP result
    print(f"\n[Phase 4] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        lbfgs_deadline = min(deadline - 200, t_start + 1400)
        a_lb = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=100.0, aggressive=True)
        update_best(a_lb, _score(a_lb), "lbfgs_polish")

    # Phase 5: Active-set Newton
    print(f"\n[Phase 5] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        newton_deadline = min(deadline - 40, t_start + 1550)
        a_as, sc_as = _active_set_newton(best_seq, newton_deadline, max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "newton")

        if time.monotonic() < deadline - 20:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 10, t_start + 1580), max_iter=100, tol_active=0.0003)
            update_best(a_as2, sc_as2, "newton_tight")

    # Phase 6: SLP with tighter tolerance on current best
    print(f"\n[Phase 6] SLP tight tolerance (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        slp_deadline3 = min(deadline - 15, t_start + 1600)
        a_slp3, sc_slp3 = _slp_optimize(best_seq, slp_deadline3, max_iters=100, tol_frac=0.002, max_active=200)
        update_best(a_slp3, sc_slp3, "slp_tight")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        restart_configs = [
            ("pert_005", 0.05),
            ("pert_010", 0.10),
            ("pert_020", 0.20),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, sigma, rng)
            # Quick SLP on perturbed
            a_r, sc_r = _slp_optimize(p_start, min(deadline - 5, time.monotonic() + 60),
                                       max_iters=50, tol_frac=0.01, max_active=200)
            update_best(a_r, _score(a_r), f"restart_{label}")

            if _score(a_r) < best_score_val + 0.003 and time.monotonic() < deadline - 5:
                a_r2 = _run_lbfgs(a_r, min(deadline - 2, time.monotonic() + 30), min_time_remaining=2.0)
                update_best(a_r2, _score(a_r2), f"restart_{label}_lbfgs")

    # Phase 8: Subgradient polish
    print(f"\n[Phase 8] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _subgrad_polish(best_seq, min(deadline - 2, t_start + 1750),
                                         n_max_steps=2000, lr_init=0.003)
        update_best(a_fin, sc_fin, "sg_polish")

    # Phase 9: Final active-set Newton with very tight tolerance
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin2, sc_fin2 = _active_set_newton(best_seq, deadline - 1, max_iter=80, tol_active=0.0001)
        update_best(a_fin2, sc_fin2, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
