"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses KKT-based multiplicative updates, adaptive active-set optimization,
spectral analysis, and L-BFGS with log-sum-exp smoothing.
Key innovation: Direct KKT condition satisfaction via multiplicative weight updates
that drive the convolution gradient toward uniformity at active peaks.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * t)) / 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _kkt_multiplicative_update(a, active_frac, eta, min_active=5):
    """
    KKT-based multiplicative update.
    Drives the gradient (convolution with uniform weight on active peaks)
    toward uniformity by multiplicatively adjusting a.
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Active set: top fraction of peaks
    n_active = max(min_active, int(n * active_frac))
    threshold = np.sort(c)[::-1][min(n_active - 1, len(c) - 1)]
    active = c >= threshold
    
    n_act = int(active.sum())
    if n_act == 0:
        return a, c_max
    
    # Weight vector
    w = np.zeros_like(c)
    w[active] = 1.0 / n_act
    
    # Gradient: g[i] = sum_k w[k] * a[k-i] (with bounds)
    # This is the convolution of w (reversed) with a, taking the middle part
    w_rev = w[::-1]
    wc = _full_convolve(w_rev, a)
    g = wc[n - 1:2 * n - 1]  # shape (n,)
    
    # Target: g should be constant for all i where a[i] > 0
    # Use the mean of g where a > 0 as the target
    pos = a > 1e-15
    if pos.sum() < 2:
        return a, c_max
    
    target = np.mean(g[pos])
    if target < 1e-30:
        return a, c_max
    
    # Multiplicative update: a[i] *= (target / g[i])^eta
    ratio = target / np.maximum(g, 1e-30)
    ratio = np.clip(ratio, 0.01, 100.0)
    a_new = a * (ratio ** eta)
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new)
    
    return a_new, c_max

            no_improve += 1

            tol_frac = min(0.2, tol_frac * 1.5)

            no_improve = 0
            stall_count += 1

                break

        n_active = max(5, int(n * active_frac))

        w_rev = w[::-1]
        wc = _full_convolve(w_rev, a)
        g = wc[n - 1:2 * n - 1]

        pos = a > 1e-12
        if pos.sum() < 2:
            break
        target = np.mean(g[pos])
        if target < 1e-30:
            break
        
        # Multiplicative update
        ratio = target / np.maximum(g, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_new = a * (ratio ** eta)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

            a = a_new

                c_half = _full_convolve(a_half, a_half)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.002)

    betas = base_beta * np.array([1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.2, 0.15,
                                   0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005,
                                   0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                   0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max_iter based on remaining time and dimension
        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _active_set_gradient_descent(a_init, deadline, min_time_remaining=10.0):
    """
    Active-set gradient descent with adaptive learning rate.
    Uses the subgradient of max(c) with respect to a.
    """
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = 0.05
    active_tol = 0.01  # fraction of max to include in active set
    no_improve = 0
    max_stall = 200
    stall_count = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Slightly tighten active set
            active_tol = max(0.001, active_tol * 0.98)
        else:
            no_improve += 1
        
        if no_improve >= max_stall:
            a = best_a.copy()
            active_tol = min(0.05, active_tol * 1.5)
            lr = 0.05
            no_improve = 0
            stall_count += 1
            if stall_count > 15:
                break
        
        # Active set
        threshold = c_max * (1 - active_tol)
        active = c >= threshold
        if active.sum() < 3:
            top_idx = np.argsort(c)[::-1][:max(3, int(n * 0.003))]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
        
        n_act = int(active.sum())
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        
        # Gradient: d max(c)/da ≈ (w * a_rev)[n-1:2n-1]
        w_rev = w[::-1]
        wc = _full_convolve(w_rev, a)
        g = wc[n - 1:2 * n - 1]
        
        # Center the gradient (to maintain sum constraint)
        g_centered = g - np.mean(g)
        g_norm = np.linalg.norm(g_centered)
        
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        alpha = min(lr, 0.5 / (g_norm + 1e-15))
        
        # Line search
        improved = False
        for ls in range(50):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.8
            if lr < 1e-8:
                lr = 0.05
                no_improve += 50  # force stall detection
    
    return best_a, _score(best_a)

def _spectral_flatten(a, deadline, min_time_remaining=10.0):
    """
    Try to flatten the power spectrum in frequency domain.
    """
    n = len(a)
    a = a.copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    pad_n = _next_pow2(2 * n - 1)
    
    for step in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        A = np.fft.rfft(a, pad_n)
        power = np.abs(A) ** 2
        
        # Try to flatten: target is the mean power
        mean_power = np.mean(power)
        
        # Adjust: reduce power at peaks
        # In time domain, this corresponds to reducing autocorrelation peaks
        # Use a soft threshold on the power spectrum
        alpha = 0.1
        target = mean_power * (1 + 0.5 * alpha)
        power_clipped = np.minimum(power, target)
        
        # New A from clipped power (maintain phase)
        if np.any(power > 0):
            phase = A / np.maximum(np.abs(A), 1e-30)
            A_new = np.sqrt(power_clipped) * phase
            a_new = np.fft.irfft(A_new, pad_n)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new
        
        # Decay alpha
        alpha *= 0.95
    
    return best_a, _score(best_a)

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective dimension."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.001, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03, 0.05, 0.07, 0.1]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    
    return best_a, best_sc

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. KKT-based multiplicative weight updates for direct peak equalization
2. Adaptive active-set selection with automatic expansion/contraction
3. Proximal gradient steps as alternative to multiplicative updates
4. Multi-resolution warmup (small n → large n)
5. Perturbation-based restart with diversity
6. Tighter L-BFGS convergence with adaptive beta schedule
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * t)) / 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _compute_gradient(a, w):
    """
    Compute g[j] = sum_k w[k] * a[k-j] (with bounds).
    This is the subgradient of the weighted convolution.
    Uses: g = (a_rev * w)[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    aw = _full_convolve(a_rev, w)
    g = aw[n - 1:2 * n - 1]
    return g

def _kkt_multiplicative(a, eta, active_tol, min_active):
    """
    One step of KKT multiplicative update.
    Returns (a_new, c_max, n_active).
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Active set: peaks within active_tol of max
    threshold = c_max * (1.0 - active_tol)
    active = c >= threshold
    n_act = int(active.sum())
    
    # Ensure minimum active set size
    if n_act < min_active:
        top_idx = np.argsort(c)[::-1][:min_active]
        active = np.zeros_like(c, dtype=bool)
        active[top_idx] = True
        n_act = min_active
    
    # Weight vector
    w = np.zeros_like(c)
    w[active] = 1.0 / n_act
    
    # Compute gradient
    g = _compute_gradient(a, w)
    
    # Target: mean gradient on support
    pos = a > 1e-15
    if pos.sum() < 2:
        return a, c_max, n_act
    target = np.mean(g[pos])
    if target < 1e-30:
        return a, c_max, n_act
    
    # Multiplicative update
    ratio = target / np.maximum(g, 1e-15)
    ratio = np.clip(ratio, 0.001, 1000.0)
    a_new = a * (ratio ** eta)
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new)
    
    return a_new, c_max, n_act

def _kkt_optimize(a_init, deadline, min_time_remaining=10.0,
                  eta_start=0.3, active_tol_start=0.1,
                  active_tol_end=0.001, min_active_init=20):
    """
    KKT-based multiplicative optimization.
    """
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    eta = eta_start
    active_tol = active_tol_start
    min_active = min_active_init
    no_improve = 0
    stall_count = 0
    max_stall = 200
    
    while time.monotonic() + min_time_remaining < deadline:
        a_new, c_max, n_act = _kkt_multiplicative(a, eta, active_tol, min_active)
        
        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        sc_old = 2.0 * n * c_max
        
        if sc_new < best_sc - 1e-15:
            best_a = a_new.copy()
            best_sc = sc_new
            a = a_new
            no_improve = 0
            # Tighten
            active_tol = max(active_tol_end, active_tol * 0.995)
            min_active = max(5, int(min_active * 0.999))
        elif sc_new < sc_old - 1e-15:
            # Good step but not new best
            a = a_new
            no_improve = 0
        else:
            # Bad step - reduce eta
            a = a_new  # still take the step to avoid getting stuck
            no_improve += 1
            eta = max(0.01, eta * 0.8)
        
        if no_improve >= max_stall:
            a = best_a.copy()
            eta = eta_start
            active_tol = min(0.3, active_tol * 1.5)
            no_improve = 0
            stall_count += 1
            if stall_count > 20:
                break
    
    return best_a, _score(best_a)

def _proximal_gradient(a_init, deadline, min_time_remaining=10.0, lr_init=0.05):
    """
    Proximal gradient method for the KKT conditions.
    Uses projected gradient: a_new = max(0, a - lr * (g - mean_g))
    """
    n = len(a_init)
    a = a_init.copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = lr_init
    active_tol = 0.05
    no_improve = 0
    stall_count = 0
    max_stall = 300
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            active_tol = max(0.001, active_tol * 0.99)
        else:
            no_improve += 1
        
        if no_improve >= max_stall:
            a = best_a.copy()
            lr = lr_init
            active_tol = min(0.2, active_tol * 1.5)
            no_improve = 0
            stall_count += 1
            if stall_count > 20:
                break
        
        # Active set
        threshold = c_max * (1.0 - active_tol)
        active = c >= threshold
        if active.sum() < 5:
            top_idx = np.argsort(c)[::-1][:max(5, int(n * 0.003))]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
        n_act = int(active.sum())
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        
        # Gradient
        g = _compute_gradient(a, w)
        g_centered = g - np.mean(g)
        g_norm = np.linalg.norm(g_centered)
        
        if g_norm < 1e-15:
            break
        
        # Adaptive step size (Armijo-like)
        alpha = min(lr, 1.0 / (g_norm * n_act + 1e-15))
        
        improved = False
        for _ in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.85
            if lr < 1e-9:
                lr = lr_init * 0.3
    
    return best_a, _score(best_a)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted
