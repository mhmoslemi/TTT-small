1. Identify elements to reduce (high gradient contribution to active peaks)
2. Identify elements to increase (low "potential" = low max_k a[k-j])

"""
Optimized step-function search using adaptive active-set minimax with impact-aware
peak redistribution, multi-temperature log-sum-exp annealing, structured
perturbation restarts, and multi-n exploration.
Key innovations: adaptive active-set thresholding, spectral-aware redistribution,
structured non-symmetric perturbations, and aggressive LSE cooling schedules.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remaining=15.0):
    """L-BFGS with aggressive multi-temperature annealing."""
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, _score(current_a)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # Longer, more gradual cooling schedule
    betas = base_beta * np.array(
        [1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02,
         0.012, 0.007, 0.004, 0.002, 0.0012, 0.0007, 0.0004,
         0.0002, 0.00012, 0.00007, 0.00004, 0.00002, 0.00001]
    )

    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10.0:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        # More iterations at lower temperatures (they matter more)
        temp_factor = 1.0 + 0.5 * (i / len(betas))
        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 12) / temp_factor)))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Try restarting from slightly perturbed position
                z_pert = z + 0.01 * np.random.default_rng(int(time.monotonic() * 1000)).standard_normal(n)
                z_pert = z_pert - z_pert.mean()
                result2 = minimize(
                    prob, z_pert, jac=prob.grad, method='L-BFGS-B',
                    options={'maxiter': max_iter // 3, 'ftol': 1e-18,
                             'gtol': 1e-14, 'maxcor': 80, 'maxls': 40}
                )
                a_new2 = _softmax(result2.x)
                a_new2 = np.maximum(a_new2, 0.0)
                s_new2 = a_new2.sum()
                if s_new2 > 1e-15:
                    a_new2 = a_new2 / s_new2
                    sc_new2 = _score(a_new2)
                    if sc_new2 < current_sc - 1e-16:
                        current_a = a_new2
                        current_sc = sc_new2
                        z = np.log(np.maximum(a_new2, 1e-30))
                        z = z - z.mean()
        except Exception:

def _active_set_optimize(a_init, deadline, tol_start=0.03, tol_end=0.0005, lr_start=0.02):
    """Adaptive active-set gradient descent for minimax autocorrelation."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall_count = 0

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set
        threshold = c_max - tol * c_max
        active = c >= threshold
        n_active = int(active.sum())

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Cap for efficiency
        max_active = max(20, min(n_active, 8 * n))
        if n_active > max_active:
            top_idx = np.argsort(c)[::-1][:max_active]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = max_active

        # Weighted gradient
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Center to preserve sum
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Backtracking line search
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if no_improve % 15 == 0:
                tol = max(tol_end, tol * 0.75)
                lr *= 0.65
            if no_improve >= 150:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.4
                no_improve = 0
                stall_count += 1
                if stall_count > 15:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.003)

    return best_a, best_sc


def _peak_shave(a_init, deadline, n_trials=300):
    """Targeted peak shaving with adaptive reduction and smart redistribution."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)

        if k_lo >= k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()

        if total < 1e-15:
            continue

        # Adaptive fraction: reduce more when peak is dominant
        dominance = contrib.max() / total
        frac = rng.uniform(0.02, 0.05 + 0.05 * dominance)
        n_reduce = max(1, int(len(contrib) * frac))
        top_contrib_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_contrib_idx:
            red = a_try[idx] * rng.uniform(0.03, 0.1)
            a_try[idx] -= red
            reduction += red

        if reduction < 1e-12:
            continue

        # Smart redistribution: prefer low-value elements outside the peak window
        low_mask = np.ones(n, dtype=bool)
        for idx in top_contrib_idx:
            low_mask[idx] = False
        for jv in j_range:
            if 0 <= jv < n:
                low_mask[jv] = False

        if low_mask.sum() < 3:
            low_mask = np.ones(n, dtype=bool)
            for idx in top_contrib_idx:
                low_mask[idx] = False
            if low_mask.sum() < 3:
                continue

        # Weight by inverse current value (add more to smaller elements)
        targets = a_try[low_mask]
        weights = 1.0 / (targets + 0.01 * targets.mean() + 1e-10)
        weights = weights / weights.sum()
        a_try[low_mask] += reduction * weights

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _structured_perturb(a, kind, rng):

    n = len(a)
    t = np.linspace(0, 1, n)

    if kind == "ramp":
        weight = 0.5 + 0.5 * t
        a_new = a * weight
    elif kind == "ramp_inv":
        weight = 0.5 + 0.5 * (1 - t)
        a_new = a * weight
    elif kind == "quadratic":
        weight = 1.0 - 0.5 * (2 * t - 1) ** 2
        a_new = a * weight
    elif kind == "quadratic_inv":
        weight = 0.5 + 0.5 * (2 * t - 1) ** 2
        a_new = a * weight
    elif kind == "split_shift":
        mid = n // 2
        shift = int(rng.integers(1, max(2, n // 20)))
        a_new = a.copy()
        a_new[mid:] = a[mid - shift:mid] if shift < mid else np.zeros(n - mid)
        a_new[:mid] = a[:max(0, mid - shift)] + a[(mid + shift):]
    elif kind == "cos_mod":
        weight = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 11 * t)
        a_new = a * weight
    elif kind == "edge_boost":
        edge = np.minimum(t, 1 - t)
        weight = 1.0 + 0.5 * np.exp(-50 * edge ** 2)
        a_new = a * weight
    elif kind == "center_dip":
        weight = 1.0 - 0.3 * np.exp(-5 * (t - 0.5) ** 2)
        a_new = a * weight
    else:
        a_new = a * (1.0 + 0.05 * rng.standard_normal(n))

    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "lopsided":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Active-set on parent
    P2: L-BFGS LSE annealing
    P3: Peak shaving
    P4: Second L-BFGS pass
    P5: Multi-n exploration (16384, 24576)
    P6: Structured restarts
    P7: Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent: uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Active-set on parent ===
    print(f"\n[P1] Active-set (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200.0, t_start + 300.0)
    a1, sc1 = _active_set_optimize(best_a, p1_deadline, tol_start=0.025, tol_end=0.0005, lr_start=0.02)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [P1 active-set] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (sc={sc1:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS LSE ===
    print(f"\n[P2] L-BFGS LSE (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 800.0, t_start + 650.0)
    a2, sc2 = _run_lbfgs(_normalize(best_a), p2_deadline, min_remaining=20.0)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [P2 lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (sc={sc2:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak shaving ===
    print(f"\n[P3] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 500.0, t_start + 850.0)
    a3, sc3 = _peak_shave(best_a, p3_deadline, n_trials=400)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [P3 shave] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (sc={sc3:.10f}) t={time.monotonic()-t_start:.1f}s")

    # Re-optimize after peak shaving
    if time.monotonic() < deadline - 400.0:
        p3b_deadline = min(deadline - 400.0, t_start + 1000.0)
        a3b, sc3b = _active_set_optimize(best_a, p3b_deadline, tol_start=0.01, tol_end=0.0002, lr_start=0.015)
        if sc3b < best_score:
            best_a = a3b.copy()
            best_score = sc3b
            print(f"  NEW BEST: {best_score:.10f} [P3b re-opt] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second L-BFGS pass ===
    print(f"\n[P4] L-BFGS LSE round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300.0:
        p4_deadline = min(deadline - 300.0, t_start + 1200.0)
        a4, sc4 = _run_lbfgs(_normalize(best_a), p4_deadline, min_remaining=20.0)
        if sc4 < best_score:
            best_a = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4 lbfgs2] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (sc={sc4:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_targets = [16384, 24576, 8192]
    # Avoid duplicating current n
    n_targets = [nt for nt in n_targets if nt != n_best]

    for n_try in n_targets:
        if time.monotonic() > deadline - 100.0:
            break
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        remaining = deadline - time.monotonic() - 80.0
        if remaining < 60:
            continue

        # Active set first (fast), then L-BFGS
        half_time = remaining * 0.4
        a_opt, sc_opt = _active_set_optimize(a_init, t_start + time.monotonic() - t_start + half_time,
                                              tol_start=0.02, tol_end=0.001, lr_start=0.015)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            n_best = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P5 active] t={time.monotonic()-t_start:.1f}s")

        remaining2 = deadline - time.monotonic() - 60.0
        if remaining2 > 40:
            a_opt2, sc_opt2 = _run_lbfgs(_normalize(a_opt), deadline - 50.0, min_remaining=10.0)
            if sc_opt2 < best_score:
                best_a = a_opt2.copy()
                best_score = sc_opt2
                n_best = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P5 lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Structured restarts ===
    print(f"\n[P6] Structured restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60.0:
        restart_kinds = ["ramp", "ramp_inv", "quadratic", "lope-sided", "edge_boost", "center_dip"]
        # Fix typo
        restart_kinds = ["ramp", "ramp_inv", "quadratic", "lopsided", "edge_boost", "center_dip"]

        for kind in restart_kinds:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 20:
                break

            a_r = _structured_perturb(best_a, kind, rng)
            # Quick active-set on perturbed
            a_opt, sc_opt = _active_set_optimize(a_r, deadline - 15.0,
                                                  tol_start=0.015, tol_end=0.001, lr_start=0.012)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [P6 {kind}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_optimize(best_a, deadline - 2.0,
                                              tol_start=0.002, tol_end=0.0001,
                                              lr_start=0.008, )
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P7 polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny peak shave
        if time.monotonic() < deadline - 2.0:
            a_ps, sc_ps = _peak_shave(best_a, deadline - 1.0, n_trials=30)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  NEW BEST: {best_score:.10f} [P7 shave] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
