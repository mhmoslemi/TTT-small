def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _autocorr_grad(a, k):
    """Gradient of c[k] = sum_i a[i]*a[k-i] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            g[i] += a[j]
    return g

def _cg_solve(matvec, rhs, max_cg=500, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)

        p = r_new + (rs_new / rs) * p
        x = x_new
        r = r_new
        rs = rs_new

def _next_pow2(n):

def _full_convolve(a, b):
    n, m = len(a), len(b)

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

        r = r_new
        rs = rs_new
