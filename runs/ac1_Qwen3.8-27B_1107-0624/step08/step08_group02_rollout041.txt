def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
Multi-scale optimizer with coordinate descent and active-set Newton for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses coarse-to-fine strategy, smooth L-BFGS adaptation,
and 1D coordinate optimization for final polishing.
"""
import numpy as np
import time
import sys
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _resample_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _grad_from_weights(a, w, n):
    """Compute 2*(w convolved with reversed a), taking n elements from position n-1."""
    result = _full_convolve(a[::-1], w)
    return 2.0 * result[n - 1:2 * n - 1]


def _hankel_mv(v, w, n):
    """Hankel matrix-vector product: H[i,j] = 2*w[i+j]. Computed as 2*correlate(w, v)."""
    result = _full_convolve(v[::-1], w)
    return 2.0 * result[n - 1:2 * n - 1]


def _cg_solve_hankel(w, n, rhs, max_iter=300):
    """CG solver for Hankel system: H x = rhs where H[i,j] = 2*w[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_mv(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < 1e-24 * rs0:
            break
    return x


class _SmoothFunc:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def obj(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -500, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -500, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        h = _grad_from_weights(a, w, self.n)
        return h * a


def _lbfgs_smooth(a_init, deadline, min_remaining=10.0, max_rounds=50):
    """Adaptive-temperature smooth L-BFGS optimization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init.copy()

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    beta = max(gap * 0.25, max_c * 0.003)
    min_beta = max_c * 1e-12

    for rnd in range(max_rounds):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 8192.0)
        est_time = 0.003 * scale
        max_iter = min(10000, max(20, int(remaining / (est_time * 30))))

        sf = _SmoothFunc(beta, n)
        try:
            result = minimize(
                sf.obj, z, jac=sf.grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                         'maxcor': 60, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-17:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(10, len(c_sorted_new) - 1)]

        if new_gap < 1e-12 * c_max_new:
            beta = min(beta * 0.05, c_max_new * 1e-6)
        elif new_gap < 1e-9 * c_max_new:
            beta = min(beta * 0.1, c_max_new * 1e-5)
        elif new_gap < 1e-6 * c_max_new:
            beta = min(beta * 0.25, c_max_new * 5e-5)
        elif new_gap < 1e-3 * c_max_new:
            beta = min(beta * 0.5, c_max_new * 5e-4)
        else:
            beta = min(beta * 0.7, c_max_new * 0.005)

        beta = min(beta, c_max_new * 0.002)
        if beta < min_beta:
            beta = min_beta

    return current_a


def _newton_active(a_init, deadline, max_rounds=200, min_remaining=5.0):
    """Active-set Newton method with CG for the linear system."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() + min_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Determine active set with adaptive tolerance
        active = None
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 2 and n_active <= max(10, n // 100):
                break
            if n_active >= 1 and tol_frac >= 1e-3:
                break
        if active is None or not active.any():
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        n_active = int(active.sum())
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        # Gradient
        g = _grad_from_weights(a, w, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Hessian solve: H @ x = g_t (where H[i,j] = 2*w[i+j])
        rhs = g_t.copy()
        Hg = _cg_solve_hankel(w, n, rhs, max_iter=500)
        H1 = _cg_solve_hankel(w, n, np.ones(n), max_iter=500)

        sum_Hg = float(np.ones(n) @ Hg)
        sum_H1 = float(np.ones(n) @ H1)
        if abs(sum_H1) < 1e-20 or not np.isfinite(sum_H1):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = sum_Hg / sum_H1
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_max if False else c_try.max())
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _coord_descent(a_init, deadline, max_sweeps=50, min_remaining=5.0):
    """Coordinate descent in the original space (a >= 0, sum = 1)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)

    for sweep in range(max_sweps):
        if time.monotonic() + min_remaining > deadline:
            break
        any_change = False
        for j in range(n):
            if time.monotonic() + min_remaining > deadline:
                break
            # Find optimal a_j
            # Affected lags: k in [j, j+n-1] intersect [0, 2n-2]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            # Precompute coefficients
            # For k in [k_lo, k_hi], k != 2j: conv_new[k] = c[k] + 2*(t - a[j]) * a[k-j]
            # For k = 2j: conv_new[2j] = c[2j] + t^2 - a[j]^2
            # For k outside [k_lo, k_hi]: unchanged

            a_j = a[j]
            if a_j < 1e-15:
                # Try setting to small positive value
                t_trial = 0.01 / n
                delta = t_trial
                affected = np.arange(k_lo, k_hi + 1)
                if len(affected) == 0:
                    continue
                k_idx = affected - j  # k - j for each affected k
                k_idx = k_idx[(k_idx >= 0) & (k_idx < n)]
                new_vals = c[affected].copy()
                mask = (affected != 2 * j)
                new_vals[mask] += 2.0 * delta * a[k_idx[mask]]
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    new_vals[affected == 2 * j] = c[2 * j] + t_trial ** 2
                new_max = float(new_vals.max())
                unaff_max = float(np.delete(c, affected).max()) if len(affected) < len(c) else 0.0
                total_max = max(new_max, unaff_max)
                if total_max < c.max() - 1e-15:
                    a[j] = t_trial
                    a[j + 1:] = a[j + 1:].copy()  # no-op, just to trigger recomputation
                    c = _full_convolve(a, a)
                    s = a.sum()
                    a = a / s
                    c = _full_convolve(a, a)
                    any_change = True
                continue

            # Golden section search for optimal t
            # The function is roughly unimodal near the current value
            margin = max(0.1 * a_j, 0.01 / n)
            lo = max(0.0, a_j - 3.0 * margin)
            hi = a_j + 3.0 * margin
            if hi <= lo:
                hi = lo + margin

            best_t = a_j
            best_val = float(c.max())

            # Evaluate at a few points
            for t_candidate in [0.0, lo, (lo + hi) / 2, a_j, hi]:
                t_candidate = max(0.0, t_candidate)
                delta = t_candidate - a_j
                affected = np.arange(k_lo, k_hi + 1)
                if len(affected) == 0:
                    continue
                k_idx = affected - j
                k_idx_valid = (k_idx >= 0) & (k_idx < n)
                new_vals = c[affected].copy()
                mask = (affected != 2 * j) & k_idx_valid
                new_vals[mask] += 2.0 * delta * a[k_idx[mask]]
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    idx_2j = np.where(affected == 2 * j)[0]
                    if len(idx_2j) > 0:
                        new_vals[idx_2j] = c[2 * j] + t_candidate ** 2 - a_j ** 2
                new_max = float(new_vals.max())
                # Max of unaffected parts
                if len(affected) < len(c):
                    unaff = c.copy()
                    unaff[affected] = -np.inf
                    unaff_max = float(unaff.max())
                else:
                    unaff_max = 0.0
                total_max = max(new_max, unaff_max)
                if total_max < best_val:
                    best_val = total_max
                    best_t = t_candidate

            # Refine with a few more points around best_t
            if best_t != a_j:
                for frac in [0.5, 0.7, 0.3, 1.3, 0.8, 1.2]:
                    t_cand = a_j + frac * (best_t - a_j)
                    if t_cand < 0:
                        t_cand = 0.0
                    delta = t_cand - a_j
                    affected = np.arange(k_lo, k_hi + 1)
                    k_idx = affected - j
                    k_idx_valid = (k_idx >= 0) & (k_idx < n)
                    new_vals = c[affected].copy()
                    mask = (affected != 2 * j) & k_idx_valid
                    new_vals[mask] += 2.0 * delta * a[k_idx[mask]]
                    if 2 * j >= k_lo and 2 * j <= k_hi:
                        idx_2j = np.where(affected == 2 * j)[0]
                        if len(idx_2j) > 0:
                            new_vals[idx_2j] = c[2 * j] + t_cand ** 2 - a_j ** 2
                    new_max = float(new_vals.max())
                    if len(affected) < len(c):
                        unaff = c.copy()
                        unaff[affected] = -np.inf
                        unaff_max = float(unaff.max())
                    else:
                        unaff_max = 0.0
                    total_max = max(new_max, unaff_max)
                    if total_max < best_val:
                        best_val = total_max
                        best_t = t_cand

            if best_t != a_j and best_val < c.max() - 1e-16:
                a[j] = best_t
                c = _full_convolve(a, a)
                any_change = True

        if not any_change:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.5 + 0.01
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "peaked_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization driver using multi-scale strategy."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}", flush=True)
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}", flush=True)
        _update_best(parent, _score(parent), "init")

    # Phase 1: Try different profiles at smaller scale to find good shape
    print(f"\n[Phase 1] Multi-scale shape finding (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_small = max(64, n_parent // 16)
    if n_small < 64:
        n_small = 64

    best_small_a = _resample_to_n(parent, n_small)
    best_small_sc = _score(best_small_a)

    profiles = ["tapered", "asym_taper", "left_heavy", "bimodal", "peaked_center"]
    for prof_kind in profiles:
        if time.monotonic() > deadline - 600:
            break
        a_p = _make_profile(n_small, prof_kind)
        sc_p = _score(a_p)
        print(f"  Profile '{prof_kind}' (n={n_small}): score={sc_p:.10f}", flush=True)
        if sc_p < best_small_sc:
            best_small_sc = sc_p
            best_small_a = a_p

    # Also try perturbed parent at small scale
    for sigma in [0.1, 0.3]:
        if time.monotonic() > deadline - 550:
            break
        a_pert = _perturb(_resample_to_n(parent, n_small), sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_small_sc:
            best_small_sc = sc_pert
            best_small_a = a_pert

    # Quick optimize at small scale
    if time.monotonic() < deadline - 500:
        a_small_opt = _lbfgs_smooth(best_small_a, min(deadline - 500, time.monotonic() + 60),
                                     min_remaining=400.0, max_rounds=15)
        sc_small = _score(a_small_opt)
        _update_best(a_small_opt, sc_small, "small_scale")
        if sc_small < best_small_sc:
            best_small_a = a_small_opt
            best_small_sc = sc_small

    # Multi-scale up
    print(f"  Multi-scale up from n={n_small} to n={n_parent}", flush=True)
    n_cur = n_small
    a_cur = best_small_a.copy()
    target_n = n_parent

    while n_cur < target_n:
        n_next = min(n_cur * 2, target_n)
        a_cur = _resample_to_n(a_cur, n_next)
        if time.monotonic() < deadline - 200:
            a_cur = _lbfgs_smooth(a_cur, min(deadline - 200, time.monotonic() + 40),
                                   min_remaining=150.0, max_rounds=8)
        sc_cur = _score(a_cur)
        _update_best(a_cur, sc_cur, f"multiscale_n{n_next}")
        n_cur = n_next

    # Phase 2: Main optimization at full n
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic()-t_start:.1f}s)", flush=True)
    start_a = best_seq if best_seq is not None else a_cur
    if len(start_a) != n_parent:
        start_a = _resample_to_n(start_a, n_parent)
    a_main = _lbfgs_smooth(start_a, deadline, min_remaining=400.0, max_rounds=40)
    _update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 3: Newton polish
    print(f"\n[Phase 3] Newton polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_np, sc_np = _newton_active(best_seq, deadline, max_rounds=100, min_remaining=250.0)
        _update_best(a_np, sc_np, "newton_1")

    # Phase 4: Coordinate descent
    print(f"\n[Phase 4] Coordinate descent (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_cd = _coord_descent(best_seq, deadline, max_sweeps=30, min_remaining=150.0)
        _update_best(a_cd, _score(a_cd), "coord_desc")

    # Phase 5: Another round of Newton + CD
    print(f"\n[Phase 5] Newton + CD round 2 (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_np2, _ = _newton_active(best_seq, deadline, max_rounds=80, min_remaining=120.0)
        _update_best(a_np2, _score(a_np2), "newton_2")
        if time.monotonic() < deadline - 80:
            a_cd2 = _coord_descent(best_seq, deadline, max_sweeps=20, min_remaining=60.0)
            _update_best(a_cd2, _score(a_cd2), "coord_desc_2")

    # Phase 6: Try from different starting points
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_opts = [
            (_perturb(best_seq, 0.03, rng), "perturb_3pct"),
            (_perturb(best_seq, 0.08, rng), "perturb_8pct"),
            (_make_profile(n_parent, "asym_taper"), "profile_asym"),
        ]
        for a_r, label in restart_opts:
            if time.monotonic() > deadline - 30:
                break
            a_r_opt = _lbfgs_smooth(a_r, deadline, min_remaining=20.0, max_rounds=10)
            _update_best(a_r_opt, _score(a_r_opt), f"restart_{label}")

    # Phase 7: Final intense Newton
    print(f"\n[Phase 7] Final Newton (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active(best_seq, deadline, max_rounds=200, min_remaining=2.0)
        _update_best(a_fin, sc_fin, "final_newton")

        # Final coordinate descent
        if time.monotonic() < deadline - 2:
            a_cd_fin = _coord_descent(best_seq, deadline, max_sweeps=10, min_remaining=1.0)
            _update_best(a_cd_fin, _score(a_cd_fin), "final_cd")

    # Final validation
    print(f"\n{'='*60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
