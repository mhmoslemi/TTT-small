"""
Improved active-set QP optimizer with preconditioned CG, 
progressive tolerance tightening, non-negativity active set,
and multi-phase optimization strategy for minimizing the 
step-function convolution score.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,j] = 2*W[i+j], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _row_sums_hankel(W, n):
    """Compute row sums of the Hankel matrix H[i,j] = 2*W[i+j]."""
    # Row i sum = 2 * sum_{j=0}^{n-1} W[i+j]
    # Use prefix sum for efficiency
    prefix = np.zeros(len(W) + 1)
    prefix[1:] = np.cumsum(W)
    row_sums = 2.0 * (prefix[n:n + n] - prefix[:n])
    return row_sums


def _pcg_solve(W, n, rhs, max_iter=5000, tol=1e-15):
    """Preconditioned CG to solve H*x = rhs.
    H is the Hankel matrix with H[i,j] = 2*W[i+j].
    Uses diagonal preconditioner based on row sums."""
    x = np.zeros(n)
    r = rhs.copy()
    
    # Preconditioner: diagonal = row sums (clipped to avoid zeros)
    D = _row_sums_hankel(W, n)
    D_min = np.min(D[D > 0]) if np.any(D > 0) else 1.0
    D = np.maximum(D, D_min * 1e-6)
    z = r / D
    
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = r / D
        rz_new = float(r @ r_new)
        if not np.isfinite(rz_new) or abs(rz_new) < 1e-30:
            break
        beta = rz_new / rz
        p = r_new + beta * p
        rz = rz_new
        if abs(rz) < tol * tol * abs(rz0):
            break
        # Periodic restart for numerical stability
        if i > 0 and i % 1000 == 0:
            p = r_new.copy()
    
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

    base_beta = max(gap * 0.3, max_c * 0.002)

                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                           0.000001, 0.0000005, 0.0000002, 0.0000001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                           0.000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break

        if remaining < 3:

        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))

Improved active-set QP optimizer with preconditioned CG,
progressive tolerance tightening, non-negativity active set handling,
and refined multi-phase strategy for minimizing step-function convolution score.
Key improvements: (1) preconditioned CG with row-sum diagonal, (2) proper
non-negativity active set within QP, (3) tighter tolerance schedule,
(4) warm-started CG across tolerance levels, (5) adaptive time allocation.

_smo_beta = 0.0
_smo_n = 0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,j] = 2*W[i+j], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _row_sums_hankel(W, n):
    """Compute row sums of Hankel matrix H[i,j] = 2*W[i+j]."""
    prefix = np.zeros(len(W) + 1)
    prefix[1:] = np.cumsum(W)
    row_sums = 2.0 * (prefix[n:n + n] - prefix[:n])
    return row_sums


def _pcg_solve(W, n, rhs, x_init, max_iter=8000, tol=1e-16, support=None):
    """Preconditioned CG to solve (H restricted to support) * x = rhs.
    H[i,j] = 2*W[i+j]. Uses row-sum diagonal preconditioner."""
    if support is None:
        support = np.ones(n, dtype=bool)
    n_sup = int(support.sum())
    if n_sup == 0:
        return np.zeros(n)

    x = np.zeros(n)
    x[support] = x_init[support] if x_init is not None else 0.0

    # Compute residual
    r = rhs[support].copy()
    Hp_full = _hankel_matvec(x, W, n)
    r = r - Hp_full[support]

    # Preconditioner: row sums restricted to support
    D_full = _row_sums_hankel(W, n)
    D = D_full[support].copy()
    pos = D > 1e-20
    if pos.any():
        D_min = np.min(D[pos])
    else:
        D_min = 1.0
    D = np.maximum(D, D_min * 1e-6)

    z = r / D
    p = z.copy()
    rz = float(r @ z)
    rz0 = abs(rz)
    if rz0 < 1e-30:
        x[support] = x_init[support] if x_init is not None else 0.0
        return x

    for i in range(max_iter):
        Hp_full = _hankel_matvec(p, W, n)
        Hp = Hp_full[support]
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        if not np.isfinite(alpha):
            break
        x[support] = x[support] + alpha * p
        r = r - alpha * Hp
        z_new = r / D
        rz_new = float(r @ z_new)
        if not np.isfinite(rz_new) or abs(rz_new) < 1e-30:
            break
        if rz_new > rz * 1.5:
            # Numerical breakdown, restart
            p = z_new.copy()
            rz = rz_new
            continue
        beta_cg = rz_new / rz
        if not np.isfinite(beta_cg):
            p = z_new.copy()
            rz = rz_new
            continue
        p = z_new + beta_cg * p
        rz = rz_new
        if abs(rz) < tol * tol * rz0:
            break
        # Periodic restart for stability
        if (i + 1) % 2000 == 0:
            p = z_new.copy()

    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

def _active_set_qp(W, n, a_init, max_iters=50, cg_max=5000, tol=1e-15):
    """Solve min a^T H_W a s.t. sum(a)=1, a>=0 using active-set method.
    W is the weight vector for active convolution lags (length 2n-1)."""
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    support = a > 1e-12
    n_sup = int(support.sum())
    if n_sup == 0:
        support[:] = True
        n_sup = n

    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n) / 2.0)

    for outer in range(max_iters):
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            n_sup = n

        # Solve H_SS y = 1_S using PCG with warm start
        rhs_full = np.zeros(n)
        rhs_full[support] = 1.0
        x_init_full = np.zeros(n)
        x_init_full[support] = a[support]

        y = _pcg_solve(W, n, rhs_full, x_init_full, max_iter=cg_max,
                       tol=tol, support=support)

        oney = float(np.sum(y[support]))
        if abs(oney) < 1e-20 or not np.isfinite(oney):
            break

        a_new = np.zeros(n)
        a_new[support] = y[support] / oney
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        val_new = float(a_new @ _hankel_matvec(a_new, W, n) / 2.0)
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-18:
            best_val = val_new
            best_a = a_new.copy()

        # Check KKT conditions for non-negativity
        grad_full = _hankel_matvec(a_new, W, n)
        lam = float(np.sum(grad_full[support])) / n_sup if n_sup > 0 else 0.0

        # Find violated non-negativity: a_i = 0 but grad_i < lam
        violated = (~support) & (grad_full < lam - 1e-10 * (np.max(np.abs(grad_full)) + 1e-15))
        if violated.any():
            # Add to support
            support[violated] = True
            a_new[violated] = 0.0
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            a = a_new
            continue

        # Remove slack variables from support
        slack = support & (a_new < 1e-12)
        if slack.any():
            support[slack] = False
            a = a_new
            continue

        # Check convergence
        resid = np.max(np.abs(grad_full[support] - lam))
        off_resid = 0.0
        if (~support).any():
            off_resid = np.max(lam - grad_full[~support])
        total_resid = max(resid, off_resid)
        scale = np.max(np.abs(grad_full)) + 1e-15
        if total_resid < 1e-12 * scale:
            a = a_new
            break

        a = a_new

    return best_a


def _active_set_polish(a_init, deadline, max_rounds=200, tol_schedule=None):

    n = len(a_init)

                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_sc = sc_try
                    best_a = a_try.copy()
                    break
                alpha *= 0.5
                if alpha < 1e-14:
                    break

        # Advance tolerance
        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=300):
    """Direct convolution flattening."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_avg = float(np.sum(c)) / len(c)
        c_max = float(c.max())
        excess = np.maximum(c - c_avg, 0)
        if excess.max() < 1e-15:
            break

        a_rev = a[::-1]
        ec = _full_convolve(excess, a_rev)
        grad_part = ec[n - 1:2 * n - 1]

        step = 0.05 / (1.0 + 0.005 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a + 0.3 * (a_new - a)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS beta continuation,
    preconditioned-CG active-set QP with progressive tightening,
    non-negativity active set, flattening polish, and multi-n exploration.
    """
    t_start = time.monotonic()
    total_budget = max(1.0, float(budget_s) - 15.0)
    deadline = t_start + total_budget
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS beta continuation
    t_phase1 = time.monotonic()
    t_limit_1 = t_start + total_budget * 0.25
    print(f"\n[Phase 1] L-BFGS (budget {t_limit_1 - t_start:.0f}s)")
    a1 = _run_lbfgs(parent, t_limit_1, min_time_remaining=60.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs")
    print(f"  L-BFGS done, score={_score(a1):.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Active-set QP with PCG (main phase)
    t_phase2 = time.monotonic()
    t_limit_2 = t_start + total_budget * 0.70
    print(f"\n[Phase 2] Active-set QP PCG (budget {t_limit_2 - t_start:.0f}s)")
    if best_seq is not None:
        a2, sc2 = _active_set_polish(best_seq, t_limit_2, max_rounds=300)
        update_best(a2, sc2, "active_qp")
        print(f"  Active-set QP done, score={sc2:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 3: Flatten polish
    t_limit_3 = t_start + total_budget * 0.80
    print(f"\n[Phase 3] Flatten polish")
    if best_seq is not None and time.monotonic() < t_limit_3:
        a3, sc3 = _flatten_polish(best_seq, t_limit_3, max_iters=200)
        update_best(a3, sc3, "flatten")

    # Phase 4: Second round of tighter active-set QP
    t_limit_4 = t_start + total_budget * 0.90
    print(f"\n[Phase 4] Tighter active-set QP round 2")
    if best_seq is not None and time.monotonic() < t_limit_4:
        tight_schedule = [0.00001, 0.000005, 0.000002, 0.000001,
                          0.0000005, 0.0000002, 0.0000001,
                          0.00000005, 0.00000002, 0.00000001,
                          0.000000005, 0.000000002, 0.000000001]
        a4, sc4 = _active_set_polish(best_seq, t_limit_4,
                                     max_rounds=100, tol_schedule=tight_schedule)
        update_best(a4, sc4, "active_qp_tight")

    # Phase 5: Multi-n exploration
    t_limit_5 = t_start + total_budget * 0.95
    print(f"\n[Phase 5] Multi-n exploration")
    n_candidates = [32768, 10000, 25000]
    for n_try in n_candidates:
        if time.monotonic() > t_limit_5:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        sc_init = _score(a_init)
        # Quick L-BFGS
        a_opt = _run_lbfgs(a_init, t_limit_5, min_time_remaining=20.0, aggressive=False)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < t_limit_5 - 10:
            a_q, sc_q = _active_set_polish(a_opt, t_limit_5 - 5, max_rounds=50)
            update_best(a_q, sc_q, f"n_{n_try}_qp")

    # Phase 6: Final ultra-tight polish
    t_final = deadline
    print(f"\n[Phase 6] Final polish")
    if best_seq is not None and time.monotonic() < t_final - 5:
        ultra_schedule = [0.000000001, 0.0000000005, 0.0000000002,
                          0.0000000001, 0.00000000005]
        a_fin, sc_fin = _active_set_polish(best_seq, t_final - 3,
                                           max_rounds=50, tol_schedule=ultra_schedule)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < t_final - 3:
            a_fl, sc_fl = _flatten_polish(best_seq, t_final - 2, max_iters=50)
            update_best(a_fl, sc_fl, "final_flatten")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
