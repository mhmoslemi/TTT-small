import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import scipy.sparse as sp

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_iteration(f, M, max_iter=100):
    """One step of the LP-based iteration.
    Find g maximizing sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    """
    n = len(f)
    # Build constraint matrix A where A[k, m] = f[k-m] if 0 <= k-m < n
    # A is (2n-1) x n
    # Use sparse format
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for m in range(lo, hi + 1):
            rows.append(k)
            cols.append(m)
            vals.append(f[k - m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

    """Nesterov accelerated subgradient method."""
    n = len(a_init)
    y = a_init.copy()
    y = np.maximum(0, y)
    y /= y.sum()
    x = y.copy()
    best_a = y.copy()
    best_score = _score(y)
    
    t = 1
    gamma = 2.0 / (t + 1.5)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        
        # Find active set
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]
        # Only include lags that are close to max
        active = [k for k in active if c[k] >= c_max * 0.999]
        if not active:
            active = [int(np.argmax(c))]
        
        # Compute subgradient (average of gradients of active lags)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            # gradient of c[k] w.r.t. a[m]: a[k-m] + a[m] for valid m
            for m in range(lo, hi + 1):
                g[m] += a[k - m] + a[m]
        g /= len(active)
        
        # Project gradient onto sum=0 subspace (tangent to the constraint sum(a)=1)
        g = g - g.mean()
        
        # Step size
        lr = lr0 / (1.0 + 0.001 * it)
        step = lr * g
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:

        s = x_new.sum()
        if s < 1e-15:

        y_new = x_new + ((t - 1) / t_new) * (x_new - x)

        else:

        x = x_new
        y = y_new
        
        # Track best
        sc = _score(x)
        if sc < best_score:
            best_score = sc
            best_a = x.copy()
    
    return best_a, best_score

Let me think about the active-set Newton more carefully.

Given a set of active lags $S = \{k_1, ..., k_m\}$, we want to find $a$ such that:
- $c[k_i] = c[k_j]$ for all $i, j$ (equalized)
- $\sum a_j = 1$
- $a_j \geq 0$
- The gradient of $\max c$ in the feasible directions is zero (KKT)

This is a system of $m + 1$ equations (m-1 equalization + 1 normalization) in $n$ unknowns. We can solve it using a Newton method on the active constraints.

Define $F(a) = [c[k_1] - c[k_m], c[k_2] - c[k_m], ..., c[k_{m-1}] - c[k_m], \sum a_j - 1]^T$

We want $F(a) = 0$. The Jacobian is:
$J[i, j] = \frac{\partial c[k_i]}{\partial a_j} - \frac{\partial c[k_m]}{\partial a_j}$ for $i < m$
$J[m, j] = 1$ for all $j$

Each $\frac{\partial c[k]}{\partial a_j} = a[k-j] + a[j]$ (if valid).

The Jacobian is $m \times n$ (with $m \ll n$). We can solve the underdetermined system $J \delta = -F$ for the minimum-norm $\delta$ using:
$\delta = -J^T (J J^T)^{-1} F$

Then update $a \leftarrow a + \alpha \delta$ (with line search).

This is essentially a Gauss-Newton method for the active constraints.

So $J$ has a "difference of shifts" structure, which can be exploited for fast matrix-vector products.

The matrix-vector product $Jv$ (where $v \in \mathbb{R}^n$) can be computed as:
$(Jv)[i] = \sum_j (a[k_i - j] - a[k_m - j]) v[j]$ (for valid $j$)

$= (a * v)[k_i] - (a * v)[k_m]$ (approximately, with boundary effects)

This can be computed with a single FFT of $v$ and pointwise multiplication with the FFT of $a$!

4. Compute $\delta = -J^T \alpha$ (using FFT-based matvec)
5. Line search: find $\alpha$ (step size) such that the objective decreases

This is very efficient! Each Newton iteration costs O(m * n log n) for the FFT operations, plus O(m^3) for the small matrix solve.

"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _lp_iteration_step(f, n_max_constraints=None):
    """One LP iteration: find g maximizing sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    M = float(_fft_convolve(f, f).max())
    if M < 1e-15:
        return f.copy()
    
    # Build constraint matrix A (2n-1) x n
    # A[k, m] = f[k-m] if 0 <= k-m < n, else 0
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            rows.extend(range(k, k + 1))  # placeholder
            # Actually let's build it more efficiently
            rows.extend([k] * (hi - lo + 1))
            cols.extend(indices.tolist())
            vals.extend(f[k - indices].tolist())
    
    # Remove first batch of placeholder rows
    # Actually let me rebuild properly
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            for idx in indices:
                rows.append(k)
                cols.append(int(idx))
                vals.append(float(f[k - idx]))
    
    if not rows:
        return f.copy()
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(0, res.x)
        g_sum = g.sum()
        if g_sum > 1e-15:
            return g / g_sum * f.sum()
    return f.copy()

    """Iterative LP-based optimization."""
    n = len(a_init)

        return a_init, _score(a_init)

            break
        it += 1
        
        t0 = time.monotonic()

        lp_time = time.monotonic() - t0

            break
        
        g = g / g.sum()
        
        # Try several step sizes
        improved = False

            a_new = np.maximum(0, a_new)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new /= s_new
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            if lp_time > 30:
                # LP too slow, stop
                break
            # Try smaller step
            t = 0.01
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(0, a_new)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new /= s_new
                sc = _score(a_new)
                if sc < best_score:

                    best_a = a_new.copy()
                    a = a_new
                else:
                    break

def _nesterov_subgradient(a_init, deadline, max_iter=500000, top_k=5, 
                           lr_start=0.01, lr_end=0.001):
    """Nesterov accelerated subgradient method with adaptive active set."""
    n = len(a_init)
    x = a_init.copy()
    x = np.maximum(0, x)
    s = x.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    x /= s
    y = x.copy()
    best_a = x.copy()
    best_score = _score_norm(x)
    
    t_nonce = 1.0
    patience = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        
        # Find active set with hysteresis
        sorted_idx = np.argsort(c)[::-1]
        active = []
        for idx in sorted_idx[:top_k + 3]:
            if c[idx] >= c_max * (1 - 0.001):
                active.append(int(idx))
            if len(active) >= top_k:
                break
        if not active:
            active = [int(np.argmax(c))]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                indices = np.arange(lo, hi + 1)
                g[lo:hi+1] += a_init[0]*0 + y[k - indices] + y[indices]
        g /= len(active)
        
        # Project to sum=0 subspace
        g = g - g.mean()
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Nesterov step
        step = lr * g / g_norm
        x_new = x - step
        x_new = np.maximum(0, x_new)
        s_new = x_new.sum()
        if s_new < 1e-15:
            break
        x_new /= s_new
        
        t_new = 1.0 + np.sqrt(1.0 + 4.0 * t_nonce * t_nonce) / 2.0
        momentum = (t_nonce - 1.0) / t_new
        y_new = x_new + momentum * (x_new - x)
        y_new = np.maximum(0, y_new)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new /= s_y
        
        t_nonce = t_new
        x = x_new
        y = y_new
        
        # Track best (evaluate less frequently for speed)
        if it % 50 == 0 or it < 10:
            sc = _score_norm(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()
                patience = 0
            else:
                patience += 1
                if patience > 200:
                    # Restart momentum
                    y = x.copy()
                    t_nonce = 1.0
                    patience = 0
    
    sc_final = _score_norm(x)
    if sc_final < best_score:
        best_score = sc_final
        best_a = x.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, T_start=None, T_end=None,

        # Softmax weights over active lags
        c_active = c[c_sorted[:num_active]]
        T_sm = max((c_active.max() - c_active.min()) * 0.05, 1e-10)
        w = np.exp((c_active - c_active.max()) / T_sm)
        w /= w.sum()
        
        g = np.zeros(n)
        for idx in range(num_active):
            k = int(c_sorted[idx])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                indices = np.arange(lo, hi + 1)
                g[lo:hi+1] += w[idx] * 2.0 * a[k - indices]
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        # Log-space gradient
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update /= update_norm
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = np.exp(z - z.max())
            a_cur /= a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
                if patience > 5:
                    current_lr_mult *= 0.8
                    patience = 0
                    if current_lr_mult < 0.05:
                        z = np.log(np.maximum(best_a, 1e-15))
                        z = z - z.mean()
                        z += 0.005 * np.random.randn(n)
                        z = z - z.mean()
                        m_buf = np.zeros(n)
                        v_buf = np.zeros(n)
                        current_lr_mult = 0.3
    
    a_final = np.exp(z - z.max())
    a_final /= a_final.sum()
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score


def _symmetric_optimize(a_init, deadline, max_iter=100000):
    """Optimize under the constraint a_j = a_{n-1-j} (palindromic)."""
    n = len(a_init)
    # Work with half the sequence
    half_n = (n + 1) // 2
    a_half = a_init[:half_n].copy()
    a_half = np.maximum(0, a_half)
    
    def _full_from_half(half):
        if n % 2 == 0:
            return np.concatenate([half, half[::-1]])
        else:
            return np.concatenate([half, half[-2::-1]])
    
    a = _full_from_half(a_half)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a /= s
    best_a = a.copy()
    best_score = _score(a)
    
    z = np.log(np.maximum(a_half, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(half_n)
    v_buf = np.zeros(half_n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr_start, lr_end = 0.02, 0.002
    patience = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a_half_cur = np.exp(z - z.max())
        a_half_cur /= a_half_cur.sum()
        a_cur = _full_from_half(a_half_cur)
        a_cur /= a_cur.sum()
        c = _fft_convolve(a_cur, a_cur)
        
        # Gradient of max c w.r.t. a_half
        sorted_idx = np.argsort(c)[::-1]
        num_active = min(5, len(sorted_idx))
        c_active = c[sorted_idx[:num_active]]
        T_sm = max((c_active.max() - c_active.min()) * 0.05, 1e-10)
        w = np.exp((c_active - c_active.max()) / T_sm)
        w /= w.sum()
        
        g_full = np.zeros(n)
        for idx in range(num_active):
            k = int(sorted_idx[idx])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                indices = np.arange(lo, hi + 1)
                g_full[lo:hi+1] += w[idx] * 2.0 * a_cur[k - indices]
        
        # Project gradient to symmetric subspace
        g_sym = 0.5 * (g_full + g_full[::-1])
        g_half = g_sym[:half_n]
        
        # Log-space gradient
        a_h = a_half_cur
        W = float(np.dot(a_h, g_half))
        z_grad = a_h * (g_half - W)
        
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update /= update_norm
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t % 200 == 0:
            a_half_eval = np.exp(z - z.max())
            a_half_eval /= a_half_eval.sum()
            a_eval = _full_from_half(a_half_eval)
            a_eval /= a_eval.sum()
            sc = _score(a_eval)
            if sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
                patience = 0
            else:
                patience += 1
                if patience > 5:
                    lr_mult *= 0.8
                    patience = 0
    
    return best_a, _score(best_a)


def _coordinate_polish(a, deadline, max_rounds=200):
    """Coordinate descent on elements affecting the peak."""
    n = len(a)
    a = a.copy()
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a /= s
    best_a = a.copy()
    best_score = _score(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        k_max = int(np.argmax(c))
        improved = False
        
        # Elements contributing to the peak
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        # Try the most impactful coordinates first
        contrib = np.zeros(hi - lo + 1)
        for j in range(lo, hi + 1):
            contrib[j - lo] = a[j] * a[k_max - j]
        
        order = np.argsort(contrib)[::-1]
        
        for idx in order[:min(50, len(order))]:
            j = lo + int(idx)
            old_val = best_a[j]
            if old_val < 1e-12:
                continue
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s_cand = a_cand.sum()
                if s_cand < 1e-15:
                    continue
                a_cand /= s_cand
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = 2.0 * n * float(c_cand.max())
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    c_max = float(c.max())
                    improved = True
                    break
            if improved:
                break
        
        if not improved and rnd > 20:
            # Try random coordinates
            any_improve = False
            for _ in range(100):
                j = np.random.randint(0, n)
                old_val = best_a[j]
                if old_val < 1e-12:
                    continue
                scale = np.random.choice([0.7, 0.85, 1.15, 1.3])
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s_cand = a_cand.sum()
                if s_cand < 1e-15:
                    continue
                a_cand /= s_cand
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = 2.0 * n * float(c_cand.max())
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    any_improve = True
                    break
            if not any_improve:
                break
    
    return best_a, _score(best_a)


def _multi_n_search(a_8192, deadline, n_values=[4096, 16384]):
    """Try different sequence lengths by interpolating/decimating."""
    best_a = a_8192.copy()
    best_score = _score(a_8192)
    n_orig = len(a_8192)
    
    for n_target in n_values:
        if time.monotonic() > deadline - 60:
            break
        
        # Interpolate from n_orig to n_target
        if n_target > n_orig:
            # Upsample using linear interpolation
            x_orig = np.linspace(0, 1, n_orig)
            x_new = np.linspace(0, 1, n_target)
            a_interp = np.interp(x_new, x_orig, a_8192)
            a_interp = np.maximum(0, a_interp)
            a_interp /= a_interp.sum()
        else:
            # Downsample by averaging
            ratio = n_orig // n_target
            a_dec = a_8192[:n_target * ratio].reshape(n_target, ratio).mean(axis=1)
            a_interp = np.maximum(0, a_dec)
            a_interp /= a_interp.sum()
        
        # Quick Adam optimization
        c = _fft_convolve(a_interp, a_interp)
        max_c = float(c.max())
        a_opt, sc = _adam_logspace(
            a_interp, 
            deadline=min(deadline - 30, time.monotonic() + max(10, (deadline - time.monotonic()) / 2)),
            T_start=max_c * 0.01,
            T_end=max_c * 0.0001,
            lr_start=0.02,
            lr_end=0.002,
            max_iter=100000,
            active_k=5,
            eval_interval=200
        )
        full_sc = _score(a_opt)
        print(f"  n={n_target}: score={full_sc:.10f}")
        
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: LP-based iteration, active-set Newton,
    Nesterov subgradient, Adam in log-space, symmetric optimization,
    and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
            n = len(a)
            print(f"Loaded parent: n={n}")
        else:
            raise ValueError("sum too small")
    except Exception as e:
        n = 8192
        a = np.ones(n) / n
        print(f"Using default uniform init (n={n}): {e}")
    
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")
    
    # Phase 1: Active-set Newton (fast, 0-300s)
    phase1_end = min(deadline, t_start + 300.0)
    print(f"Phase 1: Active-set Newton")
    a_as, sc_as = _active_set_newton(best_a, phase1_end, max_iter=3000, max_active=8)
    full_sc_as = _score(a_as)
    print(f"Phase 1 done: score={full_sc_as:.10f}, t={time.monotonic()-t_start:.1f}s")
    if full_sc_as < best_score:
        best_score = full_sc_as
        best_a = a_as.copy()
    
    # Phase 2: Adam log-space (300-700s)
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Adam log-space")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_opt, sc_opt = _adam_logspace(
            best_a, phase2_end,
            T_start=max_cb * 0.01, T_end=max_cb * 0.0005,
            lr_start=0.03, lr_end=0.003,
            max_iter=300000, active_k=5, eval_interval=200
        )
        full_sc_opt = _score(a_opt)
        print(f"Phase 2 done: score={full_sc_opt:.10f}, t={time.monotonic()-t_start:.1f}s")
        if full_sc_opt < best_score:
            best_score = full_sc_opt
            best_a = a_opt.copy()
        
        # Second Adam run with different params
        phase2b_end = min(deadline, t_start + 900.0)
        if time.monotonic() < phase2b_end:
            a_opt2, sc_opt2 = _adam_logspace(
                best_a, phase2b_end,
                T_start=max_cb * 0.003, T_end=max_cb * 0.0001,
                lr_start=0.015, lr_end=0.001,
                max_iter=200000, active_k=7, eval_interval=150
            )
            full_sc_opt2 = _score(a_opt2)
            print(f"Phase 2b done: score={full_sc_opt2:.10f}, t={time.monotonic()-t_start:.1f}s")
            if full_sc_opt2 < best_score:
                best_score = full_sc_opt2
                best_a = a_opt2.copy()
    
    # Phase 3: Nesterov subgradient (900-1100s)
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Nesterov subgradient")
        a_nes, sc_nes = _nesterov_subgradient(
            best_a, phase3_end, max_iter=300000, top_k=5,
            lr_start=0.008, lr_end=0.001
        )
        full_sc_nes = _score(a_nes)
        print(f"Phase 3 done: score={full_sc_nes:.10f}, t={time.monotonic()-t_start:.1f}s")
        if full_sc_nes < best_score:
            best_score = full_sc_nes
            best_a = a_nes.copy()
    
    # Phase 4: Active-set Newton again (1100-1300s)
    phase4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Active-set Newton (refined)")
        a_as2, sc_as2 = _active_set_newton(best_a, phase4_end, max_iter=5000, max_active=10)
        full_sc_as2 = _score(a_as2)
        print(f"Phase 4 done: score={full_sc_as2:.10f}, t={time.monotonic()-t_start:.1f}s")
        if full_sc_as2 < best_score:
            best_score = full_sc_as2
            best_a = a_as2.copy()
    
    # Phase 5: Symmetric optimization (1300-1500s)
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Symmetric optimization")
        a_sym, sc_sym = _symmetric_optimize(best_a, phase5_end, max_iter=80000)
        full_sc_sym = _score(a_sym)
        print(f"Phase 5 done: score={full_sc_sym:.10f}, t={time.monotonic()-t_start:.1f}s")
        if full_sc_sym < best_score:
            best_score = full_sc_sym
            best_a = a_sym.copy()
    
    # Phase 6: Coordinate descent polish (1500-1650s)
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end:
        print(f"Phase 6: Coordinate descent")
        a_cd, sc_cd = _coordinate_polish(best_a, phase6_end, max_rounds=300)
        full_sc_cd = _score(a_cd)
        print(f"Phase 6 done: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.1f}s")
        if full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()
    
    # Phase 7: Random restarts with quick optimization (1650-1770s)
    phase7_end = min(deadline, t_start + 1770.0)
    if time.monotonic() < phase7_end:
        print(f"Phase 7: Random restarts")
        n_restarts = 4
        time_per = max(15.0, (phase7_end - time.monotonic()) / n_restarts)
        for r in range(n_restarts):
            if time.monotonic() > phase7_end - 5:
                break
            sigma = 0.003 + 0.002 * r
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(len(z))
            z = z - z.mean()
            a_pert = np.exp(z - z.max())
            a_pert /= a_pert.sum()
            
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(phase7_end - 5, time.monotonic() + time_per * 0.7)
            a_pert_opt, sc_pert = _adam_logspace(
                a_pert, sub_dl,
                T_start=max_cp * 0.005, T_end=max_cp * 0.0001,
                lr_start=0.02, lr_end=0.002,
                max_iter=50000, active_k=5, eval_interval=200
            )
            full_sc_pert = _score(a_pert_opt)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_sc_pert:.10f}")
            if full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()
    
    # Phase 8: Final polish (1770-deadline)
    if time.monotonic() < deadline - 10:
        print(f"Phase 8: Final polish")
        a_final, sc_final = _active_set_newton(best_a, deadline - 10, max_iter=2000, max_active=10)
        full_sc_final = _score(a_final)
        if full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()
        
        if time.monotonic() < deadline - 5:
            a_cd_f, sc_cd_f = _coordinate_polish(best_a, deadline - 5, max_rounds=100)
            full_sc_cd_f = _score(a_cd_f)
            if full_sc_cd_f < best_score:
                best_score = full_sc_cd_f
                best_a = a_cd_f.copy()
    
    # Final validation
    final = np.asarray(best_a, dtype=np.float64).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
