import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_raw(a):
    """Score assuming sum is already 1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _qp_direction(a, c, alpha=0.05, K_max=30):
    """
    Compute the QP direction to reduce the peak by factor (1-alpha).
    Returns d (direction vector with sum(d)=0) or None if infeasible.
    """
    n = len(a)
    M = float(c.max())
    M_new = M * (1.0 - alpha)
    
    # Active set: all lags above M_new
    active = np.where(c > M_new)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > K_max:
        # Keep only the top K_max
        top_idx = np.argsort(c[active])[::-1][:K_max]
        active = active[top_idx]
    
    K = len(active)
    if K == 0:
        return None
    
    # Residuals: how much each active lag needs to be reduced
    r = M_new - c[active]  # all negative or zero
    
    # Build KKT matrix
    autoc = _autocorr(a)
    B = np.zeros((K, K))
    for i in range(K):
        for j in range(K):
            m = abs(active[i] - active[j])
            B[i,j] = 4.0 * autoc[m] if m < n else 0.0
    
    # G = B - (4/n) * ones * ones^T
    ones = np.ones(K)
    G = B - (4.0/n) * np.outer(ones, ones)
    
    # Solve G * lambda = -r (note: -r is positive since r is negative)
    try:
        lam = np.linalg.solve(G + 1e-10 * np.eye(K), -r)
    except:
        lam = np.linalg.lstsq(G, -r, rcond=None)[0]
    
    # Check feasibility: lambda should be >= 0 for the active set to be correct
    # If not, some constraints are not truly active; we'll handle this simply
    lam = np.maximum(lam, 0)
    
    # Compute d
    mu = -2.0 * np.sum(lam) / n
    d = np.full(n, mu)  # start with mu * 1
    for i in range(K):
        ki = active[i]
        if lam[i] > 1e-15:
            # d[j] -= lam[i] * 2 * a[ki-j] for valid j
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            d[lo:hi+1] -= lam[i] * 2.0 * a[ki-hi:ki-lo+1][::-1]
    
    # Ensure sum(d) = 0
    d -= d.mean()
    
    # Sanity check
    if not np.all(np.isfinite(d)):
        return None
    
    return d

One more thought: I should also try a different approach entirely - using the frequency domain directly.

In the frequency domain, the DFT of the convolution c = a*a is C[k] = A[k]^2 (where A is the DFT of the zero-padded a). The peak in the time domain corresponds to the maximum of the IDFT of C = A^2.

We want to minimize max_c = max_k |IDFT(A^2)[k]|.

This is equivalent to: find A (constrained by A being the DFT of a non-negative real sequence of length n, zero-padded to N) that minimizes the peak of IDFT(A^2).

The constraint that a is non-negative and real is:
- A is real-even: A[k] = A[N-k] (conjugate symmetry for real signals)
- a = IDFT(A), a[j] >= 0 for j = 0, ..., n-1, a[j] = 0 for j >= n

This is a complex constraint. But we can work with it iteratively:
1. Start with some A
2. Compute a = IDFT(A), clip to a[j] >= 0 for j < n, set a[j] = 0 for j >= n
3. Recompute A = DFT(a)
4. Modify A to reduce the peak of IDFT(A^2)
5. Repeat

Step 4 is the key. In the frequency domain, IDFT(A^2)[k] = (1/N) sum_m A[m]^2 * e^{2πimk/N}. To reduce the peak, we want to reduce the "coherence" of A^2, i.e., make A^2 more "incoherent" (spread out in frequency).

One approach: at each iteration, find the frequency bin m where |A[m]|^2 is largest, and reduce it (while maintaining the constraint that a is real and non-negative).

This is essentially a "spectral peak reduction" algorithm.

"""
Step function optimization using active-set QP with autocorrelation-based KKT solve,
pairwise coordinate swaps, and iterated local search.
Minimizes 2n*max(self-convolution)/(sum)^2 for non-negative sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _autocorr(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    r = np.fft.irfft(A * np.conj(A), N)
    return r[:n]


def _qp_direction(a, c, alpha, K_max):
    n = len(a)
    M = float(c.max())
    M_new = M * (1.0 - alpha)
    
    # Active set: lags that need to be reduced
    active = np.where(c >= M_new)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    if len(active) > K_max:
        order = np.argsort(c[active])[::-1]
        active = active[order[:K_max]]
    
    K = len(active)
    r_target = M_new - c[active]  # negative or zero
    
    # Build KKT matrix using autocorrelation
    autoc = _autocorr(a)
    B = np.zeros((K, K))
    for i in range(K):
        for j in range(K):
            m = abs(int(active[i]) - int(active[j]))
            B[i, j] = 4.0 * autoc[m] if m < n else 0.0
    
    ones = np.ones(K)
    G = B - (4.0 / n) * np.outer(ones, ones)
    
    # Regularize for numerical stability
    G += 1e-10 * np.eye(K)
    
    try:
        lam = np.linalg.solve(G, -r_target)
    except np.linalg.LinAlgError:
        lam = np.linalg.lstsq(G, -r_target, rcond=None)[0]
    
    if not np.all(np.isfinite(lam)):
        return None
    
    # Clip negative lambdas (approximate active set handling)
    lam = np.maximum(lam, 0.0)
    
    # Compute direction d
    mu = -2.0 * np.sum(lam) / n
    d = np.full(n, mu, dtype=np.float64)
    for i in range(K):
        ki = int(active[i])
        li = lam[i]
        if li > 1e-15:
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            if lo <= hi:
                d[lo:hi+1] -= li * 2.0 * a[ki-hi:ki-lo+1][::-1]
    
    # Center
    d -= d.mean()
    
    if not np.all(np.isfinite(d)) or np.linalg.norm(d) < 1e-20:
        return None
    
    return d
