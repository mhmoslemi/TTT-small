def _next_pow2(n):

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)

    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].mean())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].mean())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved and alpha < 1e-12:
            break

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=16):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                                   0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.002, 0.001])
    betas = betas[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128, 256, 512]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(200, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lp_based_step(f, M, max_constraints=200, max_rounds=30, deadline=None):
    n = len(f)
    c = _full_convolve(f, f)
    top_k = np.argsort(c)[::-1][:min(50, 2 * n - 1)]
    active_rows = set(int(x) for x in top_k[:20])

    g_best = np.ones(n) / n

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        rows = sorted(active_rows)
        A_ub = np.zeros((len(rows), n))
        for idx, row in enumerate(rows):
            j_low = max(0, row - (n - 1))
            j_high = min(n - 1, row)
            if j_low <= j_high:
                A_ub[idx, j_low:j_high + 1] = f[row - j_high:row - j_low + 1][::-1]

        b_ub = np.full(len(rows), M)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 30})
            if not res.success:
                break
            g = res.x
            g = np.maximum(g, 0)
            s = g.sum()
            if s < 1e-15:
                break
            g = g / s
            fg = _full_convolve(f, g)
            violated = np.where(fg > M * fg.max() / fg.max() + 1e-12)[0]
            violated = np.where(fg > M + 1e-12)[0]
            if len(violated) == 0:
                g_best = g
                break
            n_add = min(20, len(violated))
            idx_add = np.random.default_rng(rnd).choice(violated, n_add, replace=False)
            for r in idx_add:
                active_rows.add(int(r))
            g_best = g
        except Exception:
            break

    return g_best


def _lp_outer_loop(f_init, deadline, n_iters=5, t_mix=0.3):
    n = len(f_init)
    a = f_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        g = _lp_based_step(a, M, max_constraints=100, max_rounds=20, deadline=deadline)
        if g is None or float(g.sum()) < 1e-10:
            break
        g = np.maximum(g, 0)
        s_g = g.sum()
        if s_g < 1e-10:
            break
        g = g / s_g

        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new
        t_mix = max(0.1, t_mix * 0.7)

    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _topk_equalize(a, deadline, n_steps=200, top_k=5):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx]
        mean_val = vals.mean()
        max_val = vals.max()
        if max_val - mean_val < 1e-14:
            break

        # Gradient of (c[top_idx].max() - c[top_idx].mean())
        # Approximate: push down the max, push up the mean
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / top_k
        w[int(top_idx[0])] -= 0.5
        w[int(np.argmax(c))] += 0.5

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = h - np.dot(w, c)

        lr = 0.01 / (1.0 + 0.01 * step)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining aggressive LSE, LP-based mixing,
    top-K equalization, adaptive Newton, and multi-n exploration.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        n_parent = 8192
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Aggressive LSE on parent (400s) ===
    print(f"\n[Phase 1] Aggressive LSE (target: {min(400, deadline - t_start - 600):.0f}s)")
    p1_deadline = min(deadline - 600, t_start + 400)
    if time.monotonic() < p1_deadline:
        a_opt, sc_opt = _lbfgs_lse(_best_seq.copy(), p1_deadline, min_remaining=10, n_stages=16)
        upd(a_opt, sc_opt, "lse_aggressive")

    # === Phase 2: Power-sum refinement (150s) ===
    print(f"\n[Phase 2] Power-sum L-BFGS")
    p2_deadline = min(deadline - 450, time.monotonic() + 150)
    if time.monotonic() < p2_deadline:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), p2_deadline, min_remaining=10)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 3: LP-based mixing (150s) ===
    print(f"\n[Phase 3] LP-based mixing")
    p3_deadline = min(deadline - 300, time.monotonic() + 150)
    if time.monotonic() < p3_deadline:
        a_lp, sc_lp = _lp_outer_loop(_best_seq.copy(), p3_deadline, n_iters=4, t_mix=0.25)
        upd(a_lp, sc_lp, "lp_mix")
        # Then LSE polish after LP
        if _score(a_lp) < _best_score * 1.002 and time.monotonic() < p3_deadline - 5:
            a_lp2, sc_lp2 = _lbfgs_lse(a_lp, p3_deadline - 2, min_remaining=3, n_stages=6)
            upd(a_lp2, sc_lp2, "lp_mix_lse")

    # === Phase 4: Top-K equalization + Newton (150s) ===
    print(f"\n[Phase 4] Top-K equalize + Newton")
    p4_deadline = min(deadline - 200, time.monotonic() + 150)
    if time.monotonic() < p4_deadline:
        a_eq, sc_eq = _topk_equalize(_best_seq, p4_deadline - 30, n_steps=300, top_k=8)
        upd(a_eq, sc_eq, "topk_eq")
        if time.monotonic() < p4_deadline - 10:
            a_nw, sc_nw = _newton_active_set(a_eq, p4_deadline - 5, max_iter=150, tol_frac=0.001)
            upd(a_nw, sc_nw, "topk_newton")

    # === Phase 5: Second LSE pass with smaller beta (200s) ===
    print(f"\n[Phase 5] Second LSE pass (low beta)")
    p5_deadline = min(deadline - 150, time.monotonic() + 200)
    if time.monotonic() < p5_deadline:
        a_lse2, sc_lse2 = _lbfgs_lse(_best_seq.copy(), p5_deadline, min_remaining=10, n_stages=12)
        upd(a_lse2, sc_lse2, "lse_pass2")

    # === Phase 6: Newton polish (100s) ===
    print(f"\n[Phase 6] Newton active-set polish")
    p6_deadline = min(deadline - 100, time.monotonic() + 100)
    if time.monotonic() < p6_deadline:
        a_nw2, sc_nw2 = _newton_active_set(_best_seq, p6_deadline, max_iter=200, tol_frac=0.0005)
        upd(a_nw2, sc_nw2, "newton_polish")

    # === Phase 7: Subgradient + Newton alternating (150s) ===
    print(f"\n[Phase 7] SG + Newton alternating")
    p7_deadline = min(deadline - 80, time.monotonic() + 150)
    if time.monotonic() < p7_deadline:
        a_cur = _best_seq.copy()
        for rnd in range(3):
            if time.monotonic() > p7_deadline - 20:
                break
            sg_dead = min(p7_deadline - 20, time.monotonic() + 30)
            a_sg, sc_sg = _subgrad_polish(a_cur, sg_dead, n_steps=2000, lr_init=0.006, tol_frac=0.001)
            if sc_sg < _score(a_cur):
                a_cur = a_sg
            upd(a_cur, _score(a_cur), f"alt_sg_{rnd}")
            nw_dead = min(p7_deadline - 10, time.monotonic() + 25)
            a_nw3, sc_nw3 = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.001)
            if sc_nw3 < _score(a_cur):
                a_cur = a_nw3
            upd(a_cur, _score(a_cur), f"alt_nw_{rnd}")

    # === Phase 8: Multi-n exploration (250s) ===
    print(f"\n[Phase 8] Multi-n exploration")
    p8_deadline = min(deadline - 60, time.monotonic() + 250)
    n_candidates = [4096, 8192, 16384, 24576, 32768]
    if n_parent not in n_candidates and 1024 <= n_parent <= 65536:
        n_candidates.append(n_parent)
        n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > p8_deadline - 30:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        sc_init = _score(a_init)
        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, p8_deadline - 20, min_remaining=5, n_stages=8)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        if _score(a_opt_n) < _best_score * 1.003 and time.monotonic() < p8_deadline - 15:
            nw_dead = min(p8_deadline - 10, time.monotonic() + 40)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=80, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 9: Diverse restarts (150s) ===
    print(f"\n[Phase 9] Diverse restarts")
    p9_deadline = min(deadline - 30, time.monotonic() + 150)
    n_restarts = 5
    for i in range(n_restarts):
        if time.monotonic() > p9_deadline - 20:
            break
        if i == 0:
            p_start = _perturb_log(_best_seq, 0.02, rng)
        elif i == 1:
            p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.25)
        elif i == 2:
            p_start = _perturb_block_swap(_best_seq, rng, n_swaps=3)
        elif i == 3:
            p_start = _perturb_amplitude(_best_seq, rng, sigma=0.06)
        else:
            p_start = _perturb_log(_best_seq, 0.04, rng)

        a_r, sc_r = _lbfgs_lse(p_start, p9_deadline - 15, min_remaining=5, n_stages=8)
        upd(a_r, sc_r, f"restart_{i}")

        if sc_r < _best_score * 1.002 and time.monotonic() < p9_deadline - 10:
            nw_dead = min(p9_deadline - 5, time.monotonic() + 20)
            a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.002)
            upd(a_r2, sc_r2, f"restart_{i}_nw")

    # === Phase 10: Final ultra-precise polish (30s) ===
    print(f"\n[Phase 10] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 64, 'maxls': 150}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=60, tol_frac=0.0003)
            upd(a_fin2, sc_fin2, "final_newton")

        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=2000, lr_init=0.003, tol_frac=0.0003)
            upd(a_fin3, sc_fin3, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
