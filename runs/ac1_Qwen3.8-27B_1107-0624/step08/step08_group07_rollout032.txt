```python
def _hessian_apply(a, active_lags, d):
    """Apply H = (2/m) * sum P_k to vector d."""
    m = len(active_lags)
    n = len(a)
    result = np.zeros(n)
    for k in active_lags:
        # P_k maps j -> k-j (with valid range)
        # (P_k @ d)[j] = d[k-j] if 0 <= k-j < n, else 0
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # (P_k @ d)[j] = d[k-j] for j in [lo, hi]
        # i.e., result[j] += d[k-j] for j in [lo, hi]
        result[lo:hi+1] += d[k - hi : k - lo + 1][::-1]
    return (2.0 / m) * result
```

"""
Optimized step function search using Newton-like active-set refinement
with Hessian structure exploitation, multi-temperature Adam, and
adaptive coordinate descent. Targets n=8192 with careful time allocation.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2*n*max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient_at_lags(a, c, lags):
    n = len(a)
    c_lags = c[lags]
    # Use softmax weighting over active lags
    range_c = c_lags.max() - c_lags.min()
    if range_c < 1e-15:
        w = np.ones(len(lags)) / len(lags)
    else:
        T = range_c * 0.1 + 1e-10
        w = np.exp((c_lags - c_lags.max()) / T)
        w = w / w.sum()
    g = np.zeros(n)
    for idx, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        g[lo:hi+1] += w[idx] * 2.0 * a[k-hi:k-lo+1][::-1]
    return g

    """Compute H @ d where H = (2/m) * sum_k P_k.
    P_k is the permutation that maps j -> k-j (with zero-padding)."""
    m = len(active_lags)
    n = len(a)
    result = np.zeros(n)
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # (P_k @ d)[j] = d[k-j] for j in [lo, hi]
            # d[k-j] for j in [lo, hi] = d[k-hi : k-lo+1] reversed
            result[lo:hi+1] += d[k-hi:k-lo+1][::-1]
    return (2.0 / m) * result

def _project_to_sum_zero(d):
    """Project d to the subspace sum(d) = 0."""
    return d - d.mean()

            z = z_new

            z = z_new

                    m_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _newton_active_set(a, deadline, max_iter=5000, top_k=10,

    H = (2/m) * sum P_k for active lags.

            active = active[top_idx]

        if best_alpha > 0:

        best_sc = best_score
        best_a_cand = None
        for alpha in alphas:
            if alpha <= 0:
                continue
            a_cand = a - alpha * g_proj
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()

def _cg_newton_step(a, active_lags, g_proj, max_cg_iter=50, lambda_reg=1e-4):
    """Solve (H + lambda*I) d = -g_proj using CG, with projection to sum=0.
    Uses the structure of H = (2/m) * sum P_k."""
    n = len(a)
    m = len(active_lags)
    
    def matvec(v):
        """Compute (H + lambda*I) @ v, projected to sum=0."""
        Hv = _hessian_mv(a, active_lags, v)
        result = Hv + lambda_reg * v
        result = result - result.mean()  # project to sum=0
        return result
    
    # CG with projection
    r = -g_proj.copy()
    r = r - r.mean()
    p = r.copy()
    r_norm_sq = float(r @ r)
    d = np.zeros(n)
    
    for i in range(max_cg_iter):
        Ap = matvec(p)
        p_Ap = float(p @ Ap)
        if abs(p_Ap) < 1e-15:
            break
        alpha = r_norm_sq / p_Ap
        if not np.isfinite(alpha) or abs(alpha) > 1e6:
            break
        d += alpha * p
        d = d - d.mean()  # maintain sum=0
        r_new = r - alpha * Ap
        r_new = r_new - r_new.mean()
        r_new_norm_sq = float(r_new @ r_new)
        if r_new_norm_sq < 1e-20:
            r = r_new
            break
        beta = r_new_norm_sq / r_norm_sq
        if not np.isfinite(beta):
            break
        p = r_new + beta * p
        p = p - p.mean()
        r = r_new
        r_norm_sq = r_new_norm_sq
    
    return d


def _newton_refine(a, deadline, max_iter=3000, top_k=15, tol=0.995):
    """Newton refinement using CG to solve the Hessian system."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active lags
        tol_val = c_max * tol
        active = np.where(c >= tol_val)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > top_k:
            top_idx = np.argsort(c[active])[::-1][:top_k]
            active = active[top_idx]
        
        # Compute gradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        g_proj = g - g.mean()
        
        g_norm = float(np.linalg.norm(g_proj))
        if g_norm < 1e-12:
            break
        
        # Newton step via CG
        d_newton = _cg_newton_step(a, active, g_proj, max_cg_iter=30, 
                                     lambda_reg=1e-4)
        d_newton = d_newton - d_newton.mean()
        
        # Also compute gradient step
        d_grad = -g_proj / g_norm
        
        # Try Newton step vs gradient step
        best_sc = best_score
        best_cand = None
        
        # Try Newton direction with various step sizes
        d_n_norm = float(np.linalg.norm(d_newton))
        if d_n_norm > 1e-15:
            d_newton_norm = d_newton / d_n_norm
            for scale in [1.0/d_n_norm * 0.01, 1.0/d_n_norm * 0.005, 
                         1.0/d_n_norm * 0.001, 0.01, 0.005, 0.001]:
                a_cand = a + scale * d_newton
                a_cand = np.maximum(0, a_cand)
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_cand = a_cand
        
        # Try gradient direction
        for scale in [0.01, 0.005, 0.001, 0.0005]:
            a_cand = a - scale * d_grad
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_cand = a_cand
        
        if best_cand is not None:
            a = best_cand
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
        else:
            break
    
    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=50000, top_k=5, initial_lr=0.003):
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
        
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        g_tangent = g - float(np.sum(g)) * a  # This is wrong, should be g - (g@1/n)*1 or similar
        # Actually for the constraint sum(a)=1, the tangent space is {d: sum(d)=0}
        # Project g onto tangent space: g_proj = g - (sum(g)/n) * 1
        g_tangent = g - g.mean()
        
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break
        
        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9995
        else:
            lr *= 0.98
            if lr < 1e-8:
                break
    
    return best_a, _score_norm(best_a)
