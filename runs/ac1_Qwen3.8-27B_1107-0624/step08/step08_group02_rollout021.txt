import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k = k << 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    """Compute H_W @ v where H_W is the Hankel matrix defined by W."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    """Solve H_W @ x = rhs using conjugate gradient with FFT acceleration."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.01:
            # Restart
            x = x_new
            r = rhs - _hankel_matvec(x, W, n)
            p = r.copy()
            rs = float(r @ r)
            if rs < tol * tol * rs0:
                break
            continue
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _smoobj(z, beta, n):

    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.3 * 0.3))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) / 0.35) ** 4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        # Flat in the middle, taper at edges
        w = 0.3  # width of flat region
        a = np.where(np.abs(t - 0.5) < w, 1.0,
                     0.5 * (1 + np.cos(np.pi * (np.abs(t - 0.5) - w) / (0.5 - w))))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell_mix":
        t = np.linspace(0, 1, n)
        a = 0.6 * np.exp(-((t - 0.5) / 0.4) ** 2) + 0.4 * np.exp(-((t - 0.5) / 0.2) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _run_lbfgs_phase(a_init, deadline, min_time_remaining=30.0, beta_start=None):
    """Run L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    if beta_start is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
        beta_start = max(gap * 0.3, max_c * 0.002)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    beta = beta_start
    min_beta = max_c * 1e-10
    
    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        scale = max(1.0, n / 8192.0)
        # Estimate time per iteration
        est_time_per_iter = 0.002 * scale
        max_iter = min(10000, max(20, int(remaining / (est_time_per_iter * 30))))
        
        try:
            obj_fn = z.copy()  # placeholder
            result = minimize(
                lambda zz: _smoobj(zz, beta, n),
                z,
                jac=lambda zz: _smograd(zz, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-17:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception as e:
            print(f"  L-BFGS error at beta={beta:.2e}: {e}")
        
        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        
        if new_gap < 1e-9 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.4
        else:
            beta *= 0.5
        
        beta = min(beta, c_max_new * 0.001)
    
    return current_a

def _symmetric_newton(a_init, deadline, max_rounds=100):

        half = a_init[:n//2].copy()

        if n % 2 == 1:
            return np.concatenate([h, h[1:][::-1]])

So the active set at the optimum (where we've optimized the shape) should be just the center, or a very small number of indices near the center.

If the active set is just the center (`{n-1}`), then:
- `W = H_{n-1}` (the Hankel matrix for the center convolution)
- `H_{n-1} @ a` is the "cross-correlation" of `a` with its reverse, evaluated at each position.
- For symmetric `a`, `H_{n-1} @ a = a` (it's the identity on the symmetric subspace).
- The Newton step becomes: minimize `a^T a` (subject to `sum(a) = 1`, `a >= 0`), which gives `a = uniform`.
