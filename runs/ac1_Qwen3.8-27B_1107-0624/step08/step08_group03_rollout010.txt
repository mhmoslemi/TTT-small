Uses L-BFGS with power-sum smoothing, followed by FFT-accelerated
active-set QP with trust region, and ADMM polishing.

import time
from scipy.optimize import minimize

def _next_pow2(m):
    if m <= 1:
        return 1
    k = 1
    while k < m:
        k <<= 1
    return k

def _full_convolve(a, b=None):
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _next_pow2(m):
    if m <= 1:
        return 1
    k = 1
    while k < m:

    return k


def _full_convolve(a, b=None):
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _hess_matvec(x, m, n):
    """Compute H*x where (H*x)_j = 2*conv(m, flip(x))[n-1+j]"""
    x_flip = x[::-1]
    out_len = n + len(m) - 1
    N = _next_pow2(out_len)
    fm = np.fft.rfft(m, N)
    fx = np.fft.rfft(x_flip, N)
    result = np.fft.irfft(fm * fx, N)[:out_len]
    return 2.0 * result[n - 1:2 * n - 1]


def _cg_solve_mvp(mvp_func, rhs, max_iter=200, tol=1e-12):
    """Conjugate gradient solve using a matrix-vector product function."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    target = tol * rhs_norm
    for i in range(max_iter):
        Ap = mvp_func(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < target:
            break
        beta = rs_new / rs
        p = r + beta * p
        r = r - alpha * Ap  # already done
        rs = rs_new
    return x

        diff = c_cur - target
        # grad = 2 * rho * 2 * conv(diff, flip(a))[n-1:2n-1]
        diff_conv = _full_convolve(diff, a[::-1])
        grad_a = 4.0 * rho * diff_conv[n - 1:2 * n - 1]

        # Subtract mean to handle sum constraint
        grad_a_centered = grad_a - np.mean(grad_a)

        # Simple gradient step with projection
        lr = 1.0 / (8.0 * rho * np.max(np.abs(diff_conv[n-1:2*n-1])) + 1e-10)
        lr = min(lr, 0.1)
        a_new = a - lr * grad_a_centered
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        a = a_new

        # t, u update: min t + (rho/2)||u - conv(a,a) + z||^2 s.t. u_i <= t
        c_a = _full_convolve(a, a)
        # For each i: min (rho/2)(u_i - c_a_i + z_i)^2 s.t. u_i <= t
        # Together with min t
        # Optimal: u_i = min(t, c_a_i - z_i), t = max_i (c_a_i - z_i)
        # Actually: t = max_i (c_a_i - z_i) and u_i = min(t, c_a_i - z_i) = c_a_i - z_i if c_a_i - z_i < t, else t
        # Hmm, let me think again.
        # min_{t, u} t + (rho/2) sum_i (u_i - c_a_i + z_i)^2 s.t. u_i <= t
        # For fixed t: u_i = min(t, c_a_i - z_i)
        # Then: min_t t + (rho/2) sum_i (min(t, c_a_i - z_i) - c_a_i + z_i)^2
        # = min_t t + (rho/2) sum_i max(0, t - c_a_i + z_i)^2
        # This is convex in t. Solve by bisection.
        # f(t) = t + (rho/2) sum_i max(0, t - (c_a_i - z_i))^2
        # f'(t) = 1 + rho * sum_i max(0, t - (c_a_i - z_i))
        # Set = 0: 1 + rho * sum_i max(0, t - d_i) = 0 where d_i = c_a_i - z_i
        # This requires sum_i max(0, t - d_i) = -1/rho, which is impossible for t >= max(d_i)
        # So the minimum is at t = max(d_i) (boundary of the feasible region for the subproblem)
        # Wait no, t is free (can be negative). Let's just use bisection.
        
        d = c_a - z  # d_i = c_a_i - z_i
        # f(t) = t + (rho/2) * sum_i max(0, t - d_i)^2
        # f'(t) = 1 + rho * sum_i max(0, t - d_i)
        # We want f'(t) = 0 => sum_i max(0, t - d_i) = -1/rho
        # Since LHS >= 0 and RHS < 0, f' > 0 always, so minimum is at t -> -inf
        # That can't be right. Let me reconsider.
        
        # Actually the constraint is u_i <= t, and the objective includes t (linear).
        # If t -> -inf, then u_i <= t -> -inf, and the quadratic term blows up.
        # The minimum should be finite.
        
        # Let me just use: t_new = max(d) (the smallest t such that u_i = d_i <= t for all i)
        # This is the optimal solution when the quadratic doesn't pull t down.
        # Actually for large rho, t ≈ max(d). For small rho, t can be smaller.
        # Let's just use t = max(d) as an approximation (exact for rho -> inf).
        
        t_new = float(np.max(d))
        u_new = np.minimum(t_new, d)
        
        # Update dual
        z = z + (c_a - u_new)
        u = u_new
        t = t_new

        if it % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
            # Adaptive rho
            if it > 0 and it % 100 == 0:
                rho *= 1.5
                z = z / 1.5

    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _momentum_descent(a_init, deadline, max_steps=5000, lr_init=0.02, momentum=0.9):
    """Momentum gradient descent in log-space with top-k weighting."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-k weighted gradient
        k = min(50, len(c) // 5)
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[top_k] = 1.0 / k

        a_flip = a[::-1]
        wc = _full_convolve(w, a_flip)
        h = wc[n - 1:2 * n - 1]
        W = np.dot(w, c)
        grad_a = 2.0 * a * (h - W)

        # Convert to z-gradient
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        step = lr / (1.0 + 0.001 * i)
        v = momentum * v - step * grad_z
        z = z + v
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.01, lr_init)
            else:
                lr *= 0.92
                if i > 200 and (best_sc - sc) > 0.01:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    lr = lr_init * 0.5

    return best_a, best_sc


def _voronoi_split_refine(a_init, deadline):
    """
    Refine by splitting the sequence into segments and optimizing
    the relative heights of segments, then redistributing.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    # Try different segment sizes
    for seg_size in [16, 32, 64, 128]:
        if time.monotonic() > deadline:
            break
        n_seg = n // seg_size
        if n_seg < 2:
            continue

        # Current segment means
        seg_means = np.array([a[i*seg_size:(i+1)*seg_size].mean() for i in range(n_seg)])
        seg_sums = np.array([a[i*seg_size:(i+1)*seg_size].sum() for i in range(n_seg)])

        # Optimize segment weights to minimize max autocorrelation
        # This is a smaller problem (n_seg variables)
        # Use simplex method: parametrize as softmax
        from scipy.optimize import minimize as _min

        best_seg_sums = seg_sums.copy()
        for restart in range(3):
            z_seg = np.log(np.maximum(seg_sums, 1e-10)) + 0.1 * np.random.standard_normal(n_seg)
            z_seg = z_seg - z_seg.mean()

            def seg_obj(z_s):
                w_s = _softmax(z_s)
                a_try = np.zeros(n)
                for i in range(n_seg):
                    a_try[i*seg_size:(i+1)*seg_size] = w_s[i] / seg_size
                return _score(a_try)

            try:
                res = _min(seg_obj, z_seg, method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
                w_s = _softmax(res.x)
                a_try = np.zeros(n)
                for i in range(n_seg):
                    a_try[i*seg_size:(i+1)*seg_size] = w_s[i] / seg_size
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    best_seg_sums = w_s * seg_sums.sum()
            except Exception:
                pass

    return best_a, best_sc


def _diversify_and_optimize(a_base, deadline, rng, min_time_remaining=30):
    """Try perturbations of a_base and optimize each."""
    n = len(a_base)
    best_a = a_base.copy()
    best_sc = _score(a_base)

    sigmas = [0.01, 0.03, 0.05, 0.1, 0.2]
    for sigma in sigmas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_pert = a_base.copy()
        z = np.log(np.maximum(a_pert, 1e-30))
        z += sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS polish
        remaining = min(deadline - time.monotonic() - min_time_remaining, 120)
        if remaining < 10:
            continue
        a_opt = _lbfgs_power_sum(a_pert, time.monotonic() + remaining, min_time_remaining=5)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc - 1e-10:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Diversify sigma={sigma}: {sc_opt:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS power-sum from parent
    Phase 2: Active-set QP refinement (multiple tolerance levels)
    Phase 3: ADMM polishing
    Phase 4: Momentum descent
    Phase 5: Try different n values
    Phase 6: Diversify and re-optimize
    Phase 7: Final tight active-set QP
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score_val:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={parent_sc:.10f}")
        update_best(parent, parent_sc, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS power-sum optimization
    print(f"\n[Phase 1] L-BFGS power-sum (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_p1 = _lbfgs_power_sum(parent, deadline - 500, min_time_remaining=400)
        update_best(a_p1, _score(a_p1), "lbfgs_p1")

    # Phase 2: Active-set QP (main refinement)
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_p2, sc_p2 = _active_set_qp_v2(best_seq, min(deadline - 100, time.monotonic() + 300),
                                         max_rounds=200, tol_active=0.001)
        update_best(a_p2, sc_p2, "as_qp_v2")

    # Phase 3: Tighter active-set QP
    print(f"\n[Phase 3] Tighter AS-QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_p3, sc_p3 = _active_set_qp_v2(best_seq, min(deadline - 50, time.monotonic() + 120),
                                         max_rounds=150, tol_active=0.0003)
        update_best(a_p3, sc_p3, "as_qp_tight")

    # Phase 4: ADMM
    print(f"\n[Phase 4] ADMM (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_p4, sc_p4 = _admm_refine(best_seq, min(deadline - 40, time.monotonic() + 80),
                                     max_iter=300, rho_init=0.5)
        update_best(a_p4, sc_p4, "admm")

    # Phase 5: Momentum descent
    print(f"\n[Phase 5] Momentum descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_p5, sc_p5 = _momentum_descent(best_seq, min(deadline - 30, time.monotonic() + 60),
                                         max_steps=3000, lr_init=0.01, momentum=0.95)
        update_best(a_p5, sc_p5, "momentum")

    # Phase 6: Try different n
    print(f"\n[Phase 6] Different n (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [8192, 32768]:
        if time.monotonic() > deadline - 60:
            break
        if best_seq is not None:
            a_try = _interpolate_to_n(best_seq, n_try)
        else:
            a_try = _make_profile(n_try, "uniform")
        sc_try = _score(a_try)
        print(f"  n={n_try}: init score={sc_try:.10f}")
        if sc_try < best_score_val + 0.01:
            # Optimize
            a_opt = _lbfgs_power_sum(a_try, min(deadline - 30, time.monotonic() + 90),
                                      min_time_remaining=20)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n{n_try}_lbfgs")
            if time.monotonic() < deadline - 20:
                a_qp, sc_qp = _active_set_qp_v2(a_opt, min(deadline - 10, time.monotonic() + 60),
                                                  max_rounds=50, tol_active=0.001)
                update_best(a_qp, sc_qp, f"n{n_try}_as")

    # Phase 7: Diversify
    print(f"\n[Phase 7] Diversify (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_div, sc_div = _diversify_and_optimize(best_seq, deadline, rng, min_time_remaining=20)
        update_best(a_div, sc_div, "diversify")

    # Phase 8: Structured initializations on best n
    print(f"\n[Phase 8] Structured inits (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else n_parent
    if time.monotonic() < deadline - 30:
        for kind in ["sine", "bimodal", "cos_asym"]:
            if time.monotonic() > deadline - 10:
                break
            a_struct = _make_profile(n_best, kind, rng)
            sc_struct = _score(a_struct)
            if sc_struct < best_score_val + 0.05 and time.monotonic() < deadline - 15:
                a_s_opt = _active_set_qp_v2(a_struct, min(deadline - 8, time.monotonic() + 30),
                                             max_rounds=30, tol_active=0.005)
                update_best(a_s_opt, _score(a_s_opt), f"struct_{kind}")

    # Phase 9: Final ultra-tight active-set
    print(f"\n[Phase 9] Final tight AS-QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_qp_v2(best_seq, deadline - 2,
                                           max_rounds=100, tol_active=0.00005)
        update_best(a_fin, sc_fin, "final_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
