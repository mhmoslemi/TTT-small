"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active.astype(np.float64), n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(np.sum(c[active]))
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.sum(c_try[active]))
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _effective_active_count(c, c_max, beta):
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 1
    w = e / s
    return 1.0 / float(np.sum(w * w))


def _adaptive_lbfgs_lse(a_init, deadline, min_remaining=30.0, n_max_stages=20):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.01)

    n_stages_done = 0
    prev_beta = base_beta
    patience = 0

    while time.monotonic() + min_remaining < deadline and n_stages_done < n_max_stages:
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        eff_count = _effective_active_count(c_cur, max_c, prev_beta)
        if n_stages_done == 0:
            beta = base_beta
        else:
            target_eff = min(eff_count * 0.5, 50.0)
            beta = prev_beta * max(0.4, min(0.8, target_eff / max(eff_count, 1.0)))
            beta = max(beta, max_c * 0.00001)

        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 150}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-15:
                improvement = current_sc - sc_new
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                c_cur = _full_convolve(current_a, current_a)
                max_c = float(c_cur.max())
                patience = 0
            else:
                patience += 1
                if patience > 3:
                    break
        except Exception:
            break

        prev_beta = beta
        n_stages_done += 1

    return current_a, current_sc


def _pocs_spectral_flatten(a_init, deadline, n_iter=200, omega=1.0):
    n = len(a_init)
    M = _next_pow2(2 * n)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        mag = np.abs(A)
        mag[0] = 0
        nonzero = mag > 1e-30
        if not np.any(nonzero):
            break
        target_mag = np.mean(mag[nonzero])

        A_flat = np.zeros_like(A)
        A_flat[0] = A[0]
        for k in range(1, M):
            if mag[k] > 1e-30:
                A_flat[k] = A[k] / mag[k] * target_mag
            else:
                A_flat[k] = 0

        a_new_full = np.fft.ifft(A_flat).real
        a_new = a_new_full[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        a = (1.0 - omega) * a + omega * a_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = a_new
            s = a.sum()
        a = a / s

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, n_steps=10000, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    momentum = 0.9
    lr_base = 0.005

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        t = 1.0 / (np.sqrt(2.0) * (i + 1) + 1.0)
        z_new = z + momentum * (z - z_prev)
        z_prev = z

        a = _softmax(z_new)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        lr = lr_base * t
        z = z_new - lr * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _make_initialization(n, kind="ramp_asym"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp_inc":
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif kind == "ramp_dec":
        a = np.arange(n, 0, -1, dtype=np.float64)
        return a / a.sum()
    elif kind == "ramp_asym":
        t = np.linspace(0, 1, n)
        a = 0.3 + 1.4 * t
        return a / a.sum()
    elif kind == "exp_decay":
        a = np.exp(-2.0 * np.linspace(0, 1, n))
        return a / a.sum()
    elif kind == "exp_decay_rev":
        a = np.exp(2.0 * np.linspace(0, 1, n))
        return a / a.sum()
    elif kind == "sigmoid":
        t = np.linspace(-5, 5, n)
        a = 1.0 / (1.0 + np.exp(-t))
        return a / a.sum()
    elif kind == "quadratic_ramp":
        t = np.linspace(0, 1, n)
        a = 0.2 + 2.0 * t ** 2
        return a / a.sum()
    elif kind == "cubic_ramp":
        t = np.linspace(0, 1, n)
        a = 0.1 + 3.0 * t ** 3
        return a / a.sum()
    elif kind == "two_plateau":
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.5, 1.5, 0.5)
        return a / a.sum()
    elif kind == "smooth_ramp_plateau":
        t = np.linspace(0, 1, n)
        a = 0.3 + 2.0 * np.sigmoid((t - 0.3) * 8.0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):

        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


_global_best_seq = None
_global_best_score = np.inf


def _update_best(p, sc=None):
    global _global_best_seq, _global_best_score
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return
    if sc is None:
        sc = _score(p)
    if np.isfinite(sc) and sc < _global_best_score:
        _global_best_seq = p.copy()
        _global_best_score = float(sc)


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_best_seq, _global_best_score
    _global_best_seq = None
    _global_best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def log(msg):
        print(f"[{time.monotonic() - t_start:7.1f}s] {msg}")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    # Phase 0: Quick diverse exploration
    log("Phase 0: Diverse initialization exploration")
    target_n = 16384
    if n_parent == target_n:
        target_n = n_parent

    candidate_n = 1024
    best_init = None
    best_init_sc = np.inf
    kinds = ["ramp_asym", "ramp_inc", "ramp_dec", "exp_decay", "uniform",
             "sigmoid", "quadratic_ramp", "two_plateau", "smooth_ramp_plateau", "exp_decay_rev"]
    for kind in kinds:
        a_try = _make_initialization(candidate_n, kind)
        sc_try = _score(a_try)
        if sc_try < best_init_sc:
            best_init_sc = sc_try
            best_init = a_try.copy()

    log(f"  Best init (n={candidate_n}): {best_init_sc:.10f}")

    # Use parent if available and good
    if parent is not None:
        sc_parent = _score(parent)
        log(f"  Parent (n={n_parent}): {sc_parent:.10f}")
        _update_best(parent, sc_parent)

    # Phase 1: Multi-scale L-BFGS (coarse to fine)
    log("Phase 1: Multi-scale L-BFGS with adaptive LSE")
    scales = [256, 1024, 4096, 8192, target_n]
    current_a = best_init
    current_sc = best_init_sc

    for n_scale in scales:
        if time.monotonic() + 100 > deadline:
            break
        if len(current_a) != n_scale:
            current_a = _resize_sequence(current_a, n_scale)

        if n_scale <= 4096:
            phases = 3
        elif n_scale <= 8192:
            phases = 3
        else:
            phases = 4

        for ph in range(phases):
            if time.monotonic() + 60 > deadline:
                break
            a_opt, sc_opt = _adaptive_lbfgs_lse(current_a, deadline, min_remaining=50)
            if sc_opt < current_sc - 1e-15:
                current_a = a_opt
                current_sc = sc_opt
                _update_best(current_a, sc_opt)
                log(f"  n={n_scale} phase{ph}: {sc_opt:.10f}")

    log(f"  Multi-scale best: {current_sc:.10f} (n={len(current_a)})")

    # Phase 2: Full L-BFGS at target size
    log("Phase 2: Full L-BFGS at target size")
    if time.monotonic() < deadline - 300:
        a_full, sc_full = _adaptive_lbfgs_lse(_global_best_seq.copy(), deadline, min_remaining=250)
        if sc_full < _global_best_score:
            _update_best(a_full, sc_full)
            log(f"  Full L-BFGS: {sc_full:.10f}")

    # Phase 3: POCS spectral flattening
    log("Phase 3: POCS spectral flattening")
    if time.monotonic() < deadline - 200:
        pocs_deadline = min(deadline - 180, time.monotonic() + 120)
        a_pocs, sc_pocs = _pocs_spectral_flatten(_global_best_seq.copy(), pocs_deadline, n_iter=100, omega=0.8)
        if sc_pocs < _global_best_score:
            _update_best(a_pocs, sc_pocs)
            log(f"  POCS: {sc_pocs:.10f}")

        # Follow up with L-BFGS after POCS
        if time.monotonic() < deadline - 150:
            a_after, sc_after = _adaptive_lbfgs_lse(_global_best_seq.copy(), deadline, min_remaining=120)
            if sc_after < _global_best_score:
                _update_best(a_after, sc_after)
                log(f"  Post-POCS L-BFGS: {sc_after:.10f}")

    # Phase 4: Newton active-set
    log("Phase 4: Newton active-set")
    if time.monotonic() < deadline - 150:
        nw_dead = min(deadline - 130, time.monotonic() + 150)
        a_nw, sc_nw = _newton_active_set(_global_best_seq, nw_dead, max_iter=300, tol_frac=0.001)
        if sc_nw < _global_best_score:
            _update_best(a_nw, sc_nw)
            log(f"  Newton: {sc_nw:.10f}")

    # Phase 5: Nesterov subgradient
    log("Phase 5: Nesterov-accelerated subgradient")
    if time.monotonic() < deadline - 100:
        ns_dead = min(deadline - 80, time.monotonic() + 100)
        a_ns, sc_ns = _nesterov_subgrad(_global_best_seq, ns_dead, n_steps=8000, lr_init=0.004, tol_frac=0.001)
        if sc_ns < _global_best_score:
            _update_best(a_ns, sc_ns)
            log(f"  Nesterov: {sc_ns:.10f}")

    # Phase 6: Alternating Newton + Subgradient
    log("Phase 6: Alternating Newton + Subgradient")
    if time.monotonic() < deadline - 100:
        a_cur = _global_best_seq.copy()
        for rnd in range(4):
            if time.monotonic() > deadline - 80:
                break
            nw_dead = min(deadline - 80, time.monotonic() + 30)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.002)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            _update_best(a_cur, _score(a_cur))

            sg_dead = min(deadline - 80, time.monotonic() + 25)
            a_sg, sc_sg = _subgrad_polish(a_cur, sg_dead, n_steps=3000, lr_init=0.004, tol_frac=0.0008)
            if sc_sg < _score(a_cur):
                a_cur = a_sg
            _update_best(a_cur, _score(a_cur))
            log(f"  Round {rnd}: {_global_best_score:.10f}")

    # Phase 7: Diverse restarts
    log("Phase 7: Diverse restarts with perturbations")
    if time.monotonic() < deadline - 60:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            if i == 0:
                p_start = _perturb_log(_global_best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_global_best_seq, rng, n_modes=8, max_amp=0.3)
            elif i == 2:
                p_start = _perturb_block_swap(_global_best_seq, rng, n_swaps=3)
            elif i == 3:
                p_start = _perturb_log(_global_best_seq, 0.04, rng)
            else:
                p_start = _make_initialization(len(_global_best_seq), kinds[i % len(kinds)])

            a_r, sc_r = _adaptive_lbfgs_lse(p_start, deadline, min_remaining=20)
            _update_best(a_r, sc_r)
            log(f"  Restart {i}: {sc_r:.10f}")

            if _global_best_score < 1.503 and time.monotonic() < deadline - 15:
                nw_dead = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(_global_best_seq, nw_dead, max_iter=40, tol_frac=0.002)
                _update_best(a_r2, sc_r2)

    # Phase 8: Final aggressive polish
    log("Phase 8: Final polish")
    if _global_best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_global_best_seq)
        c_fin = _full_convolve(_global_best_seq, _global_best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_global_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()

        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            # Ultra-low beta LSE
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-14, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                _update_best(a_fin, sc_fin)
                log(f"  Ultra-LSE: {sc_fin:.10f}")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_global_best_seq, deadline - 2, max_iter=60, tol_frac=0.0005)
            _update_best(a_fin2, sc_fin2)
            log(f"  Final Newton: {sc_fin2:.10f}")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_global_best_seq, deadline - 1, n_steps=5000, lr_init=0.003, tol_frac=0.0005)
            _update_best(a_fin3, sc_fin3)
            log(f"  Final SG: {sc_fin3:.10f}")

    # Final validation
    log("=" * 60)
    if _global_best_seq is not None:
        final = np.clip(_global_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            log("FALLBACK to uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        log(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    log(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
