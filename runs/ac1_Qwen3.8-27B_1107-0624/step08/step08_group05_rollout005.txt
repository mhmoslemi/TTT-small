"""
Adaptive multi-scale Newton optimizer with progressive active-set refinement,
subgradient momentum, coordinate-wise line search, and multi-size exploration
for minimizing step function autocorrelation peak ratio.
"""
import numpy as np
import time
from scipy.optimize import minimize


# ============================================================
# Core utilities (top-level, no closures)
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# LSE objectives for smooth approximation of max
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


# ============================================================
# Power-sum objectives
# ============================================================

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


# ============================================================
# Active set Newton with adaptive Hessian
# ============================================================

def _hess_matvec(v, active_f, n):
    """Compute H*v where H comes from the active set constraint."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-12, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set_adaptive(a_init, deadline, max_iter_per_stage=200,
                                  stages=(0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001)):
    """
    Newton method with progressively shrinking active set.
    stages: fraction of max_c used as tolerance for active set identification.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for stage_frac in stages:
        if time.monotonic() > deadline:
            break
        for it in range(max_iter_per_stage):
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            if c_max < 1e-30:
                break
            tol = stage_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True

            active_f = active.astype(np.float64)
            grad = _hess_matvec(a, active_f, n)
            grad_t = grad - np.mean(grad)
            gnorm = float(np.linalg.norm(grad_t))
            if gnorm < 1e-14:
                break

            mu = 1e-8

            def _hv(v):
                return _hess_matvec(v, active_f, n) + mu * v

            dx = _cg_solve(_hv, -grad_t, tol=1e-12, max_iter=300)
            dx = dx - np.mean(dx)

            f_old = float(c[active].sum())
            alpha = 1.0
            a_new = a.copy()
            for ls in range(50):
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try[active].sum())
                if f_try < f_old - 1e-14 * alpha * gnorm:
                    a_new = a_try
                    break
                alpha *= 0.5
            else:
                if alpha < 1e-14:
                    break
                a_new = a

            a = a_new
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc

def _subgrad_momentum(a_init, deadline, n_steps=3000, lr_init=0.01,
                      tol_frac=0.001, momentum=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    momentum_vec = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Compute subgradient of max(c)
        a_rev = a[::-1]
        active_f = active.astype(np.float64)
        wc = _full_convolve(active_f, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(active_f, c)) / active.sum()
        grad_a = 2.0 * n * (h / active.sum() - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        momentum_vec = momentum * momentum_vec - step * grad_z
        z = z + momentum_vec
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
            else:
                lr *= 0.95
                if lr < 1e-6:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum_vec = np.zeros(n)
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _coord_line_search(a_init, deadline, n_top=200, search_width=0.05):
    """
    For the top n_top coordinates by |a|, do a 1D search to find optimal
    perturbation that minimizes the peak convolution value.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Identify important coordinates (those with large values or near peak)
    c = _full_convolve(a, a)
    peak_idx = int(np.argmax(c))
    # Coordinates that most affect the peak
    influence = np.zeros(n)
    for j in range(n):
        k = peak_idx - j
        if 0 <= k < n:
            influence[j] = a[k]
    top_coords = np.argsort(influence)[::-1][:n_top]

    for coord_idx in top_coords:
        if time.monotonic() > deadline:
            break
        i = coord_idx
        a_orig = a[i]

        # Try a range of perturbations
        # The effect: c[k] changes by 2*delta*a[k-i] (approximately)
        # We want to find delta that minimizes max(c + 2*delta*a_shift)
        best_delta = 0.0
        best_val = best_sc
        a_shift = np.zeros(2 * n - 1)
        lo = max(0, i)
        hi = min(2 * n - 2, (n - 1) + i)
        for k in range(lo, hi + 1):
            a_shift[k] = a[k - i]

        # Ternary search on delta in [-a_orig, a_orig * search_width / a_orig * 0.1]
        # Simplified: search in [-0.5*a_orig, 0.5*a_orig]
        lo_d = -0.5 * a_orig
        hi_d = 0.5 * a_orig
        if a_orig < 1e-15:
            continue

        for _ in range(25):
            m1 = lo_d + (hi_d - lo_d) / 3.0
            m2 = hi_d - (hi_d - lo_d) / 3.0

            a_try1 = a.copy()
            a_try1[i] = a[i] + m1
            a_try1 = np.maximum(a_try1, 0)
            s1 = a_try1.sum()
            if s1 > 1e-15:
                a_try1 /= s1
                c1 = _full_convolve(a_try1, a_try1)
                v1 = float(c1.max())
            else:
                v1 = np.inf

            a_try2 = a.copy()
            a_try2[i] = a[i] + m2
            a_try2 = np.maximum(a_try2, 0)
            s2 = a_try2.sum()
            if s2 > 1e-15:
                a_try2 /= s2
                c2 = _full_convolve(a_try2, a_try2)
                v2 = float(c2.max())
            else:
                v2 = np.inf

            if v1 < v2:
                hi_d = m2
            else:
                lo_d = m1

        best_delta = (lo_d + hi_d) / 2.0
        a_new = a.copy()
        a_new[i] = max(0, a[i] + best_delta)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new /= s_new
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new

    return best_a, best_sc

def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=12):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.7, 0.4, 0.25, 0.15, 0.08,
                                   0.04, 0.02, 0.01, 0.005, 0.002, 0.001])
    betas = betas[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-13, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):

    current_a = a_init.copy()

    z = np.log(np.maximum(current_a, 1e-30))

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.08):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 10)
        w = np.linspace(0, 1, taper) ** 2
        a[:taper] = w
        a[-taper:] = w[::-1]
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Find top n_peaks values and set threshold
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(n_peaks - 1, len(c_sorted) - 1)]
        active = c >= threshold
        active_f = active.astype(np.float64)

        # Weight active set by how close to max
        weights = (c[active] - threshold) / (c_max - threshold + 1e-30) + 0.1
        weighted_active = np.zeros_like(c)
        weighted_active[active] = weights / weights.sum()

        grad = _hess_matvec(a, weighted_active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-7

        def _hv(v):
            return _hess_matvec(v, weighted_active, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-12, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[weighted_active > 0].sum() * (c_max - threshold + 1) / (c_max + 1))
        # Use the actual max as our objective for line search
        alpha = 1.0
        a_new = a.copy()
        sc_old = _score(a)
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc_old:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-14:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer with adaptive multi-scale Newton,
    peak-flooding, coordinate line search, subgradient momentum,
    and multi-size exploration.
    """
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def _upd(p, sc=None, label=""):
        nonlocal _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")
            return True
        return False

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        _upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (45s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 45)
    for n_try in [2048, 4096, 8192]:
        if time.monotonic() > expl_deadline:
            break
        for kind in ["uniform", "tapered", "cos2", "flat_taper", "bimodal"]:
            if time.monotonic() > expl_deadline:
                break
            a_try = _make_profile(n_try, kind)
            sc_try = _score(a_try)
            _upd(a_try, sc_try, f"expl_n{n_try}_{kind}")

    # Quick LSE on best found so far
    if _best_seq is not None and time.monotonic() < deadline - 600:
        a_opt, sc_opt = _lbfgs_lse(_best_seq.copy(), min(deadline, t_start + 90), min_remaining=500)
        _upd(a_opt, sc_opt, "expl_lbfgs_quick")

    # === Phase 2: Adaptive multi-scale Newton (500s) ===
    print(f"\n[Phase 2] Adaptive multi-scale Newton")
    if time.monotonic() < deadline - 500:
        nw_dead = min(deadline - 450, time.monotonic() + 500)
        a_nw, sc_nw = _newton_active_set_adaptive(
            _best_seq.copy(), nw_dead,
            max_iter_per_stage=100,
            stages=(0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001)
        )
        _upd(a_nw, sc_nw, "newton_adaptive")

    # === Phase 3: Peak-flooding Newton (200s) ===
    print(f"\n[Phase 3] Peak-flooding Newton")
    if time.monotonic() < deadline - 200:
        pf_dead = min(deadline - 180, time.monotonic() + 200)
        a_pf, sc_pf = _peak_flood_newton(_best_seq.copy(), pf_dead, n_peaks=3, max_iter=400)
        _upd(a_pf, sc_pf, "peak_flood_3")
        if _score(_best_seq) < _best_score:
            a_pf2, sc_pf2 = _peak_flood_newton(_best_seq.copy(), min(deadline - 180, time.monotonic() + 100), n_peaks=5, max_iter=200)
            _upd(a_pf2, sc_pf2, "peak_flood_5")

    # === Phase 4: L-BFGS LSE main (300s) ===
    print(f"\n[Phase 4] L-BFGS LSE main")
    if time.monotonic() < deadline - 300:
        a_lb, sc_lb = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=280, n_stages=14)
        _upd(a_lb, sc_lb, "lbfgs_lse_main")

        # Power sum
        if time.monotonic() < deadline - 100:
            a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=80)
            _upd(a_ps, sc_ps, "lbfgs_power_sum")

    # === Phase 5: Single-peak Newton (150s) ===
    print(f"\n[Phase 5] Single-peak Newton")
    if time.monotonic() < deadline - 150:
        sp_dead = min(deadline - 130, time.monotonic() + 150)
        a_sp, sc_sp = _newton_single_peak(_best_seq.copy(), sp_dead, max_iter=400, tol_frac=0.0005)
        _upd(a_sp, sc_sp, "single_peak_newton")

    # === Phase 6: Coordinate line search (150s) ===
    print(f"\n[Phase 6] Coordinate line search")
    if time.monotonic() < deadline - 150:
        cl_dead = min(deadline - 130, time.monotonic() + 150)
        n_top = min(300, len(_best_seq) // 50)
        a_cl, sc_cl = _coord_line_search(_best_seq.copy(), cl_dead, n_top=n_top, search_width=0.05)
        _upd(a_cl, sc_cl, "coord_line_search")

    # === Phase 7: Subgradient momentum (150s) ===
    print(f"\n[Phase 7] Subgradient momentum")
    if time.monotonic() < deadline - 150:
        sg_dead = min(deadline - 130, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_momentum(_best_seq.copy(), sg_dead, n_steps=5000, lr_init=0.008, tol_frac=0.001, momentum=0.9)
        _upd(a_sg, sc_sg, "subgrad_momentum")

    # === Phase 8: Newton + coordinate alternating (150s) ===
    print(f"\n[Phase 8] Newton + coordinate alternating")
    if time.monotonic() < deadline - 150:
        a_cur = _best_seq.copy()
        for round_i in range(4):
            if time.monotonic() > deadline - 120:
                break
            # Newton step
            nw_d = min(deadline - 120, time.monotonic() + 35)
            a_nw2, sc_nw2 = _newton_active_set_adaptive(
                a_cur, nw_d, max_iter_per_stage=50,
                stages=(0.1, 0.05, 0.02, 0.01, 0.005, 0.002)
            )
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            _upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")

            # Coordinate step
            cl_d = min(deadline - 120, time.monotonic() + 25)
            a_cl2, sc_cl2 = _coord_line_search(a_cur, cl_d, n_top=100, search_width=0.03)
            if sc_cl2 < _score(a_cur):
                a_cur = a_cl2
            _upd(a_cur, _score(a_cur), f"alt_coord_{round_i}")

    # === Phase 9: Multi-size exploration (200s) ===
    print(f"\n[Phase 9] Multi-size exploration")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates and n_parent >= 2048:
        n_candidates.append(n_parent)
    if len(_best_seq) not in n_candidates:
        n_candidates.append(len(_best_seq))
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        # Quick LSE
        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, deadline, min_remaining=60, n_stages=8)
        _upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        # Quick Newton
        if time.monotonic() < deadline - 50:
            nw_d = min(deadline - 40, time.monotonic() + 60)
            a_opt_n2, sc_opt_n2 = _newton_active_set_adaptive(
                a_opt_n, nw_d, max_iter_per_stage=50,
                stages=(0.2, 0.1, 0.05, 0.02, 0.01)
            )
            _upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 10: Diverse restarts (200s) ===
    print(f"\n[Phase 10] Diverse restarts")
    if time.monotonic() < deadline - 60:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq, rng, max_amp=0.2)
            elif i == 2:
                p_start = _perturb_block_swap(_best_seq, rng, n_swaps=4)
            elif i == 3:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.05)
            elif i == 4:
                p_start = _perturb_log(_best_seq, 0.04, rng)
            elif i == 5:
                p_start = _perturb_fourier_phase(_best_seq, rng, max_amp=0.4)
            elif i == 6:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.12)
            else:
                p_start = _perturb_block_swap(_best_seq, rng, n_swaps=8)

            # Quick LSE on perturbed
            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=30, n_stages=6)
            _upd(a_r, sc_r, f"restart_{i}")

            # Newton polish on best
            if _score(a_r) < _best_score * 1.005 and time.monotonic() < deadline - 20:
                nw_d = min(deadline - 15, time.monotonic() + 30)
                a_r2, sc_r2 = _newton_active_set_adaptive(
                    a_r, nw_d, max_iter_per_stage=40,
                    stages=(0.1, 0.05, 0.02, 0.01, 0.005)
                )
                _upd(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 11: Final aggressive polish (180s) ===
    print(f"\n[Phase 11] Final aggressive polish")
    if _best_seq is not None and time.monotonic() < deadline - 10:
        # Ultra-low beta LSE
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-14, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                _upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final adaptive Newton
        if time.monotonic() < deadline - 8:
            a_fin2, sc_fin2 = _newton_active_set_adaptive(
                _best_seq.copy(), deadline - 5,
                max_iter_per_stage=80,
                stages=(0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002)
            )
            _upd(a_fin2, sc_fin2, "final_newton_tight")

        # Final coordinate search
        if time.monotonic() < deadline - 5:
            a_fin3, sc_fin3 = _coord_line_search(
                _best_seq.copy(), deadline - 3, n_top=200, search_width=0.03
            )
            _upd(a_fin3, sc_fin3, "final_coord")

        # Final subgradient
        if time.monotonic() < deadline - 3:
            a_fin4, sc_fin4 = _subgrad_momentum(
                _best_seq.copy(), deadline - 1,
                n_steps=3000, lr_init=0.003, tol_frac=0.0005, momentum=0.95
            )
            _upd(a_fin4, sc_fin4, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
