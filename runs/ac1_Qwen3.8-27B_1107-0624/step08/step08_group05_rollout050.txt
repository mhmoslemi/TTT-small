"""
LP-based Chebyshev refinement combined with adaptive L-BFGS and spectral initialization
for minimizing the step function autocorrelation peak score: 2n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. LP-based alternating Chebyshev refinement (solves exact LP subproblems)
2. Analytical anti-palindromic initialization
3. Scale-space warm start (optimize at smaller n, then upscale)
4. Adaptive active-set management with dual-feasibility checks

import numpy as np

# Core utilities

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_constraint_matrix(a, active_indices, n):
    """Build the constraint matrix C where C[k, m] = a[k - m] if valid."""
    n_active = len(active_indices)
    C = np.zeros((n_active, n))
    for i, k in enumerate(active_indices):
        # For variable b[m], coefficient is a[k-m] if 0 <= k-m < n
        for m in range(n):
            idx = k - m
            if 0 <= idx < n:
                C[i, m] = a[idx]
    return C


def _build_constraint_matrix_fast(a, active_indices, n):
    """Build constraint matrix using vectorized operations."""
    n_active = len(active_indices)
    C = np.zeros((n_active, n))
    m = np.arange(n)  # variable indices
    for i, k in enumerate(active_indices):
        idx = k - m  # a[k-m] for each m
        valid = (idx >= 0) & (idx < n)
        C[i, valid] = a[idx[valid]]
    return C

def _solve_lp_primal(a, M, active_indices, n):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for k in active_indices, b >= 0.
    Returns optimal b vector.
    """
    C = _build_constraint_matrix_fast(a, active_indices, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(len(active_indices))
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        b_opt = np.maximum(result.x, 0)
        s = b_opt.sum()
        if s > 1e-15:
            return b_opt / s
    return None


def _solve_lp_dual(a, M, active_indices, n):
    """
    Solve the dual LP: min M * sum(lambda) s.t. C^T @ lambda >= 1, lambda >= 0.
    Returns optimal lambda and the implied b solution.
    """
    C = _build_constraint_matrix_fast(a, active_indices, n)
    n_active = len(active_indices)
    
    # Dual: min M * 1^T lambda s.t. C^T lambda >= 1, lambda >= 0
    c_obj = M * np.ones(n_active)
    A_ub = -C.T  # n x n_active
    b_ub = -np.ones(n)
    bounds = [(0, None)] * n_active
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success or result.x is None:
        return None, None
    
    lambda_star = np.maximum(result.x, 0)
    
    # Recover primal from complementary slackness
    # b(m) > 0 only where (C^T lambda)(m) <= 1 + tol
    Ct_lambda = C.T @ lambda_star
    tol_cs = 1e-6
    support = Ct_lambda <= (1.0 + tol_cs)
    
    if support.sum() == 0:
        return lambda_star, None
    
    # On support, solve C @ b = M * 1 (restricted to active rows and support columns)
    C_sub = C[:, support]
    b_sub = np.zeros(support.sum())
    
    if C_sub.shape[0] >= C_sub.shape[1]:
        # Overdetermined or square - use least squares
        b_sub, _, _, _ = np.linalg.lstsq(C_sub, M * np.ones(n_active), rcond=None)
        b_sub = np.maximum(b_sub, 0)
    else:
        # Underdetermined - solve LP: max sum(b_sub) s.t. C_sub @ b_sub = M*1, b_sub >= 0
        c_sub = -np.ones(support.sum())
        A_eq = C_sub
        b_eq = M * np.ones(n_active)
        res_sub = linprog(c_sub, A_eq=A_eq, b_ub=b_eq, bounds=[(0, None)] * support.sum(), method='highs')
        if res_sub.success:
            b_sub = np.maximum(res_sub.x, 0)
        else:
            b_sub, _, _, _ = np.linalg.lstsq(C_sub, M * np.ones(n_active), rcond=None)
            b_sub = np.maximum(b_sub, 0)
    
    b_full = np.zeros(n)
    b_full[support] = b_sub
    s = b_full.sum()
    if s > 1e-15:
        b_full = b_full / s

# L-BFGS with LSE smoothing (improved)
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _lbfgs_lse_improved(a_init, deadline, min_remaining=60.0, n_stages=20):
    """Improved L-BFGS with more aggressive beta reduction."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # More stages, geometric schedule
    betas = base_beta * (0.75 ** np.arange(n_stages))
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # More iterations at low beta
        if i < n_stages // 3:
            max_iter = 2000
        elif i < 2 * n_stages // 3:
            max_iter = 3000
        else:
            max_iter = 5000
        
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(max_iter, max(100, int(remaining / (est_time * 10))))
        
        try:
            result = minimize(
                _lse_objective, z, args=(float(beta), n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 200,
                         'maxfun': max_iter * 5}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc

# Subgradient polish
# ============================================================

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _subgrad_polish(a_init, deadline, n_steps=10000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_lr = lr_init
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        # Polyak-like step size
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = best_patience_lr * 0.5
                    best_patience_lr = lr
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

# Analytical initializations

def _make_anti_palindromic(n):
    """Sequence that is large on one side, small on the other (anti-palindromic)."""
    t = np.linspace(0, 1, n)
    a = 0.1 + 0.9 * t
    return a / a.sum()


def _make_cosine_tapered(n):
    """Cosine tapered with slight asymmetry."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 0.8 * (1.0 + 0.2 * np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_ramped(n, asymmetry=0.3):
    """Ramp with controllable asymmetry."""
    t = np.linspace(0, 1, n)
    a = 1.0 + asymmetry * (2 * t - 1)
    a = np.maximum(a, 0.05)
    return a / a.sum()


def _make_dual_peak(n):
    """Two peaks at 1/3 and 2/3 positions."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 1/3) / 0.1) ** 2) + np.exp(-0.5 * ((t - 2/3) / 0.1) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resize_sequence(a, n_target):
    if len(a) == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

# Scale-space optimization
# ============================================================

def _scale_space_optimize(start_n, target_n, deadline, min_remaining=300):
    """Optimize at increasing sizes for a good warm start."""
    n = start_n
    a = _make_anti_palindromic(n)
    best_sc = _score(a)
    
    # Quick optimize at start_n
    d1 = min(deadline - min_remaining, time.monotonic() + 30)
    a, sc = _lbfgs_lse_improved(a, d1, min_remaining=5, n_stages=8)
    best_sc = min(best_sc, sc)
    
    # Progressively increase n
    while n < target_n:
        n_new = min(n * 2, target_n)
        if time.monotonic() + min_remaining > deadline:
            break
        a = _resize_sequence(a, n_new)
        d2 = min(deadline - min_remaining, time.monotonic() + 40)
        a, sc = _lbfgs_lse_improved(a, d2, min_remaining=5, n_stages=10)
        best_sc = min(best_sc, sc)
        n = n_new
    
    return a, best_sc

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer with LP-based Chebyshev refinement.
    Combines analytical initialization, scale-space warm start,
    adaptive L-BFGS, LP-based refinement, and final gradient polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    _best = {'seq': None, 'score': np.inf}
    
    def upd(p, sc=None, label=""):
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best['score']:
            _best['seq'] = p.copy()
            _best['score'] = float(sc)
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_p = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_p:.10f}")
        upd(parent, sc_p, "parent")
    else:
        parent = _make_anti_palindromic(16384)
        upd(parent, _score(parent), "init")
    
    target_n = max(8192, n_parent) if n_parent >= 4096 else 16384
    # Cap at reasonable size
    target_n = min(target_n, 32768)
    
    # === Phase 1: Analytical initializations (20s) ===
    print(f"\n[Phase 1] Analytical initializations (target_n={target_n})")
    p1_deadline = min(deadline, t_start + 20)
    
    inits = {
        "anti_pal": _make_anti_palindromic(target_n),
        "cos_taper": _make_cosine_tapered(target_n),
        "ramp_03": _make_ramped(target_n, 0.3),
        "ramp_05": _make_ramped(target_n, 0.5),
        "dual_peak": _make_dual_peak(target_n),
    }
    
    for name, a_try in inits.items():
        if time.monotonic() > p1_deadline:
            break
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"init_{name}")
    
    # If parent is available, blend with best initial
    if parent is not None and _best['seq'] is not None:
        for t_blend in [0.3, 0.5, 0.7]:
            blended = (1 - t_blend) * parent + t_blend * _best['seq']
            blended = np.maximum(blended, 0)
            s = blended.sum()
            if s > 0.01:
                blended = blended / s
                sc_b = _score(blended)
                if time.monotonic() < p1_deadline:
                    upd(blended, sc_b, f"blend_{t_blend}")
    
    if _best['seq'] is None:
        _best['seq'] = _make_anti_palindromic(target_n)
        _best['score'] = _score(_best['seq'])
    
    # === Phase 2: Scale-space warm start (100s) ===
    print(f"\n[Phase 2] Scale-space warm start")
    if n_parent < 4096 and time.monotonic() < deadline - 1000:
        p2_deadline = min(deadline - 1000, time.monotonic() + 100)
        a_ss, sc_ss = _scale_space_optimize(1024, target_n, p2_deadline, min_remaining=1000)
        upd(a_ss, sc_ss, "scale_space")
    else:
        # Upscale parent to target_n if needed
        if n_parent != target_n:
            a_up = _resize_sequence(parent, target_n)
            upd(a_up, _score(a_up), "upscale_parent")
            if time.monotonic() < deadline - 1000:
                p2_deadline = min(deadline - 1000, time.monotonic() + 80)
                a_opt, sc_opt = _lbfgs_lse_improved(a_up, p2_deadline, min_remaining=5, n_stages=12)
                upd(a_opt, sc_opt, "upscale_opt")
    
    # === Phase 3: Main L-BFGS with LSE (500s) ===
    print(f"\n[Phase 3] Main L-BFGS LSE")
    if time.monotonic() < deadline - 600:
        p3_deadline = min(deadline - 600, time.monotonic() + 500)
        a_main, sc_main = _lbfgs_lse_improved(_best['seq'].copy(), p3_deadline, 
                                               min_remaining=600, n_stages=20)
        upd(a_main, sc_main, "lbfgs_main")
    
    # === Phase 4: LP-based Chebyshev refinement (500s) ===
    print(f"\n[Phase 4] LP-based Chebyshev refinement")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, time.monotonic() + 500)
        
        # Multiple LP refinement passes with different tolerance
        for tol_frac in [0.01, 0.005, 0.003, 0.002, 0.001]:
            if time.monotonic() > p4_deadline:
                break
            a_lp, sc_lp = _lp_chebyshev_refine(_best['seq'].copy(), p4_deadline,
                                               max_iters=50, tol_frac=tol_frac, use_dual=True)
            upd(a_lp, sc_lp, f"lp_{tol_frac}")
            
            # Also try without dual (direct primal)
            if time.monotonic() < p4_deadline:
                a_lp2, sc_lp2 = _lp_chebyshev_refine(_best['seq'].copy(), p4_deadline,
                                                     max_iters=30, tol_frac=tol_frac, use_dual=False)
                upd(a_lp2, sc_lp2, f"lp_primal_{tol_frac}")
    
    # === Phase 5: L-BFGS after LP (200s) ===
    print(f"\n[Phase 5] L-BFGS after LP")
    if time.monotonic() < deadline - 150:
        p5_deadline = min(deadline - 150, time.monotonic() + 200)
        a_post, sc_post = _lbfgs_lse_improved(_best['seq'].copy(), p5_deadline,
                                               min_remaining=150, n_stages=15)
        upd(a_post, sc_post, "lbfgs_post_lp")
    
    # === Phase 6: More LP refinement (200s) ===
    print(f"\n[Phase 6] LP refinement round 2")
    if time.monotonic() < deadline - 100:
        p6_deadline = min(deadline - 100, time.monotonic() + 200)
        for tol_frac in [0.002, 0.001, 0.0005]:
            if time.monotonic() > p6_deadline:
                break
            a_lp3, sc_lp3 = _lp_chebyshev_refine(_best['seq'].copy(), p6_deadline,
                                                 max_iters=40, tol_frac=tol_frac, use_dual=True)
            upd(a_lp3, sc_lp3, f"lp_r2_{tol_frac}")
    
    # === Phase 7: Subgradient polish (100s) ===
    print(f"\n[Phase 7] Subgradient polish")
    if time.monotonic() < deadline - 30:
        p7_deadline = min(deadline - 30, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(_best['seq'].copy(), p7_deadline,
                                      n_steps=10000, lr_init=0.005, tol_frac=0.0005)
        upd(a_sg, sc_sg, "sg_polish")
    
    # === Phase 8: Final aggressive LP + LSE (60s) ===
    print(f"\n[Phase 8] Final aggressive optimization")
    if time.monotonic() < deadline - 10:
        # Final ultra-low beta LSE
        n_fin = len(_best['seq'])
        c_fin = _full_convolve(_best['seq'], _best['seq'])
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best['seq'], 1e-30))
        z_fin = z_fin - z_fin.mean()
        
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.008)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass
        
        # Final LP
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _lp_chebyshev_refine(_best['seq'].copy(), deadline - 5,
                                                   max_iters=30, tol_frac=0.001, use_dual=True)
            upd(a_fin2, sc_fin2, "final_lp")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best['seq'] is not None:
        final = np.clip(_best['seq'], 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            print("FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
