"""
Optimized solver for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-scale optimization, aggressive log-sum-exp L-BFGS, adaptive active-set Newton,
and post-Newton coordinate descent. Key improvements over previous: tighter convergence,
adaptive active sets, multi-resolution warm starts, and parameterized family exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_newton(a_init, deadline, max_iter=500, tol_active=0.0005, max_top_frac=0.3):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        max_top = int(max_top_frac * (2 * n - 1))
        if n_active > max_top:
            n_active = max_top
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True
        m = n_active

        A_vec = active.astype(np.float64)

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        mu = 1e-7 * (c_max / max(m, 1)) + 1e-9

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            rhs_norm = np.linalg.norm(rhs)
            if rhs_norm < 1e-15:
                return x
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-14 * (1.0 + rhs_norm):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-12:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-14 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - f_old) < 1e-14 * f_old:
            break

    return best_a, best_sc


def _coord_descent(a_init, deadline, max_rounds=200, n_coords=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c_max = float(c.max())
        tol = max(0.001 * c_max, 0.0005 * c_max)

        n_active = int(active.sum())
        if n_active == 0:

        c_new = _full_convolve(a, a)

def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0, aggressive=True):

                                      0.03, 0.015, 0.007, 0.003, 0.001,

    current_a = a_init.copy()

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.03):

def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + t
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.3) ** 2) / 0.03) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.03)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        a = np.ones(n)
        a[n // 4:3 * n // 4] *= 1.5
        return a / a.sum()
    elif kind == "quasiperiodic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 7 * t) + 0.1 * np.sin(2 * np.pi * 13 * t + 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrink_and_reopt(a, deadline, max_candidates=10):

    print(f"\n[Phase 1] Multi-scale warm start (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_ms = _multi_scale_optimize(deadline - 400, n_target, min_time_remaining=200.0)
        if a_ms is not None:
            update_best(a_ms, _score(a_ms), "multi_scale")

    # Phase 2: Aggressive L-BFGS from best
    print(f"\n[Phase 2] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_main = _lbfgs_smooth(best_seq, deadline - 300, min_time_remaining=200.0, aggressive=True)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 3: Active-set Newton (loose)
    print(f"\n[Phase 3] Active-set Newton loose (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 100, time.monotonic() + 120),
                                         max_iter=300, tol_active=0.005)
        update_best(a_as, sc_as, "as_loose")

    # Phase 4: L-BFGS again from improved
    print(f"\n[Phase 4] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_r2 = _lbfgs_smooth(best_seq, deadline - 100, min_time_remaining=60.0, aggressive=True)
        update_best(a_r2, _score(a_r2), "lbfgs_r2")

    # Phase 5: Active-set Newton (tight)
    print(f"\n[Phase 5] Active-set Newton tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 40, time.monotonic() + 100),
                                           max_iter=400, tol_active=0.001)
        update_best(a_as2, sc_as2, "as_tight")

    # Phase 6: Coordinate descent
    print(f"\n[Phase 6] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_cd, sc_cd = _coord_descent(best_seq, min(deadline - 30, time.monotonic() + 80),
                                     max_rounds=300, n_coords=1000)
        update_best(a_cd, sc_cd, "coord_descent")

    # Phase 7: Mirror descent polish
    print(f"\n[Phase 7] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_md, sc_md = _mirror_descent(best_seq, min(deadline - 15, time.monotonic() + 60),
                                      max_steps=3000, lr_init=0.02)
        update_best(a_md, sc_md, "mirror_descent")

    # Phase 8: Shrinking
    print(f"\n[Phase 8] Shrinking (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sh, sc_sh = _shrink_and_reopt(best_seq, min(deadline - 10, time.monotonic() + 30),
                                        max_candidates=8)
        update_best(a_sh, sc_sh, "shrink")
        if sc_sh < best_score + 1e-10 and time.monotonic() < deadline - 15:
            a_sh2, sc_sh2 = _active_set_newton(a_sh, min(deadline - 5, time.monotonic() + 20),
                                                max_iter=100, tol_active=0.001)
            update_best(a_sh2, sc_sh2, "shrink_as")

    # Phase 9: Diverse restarts
    print(f"\n[Phase 9] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _lbfgs_smooth(p_start, deadline, min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{sigma}")
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 8:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 3, time.monotonic() + 15),
                                                  max_iter=100, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{sigma}_as")

    # Phase 10: Final tight active-set Newton
    print(f"\n[Phase 10] Final AS Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1, max_iter=500, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _coord_descent(a_fin, deadline - 0.5, max_rounds=100, n_coords=500)
            update_best(a_fin2, sc_fin2, "final_cd")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
