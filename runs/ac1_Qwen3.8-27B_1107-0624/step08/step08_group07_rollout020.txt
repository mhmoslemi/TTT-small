def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

            c_best = c_try.copy()
            best_val = val

            T_smooth = c_max * smooth_T_frac * (1 - frac * 0.5)
            g = _smoothed_gradient(a, c, T_smooth)
        else:
            threshold = c_max * (1 - tol_frac)
            active = np.where(c >= threshold)[0]
            if len(active) > top_k:
                sorted_a = active[np.argsort(c[active])[::-1][:top_k]]
                active = sorted_a
            elif len(active) == 0:
                active = np.array([np.argmax(c)])
            g = _subgradient_direct(a, c, active)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > 1.5:
            update = update * (1.5 / update_norm)

        z_new = z - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        # Backtracking check every few steps
        if t % (eval_interval * 3) == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_unit(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z.copy()
                m_buf *= 0.3
                v_buf *= 0.3

        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_unit(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

            if patience > 15 and patience > best_patience:
                best_patience = patience
                # Perturb and restart
                sigma = 0.003 + 0.002 * (patience // 20)
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                z += sigma * np.random.randn(n)
                z = np.clip(z, -60.0, 60.0)
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0
                best_patience = 0

    a_final = _softmax_stable(z)
    sc_final = _score_unit(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _line_search_polish(a_init, deadline, max_rounds=500, top_k=5, tol_frac=0.005):
    """Subgradient with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = _score_unit(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        threshold = c_max * (1 - tol_frac)
        active = np.where(c >= threshold)[0]
        if len(active) > top_k:
            sorted_a = active[np.argsort(c[active])[::-1][:top_k]]
            active = sorted_a
        elif len(active) == 0:
            active = np.array([np.argmax(c)])

        g = _subgradient_direct(a, c, active)
        g_tan = g - np.mean(g)

        direction = -g_tan
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        direction /= dir_norm

        best_val = c_max
        best_a_try = a.copy()
        step = 0.05
        for _ in range(25):
            a_try = a + step * direction
            a_try = np.maximum(0, a_try)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try /= s_try
            c_try = _fft_convolve(a_try, a_try)
            val = float(c_try.max())
            if val < best_val:
                best_val = val
                best_a_try = a_try.copy()
            step *= 0.6
            if step < 1e-10:
                break

        if best_val < c_max * (1 - 1e-9):
            a = best_a_try
            sc = 2.0 * n * best_val
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # Try with single peak
            active_single = np.array([np.argmax(c)])
            g_s = _subgradient_direct(a, c, active_single)
            g_tan_s = g_s - np.mean(g_s)
            dir_s = -g_tan_s
            ds = np.linalg.norm(dir_s)
            if ds < 1e-15:
                break
            dir_s /= ds
            best_val_s = c_max
            step = 0.03
            for _ in range(20):
                a_try = a + step * dir_s
                a_try = np.maximum(0, a_try)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    step *= 0.5
                    continue
                a_try /= s_try
                c_try = _fft_convolve(a_try, a_try)
                val = float(c_try.max())
                if val < best_val_s:
                    best_val_s = val
                    best_a_try = a_try.copy()
                step *= 0.6
                if step < 1e-10:
                    break
            if best_val_s < c_max * (1 - 1e-9):
                a = best_a_try
                sc = 2.0 * n * best_val_s
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            else:
                break

    return best_a, best_score


def _active_set_equalization(a_init, deadline, max_iter=2000, tol=0.003, max_active=12):
    """Equalize active peaks using minimum-norm correction."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = _score_unit(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        threshold = c_max * (1 - tol)
        active = np.where(c >= threshold)[0]
        if len(active) < 2:
            break
        if len(active) > max_active:
            sorted_a = active[np.argsort(c[active])[::-1][:max_active]]
            active = sorted_a

        K = len(active)
        # Build Jacobian rows: J[i,j] = 2*a[k_i - j] if valid
        M = np.zeros((K + 1, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        M[K, :] = 1.0  # sum constraint row

        # Target: make all c_k equal to mean, keep sum = 0
        c_active = c[active]
        target = np.mean(c_active)
        b = np.concatenate([target - c_active, [0.0]])

        # Minimum norm: d = M^T (M M^T)^{-1} b
        G = M @ M.T
        try:
            G_inv_b = np.linalg.solve(G + 1e-10 * np.eye(G.shape[0]), b)
            d = M.T @ G_inv_b
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M, b, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d /= d_norm

        # Try multiple step sizes
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s_cand = a_cand.sum()
            if s_cand < 1e-15:
                continue
            a_cand /= s_cand
            c_cand = _fft_convolve(a_cand, a_cand)
            sc = float(c_cand.max()) * 2.0 * n
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _coordinate_polish(a_init, deadline, max_rounds=200):
    """Targeted coordinate descent on elements affecting the peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = _score_unit(a)
    c = _fft_convolve(a, a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        # Try modifying contributing elements
        for j in range(lo, hi + 1, max(1, (hi - lo + 1) // 50)):
            old_val = a[j]
            if old_val < 1e-12:
                continue
            for scale in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_try = a.copy()
                a_try[j] = old_val * scale
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try /= s_try
                c_try = _fft_convolve(a_try, a_try)
                sc = float(c_try.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    c = c_try
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try broader search
            any_improve = False
            step_size = max(1, n // 100)
            for j in range(0, n, step_size):
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                for scale in [0.5, 0.75, 1.3, 1.6]:
                    a_try = a.copy()
                    a_try[j] = old_val * scale
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try /= s_try
                    c_try = _fft_convolve(a_try, a_try)
                    sc = float(c_try.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_try.copy()
                        a = a_try
                        c = c_try
                        any_improve = True
                        improved = True
                        break
                if improved:
                    break
            if not any_improve:
                break

    return best_a, best_score


def _get_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []

    # Parent
    if parent is not None and parent.sum() > 0.01:
        p = parent.copy()
        p = np.maximum(0, p)
        p = p / p.sum()
        inits.append(('parent', p))

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Exponential
    x = rng.exponential(1.0, n)
    inits.append(('exp', x / x.sum()))

    # Power law
    j = np.arange(n) + 1
    x = j ** (-0.3)
    inits.append(('powerlaw', x / x.sum()))

    # Cosine modulated
    x = 1.0 + 0.3 * np.cos(2 * np.pi * j / n)
    x = np.maximum(x, 0.01)
    inits.append(('cosine', x / x.sum()))

    # Two-lobe
    x = np.exp(-((j - n * 0.25) ** 2) / (n * 0.05)) + np.exp(-((j - n * 0.75) ** 2) / (n * 0.05))
    inits.append(('two_lobe', x / x.sum()))

    # Sparse random
    x = np.zeros(n)
    num_sparse = n // 4
    idx = rng.choice(n, num_sparse, replace=False)
    x[idx] = rng.uniform(0.5, 1.0, num_sparse)
    inits.append(('sparse', x / x.sum()))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: diverse initialization -> Adam -> line search polish
    -> active set equalization -> coordinate descent -> diversification restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() <= 0.01:
            parent = None
    except Exception:
        parent = None

    n = 8192

    # Generate initializations
    inits = _get_initializations(n, parent, rng)
    print(f"Generated {len(inits)} initializations, n={n}")

    # Phase 1: Quick evaluation of all inits (0-30s)
    phase1_end = min(deadline, t_start + 30.0)
    best_a = None
    best_score = np.inf
    best_name = ''
    for name, init in inits:
        if time.monotonic() > phase1_end:
            break
        sc = _score_unit(init)
        print(f"  Init '{name}': score={sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_a = init.copy()
            best_name = name
    print(f"Phase 1 best: '{best_name}' score={best_score:.10f}")

    # Phase 2: Adam optimization from best init (30-400s)
    phase2_end = min(deadline, t_start + 400.0)
    print(f"Phase 2: Adam optimization, {phase2_end - t_start:.0f}s budget")
    c_best = _fft_convolve(best_a, best_a)
    max_cb = float(c_best.max())

    # Run multiple Adam passes with decreasing temperature
    n_passes = 3
    time_per_pass = (phase2_end - time.monotonic()) / n_passes
    for p in range(n_passes):
        sub_dl = min(deadline, time.monotonic() + time_per_pass)
        T_s = max_cb * [0.02, 0.005, 0.001][p]
        T_e = T_s * 0.01
        lr_s = [0.04, 0.02, 0.01][p]
        lr_e = [0.002, 0.001, 0.0003][p]

        a_opt, sc_opt = _adam_optimize(
            a_init=best_a, deadline=sub_dl,
            T_start=T_s, T_end=T_e, lr_start=lr_s, lr_end=lr_e,
            max_iter=500000, top_k=6, tol_frac=0.008,
            eval_interval=40, use_smoothed=(p == 0), smooth_T_frac=0.05
        )
        full_sc = _score_unit(a_opt)
        print(f"  Adam pass {p}: score={full_sc:.10f} (T={T_s:.6f}), t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())

    # Phase 3: Line search polish (400-700s)
    phase3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Line search polish")
        a_ls, sc_ls = _line_search_polish(best_a, phase3_end,
                                           max_rounds=2000, top_k=6, tol_frac=0.005)
        full_sc_ls = _score_unit(a_ls)
        print(f"  Line search: score={full_sc_ls:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_ls < best_score:
            best_score = full_sc_ls
            best_a = a_ls.copy()

    # Phase 4: Active set equalization (700-950s)
    phase4_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Active set equalization")
        a_as, sc_as = _active_set_equalization(best_a, phase4_end,
                                                max_iter=3000, tol=0.002, max_active=12)
        full_sc_as = _score_unit(a_as)
        print(f"  Active set: score={full_sc_as:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_as < best_score:
            best_score = full_sc_as
            best_a = a_as.copy()

        # Second round of line search after equalization
        if time.monotonic() < phase4_end - 30:
            a_ls2, sc_ls2 = _line_search_polish(best_a, phase4_end - 10,
                                                 max_rounds=1000, top_k=6, tol_frac=0.004)
            full_sc_ls2 = _score_unit(a_ls2)
            if full_sc_ls2 < best_score:
                best_score = full_sc_ls2
                best_a = a_ls2.copy()

    # Phase 5: Second Adam round with lower temperature (950-1200s)
    phase5_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Refined Adam (low T)")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_opt2, sc2 = _adam_optimize(
            a_init=best_a, deadline=phase5_end,
            T_start=max_cb * 0.002, T_end=max_cb * 0.00005,
            lr_start=0.008, lr_end=0.0002,
            max_iter=800000, top_k=8, tol_frac=0.005,
            eval_interval=30
        )
        full_sc2 = _score_unit(a_opt2)
        print(f"  Refined Adam: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # Phase 6: Coordinate descent (1200-1400s)
    phase6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase6_end:
        print(f"Phase 6: Coordinate descent polish")
        a_cd, sc_cd = _coordinate_polish(best_a, phase6_end, max_rounds=300)
        full_sc_cd = _score_unit(a_cd)
        print(f"  Coord descent: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # Phase 7: Diversification restarts (1400-1700s)
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end:
        print(f"Phase 7: Diversification restarts")
        n_restarts = 6
        time_per_restart = max(15.0, (phase7_end - time.monotonic()) / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > phase7_end - time_per_restart * 0.5:
                break

            # Different perturbation strategies
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()

            if r == 0:
                # Small perturbation
                z += 0.005 * rng.standard_normal(n)
            elif r == 1:
                # Medium perturbation
                z += 0.02 * rng.standard_normal(n)
            elif r == 2:
                # Structured perturbation (low frequency)
                freq = rng.integers(1, 20)
                z += 0.01 * np.cos(2 * np.pi * freq * np.arange(n) / n)
            elif r == 3:
                # Sparse perturbation
                n_pert = n // 10
                idx = rng.choice(n, n_pert, replace=False)
                z[idx] += 0.05 * rng.randn(n_pert)
            elif r == 4:
                # Scale perturbation (multiply by random positive)
                z += 0.03 * rng.standard_normal(n)
            else:
                # Very large perturbation
                z += 0.1 * rng.standard_normal(n)

            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            a_pert = _softmax_stable(z)

            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())

            sub_dl = min(phase7_end - 5, time.monotonic() + time_per_restart * 0.7)
            a_pert_opt, sc_pert = _adam_optimize(
                a_init=a_pert, deadline=sub_dl,
                T_start=max_cp * 0.003, T_end=max_cp * 0.00005,
                lr_start=0.015, lr_end=0.0005,
                max_iter=100000, top_k=6, tol_frac=0.006,
                eval_interval=50
            )
            full_sc = _score_unit(a_pert_opt)
            print(f"  Restart[{r}]: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 8: Final polish (1700-deadline)
    if time.monotonic() < deadline - 20:
        print(f"Phase 8: Final polish")
        # Quick line search
        a_fin, sc_fin = _line_search_polish(best_a, deadline - 30,
                                             max_rounds=500, top_k=8, tol_frac=0.003)
        full_sc_fin = _score_unit(a_fin)
        print(f"  Final line search: score={full_sc_fin:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_fin < best_score:
            best_score = full_sc_fin
            best_a = a_fin.copy()

        # Quick active set
        if time.monotonic() < deadline - 15:
            a_as_f, sc_as_f = _active_set_equalization(best_a, deadline - 10,
                                                        max_iter=1000, tol=0.001, max_active=10)
            full_sc_asf = _score_unit(a_as_f)
            if full_sc_asf < best_score:
                best_score = full_sc_asf
                best_a = a_as_f.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) * 0.01

    sc_check = _score(final)
    print(f"FINAL: score={sc_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
