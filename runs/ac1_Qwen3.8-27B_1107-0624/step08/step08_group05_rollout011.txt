Optimized step function autocorrelation minimizer using:

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho_last = rho[-1]
    theta = (cv[rho_last] - 1.0) / (rho_last + 1.0)
    w = np.maximum(v - theta, 0.0)
    ws = w.sum()
    if ws < 1e-300:
        return np.full(n, target / n)
    return w * (target / ws)


def _active_subgrad(a):
    """Compute subgradient of max_k (a*a)_k w.r.t. a. O(n) for single active lag."""
    n = len(a)
    b = _full_convolve(a, a)
    b_max = float(b.max())
    tol = max(0.001 * b_max, 1e-14)
    active = b >= (b_max - tol)
    n_active = active.sum()
    if n_active == 0:
        idx = int(np.argmax(b))
        grad = 2.0 * np.zeros(n)
        shift = idx
        lo = max(0, shift - (n - 1))
        hi = min(n, shift + 1)
        for i in range(lo, hi):
            j = shift - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        return grad, b_max, 1
    # For multiple active lags, compute gradient as weighted sum
    # grad_i = 2 * sum_l alpha_l * a[k_l - i]
    # Use FFT for efficiency if n_active is large
    if n_active < 50:
        active_indices = np.nonzero(active)[0]
        weights = np.ones(n_active) / n_active
        grad = np.zeros(n)
        for idx, w in zip(active_indices, weights):
            lo = max(0, idx - (n - 1))
            hi = min(n, idx + 1)
            for i in range(lo, hi):
                j = idx - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a[j]
    else:
        h = np.zeros(2 * n - 1)
        h[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(h, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad, b_max, n_active


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    val = np.sum(np.power(c_safe, p)) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    sum_cp = float(np.sum(np.power(c_safe, p)))
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result
