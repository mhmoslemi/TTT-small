"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, FISTA (Nesterov momentum),
aggressive active-set gradient descent, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoothProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoothed_obj(z, self.beta, self.n)

    def grad(self, z):
        return _smoothed_grad(z, self.beta, self.n)


def _fista(z0, beta, n, max_iter=10000, deadline=None):
    """Nesterov accelerated gradient descent on smoothed objective."""
    z = z0.copy()
    z_prev = z0.copy()
    t = 1.0
    # Estimate initial L
    g0 = _smoothed_grad(z, beta, n)
    gnorm = np.linalg.norm(g0)
    znorm = np.linalg.norm(z)
    L = max(1.0, gnorm / max(1e-10, znorm) * 10)
    
    f_curr = _smoothed_obj(z, beta, n)
    
    for k in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t_prev - 1.0) / t
        y = z + momentum * (z - z_prev)

        gy = _smoothed_grad(y, beta, n)

        # Line search with backtracking
        f_y = _smoothed_obj(y, beta, n)
        for ls in range(25):
            z_new = y - gy / L
            diff = z_new - y
            model = f_y + np.dot(gy, diff) + 0.5 * L * np.dot(diff, diff)
            f_new = _smoothed_obj(z_new, beta, n)
            if f_new <= model + 1e-10:
                break
            L *= 2.0
        else:
            # Fallback: just take the step
            z_new = y - gy / L

        z_prev = z
        z = z_new

    return z


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])

    best_a = a_init.copy()
    best_sc = _score(a_init)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 15))))

        prob = _SmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception as e:
            print(f"  L-BFGS error at beta={beta:.6f}: {e}")

    return best_a, best_sc


def _fista_phase(a_init, deadline, min_time_remaining=30.0):
    """FISTA with beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Beta schedule
    betas = [max_c * f for f in [0.01, 0.005, 0.0025, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        max_iter = min(20000, max(100, int(remaining / 0.015)))
        
        z = _fista(z, beta, n, max_iter=max_iter, deadline=min(deadline - min_time_remaining, time.monotonic() + remaining * 0.9))
        
        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    FISTA beta={beta:.6f}: score={best_sc:.10f}")
    
    return best_a, best_sc


def _active_set_momentum(a_init, deadline, min_time_remaining=30.0):
    """Active-set gradient descent with momentum and adaptive learning rate."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.01
    momentum_coeff = 0.9
    v = np.zeros(n)  # momentum buffer
    tol_frac_start = 0.01
    tol_frac = tol_frac_start
    no_improve = 0
    stall_count = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max  # since sum(a) = 1
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Adaptively tighten tolerance
            if tol_frac > 0.0001:
                tol_frac = max(0.0001, tol_frac * 0.99)
        else:
            no_improve += 1
        
        if no_improve > 100:
            lr *= 0.8
            if no_improve > 300:
                stall_count += 1
                if stall_count > 10:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                no_improve = 0
                lr = 0.01
                tol_frac = max(0.0001, tol_frac * 0.5)
        
        # Find active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Compute gradient (subgradient of max, averaged over active set)
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h  # gradient of max_c w.r.t. a (approximately)
        
        # Center the gradient (maintain sum = 1 in a multiplicative way)
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        
        if gnorm < 1e-18:
            break
        
        # Momentum update
        v = momentum_coeff * v - lr * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < current_sc - 1e-14:
            a = a_new
            # Increase learning rate slightly
            lr = min(0.05, lr * 1.01)
        else:
            # Backtrack
            lr *= 0.7
            v = momentum_coeff * v
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
    
    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        # Flat in the middle, tapered at edges
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper_w = 0.1
        for i in range(n):
            if t[i] < taper_w:
                a[i] = 0.3 + 0.7 * t[i] / taper_w
            elif t[i] > 1 - taper_w:
                a[i] = 0.3 + 0.7 * (1 - t[i]) / taper_w
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
