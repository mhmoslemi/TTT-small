LSE-smoothed L-BFGS, top-k active-set refinement, paired coordinate descent,
Fourier-domain perturbations, and adaptive multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities (top-level, no closures)
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (t - 0.5) ** 2 * 4.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.3 * np.exp(-3.0 * (1.0 - t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

# ============================================================
# LSE objective and gradient
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

def _topk_lse_objective(z, k_top, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if k_top >= len(c):
        idx = slice(None)
    else:
        top_vals = np.sort(c)[::-1][:k_top]
        threshold = top_vals[-1]
        idx = c >= threshold
    c_sub = c[idx]
    c_max = float(c_sub.max())
    shifted = (c_sub - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _topk_lse_gradient(z, k_top, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if k_top >= len(c):
        idx = np.ones_like(c, dtype=bool)
    else:
        top_vals = np.sort(c)[::-1][:k_top]
        threshold = top_vals[-1]
        idx = c >= threshold
    c_sub = c[idx]
    c_max = float(c_sub.max())
    shifted = (c_sub - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w_sub = np.zeros_like(c_sub)
        w_sub[int(np.argmax(c_sub))] = 1.0
    else:
        w_sub = e / s
    w = np.zeros_like(c)
    w[idx] = w_sub
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

# ============================================================
# Newton active-set method
# ============================================================

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

# L-BFGS wrapper with adaptive LSE
# ============================================================

def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=None):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    if n_stages is None:
        n_stages = 16
    betas = base_beta * np.array([1.0, 0.7, 0.45, 0.25, 0.15, 0.09,
                                   0.05, 0.03, 0.018, 0.01, 0.006,
                                   0.003, 0.0015, 0.0008, 0.0004,
                                   0.0002])[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

def _lbfgs_topk(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    configs = [(50, 0.005), (20, 0.003), (10, 0.002), (5, 0.001), (3, 0.0005), (2, 0.0003)]

    for k_top, beta_frac in configs:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = max_c * beta_frac
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _topk_lse_objective, z, args=(k_top, beta, n),
                jac=_topk_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _paired_cd(a_init, deadline, n_pairs=5000, tol_frac=0.003):
    """Paired coordinate descent: adjust pairs (i,j) while preserving sum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    t_start = time.monotonic()

    # Focus on pairs contributing to the peak
    c_max = float(c.max())
    peak_idx = int(np.argmax(c))

    rng = np.random.default_rng(12345)
    step = 0

    while step < n_pairs and time.monotonic() < deadline:
        # Choose pair: bias toward pairs contributing to peak
        if step % 3 == 0:
            # Pair summing to peak index
            i = int(rng.integers(0, min(peak_idx + 1, n)))
            j = peak_idx - i
            if j < 0 or j >= n or j == i:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
        elif step % 3 == 1:
            # Adjacent-ish pair
            i = int(rng.integers(0, n - 1))
            j = i + 1
        else:
            # Random pair
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

        if i == j or i >= n or j >= n:
            step += 1
            continue

        pair_sum = a[i] + a[j]
        if pair_sum < 1e-15:
            step += 1
            continue

        # Find optimal t = a[i] (with a[j] = pair_sum - t)
        # Minimize max of affected c values
        best_t = a[i]
        best_obj = np.inf

        # Try a grid of t values
        # The objective changes only in a bounded range of t
        # Use a coarse-to-fine approach
        for lo_frac, hi_frac in [(0.0, 1.0), (0.2, 0.8)]:
            lo = lo_frac * pair_sum
            hi = hi_frac * pair_sum
            if lo >= hi:
                continue
            n_grid = 8
            for t in np.linspace(lo, hi, n_grid):
                a[i] = t
                a[j] = pair_sum - t
                # Compute affected c values quickly
                # Only recompute affected indices
                max_c_try = 0.0
                for k in range(max(0, i - (n-1)), min(2*n-1, i + n)):
                    pass  # too slow for Python loop
                # Use full convolve for simplicity (slower but correct)
                c_try = _full_convolve(a, a)
                obj = float(c_try.max())
                if obj < best_obj:
                    best_obj = obj
                    best_t = t

        a[i] = best_t
        a[j] = pair_sum - best_t
        c = _full_convolve(a, a)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        step += 1

    return best_a, best_sc

def _paired_cd_fast(a_init, deadline, n_iter=2000):
    """Faster paired CD using precomputed structure."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    rng = np.random.default_rng(67890)

    for step in range(n_iter):
        if time.monotonic() > deadline:
            break
        if step % 100 == 0:
            c = _full_convolve(a, a)

        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Choose pair
        if step % 4 == 0:
            i = int(rng.integers(0, n))
            j = min(2 * (n - 1) - i, n - 1) if 2*(n-1) - i < n else int(rng.integers(0, n))
            if i == j:
                j = (i + 1) % n
        elif step % 4 == 1:
            # Target peak
            i = int(rng.integers(0, min(peak_idx, n - 1) + 1))
            j = peak_idx - i
            if j < 0 or j >= n or j == i:
                j = (i + n // 2) % n
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (i + 1) % n

        pair_sum = a[i] + a[j]
        if pair_sum < 1e-12:
            continue

        # Try small perturbation around current a[i]
        delta = pair_sum * 0.05
        candidates = [a[i] + d * delta for d in [-1, -0.5, 0.5, 1.0]]
        candidates = [max(0, min(d, pair_sum)) for d in candidates]
        best_t = a[i]
        best_obj = np.inf

        for t in candidates:
            a[i] = t
            a[j] = pair_sum - t
            # Quick score check via affected convolution values
            c_partial = _full_convolve(a, a)
            obj = float(c_partial.max())
            if obj < best_obj:
                best_obj = obj
                best_t = t

        a[i] = best_t
        a[j] = pair_sum - best_t

        if step % 50 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

# Perturbation methods
# ============================================================

def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

def _perturb_cross(a, rng, sigma=0.03):
    """Cross-perturb: swap mass between symmetric positions."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    n_swaps = n // 50
    for _ in range(n_swaps):
        i = int(rng.integers(0, n // 2))
        j = n - 1 - i
        swap = sigma * rng.standard_normal()
        z[i] += swap
        z[j] -= swap
    return _softmax(z)

# ============================================================
# Annealing-based search
# ============================================================

def _anneal_search(a_init, deadline, min_remaining=60.0):
    """Simulated annealing style: occasionally large jumps."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(99999)

    t_start = time.monotonic()
    total_time = deadline - t_start
    if total_time < 10:
        return best_a, best_sc

    current_temp = 0.05
    min_temp = 0.001
    n_steps = min(200, int(total_time / 1.5))

    for step in range(n_steps):
        if time.monotonic() > deadline - min_remaining:
            break
        progress = step / max(1, n_steps)
        temp = current_temp * (min_temp / current_temp) ** progress

        # Random perturbation
        sigma = temp
        a_try = _perturb_log(a, sigma, rng)
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        elif sc_try < best_sc * (1.0 + temp * 2):
            # Accept with some probability (annealing)
            if rng.random() < 0.3:
                a = a_try

        # After each step, do a quick LSE refinement
        if step % 5 == 0 and time.monotonic() < deadline - min_remaining - 20:
            z = np.log(np.maximum(a, 1e-30))
            z = z - z.mean()
            c_cur = _full_convolve(a, a)
            beta = float(c_cur.max()) * 0.01
            try:
                result = minimize(
                    _lse_objective, z, args=(beta, n),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-14, 'maxcor': 30}
                )
                a_refined = _softmax(result.x)
                sc_refined = _score(a_refined)
                if sc_refined < best_sc:
                    best_sc = sc_refined
                    best_a = a_refined.copy()
                if sc_refined < _score(a):
                    a = a_refined
            except Exception:
                pass

    return best_a, best_sc

# ============================================================
# Main optimization function
# ============================================================

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-stage optimizer: LSE L-BFGS, top-k active set, Newton refinement,
    paired coordinate descent, annealing, and multi-n exploration.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (30s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 30)
    n_expl = max(2048, n_parent // 8) if n_parent > 4096 else 2048
    profiles = ["uniform", "tapered", "cos2", "wheatstone", "gauss",
                "bimodal", "u_shape", "asym"]
    for kind in profiles:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 2: Main LSE L-BFGS (450s) ===
    print(f"\n[Phase 2] Main LSE L-BFGS")
    if time.monotonic() < deadline - 450:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=400, n_stages=16)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 3: Top-k LSE (200s) ===
    print(f"\n[Phase 3] Top-k LSE")
    if time.monotonic() < deadline - 200:
        a_topk, sc_topk = _lbfgs_topk(_best_seq.copy(), deadline, min_remaining=180)
        upd(a_topk, sc_topk, "topk_lse")

    # === Phase 4: Newton active-set (180s) ===
    print(f"\n[Phase 4] Newton active-set")
    if time.monotonic() < deadline - 150:
        nw_dead = min(deadline - 130, time.monotonic() + 180)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 5: Second LSE pass with refined start (200s) ===
    print(f"\n[Phase 5] Second LSE pass")
    if time.monotonic() < deadline - 150:
        a_lse2, sc_lse2 = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=130, n_stages=12)
        upd(a_lse2, sc_lse2, "lse_pass2")

    # === Phase 6: Paired coordinate descent (250s) ===
    print(f"\n[Phase 6] Paired coordinate descent")
    if time.monotonic() < deadline - 100:
        cd_dead = min(deadline - 80, time.monotonic() + 250)
        a_cd, sc_cd = _paired_cd_fast(_best_seq, cd_dead, n_iter=1500)
        upd(a_cd, sc_cd, "paired_cd")

        # Newton polish after CD
        if sc_cd <= _best_score + 1e-10 and time.monotonic() < deadline - 60:
            nw_dead2 = min(deadline - 50, time.monotonic() + 60)
            a_nw2, sc_nw2 = _newton_active_set(_best_seq, nw_dead2, max_iter=100, tol_frac=0.002)
            upd(a_nw2, sc_nw2, "newton_after_cd")

    # === Phase 7: Subgradient polish (100s) ===
    print(f"\n[Phase 7] Subgradient polish")
    if time.monotonic() < deadline - 60:
        sg_dead = min(deadline - 40, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=3000, lr_init=0.006, tol_frac=0.0008)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 8: Multi-n exploration (250s) ===
    print(f"\n[Phase 8] Multi-n exploration")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates and n_parent >= 2048:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 40:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        # Quick LSE
        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, deadline, min_remaining=30, n_stages=8)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        # Newton polish if promising
        if sc_opt_n < _best_score * 1.002 and time.monotonic() < deadline - 25:
            nw_dead = min(deadline - 20, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=60, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 9: Diverse restarts with annealing (200s) ===
    print(f"\n[Phase 9] Diverse restarts")
    if time.monotonic() < deadline - 50:
        restart_configs = [
            (lambda: _perturb_log(_best_seq, 0.04, rng), "restart_log"),
            (lambda: _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3), "restart_fourier"),
            (lambda: _perturb_block_swap(_best_seq, rng, n_swaps=4), "restart_block"),
            (lambda: _perturb_amplitude(_best_seq, rng, sigma=0.1), "restart_amp"),
            (lambda: _perturb_cross(_best_seq, rng, sigma=0.05), "restart_cross"),
        ]
        for gen_fn, label in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=20, n_stages=6)
            upd(a_r, sc_r, label)

            if sc_r < _best_score * 1.003 and time.monotonic() < deadline - 15:
                nw_dead = min(deadline - 12, time.monotonic() + 30)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, label + "_newton")

    # === Phase 10: Annealing search (100s) ===
    print(f"\n[Phase 10] Annealing search")
    if time.monotonic() < deadline - 30:
        ann_dead = min(deadline - 25, time.monotonic() + 100)
        a_ann, sc_ann = _anneal_search(_best_seq, ann_dead, min_remaining=20)
        upd(a_ann, sc_ann, "anneal")

    # === Phase 11: Final aggressive polish (50s) ===
    print(f"\n[Phase 11] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-14, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=80, tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient with very tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=2000, lr_init=0.003, tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
