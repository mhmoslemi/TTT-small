The optimal solution balances these: it wants to make q_{n-1} as small as possible while keeping the other q_k values from getting too large.

"""
Multi-strategy optimizer for step function convolution minimization.
Uses primal-dual multiplicative weights, L-BFGS with adaptive smoothing,
active-set QP, palindromic projection, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "fejer":
        t = np.linspace(0, 1, n)
        a = np.maximum(np.sin(np.pi * t) / (np.pi * t), 0.01)
        a = a ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _palindromize(a):
    """Project onto palindromic sequences."""
    n = len(a)
    a_p = 0.5 * (a + a[::-1])
    a_p = np.maximum(a_p, 0)
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s


def _hankel_matvec(v, h, n):
    """Compute H*v where H is the n×n Hankel matrix with symbol h (length 2n-1).
    H[i,j] = h[i+j]"""
    v_rev = v[::-1]
    out_len = len(v_rev) + len(h) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(v_rev, N)
    F_H = np.fft.rfft(h, N)
    conv = np.fft.irfft(F_V * F_H, N)[:out_len]
    # conv[k] = sum_i v_rev[i] * h[k-i] = sum_i v[n-1-i] * h[k-i]
    # We want (H*v)[j] = sum_i h[i+j] * v[i]
    # Let's verify: conv[j + n - 1] = sum_i v[n-1-i] * h[j+n-1-i]
    # Let m = n-1-i, so i = n-1-m:
    # = sum_m v[m] * h[j+n-1-(n-1-m)] = sum_m v[m] * h[j+m]
    # Yes! So (H*v) = conv[n-1 : 2n-1]
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(h, n, rhs, max_iter=500, tol=1e-12, reg=0.0):
    """Solve H*x = rhs where H is Hankel with symbol h, using CG.
    If reg > 0, solves (H + reg*I)*x = rhs."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, h, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        if rs_new > rs * 1.1:
            # CG breakdown, try steepest descent step
            alpha_sd = rs / pHp if pHp > 0 else 1e-10
            x_new = x + alpha_sd * p
            r_new = r - alpha_sd * Hp
            rs_new = float(r_new @ r_new)
            if rs_new >= rs:
                break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                           0.000001, 0.0000005, 0.0000002, 0.0000001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                           0.00002, 0.00001, 0.000005, 0.000002, 0.000001])

    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "fejer":
        t = np.linspace(0, 1, n)
        a = np.maximum(np.sin(np.pi * t) / (np.pi * t + 1e-15), 0.01)
        a = a ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _palindromize(a):
    n = len(a)
    a_p = 0.5 * (a + a[::-1])
    a_p = np.maximum(a_p, 0)
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s


def _hankel_matvec(v, h, n):
    v_rev = v[::-1]
    out_len = len(v_rev) + len(h) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(v_rev, N)
    F_H = np.fft.rfft(h, N)
    conv = np.fft.irfft(F_V * F_H, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(h, n, rhs, max_iter=500, tol=1e-12, reg=0.0):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, h, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        if abs(alpha) > 1e10:
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        if rs_new > rs * 1.05:
            alpha_sd = rs / max(pHp, 1e-30)
            x_new = x + alpha_sd * p
            r_new = r - alpha_sd * Hp
            rs_new = float(r_new @ r_new)
            if rs_new >= rs:
                break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


class _SmoProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def fun(self, z):
        return _smoobj(z, self.beta, self.n)

    def jac(self, z):
        return _smograd(z, self.beta, self.n)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                           0.000001, 0.0000005, 0.0000002, 0.0000001,
                           0.00000005, 0.00000002, 0.00000001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001])

    betas = base_beta * ratios
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 12))))

        try:
            prob = _SmoProblem(beta, n)
            result = minimize(
                prob.fun,
                z,
                jac=prob.jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _multiplicative_weights(a_init, deadline, max_iters=2000):
    """Primal-dual multiplicative weights for soft active set identification."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Initialize lambda uniformly
    n_lags = 2 * n - 1
    lam = np.ones(n_lags) / n_lags
    eta = 5.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Build Hankel symbol from lambda
        h = lam.copy()
        h_ext = np.zeros(2 * n - 1)
        h_ext[:n_lags] = h

        # Solve H*x = 1 where H[i,j] = h[i+j]
        ones = np.ones(n)
        x = _cg_solve_hankel(h_ext, n, ones, max_iter=300, reg=1e-8)
        one_x = float(np.sum(x))
        if abs(one_x) < 1e-20 or not np.isfinite(one_x):
            # Fallback: use current a
            c = _full_convolve(a, a)
            reg_val = 1e-6
            x = _cg_solve_hankel(h_ext + reg_val * np.arange(2 * n - 1), n, ones,
                                  max_iter=200, reg=1e-6)
            one_x = float(np.sum(x))
            if abs(one_x) < 1e-20 or not np.isfinite(one_x):
                break

        a_new = x / one_x
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()

        # Update lambda based on convolution values
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Exponential update: concentrate on high lags
        log_lam = np.log(lam + 1e-300)
        log_lam += eta * (c - c.max()) / c_max
        log_lam -= log_lam.max()
        lam_new = np.exp(log_lam)
        lam_new = lam_new / lam_new.sum()
        lam = lam_new

        # Adaptive eta: decrease over time
        eta = 5.0 / (1.0 + 0.02 * it)

        # Convergence check
        if it % 20 == 0:
            c_check = _full_convolve(best_a, best_a)
            spread = c_check.max() - np.median(c_check)
            if spread < 1e-10 * c_check.max():
                break

    return best_a, best_sc


def _active_qp_polish(a_init, deadline, max_rounds=80):
    """Active-set QP with progressive tolerance tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                    0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                    0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 5

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Build weighted Hankel symbol
        h_ext = np.zeros(2 * n - 1)
        h_ext[active] = 1.0 / n_active

        # Solve: min a^T H a s.t. sum(a)=1, a>=0
        # KKT: 2*H*a = mu*1 - nu, a>=0, nu>=0, a*nu=0
        ones = np.ones(n)
        x = _cg_solve_hankel(h_ext, n, ones, max_iter=400, reg=0.0)
        one_x = float(np.sum(x))
        if abs(one_x) > 1e-20 and np.isfinite(one_x):
            a_qp = x / one_x
            a_qp = np.maximum(a_qp, 0)
            s_qp = a_qp.sum()
            if s_qp > 1e-15:
                a_qp = a_qp / s_qp
                sc_qp = _score(a_qp)
                if sc_qp < best_sc - 1e-14:
                    best_sc = sc_qp
                    best_a = a_qp.copy()
                    a = a_qp.copy()
                    continue

        # Fallback: projected gradient step
        grad = _hankel_matvec(a, h_ext, n)
        lam_val = np.mean(grad)
        grad_centered = grad - lam_val
        step = 0.5 / (np.max(np.abs(grad)) + 1e-15)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=300):
    """Direct convolution flattening via projected gradient on excess."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_target = float(np.sum(c)) / len(c)
        c_max = float(c.max())

        excess = np.maximum(c - c_target, 0)
        excess_max = float(excess.max())
        if excess_max < 1e-15:
            break

        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]

        step = 0.5 / (1.0 + 0.05 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a_new = 0.5 * a + 0.5 * a_new
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, max_steps=2000):
    """Subgradient with Nesterov momentum on log-sum-exp smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.05
    momentum = 0.85
    beta = 0.01 * _full_convolve(a_init, a_init).max()

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive beta
        beta = max(1e-8, c_max * 0.001 / (1.0 + 0.01 * i))

        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = z_prev + momentum * (z_new - z) + (1 - momentum) * (-step * grad)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    patience = 0
                    lr *= 0.7
                    if lr < 0.001:
                        break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.02, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: multiplicative weights, L-BFGS beta continuation,
    active-set QP, flattening, subgradient, multi-n exploration, restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS aggressive from parent
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=800.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs_p1")

    # Phase 1b: Palindromic projection + re-optimize
    print(f"\n[Phase 1b] Palindromic (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 700:
        a_pal = _palindromize(best_seq)
        sc_pal = _score(a_pal)
        if sc_pal < best_score + 0.01:
            a_pal_opt = _run_lbfgs(a_pal, deadline, min_time_remaining=700.0, aggressive=True)
            update_best(a_pal_opt, _score(a_pal_opt), "palindromic")

    # Phase 2: Multiplicative weights (new approach)
    print(f"\n[Phase 2] Multiplicative weights (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        t_end = min(deadline - 600, time.monotonic() + 150)
        a2, sc2 = _multiplicative_weights(best_seq, t_end, max_iters=1000)
        update_best(a2, sc2, "mult_weights")

    # Phase 3: Active-set QP
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        t_end = min(deadline - 500, time.monotonic() + 150)
        a3, sc3 = _active_qp_polish(best_seq, t_end, max_rounds=80)
        update_best(a3, sc3, "active_qp_1")

    # Phase 4: Flatten
    print(f"\n[Phase 4] Flatten (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 420:
        t_end = min(deadline - 420, time.monotonic() + 60)
        a4, sc4 = _flatten_polish(best_seq, t_end, max_iters=200)
        update_best(a4, sc4, "flatten_1")

    # Phase 5: Subgradient
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 380:
        t_end = min(deadline - 380, time.monotonic() + 80)
        a5, sc5 = _subgradient_polish(best_seq, t_end, max_steps=1500)
        update_best(a5, sc5, "subgrad_1")

    # Phase 6: Another active-set QP round (tighter)
    print(f"\n[Phase 6] Active-set QP r2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not
