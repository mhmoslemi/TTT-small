    base_beta = max(gap, max_c * 0.002)

    # More aggressive schedule: start larger, end smaller, more steps
    betas = base_beta * np.array([
        2.0, 1.0, 0.75, 0.5, 0.35, 0.25, 0.175, 0.125, 0.085,
        0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # More iterations at fine betas
        if i < len(betas) // 2:
            est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
            max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 15))))
        else:
            est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
            max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _smoobj_fn, z, args=(beta, n),
                jac=lambda zz, b, nn: _smograd_fn(zz, b, nn),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 4 == 0:
                    print(f"    L-BFGS beta_idx={i}, beta={beta:.6f}: score={current_sc:.12f}")
        except Exception:
            pass

    return current_a


def _momentum_subgradient(a_init, deadline, min_time_remaining=60.0):
    """Nesterov-accelerated subgradient method in the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Nesterov momentum parameters
    momentum = 0.9
    eta = 0.005  # initial step size
    min_eta = 1e-9
    no_improve = 0
    max_stall = 300

    # Yekiel-C走完: maintain a "lookahead" point
    y = a.copy()

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(y, y)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = y.copy()
            no_improve = 0

        # Find active peaks (top fraction)
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute subgradient at y
        subgrad = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            subgrad[lo:hi + 1] += 2.0 * y[k - np.arange(lo, hi + 1)]
        subgrad /= n_active

        # Project to tangent of simplex (zero mean)
        subgrad -= subgrad.mean()

        # Gradient norm for step size
        gnorm = np.linalg.norm(subgrad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        alpha = min(eta / (1.0 + gnorm * 0.1), 1.0)

        # Nesterov update
        a_new = y - alpha * subgrad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        # Check improvement with backtracking
        improved = False
        for _ in range(30):
            c_try = _full_convolve(a_new, a_new)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-15:
                # Nesterov momentum update
                y = a_new + momentum * (a_new - a)
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-10:
                    y = y / s_y
                a = a_new
                improved = True
                break
            alpha *= 0.5
            a_new = y - alpha * subgrad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                eta *= 0.5
                momentum = min(0.95, momentum + 0.01)
            if no_improve >= max_stall:
                if eta < min_eta:
                    break
                # Reset
                y = best_a.copy()
                a = best_a.copy()
                eta = max(eta, 0.001)
                momentum = 0.8
                no_improve = 0
                max_stall = int(max_stall * 1.3)
        else:
            # Gradually decrease step size
            eta = max(min_eta, eta * 0.999)

    return best_a, best_sc


def _active_set_projection(a_init, deadline, min_time_remaining=30.0):
    """Active-set method: minimize the top peaks by direct projection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.02
    min_lr = 1e-7
    tol_frac = 0.003
    min_tol = 0.00005
    no_improve = 0
    max_stall = 500

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Equal weight on active peaks
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Gradient: h[i] = sum_k w[k] * a[k-i]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Direction: reduce a where h > W (contributes to peaks)
        grad = h - W
        grad -= grad.mean()  # project to simplex tangent

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm * 1.5 + 1e-15))

        # Backtracking line search
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.7
            if no_improve % 30 == 0:
                tol_frac = max(min_tol, tol_frac * 0.7)
            if no_improve >= max_stall:
                if lr < min_lr or tol_frac <= min_tol * 1.1:
                    break
                a = best_a.copy()
                lr = max(min_lr, lr_init * 0.3) if 'lr_init' in dir() else max(min_lr, 0.01)
                tol_frac = max(min_tol, tol_frac * 0.5)
                no_improve = 0
        else:
            if tol_frac > min_tol:
                tol_frac = max(min_tol, tol_frac * 0.98)
            lr = min(0.02, lr * 1.01)

    return best_a, best_sc


def _directed_perturbation(a_init, deadline, rng, min_time_remaining=30.0):
    """Try directed perturbations that target specific peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Find top peaks
    top_peaks = np.argsort(c)[::-1][:min(20, len(c))]

    for t in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Pick a random peak to target
        peak_idx = top_peaks[rng.integers(len(top_peaks))]

        # Find which a[i] contribute most to this peak
        i_lo = max(0, peak_idx - (n - 1))
        i_hi = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = peak_idx - i
            contrib[i] = a[j]
            if i != j:
                contrib[j] += a[i]

        # Try reducing top contributors
        n_reduce = max(2, int(n * 0.003))
        top_contrib = np.argsort(contrib)[::-1][:n_reduce]

        # Try increasing bottom contributors (mass redistribution)
        n_incre = max(2, int(n * 0.003))
        bot_contrib = np.argsort(contrib)[:n_incre]

        # Random direction in the span of these
        direction = np.zeros(n)
        for i in top_contrib:
            direction[i] -= 1.0
        for i in bot_contrib:
            direction[i] += 1.0

        # Randomize
        direction += 0.1 * rng.standard_normal(n)
        direction -= direction.mean()

        # Small step
        step = 0.01 / (np.linalg.norm(direction) + 1e-10)
        a_try = a + step * direction
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            # Update peaks
            c = c_try
            c_max = float(c.max())
            top_peaks = np.argsort(c)[::-1][:min(20, len(c))]

    return best_a, best_sc


def _local_search(a_init, deadline, rng, min_time_remaining=30.0):
    """Local search: random log-perturbations + quick re-optimization."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    attempts = 0
    max_attempts = 50

    while time.monotonic() + min_time_remaining < deadline and attempts < max_attempts:
        attempts += 1
        # Random perturbation in log domain
        sigma = 0.05 / np.sqrt(attempts)
        z = np.log(np.maximum(a_init, 1e-30))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_try = _softmax_fn(z)

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_a = a_try.copy()
            best_sc = sc_try
            a_init = a_try
            print(f"    Local search iter {attempts}: NEW BEST {best_sc:.12f}")
            # Quick L-BFGS from new point
            remaining = deadline - min_time_remaining - time.monotonic()
            if remaining > 30:
                a_opt = _run_lbfgs_aggressive(a_try, deadline, min_time_remaining=min_time_remaining)
                sc_opt = _score(a_opt)
                if sc_opt < best_sc:
                    best_a = a_opt.copy()
                    best_sc = sc_opt
                    a_init = a_opt
                    print(f"    After re-opt: {best_sc:.12f}")

        elif attempts > 10 and sc_try > best_sc * 1.01:
            # Getting worse, reduce perturbation
            sigma *= 0.5

    return best_a, best_sc


def _multi_peak_gradient(a_init, deadline, num_peaks=5, min_time_remaining=30.0):
    """Target the top num_peaks peaks simultaneously with direct gradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.01
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0

        # Top peaks
        top_idx = np.argsort(c)[::-1][:num_peaks]
        top_vals = c[top_idx]

        # Weight: inverse of value (penalize more for higher peaks)
        w = np.zeros_like(c)
        for k in top_idx:
            w[k] = 1.0 / (top_vals[k] + 1e-30)
        w = w / w.sum()

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = h - W
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        alpha = lr / (gnorm + 1e-10)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.7
            if no_improve > 200 and lr < 1e-7:
                break
        else:
            lr = min(0.01, lr * 1.01)

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample_fn(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage_optimize(a, deadline):
    """Try zeroing out the smallest elements to see if score improves."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, momentum subgradient,
    active-set projection, directed perturbations, and local search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.12f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.12f}")

    # === Phase 1: L-BFGS warm-up ===
    print(f"\n[P1] L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 300)
    a1 = _run_lbfgs_aggressive(best_seq, p1_deadline, min_time_remaining=60.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  P1 NEW BEST: {best_score:.12f} (n={len(best_seq)})")
    else:
        print(f"  P1: no improvement (best={best_score:.12f})")

    # === Phase 2: Momentum subgradient ===
    print(f"\n[P2] Momentum subgradient (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, time.monotonic() + 200)
    a2, sc2 = _momentum_subgradient(best_seq, p2_deadline, min_time_remaining=60.0)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  P2 NEW BEST: {best_score:.12f} (n={len(best_seq)})")
    else:
        print(f"  P2: no improvement (best={best_score:.12f})")

    # === Phase 3: L-BFGS again from improved point ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, time.monotonic() + 200)
        a3 = _run_lbfgs_aggressive(best_seq, p3_deadline, min_time_remaining=60.0)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  P3 NEW BEST: {best_score:.12f}")

    # === Phase 4: Active-set projection ===
    print(f"\n[P4] Active-set projection (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p4_deadline = min(deadline - 250, time.monotonic() + 150)
        a4, sc4 = _active_set_projection(best_seq, p4_deadline, min_time_remaining=50.0)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  P4 NEW BEST: {best_score:.12f}")

    # === Phase 5: Multi-peak gradient ===
    print(f"\n[P5] Multi-peak gradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p5_deadline = min(deadline - 200, time.monotonic() + 100)
        a5, sc5 = _multi_peak_gradient(best_seq, p5_deadline, num_peaks=5,
                                       min_time_remaining=40.0)
        if sc5 < best_score:

            best_score = sc5

    # === Phase 6: Directed perturbation ===
    print(f"\n[P6] Directed perturbation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p6_deadline = min(deadline - 150, time.monotonic() + 80)
        a6, sc6 = _directed_perturbation(best_seq, p6_deadline, rng,
                                         min_time_remaining=40.0)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  P6 NEW BEST: {best_score:.12f}")

    # === Phase 7: Local search with re-optimization ===
    print(f"\n[P7] Local search (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:

        a8, sc8 = _shrinkage_optimize(best_seq, deadline - 60)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  P8 NEW BEST: {best_score:.12f}")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 50:
                a8b = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=20.0)
                sc8b = _score(a8b)
                if sc8b < best_score:
                    best_seq = a8b.copy()
                    best_score = sc8b
                    print(f"  P8b NEW BEST: {best_score:.12f}")

    print(f"\n[P9] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        a9 = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=3.0)
        sc9 = _score(a9)
        if sc9 < best_score:
            best_seq = a9.copy()
            best_score = sc9
            print(f"  P9 NEW BEST: {best_score:.12f}")

    # === Phase 10: Ultra-fine active set ===
    print(f"\n[P10] Ultra-fine active set (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 3 - time.monotonic()
    if remaining > 5:
        a10, sc10 = _active_set_projection(best_seq, deadline, min_time_remaining=1.0)
        if sc10 < best_score:
            best_seq = a10.copy()
            best_score = sc10
            print(f"  P10 NEW BEST: {best_score:.12f}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
