        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.5 / n
    momentum = np.zeros(n)
    beta_m = 0.9

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Subgradient: d/d a_i of conv[k_star] = a[k_star - i] (if valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
            else:
                # If k_star - i is out of bounds, the term doesn't exist
                # But we also need to consider the reverse: a[i] * a[k_star - i]
                # Actually conv[k] = sum_i a[i]*a[k-i], so d/d a[i] = a[k-i] if valid
                # AND d/d a[j] = a[k-j] if valid where j = k-i
                # Wait, conv[k] = sum_i a[i] * a[k-i]
                # d/d a[m] = a[k-m] (if 0 <= k-m < n) + a[m-k] (if 0 <= m-k < n)
                # For k = k_star: d/d a[m] = a[k_star - m] (if valid) + a[m - k_star] (if valid)
                pass

        # Recompute properly
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        # Normalize to account for scale invariance
        grad_t = grad - np.mean(grad)
        momentum = beta_m * momentum + lr * grad_t
        a_try = a - momentum
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            a = best_a.copy()
            continue
        a_try = a_try / s_try
        a = a_try

    return best_a, _score(best_a)


def _try_multiple_starts(deadline, best_score, best_a):
    """Try multiple random and structured starting points with quick optimization."""
    rng = np.random.default_rng(999)
    n_base = len(best_a)

    starts = []
    # Slightly perturbed uniform
    for sigma in [0.01, 0.05, 0.1, 0.2]:
        z = np.zeros(n_base) + sigma * rng.standard_normal(n_base)
        a = _softmax(z)
        starts.append(a)

    # Random
    for _ in range(5):
        a = rng.random(n_base)
        starts.append(a)

    # Structured
    t = np.linspace(0, 1, n_base)
    starts.append(np.maximum(np.sin(np.pi * t), 0))
    starts.append(np.maximum(1 - 2 * np.abs(t - 0.5), 0))
    starts.append(np.exp(-20 * (t - 0.5) ** 2))

    for a in starts:
        if time.monotonic() > deadline - 30:
            break
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  Better start found: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    # === Phase 0: Verify uniform is optimal (score should be exactly 2.0) ===
    print("=" * 60)
    print("Phase 0: Verifying theoretical optimum")
    for n in [256, 1024, 4096, 12288, 32768]:
        a = np.ones(n)
        sc = _score(a)
        print(f"  Uniform n={n}: score={sc:.15f}")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # === Phase 1: Try parent sequence ===
    print("\nPhase 1: Parent sequence")
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0, 1000)
        if len(parent) > 0 and parent.sum() > 0.01:
            sc = _score(parent)
            print(f"  Parent: n={len(parent)}, score={sc:.15f}")
            if sc < best_score:
                best_score = sc
                best_a = parent.copy()
    except Exception as e:
        print(f"  Parent not available: {e}")

    # === Phase 2: L-BFGS optimization from best ===
    print(f"\nPhase 2: L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 600.0, t_start + 400.0)
    a_opt = _run_lbfgs(best_a, p2_deadline, min_remaining=20.0)
    sc_opt = _score(a_opt)
    print(f"  L-BFGS result: {sc_opt:.15f}")

        best_score = sc_opt
        best_a = a_opt.copy()
        print(f"  NEW BEST: {best_score:.15f}")

    # === Phase 3: Active-set polishing ===
    print(f"\nPhase 3: Active-set polish (t={time.monotonic()-t_start:.1f}s)")

    if sc_pol < best_score:
        best_score = sc_pol
        best_a = a_pol.copy()
        print(f"  NEW BEST: {best_score:.15f}")

    # === Phase 4: Try different n values ===
    print(f"\nPhase 4: Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_current = len(best_a)
    for n_try in [n_current // 2, n_current * 2, 8192, 16384, 24576, 32768]:
        if time.monotonic() > deadline - 200.0:
            break
        if n_try < 64 or n_try > 65536 or n_try == n_current:
            continue
        # Resample
        x_old = np.linspace(0, 1, n_current)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        print(f"  Resampled n={n_try}: score={sc_try:.15f}")
        if sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
            print(f"  NEW BEST: {best_score:.15f}")

        # Quick L-BFGS on this n
        if time.monotonic() < deadline - 100.0:
            a_opt2 = _run_lbfgs(a_try, min(deadline - 80.0, time.monotonic() + 60.0), min_remaining=10.0)

    if time.monotonic() < deadline - 60.0:

    # === Phase 6: Final polish ===
    print(f"\nPhase 6: Final polishing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:

            best_a = a_fin.copy()
            print(f"  NEW BEST: {best_score:.15f}")

        if remaining > 10.0:
            a_fin2 = _run_lbfgs(best_a, deadline - 2.0, min_remaining=5.0)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_score = sc_fin2
                best_a = a_fin2.copy()
                print(f"  NEW BEST: {best_score:.15f}")

    # === Final output ===
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.15f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    print(f"Theoretical minimum for non-negative sequences: 2.0")
    print(f"Achieved: {sc_final:.15f}")

    return [float(x) for x in final]
