For c to be non-negative, we need the IDFT of A^2 to be non-negative. This is the constraint.

Now, to minimize max(c), we want c to be as flat as possible. The "flattest" c (subject to the constraints) would be c[k] = 1/(2n-1) for all k, which gives C_0 = 1, C_m = 0 for m ≠ 0. This requires A_0 = ±1 and A_m = 0 for m ≠ 0 (since A_m^2 = C_m). So A_0 = 1 (since sum(a) = 1 > 0), and A_m = 0 for m ≠ 0. This gives a as the Dirac delta, which contradicts the non-negativity of c (actually it doesn't, the delta is non-negative, but max(c) = 1 which gives score = 2n, which is terrible).

Wait, I think I made an error. If c[k] = 1/(2n-1) for all k, then max(c) = 1/(2n-1), and score = 2n/(2n-1) ≈ 1. But this c is a constant, whose DFT is C_0 = sum(c) = 1, C_m = 0 for m ≠ 0. So A_0^2 = 1, A_m^2 = 0. This means A_0 = 1, A_m = 0. The inverse DFT of this is a[k] = 1/N for all k (uniform). But the uniform a gives a TRIANGULAR c, not a constant c!

The issue is that the convolution of two length-n sequences gives a length-(2n-1) result, and the DFT relationship is:
If we zero-pad a to length N >= 2n-1, then DFT_N(a * a) = DFT_N(a)^2.

But the "c[k] = 1/(2n-1) for all k" is a length-(2n-1) sequence. Its DFT of length N is not simply C_0 = 1, C_m = 0. It's C_0 = 1, and C_m = sum_k (1/(2n-1)) * exp(-2πi mk/N) = (1/(2n-1)) * sum_{k=0}^{2n-2} exp(-2πi mk/N) for m ≠ 0.

So the relationship is more subtle. Let me just accept that the problem is non-trivial and focus on a practical algorithm.

**Let me now write the code.**

I'll use a combination of:
1. p-norm gradient descent (with projected gradient onto the simplex)
2. Active-set subgradient method
3. Mirror descent (exponential weights)
4. Multiple restarts with different initializations
5. Adaptive p schedule

The key innovation over the previous code:
- Use projected gradient descent in the a-space directly (not the z-space with softmax)
- This allows for a cleaner projection onto the simplex
- Use the p-norm with a more aggressive schedule
- Add a "Fourier flattening" initialization
- Use mirror descent as an alternative to L-BFGS

"""
Optimized step-function search using p-norm gradient descent with adaptive
scheduling, Fourier-domain initialization, mirror descent, and active-set
polishing. Targets score below 1.5027.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to work with sum=1
    v = v.copy()
    # Lagrangian projection
    # min sum (v_i - x_i)^2 s.t. x_i >= 0, sum(x) = target_sum
    # Solution: x_i = max(v_i - tau, 0) where tau is chosen so sum = target_sum
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.maximum(v - mid, 0.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-15:
            break
    x = np.maximum(v - mid, 0.0)
    # Fix any numerical issues
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        positive = x > 0
        if positive.any():
            x[positive] += diff / positive.sum()
            x = np.maximum(x, 0)
            # Re-normalize
            x = x / x.sum() * target_sum
    return x


def _pnorm_obj_grad(a, p, n):
    """
    Compute the p-norm objective and its gradient.
    F_p(a) = 2n * (sum c_k^p)^(1/p) / (sum a)^2
    Returns (objective, gradient).
    """
    s = float(a.sum())
    c = _full_convolve(a, a)
    c_p = np.power(np.maximum(c, 0), p)
    M_p = float(np.sum(c_p))
    M = M_p ** (1.0 / p)
    
    obj = 2.0 * n * M / (s * s)
    
    # Gradient
    # dF/da_i = 2n * [M^(1-p) * sum_k c_k^(p-1) * a[k-i]] / s^2 - 4n*M / s^3
    #         = 2n * h_i / (M^(p-1) * s^2) - 4n*M / s^3
    # where h_i = sum_k c_k^(p-1) * a[k-i] (full convolution of c^(p-1) with reversed a)
    
    c_pw1 = np.power(np.maximum(c, 0), p - 1)
    a_rev = a[::-1]
    h = _full_convolve(c_pw1, a_rev)
    h = h[n-1:2*n-1]  # Extract the relevant part
    
    grad = 2.0 * n * h / (M_p * s * s) - 4.0 * n * M / (s * s * s)
    
    return obj, grad


def _max_obj_grad(a, n, top_k=10, tol_frac=0.0):
    """
    Compute objective based on max of convolution and subgradient.
    If tol_frac > 0, use averaged subgradient over top peaks within tol_frac of max.
    """
    s = float(a.sum())
    c = _full_convolve(a, a)
    c_max = float(c.max())
    obj = 2.0 * n * c_max / (s * s)
    
    # Find active set
    if tol_frac > 0:
        threshold = c_max * (1.0 - tol_frac)
        active = c >= threshold
    else:
        # Use top-k
        k = min(top_k, len(c))
        top_indices = np.argpartition(c, -k)[-k:]
        active = np.zeros_like(c, dtype=bool)
        active[top_indices] = True
    
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    h = h[n-1:2*n-1]
    
    # Subgradient: 2n/s^2 * h - 4n*c_max/s^3 (with h being the "directional" part)
    # Actually: d(max c)/da_i = sum_k w[k] * dc[k]/da_i = sum_k w[k] * a[k-i] = h_i
    # dF/da_i = 2n/s^2 * h_i - 4n*c_max/s^3
    grad = 2.0 * n * h / (s * s) - 4.0 * n * c_max / (s * s * s)
    
    return obj, grad


def _optimize_pnorm(a_init, deadline, p_schedule=None, min_remaining=10.0):
    """
    Optimize using p-norm gradient descent with adaptive p schedule.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256, 512]
    
    for p in p_schedule:
        if time.monotonic() + min_remaining >= deadline:
            break
        
        # Phase for this p value
        phase_deadline = min(deadline - min_remaining, time.monotonic() + (deadline - time.monotonic()) * 0.3)
        
        lr = 0.1
        no_improve = 0
        max_no_improve = 500
        
        while time.monotonic() < phase_deadline:
            obj, grad = _pnorm_obj_grad(a, p, n)
            
            # Subtract mean for projection (outline: we project onto simplex)
            # Actually, since we're working on the simplex, we should use the
            # "simplex gradient" which has zero sum
            grad_mean = grad.mean()
            grad_centered = grad - grad_mean
            
            gnorm = np.linalg.norm(grad_centered)
            if gnorm < 1e-14:
                break
            
            # Adaptive step size
            alpha = lr / (gnorm + 1e-10)
            
            # Try the step
            a_try = a - alpha * grad_centered
            a_try = _project_simplex(a_try, 1.0)
            
            sc_try = _score(a_try)
            
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()
                no_improve = 0
            elif sc_try < obj + 0.01:  # Accept if not too bad (for p-norm, obj > true score)
                no_improve = 0
            
            if sc_try <= best_score + 1e-10:
                a = a_try
                no_improve = 0
            else:
                no_improve += 1
                lr *= 0.7
                if lr < 1e-8:
                    break
            
            if no_improve >= max_no_improve:
                break
        
        # Check if we improved
        if _score(a) < best_score:
            best_score = _score(a)
            best_a = a.copy()
    
    return best_a, best_score


def _optimize_mirror_descent(a_init, deadline, min_remaining=10.0):
    """
    Mirror descent (exponential weights) on the simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    t = 0
    while time.monotonic() + min_remaining < deadline:
        t += 1
        lr = 1.0 / math.sqrt(t + 1.0)
        
        obj, grad = _max_obj_grad(a, n, top_k=20, tol_frac=0.05)
        
        # Mirror descent update: a_new_i ∝ a_i * exp(-lr * grad_i)
        # Clip to avoid overflow
        exponent = -lr * grad
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 0.0)
        s_new2 = a_new.sum()
        if s_new2 < 1e-15:
            break
        a_new = a_new / s_new2
        
        sc_new = _score(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Accept if it doesn't increase the true objective too much
        if sc_new <= best_score * 1.001:
            a = a_new
        else:
            # Don't accept, but don't break
            lr *= 0.5
    
    return best_a, best_score


def _active_set_optimize(a_init, deadline, min_remaining=5.0):
    """
    Active-set subgradient optimization with adaptive peak tracking.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    top_k = 5
    tol_frac = 0.01
    lr = 0.05
    no_improve = 0
    
    while time.monotonic() + min_remaining < deadline:
        obj, grad = _max_obj_grad(a, n, top_k=top_k, tol_frac=tol_frac)
        
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-14:
            break
        
        alpha = lr / (gnorm + 1e-10)
        
        # Backtracking
        a_try = a
        alpha_try = alpha
        improved = False
        
        for bt in range(30):
            a_try = a - alpha_try * grad_centered
            a_try = _project_simplex(a_try, 1.0)
            sc_try = _score(a_try)
            
            if sc_try < best_score - 1e-12:
                improved = True
                break
            alpha_try *= 0.5
        
        if improved:
            a = a_try
            best_score = sc_try
            best_a = a.copy()
            no_improve = 0
            # Slightly increase lr
            lr = min(0.1, lr * 1.02)
            # Tighten active set
            if tol_frac > 0.001:
                tol_frac *= 0.95
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve % 50 == 0:
                top_k = max(1, top_k // 2)
                tol_frac = max(0.0001, tol_frac * 0.5)
            if no_improve > 500:
                break
    
    return best_a, best_score


def _make_fourier_init(n):
    """
    Create an initialization by designing in the Fourier domain.
    We want A(f)^2 (the spectrum of the self-convolution) to be as flat as possible.
    """
    N = _next_pow2(2 * n - 1)
    
    # Target: flat spectrum with DC = 1
    # A_0 = 1, A_m = r * exp(i*theta_m) for m != 0
    # We want A^2 to be as flat as possible
    # A^2_0 = 1, A^2_m = r^2 * exp(2i*theta_m)
    # For A^2 to be "flat", we want |A^2_m| to be constant
    # So |A_m| = r for all m != 0
    # And A^2 needs to be Hermitian: A^2_{N-m} = conj(A^2_m)
    # If A_m = r * exp(i*theta_m), then A^2_m = r^2 * exp(2i*theta_m)
    # And A^2_{N-m} = r^2 * exp(2i*theta_{N-m})
    # We need A^2_{N-m} = conj(A^2_m) = r^2 * exp(-2i*theta_m)
    # So theta_{N-m} = -theta_m (mod pi)
    
    # Simplest: make A real, so theta_m = 0 or pi
    # A_m = r * cos(m * phi) for some phase phi (this gives a "modulated" sequence)
    # Or: A_m = r for all m (gives uniform a back)
    
    # Better: use a "raised cosine" shape
    # A_m = r * (0.5 + 0.5 * cos(pi * m / (N/2))) for m <= N/2
    # This gives a smooth transition from DC to zero frequency
    
    m = np.arange(N // 2 + 1)
    # Normalize so that A_0 = 1
    A_mag = np.ones(N // 2 + 1)
    A_mag[0] = 1.0
    # Taper: reduce the high-frequency components
    taper = np.cos(np.pi * m / (N // 2)) ** 2  # Hann-like taper
    A_mag[1:] = 0.5 * taper[1:]
    
    # Make it Hermitian symmetric (real a)
    A_full = np.zeros(N, dtype=complex)
    A_full[:N//2+1] = A_mag
    for i in range(1, N//2 + 1):
        if i < N - i:
            A_full[N - i] = A_mag[i]  # Real => symmetric
    
    a = np.real(np.fft.ifft(A_full))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    
    return a


def _make_tapered_init(n, power=0.5):
    """Create a tapered initialization."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_wedge_init(n):
    """Create a wedge-shaped initialization (high in middle, low at edges)."""
    t = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_anti_wedge_init(n):
    """Create an anti-wedge (high at edges, low in middle)."""
    t = np.linspace(0, 1, n)
    a = np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 0.05)
    return a / a.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _perturb(a, sigma, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(np.clip(z - z.max(), -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return e / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)

    s = x.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    return x * (target / s)


def _make_fourier_init(n):
    """Fourier-domain flat-spectrum initialization."""
    N = _next_pow2(2 * n - 1)
    m = np.arange(N // 2 + 1)

    # Design |A(f)| to be as flat as possible
    # A(0) = 1 (DC), target |A(f)| ~ 1/sqrt(2n) for f != 0
    # Use a smooth taper
    A_mag = np.ones(N // 2 + 1)
    A_mag[0] = 1.0
    # Raised cosine taper (Hann window in frequency)
    taper = 0.5 + 0.5 * np.cos(np.pi * m / (N // 2))
    A_mag[1:] = 0.3 * taper[1:]

    # Build full DFT (real signal => Hermitian symmetry)
    A_full = np.zeros(N, dtype=complex)
    A_full[:N // 2 + 1] = A_mag.astype(complex)
    for i in range(1, N // 2 + 1):
        if i < N - i:
            A_full[N - i] = A_mag[i]

    a = np.real(np.fft.ifft(A_full))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_tapered_init(n, power=0.7):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_wedge_init(n):
    t = np.linspace(0, 1, n)
    a = 1.0 - 0.7 * np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cos_mod_init(n):
    t = np.arange(n, dtype=float)
    a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _downsample(a, factor):
    n = len(a)
    new_n = n // factor
    if new_n < 4:
        return a.copy()
    # Average in groups
    a_pad = a[:new_n * factor]
    a_small = a_pad.reshape(new_n, factor).mean(axis=1)
    a_small = np.maximum(a_small, 0.0)
    s = a_small.sum()
    if s > 1e-15:
        a_small = a_small / s
    return a_small


def _perturb(a, sigma, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(np.clip(z - z.max(), -700, 0.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return e / s


def _optimize_at_scale(a_init, deadline, min_remaining=10.0, aggressive=False):
    """Run a full optimization cycle at a given scale."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # L-BFGS phase
    lbfgs_deadline = min(deadline - min_remaining, time.monotonic() + (deadline - time.monotonic()) * 0.55)
    a1 = _run_lbfgs(best_a, lbfgs_deadline, min_remaining=min_remaining * 0.3, aggressive=aggressive)
    sc1 = _score(a1)
    if sc1 < best_sc:
        best_a = a1.copy()
        best_sc = sc1

    # p-norm gradient descent
    pn_deadline = min(deadline - min_remaining * 0.5, time.monotonic() + (deadline - time.monotonic()) * 0.25)
    if time.monotonic() < pn_deadline:
        a2, sc2 = _pnorm_grad_descent(best_a, pn_deadline,
                                       p_values=[16, 32, 64, 128, 256],
                                       min_remaining=3.0)
        if sc2 < best_sc:
            best_a = a2.copy()
            best_sc = sc2

    # Active-set polish
    ps_deadline = min(deadline - min_remaining * 0.3, time.monotonic() + (deadline - time.monotonic()) * 0.2)
    if time.monotonic() < ps_deadline:
        a3, sc3 = _active_set_polish(best_a, ps_deadline, min_remaining=2.0)
        if sc3 < best_sc:
            best_a = a3.copy()
            best_sc = sc3

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimizer: coarse-to-fine L-BFGS + p-norm GD + active-set polish,
    with Fourier initialization and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    target_n = 12288

    if parent is not None:
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}")
    else:
        n_parent = target_n
        print("No parent available, using synthetic initializations.")

    # === Multi-scale coarse-to-fine ===
    print(f"\n[M1] Multi-scale coarse-to-fine (t={time.monotonic()-t_start:.1f}s)")

    # Build multi-scale chain: start small, go to target
    scales = []
    n_cur = target_n
    while n_cur > 256:
        scales.insert(0, n_cur)
        n_cur = max(256, n_cur // 2)
    if scales[0] != 256:
        scales[0] = 256
    if target_n not in scales:
        scales.append(target_n)

    print(f"  Scales: {scales}")

    # Initialize at finest scale
    if parent is not None:
        a_current = _resample(parent, target_n)
    else:
        a_current = _make_fourier_init(target_n)
    sc_current = _score(a_current)
    print(f"  Initial (n={target_n}): score={sc_current:.10f}")

    best_a = a_current.copy()
    best_score = sc_current

    # Optimize from coarse to fine
    time_per_scale = (deadline - t_start - 200.0) / max(1, len(scales))

    for i, n_scale in enumerate(scales):
        scale_deadline = min(deadline - 150.0, t_start + 200.0 + time_per_scale * (i + 1))
        remaining_alloc = scale_deadline - time.monotonic() - 30.0
        if remaining_alloc < 10.0:
            continue

        # Get initialization for this scale
        if n_scale == target_n:
            a_scale = a_current.copy()
        elif parent is not None:
            a_scale = _resample(parent, n_scale)
            # Blend with upsampled previous scale result
            if i > 0 and len(a_current) != n_scale:
                a_up = _resample(a_current, n_scale)
                a_scale = 0.7 * a_scale + 0.3 * a_up
                a_scale = np.maximum(a_scale, 0.0)
                a_scale = a_scale / a_scale.sum()
            else:
                a_scale = _resample(parent, n_scale)
        else:
            a_scale = _make_fourier_init(n_scale)

        sc_scale_init = _score(a_scale)
        print(f"  Scale n={n_scale}: init score={sc_scale_init:.10f}")

        a_opt, sc_opt = _optimize_at_scale(a_scale, scale_deadline, min_remaining=15.0,
                                            aggressive=(n_scale >= 4096))
        print(f"  Scale n={n_scale}: optimized score={sc_opt:.10f}")

        if n_scale == target_n:
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [multi-scale]")
            a_current = a_opt
        else:
            # Upscale for next iteration
            a_current = _resample(a_opt, target_n)
            # Also try to use this as a blend for the next scale

    # === Phase 2: Aggressive L-BFGS on best ===
    print(f"\n[P2] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400.0, t_start + 600.0)
    a2 = _run_lbfgs(best_a, p2_deadline, min_remaining=30.0, aggressive=True)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [aggressive_lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc2:.10f} vs {best_score:.10f}")

    # === Phase 3: p-norm gradient descent ===
    print(f"\n[P3] p-norm GD (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300.0, t_start + 800.0)
    a3, sc3 = _pnorm_grad_descent(best_a, p3_deadline,
                                   p_values=[16, 32, 64, 128, 256, 512, 1024],
                                   min_remaining=20.0)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [pnorm] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc3:.10f} vs {best_score:.10f}")

    # === Phase 4: Active-set polish ===
    print(f"\n[P4] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 200.0, t_start + 1000.0)
    a4, sc4 = _active_set_polish(best_a, p4_deadline, min_remaining=15.0,
                                  tol_start=0.003, tol_end=0.00001, lr_start=0.03)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc4:.10f} vs {best_score:.10f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n[P5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)

    restart_inits = [
        ("fourier", _make_fourier_init(n_best)),
        ("tapered", _make_tapered_init(n_best)),
        ("wedge", _make_wedge_init(n_best)),
        ("cos_mod", _make_cos_mod_init(n_best)),
        ("pert_small", _perturb(best_a, 0.03, seed=777)),
        ("pert_med", _perturb(best_a, 0.08, seed=888)),
        ("pert_large", _perturb(best_a, 0.15, seed=999)),
    ]

    # If parent has different length, try it directly
    if parent is not None and len(parent) != n_best:
        restart_inits.append(("parent_direct", parent.copy()))
        restart_inits.append(("parent_resampled", _resample(parent, n_best)))

    for label, a_r in restart_inits:
        if time.monotonic() > deadline - 80.0:
            break
        remaining = deadline - time.monotonic() - 40.0
        if remaining < 20.0:
            break

        sc_r = _score(a_r)
        if sc_r < best_score:
            best_a = a_r.copy()
            best_score = sc_r
            print(f"  NEW BEST: {best_score:.10f} [init_{label}]")
            continue

        alloc = min(remaining * 0.35, 120.0)
        if alloc < 10.0:
            continue

        a_opt, sc_opt = _optimize_at_scale(a_r, t_start + time.monotonic() - t_start + alloc,
                                            min_remaining=5.0, aggressive=False)
        # Fix: use absolute time
        a_opt, sc_opt = _optimize_at_scale(a_r, time.monotonic() + alloc,
                                            min_remaining=5.0, aggressive=False)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final L-BFGS + polish ===
    print(f"\n[P6] Final optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10.0:
        # One more aggressive L-BFGS
        lbfgs_dl = min(deadline - 10.0, time.monotonic() + 120.0)
        a_fin = _run_lbfgs(best_a, lbfgs_dl, min_remaining=8.0, aggressive=True)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

        # Final active-set
        if time.monotonic() < deadline - 5.0:
            a_pol, sc_pol = _active_set_polish(best_a, deadline - 3.0, min_remaining=2.0,
                                               tol_start=0.001, tol_end=0.000005, lr_start=0.01)
            if sc_pol < best_score:
                best_a = a_pol.copy()
                best_score = sc_pol
                print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Try slightly different n ===
    print(f"\n[P7] N exploration (t={time.monotonic()-t_start:.1f}s)")
    for n_try in [16384, 14336, 10240]:
        if time.monotonic() > deadline - 30.0:
            break
        a_try = _resample(best_a, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try}: resampled score={sc_try:.10f}")

        if sc_try < best_score:
            best_a = a_try.copy()
            best_score = sc_try
            print(f"  NEW BEST: {best_score:.10f} [n={n_try}]")

        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30.0 and sc_try < best_score + 0.01:
            a_opt, sc_opt = _optimize_at_scale(a_try, time.monotonic() + min(remaining * 0.6, 100.0),
                                                min_remaining=5.0, aggressive=True)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [opt_n={n_try}] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
