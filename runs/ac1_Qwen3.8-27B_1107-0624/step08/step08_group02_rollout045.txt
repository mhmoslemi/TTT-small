def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _hankel_diag(W, n):
    """Compute diagonal of the Hankel matrix H_W."""
    # H_W @ a: element j is 2*sum_k W_k * a_rev[k] * (appropriate indexing)
    # The diagonal element is 2*W[n-1-j] (approximately)
    # More precisely, (H_W)_{j,j} = 2 * W[n-1-j] if n-1-j is in range
    diag = np.zeros(n)
    for j in range(n):
        k = n - 1 - j
        if 0 <= k < len(W):
            diag[j] = 2.0 * W[k]
    return np.maximum(diag, 1e-30)


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14, precond=True):
    """CG solver with diagonal preconditioning and restarts."""
    n_inner = len(rhs)
    x = np.zeros(n_inner)
    r = rhs.copy()
    
    if precond:
        d = _hankel_diag(W, n)
        z = r / d
    else:
        z = r.copy()
    
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if abs(rz0) < 1e-30:
        return x
    
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            # Restart
            x = np.maximum(x, 0)
            r = rhs - _hankel_matvec(x, W, n)
            if precond:
                d = _hankel_diag(W, n)
                z = r / d
            else:
                z = r.copy()
            p = z.copy()
            rz = float(r @ z)
            rz0 = rz
            if abs(rz0) < 1e-30:
                break
            continue
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        if precond:
            z_new = r / d
        else:
            z_new = r.copy()
        rz_new = float(r @ z_new)
        if abs(rz_new) < tol * tol * abs(rz0):
            break
        beta = rz_new / rz if abs(rz) > 1e-30 else 0.0
        p = z_new + beta * p
        rz = rz_new
        # Periodic restart
        if i % 50 == 49:
            r = rhs - _hankel_matvec(x, W, n)
            if precond:
                z = r / d
            else:
                z = r.copy()
            p = z.copy()
            rz = float(r @ z)
            if abs(rz) < 1e-30:
                break
    
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=200, verbose=False):
    """Improved Newton active set with multi-phase tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    # Multi-phase tolerance: start loose, tighten
    phases = [
        (0.005, 30),   # (tol_frac, max_rounds_in_phase)
        (0.002, 30),
        (0.0005, 40),
        (0.0001, 40),
        (0.00002, 40),
        (0.000005, 50),
    ]

    total_round = 0
    for phase_idx, (tol_frac, phase_rounds) in enumerate(phases):
        if time.monotonic() > deadline:
            break
        if total_round >= max_rounds:
            break
        
        for rnd in range(phase_rounds):
            total_round += 1
            if time.monotonic() > deadline:
                break
            if total_round >= max_rounds:
                break
            
            c = _full_convolve(a, a)
            c_max = float(c.max())
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active < 1:
                active[int(np.argmax(c))] = True
                n_active = 1
            
            # For very large active sets, focus on the top ones
            if n_active > 500:
                c_sorted_idx = np.argsort(c)[::-1][:min(n_active, 300)]
                active = np.zeros_like(c, dtype=bool)
                active[c_sorted_idx] = True
                n_active = len(c_sorted_idx)
            
            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / n_active
            
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            
            # KKT residual check
            if gn < 1e-12 * n * c_max:
                if verbose:
                    print(f"    Newton converged at phase {phase_idx}, round {rnd}, residual={gn:.2e}")
                break
            
            # Solve H dx = -g_t (with sum constraint handled by H1)
            Hg = _cg_solve(W, n, g, max_iter=500, tol=1e-14)
            H1 = _cg_solve(W, n, np.ones(n), max_iter=300, tol=1e-14)
            H1_norm = float(np.linalg.norm(H1))
            
            if H1_norm < 1e-20 or not np.isfinite(H1_norm):
                dx = -g_t / max(gn, 1e-30)
            else:
                lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
                dx = -Hg + lam * H1
                dx = dx - np.mean(dx)
                dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    dx = -g_t / max(gn, 1e-30)
            
            # Cubic line search
            alpha = 1.0
            step_done = False
            a0 = a.copy()
            f0 = c_max
            df0 = float(g_t @ dx)  # directional derivative (approx)
            
            for ls in range(60):
                a_try = a0 + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.4
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                
                # Armijo condition with strong decrease
                armijo_ok = c_max_try < f0 - 0.01 * alpha * max(abs(df0), 1e-30)
                strong_ok = c_max_try < f0 * (1.0 - 0.001 * alpha)
                
                if armijo_ok or strong_ok or c_max_try < f0 - 1e-15:
                    a = a_try
                    step_done = True
                    break
                alpha *= 0.5
                if alpha < 1e-15:
                    break
            
            if not step_done:
                # Subgradient fallback with multiple step sizes
                for trial in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
                    a_try = a - trial * g_t / max(gn, 1e-30)
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max - 1e-15:
                        a = a_try
                        step_done = True
                        break
            
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if not step_done and no_improve > 8:
                if verbose:
                    print(f"    Newton stalled at phase {phase_idx}, round {rnd}")
                break
            elif no_improve > 15:
                break

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_a if False else current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.003 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        elif sc_improved > 1e-14:
            beta *= 0.3
        else:
            beta *= 0.5

        beta = min(beta, c_max_new * 0.001)
        if beta < min_beta:
            break

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.0 + 0.01
        return a / a.sum()
    elif kind == "flat_spec":
        # Try to create a sequence with flat power spectrum
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.sin(2 * np.pi * 3 * t) + 0.05 * np.cos(2 * np.pi * 7 * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-t**2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_structured(a, rng, mode="lowfreq"):
    """Add structured (low-frequency) perturbations to break symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    if mode == "lowfreq":
        # Add low-frequency perturbation
        freqs = rng.choice(np.arange(1, min(n // 2, 100)), size=5, replace=False)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.01, 0.05, size=5)
        t = np.arange(n)
        pert = np.zeros(n)
        for f, p, amp in zip(freqs, phases, amps):
            pert += amp * np.sin(2 * np.pi * f * t / n + p)
        z = z + pert
    elif mode == "highfreq":
        freqs = rng.choice(np.arange(n // 4, n // 2), size=5, replace=False)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.005, 0.02, size=5)
        t = np.arange(n)
        pert = np.zeros(n)
        for f, p, amp in zip(freqs, phases, amps):
            pert += amp * np.sin(2 * np.pi * f * t / n + p)
        z = z + pert
    elif mode == "symmetric_break":
        # Break symmetry: add a small ramp
        t = np.linspace(0, 1, n)
        z = z + 0.02 * (t - 0.5) * rng.standard_normal()
    else:
        z = z + 0.01 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_around(a_init, deadline, max_steps=2000):
    """Subgradient method with adaptive step size around a good point."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with moderate tolerance
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.03)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.008

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick n-scan to find optimal length
    print(f"\n[Phase 1] n-scan (t={time.monotonic() - t_start:.1f}s)")
    n_options = [n_parent]
    for delta in [-2048, -1024, 1024, 2048, 4096]:
        n_try = n_parent + delta
        if 4096 <= n_try <= 65536 and n_try != n_parent:
            n_options.append(n_try)
    
    best_n = n_parent
    best_n_a = parent.copy()
    best_n_sc = _score(parent)
    
    for n_try in n_options:
        if time.monotonic() > deadline - 1200:
            break
        a_try = _resample(parent, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try}: score={sc_try:.10f}")
        if sc_try < best_n_sc:
            best_n_sc = sc_try
            best_n = n_try
            best_n_a = a_try.copy()

    print(f"  Best n={best_n}, score={best_n_sc:.10f}")
    start_a = best_n_a
    n_work = best_n

    # Phase 2: Quick L-BFGS to get a good starting point
    print(f"\n[Phase 2] Quick L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = min(deadline - 900, time.monotonic() + 300)
    a_lb = _run_adaptive_lbfgs(start_a, lbfgs_deadline, min_time_remaining=60.0)
    sc_lb = _score(a_lb)
    update_best(a_lb, sc_lb, "lbfgs_quick")
    
    # Phase 3: Main Newton optimization
    print(f"\n[Phase 3] Main Newton (t={time.monotonic() - t_start:.1f}s)")
    newton_deadline = min(deadline - 500, time.monotonic() + 400)
    a_np, sc_np = _newton_active_polish(
        best_seq if best_seq is not None else a_lb,
        newton_deadline,
        max_rounds=300,
        verbose=True
    )
    update_best(a_np, sc_np, "newton_main")

    # Phase 4: Perturb-restart loop
    print(f"\n[Phase 4] Perturb-restart (t={time.monotonic() - t_start:.1f}s)")
    perturb_modes = ["lowfreq", "highfreq", "symmetric_break", "lowfreq", "highfreq"]
    perturb_sigmas = [0.02, 0.01, 0.03, 0.015, 0.008]
    
    for i in range(len(perturb_modes)):
        if time.monotonic() > deadline - 200:
            break
        
        base = best_seq if best_seq is not None else a_np
        mode = perturb_modes[i]
        sigma = perturb_sigmas[i]
        
        a_pert = _perturb_structured(base, rng, mode=mode)
        # Also mix with random perturbation
        a_pert2 = _perturb(base, sigma, rng)
        
        # Quick L-BFGS on perturbed
        pert_deadline = min(deadline - 150, time.monotonic() + 80)
        a_pr = _run_adaptive_lbfgs(a_pert, pert_deadline, min_time_remaining=30.0)
        sc_pr = _score(a_pr)
        
        a_pr2 = _run_adaptive_lbfgs(a_pert2, min(deadline - 140, time.monotonic() + 60), min_time_remaining=30.0)
        sc_pr2 = _score(a_pr2)
        
        best_pert = a_pr if sc_pr < sc_pr2 else a_pr2
        best_pert_sc = min(sc_pr, sc_pr2)
        update_best(best_pert, best_pert_sc, f"perturb_{i}_{mode}")
        
        # Newton polish on best perturbation
        if best_pert_sc < best_score + 0.001 and time.monotonic() < deadline - 100:
            np_deadline = min(deadline - 80, time.monotonic() + 60)
            a_npr, sc_npr = _newton_active_polish(best_pert, np_deadline, max_rounds=80)
            update_best(a_npr, sc_npr, f"perturb_{i}_np")

    # Phase 5: Subgradient diversification
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        sg_deadline = min(deadline - 120, time.monotonic() + 80)
        a_sg, sc_sg = _subgrad_around(best_seq, sg_deadline, max_steps=1500)
        update_best(a_sg, sc_sg, "subgrad")
        
        if sc_sg < best_score + 0.0005 and time.monotonic() < deadline - 80:
            np_deadline = min(deadline - 60, time.monotonic() + 50)
            a_sgnp, sc_sgnp = _newton_active_polish(a_sg, np_deadline, max_rounds=100)
            update_best(a_sgnp, sc_sgnp, "subgrad_np")

    # Phase 6: Try different n values from best sequence
    print(f"\n[Phase 6] Multi-n refinement (t={time.monotonic() - t_start:.1f}s)")
    n_refine = [len(best_seq) + 1024, len(best_seq) - 1024, len(best_seq) + 512]
    for n_r in n_refine:
        if time.monotonic() > deadline - 50:
            break
        if n_r < 4096 or n_r > 65536:
            continue
        a_r = _resample(best_seq, n_r)
        sc_r_init = _score(a_r)
        print(f"  Resample to n={n_r}: score={sc_r_init:.10f}")
        
        if sc_r_init < best_score + 0.01 and time.monotonic() < deadline - 30:
            r_deadline = min(deadline - 20, time.monotonic() + 40)
            a_rr, sc_rr = _newton_active_polish(a_r, r_deadline, max_rounds=50)
            update_best(a_rr, sc_rr, f"n_{n_r}")

    # Phase 7: Final ultra-fine Newton
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        final_deadline = deadline - 2
        a_fin, sc_fin = _newton_active_polish(best_seq, final_deadline, max_rounds=200, verbose=True)
        update_best(a_fin, sc_fin, "final_newton")

    # One more L-BFGS at very small beta
    if time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 1e-11
        remaining = deadline - 3 - time.monotonic()
        if remaining > 2:
            max_iter = min(3000, max(50, int(remaining / 0.005)))
            try:
                obj_fn = _SmoObjAdapter(beta_fin, n)
                grad_fn = _SmoGradAdapter(beta_fin, n)
                result = minimize(
                    obj_fn, z, jac=grad_fn,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                             'maxcor': 100, 'maxls': 300}
                )
                a_flb = _softmax(result.x)
                sc_flb = _score(a_flb)
                update_best(a_flb, sc_flb, "final_lbfgs")
            except Exception:
                pass

        # Final Newton after last L-BFGS
        if time.monotonic() < deadline - 1:
            a_veryfin, sc_veryfin = _newton_active_polish(best_seq, deadline - 0.5, max_rounds=50)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
