"""
Optimized step function search using active-set gradient methods,
L-BFGS with adaptive log-sum-exp smoothing, and multi-start strategy.
Targets minimization of 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(x):
    k = 1
    while k < x:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smoothed(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * full_convolve(a, w[::-1])[n-1:2n-1][::-1]
    """
    n = len(a)
    conv_aw = _full_convolve(a, w[::-1])
    return 2.0 * conv_aw[n-1:2*n-1][::-1]

    """Compute gradient of the smoothed max using active set.
    If top_k is given, use uniform weights on top k peaks.
    Otherwise, use exponential weighting with gap epsilon.

    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if top_k is not None:
        idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros_like(c)
        w[idx] = 1.0 / top_k
    else:
        # Exponential weighting
        shifted = (c - c_max) / epsilon
        e = np.exp(np.clip(shifted, -100, 0))
        s = e.sum()
        w = e / s
    
    return _grad_smoothed(a, w), c, c_max


def _optimize_active_set(a_init, deadline, min_remaining=30.0):
    """Active-set gradient descent with adaptive parameters."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.1
    epsilon = 0.01 * a.max()  # initial smoothing
    top_k = max(5, n // 100)  # number of peaks to target
    no_improve = 0
    t_start = time.monotonic()
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Compute gradient using active set
        # Use top_k peaks with uniform weight
        idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros_like(c)
        w[idx] = 1.0 / top_k
        
        grad = _grad_smoothed(a, w)
        
        # Project gradient onto the tangent space of the simplex
        # (remove the mean component)
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)
        
        if gnorm < 1e-15:
            break
        
        # Line search
        a_best = a
        sc_best = current_sc
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        
        for _ in range(50):
            a_try = a - alpha * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc_best - 1e-13:
                a_best = a_try
                sc_best = sc_try
                break
            alpha *= 0.5
        
        if sc_best < current_sc - 1e-13:
            a = a_best
            # Gradual parameter adjustment
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.7
                top_k = max(3, int(top_k * 0.8))
            if no_improve > 200:
                a = best_a.copy()
                lr = 0.1
                top_k = max(5, n // 100)
                no_improve = 0
        
        # Periodically increase top_k to capture more peaks
        if no_improve == 0 and top_k < n // 10:
            top_k = int(top_k * 1.05)
    
    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_remaining=30.0):
    """L-BFGS optimization with log-sum-exp smoothing in log domain."""
    n = len(a_init)
    
    class Prob:
        def __init__(self, beta):
            self.beta = beta
            self.n = n
        
        def __call__(self, z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / self.beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * self.n * c_max
            return 2.0 * self.n * (c_max + self.beta * np.log(s))
        
        def grad(self, z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / self.beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            grad_a = _grad_smoothed(a, w)
            # Jacobian of softmax: grad_z = a * (grad_a - a @ grad_a)
            W = float(np.dot(a, grad_a))
            return a * (grad_a - W)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Adaptive beta sequence
    c = _full_convolve(current_a, current_a)
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)]
    if gap < c_sorted[0] * 0.001:
        gap = c_sorted[0] * 0.001
    base_beta = max(gap, c_sorted[0] * 0.005)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        prob = Prob(beta)
        max_iter = min(3000, max(50, int(remaining / 0.01)))
        
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxcor': 50, 'maxls': 30}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc
