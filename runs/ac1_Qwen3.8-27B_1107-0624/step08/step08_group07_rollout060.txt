```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1 and (v >= 0).all():
        return v / v.sum()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def _fft_convolve(a, b):
    """Fast convolution using FFT."""
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = 1
    while N < out_len:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]

def _score_norm(a):
    """Score = 2*n*max(conv) (assuming sum(a)=1)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())

def _subgradient(c, a, active_set):

    n = len(a)
    g = np.zeros(n)
    for k in active_set:

        hi = min(k, n - 1)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(active_set)
    return g
```

Actually, let me think about this differently. The key challenge is the `max` in the objective. What if we reformulate using the "soft-min" or use a different approach entirely?

**Approach: Alternating Minimization**

1. Fix the active set A (the lags where the convolution is at its maximum)
2. Given A, solve the convex(?) problem: minimize max_{k in A} convolve(a,a)[k] subject to sum(a)=1, a≥0
3. Update A based on the solution
4. Repeat

Step 2 is the key. If A = {k1, k2, ..., km}, we want to minimize the maximum of m quadratic functions. This is still non-convex in general.

But if we linearize: at the current point a*, the constraint convolve(a,a)[k] ≤ t can be approximated by:
`2 * sum_j a*[j] * (a[j] - a*[j]) + convolve(a*,a*)[k] ≤ t` (first-order approximation)

This gives us a set of linear constraints, and the problem becomes:
```
minimize t
subject to: 2 * sum_j a*[j] * a[j] ≤ t - convolve(a*,a*)[k] + 2*sum_j (a*[j])^2, for k in A
            sum(a) = 1
            a ≥ 0
```

This is a linear program! We could solve it with scipy's linprog or a specialized solver.

But with n=8192, the LP would have 8192 variables and m+1 constraints (m active peaks + sum constraint). This is actually quite manageable!

Let me think about this more:
- Variables: a[0], ..., a[n-1], t (total n+1 variables)
- Constraints: 
  - m linear inequalities (one per active peak)
  - 1 equality (sum = 1)
  - n inequalities (a[j] ≥ 0)

Total: m + n + 1 constraints. With m ≤ 20 and n = 8192, this is about 8213 constraints.

Using scipy.optimize.linprog with the HiGHS solver, this should be fast.

The algorithm:
1. Start with some active set A
2. Solve the LP to get a new a
3. Compute the actual convolution
4. Update the active set
5. Repeat

This is essentially a "cutting plane" or "outer approximation" method. It's an exact method for the min-max of quadratic functions (if we add valid cuts), and it should converge to a local minimum.

```python
"""
Optimized step function search using:
1. Subgradient method with Nesterov momentum and adaptive step size
2. LP-based cutting plane refinement for local optimization
3. Multi-restart with structured perturbations
4. Final coordinate descent polish
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cmax(a):
    """Returns max of convolution (assuming sum(a)=1)."""
    c = _fft_convolve(a, a)
    return float(c.max()), c


def _subgrad_at_lags(a, lags):
    """Compute average subgradient at given lags."""
    n = len(a)
    g = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(lags)
    return g


def _bb_step(s, y):
    """Barzilai-Borwein step size."""
    sy = float(np.dot(s, y))
    if abs(sy) < 1e-15:
        return 0.01
    yy = float(np.dot(y, y))
    ss = float(np.dot(s, s))
    if yy > 1e-15:
        return sy / yy
    if ss > 1e-15:
        return sy / ss
    return 0.01


def _optimize_nesterov(a_init, deadline, max_iter=500000, active_tol_schedule=None):
    """
    Nesterov-accelerated subgradient method for min max(convolve(a,a)).
    Uses BB step size and adaptive active set.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    
    y = a.copy()  # Nesterov lookahead point
    
    best_a = a.copy()
    best_cmax, best_c = _cmax(a)
    
    lr = 0.05
    lr_min = 1e-8
    lr_max = 0.5
    prev_cmax = best_cmax
    prev_a = a.copy()
    prev_g = None
    patience = 0
    
    if active_tol_schedule is None:
        active_tol_schedule = [(0, 0.001), (500, 0.01), (2000, 0.05), (5000, 0.1)]
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Determine active set tolerance
        tol = active_tol_schedule[-1][1]
        for thresh, t_val in active_tol_schedule:
            if it < thresh:
                tol = t_val
                break
        
        # Compute gradient at y (Nesterov point)
        c_y, c_full = _cmax(y)
        threshold = c_y * (1 - tol)
        active = np.where(c_full >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_full)])
        if len(active) > 30:
            active = active[np.argsort(c_full[active])[::-1][:30]]
        
        g = _subgrad_at_lags(y, active)
        
        # BB step size
        if prev_g is not None:
            s_vec = y - prev_a  # approximate step in y-space
            y_vec = g - prev_g
            bb_lr = max(1e-8, min(_bb_step(s_vec, y_vec), lr_max))
            lr = 0.5 * lr + 0.5 * bb_lr
        
        # Nesterov update
        # y_new = a - lr * g (gradient step)
        # a_new = y_new (for simplicity, or use momentum)
        # Standard Nesterov: y = a + beta*(a - a_prev), then a_new = y - lr*g(y)
        # Let's use a simpler variant:
        # a_new = a - lr * g
        # y = a_new + momentum * (a_new - a_prev)
        
        momentum = 0.9  # Nesterov-like momentum
        a_new = a - lr * g
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(0, a_new)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        # Evaluate
        c_new_max, c_new_full = _cmax(a_new)
        
        if c_new_max < best_cmax - 1e-12:
            best_cmax = c_new_max
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Update momentum
        if it > 0:
            beta = (1 + np.sqrt(1 + 4 * (it-1)**2)) / (2 * it**2 + 2)  # Nesterov beta
            # Actually, standard Nesterov uses beta_t = (t-1)/(t+2)
            beta = (it - 1) / (it + 2) if it > 1 else 0
            y = a_new + beta * (a_new - a)
        else:
            y = a_new
        
        # Adjust learning rate
        if c_new_max > c_y:
            lr = max(lr * 0.7, lr_min)
        elif c_new_max < c_y * 0.999:
            lr = min(lr * 1.05, lr_max)
```

```python
"""
Optimized step function search using:
1. Nesterov-accelerated subgradient with Barzilai-Borwein step size
2. LP-based cutting plane refinement (scipy HiGHS)
3. Multi-restart with structured perturbations
4. Final coordinate descent and active-set polish

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cmax(a):
    c = _fft_convolve(a, a)
    return float(c.max()), c


def _subgrad_at_lags(a, lags):
    n = len(a)
    g = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(lags)
    return g


def _project_simplex(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _bb_step_size(s_vec, y_vec):
    sy = float(np.dot(s_vec, y_vec))
    if abs(sy) < 1e-18:
        return 0.01
    yy = float(np.dot(y_vec, y_vec))
    ss = float(np.dot(s_vec, s_vec))
    if yy > 1e-18:
        return max(1e-10, sy / yy)
    if ss > 1e-18:
        return max(1e-10, sy / ss)
    return 0.01

    a = a / s

    best_cmax, _ = _cmax(a)

    lr = lr_init
    lr_min = 1e-9
    lr_max = 1.0
    prev_a = a.copy()
    prev_g = None
    patience = 0
    momentum = 0.0
    y = a.copy()  # Nesterov lookahead

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive active set size
        if it < 100:
            active_k = active_k_init
            tol = 0.005

        else:

            tol = 0.15

        c_y = _fft_convolve(y, y)

    all active peaks simultaneously.
    """

        tol = 0.01

        if len(active) < 2:

        if len(active) > 20:
            active = active[np.argsort(c[active])[::-1][:20]]

        K = len(active)
        # Build system: for each active k, we want to reduce c[k]
        # Direction d should satisfy: for each k in active, dc[k]/da · d < 0
        # Minimum norm: d = M^T (M M^T)^{-1} * (-1, ..., -1)^T
        # where M[i, j] = dc[k_i]/da[j] = 2*a[k_i-j] (bounded)

        for i, k in enumerate(active):

            hi = min(k, n - 1)

        # Add zero-sum constraint
        M_full = np.vstack([M, np.ones(n) * 0.001])
        b = np.concatenate([-np.ones(K), [0.0]])

        try:
            G = M_full @ M_full.T
            G_reg = G + 1e-10 * np.eye(G.shape[0])
            G_inv_b = np.linalg.solve(G_reg, b)
            d = M_full.T @ G_inv_b
        except np.linalg.LinAlgError:
            break

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Try step sizes
        improved = False

        if not improved and rnd > 100:
            # Escalate: try more coordinates
            if rnd % 100 == 0:
                continue
            if rnd > 500:
                break

    return best_a, 2.0 * n * best_cmax


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Subgradient with Nesterov (0-300s)
    Phase 2: LP cutting plane refinement (300-550s)
    Phase 3: Active-set equalization (550-700s)
    Phase 4: Multi-restart with subgrad + LP (700-1400s)
    Phase 5: Final polish with coord descent + active set (1400-1790s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Nesterov subgradient (0-250s)
    p1_end = min(deadline, t_start + 250.0)
    print(f"Phase 1: Nesterov subgradient (deadline={p1_end - t_start:.0f}s)")
    a_opt, sc = _optimize_subgrad(a, p1_end, max_iter=200000, lr_init=0.1)
    full_sc = _score(a_opt)
    print(f"  Phase 1: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 1b: Smoothed optimization (250-400s)
    p1b_end = min(deadline, t_start + 400.0)
    if time.monotonic() < p1b_end:
        print(f"Phase 1b: Smoothed min-max (deadline={p1b_end - t_start:.0f}s)")
        a_opt2, sc2 = _optimize_smoothed(best_a, p1b_end, max_iter=100000, T_ratio=0.005, lr_init=0.03)
        full_sc2 = _score(a_opt2)
        print(f"  Phase 1b: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # Phase 2: LP cutting plane (400-650s)
    p2_end = min(deadline, t_start + 650.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: LP cutting plane (deadline={p2_end - t_start:.0f}s)")
        a_lp, sc_lp = _lp_refine(best_a, p2_end, max_rounds=100)
        full_sc_lp = _score(a_lp)
        print(f"  Phase 2: score={full_sc_lp:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_lp < best_score:
            best_score = full_sc_lp
            best_a = a_lp.copy()

    # Phase 3: Active-set equalization (650-800s)
    p3_end = min(deadline, t_start + 800.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Active-set equalization (deadline={p3_end - t_start:.0f}s)")
        a_as, sc_as = _active_set_polish(best_a, p3_end, max_iter=3000)
        full_sc_as = _score(a_as)
        print(f"  Phase 3: score={full_sc_as:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_as < best_score:
            best_score = full_sc_as
            best_a = a_as.copy()

    # Phase 4: Multi-restart (800-1450s)
    p4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Multi-restart (deadline={p4_end - t_start:.0f}s)")
        n_restarts = 6
        time_per = max(30.0, (p4_end - time.monotonic()) / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > p4_end - time_per:
                break

            # Different perturbation strategies
            if r == 0:
                sigma = 0.003
            elif r == 1:
                sigma = 0.01
            elif r == 2:
                sigma = 0.005
                # Structured: sinusoidal perturbation
                pert = 0.01 * np.sin(2 * np.pi * (r+3) * np.arange(n) / n)
                a_pert = best_a + pert * best_a
                a_pert = _project_simplex(a_pert)
            elif r == 3:
                sigma = 0.008
                # Structured: block perturbation
                block = n // 8
                pert = np.zeros(n)
                for b in range(8):
                    pert[b*block:(b+1)*block] = 0.02 * rng.standard_normal()
                a_pert = best_a + pert * best_a
                a_pert = _project_simplex(a_pert)
            elif r == 4:
                sigma = 0.015
            else:
                sigma = 0.002

            if r < 2 or r == 4 or r == 5:
                a_pert = best_a + sigma * rng.standard_normal(n)
                a_pert = _project_simplex(a_pert)

            # Quick subgradient optimization
            sub_dl = min(p4_end - 10, time.monotonic() + time_per * 0.5)
            a_r, sc_r = _optimize_subgrad(a_pert, sub_dl, max_iter=50000, lr_init=0.05)

            # Quick LP refinement
            sub_dl2 = min(p4_end - 10, time.monotonic() + time_per * 0.35)
            if time.monotonic() < sub_dl2:
                a_r, sc_r2 = _lp_refine(a_r, sub_dl2, max_rounds=30)

            full_sc_r = _score(a_r)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_sc_r:.10f} t={time.monotonic()-t_start:.0f}s")
            if full_sc_r < best_score:
                best_score = full_sc_r
                best_a = a_r.copy()

    # Phase 5: Final polish (1450-1790s)
    if time.monotonic() < deadline - 5:
        print(f"Phase 5: Final polish (deadline={deadline - t_start:.0f}s)")

        # Active set polish
        p5a_end = min(deadline - 30, t_start + 1600.0)
        if time.monotonic() < p5a_end:
            a_p, sc_p = _active_set_polish(best_a, p5a_end, max_iter=2000)
            full_sc_p = _score(a_p)
            print(f"  Active polish: score={full_sc_p:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc_p < best_score:
                best_score = full_sc_p
                best_a = a_p.copy()

        # LP refinement
        p5b_end = min(deadline - 15, t_start + 1720.0)
        if time.monotonic() < p5b_end:
            a_lp2, sc_lp2 = _lp_refine(best_a, p5b_end, max_rounds=50)
            full_sc_lp2 = _score(a_lp2)
            print(f"  LP polish: score={full_sc_lp2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc_lp2 < best_score:
                best_score = full_sc_lp2
                best_a = a_lp2.copy()

        # Coordinate descent
        p5c_end = min(deadline - 5, t_start + 1780.0)
        if time.monotonic() < p5c_end:
            a_cd, sc_cd = _coord_descent(best_a, p5c_end, max_rounds=2000)
            full_sc_cd = _score(a_cd)
            print(f"  Coord descent: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc_cd < best_score:
                best_score = full_sc_cd
                best_a = a_cd.copy()

        # One more subgradient pass
        if time.monotonic() < deadline - 3:
            a_sg, sc_sg = _optimize_subgrad(best_a, deadline - 3, max_iter=50000, lr_init=0.02)
            full_sc_sg = _score(a_sg)
            if full_sc_sg < best_score:
                best_score = full_sc_sg
                best_a = a_sg.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
