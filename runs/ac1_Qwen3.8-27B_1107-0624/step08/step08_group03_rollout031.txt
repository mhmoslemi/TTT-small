"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Aggressive L-BFGS with log-sum-exp continuation, projected subgradient,
peak-targeted refinement, active-set Newton polish, and multi-scale n exploration.
Key improvements over previous: more aggressive beta schedule with reheating,
Lp-norm alternative smoothing, better active-set adaptation, and structured
bump initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lp_obj(z, p, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return 0.0
    normalized = c / c_max
    return 2.0 * n * c_max * float(np.sum(normalized ** p)) / (2 * n - 1)


def _lp_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(z)
    normalized = c / c_max
    w = p * normalized ** (p - 1)
    w = w / w.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_one_step(a, n, top_k, step):
    c = _full_convolve(a, a)
    top_idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros_like(c)
    w[top_idx] = 1.0 / top_k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 2.0 * a * (h - W)
    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new


def _run_subgrad(a_init, deadline, max_steps=5000, top_k=10, lr_init=0.02):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a_new = _subgrad_one_step(a, n, top_k, lr)
        a = a_new
        if i % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    a = best_a.copy()
                    patience = 0
                    lr = lr_init * 0.5

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _peak_refine(a_init, deadline, max_rounds=100, n_peaks=5):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        n_pk = min(n_peaks, len(c))
        top_pk = np.argsort(c)[::-1][:n_pk]
        w = np.zeros_like(c)
        for k in top_pk:
            w[k] = 1.0 / n_pk
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        f_old = c_max
        accepted = False
        step = max(0.5 / (1.0 + 0.05 * rnd), 0.01)
        for ls in range(40):
            a_try = a - step * grad
            a_try = _project_simplex(a_try)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                accepted = True
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5
        if not accepted:
            break

    return best_a, best_sc


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:3 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 3 * n

        A_vec = active.astype(np.float64)
        m = n_active
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-5 * (c_max / max(m, 1)) + 1e-10

        def hess_vec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        x = np.zeros(n)
        r = g_t - hess_vec(x)
        p = r.copy()
        rs = float(np.dot(r, r))
        for cg_it in range(300):
            Ap = hess_vec(p)
            pAp = float(np.dot(p, Ap))
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r = r - alpha * Ap
            rs_new = float(np.dot(r, r))
            if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(g_t)):
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        w_solve = np.zeros(n)
        r2 = np.ones(n) - hess_vec(w_solve)
        p2 = r2.copy()
        rs2 = float(np.dot(r2, r2))
        for cg_it in range(300):
            Ap2 = hess_vec(p2)
            pAp2 = float(np.dot(p2, Ap2))
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            w_solve += alpha2 * p2
            r2 = r2 - alpha2 * Ap2
            rs2_new = float(np.dot(r2, r2))
            if np.sqrt(rs2_new) < 1e-13:
                break
            p2 = r2 + (rs2_new / rs2) * p2
            rs2 = rs2_new

        sum_w = float(w_solve.sum())
        if abs(sum_w) < 1e-15:
            dx = -x
            dx = dx - np.mean(dx)
        else:
            lam = float(x.sum()) / sum_w
            dx = -x + lam * w_solve
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                      0.015, 0.0075, 0.0037, 0.0018,
                                      0.0009, 0.00045, 0.00022, 0.00011,
                                      0.00005, 0.000025, 0.000012, 0.000006,
                                      0.000003, 0.0000015])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.0035, 0.0017,
                                      0.0008, 0.0004, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 10:
            break

        if n <= 4096:
            est_per_iter = 0.002
        elif n <= 8192:
            est_per_iter = 0.004
        elif n <= 16384:
            est_per_iter = 0.01
        else:
            est_per_iter = 0.025 * (n / 32768.0)

        max_iter = min(8000, max(50, int(remaining / (est_per_iter * 8))))

        try:
            result = minimize(
                _ls_obj, z,
                jac=_ls_grad,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-12,
                    'maxcor': 80,
                    'maxls': 100,
                    'maxfun': max_iter * 3,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    # Reheating: go back up slightly then down again
    if time.monotonic() + min_time_remaining < deadline:
        z = np.log(np.maximum(current_a, 1e-30))
        z = z - z.mean()
        reheat_betas = base_beta * np.array([0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2.5e-5, 1e-5, 5e-6, 2e-6, 1e-6])
        for beta in reheat_betas:
            if time.monotonic() + min_time_remaining > deadline:
                break
            remaining = deadline - min_time_remaining - time.monotonic()
            if remaining < 10:
                break
            if n <= 8192:
                est_per_iter = 0.004
            elif n <= 16384:
                est_per_iter = 0.01
            else:
                est_per_iter = 0.025 * (n / 32768.0)
            max_iter = min(5000, max(30, int(remaining / (est_per_iter * 10))))
            try:
                result = minimize(
                    _ls_obj, z,
                    jac=_ls_grad,
                    args=(beta, n),
                    method='L-BFGS-B',
                    options={
                        'maxiter': max_iter,
                        'ftol': 1e-18,
                        'gtol': 1e-12,
                        'maxcor': 80,
                        'maxls': 100,
                        'maxfun': max_iter * 3,
                    }
                )
                a_new = _softmax(result.x)
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass

    return current_a


def _run_lbfgs_lp(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128, 256, 512, 1024]
    for p in p_values:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        if n <= 8192:
            est_per_iter = 0.004
        elif n <= 16384:
            est_per_iter = 0.01
        else:
            est_per_iter = 0.025 * (n / 32768.0)
        max_iter = min(3000, max(30, int(remaining / (est_per_iter * 10))))
        try:
            result = minimize(
                _lp_obj, z,
                jac=_lp_grad,
                args=(p, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-10,
                    'maxcor': 60,
                    'maxls': 50,
                    'maxfun': max_iter * 3,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump_sin2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump_sin4":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / 0.1)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump_quartic":
        t = np.linspace(0, 1, n)
        x = 2 * t - 1
        a = (1 - x ** 2) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "plateau":
        a = np.ones(n)
        a[n // 4:3 * n // 4] = 1.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    - Aggressive L-BFGS with log-sum-exp continuation + reheating
    - Lp-norm smoothing alternative
    - Projected subgradient
    - Peak-targeted refinement
    - Active-set Newton (multiple tolerance levels)
    - Multi-n exploration (16384, 32768)
    - Diverse restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        sc_parent = _score(parent)
        print(f"No parent, using uniform n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    # Phase 1: Aggressive L-BFGS from parent
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs(parent, deadline - 600, min_time_remaining=400.0, aggressive=True)
        update_best(a1, _score(a1), "lbfgs_aggr")

    # Phase 2: Lp-norm L-BFGS
    print(f"\n[Phase 2] Lp-norm L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a2 = _run_lbfgs_lp(best_seq, deadline - 200, min_time_remaining=100.0)
        update_best(a2, _score(a2), "lp_lbfgs")

    # Phase 3: Peak-targeted refinement
    print(f"\n[Phase 3] Peak refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a3, sc3 = _peak_refine(best_seq, min(deadline - 80, time.monotonic() + 80), max_rounds=100, n_peaks=8)
        update_best(a3, sc3, "peak_refine_8")
        if time.monotonic() < deadline - 60:
            a3b, sc3b = _peak_refine(best_seq, min(deadline - 40, time.monotonic() + 50), max_rounds=80, n_peaks=3)
            update_best(a3b, sc3b, "peak_refine_3")

    # Phase 4: Projected subgradient
    print(f"\n[Phase 4] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a4, sc4 = _run_subgrad(best_seq, min(deadline - 50, time.monotonic() + 60), max_steps=3000, top_k=8, lr_init=0.015)
        update_best(a4, sc4, "subgrad")

    # Phase 5: Active-set Newton (loose)
    print(f"\n[Phase 5] Active-set Newton loose (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a5, sc5 = _active_set_newton(best_seq, min(deadline - 30, time.monotonic() + 50), max_iter=150, tol_active=0.002)
        update_best(a5, sc5, "as_newton_loose")

    # Phase 6: Try n=32768
    print(f"\n[Phase 6] n=32768 exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        n_large = 32768
        a_large_init = _interpolate_to_n(best_seq, n_large) if best_seq is not None else _make_profile(n_large, "uniform")
        sc_large_init = _score(a_large_init)
        print(f"  n={n_large} init score: {sc_large_init:.10f}")
        update_best(a_large_init, sc_large_init, f"interp_n{n_large}")

        if time.monotonic() < deadline - 100:
            a_large = _run_lbfgs(a_large_init, deadline - 80, min_time_remaining=50.0, aggressive=True)
            update_best(a_large, _score(a_large), f"lbfgs_n{n_large}")

        if best_seq is not None and time.monotonic() < deadline - 50:
            a_large2, sc_large2 = _peak_refine(a_large, min(deadline - 30, time.monotonic() + 30), max_rounds=50, n_peaks=5)
            update_best(a_large2, sc_large2, f"peak_n{n_large}")

        if best_seq is not None and time.monotonic() < deadline - 20:
            a_large3, sc_large3 = _active_set_newton(a_large, min(deadline - 10, time.monotonic() + 20), max_iter=80, tol_active=0.001)
            update_best(a_large3, sc_large3, f"as_newton_n{n_large}")

    # Phase 7: Structured initializations
    print(f"\n[Phase 7] Structured inits (t={time.monotonic()-t_start:.1f}s)")
    structured_profiles = ["bump_sin2", "bump_sin4", "bump_gauss", "bump_quartic"]
    for kind in structured_profiles:
        if time.monotonic() > deadline - 80:
            break
        a_struct = _make_profile(n_parent, kind, rng)
        sc_struct = _score(a_struct)
        print(f"  {kind}: score={sc_struct:.10f}")
        if sc_struct < best_score + 0.05 and time.monotonic() < deadline - 40:
            a_struct_opt = _run_lbfgs(a_struct, deadline - 20, min_time_remaining=20.0, aggressive=True)
            update_best(a_struct_opt, _score(a_struct_opt), f"struct_{kind}")

    # Phase 8: Diverse restarts from current best
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_sigs = [0.02, 0.05, 0.10]
        for sigma in restart_sigs:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            if sc_start < best_score + 0.01:
                a_r = _run_lbfgs(p_start, deadline - 5, min_time_remaining=5.0)
                update_best(a_r, _score(a_r), f"restart_s{sigma}")

    # Phase 9: Active-set Newton (tight) - final polish
    print(f"\n[Phase 9] Active-set Newton tight (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a9, sc9 = _active_set_newton(best_seq, min(deadline - 5, time.monotonic() + 30), max_iter=200, tol_active=0.0005)
        update_best(a9, sc9, "as_tight")

    # Phase 10: Very tight active-set polish
    if best_seq is not None and time.monotonic() < deadline - 3:
        a10, sc10 = _active_set_newton(best_seq, deadline - 1, max_iter=150, tol_active=0.0001)
        update_best(a10, sc10, "as_very_tight")

    # Phase 11: Final peak refinement
    if best_seq is not None and time.monotonic() < deadline - 2:
        a11, sc11 = _peak_refine(best_seq, deadline - 1, max_rounds=50, n_peaks=3)
        update_best(a11, sc11, "final_peak")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
