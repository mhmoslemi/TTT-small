def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to work with sum = 1

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w * target_sum

def _nesterov_subgrad(a, deadline, min_time_remaining=30.0, active_frac=0.02):
    """Nesterov-accelerated proximal gradient for min max_k c[k] s.t. a>=0, sum(a)=1."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    t = 1.0
    step_size = 1.0 / (2.0 * n * a.max())  # initial step
    no_improve = 0
    max_stall = 200
    
    while time.monotonic() + min_time_remaining < deadline:
        # Nesterov extrapolation
        momentum = (t - 1) / (t + 2)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
            y_sum = 1.0
        y = y / y_sum
        
        # Compute convolution
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_stall:
                break
        
        # Determine active set
        n_active = max(3, int(n * active_frac))
        c_sorted_idx = np.argsort(c)[::-1]
        active_indices = c_sorted_idx[:n_active]
        active_mask = np.zeros(2 * n - 1)
        active_mask[active_indices] = 1.0 / n_active
        
        # Compute subgradient
        g = _subgradient_f(y, active_mask)
        
        # Normalize gradient
        g_mean = g.mean()
        g_centered = g - g_mean  # Center to respect the sum constraint
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        alpha = step_size
        improved = False
        for _ in range(50):
            y_try = y - alpha * g_centered
            a_try = _proj_simplex(y_try, 1.0)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                step_size = alpha * 1.05
                break
            alpha *= 0.8
        
        if not improved:
            step_size *= 0.7
            if step_size < 1e-15:
                break
        
        a_prev = a if improved else a_prev
        t += 1
    
    return best_a, best_sc


def _multiplicative_shave(a, deadline, min_time_remaining=30.0):
    """Multiplicative peak shaving: a[i] *= (1 - eta * contribution_to_peaks[i]).
    Preserves positivity by construction."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    eta = 0.01
    active_frac = 0.01
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
        
        # Active set
        n_active = max(3, int(n * active_frac))
        active_idx = np.argsort(c)[::-1][:n_active]
        
        # Compute contribution of each a[i] to the active peaks
        # a[i] contributes to c[k] for k in [i, i+n-1] (as a[i]*a[k-i])
        # and for k in [0, 2i] (as a[k-i]*a[i])
        # Total contribution: sum_{k active} w_k * (a[k-i] + a[i]) * 1_{valid}
        # = sum_{k active} w_k * 2 * a[k-i] * 1_{valid} (from our formula)
        
        # Use the subgradient computation
        active_mask = np.zeros(2 * n - 1)
        active_mask[active_idx] = 1.0 / n_active
        g = _subgradient_f(a, active_mask)  # This gives 2 * contribution
        
        # Multiplicative update
        # a_new[i] = a[i] * exp(-eta * g[i] / c_max)
        # This is a multiplicative gradient step in the log domain
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * g / c_max
        log_a -= log_a.mean()  # Keep the scale
        a = np.exp(np.clip(log_a, -50, 50))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Adapt eta and active_frac
        if current_sc > best_sc:
            eta *= 0.8
        else:
            eta = min(0.05, eta * 1.1)
            active_frac = min(0.1, active_frac * 1.05)
    
    return best_a, best_sc


def _make_asymmetric_profile(n, skew=0.3):
    """Create an asymmetric profile: a[i] = exp(skew * (2*i/n - 1)) * cos^0.5(pi*i/n)"""
    t = np.linspace(0, 1, n)
    shape = np.exp(skew * (2 * t - 1)) * (np.cos(np.pi * t / 2)) ** 0.5
    shape = np.maximum(shape, 1e-10)
    return shape / shape.sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "symmetric_taper":
        t = np.linspace(0, 1, n)
        a = (np.cos(np.pi * t / 2)) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_exp":
        t = np.linspace(0, 1, n)
        a = np.exp(0.4 * (2 * t - 1)) * (1.0 - 0.3 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = (np.cos(np.pi * t / 2)) ** 0.5 * np.exp(0.3 * (2*t-1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raise_cos_asym":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * (t - 0.4))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    c_max = float(c_cur.max())
    if c_max < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.003)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Objective: 2n * logsumexp(c/beta) * beta (approximation of 2n * max(c))
        # Actually: 2n * (c_max + beta * log(sum(exp((c-c_max)/beta))))
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 15))))
        
        def obj(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def grad(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            return 4.0 * n * a * (h - W)
        
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-16,
                                      'gtol': 1e-13, 'maxcor': 80, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _refine_qp(a, deadline, min_time_remaining=20.0):
    """QP-based final refinement using cvxpy if available."""
    n = len(a)
    if not HAS_CVXPY:
        return a, _score(a)
    
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Try a few QP steps
    for _ in range(5):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        n_active = max(3, int(n * 0.005))
        active_idx = np.argsort(c)[::-1][:n_active]
        target = c_max * 0.995
        
        # Build constraint matrix
        B_rows = []
        rhs = []
        for k in active_idx:
            i_lo = max(0, k - (n - 1))
            i_hi = min(n -
