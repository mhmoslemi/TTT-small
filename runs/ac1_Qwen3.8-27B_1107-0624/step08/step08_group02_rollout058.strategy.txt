def _make_profile(n, kind="uniform"):
    """Generate a normalized profile sequence"""
    if kind == "uniform":
        return np.ones(n)/n
    elif kind == "tapered":
        t = np.linspace(0,1,n)
        a = np.sin(np.pi*t)**0.7
        a = np.maximum(a, 1e-10)
        return a/a.sum()

        a = np.exp(-3.0*t)

def _simplex_proj(a):
    """Project 'a' onto a simplex"""
    n = len(a)
    y = np.sort(a)[::-1]
    for i in range(n):
        if y[i] < 0 if i == 0 else y[i] < y[0] + (np.sum(y) - np.sum(y[0:i+1]) * (i+1)/np.sum(y)):
            y[i] -= (np.sum(y) - np.sum(y[0:i+1]) * (i+1))/np.sum(y)
    return y
