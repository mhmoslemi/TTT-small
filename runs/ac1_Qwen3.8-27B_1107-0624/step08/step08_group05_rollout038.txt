1. Use LSE + L-BFGS to get into the right basin (find the correct active set).
2. Use Newton active set to converge to the optimum within that basin.
3. Use stochastic perturbations to try different basins.
4. Use multi-n to try different problem sizes.

Let me now write the code. I'll focus on:
1. A clean, efficient implementation of the FFT-based convolution and mat-vec.
2. An adaptive Newton active set method with preconditioning.

4. Stochastic restarts.

One more thought: the previous code uses `n_parent` which is 16384. Let me also try n = 32768 to see if a longer sequence gives a better score. The score has a factor of n in the numerator, so longer sequences have a "penalty" but also have more degrees of freedom to reduce the peak.

Actually, the score is 2n * max(c) / (sum a)^2. For a sequence of length n with sum = 1, the "best possible" max(c) is at least 1/(2n-1) (from the average). So the best possible score is at least 2n/(2n-1) ≈ 1 + 1/(2n-1). For n = 16384, this is ≈ 1.00003. For n = 32768, it's ≈ 1.000015. So longer sequences have a slightly better theoretical lower bound, but the practical improvement is very small.

However, the practical optimum for non-negative sequences is much above 1. The question is whether a longer sequence can achieve a lower practical optimum. I think the answer is yes, but the improvement diminishes with n.

"""
Multi-phase optimizer for step function autocorrelation peak minimization.
Uses adaptive Newton active-set with FFT-based Hankel operations,
LSE-smoothed L-BFGS for basin finding, stochastic restarts,
and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities (top-level, no closures)
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hess_matvec(v, active, n):
    """Compute H*v where H[i,j] = 2 * active[i+j] (Hankel structure)."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _active_gradient(a, c, n, tol_frac):
    """Compute gradient of sum of active convolution values."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_conc":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.05 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 1e-15)
    return a_target / a_target.sum()

def _cg_solve_precon(hess_func, b, diag_precon, tol, max_iter):
    """Preconditioned CG for solving H*x = b."""
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    z = r / np.maximum(diag_precon, 1e-20)
    p = z.copy()
    rz = float(np.dot(r, z))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rz / pH
        x = x + alpha * p
        r = r - alpha *Hp
        r_norm = float(np.sqrt(np.dot(r, r)))
        if r_norm < tol * b_norm:
            break
        z_new = r / np.maximum(diag_precon, 1e-20)
        rz_new = float(np.dot(r, z_new))
        if abs(rz) < 1e-30:
            break
        beta_cg = rz_new / rz
        p = z_new + beta_cg * p
        rz = rz_new
    return x

def _newton_active_set(a_init, deadline, max_iter, tol_frac, mu_base):
    """
    Newton active set method with adaptive tolerance, preconditioning,
    and damped line search.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-30)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    active_f = None
    prev_active = None
    patience = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance: tighten as we converge
        tol = max(tol_frac * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Check if active set changed
        if prev_active is not None:
            if not np.array_equal(active, prev_active):
                patience = 0
        prev_active = active.copy()
        active_f = active.astype(np.float64)
        
        # Compute gradient of sum(active c[k])
        # grad_a[i] = 2 * sum_{k in active} a[k-i] (with appropriate indexing)
        # Using the mat-vec trick: grad = H * a / |active|
        # Actually, the gradient of sum_{k in active} c[k] w.r.t. a is:
        # g[i] = 2 * sum_{k in active: 0<=k-i<n} a[k-i]
        # This is the same as 2 * (active_convolve)
        
        # Compute gradient using the same trick as _active_gradient but for the normalized sum
        w = active.astype(np.float64) / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)  # gradient of (1/|A|) * sum_{k in A} c[k]
        
        # Project out the sum constraint
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        
        if gnorm < 1e-14:
            break
        
        # Adaptive regularization
        # Estimate condition number from the diagonal of H
        H_diag = float(np.sum(active_f))  # H[i,i] = 2*active[i] approximately
        # Actually H[i,i] = 2 * active[i] (since m+p=i means k=i, so active[i])
        # More precisely: H[i,i] = 2 * 1[i in active_set_range]
        # For our Hankel H[m,p] = 2*active[m+p], H[i,i] = 2*active[2i]
        # But the relevant ones are those where 2i <= 2n-2
        
        # Use a simple preconditioner: diagonal of H + mu
        H_diag_vec = 2.0 * active_f[0:min(2*n-1, len(active_f))]  # approximate
        # Actually let's just use a constant preconditioner
        mu = mu_base * max(1.0, c_max)
        
        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v
        
        # Preconditioner: use diagonal approximation
        # D[i] = H[i,i] + mu = 2*active[2i] + mu (for 2i < 2n-1)
        diag_precon = np.ones(n) * mu
        # Better: compute H * ones to get row sums, use as preconditioner
        ones = np.ones(n)
        H_ones = _hess_matvec(ones, active_f, n)
        diag_precon = np.maximum(H_ones, mu)
        
        # Solve (H + mu*I) dx = -grad_t
        dx = _cg_solve_precon(_hv, -grad_t, diag_precon, tol=1e-10, max_iter=150)
        dx = dx - np.mean(dx)
        
        # Line search withArmijo conditions
        f_old = float(np.sum(c[active]))
        alpha = 1.0
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.sum(c_try[active]))
            if f_try < f_old - 1e-4 * alpha * gnorm:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved and alpha < 1e-12:
            break
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 30:
                # Restart from best
                a = best_a.copy()
                patience = 0
                tol_frac = max(tol_frac * 0.5, 1e-4)
    
    return best_a, best_sc

def _lbfgs_lse(a_init, deadline, min_remaining, n_stages):
    n = len(a_init)
    current_a = a_init.copy().astype(np.float64)
    current_sc = _score(current_a)
    
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # More stages, more aggressive cooling
    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18,
                                  0.12, 0.08, 0.05, 0.03, 0.02,
                                  0.01, 0.006, 0.003, 0.0015, 0.0008])
    betas = betas[:n_stages]
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))
        
        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc

def _subgrad_polish(a_init, deadline, n_steps, lr_init, tol_frac):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        gn = float(np.linalg.norm(grad_z))
        if gn < 1e-14:
            break
        
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.95
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3
                    tol_frac = max(tol_frac * 0.5, 1e-5)
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)

def _perturb_fourier_phase(a, rng, n_modes, max_amp):
    n = len(a)
    A = np.fft.rfft(a)
    n_pert = min(n_modes, len(A))
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    # Only perturb the first n_modes
    for i in range(n_pert):
        A[i] = A[i] * np.exp(1j * phase_pert[i])
    # Keep DC and possibly Nyquist fixed
    A[0] = np.abs(A[0])
    if n % 2 == 0:
        A[-1] = np.abs(A[-1])
    a_new = np.fft.irfft(A, n)
    a_new = np.maximum(a_new, 1e-30)
    return a_new / a_new.sum()

def _perturb_block(a, rng, n_blocks):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(8, n // n_blocks)
    for _ in range(n_blocks // 2):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

def _perturb_symmetric(a, rng, sigma):
    """Perturb the symmetric part of the sequence."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_antisym = (a - a[::-1]) / 2.0
    # Perturb the antisymmetric part
    pert = sigma * rng.standard_normal(n)
    pert = (pert - pert[::-1]) / 2.0  # make it antisymmetric
    a_new = a_sym + a_antisym + pert
    a_new = np.maximum(a_new, 1e-30)
    return a_new / a_new.sum()

Multi-phase optimizer for step function autocorrelation peak minimization.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

    n = len(a)
    s = float(a.sum())

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hess_matvec(v, active_f, n):
    """Compute H*v where H[i,j] = 2*active_f[i+j] (Hankel structure via FFT)."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _active_gradient(a, c, n, tol_frac):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.18 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.1) ** 2) / 0.01) + np.exp(-((t - 0.9) ** 2) / 0.01) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "arch":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t) - 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 1e-15)
    return a_target / a_target.sum()


# ============================================================
# Preconditioned CG
# ============================================================

def _cg_solve_precon(hess_func, b, diag_precon, tol, max_iter):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    z = r / np.maximum(diag_precon, 1e-20)
    p = z.copy()
    rz = float(np.dot(r, z))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rz / pH
        if not np.isfinite(alpha) or abs(alpha) > 1e6:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_norm = float(np.sqrt(np.dot(r, r)))
        if r_norm < tol * b_norm:
            break
        z_new = r / np.maximum(diag_precon, 1e-20)
        rz_new = float(np.dot(r, z_new))
        if abs(rz) < 1e-30:
            break
        beta_cg = rz_new / rz
        if not np.isfinite(beta_cg):
            break
        p = z_new + beta_cg * p
        rz = rz_new
    return x


# ============================================================
# Newton Active Set (improved with adaptive tolerance, preconditioning)
# ============================================================

def _newton_active_set(a_init, deadline, max_iter, tol_frac, mu_base):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-30)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    prev_active = None
    patience = 0
    current_tol = tol_frac

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: tighten over iterations
        tol = max(current_tol * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Detect active set change
        if prev_active is not None and not np.array_equal(active, prev_active):
            patience = 0
            # Tighten tolerance when active set stabilizes check
        prev_active = active.copy()
        active_f = active.astype(np.float64)

        # Compute gradient of (1/|A|) * sum_{k in A} c[k]
        w = active_f / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)

        # Project out sum constraint
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))

        if gnorm < 1e-13:
            break

        # Regularization parameter
        mu = mu_base * max(1.0, c_max * n)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        # Preconditioner from row sums
        ones_vec = np.ones(n)
        H_ones = _hess_matvec(ones_vec, active_f, n)
        diag_precon = np.maximum(H_ones + mu, mu * 0.1)

        # Solve Newton system
        dx = _cg_solve_precon(_hv, -grad_t, diag_precon, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)
        dxnorm = float(np.linalg.norm(dx))

        if dxnorm < 1e-15:
            break

        # Damped line search (Armijo)
        f_old = float(np.sum(c[active]))
        alpha = 1.0
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.sum(c_try[active]))
            if f_try < f_old - 1e-4 * alpha * gnorm:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved and alpha < 1e-12:
            # Try a steepest descent step as fallback
            step = 0.01 / max(gnorm, 1e-10)
            a_try = a - step * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a = a_try

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            patience = 0
            # Gradually tighten tolerance
            if current_tol > 0.0001:
                current_tol = max(current_tol * 0.95, 0.0001)
        else:
            patience += 1
            if patience > 40:
                a = best_a.copy()
                patience = 0
                current_tol = max(current_tol * 0.7, 0.00005)
                mu_base = max(mu_base * 0.5, 1e-10)

    # Final check
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


# ============================================================
# LSE + L-BFGS
# ============================================================

def _lbfgs_lse(a_init, deadline, min_remaining, n_stages):
    n = len(a_init)
    current_a = a_init.copy().astype(np.float64)
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.4, max_c * 0.004)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Cooling schedule: aggressive early, very fine late
    ratios = np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18,
                       0.12, 0.08, 0.055, 0.035, 0.022,
                       0.014, 0.008, 0.005, 0.003, 0.0018,
                       0.001, 0.0006])
    ratios = ratios[:n_stages]
    betas = base_beta * ratios

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(6000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 150}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


# ============================================================
# Subgradient polish
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps, lr_init, tol_frac):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        gn = float(np.linalg.norm(grad_z))
        if gn < 1e-13:
            break

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 80 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.94
                if patience > 35:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3
                    tol_frac = max(tol_frac * 0.6, 1e-5)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes, max_amp):
    n = len(a)
    A = np.fft.rfft(a)
    n_pert = min(n_modes, len(A))
    for i in range(n_pert):
        A[i] = A[i] * np.exp(1j * rng.uniform(-max_amp, max_amp))
    A[0] = np.abs(A[0])
    if n % 2 == 0 and len(A) > 1:
        A[-1] = np.abs(A[-1])
    a_new = np.fft.irfft(A, n)
    a_new = np.maximum(a_new, 1e-30)
    return a_new / a_new.sum()


def _perturb_block(a, rng, n_blocks):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(16, n // n_blocks)
    for _ in range(n_blocks // 2):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_antisymmetric(a, rng, sigma):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_anti = (a - a[::-1]) / 2.0
    pert = sigma * rng.standard_normal(n)
    pert = (pert - pert[::-1]) / numpy_max_abs(pert, 1e-30)
    a_new = a_sym + a_anti + pert
    a_new = np.maximum(a_new, 1e-30)
    return a_new / a_new.sum()


def _numpy_max_abs(x, default):
    m = float(np.max(np.abs(x)))
    return m if m > default else default

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: LSE+L-BFGS for basin refinement, adaptive Newton
    active-set with preconditioned CG for local quadratic convergence,
    stochastic restarts with diverse perturbations, and multi-n exploration.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            old = _best_score
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s (improved by {old-sc:.6f})")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse check (15s) ===
    print(f"\n[Phase 1] Quick init check")
    expl_deadline = min(deadline, t_start + 15)
    n_base = n_parent if n_parent > 0 else 16384
    for kind in ["tapered", "cos_mod", "gauss", "arch"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_base, kind)
        sc_try = _score(a_try)
        if sc_try < _best_score:
            upd(a_try, sc_try, f"init_{kind}")

    # === Phase 2: LSE + L-BFGS refinement (250s) ===
    print(f"\n[Phase 2] LSE + L-BFGS refinement")
    if _best_seq is not None and time.monotonic() < deadline - 220:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=200, n_stages=14)
        upd(a_main, sc_main, "lse_main")

    # === Phase 3: Newton active set - coarse (300s) ===
    print(f"\n[Phase 3] Newton coarse")
    if _best_seq is not None and time.monotonic() < deadline - 280:
        nw_dead = deadline - 260
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=600, tol_frac=0.003, mu_base=1e-5)
        upd(a_nw, sc_nw, "newton_coarse")

    # === Phase 4: Newton active set - medium (300s) ===
    print(f"\n[Phase 4] Newton medium")
    if _best_seq is not None and time.monotonic() < deadline - 270:
        nw_dead = deadline - 250
        a_nw2, sc_nw2 = _newton_active_set(_best_seq, nw_dead, max_iter=600, tol_frac=0.001, mu_base=1e-6)
        upd(a_nw2, sc_nw2, "newton_medium")

    # === Phase 5: Newton active set - fine (300s) ===
    print(f"\n[Phase 5] Newton fine")
    if _best_seq is not None and time.monotonic() < deadline - 270:
        nw_dead = deadline - 250
        a_nw3, sc_nw3 = _newton_active_set(_best_seq, nw_dead, max_iter=800, tol_frac=0.0003, mu_base=1e-7)
        upd(a_nw3, sc_nw3, "newton_fine")

    # === Phase 6: LSE + Newton cycle (200s) ===
    print(f"\n[Phase 6] LSE+Newton cycle")
    if _best_seq is not None and time.monotonic() < deadline - 170:
        a_cur = _best_seq.copy()
        for cyc in range(3):
            if time.monotonic() > deadline - 150:
                break
            # Short LSE
            lse_d = min(deadline - 150, time.monotonic() + 30)
            a_l, sc_l = _lbfgs_lse(a_cur, lse_d,
