"""
Optimized minimizer for 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set Newton with FFT-based CG, smoothed L-BFGS, and subgradient diversification.
Key innovations: adaptive active set with merit function, preconditioned CG,
symmetric restriction option, and multi-scale exploration.

import numpy as np
import time
import sys
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_mv(v, w, n):
    """Compute H_w @ v where (H_w)_{i,j} = w[i+j] for i,j in [0,n-1]."""
    v_rev = v[::-1]
    out_len = len(v_rev) + len(w) - 1
    N = _next_pow2(out_len)
    F_v = np.fft.rfft(v_rev, N)
    F_w = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_v * F_w, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _preconditioner(w, n):
    """Diagonal preconditioner: row sums of H_w."""
    # Row i of H_w has entries w[i], w[i+1], ..., w[i+n-1]
    # Row sum for row i = sum_{j=0}^{n-1} w[i+j]
    padded = np.zeros(2 * n)
    padded[:len(w)] = w
    row_sums = np.array([padded[i:i + n].sum() for i in range(n)])
    row_sums = np.maximum(row_sums, 1e-30)
    return row_sums


def _cg_solve(w, n, rhs, max_iter=500, tol=1e-14):
    """Solve H_w @ x = rhs using preconditioned CG."""
    x = np.zeros(n)
    # Apply preconditioner
    prec = _preconditioner(w, n)
    r = rhs - _hankel_mv(x, w, n)
    z = r / prec
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_mv(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        nz = r_new / prec
        rz_new = float(r_new @ nz)
        if not np.isfinite(rz_new) or rz_new > rz * 1.05:
            break
        beta_cg = rz_new / rz if rz != 0 else 0.0
        p = nz + beta_cg * p
        x = x_new
        r = r_new
        z = nz
        rz = rz_new
        if rz < tol * tol * rz0:
            break
    return x


def _active_set_newton(a_init, deadline, max_rounds=200, verbose=False):
    """Active set Newton method with adaptive tolerance and CG inner solve."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    tol_frac = 0.001

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-18:
            break

        # Adaptive active set
        active = c >= (c_max - tol_frac * c_max)
        n_active = int(active.sum())
        if n_active < 1:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size to avoid ill-conditioning
        if n_active > n:
            idx = np.argsort(c)[::-1][:n]
            active = np.zeros_like(active)
            active[idx] = True
            n_active = n

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        # Compute gradient: grad_i = 2 * (H_w @ a)_i - 2 * mean(H_w @ a) * (for sum constraint)
        Ha = _hankel_mv(a, w, n)
        lam = float(np.mean(Ha))
        grad = 2.0 * (Ha - lam)

        # Check convergence
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14 * n:
            break

        # Newton step via CG
        H1 = _cg_solve(w, n, np.ones(n), max_iter=300, tol=1e-15)
        Hg = _cg_solve(w, n, grad, max_iter=500, tol=1e-14)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20:
            dx = -grad / max(gnorm, 1e-30)
        else:
            lam2 = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam2 * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -grad / max(gnorm, 1e-30)

        # Line search (backtracking with sufficient decrease)
        c_max_orig = c_max
        alpha = 1.0
        step_accepted = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Sufficient decrease condition
            if c_max_try < c_max_orig - 0.001 * alpha * gnorm * np.max(np.abs(dx)) + 1e-18:
                a = a_try
                step_accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_accepted:
            # Subgradient fallback
            for trial_lr in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial_lr * grad / max(gnorm, 1e-30)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if float(c_try.max()) < c_max_orig - 1e-15:

                    step_accepted = True
                    break

def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _lbfgs_smoothed(a_init, deadline, min_time_remaining=50.0, verbose=False):

        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.03)
            else:
                patience += 1
                lr = max(lr * 0.8, lr_min)
                if patience > 80:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.008

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(np.pi * t)) ** 1.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.6
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "exp_sym":
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t ** 2)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Create a plateau in the middle
        a[:int(0.2 * n)] *= 0.3
        a[int(0.8 * n):] *= 0.3
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _spectral_init(n, rng):
    """Initialize using spectral method: create A with flat-ish magnitude, IFFT, project."""
    N = _next_pow2(2 * n - 1)
    # Create a frequency response with controlled shape
    freq = np.fft.rfftfreq(N)
    # Raised cosine in frequency
    fc = 0.5
    A = np.ones(N // 2 + 1)
    for k in range(len(A)):
        if abs(freq[k]) > fc:
            A[k] = 0.0
        elif abs(freq[k]) > fc * 0.8:
            A[k] = 0.5 + 0.5 * np.cos(np.pi * (abs(freq[k]) - fc * 0.8) / (fc * 0.2))
    # Add random phase
    phase = rng.standard_normal(len(A))
    A = A * np.exp(1j * 0.1 * phase)
    A[0] = float(np.abs(A[0]))  # Keep DC real
    # IFFT
    a_full = np.fft.irfft(A, N)[:n]
    a_full = np.maximum(a_full, 0)
    s = a_full.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_full / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}", flush=True)
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}", flush=True)
        update_best(parent, _score(parent), "init")

    # === Phase 1: Profile exploration + quick optimization (0-120s) ===
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)", flush=True)
    profiles = ["tapered", "cos2", "asym", "exp_sym", "bimodal", "raised_cos", "plateau"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 600:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p

    # Spectral init
    if time.monotonic() < deadline - 500:
        a_spec = _spectral_init(n_parent, rng)
        sc_spec = _score(a_spec)
        if sc_spec < best_profile_sc:
            best_profile_sc = sc_spec
            best_profile_a = a_spec

    # Quick L-BFGS on best
    if time.monotonic() < deadline - 500:
        a_q = _lbfgs_smoothed(best_profile_a, min(deadline - 500, time.monotonic() + 90),
                              min_time_remaining=350.0)
        sc_q = _score(a_q)
        update_best(a_q, sc_q, "quick_lbfgs")
        if sc_q < best_profile_sc:
            best_profile_a = a_q
            best_profile_sc = sc_q

    # Quick active set Newton
    if time.monotonic() < deadline - 400:
        a_asn, sc_asn = _active_set_newton(best_profile_a,
                                           min(deadline - 400, time.monotonic() + 60),
                                           max_rounds=50)
        update_best(a_asn, sc_asn, "quick_asn")

    # === Phase 2: Main active set Newton (120-700s) ===
    print(f"\n[Phase 2] Main active set Newton (t={time.monotonic() - t_start:.1f}s)", flush=True)
    start_a = best_seq if best_seq is not None else best_profile_a
    a_main, sc_main = _active_set_newton(start_a, deadline, max_rounds=200, verbose=True)
    update_best(a_main, sc_main, "asn_main")

    # === Phase 3: L-BFGS polish (700-1000s) ===
    print(f"\n[Phase 3] LBFGS polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_l = _lbfgs_smoothed(best_seq, min(deadline - 300, time.monotonic() + 200),
                              min_time_remaining=200.0, verbose=True)
        update_best(a_l, _score(a_l), "lbfgs_polish")

        # Newton after L-BFGS
        if time.monotonic() < deadline - 200:
            a_np, sc_np = _active_set_newton(best_seq,
                                             min(deadline - 200, time.monotonic() + 80),
                                             max_rounds=80)
            update_best(a_np, sc_np, "post_lbfgs_asn")

    # === Phase 4: Diversification with restarts (1000-1300s) ===
    print(f"\n[Phase 4] Diversification (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if best_seq is not None:
        for i, sigma in enumerate([0.03, 0.1, 0.2]):
            if time.monotonic() > deadline - 200:
                break
            a_pert = _perturb_log(best_seq, sigma, rng)
            sc_pert = _score(a_pert)
            if sc_pert < best_score + 0.01:
                a_opt, sc_opt = _active_set_newton(a_pert,
                                                   min(deadline - 200, time.monotonic() + 60),
                                                   max_rounds=50)
                update_best(a_opt, sc_opt, f"pert_{sigma}")

    # Subgradient exploration
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgradient_descent(best_seq, min(deadline - 150, time.monotonic() + 80),
                                           max_steps=3000)
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.005 and time.monotonic() < deadline - 100:
            a_np2, sc_np2 = _active_set_newton(a_sg, min(deadline - 100, time.monotonic() + 50),
                                                max_rounds=40)
            update_best(a_np2, sc_np2, "subgrad_asn")

    # === Phase 5: Multi-n exploration (1300-1600s) ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)", flush=True)
    base_n = len(best_seq) if best_seq is not None else n_parent
    n_candidates = []
    if base_n >= 8192:
        n_candidates = [32768, base_n]
    else:
        n_candidates = [base_n, 16384, 32768]
    n_candidates = [n for n in n_candidates if n != base_n] + [base_n]

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 100:
            break
        print(f"  Trying n={n_try}...", flush=True)
        a_init = _resample(best_seq if best_seq is not None else parent, n_try)
        # Quick optimize
        a_opt, sc_opt = _active_set_newton(a_init,
                                           min(deadline - 100, time.monotonic() + 80),
                                           max_rounds=60)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 60:
            a_l2 = _lbfgs_smoothed(a_opt, min(deadline - 60, time.monotonic() + 40),
                                    min_time_remaining=40.0)
            update_best(a_l2, _score(a_l2), f"n_{n_try}_lbfgs")
            if time.monotonic() < deadline - 30:
                a_np3, sc_np3 = _active_set_newton(a_l2, min(deadline - 30, time.monotonic() + 25),
                                                    max_rounds=40)
                update_best(a_np3, sc_np3, f"n_{n_try}_final")

    # === Phase 6: Final ultra-fine polish (1600-1770s) ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-tight active set
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 2, max_rounds=100)
        update_best(a_fin, sc_fin, "final_asn")

        # One more L-BFGS at very small beta
        if best_seq is not None and time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _fft_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-9
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                try:
                    max_iter = min(3000, max(20, int(remaining / 0.01)))
                    result = minimize(
                        lambda zz: _smoothed_obj(zz, beta_fin, n),
                        z,
                        jac=lambda zz: _smoothed_grad(zz, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-17,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Last Newton
        if best_seq is not None and time.monotonic() < deadline - 1:
            a_veryfin, sc_veryfin = _active_set_newton(best_seq, deadline - 0.5, max_rounds=50)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
