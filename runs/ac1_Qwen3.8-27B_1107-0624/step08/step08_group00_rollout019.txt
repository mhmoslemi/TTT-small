"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Softmax parameterization with L-BFGS on log-sum-exp smoothed peak
2. Aggressive beta annealing with periodic perturbation for local minimum escape
3. Multi-n and multi-initialization search
4. Final exact-peak subgradient polish with line search
5. Direct peak shaving via subgradient on top-K peaks

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_peak_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _soft_peak_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SoftPeakProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _soft_peak_obj(z, self.beta, self.n)

    def grad(self, z):
        return _soft_peak_grad(z, self.beta, self.n)


def _run_lbfgs_round(a_init, beta, deadline, max_iter=5000, min_time_remaining=30.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    prob = _SoftPeakProb(beta, n)
    try:
        result = minimize(
            prob, z, jac=prob.grad, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-18,
                'gtol': 1e-15,
                'maxcor': 80,
                'maxls': 50,
            }
        )
        a_new = _softmax(result.x)
        return a_new
    except Exception:
        return a_init.copy()


def _anneal_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.005:
        gap = max_c * 0.005
    base_beta = max(gap, max_c * 0.01)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    n_betas = 25
    betas = base_beta * np.exp(np.linspace(0, -8, n_betas))

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))
        max_iter = min(max_iter, 10000)

        prob = _SoftPeakProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0:
                print(f"    beta={beta:.4e}, score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _perturb_restart(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _peak_shave_subgrad(a, top_k, lr, n_steps, deadline, min_time_remaining=10.0):
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Get top K peak indices
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / top_k

        # Compute a-space gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * h - 4.0 * W  # proportional to a-space gradient (without the 'a*' factor)

        # In a-space with constraint sum(a)=1, project gradient
        grad_proj = grad_a - np.mean(grad_a)
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-15:
            break

        # Line search
        alpha = lr
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            break

    return best_a, best_sc


def _make_init(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin":
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        f = rng.uniform(1, 20)
        a = 1.0 + 0.5 * np.sin(f * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "multi_sin":
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            f = rng.uniform(1, 50)
            ph = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.sin(f * t + ph)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        f = rng.uniform(5, 50)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * t**2 + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "raised_cosh":
        t = np.linspace(-1, 1, n)
        a = 1.0 / np.cosh(2.0 * t * rng.uniform(0.5, 3.0))
        a += 0.1
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        w = rng.uniform(0.02, 0.1)
        a = np.exp(-((t - c1)**2) / (2*w**2)) + np.exp(-((t - c2)**2) / (2*w**2)) + 0.05
        return a / a.sum()
    elif kind == "random_smooth":
        freqs = rng.uniform(1, 30, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        t = np.linspace(0, 2*np.pi, n, endpoint=False)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.sin(f * t + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Main annealing L-BFGS from parent ===
    print(f"\n[P1] Annealing L-BFGS from parent (t={time.monotonic()-t_start:.1f}s)")
    t1_start = time.monotonic()
    a_opt = _anneal_lbfgs(parent, deadline, min_time_remaining=800)
    sc_opt = _score(a_opt)
    if sc_opt < best_score:
        best_seq = a_opt.copy()
        best_score = sc_opt
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [P1]")
    else:
        print(f"  P1: no improvement (best={best_score:.10f})")
    print(f"  P1 done in {time.monotonic()-t1_start:.1f}s")

    # === Phase 2: Peak shaving + perturbation loop ===
    print(f"\n[P2] Peak shave + perturb (t={time.monotonic()-t_start:.1f}s)")
    t2_start = time.monotonic()
    n_curr = len(best_seq)
    a_work = best_seq.copy()
    stall = 0

    for iteration in range(300):
        if time.monotonic() + 400 > deadline:
            break

        # Peak shave with top-K
        top_k = max(3, int(n_curr * 0.002))
        lr = 0.02
        a_work, sc_work = _peak_shave_subgrad(
            a_work, top_k, lr, n_steps=100,
            deadline=deadline, min_time_remaining=400
        )
        if sc_work < best_score - 1e-12:
            best_seq = a_work.copy()
            best_score = sc_work
            stall = 0
            if iteration % 10 == 0:
                print(f"  iter {iteration}: score={best_score:.10f}")
        else:
            stall += 1

        if stall >= 5:
            # Perturb and try L-BFGS
            sigma = 0.05 + 0.02 * stall
            a_pert = _perturb_restart(best_seq, sigma, rng)
            c_p = _full_convolve(a_pert, a_pert)
            beta_p = max(float(c_p.max()) * 0.005, 1e-6)
            a_pert = _run_lbfgs_round(a_pert, beta_p, deadline,
                                      max_iter=2000, min_time_remaining=400)
            sc_pert = _score(a_pert)
            if sc_pert < best_score:
                best_seq = a_pert.copy()
                best_score = sc_pert
                a_work = a_pert.copy()
                stall = 0
                print(f"  P2 perturbed: NEW BEST {best_score:.10f} at iter {iteration}")
            else:
                a_work = best_seq.copy()

        if stall > 30:
            break

    print(f"  P2 done in {time.monotonic()-t2_start:.1f}s, best={best_score:.10f}")

    # === Phase 3: Diversified initializations ===
    print(f"\n[P3] Diverse inits (t={time.monotonic()-t_start:.1f}s)")
    t3_start = time.monotonic()
    inits_to_try = [
        ("multi_sin", 0), ("chirp", 1), ("random_smooth", 2),
        ("sin", 3), ("bimodal", 4), ("raised_cosh", 5),
        ("multi_sin", 6), ("chirp", 7),
    ]
    n_try_list = [n_parent]
    if n_parent != 8192:
        n_try_list.append(8192)
    if n_parent != 32768:
        n_try_list.append(32768)

    for n_try in n_try_list:
        if time.monotonic() + 150 > deadline:
            break
        for kind, sub_idx in inits_to_try[:4]:
            if time.monotonic() + 30 > deadline - 200:
                break
            a_init = _make_init(n_try, kind, rng)
            # Quick L-BFGS
            c_i = _full_convolve(a_init, a_init)
            beta_i = max(float(c_i.max()) * 0.01, 1e-5)
            a_i = _run_lbfgs_round(a_init, beta_i, deadline,
                                   max_iter=3000, min_time_remaining=250)
            sc_i = _score(a_i)
            # Second round with smaller beta
            if time.monotonic() + 20 < deadline - 200:
                c_i2 = _full_convolve(a_i, a_i)
                beta_i2 = max(float(c_i2.max()) * 0.002, 1e-7)
                a_i = _run_lbfgs_round(a_i, beta_i2, deadline,
                                       max_iter=3000, min_time_remaining=200)
                sc_i2 = _score(a_i)
                if sc_i2 < sc_i:
                    sc_i = sc_i2
                    c_i = c_i2

            if sc_i < best_score:
                best_seq = a_i.copy()
                best_score = sc_i
                print(f"  P3 n={n_try}, {kind}_{sub_idx}: NEW BEST {best_score:.10f}")
            else:
                print(f"  P3 n={n_try}, {kind}_{sub_idx}: {sc_i:.10f} (best={best_score:.10f})")

    print(f"  P3 done in {time.monotonic()-t3_start:.1f}s")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    t4_start = time.monotonic()
    for n_new in [8192, 16384, 32768, 65536]:
        if time.monotonic() + 60 > deadline:
            break
        if n_new == len(best_seq):
            continue
        a_resampled = _resample(best_seq, n_new)
        sc_res = _score(a_resampled)
        print(f"  n={n_new}: resampled score={sc_res:.10f}")
        if time.monotonic() + 40 > deadline:
            if sc_res < best_score:
                best_seq = a_resampled.copy()
                best_score = sc_res
            continue
        c_r = _full_convolve(a_resampled, a_resampled)
        beta_r = max(float(c_r.max()) * 0.005, 1e-6)
        a_r = _run_lbfgs_round(a_resampled, beta_r, deadline,
                               max_iter=4000, min_time_remaining=60)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_seq = a_r.copy()
            best_score = sc_r
            print(f"  P4 n={n_new}: NEW BEST {best_score:.10f}")
        # Second round
        if time.monotonic() + 20 < deadline - 40:
            c_r2 = _full_convolve(a_r, a_r)
            beta_r2 = max(float(c_r2.max()) * 0.001, 1e-8)
            a_r2 = _run_lbfgs_round(a_r, beta_r2, deadline,
                                    max_iter=3000, min_time_remaining=40)
            sc_r2 = _score(a_r2)
            if sc_r2 < best_score:
                best_seq = a_r2.copy()
                best_score = sc_r2
                print(f"  P4 n={n_new} round2: NEW BEST {best_score:.10f}")

    print(f"  P4 done in {time.monotonic()-t4_start:.1f}s")

    # === Phase 5: Final aggressive polish ===
    print(f"\n[P5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    t5_start = time.monotonic()
    a_final = best_seq.copy()
    n_f = len(a_final)

    # Ultra-fine L-BFGS
    c_f = _full_convolve(a_final, a_final)
    max_cf = float(c_f.max())
    # Try several very small betas
    for beta_try in [max_cf * 0.0005, max_cf * 0.0002, max_cf * 0.0001, max_cf * 5e-5]:
        if time.monotonic() + 15 > deadline:
            break
        a_f = _run_lbfgs_round(a_final, beta_try, deadline,
                               max_iter=3000, min_time_remaining=15)
        sc_f = _score(a_f)
        if sc_f < best_score:
            best_seq = a_f.copy()
            best_score = sc_f
            a_final = a_f.copy()
            print(f"  P5 beta={beta_try:.2e}: NEW BEST {best_score:.10f}")

    # Exact peak shaving
    if time.monotonic() < deadline - 10:
        top_k_final = max(1, int(n_f * 0.0005))
        a_f2, sc_f2 = _peak_shave_subgrad(
            a_final, top_k_final, lr=0.01, n_steps=200,
            deadline=deadline, min_time_remaining=5
        )
        if sc_f2 < best_score:
            best_seq = a_f2.copy()
            best_score = sc_f2
            print(f"  P5 peak shave: NEW BEST {best_score:.10f}")

    print(f"  P5 done in {time.monotonic()-t5_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
