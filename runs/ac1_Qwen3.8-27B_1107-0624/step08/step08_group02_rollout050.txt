"""
Optimized minimizer for 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive log-sum-exp smoothing with L-BFGS, Newton active-set polish,
iterative proportional fitting (IPF), subgradient diversification,
and multi-n exploration with careful time budgeting.
Key improvements: faster Newton via FFT-based CG, IPF refinement step,
sigmoid parameterization option, and smarter multi-restart strategy.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _sigmoid_01(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -500, 500)))


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _newton_active_polish(a_init, deadline, max_rounds=80):
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        active = None
        for tol_frac in [0.00005, 0.0002, 0.0008, 0.003, 0.01]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() >= 1:
                break
        if active is None or active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        step_done = False
        alpha = 1.0
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.003 * alpha) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break
                # Also try along +g_t direction (ascent for subgradient)
                a_try2 = a + trial * g_t / max(gn, 1e-30)
                a_try2 = np.maximum(a_try2, 0)
                s_try2 = a_try2.sum()
                if s_try2 < 1e-15:
                    continue
                a_try2 = a_try2 / s_try2
                c_try2 = _full_convolve(a_try2, a_try2)
                if float(c_try2.max()) < c_max - 1e-15:
                    a = a_try2
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _ipf_refine(a_init, deadline, max_iter=500):
    """Iterative Proportional Fitting to flatten the convolution."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    prev_sc = best_sc

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find the active region (where conv is close to max)
        tol = 0.01 * c_max
        active_mask = c >= (c_max - tol)
        if active_mask.sum() == 0:
            active_mask[int(np.argmax(c))] = True

        # Compute the "excess" at each position
        # a[j] contributes to conv[k] via a[j]*a[k-j]
        # We want to reduce a[j] where it contributes most to the peak
        a_rev = a[::-1]
        weighted = _full_convolve(active_mask.astype(float), a_rev)
        h = weighted[n - 1:2 * n - 1]
        h_max = float(h.max())
        if h_max < 1e-30:
            break

        # Multiplicative update: reduce where contribution is high
        damping = 0.3 + 0.2 * np.sqrt(it / max(max_iter, 1))
        update = 1.0 - damping * h / h_max
        a = a * np.maximum(update, 0.1)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 20 == 19:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()
            elif prev_sc - sc < -1e-10:
                # Score got worse, dampen more
                pass
            prev_sc = sc

    return best_a


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 5e-10
    prev_sc = current_sc
    stagnation = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_per_iter = 0.004 * scale
        max_iter = min(12000, max(20, int(remaining / (est_per_iter * 20))))

        try:
            class ObjFn:
                def __init__(self, b, nn):
                    self.b = b
                    self.nn = nn
                def __call__(self, x):
                    return _smoobj(x, self.b, self.nn)

            class GradFn:
                def __init__(self, b, nn):
                    self.b = b
                    self.nn = nn
                def __call__(self, x):
                    return _smograd(x, self.b, self.nn)

            obj_fn = ObjFn(beta, n)
            grad_fn = GradFn(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 250}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                stagnation = 0
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                stagnation += 1
        except Exception:
            stagnation += 1

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 5e-7 * c_max_new or stagnation >= 2:
            beta *= 0.12
        elif new_gap < 5e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.0003)
        if stagnation >= 3:
            break

    return current_a


def _subgrad_polish(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-11:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.05)
            else:
                patience += 1
                lr *= 0.8
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.015

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.5 + 0.01
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 2.5 + 0.01
        return a / a.sum()
    elif kind == "step_like":
        a = np.ones(n)
        third = n // 3
        a[:third] = 1.0
        a[third:2*third] = 0.6
        a[2*third:] = 0.3
        return a / a.sum()
    elif kind == "gaussian_offset":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.4) ** 2) + 0.005
        return a / a.sum()
    elif kind == "wide_gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = int(0.15 * n)
        a[:edge] = np.linspace(0.3, 1.0, edge)
        a[-edge:] = np.linspace(0.3, 1.0, edge)[::-1]
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _two_phase_optimize(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS then Newton polish in sequence."""
    n = len(a_init)
    # Phase A: L-BFGS
    a_lb = _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=min_time_remaining)
    sc_lb = _score(a_lb)

    # Phase B: Newton polish
    if time.monotonic() < deadline - 5:
        a_np, sc_np = _newton_active_polish(a_lb, deadline, max_rounds=60)
        if sc_np < sc_lb:
            return a_np
    return a_lb


def _cycle_optimize(a_init, deadline, min_time_remaining=50.0):
    """Alternate L-BFGS and Newton with increasing cycle length."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    cycle_time = min(30.0, (deadline - min_time_remaining - time.monotonic()) / 8.0)
    if cycle_time < 3:
        return current_a

    for cycle in range(10):
        if time.monotonic() + min_time_remaining > deadline:
            break
        t_end = min(time.monotonic() + cycle_time, deadline - min_time_remaining)

        # L-BFGS burst
        if time.monotonic() < t_end - 3:
            a_lb = _run_adaptive_lbfgs(current_a, t_end - 3, min_time_remaining=5.0)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb

        # Newton polish
        if time.monotonic() + 5 < min(t_end, deadline - min_time_remaining + 5):
            a_np, sc_np = _newton_active_polish(
                current_a, min(t_end, deadline - min_time_remaining + 5), max_rounds=30)
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np

        cycle_time *= 1.3
        if cycle_time > 90:
            cycle_time = 90

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "asym_taper")
        print(f"No parent, using asym_taper n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Quick diverse exploration ---
    print(f"\n[Phase 1] Diverse exploration (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["asym_taper", "tapered", "left_heavy", "right_heavy",
                "gaussian_offset", "exp_decay", "flat_top", "bimodal", "cos"]
    best_profile_a = parent.copy()
    best_profile_sc = best_score

    for prof_kind in profiles:
        if time.monotonic() > deadline - 1200:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': {sc_p:.10f} (new best profile)")

    # Perturbed parent
    for sigma in [0.03, 0.08, 0.15]:
        if time.monotonic() > deadline - 1100:
            break
        a_pert = _perturb(parent, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_profile_sc:
            best_profile_sc = sc_pert
            best_profile_a = a_pert

    # Quick optimization on best profile
    if time.monotonic() < deadline - 1000:
        t_q_end = min(time.monotonic() + 120, deadline - 1000)
        a_quick = _run_adaptive_lbfgs(best_profile_a, t_q_end, min_time_remaining=5.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_profile")
        if sc_quick < best_profile_sc:
            best_profile_a = a_quick
            best_profile_sc = sc_quick

    # --- Phase 2: Main optimization on best candidate ---
    print(f"\n[Phase 2] Main L-BFGS optimization (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_profile_a
    t_main_end = min(deadline - 400, time.monotonic() + 300)
    a_main = _run_adaptive_lbfgs(start_a, t_main_end, min_time_remaining=20.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Newton polish after main L-BFGS
    if best_seq is not None and time.monotonic() < deadline - 350:
        t_np_end = min(deadline - 350, time.monotonic() + 60)
        a_np1, sc_np1 = _newton_active_polish(best_seq, t_np_end, max_rounds=50)
        update_best(a_np1, sc_np1, "newton_after_main")

    # --- Phase 3: IPF refinement ---
    print(f"\n[Phase 3] IPF refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        t_ipf_end = min(deadline - 300, time.monotonic() + 60)
        a_ipf = _ipf_refine(best_seq, t_ipf_end, max_iter=300)
        sc_ipf = _score(a_ipf)
        update_best(a_ipf, sc_ipf, "ipf")
        if sc_ipf < best_score + 0.001 and time.monotonic() < deadline - 280:
            a_ipf_np, sc_ipf_np = _newton_active_polish(a_ipf, min(deadline - 280, time.monotonic() + 30), max_rounds=40)
            update_best(a_ipf_np, sc_ipf_np, "ipf_newton")

    # --- Phase 4: Cycle optimization ---
    print(f"\n[Phase 4] Cycle optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_cyc = _cycle_optimize(best_seq, min(deadline - 150, time.monotonic() + 180),
                                min_time_remaining=20.0)
        update_best(a_cyc, _score(a_cyc), "cycle")

    # --- Phase 5: Subgradient diversification ---
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 120, time.monotonic() + 80))
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.002 and time.monotonic() < deadline - 100:
            a_sg_np, sc_sg_np = _newton_active_polish(a_sg, min(deadline - 100, time.monotonic() + 40), max_rounds=40)
            update_best(a_sg_np, sc_sg_np, "subgrad_newton")

    # --- Phase 6: Multi-n exploration ---
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    base_seq = best_seq if best_seq is not None else parent
    n_candidates = [16384, 8192, 4096, 32768]
    # Remove n_parent to avoid redundant work
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.insert(0, n_parent)  # Keep parent n first

    for n_try in n_candidates[:4]:
        if time.monotonic() > deadline - 60:
            break
        a_init = _resample(base_seq, n_try)
        t_n_end = min(deadline - 40, time.monotonic() + 80)
        a_opt = _run_adaptive_lbfgs(a_init, t_n_end, min_time_remaining=10.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 30:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 20, time.monotonic() + 20), max_rounds=30)
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # --- Phase 7: Diverse restarts ---
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_strategies = [
            ("perturb_005", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_015", lambda: _perturb(best_seq, 0.1, rng)),
            ("right_heavy", lambda: _make_profile(len(best_seq), "right_heavy")),
            ("gaussian_offset", lambda: _make_profile(len(best_seq), "gaussian_offset")),
        ]
        # Replace lambdas with explicit calls to avoid lambda issue
        restart_list = [
            ("perturb_005", _perturb(best_seq, 0.03, rng)),
            ("perturb_015", _perturb(best_seq, 0.1, rng)),
            ("right_heavy", _make_profile(len(best_seq), "right_heavy")),
            ("gaussian_offset", _make_profile(len(best_seq), "gaussian_offset")),
        ]
        for i, (label, p_start) in enumerate(restart_list):
            if time.monotonic() > deadline - 20:
                break
            t_r_end = min(deadline - 15, time.monotonic() + 30)
            a_r = _run_adaptive_lbfgs(p_start, t_r_end, min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _newton_active_polish(a_r, deadline - 5, max_rounds=30)
                update_best(a_rp, sc_rp, f"restart_{label}_np")

    # --- Phase 8: Final polish ---
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=100)
        update_best(a_fin, sc_fin, "final_newton")

        # One more L-BFGS at ultra-small beta
        if time.monotonic() < deadline - 1:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-11
            remaining = deadline - 1 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(30, int(remaining / 0.005)))
                try:
                    class FinalObj:
                        def __init__(self, b, nn):
                            self.b = b
                            self.nn = nn
                        def __call__(self, x):
                            return _smoobj(x, self.b, self.nn)
                    class FinalGrad:
                        def __init__(self, b, nn):
                            self.b = b
                            self.nn = nn
                        def __call__(self, x):
                            return _smograd(x, self.b, self.nn)
                    result = minimize(
                        FinalObj(beta_fin, n), z, jac=FinalGrad(beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 100, 'maxls': 200}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_veryfin, sc_veryfin = _newton_active_polish(best_seq, deadline - 0.1, max_rounds=40)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "asym_taper")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
