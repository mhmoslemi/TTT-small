"""
Minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
1. FFT-based spectral FISTA with adaptive beta continuation
2. L-BFGS polishing with log-sum-exp smooth objective
3. Active-set equalization for final convergence
4. Multi-n exploration with interpolation
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_grad(c, a, beta, N):
    """Compute gradient of 2n*beta*log(sum exp(c[k]/beta)) w.r.t. a using FFT.
    Returns grad of length n. N is pre-computed FFT size."""
    n = len(a)
    shifted = (c - c.max()) / beta
    e = np.exp(np.clip(shifted, -700, 700))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    N_use = max(N, _next_pow2(conv_len))
    fw = np.fft.rfft(w, N_use)
    fa_r = np.fft.rfft(a_rev, N_use)
    result = np.fft.irfft(fw * fa_r, N_use)[:conv_len]
    grad = 2.0 * result[n - 1:2 * n - 1]
    return grad


def _spectral_obj(a, beta, n):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return float(2.0 * n * c_max)
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _lsobj(z, beta, n):
    a = _softmax(z)
    return _spectral_obj(a, beta, n)


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    N = _next_pow2(3 * n)
    grad = _spectral_grad(c, a, beta, N)
    W = float(np.dot(a, grad))
    return grad - W


def _fista_spectral(a_init, deadline, betas=None):
    """Spectral FISTA with continuation on beta."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s

    if betas is None:
        c0 = _full_convolve(a, a)
        cmax0 = float(c0.max())
        betas = cmax0 * np.array([0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                                   0.002, 0.001, 0.0005, 0.0002, 0.0001])

    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(3 * n)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        max_iters = min(3000, max(100, int(remaining / (0.05 * n / 4096))))

        y = a.copy()
        t_mom = 1.0
        step = 0.5 / (n ** 0.3)

        for it in range(max_iters):
            if time.monotonic() > deadline:
                break

            c_y = _full_convolve(y, y)
            grad_y = _spectral_grad(c_y, y, beta, N)
            g_t = grad_y - np.mean(grad_y)

            a_new = y - step * g_t
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            else:
                y = a_new.copy()
            a = a_new
            t_mom = t_new

            if it % 50 == 49:
                sc = _score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                # Adaptive step: try to increase if making good progress
                step *= 1.01

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        a = best_a

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_start=None):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    if beta_start is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
        beta_start = max(gap * 0.3, max_c * 0.005)

    betas = beta_start * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                    0.015, 0.007, 0.003, 0.001, 0.0005,
                                    0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 5))))

        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
            else:
                # Try with slightly different beta
                pass
        except Exception as e:
            print(f"    L-BFGS error at beta={beta:.2e}: {e}")

    return current_a, current_sc


def _active_set_equalize(a_init, deadline, max_iter=500):
    """Active-set equalization: iteratively reduce the spread of top autocorrelation values."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: top values within tolerance
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        m = int(active.sum())
        if m < 2:
            top_idx = np.argsort(c)[::-1][:5]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = 5

        if m > 4 * n:
            top_idx = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = 4 * n

        # We want to reduce max(c[active]) while keeping sum(c[active]) roughly constant
        # Direction: reduce a at positions that contribute most to the peak
        c_active = c[active]
        w = (c_active - c_active.min())
        w = np.maximum(w, 0)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Build full weight vector
        w_full = np.zeros_like(c)
        w_full[active] = w

        # Gradient direction
        a_rev = a[::-1]
        conv_len = len(w_full) + n - 1
        N_use = max(N, _next_pow2(conv_len))
        fw = np.fft.rfft(w_full, N_use)
        fa_r = np.fft.rfft(a_rev, N_use)
        result = np.fft.irfft(fw * fa_r, N_use)[:conv_len]
        direction = result[n - 1:2 * n - 1]
        direction = direction - np.mean(direction)

        # Line search
        f_old = c_max
        t = 0.1
        accepted = False
        for ls in range(30):
            a_try = a - t * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Try opposite direction or smaller step
            t = 0.01
            for ls in range(20):
                a_try = a - t * direction
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                if it > 20:
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - f_old) < 1e-14:
            break

    return best_a, best_sc


def _cg_solve_hess(hess_matvec, rhs, max_cg=200, tol=1e-12):
    """CG solver for H*x = rhs."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - hess_matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = hess_matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.005):
    """Newton method on active set KKT conditions."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set
        active = c >= (c_max - tol_frac * c_max)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())
        if m < 2 or m > min(2 * n, 65536):
            if m > min(2 * n, 65536):
                top_idx = np.argsort(c)[::-1][:min(2 * n, 65536)]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                m = int(active.sum())
            else:
                top_idx = np.argsort(c)[::-1][:10]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                m = 10

        # Equalize active values: find direction d such that (Q_k (a + t*d))[k] 
        # decreases more for larger c[k]
        # Simple approach: direction = -sum_k w_k * Q_k * a where w_k proportional to (c[k] - c_min_active)
        c_act = c[active]
        c_min_act = c_act.min()
        w_act = np.maximum(c_act - c_min_act, 0)
        w_sum = w_act.sum()
        if w_sum < 1e-30:
            w_act = np.ones(m) / m
        else:
            w_act = w_act / w_sum

        # Compute direction = sum_k w_k * (Q_k * a)
        # (Q_k * a)[i] = a[k-i] if 0 <= k-i < n
        # This is convolve(w_full, a_rev)[n-1:2n-1]
        w_full = np.zeros(2 * n - 1)
        w_full[active] = w_act
        a_rev = a[::-1]
        conv_len = (2 * n - 1) + n - 1
        N_use = _next_pow2(conv_len)
        fw = np.fft.rfft(w_full, N_use)
        fa_r = np.fft.rfft(a_rev, N_use)
        conv_res = np.fft.irfft(fw * fa_r, N_use)[:conv_len]
        direction = conv_res[n - 1:2 * n - 1]
        direction = direction - np.mean(direction)

        # Newton-like: also add a component to reduce the peak
        # (The above direction already does this since w is weighted by excess)

        # Line search
        f_old = c_max
        t = 0.5
        accepted = False
        for ls in range(40):
            a_try = a - t * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old - 1e-13 * t or ls >= 30:
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback: tiny step
            a_try = a - 0.01 * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old:
                    a = a_try
                else:
                    break
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - f_old) < 1e-13:
            break

        # Tighten tolerance
        tol_frac = max(0.001, tol_frac * 0.9)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_cos":
        # Asymmetric: shifted cosine
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.15)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_cos2":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.3)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = n // 3
        a[:block_size] = 1.0
        a[2 * block_size:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t ** 2 / 0.5)
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(-1, 1, n)
        a = np.exp(-(t + 0.3) ** 2 / 0.4)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_parent = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
                _update_best(parent, sc_parent, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "asym_cos")
        sc_parent = _score(parent)
        print(f"No parent, using asym_cos n={n_parent}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent, "init")

    # === Phase 1: Spectral FISTA from parent ===
    print(f"\n[Phase 1] Spectral FISTA (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        fista_dl = min(deadline - 400, time.monotonic() + 200)
        a_fista, sc_fista = _fista_spectral(parent, fista_dl)
        _update_best(a_fista, sc_fista, "fista_p1")

    # === Phase 2: L-BFGS polishing ===
    print(f"\n[Phase 2] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_lbg, sc_lbg = _run_lbfgs(best_seq, deadline, min_time_remaining=200.0)
        _update_best(a_lbg, sc_lbg, "lbfgs_p2")

    # === Phase 3: Active-set equalization ===
    print(f"\n[Phase 3] Active-set equalization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        as_dl = min(deadline - 200, time.monotonic() + 120)
        a_eq, sc_eq = _active_set_equalize(best_seq, as_dl, max_iter=300)
        _update_best(a_eq, sc_eq, "equalize_p3")

    # === Phase 4: Newton active set ===
    print(f"\n[Phase 4] Newton active set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        for tol_try in [0.01, 0.005, 0.002, 0.001]:
            if time.monotonic() > deadline - 150:
                break
            nt_dl = min(deadline - 150, time.monotonic() + 60)
            a_nt, sc_nt = _newton_active_set(best_seq, nt_dl, max_iter=150, tol_frac=tol_try)
            _update_best(a_nt, sc_nt, f"newton_t{tol_try}")

    # === Phase 5: L-BFGS again with tighter betas ===
    print(f"\n[Phase 5] L-BFGS tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_tight = mb * 0.001
        a_lbg2, sc_lbg2 = _run_lbfgs(best_seq, deadline, min_time_remaining=50.0,
                                       beta_start=beta_tight)
        _update_best(a_lbg2, sc_lbg2, "lbfgs_tight")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent:
            continue

        # Try a few profiles
        profiles = ["asym_cos", "raised_cos", "asym_gauss"]
        best_init = None
        best_init_sc = np.inf
        for prof_kind in profiles:
            a_try = _make_profile(n_try, prof_kind)
            sc_try = _score(a_try)
            if sc_try < best_init_sc:
                best_init_sc = sc_try
                best_init = a_try

        # Also interpolate parent
        a_interp = _interpolate_to_n(parent, n_try)
        sc_interp = _score(a_interp)
        if sc_interp < best_init_sc:
            best_init = a_interp
            best_init_sc = sc_interp

        a_opt, sc_opt = _run_lbfgs(best_init, deadline, min_time_remaining=30.0)
        _update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 20:
            a_nt, sc_nt = _newton_active_set(a_opt, min(deadline - 15, time.monotonic() + 30),
                                              max_iter=80, tol_frac=0.003)
            _update_best(a_nt, sc_nt, f"n_{n_try}_nt")

    # === Phase 7: Final polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        nt_dl = min(deadline - 3, time.monotonic() + 30)
        a_fin, sc_fin = _newton_active_set(best_seq, nt_dl, max_iter=200, tol_frac=0.0005)
        _update_best(a_fin, sc_fin, "final_nt")

        # Tight equalization
        if time.monotonic() < deadline - 3:
            eq_dl = min(deadline - 2, time.monotonic() + 20)
            a_eq2, sc_eq2 = _active_set_equalize(best_seq, eq_dl, max_iter=200)
            _update_best(a_eq2, sc_eq2, "final_eq")

        # Final L-BFGS with tiny beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_final = mb * 0.00005
            remaining = deadline - 5 - time.monotonic()
            if remaining > 5:
                z_fin = np.log(np.maximum(best_seq, 1e-30))
                z_fin = z_fin - z_fin.mean()
                max_iter = min(2000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj,
                        z_fin,
                        args=(beta_final, n),
                        jac=_lsgrad,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                  'maxcor': 100, 'maxls': 200}
                    )
                    a_fin2 = _softmax(result.x)
                    _update_best(a_fin2, _score(a_fin2), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
