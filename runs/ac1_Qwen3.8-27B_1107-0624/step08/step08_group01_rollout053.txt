"""
Step function optimization using multiplicative Optseq-like iterations,
LSE-smoothed gradient descent, active-set refinement, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    """Compute the objective value."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _mult_descend(a, deadline, min_remaining=10.0, alpha_init=0.05, 
                   alpha_min=1e-8, alpha_decay=0.999, n_no_improve_limit=5000,
                   use_sim_anneal=True, sa_temp=0.01):
    """
    Multiplicative gradient descent (Optseq-like) for minimizing max of convolution.
    Uses the gradient of the active peak lag.
    """
    n = len(a)
    a = a.copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    alpha = alpha_init
    no_improve = 0
    rng = np.random.default_rng(99)
    
    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max  # with sum=1
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active peak
        k_star = int(np.argmax(c))
        
        # Compute gradient: g[i] = 2*a[k_star - i] for valid i
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        g = np.zeros(n)
        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        g[i_range] = 2.0 * a[j_range]
        
        # Multiplicative update: a[i] *= exp(-alpha * g[i])
        # But g can be 0 for some i, so those don't change
        # Normalize the gradient for step size control
        g_norm = g.sum()
        if g_norm < 1e-15:
            break
        
        # Adaptive alpha
        if no_improve > 100:
            alpha *= alpha_decay
        
        # Update
        log_factor = -alpha * g / (g_norm / n)  # normalize gradient
        a_new = a * np.exp(np.clip(log_factor, -50, 50))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        accept = sc_new < current_sc  # greedy
        if use_sim_anneal and not accept and no_improve > 200:
            # Simulated annealing: accept worse solution with some probability
            temp = sa_temp * max(1.0, 1.0 - no_improve / n_no_improve_limit)
            if rng.random() < np.exp(-(sc_new - current_sc) / max(temp, 1e-10)):
                accept = True
        
        if accept:
            a = a_new
        else:
            alpha *= 0.95
            if alpha < alpha_min:
                # Try a random perturbation
                if no_improve > n_no_improve_limit:
                    a = _normalize(a * np.exp(0.01 * rng.standard_normal(n)))
                    alpha = alpha_init * 0.1
                    no_improve = 0
                else:
                    break
    
    return best_a, best_sc

"""
Step function optimization using multiplicative Optseq-like iterations with top-K
active sets, LSE-smoothed L-BFGS, simulated annealing, and multi-n exploration.
Key differences from previous approach: primary method is fast multiplicative
gradient descent with adaptive top-K weighting, combined with periodic
simulated-annealing restarts to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def _topk_mult_descend(a_init, deadline, min_remaining=10.0,
                       alpha_init=0.08, alpha_decay=0.9995,
                       top_k_start=3, top_k_end=20,
                       sa_enabled=True, sa_temp_init=0.005,
                       seed=42):
    """
    Multiplicative gradient descent using top-K active lags.
    Faster per-iteration than L-BFGS, good for long runs.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    rng = np.random.default_rng(seed)

    alpha = alpha_init
    no_improve = 0
    total_iters = 0

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        total_iters += 1

        # Adaptive top-K
        progress = min(1.0, no_improve / 5000.0)
        k_eff = int(top_k_start + (top_k_end - top_k_start) * progress)
        k_eff = max(1, min(k_eff, len(c)))

        # Get top-K lags
        top_indices = np.argpartition(c, -k_eff)[-k_eff:]
        top_indices = top_indices[np.argsort(c[top_indices])[::-1]]

        # Weight by proximity to max
        c_top = c[top_indices]
        margin = c_top[0] - c_top[-1]
        if margin > 1e-15:
            weights = np.exp(-(c_top[0] - c_top) / (margin + 1e-15))
        else:
            weights = np.ones(k_eff) / k_eff
        weights = weights / weights.sum()

        # Compute combined gradient
        g = np.zeros(n)
        for kk, ww in zip(top_indices, weights):
            kk = int(kk)
            k_lo = max(0, kk - n + 1)
            k_hi = min(n - 1, kk)
            if k_lo > k_hi:
                continue
            i_range = np.arange(k_lo, k_hi + 1)
            j_range = kk - i_range
            g[i_range] += 2.0 * ww * a[j_range]

        g_sum = g.sum()
        if g_sum < 1e-15:
            break

        # Adaptive alpha
        if no_improve > 50:
            alpha *= alpha_decay
        elif no_improve < 10:
            alpha = min(alpha_init, alpha * 1.02)

        # Multiplicative update
        scale = g_sum / n
        log_update = -alpha * g / max(scale, 1e-15)
        a_new = a * np.exp(np.clip(log_update, -30, 30))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new

        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        # Accept/reject with optional simulated annealing
        accept = sc_new < current_sc - 1e-15
        if not accept and sa_enabled and no_improve > 300:
            temp = sa_temp_init * max(0.01, 1.0 - no_improve / 10000.0)
            delta = sc_new - current_sc
            if delta < temp * 10 and rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            a = a_new
            if no_improve > 200:
                alpha = min(alpha_init, alpha * 1.05)
        else:
            alpha *= 0.97
            if alpha < 1e-10:
                if no_improve > 8000:
                    # Random restart from best
                    a = _normalize(best_a * np.exp(0.005 * rng.standard_normal(n)))
                    alpha = alpha_init * 0.5
                    no_improve = 0
                else:
                    break

    return best_a, best_sc


def _lse_obj_grad(z, beta, n):
    """LSE-smoothed objective and gradient in log-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)

        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


class _LseFunc:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def fun(self, z):
        obj, _ = _lse_obj_grad(z, self.beta, self.n)
        return obj

    def jac(self, z):
        _, g = _lse_obj_grad(z, self.beta, self.n)
        return g


def _lse_lbfgs(a_init, deadline, min_remaining=15.0):
    """LSE-smoothed L-BFGS with aggressive annealing schedule."""
    n = len(a_init)
    c_init = _full_convolve(a_init, a_init)
    c_max_init = float(c_init.max())
    if c_max_init < 1e-15:
        return a_init

    c_sorted = np.sort(c_init)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < c_max_init * 0.001:
        gap = c_max_init * 0.001
    base_beta = max(gap, c_max_init * 0.003)

    # Annealing schedule - more steps at the end
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
              0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    ratios = [max(r, 0.00001) for r in ratios]

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)

    for i, ratio in enumerate(ratios):
        if time.monotonic() > deadline - min_remaining:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10.0:
            break

        beta = base_beta * ratio
        scale = max(1.0, n / 8192.0)
        est_per_iter = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_per_iter * 20))))

        func = _LseFunc(beta, n)
        try:
            res = minimize(func.fun, z, jac=func.jac, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-18,
                                    'gtol': 1e-13, 'maxcor': 60, 'maxls': 40})
            a_new = _softmax(res.x)
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_refine(a_init, deadline, min_remaining=5.0):
    """
    Fine active-set gradient descent with very small steps.
    Focuses on equalizing the top few peaks.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    lr = 0.005
    no_improve = 0

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            break

        # Top-3 lags with equal weight
        top3 = np.argpartition(c, -3)[-3:]
        top3 = top3[np.argsort(c[top3])[::-1]]

        # Check if peaks are close (converged)
        peak_spread = c[top3[0]] - c[top3[-1]]
        if peak_spread < c_max * 1e-6:
            break

        # Equalize: push down the highest, push up the lowest
        g = np.zeros(n)
        for idx, lag in enumerate(top3):
            lag = int(lag)
            weight = 1.0 if idx == 0 else -0.3
            k_lo = max(0, lag - n + 1)
            k_hi = min(n - 1, lag)
            if k_lo > k_hi:
                continue
            i_range = np.arange(k_lo, k_hi + 1)
            j_range = lag - i_range
            g[i_range] += weight * 2.0 * a[j_range]

        # Project: remove mean to preserve sum
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        step = lr * min(1.0, 0.1 / g_norm)
        a_new = a + step * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new)

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc - 1e-14:
            a = a_new
            lr = min(0.02, lr * 1.05)
        else:
            lr *= 0.9
            if lr < 1e-6:
                break

    return best_a, best_sc


def _peak_shave_direct(a_init, deadline, min_remaining=5.0, n_trials=200, seed=77):
    """
    Direct peak shaving: find contributors to the max peak and redistribute.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)

    for _ in range(n_trials):
        if time.monotonic() > deadline - min_remaining:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find contributing indices
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            break

        # Reduce top contributors, add to low-contribution positions
        frac_reduce = rng.uniform(0.05, 0.15)
        n_reduce = max(1, int(len(contrib) * frac_reduce))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * rng.uniform(0.05, 0.15)
            a_try[idx] -= red
            reduction += red

        # Distribute to positions NOT in the peak's contributing range
        peak_mask = np.zeros(n, dtype=bool)
        peak_mask[i_range] = True
        peak_mask[j_range] = True
        free_mask = ~peak_mask
        if free_mask.sum() < 10:
            free_mask = np.ones(n, dtype=bool)
            free_mask[top_idx] = False
        if free_mask.sum() == 0:
            continue

        a_try[free_mask] += reduction / free_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        a_try = _normalize(a_try)

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "linear_ramp":
        t = np.linspace(1, n, n)
        a = t ** 0.5
        return a / a.sum()
    elif kind == "inverse_ramp":
        t = np.linspace(n, 1, n)
        a = t ** 0.5
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.2) ** 2) + np.exp(-50.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    return _normalize(a_new)


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    def _update_best(a_new, name):
        nonlocal best_a, best_score
        sc = _score(a_new)
        if sc < best_score:
            best_a = a_new.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [{name}] t={time.monotonic()-t_start:.1f}s")
        return sc

    # === Phase 1: Top-K multiplicative descent on parent (600s) ===
    print(f"\n[P1] Top-K multiplicative descent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 400.0
    a1, sc1 = _topk_mult_descend(best_a, p1_deadline, min_remaining=20.0,
                                 alpha_init=0.1, top_k_start=3, top_k_end=15,
                                 sa_enabled=True, sa_temp_init=0.003, seed=1)
    _update_best(a1, "mult_p1")

    # === Phase 2: LSE L-BFGS (300s) ===
    print(f"\n[P2] LSE L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(t_start + 650.0, deadline - 500.0)
    if time.monotonic() < p2_deadline - 30:
        a2 = _lse_lbfgs(best_a, p2_deadline, min_remaining=20.0)
        _update_best(a2, "lse_p2")

    # === Phase 3: Active set refinement (200s) ===
    print(f"\n[P3] Active set refine (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(t_start + 800.0, deadline - 350.0)
    if time.monotonic() < p3_deadline - 10:
        a3, sc3 = _active_set_refine(best_a, p3_deadline, min_remaining=10.0)
        _update_best(a3, "active_p3")

    # === Phase 4: Peak shaving (150s) ===
    print(f"\n[P4] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(t_start + 900.0, deadline - 250.0)
    if time.monotonic() < p4_deadline - 10:
        a4, sc4 = _peak_shave_direct(best_a, p4_deadline, min_remaining=10.0,
                                     n_trials=300, seed=123)
        _update_best(a4, "shave_p4")

    # Re-run multiplicative after shaving if it helped
    if sc4 < best_score + 0.0005 and time.monotonic() < deadline - 200:
        p4b_deadline = min(t_start + 1000.0, deadline - 200.0)
        a4b, sc4b = _topk_mult_descend(best_a, p4b_deadline, min_remaining=15.0,
                                        alpha_init=0.05, top_k_start=5, top_k_end=25,
                                        sa_enabled=True, sa_temp_init=0.002, seed=2)
        _update_best(a4b, "mult_p4b")

    # === Phase 5: Multi-n exploration (300s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 12288, 20480, 14336]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates, key=lambda x: abs(x - len(best_a))):
        if time.monotonic() > deadline - 100.0:
            break
        if n_try == len(best_a):
            continue
        remaining = deadline - time.monotonic() - 80.0
        if remaining < 40:
            break

        a_init = _resample(best_a, n_try)
        print(f"  Trying n={n_try}, init score={_score(a_init):.10f}")

        # Quick multiplicative optimization
        sub_deadline = time.monotonic() + min(100.0, remaining * 0.5)
        a_opt, sc_opt = _topk_mult_descend(a_init, sub_deadline, min_remaining=10.0,
                                            alpha_init=0.06, top_k_start=3, top_k_end=15,
                                            sa_enabled=False, seed=3)
        _update_best(a_opt, f"mult_n{n_try}")

        # Quick LSE if time permits
        if time.monotonic() < deadline - 60:
            a_opt2 = _lse_lbfgs(a_opt, deadline - 50.0, min_remaining=15.0)
            _update_best(a_opt2, f"lse_n{n_try}")

    # === Phase 6: Diverse restarts (200s) ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_best = len(best_a)
        restarts = [
            ("pert1", _perturb(best_a, 0.03, rng)),
            ("pert2", _perturb(best_a, 0.08, rng)),
            ("tapered", _make_profile(n_best, "tapered")),
            ("ramp", _make_profile(n_best, "linear_ramp")),
            ("inv_ramp", _make_profile(n_best, "inverse_ramp")),
            ("cosine", _make_profile(n_best, "cosine")),
        ]

        for label, a_r in restarts:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 20:
                break
            a_opt, sc_opt = _topk_mult_descend(a_r, deadline - 20.0, min_remaining=10.0,
                                                alpha_init=0.04, top_k_start=3, top_k_end=10,
                                                sa_enabled=False, seed=4)
            _update_best(a_opt, f"restart_{label}")

    # === Phase 7: Final polishing (remaining time) ===
    print(f"\n[P7] Final polishing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Fine active set
        a_fin, sc_fin = _active_set_refine(best_a, deadline - 3.0, min_remaining=3.0)
        _update_best(a_fin, "final_active")

        # Tiny multipulative
        remaining = deadline - time.monotonic() - 2.0
        if remaining > 10:
            a_fin2, sc_fin2 = _topk_mult_descend(best_a, deadline - 2.0, min_remaining=2.0,
                                                  alpha_init=0.01, top_k_start=3, top_k_end=5,
                                                  sa_enabled=False, seed=5)
            _update_best(a_fin2, "final_mult")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
