```python
"""
LP-based step function optimization using cutting-plane direction finding,
combined with Adam refinement and subgradient polish.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""
import numpy as np
import time
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _solve_direction_lp(a, M, active_lags, time_limit=12.0):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M for k in active_lags, g >= 0.
    Returns g or None if failed.
    """
    n = len(a)
    K = len(active_lags)
    if K == 0:
        return None

    # Build constraint matrix A where A[i, j] = a[k_i - j] if valid
    A = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    g_var = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(g_var))
    cons = [A @ g_var <= M]
    prob = cp.Problem(obj, cons)

    try:
        prob.solve(solver=cp.HIGHS, max_time=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            return None

    if g_var.value is None:
        return None

    g = np.maximum(0, np.array(g_var.value).flatten())
    return g


def _lp_iteration(a, deadline, max_k=300, time_per_lp=12.0):
    """
    One LP-based iteration: find optimal direction and update.
    Uses cutting-plane approach for constraint selection.
    Returns (new_a, improved).
    """
    n = len(a)
    w = _fft_convolve(a, a)
    M = float(w.max())

    # Select active lags: top peaks
    k_order = np.argsort(w)[::-1]
    active_lags = k_order[:max_k].copy()

    # Cutting-plane loop
    g = None
    for cp_iter in range(10):
        if time.monotonic() > deadline - 10:
            break

        g = _solve_direction_lp(a, M, active_lags, time_limit=time_per_lp)
        if g is None or g.sum() < 1e-15:
            break

        # Check all constraints
        ag = _fft_convolve(a, g)
        max_viol = float(ag.max() - M)

        if max_viol <= M * 1e-5:
            break

        # Add most violated constraint
        violated_k = int(np.argmax(ag))
        if violated_k not in active_lags:
            active_lags = np.append(active_lags, violated_k)
            # Keep top max_k+20 by w value
            all_k = np.unique(active_lags)
            w_vals = w[all_k]
            sorted_idx = np.argsort(w_vals)[::-1]
            active_lags = all_k[sorted_idx[:max_k + 20]]

    if g is None or g.sum() < 1e-15:
        return a, False

    # Line search
    best_t = 0
    best_sc = _score_norm(a)
    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
        a_cand = (a + t * g) / (1.0 + t * g.sum())
        a_cand = np.maximum(0, a_cand)
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_cand /= s
        sc = _score_norm(a_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t

    if best_t == 0:
        return a, False

    a_new = (a + best_t * g) / (1.0 + best_t * g.sum())
    a_new = np.maximum(0, a_new)
    a_new /= a_new.sum()
    return a_new, True


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=300000, eval_interval=100, active_k=5,
                   backtrack_interval=200):
    """Adam optimizer in log-space with adaptive temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    last_improve_t = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8) * current_lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if T < T_start * 0.01 and frac > 0.7:
            c_max = float(c.max())

            if len(active) == 0:
                active = np.array([np.argmax(c)])
            elif len(active) > active_k:
                top_idx = np.argsort(c[active])[::-1][:active_k]
                active = active[top_idx]
            g = _compute_gradient_at_lags(a, c, active)
        else:
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                p = np.zeros(len(c))
                p[np.argmax(c)] = 1.0
            else:
                p = e / s
            g = np.zeros(n)
            c_sorted = np.argsort(c)[::-1]
            num_active = min(active_k + 3, len(c_sorted))
            for idx in range(num_active):
                k = c_sorted[idx]
                if p[k] < c[-1] * 2:
                    break
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if lo <= hi:
                    g[lo:hi + 1] += p[k] * 2.0 * a[k - hi:k - lo + 1][::-1]

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if t % backtrack_interval == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.005:
                z_new = z.copy()
                m_buf *= 0.5
                v_buf *= 0.5
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve_t = t
            else:
                patience += 1

            if patience > 10 and t - last_improve_t > 500:
                current_lr_mult *= 0.7
                patience = 0
                if current_lr_mult < 0.03:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.01 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    current_lr_mult = 0.3

        best_a = a_final.copy()
    return best_a, best_score


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_at_lags(a, c, lags):
    n = len(a)
    c_lags = c[lags]
    T = (c_lags.max() - c_lags.min()) * 0.1 + 1e-10
    w = np.exp((c_lags - c_lags.max()) / T)
    w = w / w.sum()
    g = np.zeros(n)
    for idx, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        g[lo:hi + 1] += w[idx] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _subgradient_polish(a, deadline, max_iter=30000, top_k=5, initial_lr=0.003):
    """Direct subgradient method on the max convolution peak."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        g_tangent = g - np.sum(g) * a
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break

        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9998
        else:
            lr *= 0.99
            if lr < 1e-8:
                break

    return best_a, _score_norm(best_a)


def _active_set_equalization(a, deadline, max_iter=5000, tol=0.002):
    """Equalize top peaks by solving a linear system."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 12:
            sorted_a = active[np.argsort(c[active])[::-1][:12]]
            active = sorted_a

        K = len(active)
        M_mat = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M_mat[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        M_full = np.vstack([M_mat, np.ones(n)])
        b = np.concatenate([-np.ones(K), [0.0]])

        G = M_full @ M_full.T
        try:
            G_inv = np.linalg.inv(G + 1e-12 * np.eye(G.shape[0]))
            d = M_full.T @ (G_inv @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _coordinate_descent_polish(a, deadline, max_rounds=200):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        for j in range(lo, hi + 1, max(1, (hi - lo + 1) // 50)):
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.7, 0.85, 0.95, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        if not improved:
            any_improve = False
            step_size = max(1, n // 100)
            for j in range(0, n, step_size):
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.8, 1.25]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        any_improve = True
                        improved = True
                        break
                if improved:
                    break
            if not any_improve:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: LP-based direction finding + Adam + subgradient.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP-based iterations (0 - 600s)
    phase1_end = min(deadline, t_start + 600.0)
    print(f"Phase 1: LP-based iterations, budget={phase1_end - t_start:.0f}s")
    lp_improvements = 0
    no_improve_count = 0
    max_k_start = 200

    while time.monotonic() < phase1_end - 20:
        a_new, improved = _lp_iteration(a, phase1_end, max_k=max_k_start, time_per_lp=10.0)
        if improved:
            sc_new = _score(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                lp_improvements += 1
                no_improve_count = 0
                print(f"  LP iter {lp_improvements}: score={sc_new:.10f}, t={time.monotonic()-t_start:.0f}s")
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1

        if no_improve_count >= 3:
            # Try with more constraints
            max_k_start = min(500, max_k_start + 100)
            no_improve_count = 0
            if max_k_start >= 500:
                # LP plateaus; break
                print(f"  LP plateau at max_k=500, breaking")
                break

    full_sc = _score(best_a)
    print(f"Phase 1 done: score={full_sc:.10f}, LP_iters={lp_improvements}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Adam refinement (600 - 1000s)
    phase2_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_end - 30:
        print(f"Phase 2: Adam refinement")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_opt, sc = _adam_optimize(
            a_init=best_a,
            deadline=phase2_end,
            T_start=max_cb * 0.005,
            T_end=max_cb * 0.0001,
            lr_start=0.02,
            lr_end=0.001,
            max_iter=200000,
            eval_interval=80,
            active_k=5,
            backtrack_interval=150
        )
        full_sc = _score(a_opt)
        print(f"Phase 2: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # Phase 3: More LP iterations (1000 - 1300s)
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end - 20:
        print(f"Phase 3: LP iterations round 2")
        a = best_a.copy()
        lp_improvements2 = 0
        no_improve2 = 0
        while time.monotonic() < phase3_end - 15:
            a_new, improved = _lp_iteration(a, phase3_end, max_k=350, time_per_lp=8.0)
            if improved:
                sc_new = _score(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    a = a_new
                    lp_improvements2 += 1
                    no_improve2 = 0
                    print(f"  LP2 iter {lp_improvements2}: score={sc_new:.10f}, t={time.monotonic()-t_start:.0f}s")
                else:
                    no_improve2 += 1
            else:
                no_improve2 += 1
            if no_improve2 >= 4:
                break

    # Phase 4: Active-set equalization (1300 - 1450s)
    phase4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase4_end - 10:
        print(f"Phase 4: Active-set equalization")
        a_as, sc_as = _active_set_equalization(best_a, phase4_end, max_iter=3000, tol=0.003)
        full_sc_as = _score(a_as)
        print(f"Phase 4: score={full_sc_as:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_as < best_score:
            best_score = full_sc_as
            best_a = a_as.copy()

    # Phase 5: Subgradient polish (1450 - 1600s)
    phase5_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase5_end - 10:
        print(f"Phase 5: Subgradient polish")
        a_sg, sc_sg = _subgradient_polish(best_a, phase5_end, max_iter=20000, top_k=4, initial_lr=0.002)
        full_sc_sg = _score(a_sg)
        print(f"Phase 5: score={full_sc_sg:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_sg < best_score:
            best_score = full_sc_sg
            best_a = a_sg.copy()

    # Phase 6: Final Adam at very low temperature (1600 - 1720s)
    phase6_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < phase6_end - 10:
        print(f"Phase 6: Final fine Adam")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_opt2, sc2 = _adam_optimize(
            a_init=best_a,
            deadline=phase6_end,
            T_start=max_cb * 0.001,
            T_end=max_cb * 0.00001,
            lr_start=0.005,
            lr_end=0.0005,
            max_iter=150000,
            eval_interval=50,
            active_k=4,
            backtrack_interval=100
        )
        full_sc2 = _score(a_opt2)
        print(f"Phase 6: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # Phase 7: Coordinate descent + final subgradient (1720 - deadline)
    phase7_end = deadline - 5
    if time.monotonic() < phase7_end:
        print(f"Phase 7: Final polish")
        a_cd, sc_cd = _coordinate_descent_polish(best_a, phase7_end - 10, max_rounds=100)
        full_sc_cd = _score(a_cd)
        if full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

        if time.monotonic() < phase7_end - 3:
            a_sg2, sc_sg2 = _subgradient_polish(best_a, phase7_end, max_iter=5000, top_k=3, initial_lr=0.001)
            full_sc_sg2 = _score(a_sg2)
            if full_sc_sg2 < best_score:
                best_score = full_sc_sg2
                best_a = a_sg2.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
```
