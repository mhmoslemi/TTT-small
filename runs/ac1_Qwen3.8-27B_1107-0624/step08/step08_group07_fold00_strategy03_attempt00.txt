import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import rfft, irfft

def next_pow2(x):
    """Returns the smallest power of two >= x."""
    k = 1
    while k < x:
        k *= 2
    return max(2, k)

def linear_convolve(a, b):
    """Compute full linear convolution using FFT for better performance."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    a_pad = np.zeros(N, dtype=np.float64)
    b_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    b_pad[:m] = b
    
    fa = rfft(a_pad)
    fb = rfft(b_pad)
    fc = np.zeros(out_n if out_n <= N//2 else int(N/2)+1, dtype=np.complex_)
    # Only multiply first half if N even
    if N % 2 == 0:
        fc[:N//2+1] = fa[:N//2+1] * fb[:N//2+1]
    else:
        fc[:int(N/2)+1] = fa[:int(N/2)+1] * fb[:int(N/2)+1]
    
    # Need to conjugate because we did rfft and then direct multiplication
    return irfft(fc.astype('complex_'))[:out_n]

def score(a):
    """Compute the score value from a sequence."""
    n = len(a)
    a_sum = np.sum(a)
    if a_sum < 1e-15:
        return np.inf
    conv = linear_convolve(a, a)

def softmax_stable(z):
    """Compute softmax that is numerically stable."""
    m = z.max()
    stabilized = (z - m)   # Shift to avoid large numbers
    exp_vals = np.exp(stabilized)
    sum_exp = exp_vals.sum()
    return exp_vals / sum_exp if sum_exp > 1e-300 else np.ones_like(z) / len(z)

    n = len(a_init)

    a0 = np.copy(a_init)
    a0_sum = np.sum(a0)

    else:
        # fallback to better initialization for small sum
        a0 = np.ones_like(a0) * 0.01
        best_score = np.inf
    
    for i in range(max_iter):
        if time.time()>=deadline:
            break
            
        # Evaluation phase
        if i % eval_interval == 0 and backtracking:
            # Compute score and gradient
            conv_full = linear_convolve(a0, a0)
            if not np.isfinite(conv_full).all():
                a0 = np.random.rand(n).astype(np.float64) * 2.0 - 1.0  # Failed to converge, try new initialization
                a0_sum = np.sum(a0)
                if a0_sum < 0.01:

                    factor = 0.01 / a0_sum if a0_sum>0 else 0.005
                    a0 = a0 * factor
                continue
                
            max_conv = np.max(conv_full)
            score_val = float(2.0 * n * max_conv / (a0_sum**2))
            
            if score_val < best_score or not np.isfinite(score_val):

                # Compute gradient

                grad_norm = np.linalg.norm(grad)
