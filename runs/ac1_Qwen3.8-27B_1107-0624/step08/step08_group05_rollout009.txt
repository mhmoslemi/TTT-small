# ============ Core utilities ============

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============ LSE objective and gradient ============

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


# ============ Power-sum objective and gradient ============

def _ps_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    val = float(np.sum(np.power(c_safe, p)) ** (1.0 / p))
    return 2.0 * n * val


def _ps_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (np.sum(np.power(c_safe, p))) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def _hess_mv(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hv_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p_vec = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hv_func(p_vec)
        pH = float(np.dot(p_vec, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if not np.isfinite(alpha) or alpha <= 0:
            break
        x = x + alpha * p_vec
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p_vec = r + beta_cg * p_vec
        rs = r_new
    return x

def _perturb_log(a, sigma, rng):

def _perturb_fourier_phase(a, rng, max_amp=0.4):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = np.maximum(amp * (1.0 + sigma * rng.standard_normal(len(amp))), 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

        sc_try = _score_fast(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 2: Main adaptive LSE at best size (500s) ===
    print(f"\n[Phase 2] Adaptive LSE main (~500s)")
    if time.monotonic() < deadline - 500:
        a2, sc2 = _lbfgs_adaptive_lse(_best_seq.copy(), deadline, min_remaining=450, n_stages=18)
        upd(a2, sc2, "lse_main")

    # === Phase 3: Newton peak shaving (200s) ===
    print(f"\n[Phase 3] Newton peak shaving (~200s)")
    if time.monotonic() < deadline - 200:
        nw_end = min(deadline - 180, time.monotonic() + 200)
        a3, sc3 = _newton_shave(_best_seq, nw_end, max_iter=200, tol_frac=0.002)
        upd(a3, sc3, "newton_main")

    # === Phase 4: Power-sum LSE (150s) ===
    print(f"\n[Phase 4] Power-sum LSE (~150s)")
    if time.monotonic() < deadline - 150:
        a4, sc4 = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=130)
        upd(a4, sc4, "ps_main")

    # === Phase 5: Subgradient polish (100s) ===
    print(f"\n[Phase 5] Subgradient polish (~100s)")
    if time.monotonic() < deadline - 100:
        sg_end = min(deadline - 85, time.monotonic() + 100)
        a5, sc5 = _subgrad_polish(_best_seq, sg_end, n_steps=6000, lr_init=0.008, tol_frac=0.001)
        upd(a5, sc5, "sg_main")

    # === Phase 6: Direct peak shave (80s) ===
    print(f"\n[Phase 6] Direct peak shave (~80s)")
    if time.monotonic() < deadline - 80:
        ps_end = min(deadline - 70, time.monotonic() + 80)
        a6, sc6 = _peak_shave(_best_seq, ps_end, max_iter=500)
        upd(a6, sc6, "peak_shave")

    # === Phase 7: Newton + SG alternating (150s) ===
    print(f"\n[Phase 7] Newton+SG alternating (~150s)")
    if time.monotonic() < deadline - 150:
        a_cur = _best_seq.copy()
        for rnd in range(4):
            if time.monotonic() > deadline - 140:
                break
            nw_e = min(deadline - 140, time.monotonic() + 30)
            a_n, sc_n = _newton_shave(a_cur, nw_e, max_iter=100, tol_frac=0.002)
            if sc_n < _score_fast(a_cur):
                a_cur = a_n
            upd(a_cur, _score_fast(a_cur), f"alt_n_{rnd}")

            sg_e = min(deadline - 140, time.monotonic() + 25)
            a_s, sc_s = _subgrad_polish(a_cur, sg_e, n_steps=2000, lr_init=0.005, tol_frac=0.001)
            if sc_s < _score_fast(a_cur):
                a_cur = a_s
            upd(a_cur, _score_fast(a_cur), f"alt_s_{rnd}")

    # === Phase 8: Multi-n exploration (350s) ===
    print(f"\n[Phase 8] Multi-n exploration (~350s)")
    n_candidates = [2048, 4096, 8192, 12288, 16384, 24576, 32768]
    if best_n not in n_candidates and best_n >= 1024:
        n_candidates.append(best_n)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq) == n_try:
            a_init_n = _best_seq.copy()
        else:
            a_init_n = _resize_seq(_best_seq, n_try)

        a_opt_n, sc_opt_n = _lbfgs_adaptive_lse(a_init_n, deadline, min_remaining=45, n_stages=12)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        if _score_fast(a_opt_n) < _best_score * 1.005 and time.monotonic() < deadline - 40:
            nw_e = min(deadline - 35, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_shave(a_opt_n, nw_e, max_iter=80, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

            if time.monotonic() < deadline - 20:
                sg_e2 = min(deadline - 15, time.monotonic() + 20)
                a_opt_n3, sc_opt_n3 = _subgrad_polish(a_opt_n2, sg_e2, n_steps=1500, lr_init=0.004, tol_frac=0.001)
                upd(a_opt_n3, sc_opt_n3, f"n_{n_try}_sg")

    # === Phase 9: Diverse restarts (250s) ===
    print(f"\n[Phase 9] Diverse restarts (~250s)")
    if time.monotonic() < deadline - 60:
        restart_configs = [
            ("log_003", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("fourier_03", lambda: _perturb_fourier_phase(_best_seq, rng, max_amp=0.3)),
            ("block_swap", lambda: _perturb_block_swap(_best_seq, rng, n_swaps=6)),
            ("amp_01", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.08)),
            ("mix", lambda: _perturb_mix(_best_seq, rng)),
            ("rescale", lambda: _perturb_rescale(_best_seq, rng)),
            ("log_005", lambda: _perturb_log(_best_seq, 0.05, rng)),
            ("fourier_05", lambda: _perturb_fourier_phase(_best_seq, rng, max_amp=0.5)),
        ]
        for i, (name, gen_fn) in enumerate(restart_configs):
            if time.monotonic() > deadline - 40:
                break
            p_start = gen_fn()
            a_r, sc_r = _lbfgs_adaptive_lse(p_start, deadline, min_remaining=20, n_stages=10)
            upd(a_r, sc_r, f"restart_{name}")

            if sc_r < _best_score * 1.004 and time.monotonic() < deadline - 15:
                nw_e = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_shave(a_r, nw_e, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{name}_nw")

    # === Phase 10: Final aggressive polish (100s) ===
    print(f"\n[Phase 10] Final polish (~100s)")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())

        # Ultra-low temp LSE
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.003)))
            try:
                result = minimize(
                    _lse_obj, z_fin, args=(beta_final, n_fin),
                    jac=_lse_grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score_fast(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_shave(_best_seq, deadline - 3, max_iter=100, tol_frac=0.001)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=4000, lr_init=0.004, tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")

        # One more round of Newton on the best
        if time.monotonic() < deadline - 1:
            a_fin4, sc_fin4 = _newton_shave(_best_seq, deadline - 0.5, max_iter=50, tol_frac=0.0005)
            upd(a_fin4, sc_fin4, "final_newton2")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print("FALLBACK: uniform n=256")
            return [float(x) for x in final]
        # Validate with the actual evaluator
        sc_final = evaluate_sequence([float(x) for x in final])
        if not np.isfinite(sc_final) or sc_final > _best_score * 1.001:
            # Fallback to a simpler sequence
            print(f"Validation mismatch! internal={_best_score:.10f}, eval={sc_final}")
            fallback = _make_profile(256, "uniform")
            return [float(x) for x in fallback]
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score_fast(fallback):.10f}")
    return [float(x) for x in fallback]
