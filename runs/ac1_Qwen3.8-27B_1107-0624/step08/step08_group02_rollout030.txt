def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r

        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        active_found = False
        for tol_frac in [5e-6, 1e-5, 5e-5, 1e-4, 3e-4, 1e-3, 3e-3]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 1:
                active_found = True
                break
        if not active_found or n_active == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / max(float(np.ones(n) @ H1), 1e-30)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try <= c_max - 1e-18:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 5e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.0025 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 5e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.12
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.15
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _annealed_lbfgs(a_init, deadline, min_time_remaining=50.0, n_restarts=5):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    total_time = deadline - min_time_remaining - time.monotonic()
    if total_time < 10:
        return current_a
    
    time_per_restart = total_time / n_restarts
    sigma = 0.15
    
    for r in range(n_restarts):
        if time.monotonic() + min_time_remaining > deadline:
            break
        phase_deadline = min(time.monotonic() + time_per_restart * (1 + 0.5 * (r + 1) / n_restarts),
                           deadline - min_time_remaining)
        
        if r > 0:
            z = np.log(np.maximum(current_a, 1e-30))
            z = z - z.mean()
            pert = sigma * np.random.default_rng(42 + r * 1000).standard_normal(n)
            z = z + pert
            z = z - z.mean()
            current_a = _softmax(z)
            current_sc = _score(current_a)
            z = np.log(np.maximum(current_a, 1e-30))
            z = z - z.mean()
            sigma *= 0.6
        
        a_new = _run_adaptive_lbfgs(current_a, phase_deadline,
                                     min_time_remaining=min(10.0, (deadline - min_time_remaining) * 0.1))
        sc_new = _score(a_new)
        if sc_new < current_sc:
            current_a = a_new
            current_sc = sc_new
    
    return current_a


def _make_power_ramp(n, p, direction=1):
    t = np.linspace(0, 1, n)
    if direction == 1:
        a = (1.0 - t) ** p + 1e-10
    elif direction == -1:
        a = t ** p + 1e-10
    else:
        a = np.sin(np.pi * t) ** p + 1e-10
    return a / a.sum()


def _make_exp_ramp(n, alpha, direction=1):
    t = np.linspace(0, 1, n)
    if direction == 1:
        a = np.exp(-alpha * t) + 1e-10
    else:
        a = np.exp(-alpha * (1 - t)) + 1e-10
    return a / a.sum()


def _make_biased_bump(n, mu, sigma, power=1.0):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - mu) / sigma) ** 2) * (1 + power * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_hybrid(n, p1, p2, blend):
    a1 = _make_power_ramp(n, p1, 1)
    a2 = _make_power_ramp(n, p2, -1)
    a = (1 - blend) * a1 + blend * a2
    return a / a.sum()


def _try_parametric_fit(a_best, deadline):
    n = len(a_best)
    t = np.linspace(0, 1, n)
    best_sc = _score(a_best)
    best_a = a_best.copy()
    
    for p in [1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0, 6.0, 8.0, 10.0]:
        if time.monotonic() > deadline:
            break
        for alpha in [1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0]:
            if time.monotonic() > deadline:
                break
            a_try = (1 - t) ** p * np.exp(-alpha * t * (1 - t)) + 1e-15
            a_try = a_try / a_try.sum()
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_power_ramp(n_parent, 2.5, 1)
        print(f"No parent, using power ramp n={n_parent}")
        update_best(parent, _score(parent), "init")

    print(f"\n[Phase 1] Diverse initialization at n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    candidates = {}
    
    init_configs = [
        ("p2.5_r1", _make_power_ramp(n_parent, 2.5, 1)),
        ("p3_r1", _make_power_ramp(n_parent, 3.0, 1)),
        ("p4_r1", _make_power_ramp(n_parent, 4.0, 1)),
        ("p5_r1", _make_power_ramp(n_parent, 5.0, 1)),
        ("p7_r1", _make_power_ramp(n_parent, 7.0, 1)),
        ("p2_r-1", _make_power_ramp(n_parent, 2.0, -1)),
        ("p3_r-1", _make_power_ramp(n_parent, 3.0, -1)),
        ("p2_sin", _make_power_ramp(n_parent, 2.0, 0)),
        ("exp3", _make_exp_ramp(n_parent, 3.0, 1)),
        ("exp5", _make_exp_ramp(n_parent, 5.0, 1)),
        ("exp7", _make_exp_ramp(n_parent, 7.0, 1)),
        ("bump_0.3", _make_biased_bump(n_parent, 0.3, 0.25, 1.0)),
        ("bump_0.5", _make_biased_bump(n_parent, 0.5, 0.3, 0.5)),
        ("hybrid_2_3", _make_hybrid(n_parent, 2.0, 3.0, 0.7)),
        ("hybrid_3_5", _make_hybrid(n_parent, 3.0, 5.0, 0.8)),
    ]

    for name, a_init in init_configs:
        if time.monotonic() > deadline - 600:
            break
        sc = _score(a_init)
        candidates[name] = (a_init, sc)
        update_best(a_init, sc, f"init_{name}")

    z_pert = np.log(np.maximum(parent, 1e-30))
    z_pert = z_pert - z_pert.mean()
    for sigma in [0.03, 0.08, 0.15]:
        if time.monotonic() > deadline - 500:
            break
        a_pert = _softmax(z_pert + sigma * rng.standard_normal(n_parent))
        sc_pert = _score(a_pert)
        update_best(a_pert, sc_pert, f"pert_{sigma}")

    best_init_a = best_seq if best_seq is not None else parent
    print(f"  Best init score: {best_score:.10f}")

    print(f"\n[Phase 2] Annealed L-BFGS from best init (t={time.monotonic() - t_start:.1f}s)")
    a_annealed = _annealed_lbfgs(best_init_a, deadline, min_time_remaining=400.0, n_restarts=4)
    update_best(a_annealed, _score(a_annealed), "annealed_lbfgs")

    print(f"\n[Phase 3] Newton polish cycle 1 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_np1, sc_np1 = _newton_active_polish(best_seq, min(deadline - 300, time.monotonic() + 120),
                                                max_rounds=100)
        update_best(a_np1, sc_np1, "newton_1")

    print(f"\n[Phase 4] Second L-BFGS pass with tighter beta (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_2nd = _run_adaptive_lbfgs(best_seq, min(deadline - 200, time.monotonic() + 120),
                                     min_time_remaining=100.0)
        update_best(a_2nd, _score(a_2nd), "lbfgs_2nd")

    print(f"\n[Phase 5] Newton polish cycle 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_np2, sc_np2 = _newton_active_polish(best_seq, min(deadline - 150, time.monotonic() + 80),
                                                max_rounds=150)
        update_best(a_np2, sc_np2, "newton_2")

    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 32768, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        x_old = np.linspace(0, 1, len(best_seq) if best_seq is not None else n_parent)
        x_new = np.linspace(0, 1, n_try)
        a_resampled = np.interp(x_new, x_old, best_seq if best_seq is not None else parent)
        a_resampled = np.maximum(a_resampled, 0)
        s_r = a_resampled.sum()
        if s_r > 0:
            a_resampled = a_resampled / s_r
        
        a_opt = _run_adaptive_lbfgs(a_resampled, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        
        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 40:
            a_np3, sc_np3 = _newton_active_polish(a_opt, min(deadline - 30, time.monotonic() + 40),
                                                   max_rounds=60)
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    print(f"\n[Phase 7] Parametric fit + re-optimize (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_param, sc_param = _try_parametric_fit(best_seq, min(deadline - 50, time.monotonic() + 30))
        update_best(a_param, sc_param, "parametric")
        
        if sc_param < best_score + 0.005:
            a_po = _run_adaptive_lbfgs(a_param, min(deadline - 30, time.monotonic() + 25),
                                        min_time_remaining=10.0)
            update_best(a_po, _score(a_po), "parametric_opt")

    print(f"\n[Phase 8] Final ultra-tight Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=300)
        update_best(a_fin, sc_fin, "final_newton")

    print(f"\n[Phase 9] Last-moment L-BFGS squeeze (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 5e-11
        remaining = deadline - 2 - time.monotonic()
        if remaining > 2:
            max_iter = min(3000, max(30, int(remaining / 0.008)))
            try:
                obj_fn = _SmoObjAdapter(beta_fin, n)
                grad_fn = _SmoGradAdapter(beta_fin, n)
                result = minimize(
                    obj_fn, z, jac=grad_fn,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-19, 'gtol': 1e-16,
                             'maxcor': 120, 'maxls': 300}
                )
                a_flb = _softmax(result.x)
                update_best(a_flb, _score(a_flb), "final_lbfgs")
            except Exception:
                pass

    if best_seq is not None and time.monotonic() < deadline - 0.5:
        a_very, sc_very = _newton_active_polish(best_seq, deadline - 0.1, max_rounds=50)
        update_best(a_very, sc_very, "very_final")

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_power_ramp(256, 2.5, 1)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_power_ramp(8192, 2.5, 1)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
