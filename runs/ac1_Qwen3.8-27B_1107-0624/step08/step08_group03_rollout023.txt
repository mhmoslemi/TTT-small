"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
1. Analytical parametric initialization (power-law, exponential ramps)
2. L-BFGS on log-sum-exp smoothed objective for coarse optimization
3. Active-set Newton with adaptive tolerance and support set handling
4. Iterative peak reduction with exact line search
5. Multi-start with diverse initializations and local perturbations
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_grad(a, w, n):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_k w[k]*a[k-i] where 0<=k-i<n.
    This equals 2 * cross_corr(w, a)[i].
    """
    w_flip = w[::-1]
    out_len = (2 * n - 1) + n - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w_flip, N)
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fw * fa, N)[:out_len]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * result[idx]


def _hess_matvec(v, w, n):
    """Compute H*v where H[j,m] = 2*w[j+m] (Hankel matrix).
    Result has length n.
    """
    w_flip = w[::-1]
    out_len = (2 * n - 1) + n - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w_flip, N)
    fv = np.fft.rfft(v, N)
    result = np.fft.irfft(fw * fv, N)[:out_len]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * result[idx]


def _ls_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = _conv_grad(a, w, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (g / 2.0 - W)

    1. Identify active peaks (where conv is close to max)

        reg = mu_reg * c_max + 1e-10

            return _hess_matvec(v, w, n) + reg * v

            r = r_new
            rs = rs_new

        dx = dx - dx.mean()

                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < best_f:
                best_f = f_try
                best_t = t
            if f_try < f_old - 1e-14 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:

def _make_power_law(n, p, direction='decreasing'):
    """Create a power-law sequence."""
    t = np.linspace(0, 1, n)
    if direction == 'decreasing':
        a = (1.0 - t + 1e-10) ** p
    else:
        a = (t + 1e-10) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _make_exponential(n, rate, direction='decreasing'):
    """Create an exponential sequence."""
    t = np.linspace(0, 1, n)
    if direction == 'decreasing':
        a = np.exp(-rate * t)
    else:
        a = np.exp(rate * t)
    s = a.sum()
    return a / s


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        update_best(a_ls, sc_ls, "local_search")

    # === Phase 6: Try larger n ===
    print(f"\n[Phase 6] Larger n exploration (t={time.monotonic()-t_start:.1f}s)")
    if n_parent < 16384 and time.monotonic() < deadline - 300:
        n_large = 16384
        a_large_init = _interpolate_to_n(best_seq if best_seq is not None else parent, n_large)
        sc_large = _score(a_large_init)
        print(f"  n={n_large} interpolated score: {sc_large:.10f}")

        if sc_large < best_score + 0.1:
            a_large = _run_lbfgs(a_large_init, deadline - 100, min_time_remaining=50.0)
            update_best(a_large, _score(a_large), f"lbfgs_n{n_large}")

            if time.monotonic() < deadline - 50:
                a_large_as, sc_large_as = _active_set_newton(a_large, deadline - 10,
                                                              max_iter=200, tol_active=1e-4)
                update_best(a_large_as, sc_large_as, f"as_n{n_large}")

    # === Phase 7: Try smaller n (sometimes helps) ===
    print(f"\n[Phase 7] Smaller n exploration (t={time.monotonic()-t_start:.1f}s)")
    if n_parent > 4096 and time.monotonic() < deadline - 100:
        for n_small in [8192, 4096]:
            if time.monotonic() > deadline - 50:
                break
            a_small = _interpolate_to_n(best_seq, n_small)
            sc_small = _score(a_small)
            print(f"  n={n_small} interpolated score: {sc_small:.10f}")
            if sc_small < best_score + 0.05:
                a_small_as, sc_small_as = _active_set_newton(a_small, deadline - 20,
                                                              max_iter=200, tol_active=1e-5)
                update_best(a_small_as, sc_small_as, f"as_n{n_small}")

    # === Phase 8: Final intensive polishing ===
    print(f"\n[Phase 8] Final polishing (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Multiple rounds of active set with decreasing tolerance
            a_cur = best_seq.copy()
            sc_cur = best_score
            for tol in [1e-4, 1e-5, 1e-6, 1e-7]:
                chunk = min(remaining / 4, 60)
                remaining -= chunk
                if remaining < 5:
                    break
                a_cur, sc_cur = _active_set_newton(a_cur, time.monotonic() + chunk,
                                                    max_iter=100, tol_active=tol)
                update_best(a_cur, sc_cur, f"polish_tol{tol}")

            # Final peak reduction
            if remaining > 10:
                a_final, sc_final = _peak_reduction(a_cur, time.monotonic() + remaining * 0.5,
                                                     max_rounds=50)
                update_best(a_final, sc_final, "final_peak")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_power_law(256, 2.0, 'decreasing')
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_power_law(256, 2.0, 'decreasing')
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
