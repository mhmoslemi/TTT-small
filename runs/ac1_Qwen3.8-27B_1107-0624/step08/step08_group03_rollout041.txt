This is a barrier-augmented objective that penalizes violations of (a*a)_k <= t. We minimize over a (in the simplex) and t simultaneously.

As mu → ∞, the solution approaches the constrained optimum.

This is similar to the log-sum-exp approach but with a different parameterization (using t explicitly).

**Approach: Use a "chasing the leader" strategy**

Instead of trying to minimize all peaks simultaneously, focus on the current maximum peak and reduce it, while not increasing other peaks too much. This is like a "greedy" or "peeling" strategy.

Algorithm:
1. Compute autocorrelation, find peak at lag k*
2. Compute the gradient direction to reduce (a*a)_{k*}
3. Take a step in that direction
4. Check that no other peak has increased above the new value of the target peak
5. If violated, reduce the step size (backtracking)
6. Repeat

This is essentially what the active-set Newton does, but simpler.

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Novel elements:
1. Spectral initialization (flat power spectrum via random phases)
2. Multi-scale optimization (small n → large n)
3. Nesterov-accelerated mirror descent
4. Weighted peak chasing with top-K peaks
5. Adaptive reheat strategy for L-BFGS
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_init(n, rng, n_peaks=200):
    """Create a non-negative sequence with approximately flat power spectrum."""
    # Generate random phases on the frequency grid
    N = _next_pow2(2 * n - 1)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    # Create the frequency-domain signal (real, non-negative in freq domain is not required)
    # We want the time-domain signal to be real and non-negative
    # Use random phases → inverse FFT → rectify
    F = np.zeros(N // 2 + 1, dtype=complex)
    F[0] = 1.0  # DC component
    F[1:] = np.exp(1j * phases[1:])
    # Hermitian symmetry
    F_full = np.zeros(N, dtype=complex)
    F_full[:N // 2 + 1] = F
    F_full[N // 2 + 1:] = np.conj(F[1:N // 2][::-1])
    # Inverse FFT
    a = np.fft.ifft(F_full).real
    # Take first n elements, rectify
    a = a[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _msequence_init(n, rng):
    """Initialize based on a smoothed binary m-sequence."""
    # Generate a pseudo-random binary sequence
    bits = rng.integers(0, 2, n).astype(np.float64) * 2 - 1  # ±1
    # Smooth with a boxcar filter
    k = max(3, n // 64)
    kernel = np.ones(k) / k
    smoothed = np.convolve(bits, kernel, mode='same')
    # Map to non-negative
    a = np.maximum(smoothed + 1, 0.01)
    return a / a.sum()


def _tapered_init(n, window='hann'):
    """Windowed (tapered) initialization."""
    if window == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif window == 'blackman':
        t = np.arange(n) / n
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lsobj_grad(z, beta, n):
    """Combined objective and gradient for L-BFGS."""
    obj = _lsobj(z, beta, n)
    grad = _lsgrad(z, beta, n)
    return obj, grad


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    """Aggressive L-BFGS with rapid beta schedule and reheat."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    # More aggressive schedule with reheat
    betas = base_beta * np.array([
        2.0, 1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
        0.015, 0.0075, 0.0037, 0.00185, 0.0009,
        0.00045, 0.0002, 0.0001, 0.00005,
        0.00002, 0.00001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    reheat_count = 0
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 2048:
            est_time_per_iter = 0.001
        elif n <= 4096:
            est_time_per_iter = 0.002
        elif n <= 8192:
            est_time_per_iter = 0.005
        elif n <= 16384:
            est_time_per_iter = 0.010
        else:
            est_time_per_iter = 0.02

        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 15))))

        try:
            def obj_func(z_local):
                return _lsobj(z_local, beta, n)
            
            def grad_func(z_local):
                return _lsgrad(z_local, beta, n)

            result = minimize(
                obj_func,
                z,
                jac=grad_func,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Reheat: increase beta and retry
                if reheat_count < 3:
                    reheat_count += 1
                    z = np.log(np.maximum(current_a, 1e-30))
                    z = z + 0.1 * np.random.default_rng(i * 1000).standard_normal(n)
                    z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.02, momentum=0.9):
    """Nesterov-accelerated mirror descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)  # velocity for momentum
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_lr_at_reheat = lr_init

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-K weighted gradient
        K = min(20, len(c) // 5)
        top_K = np.argsort(c)[::-1][:K]
        # Weight peaks by their excess over the median
        c_med = np.median(c)
        weights = np.zeros(len(c))
        for k in top_K:
            weights[k] = (c[k] - c_med) / (c_max - c_med + 1e-30)
        weights = weights / (weights.sum() + 1e-30)

        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(weights, c))
        grad = 4.0 * n * a * (h - W)

        # Nesterov step
        v = momentum * v + (1 - momentum) * grad
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * v
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    # Reheat
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    lr = lr_init * 0.7
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _peak_chase(a_init, deadline, max_rounds=200, top_k=5):
    """Directly chase and reduce the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Get top-k peaks
        top_peaks = np.argsort(c)[::-1][:top_k]
        # Weight: emphasize the highest peaks
        peak_vals = c[top_peaks]
        w = np.zeros(len(c))
        for i, k in enumerate(top_peaks):
            w[k] = (peak_vals[i] - c[top_peaks[-1]]) / (peak_vals[0] - c[top_peaks[-1]] + 1e-30) + 0.1
        w = w / w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Adaptive step size: target reducing c_max by a fixed fraction
        # Expected reduction: step * ||grad||^2 / (2 * curvature)
        # Approximate: use a fraction of c_max
        target_reduction = c_max * 0.01  # Try to reduce by 1% each step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = min(1.0, target_reduction / (grad_norm * 10))

        a_try = a - step * grad
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            step *= 0.5
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = float(c_try.max())

        if f_try < c_max:
            a = a_try
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        else:
            # Backtrack
            for ls in range(10):
                step *= 0.5
                a_try = a - step * grad
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.002):
    """Active-set Newton for the min-max problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            n_active = 3 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(30):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 29:
                if f_try <= f_old + 1e-11:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(15):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _corr_a_A(a, A_vec, n):
    """Compute the correlation: for each i, sum_{k} A_vec[k] * a[k-i] (valid range)."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _cross_over(a1, a2, alpha, rng):
    """Create a hybrid of two sequences."""
    n = len(a1)
    # Random alpha field for diversity
    alpha_field = alpha + (1 - alpha) * rng.uniform(0, 1, n) * 0.3
    alpha_field = np.clip(alpha_field, 0.3, 0.7)
    a_new = alpha_field * a1 + (1 - alpha_field) * a2
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a1.copy()
    return a_new / s


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Innovations:
1. Spectral initialization (flat power spectrum via random phases + rectification)
2. Multi-scale optimization (small n -> large n for better basins)

import numpy as np
import time
from scipy.optimize import minimize
from functools import partial


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _spectral_init(n, rng):
    """Non-negative sequence with approximately flat power spectrum."""
    N = _next_pow2(2 * n + 1)
    F = np.zeros(N // 2 + 1, dtype=complex)
    F[0] = 1.0
    phases = rng.uniform(0, 2.0 * np.pi, N // 2)
    F[1:] = np.exp(1j * phases)
    F_full = np.zeros(N, dtype=complex)
    F_full[:N // 2 + 1] = F
    if N % 2 == 0:
        F_full[N // 2 + 1:] = np.conj(F[1:N // 2][::-1])
    else:
        F_full[N // 2 + 1:] = np.conj(F[1:N // 2 + 1][::-1])
    a = np.fft.ifft(F_full).real[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _binary_smoothed_init(n, rng):
    """Smoothed binary pseudo-random sequence mapped to non-negative."""
    bits = rng.integers(0, 2, n).astype(np.float64)
    k = max(3, n // 32)
    kernel = np.ones(k) / k
    smoothed = np.convolve(bits, kernel, mode='same')
    a = np.maximum(smoothed * 0.5 + 0.25, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _tapered_init(n, kind='hann'):
    t = np.arange(n) / max(n - 1, 1)
    if kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == 'flat_top':
        a = 0.21559 + 0.41663 * np.cos(2 * np.pi * t) + 0.27726 * np.cos(4 * np.pi * t) + 0.08358 * np.cos(6 * np.pi * t) + 0.00694 * np.cos(8 * np.pi * t)
    elif kind == 'bimodal':
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _asymmetric_init(n, skew=0.3):
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 + skew * np.sin(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with aggressive beta continuation and reheat."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap_idx = min(30, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[gap_idx]
    base_beta = max(gap * 0.4, max_c * 0.008)

    betas = base_beta * np.array([
        1.5, 1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015, 0.007, 0.0035, 0.0017, 0.00085,
        0.0004, 0.0002, 0.0001, 0.00005,
        0.00002, 0.00001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    rng_reheat = np.random.default_rng(12345)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        if n <= 1024:
            est_t = 0.0005
        elif n <= 2048:
            est_t = 0.001
        elif n <= 4096:
            est_t = 0.002
        elif n <= 8192:
            est_t = 0.004
        elif n <= 16384:
            est_t = 0.008
        else:
            est_t = 0.016

        max_iter = min(5000, max(30, int(remaining / (est_t * 20))))

        obj_fn = partial(_lsobj, beta=beta, n=n)
        grad_fn = partial(_lsgrad, beta=beta, n=n)

        try:
            result = minimize(
                obj_fn,
                z,
                jac=grad_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                if verbose:
                    print(f"    beta={beta:.2e} -> score={sc_new:.10f}")
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Reheat: perturb and try with larger effective beta
                z = np.log(np.maximum(current_a, 1e-30))
                z = z + 0.05 * rng_reheat.standard_normal(n)
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_md(a_init, deadline, max_steps=5000, lr_init=0.02, momentum=0.85, top_k=15):
    """Nesterov-accelerated mirror descent with top-K peak weighting."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = min(top_k, len(c) // 4)
        top_K = np.argsort(c)[::-1][:K]
        c_min_active = c[top_K[-1]]
        w = np.zeros(len(c))
        excess = c[top_K] - c_min_active
        total_excess = excess.sum()
        if total_excess > 1e-30:
            w[top_K] = excess / total_excess
        else:
            w[top_K] = 1.0 / K

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        v = momentum * v + (1.0 - momentum) * grad
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * v
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 40 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    lr = max(lr * 0.75, lr_init * 0.1)
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _peak_chase(a_init, deadline, max_rounds=300, top_k=5, reduction_frac=0.005):
    """Iteratively reduce top peaks with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = min(top_k, len(c))
        top_peaks = np.argsort(c)[::-1][:K]
        peak_vals = c[top_peaks]
        pmin = peak_vals[-1]
        w = np.zeros(len(c))
        for i, k in enumerate(top_peaks):
            if peak_vals[0] - pmin > 1e-30:
                w[k] = (peak_vals[i] - pmin) / (peak_vals[0] - pmin) + 0.05
            else:
                w[k] = 1.0
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        target = c_max * reduction_frac
        step = min(1.0, target / (grad_norm * 5.0))

        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = float(c_try.max())

        if f_try < c_max - 1e-14:
            a = a_try
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        else:
            for ls in range(8):
                step *= 0.5
                a_try = a - step * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    sc = _score(a)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = a.copy()
                    break

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.002):
    """Active-set Newton with CG for the Hessian system."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            n_active = 4 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-7 * (c_max / max(m, 1)) + 1e-10

        # CG with Hessian-vector product
        def _hvp(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def _cg(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - _hvp(x)
            p = r.copy()
            rs = np.dot(r, r)
            rhs_norm = np.linalg.norm(rhs)
            for _ in range(max_cg):
                Ap = _hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r = r - alpha * Ap
                rs_new = np.dot(r, r)
                if np.sqrt(rs_new) < 1e-13 * (1.0 + rhs_norm):
                    break
                p = r + (rs_new / rs) * p
                rs = rs_new
            return x

        u = _cg(g_t)
        w_sol = _cg(np.ones(n))
        sum_w = float(w_sol.sum())
        if abs(sum_w) < 1e-15:
            dx = -u - np.mean(-u)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_sol
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.98
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(35):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 or ls == 34:
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(15):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                if it > 5:
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _cross_over(a1, a2, alpha, rng):
    n = len(a1)
    # Smoothly varying mixing coefficient
    base = np.full(n, alpha)
    noise = rng.uniform(-0.15, 0.15, n)
    # Smooth the noise
    k = max(3, n // 64)
    kernel = np.ones(k) / k
    smooth_noise = np.convolve(noise, kernel, mode='same')
    mix = np.clip(base + smooth_noise, 0.2, 0.8)
    a_new = mix * a1 + (1.0 - mix) * a2
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a1.copy()
    return a_new / s

        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc
