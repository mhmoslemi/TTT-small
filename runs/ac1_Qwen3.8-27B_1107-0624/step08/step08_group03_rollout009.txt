"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-n exploration with interpolation
2. Aggressive L-BFGS with log-sum-exp smoothing (adaptive beta)
3. Active-set Newton with weighted dual variables and optimal step
4. Iterative peak equalization with adaptive active set
5. Coordinate descent final polish
6. Diverse restarts with geometric perturbation
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _interp_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    return _normalize(a_new)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_grad(a, w, n):
    """Compute gradient of sum_k w[k]*c_k(a) w.r.t. a, divided by 2.
    Result[i] = sum_k w[k] * (a[k-i] + a[i-k]) / 2 (with valid index handling)
    Actually: grad c_k / da_i = a[k-i] + a[i-k] (for valid indices)
    So sum_k w[k] * grad c_k / da_i = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    = (w * a_rev)[i] + (w_rev * a)[i] ... let me use FFT
    """
    # grad_i(c_k) = a[k-i] (when 0<=k-i<n) + a[i-k] (when 0<=i-k<n)
    # sum_k w[k]*a[k-i] = correlation of w and a at lag i
    # This is: (w * flip(a))[n-1 + i] in full convolution terms
    # Actually: sum_k w[k]*a[k-i] where k ranges so both are in [0,n-1]
    # Let's use: result = conv(w, a_flip)[offset]
    a_flip = a[::-1]
    N = _next_pow2(2 * n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_flip, N)
    corr = np.fft.irfft(fw * fa, N)
    # corr[m] = sum_j w[j] * a_flip[m-j] = sum_j w[j] * a[n-1-m+j]
    # We want sum_k w[k] * a[k-i] for k in [0,2n-2], a[k-i] valid for k-i in [0,n-1]
    # = sum_k w[k] * a[k-i] where 0<=k-i<n => i <= k < i+n
    # corr at index (n-1+i): sum_j w[j]*a_flip[n-1+i-j] = sum_j w[j]*a[n-1-(n-1+i-j)] = sum_j w[j]*a[j-i]
    # Hmm, let me just compute it directly: h[i] = sum_k w[k]*a[k-i] + w[k]*a[i-k]
    # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    # First term: corr_w_a[i] where corr_w_a[i] = sum_k w[k]*a[k-i]
    # Second term: sum_k w[k]*a[i-k] = sum_k w[i-k]*a[k] = (w_rev * a)[i] ... 
    
    # Simpler approach: 
    # h[i] = sum_k w[k]*(a[k-i] + a[i-k]) for valid indices
    # = conv(w, a)[n-1+i] + conv(w_rev, a)[n-1-i] ... getting confusing
    # Let me just use the FFT correlation directly
    # 
    # sum_k w[k]*a[k-i]: let j = k-i, so k = j+i
    # = sum_j w[j+i]*a[j] (for valid j, i.e., 0<=j<n and 0<=j+i<2n-1)

    # = ifft(fft(w) * conj(fft(a))) ... but for real sequences with the right indexing
    # Actually: np.correlate(w, a, 'full')[n-1+i] gives sum_j w[j+i]*a[j]
    # But for large arrays, FFT is faster.
    # 
    # Let me use: fft(a) * conj(fft(w)) --> gives correlation
    # corr[i] = sum_j a[j] * conj(w[j-i]) = sum_j a[j] * w[j-i] (real)
    # = sum_j w[j-i] * a[j] (for valid j)
    # We want sum_k w[k]*a[k-i] = sum_j w[j+i]*a[j] (sub j=k-i)
    # This is corr at lag -i, i.e., corr[full_n-1-i] if using full correlation
    # 
    # For real sequences: ifft(fft(a)*conj(fft(w))) = correlation where
    # result[k] = sum_j a[j+k]*w[j] (circular)
    # For linear: we need zero-padding.
    
    # Let me just do it the straightforward way with one FFT:
    # h[i] = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    # First sum: let's define u[k] = w[k] for k in [0, 2n-2], 0 otherwise
    #            and v[k] = a[k] for k in [0, n-1], 0 otherwise
    # sum_k u[k]*v[k-i] = (u * v)[i] in convolution sense... 
    # Actually this IS a cross-correlation: (u ⋆ v)[i] = sum_k u[k]*v[k-i]
    # = ifft(fft(u) * conj(fft(v))) with proper zero-padding for linear correlation
    
    N2 = _next_pow2(3 * n)

    # = sum_k w[k]*a[k-i] (k from max(0,i) to min(2n-2, i+n-1))

    Npad = _next_pow2(3 * n)

    # This is different. Let me think again.
    # sum_k w[k]*a[i-k] = sum_k w[i-k]*a[k] (rename k->i-k, k->k)
    # = (w_rev ⋆ a)[i] where w_rev[k] = w[2n-2-k]

    # sum_k w[k]*a[i-k] = conv(w, a_flip)[n-1 + i - (n-1)] ... 
    
    # OK let me just use a different approach. Compute h directly:
    # h[i] = sum_k w[k]*(a[k-i] + a[i-k])

    # And a[k-i] with k-i in [0,n-1] means k in [i, i+n-1]

    # First term: T1[i] = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k]*a[k-i]
    # Second term: T2[i] = sum_{k=max(0,i-n+1)}^{min(2n-2, i)} w[k]*a[i-k]

    #
    # Using the FFT computation above:
    # corr1[i] = sum_j w[j]*a[j-i] = T1[i] ✓

    # 
    # Therefore: h[i] = sum_k 2*w[k]*a[k-i] = 2 * sum_k w[k]*a[k-i]
    # = 2 * (cross-correlation of w and a at lag i)
    # = 2 * ifft(fft(w)*conj(fft(a)))[i] (with zero-padding for linear correlation)
    
    Npad = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, Npad)
    fa_p = np.fft.rfft(a, Npad)
    corr = np.fft.irfft(fw * np.conj(fa_p), Npad)

    # We want sum_k w[k]*a[k-i] which is corr[i] (rename j->k)
    return 2.0 * corr[:n]


def _active_set_newton(a_init, deadline, max_iter=500, tol_active=0.0005):
    """Active-set Newton with weighted peaks for optimal convergence."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Weight peaks by their height (higher peaks get more weight)
        w = np.zeros_like(c)
        active_vals = c[active]
        # Use power weighting to focus on the highest peaks
        weights = active_vals ** 10
        weights = weights / weights.sum()
        w[active] = weights

        # Compute gradient: h[i] = sum_k 2*w[k]*a[k-i]
        h = _corr_grad(a, w, n)
        W = float(np.dot(w, c))
        # The gradient of the weighted sum of active peaks w.r.t. a (in simplex)
        # We're minimizing sum_k w[k]*c[k] subject to sum(a)=1, a>=0
        # Grad in log space or just projected
        # For the Newton step, we need the Hessian of the Lagrangian
        # H[i,j] = sum_k 2*w[k] * (e_{k-i}[i==j...]) ... 
        # Hessian of c[k] w.r.t. (a_p, a_q) = 2*delta[p,k-q]*delta[... ] 
        # Actually Hessian of c[k] is: d2c[k]/(da_p da_q) = 2*delta[p+q, k] (when p+q=k)
        # Wait: c[k] = sum_j a[j]*a[k-j]
        # d2c[k]/(da_p da_q) = delta[p,q... ] 
        # dc[k]/da_p = 2*a[k-p]
        # d2c[k]/(da_p da_q) = 2*delta[k-p, q] = 2*delta[p+q, k]
        # So H[k][p,q] = 2*delta[p+q, k] (a diagonal in the "sum" index!)
        # 
        # Hessian of sum_k w[k]*c[k] w.r.t. a:
        # H[p,q] = sum_k 2*w[k]*delta[p+q, k] = 2*w[p+q] (when 0<=p+q<=2n-2, else 0)
        # This is a HASKIND matrix (constant on anti-diagonals)!
        # 
        # For the Newton step on the simplex, we need to solve:
        # H * dx + g = lambda * 1 (where g is the gradient projected)
        # sum(dx) = 0
        # 
        # The Hessian H is a Toeplitz-plus-Hankel matrix (specifically a Hankel matrix).
        # We can multiply by a vector in O(n log n) using FFT:
        # (H*v)[p] = sum_q H[p,q]*v[q] = sum_q 2*w[p+q]*v[q] (for valid p+q)
        # Let r = p+q: (H*v)[p] = sum_q 2*w[p+q]*v[q]
        # This is a cross-correlation of w and v!
        # (H*v)[p] = 2 * sum_q w[p+q]*v[q] = 2 * corr(w, v)[p] (with appropriate indexing)
        # = 2 * ifft(fft(w)*conj(fft(v)))[p] (with zero-padding)
        
        def hess_matvec(v):
            Npad = _next_pow2(3 * n)
            fw = np.fft.rfft(w, Npad)
            fv = np.fft.rfft(v, Npad)
            corr = np.fft.irfft(fw * np.conj(fv), Npad)
            return 2.0 * corr[:n]

        # Gradient of the objective (weighted sum of active peaks) on the simplex
        # We want to minimize f(a) = sum_k w[k]*c_k(a) s.t. sum(a)=1, a>=0
        # The gradient is h (computed above)
        # Projected gradient: h - mean(h) (for the sum constraint)
        g = h - np.mean(h)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Add regularization for positive definiteness
        mu_reg = 1e-10

        def hess_reg_matvec(v):
            return hess_matvec(v) + mu_reg * v

        # CG solve: (H + mu*I) * dx = g_t where g_t = h - mean(h)
        # But we also need sum(dx) = 0, so we solve:
        # (H + mu*I) * dx + lambda * 1 = g
        # sum(dx) = 0
        # This is a saddle point system. Use the trick:
        # First solve (H+mu*I)*u = g (without constraint)
        # Then dx = u - mean(u) ... no that's not quite right.
        # 
        # Actually for the equality constrained problem:
        # min 0.5*dx^T H dx + g^T dx s.t. 1^T dx = 0
        # KKT: H*dx + g + lambda*1 = 0, 1^T dx = 0
        # From first: dx = -(H+mu*I)^{-1}*(g + lambda*1)
        # From second: 1^T dx = -1^T H^{-1} g - lambda * 1^T H^{-1} 1 = 0
        # lambda = -1^T H^{-1} g / (1^T H^{-1} 1)
        
        # Solve (H+mu*I)*x = g using CG
        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_reg_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_reg_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g)
        one_vec = np.ones(n)
        w_sol = cg_solve(one_vec)
        sum_w = float(w_sol.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(np.dot(u, one_vec)) / sum_w
            dx = -u + lam * w_sol
            dx = dx - np.mean(dx)

        # Line search with bounds
        f_old = float(np.dot(w, c))
        neg_mask = (a + dx < 0) & (dx < 0)
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            # Check if the actual objective (max of c) improved
            f_try_max = float(c_try.max())
            if f_try_max < c_max + 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Try steepest descent fallback
            t = min(0.5 / (1 + 0.1 * it), t_max)
            for ls in range(20):
                a_try = a - t * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.4, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.025,
                                      0.012, 0.006, 0.003, 0.0015, 0.0007,
                                      0.00035, 0.00017, 0.000085, 0.00004,
                                      0.00002, 0.00001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.005
        elif n <= 16384:
            est_time_per_iter = 0.012
        elif n <= 32768:
            est_time_per_iter = 0.03
        else:
            est_time_per_iter = 0.06

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 12))))

        try:
            obj_fn = lambda z_local: _lsobj(z_local, beta, n)
            jac_fn = lambda z_local: _lsgrad(z_local, beta, n)
            result = minimize(
                obj_fn,
                z,
                jac=jac_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_equalize(a_init, deadline, max_rounds=100):
    """Iteratively equalize the top peaks using adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Start with top-5, expand to top-50 over iterations
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive number of active peaks
        if rnd < 10:
            n_active = min(10, 2 * n - 1)
        elif rnd < 30:
            n_active = min(30, 2 * n - 1)
        elif rnd < 60:
            n_active = min(80, 2 * n - 1)
        else:
            n_active = min(200, 2 * n - 1)

        top_idx = np.argsort(c)[::-1][:n_active]
        w = np.zeros_like(c)
        # Use power weighting
        vals = c[top_idx]
        weights = vals ** 8
        weights = weights / weights.sum()
        w[top_idx] = weights

        # Compute gradient
        h = _corr_grad(a, w, n)
        g = h - np.mean(h)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Line search
        f_old = c_max
        step = 0.1 / (1.0 + 0.05 * rnd)
        for ls in range(30):
            a_try = a - step * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _coord_descent_polish(a_init, deadline, max_passes=50):
    """Coordinate descent: update small groups of coordinates at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    # Work in log-space for the coordinates
    z = np.log(np.maximum(a, 1e-30))

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        # Randomly select a subset of coordinates to update
        n_update = min(n, max(100, n // 4))
        idx = rng.choice(n, size=n_update, replace=False)
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # For each selected coordinate, try small perturbations
        for i in idx:
            if time.monotonic() > deadline:
                break
            for delta_frac in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                a_try = a.copy()
                old_val = a_try[i]
                a_try[i] = max(0, old_val * (1 + delta_frac))
                # Renormalize
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    c_max = float(c_try.max())
                    sc = _score(a)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = a.copy()
                    break

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top-k with decay
        k = min(max(5, n // 200), len(c) // 5)
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c)
        # Exponential weighting by rank
        ranks = np.arange(k, 0, -1, dtype=np.float64)
        weights = np.exp(-0.1 * ranks)
        weights = weights / weights.sum()
        w[top_k] = weights

        h = _corr_grad(a, w, n)
        W = float(np.dot(w, c))
        g = h - np.mean(h)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * g
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, active-set Newton, peak equalization,
    and coordinate descent to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Explores multiple sequence lengths and uses diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "uniform_init")

    # Phase 1: L-BFGS on parent n
    print(f"\n[Phase 1] Aggressive L-BFGS on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a1 = _run_lbfgs(parent, deadline - 400, min_time_remaining=200, aggressive=True)
        update_best(a1, _score(a1), "lbfgs_parent")
        # Peak equalize
        if time.monotonic() < deadline - 200:
            a1p, sc1p = _peak_equalize(a1, min(deadline - 100, time.monotonic() + 100), max_rounds=50)
            update_best(a1p, sc1p, "peak_eq_parent")

    # Phase 2: Active-set Newton (tight)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a2, sc2 = _active_set_newton(best_seq, min(deadline - 200, time.monotonic() + 200),
                                     max_iter=300, tol_active=0.0005)
        update_best(a2, sc2, "as_newton_1")
        # Tighter
        if time.monotonic() < deadline - 150:
            a2b, sc2b = _active_set_newton(a2, min(deadline - 100, time.monotonic() + 100),
                                           max_iter=200, tol_active=0.0002)
            update_best(a2b, sc2b, "as_newton_tight")

    # Phase 3: Try different n values
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_parent]
    # Add nearby sizes
    for delta in [-1024, -512, 512, 1024, 2048]:
        n_try = n_parent + delta
        if 4096 <= n_try <= 32768 and n_try not in n_candidates:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue  # Already done
        a_inter = _interp_to_n(best_seq if best_seq is not None else parent, n_try)
        sc_inter = _score(a_inter)
        print(f"  n={n_try}: interp score={sc_inter:.10f}")
        
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        # Quick L-BFGS
        time_budget = min(remaining * 0.4, 120)
        if time_budget > 20:
            a_opt = _run_lbfgs(a_inter, time.monotonic() + time_budget, min_time_remaining=10, aggressive=True)
            update_best(a_opt, _score(a_opt), f"lbfgs_n{n_try}")
        
        # Active set
        if time.monotonic() < deadline - 30:
            a_as, sc_as = _active_set_newton(a_opt, min(deadline - 20, time.monotonic() + 60),
                                             max_iter=100, tol_active=0.0005)
            update_best(a_as, sc_as, f"as_n{n_try}")

    # Phase 4: Peak equalization on best
    print(f"\n[Phase 4] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_pe, sc_pe = _peak_equalize(best_seq, min(deadline - 40, time.monotonic() + 60), max_rounds=80)
        update_best(a_pe, sc_pe, "peak_eq_best")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts = [
            ("pert_001", 0.01),
            ("pert_003", 0.03),
            ("pert_005", 0.05),
            ("pert_010", 0.10),
        ]
        for label, sigma in restarts:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_r = _run_lbfgs(p_start, min(deadline - 10, time.monotonic() + remaining * 0.5),
                             min_time_remaining=5, aggressive=False)
            update_best(a_r, _score(a_r), f"restart_{label}")
            if time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 10, time.monotonic() + 30),
                                                 max_iter=80, tol_active=0.0005)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 6: Mirror descent
    print(f"\n[Phase 6] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_md, sc_md = _mirror_descent(best_seq, min(deadline - 15, time.monotonic() + 30),
                                      max_steps=3000, lr_init=0.03)
        update_best(a_md, sc_md, "mirror_descent")

    # Phase 7: Final tight active-set Newton
    print(f"\n[Phase 7] Final tight AS-Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 3, max_iter=200, tol_active=0.0001)
        update_best(a_fin, sc_fin, "final_as_tight")
        # Peak equalize
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _peak_equalize(a_fin, deadline - 2, max_rounds=30)
            update_best(a_fin2, sc_fin2, "final_peak_eq")

    # Phase 8: Coordinate descent polish
    print(f"\n[Phase 8] Coord descent polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_cd, sc_cd = _coord_descent_polish(best_seq, deadline - 2, max_passes=20)
        update_best(a_cd, sc_cd, "coord_descent")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
