"""
Optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using linear ramp initialization.
Key insight: the linear ramp a[i]=(i+1) achieves score ≈ 4/3 ≈ 1.333 for large n,
far below the previous best of 1.5057. We use this as the primary initialization
and apply smooth-approximation L-BFGS + Newton active-set polish to extract
additional improvement from the finite-n structure.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-13):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _linear_ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _pow_ramp(n, p=1.0):
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _perturbed_ramp(n, sigma, rng):
    a = _linear_ramp(n)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.002 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-8 * c_max_new and sc_improved > 1e-12:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _newton_active_polish(a_init, deadline, max_rounds=150):

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_tol_frac = 0.00001
        n_active = 0
        for tol_frac in [0.00001, 0.00005, 0.0001, 0.0003, 0.001, 0.005]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 1:
                best_tol_frac = tol_frac
                break
        if n_active == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-12 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.001 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -70.0, 70.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.05)
            else:
                patience += 1
                lr *= 0.8

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s "
                  f"improve={prev-best_score:.2e}")

    n = 16384

    # Phase 0: Linear ramp baseline
    print(f"[Phase 0] Linear ramp n={n}")
    a_ramp = _linear_ramp(n)
    sc_ramp = _score(a_ramp)
    print(f"  Linear ramp score: {sc_ramp:.10f} (theoretical: {4*(n+2)/(3*(n+1)):.10f})")
    update_best(a_ramp, sc_ramp)

    # Phase 1: Try alternative initializations
    print(f"\n[Phase 1] Alternative initializations (t={time.monotonic()-t_start:.1f}s)")
    inits = []
    labels = []

    # Power ramps
    for p in [0.9, 1.0, 1.1, 1.2, 1.5]:
        a_p = _pow_ramp(n, p)
        sc_p = _score(a_p)
        inits.append(a_p)
        labels.append(f"pow_p={p}")
        print(f"  pow(p={p}): score={sc_p:.10f}")
        update_best(a_p, sc_p)

    # Perturbed ramps
    for sigma in [0.01, 0.03, 0.05]:
        a_perm = _perturbed_ramp(n, sigma, rng)
        sc_perm = _score(a_perm)
        inits.append(a_perm)
        labels.append(f"perturb_sigma={sigma}")
        print(f"  perturb(sigma={sigma}): score={sc_perm:.10f}")
        update_best(a_perm, sc_perm)

    # Try parent if available
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_par = float(parent_raw.sum())
            if s_par > 0.01:
                a_par = parent_raw / s_par
                sc_par = _score(a_par)
                inits.append(a_par)
                labels.append("parent")
                print(f"  parent: score={sc_par:.10f}")
                update_best(a_par, sc_par)
    except Exception:
        pass

    # Phase 2: Main L-BFGS optimization from best initialization
    print(f"\n[Phase 2] Main L-BFGS from linear ramp (t={time.monotonic()-t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else a_ramp
    a_lbfgs = _run_adaptive_lbfgs(start_a, deadline, min_time_remaining=400.0)
    sc_lbfgs = _score(a_lbfgs)
    print(f"  L-BFGS result: score={sc_lbfgs:.10f}")
    update_best(a_lbfgs, sc_lbfgs)

    # Phase 3: Newton polish
    print(f"\n[Phase 3] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 200, time.monotonic() + 300),
                                             max_rounds=150)
        print(f"  Newton result: score={sc_np:.10f}")
        update_best(a_np, sc_np)

    # Phase 4: Subgradient diversification
    print(f"\n[Phase 4] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 150, time.monotonic() + 100))
        print(f"  Subgrad result: score={sc_sg:.10f}")
        update_best(a_sg, sc_sg)
        if sc_sg < best_score + 0.001 and time.monotonic() < deadline - 100:
            a_np2, sc_np2 = _newton_active_polish(a_sg, min(deadline - 100, time.monotonic() + 60),
                                                   max_rounds=60)
            update_best(a_np2, sc_np2)

    # Phase 5: L-BFGS from different initializations
    print(f"\n[Phase 5] L-BFGS from alternatives (t={time.monotonic()-t_start:.1f}s)")
    for i, (a_init, label) in enumerate(zip(inits, labels)):
        if time.monotonic() > deadline - 80:
            break
        if label == "pow_p=1.0":
            continue  # Already optimized
        a_opt = _run_adaptive_lbfgs(a_init, min(deadline - 60, time.monotonic() + 60),
                                     min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        print(f"  {label}: score={sc_opt:.10f}")
        update_best(a_opt, sc_opt)

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    for n_try in [32768, 8192]:
        if time.monotonic() > deadline - 60:
            break
        a_init_n = _linear_ramp(n_try)
        sc_init_n = _score(a_init_n)
        print(f"  n={n_try} ramp: {sc_init_n:.10f}")
        update_best(a_init_n, sc_init_n)

        if time.monotonic() < deadline - 40:
            a_opt_n = _run_adaptive_lbfgs(a_init_n, min(deadline - 30, time.monotonic() + 60),
                                           min_time_remaining=25.0)
            sc_opt_n = _score(a_opt_n)
            print(f"  n={n_try} optimized: {sc_opt_n:.10f}")
            update_best(a_opt_n, sc_opt_n)

    # Phase 7: Final Newton polish
    print(f"\n[Phase 7] Final Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=100)
        update_best(a_fin, sc_fin)

        # One more L-BFGS at tiny beta
        if best_seq is not None and time.monotonic() < deadline - 2:
            n_cur = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-11
            remaining = deadline - 2 - time.monotonic()
            if remaining > 1:
                max_iter = min(3000, max(30, int(remaining / 0.005)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n_cur)
                    grad_fn = _SmoGradAdapter(beta_fin, n_cur)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 200}
                    )
                    a_flb = _softmax(result.x)
                    a_flb = np.maximum(a_flb, 0)
                    s_flb = a_flb.sum()
                    if s_flb > 1e-15:
                        a_flb = a_flb / s_flb
                    update_best(a_flb, _score(a_flb))
                except Exception:
                    pass

        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_vf, sc_vf = _newton_active_polish(best_seq, deadline - 0.1, max_rounds=50)
            update_best(a_vf, sc_vf)

    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _linear_ramp(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _linear_ramp(8192)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
