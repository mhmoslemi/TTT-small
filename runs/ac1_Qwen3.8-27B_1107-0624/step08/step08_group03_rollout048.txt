Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses peak-tracking active-set Newton, multi-resolution optimization,
and diverse ramp-based initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

        active = c >= (c_max - tol_active)
        n_active = active.sum()
        if n_active == 0:
            active[np.argmax(c)] = True
            n_active = 1
        
        # Compute gradient of average active peak
        # grad_i = (1/n_active) * sum_{k in active} dc_k/da_i
        # dc_k/da_i = a[k-i] * (1 or 2) depending on whether k=2i
        # Approximate: grad_i ≈ (2/n_active) * sum_{k in active} a[k-i] (for bulk indices)
        # This is (2/n_active) * (indicator of active convolved with reversed a), centered
        
        # More precisely:
        # h_i = sum_{k in active} a[k-i] where 0 <= k-i < n
        # This is the cross-correlation of a with the indicator of active
        ind = active.astype(np.float64)
        # h = cross_correlation(a, ind) centered at n-1
        # h[i] = sum_k ind[k] * a[k-i] = (ind * reverse(a))[n-1+i] in full conv
        a_rev = a[::-1]
        h_full = _full_convolve(ind, a_rev)
        h = h_full[n-1:2*n-1]  # center-aligned
        
        # For the "2x" factor: when k=2i, the coefficient is 1 instead of 2
        # This is a small correction; for now, use the 2x approximation
        grad = (2.0 / n_active) * h
        # Remove mean (for the sum constraint)
        grad_centered = grad - grad.mean()
        
        # Newton step: solve H * dx = -grad_centered
        # where H is the Hessian of the average active peak
        # H = (2/n_active) * sum_{k in active} C_k
        # (H v)_i = (2/n_active) * sum_{k in active} (C_k v)_i = (1/n_active) * sum_k v[n-1+k-i]... 
        # Actually (C_k v)_i = v[k-i] (with boundary effects)
        # So (H v)_i = (2/n_active) * sum_{k in active} v[k-i] = (2/n_active) * (cross-corr of v with ind)
        # This is the same operator as for the gradient!
        
        def hess_matvec(v):
            v_rev = v[::-1]
            result = _full_convolve(ind, v_rev)
            return (2.0 / n_active) * result[n-1:2*n-1]
        
        # Solve using conjugate gradients
        # Add small regularization for numerical stability
        mu = 1e-10
        x = np.zeros(n)
        r = -grad_centered - hess_matvec(x) - mu * x
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(100):
            Ap = hess_matvec(p) + mu * p
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x_new = x + alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if sqrt(rs_new) < 1e-12 * (1 + sqrt(rs)):
                x = x_new
                break
            p = r_new + (rs_new / rs) * p
            x = x_new
            r = r_new
            rs = rs_new
        
        # x is the step direction (approximately)
        # Enforce sum(x) = 0 (for the sum constraint)
        x = x - x.mean()
        
        # Line search with non-negativity
        f_old = c_max
        t = 1.0
        for ls in range(50):
            a_try = a + t * x
            if a_try.min() < 0:
                t *= 0.5
                continue
            a_try = a_try / a_try.sum()
            c_try = _full_convolve(a_try, a_try)
