def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())

        return np.inf
    c = _full_convolve(a, a)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- Top-K smooth max for more focused gradients (only smooth over top K peaks)
- Barzilai-Borwein adaptive step sizes in subgradient phase
- Hybrid active-set method with embedded L-BFGS restarts
- Diverse initialization with multiple profile types
- Aggressive beta/K annealing schedules
- Periodic re-optimization from perturbed best to escape local minima

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a).
    w has length 2n-1. Returns array of length n."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-domain.
    Only the top K values of c contribute to the smoothing."""
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Get top K indices
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
        c_top_min = c_top.min()
    else:
        c_top = c
        c_top_min = c.min()
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        # Build full weight vector
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    # Gradient in a-space
    ga = _grad_maxconv(a, w)
    # Chain rule through softmax
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-K smoothing and adaptive K/beta annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.008)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Schedule: (beta_multiplier, K_fraction)
    # Start with large K (more smooth), end with small K (sharper)
    schedule = [
        (1.0, 0.1), (0.7, 0.08), (0.5, 0.06), (0.3, 0.04),
        (0.2, 0.03), (0.1, 0.02), (0.05, 0.015), (0.03, 0.01),
        (0.01, 0.008), (0.005, 0.005), (0.002, 0.003), (0.001, 0.002),
    ]
    
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = base_beta * beta_mult
        K = max(5, int((2 * n - 1) * K_frac))
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 20))))
        
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _subgradient_bb(a_init, deadline, min_time_remaining=60.0):
    """Active-set subgradient with Barzilai-Borwein steps and momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    n_2nm1 = 2 * n - 1
    
    active_frac = 0.01
    momentum = 0.7
    lr = 0.02
    lr_min = 2e-5
    lr_max = 0.1
    stall = 0
    max_stall = 80
    velocity = np.zeros(n)
    
    prev_a = None
    prev_g = None
    
    iteration = 0
    t_start = time.monotonic()
    last_lbfgs_time = t_start
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Select active peaks
        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        # Weights: proportional to how much above the median of active peaks
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.1
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        # Gradient
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        # Momentum step
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            # Line search
            for ls in range(25):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                # Halve
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.03)
            # Shrink active set
            active_frac = max(0.002, active_frac * 0.98)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.08, active_frac * 1.3)
        
        # Adapt on stall
        if stall > max_stall:
            active_frac = min(0.08, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.9, momentum + 0.05)
            stall = 0
        
        # Periodic L-BFGS restart
        if iteration % 500 == 0 and time.monotonic() - last_lbfgs_time > 120:
            if time.monotonic() + min_time_remaining + 90 < deadline:
                lbfgs_deadline = time.monotonic() + 80
                a_tmp = _run_lbfgs(a, lbfgs_deadline, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()
    
    return best_a, best_sc

            patience += 1
            if patience > max_patience:
                break
        
        # Find top peak
        k_peak = int(np.argmax(c))
        
        # Compute contributions to this peak
        # c[k_peak] = sum_i a[i]*a[k_peak-i]
        i_lo = max(0, k_peak - (n - 1))
        i_hi = min(n - 1, k_peak)
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_peak - i
            contrib[i] += a[j]
            if i != j:
                contrib[j] += a[i]
        
        # Reduce top contributors
        n_reduce = max(1, int(n * 0.002))
        top_contrib = np.argsort(contrib)[::-1][:n_reduce]
        
        reduction_amount = c_max * 0.005
        delta = np.zeros(n)
        for idx in top_contrib:
            if contrib[idx] > 1e-15:
                delta[idx] = -reduction_amount / (2.0 * contrib[idx] + 1e-30)
        
        # Redistribute to low contributors
        low_idx = np.argsort(contrib)[:n_reduce]
        for idx in low_idx:
            delta[idx] = reduction_amount * 0.5 / (n_reduce * (a[idx] + 1e-10) + 1e-30)
        
        delta -= delta.mean()  # Preserve sum
        
        # Line search
        alpha = 1.0
        improved = False
        for ls in range(30):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            patience += 10
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish with small steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.005
    tol = 0.005
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol of max
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        # Line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol = max(0.0001, tol * 0.7)
                lr *= 0.7
            if no_improve > 300:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.003
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.97)
            lr = min(0.01, lr * 1.01)
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS with top-K smoothing, BB subgradient,
    direct peak reduction, fine polish, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: L-BFGS on parent ===
    print(f"\n[P1] L-BFGS top-K (t={time.monotonic()-t_start:.1f}s)")
    a1 = _run_lbfgs(best_seq, deadline, min_time_remaining=600)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Subgradient BB ===
    print(f"\n[P2] Subgradient BB (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a2, sc2 = _subgradient_bb(best_seq, min(deadline - 350, time.monotonic() + 400),
                                  min_time_remaining=60)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Direct peak reduction ===
    print(f"\n[P3] Direct peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a3, sc3 = _direct_peak_reduction(best_seq, min(deadline - 250, time.monotonic() + 250),
                                         min_time_remaining=30)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [peakred] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Second L-BFGS from improved point ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a4 = _run_lbfgs(best_seq, min(deadline - 200, time.monotonic() + 250),
                        min_time_remaining=40)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Fine polish ===
    print(f"\n[P5] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a5, sc5 = _fine_polish(best_seq, min(deadline - 120, time.monotonic() + 150),
                               min_time_remaining=20)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Multi-restart exploration ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        # Try perturbed versions of best
        restart_budget = min(200, (deadline - 120 - time.monotonic()) * 0.5)
        n_restart = max(2, int(restart_budget / 40))
        for i in range(n_restart):
            if time.monotonic() > deadline - 80:
                break
            sigma = [0.03, 0.05, 0.08, 0.12, 0.06, 0.04, 0.1, 0.15][i % 8]
            a_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(a_pert, min(deadline - 60, time.monotonic() + 35),
                             min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}] t={time.monotonic()-t_start:.1f}s")
        
        # Try different profiles
        profiles = ["cos_taper", "optimal_gauss", "bimodal", "asym", "triangle"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 60:
                break
            a_prof = _make_profile(len(best_seq), prof_kind)
            a_p = _run_lbfgs(a_prof, min(deadline - 40, time.monotonic() + 30),
                             min_time_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        n_options = [16384, 12288, 24576]
        n_options = [x for x in n_options if x != n_main]
        
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs(a_res, min(deadline - 30, time.monotonic() + 40),
                               min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Final fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")
        
        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0002, 1e-8)
            K_fin = max(5, int(n * 0.001))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _TopkSmoProb(beta_fin, K_fin, n)
            max_iter = min(3000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
