"""
Adaptive active-set Newton optimizer for minimizing step function autocorrelation peak.
Uses KKT-based active set management, smearing optimality checks, and adaptive tolerance
to refine the solution beyond the previous state of the art (1.505675) toward the target

smearing-based convergence detection, and aggressive final polish.

import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# LSE objective for L-BFGS
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


# ============================================================
# Hessian vector product for active set
# ============================================================

def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-12, max_iter=500):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha < 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


# ============================================================
# Smearing computation (KKT check)
# ============================================================

def _compute_smearing(a, active, n):
    """Compute the smearing: s[j] = sum_{tau in active} a[tau-j+n-1] (valid indices).
    At KKT optimum, this should be constant on the support of a."""
    active_f = active.astype(np.float64)
    a_rev = a[::-1]
    wc = _full_convolve(active_f, a_rev)
    return wc[n - 1:2 * n - 1]


def _kkt_residual(a, active, n, tol=1e-8):
    """Compute KKT residual: how non-constant the smearing is on the support."""
    s = _compute_smearing(a, active, n)
    mask = a > tol
    if mask.sum() < 2:
        return np.inf
    s_support = s[mask]
    return float(np.std(s_support) / (np.abs(np.mean(s_support)) + 1e-15))


# ============================================================
# Adaptive active-set Newton
# ============================================================

def _newton_adaptive(a_init, deadline, max_iter=5000, tol_init=None, tol_min=1e-10):
    """Active-set Newton with adaptive tolerance and KKT-based convergence."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s_sum = a.sum()
    if s_sum < 1e-15:
        return a_init, _score(a_init)
    a = a / s_sum
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    if tol_init is None:
        tol_init = 0.03 * c_max

    tol = tol_init
    patience = 0
    max_patience = 15

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Define active set
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        active_count = int(active.sum())
        active_f = active.astype(np.float64) / active_count

        # Compute gradient (smearing)
        smearing = _compute_smearing(a, active_f, n)
        grad = 2.0 * smearing
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))

        # KKT check
        kkt_res = _kkt_residual(a, active, n, tol=1e-10)

        if gnorm < 1e-14 or kkt_res < 1e-10:
            # Converged at this tolerance level - try tightening
            patience += 1
            if patience >= max_patience:
                tol *= 0.5
                patience = 0
                if tol < tol_min:
                    break
            continue

        # Newton step
        mu = 1e-10 * max(1.0, float(np.dot(grad_t, grad_t)) / max(n, 1))

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-12, max_iter=min(500, max(50, n // 64)))
        dx = dx - np.mean(dx)

        # Line search
        alpha = 1.0
        f_old = c_max
        a_new = a.copy()
        found = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_try_max = float(c_try.max())
            # Sufficient decrease
            if c_try_max < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                found = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not found:
            if alpha < 1e-14:
                # Try a smaller tolerance step
                patience += 1
                if patience >= max_patience:
                    tol = max(tol * 0.7, tol_min)
                    patience = 0
                continue
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            patience = 0

    return best_a, best_sc


# ============================================================
# Multi-stage Newton with progressive tightening
# ============================================================

def _newton_progressive(a_init, deadline, min_remaining=30.0):
    """Run Newton with progressively tightening tolerance stages."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return best_a, best_sc

    # Stages of decreasing tolerance (as fraction of c_max)
    fractions = [0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.0015,
                 0.0008, 0.0004, 0.0002, 0.0001, 0.00005, 0.00002,
                 0.00001, 0.000005]

    for frac in fractions:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        tol = frac * c_max
        a_cur = a.copy()

        # Run Newton for this tolerance level
        stage_deadline = min(deadline - min_remaining, time.monotonic() + remaining * 0.4)
        a_new, sc_new = _newton_adaptive(a_cur, stage_deadline, max_iter=2000,
                                          tol_init=tol, tol_min=tol * 0.3)

        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            # Update c_max
            c = _full_convolve(a, a)
            c_max = float(c.max())

    return best_a, best_sc


# ============================================================
# Subgradient polish with adaptive step
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps=10000, lr_init=0.005, tol_frac=0.002):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        active_f = active.astype(np.float64) / active.sum()

        smearing = _compute_smearing(a, active_f, n)
        grad_a = 2.0 * smearing
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


# ============================================================
# L-BFGS with LSE
# ============================================================

def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=10):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                                  0.01, 0.005, 0.002, 0.001,
                                  0.0005, 0.0002])[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


# ============================================================
# Perturbation methods for diversity
# ============================================================

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block(a, rng, n_blocks=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


# ============================================================
# Targeted peak reduction
# ============================================================

def _targeted_peak_reduction(a, deadline, n_iters=2000):
    """Directly target the maximum lag and try to reduce it."""
    n = len(a)
    a_cur = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.002

    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        c_max_idx = int(np.argmax(c))
        # Gradient of c[c_max_idx]
        # (grad)[j] = 2*a[c_max_idx - j] for valid indices
        grad = np.zeros(n)
        d = c_max_idx
        for j in range(n):
            k = d - j
            if 0 <= k < n:
                grad[j] = 2.0 * a_cur[k]
        grad_t = grad - np.mean(grad)

        a_try = a_cur - lr * grad_t
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < _score(a_cur):
            a_cur = a_try
            lr = min(lr * 1.05, 0.02)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_cur.copy()
        else:
            lr *= 0.88

    return best_a, best_sc


# ============================================================
# Profile generators
# ============================================================

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.04) + np.exp(-((t - 0.7) ** 2) / 0.04)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


# ============================================================
# Resize helper
# ============================================================

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Adaptive active-set Newton optimizer with progressive tolerance tightening,
    L-BFGS warm start, KKT-based convergence detection, and targeted peak reduction.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init")

    a_work = _best_seq.copy()

    # === Phase 1: Quick L-BFGS warm start (120s) ===
    print(f"\n[Phase 1] L-BFGS warm start")
    phase1_dead = min(deadline, t_start + 120)
    if time.monotonic() < phase1_dead:
        a_l, sc_l = _lbfgs_lse(a_work, phase1_dead, min_remaining=60, n_stages=6)
        upd(a_l, sc_l, "lbfgs_warm")
        if _best_seq is not None:
            a_work = _best_seq.copy()

    # === Phase 2: Progressive Newton (main workhorse, 1200s) ===
    print(f"\n[Phase 2] Progressive Newton")
    phase2_dead = min(deadline, t_start + 1350)
    if time.monotonic() < phase2_dead:
        a_n, sc_n = _newton_progressive(a_work, phase2_dead, min_remaining=300)
        upd(a_n, sc_n, "newton_progressive")
        if _best_seq is not None:
            a_work = _best_seq.copy()

    # === Phase 3: Subgradient polish (150s) ===
    print(f"\n[Phase 3] Subgradient polish")
    phase3_dead = min(deadline, t_start + 1520)
    if time.monotonic() < phase3_dead:
        a_s, sc_s = _subgrad_polish(a_work, phase3_dead, n_steps=8000, lr_init=0.004, tol_frac=0.001)
        upd(a_s, sc_s, "sg_polish")
        if _best_seq is not None:
            a_work = _best_seq.copy()

    # === Phase 4: Newton again after SG (100s) ===
    print(f"\n[Phase 4] Post-SG Newton")
    phase4_dead = min(deadline, t_start + 1630)
    if time.monotonic() < phase4_dead:
        a_n2, sc_n2 = _newton_progressive(a_work, phase4_dead, min_remaining=80)
        upd(a_n2, sc_n2, "newton_post_sg")
        if _best_seq is not None:
            a_work = _best_seq.copy()

    # === Phase 5: Targeted peak reduction (60s) ===
    print(f"\n[Phase 5] Targeted peak reduction")
    phase5_dead = min(deadline, t_start + 1700)
    if time.monotonic() < phase5_dead:
        a_t, sc_t = _targeted_peak_reduction(a_work, phase5_dead, n_iters=5000)
        upd(a_t, sc_t, "targeted")
        if _best_seq is not None:
            a_work = _best_seq.copy()

    # === Phase 6: Diverse restarts with Newton (50s) ===
    print(f"\n[Phase 6] Diverse restarts")
    phase6_dead = min(deadline, t_start + 1760)
    if time.monotonic() < phase6_dead:
        for i in range(4):
            if time.monotonic() > phase6_dead:
                break
            if i == 0:
                p_start = _perturb_log(a_work, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(a_work, rng, max_amp=0.15)
            elif i == 2:
                p_start = _perturb_block(a_work, rng, n_blocks=4)
            else:
                p_start = _perturb_log(a_work, 0.04, rng)

            sub_dead = min(phase6_dead, time.monotonic() + 10)
            a_r, sc_r = _newton_progressive(p_start, sub_dead, min_remaining=2)
            upd(a_r, sc_r, f"restart_{i}")

    # === Phase 7: Final ultra-tight Newton (30s) ===
    print(f"\n[Phase 7] Final tight Newton")
    if _best_seq is not None and time.monotonic() < deadline - 3:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        c_max_fin = float(c_fin.max())
        a_fin, sc_fin = _newton_adaptive(_best_seq, deadline - 2, max_iter=3000,
                                          tol_init=c_max_fin * 0.001,
                                          tol_min=c_max_fin * 1e-7)
        upd(a_fin, sc_fin, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
