"""
Optimized solver for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-stage approach: L-BFGS with adaptive beta smoothing, active-set Newton polishing,
and multi-n exploration. Key improvements over previous: tighter active set management,
better CG convergence, momentum-augmented subgradient as fallback, and more aggressive
beta continuation schedule.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        if rs < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active(a_init, deadline, max_rounds=200, tight_factor=0.00005):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    tol_frac = 0.0005

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        active = c >= (c_max - tol_frac * c_max)
        n_active = int(active.sum())
        if n_active < 1:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.002 * alpha) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not step_done:
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try

                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

        if (rnd + 1) % 25 == 0:
            tol_frac = max(0.000005, tol_frac * 0.5)

    # Final tightening pass
    for final_tol in [0.000005, 0.000001]:
        if time.monotonic() > deadline:
            break
        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        active = c >= (c_max - final_tol * c_max)
        if active.sum() < 1:
            active[int(np.argmax(c))] = True
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()
        g = _hankel_matvec(best_a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            continue
        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20:
            continue
        lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
        dx = -Hg + lam * H1
        dx = dx - np.mean(dx)
        for ls in range(60):
            a_try = best_a + (0.5 ** ls) * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max - 1e-15:
                best_a = a_try
                best_sc = _score(best_a)
                break

    return best_a, best_sc


def _run_lbfgs_stages(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.0005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 5e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.002 * scale
        max_iter = min(15000, max(50, int(remaining / (est_iter_time * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 and sc_improved > 1e-14:
            beta *= 0.12
        elif new_gap < 1e-7 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _subgrad_with_momentum(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()

    lr = 0.01
    v = np.zeros(n)

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0003 * i)
        v = mu * v - step * grad
        z = z + v
        z = np.clip(z, -60.0, 60.0)

        if i % 20 == 19:

                    v = np.zeros(n)

    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = int(0.1 * n)
        ramp = np.linspace(0, 1, edge)
        a[:edge] = ramp
        a[-edge:] = ramp[::-1]
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _cyclic_optimize(a_init, deadline, min_time_remaining=100.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    cycle_dur = 30.0

    while time.monotonic() + min_time_remaining < deadline:
        t_end = min(time.monotonic() + cycle_dur, deadline - min_time_remaining)
        if time.monotonic() >= t_end:
            break

        a_lb = _run_lbfgs_stages(current_a, t_end - 5, min_time_remaining=20.0)
        sc_lb = _score(a_lb)

                max_rounds=50, tight_factor=0.0002
            )
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np

        cycle_dur = min(cycle_dur * 1.3, 120.0)

    return current_a

    print(f"\n[Phase 2] Main L-BFGS t={time.monotonic() - t_start:.1f}s")
    a_main = _run_lbfgs_stages(start_a, deadline, min_time_remaining=500.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 3: Cyclic optimization (L-BFGS + Newton alternation)
    print(f"\n[Phase 3] Cyclic opt t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_cyc = _cyclic_optimize(best_seq, deadline, min_time_remaining=250.0)
        update_best(a_cyc, _score(a_cyc), "cyclic")

    # Phase 4: Dedicated Newton polish (multi-stage tightening)
    print(f"\n[Phase 4] Newton polish t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 200:
        # Stage 1: moderate tolerance
        a_n1, sc_n1 = _newton_active(best_seq, min(deadline - 200, time.monotonic() + 120),
                                     max_rounds=80, tight_factor=0.0005)
        update_best(a_n1, sc_n1, "newton_1")
        # Stage 2: tighter
        if best_seq is not None and time.monotonic() < deadline - 80:
            a_n2, sc_n2 = _newton_active(best_seq, min(deadline - 80, time.monotonic() + 80),
                                         max_rounds=100, tight_factor=0.0001)
            update_best(a_n2, sc_n2, "newton_2")

    # Phase 5: Subgradient with momentum (diversification)
    print(f"\n[Phase 5] Subgrad+momentum t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a_sg, sc_sg = _subgrad_with_momentum(best_seq, min(deadline - 120, time.monotonic() + 80))
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.005 and time.monotonic() < deadline - 60:
            a_n3, sc_n3 = _newton_active(a_sg, min(deadline - 60, time.monotonic() + 50),
                                         max_rounds=60, tight_factor=0.0002)
            update_best(a_n3, sc_n3, "newton_3")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n t={time.monotonic() - t_start:.1f}s")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt_n = _run_lbfgs_stages(a_init_n, deadline, min_time_remaining=60.0)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")
        if sc_opt_n < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_np, sc_np = _newton_active(a_opt_n, min(deadline - 40, time.monotonic() + 30),
                                         max_rounds=40, tight_factor=0.0002)
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # Phase 7: Perturbed restarts
    print(f"\n[Phase 7] Restarts t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for sigma in [0.03, 0.08]:
            if time.monotonic() > deadline - 20:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs_stages(a_pert, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_s{sigma}")
            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _newton_active(a_r, deadline - 5, max_rounds=40, tight_factor=0.0003)
                update_best(a_rp, sc_rp, f"restart_s{sigma}_np")

    # Phase 8: Final ultra-fine Newton
    print(f"\n[Phase 8] Final polish t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _newton_active(best_seq, deadline - 1, max_rounds=100,
                                       tight_factor=0.00001)
        update_best(a_fin, sc_fin, "final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        ver = evaluate_sequence([float(x) for x in final])
        print(f"FINAL: internal_score={sc_final:.10f}, evaluator_score={ver:.10f}, "
              f"n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
