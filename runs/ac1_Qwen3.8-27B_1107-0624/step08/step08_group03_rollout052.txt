Optimized step-function designer using active-set Newton, peak shaving,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr(a, A_vec, n):
    """Compute sum_k A_vec[k] * a[k-i] for each i (i.e., correlation of a with A_vec)."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _peak_shave(a_init, deadline, max_rounds=200):
    """Explicitly target and reduce the single maximum peak."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Gradient of c[peak_idx] w.r.t. a
        # dc_k/da_j = 2*a[k-j] (if 0<=k-j<n and j!=k-j) or a[j] (if 2j=k)
        # Simplified: grad = 2 * corr(a, delta_peak, n)
        # where delta_peak is a spike at peak_idx
        # Use the formula: grad_j = a[peak_idx - j] if valid + a[peak_idx - j] if valid (from other factor)
        # = 2 * a[max(0, min(n-1, peak_idx - j))] ... but with bounds

        # Efficient: grad = 2 * (flip_a shifted)
        # flip_a[i] = a[n-1-i]
        # (dc_k/da)_j = a[k-j] if 0<=k-j<n (first factor) + a[k-j] if 0<=k-j<n (second factor)
        # For k = peak_idx: grad_j = 2*a[peak_idx - j] if 0 <= peak_idx - j < n, else a[peak_idx-j] if 2j=k, else 0

        # Use correlation: corr(a, delta, n) where delta[peak_idx] = 1
        # This gives h[j] = a[peak_idx - j] (with appropriate bounds via the FFT trick)
        h = _corr(a, np.zeros(2 * n - 1).__setitem__(peak_idx, 1.0) * np.ones(2 * n - 1) * 0 + np.where(np.arange(2 * n - 1) == peak_idx, 1.0, 0.0), n)
        # Simpler: just construct the gradient directly
        grad = np.zeros(n)
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):

            if 2 * j == peak_idx:
                grad[j] += val  # diagonal
            else:
                grad[j] += 2.0 * val

        if gnorm < 1e-14:

        f_old = c_max
        step = 0.1
        for ls in range(40):

            s_try = a_try.sum()
            if s_try < 1e-15:

            a_try = a_try / s_try

                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:

        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _local_search(a_init, deadline, n_perturb=200, sigma=0.05):
    """Random local search with refinement."""
    n = len(a_init)
    rng = np.random.default_rng(12345)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for i in range(n_perturb):
        if time.monotonic() > deadline:
            break
        z = np.log(np.maximum(best_a, 1e-30))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_try = _softmax(z)
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            # Refine
            a_ref, sc_ref = _active_set_newton(a_try, min(deadline, time.monotonic() + 10),
                                                max_iter=50, tol_active=0.001)
            if sc_ref < best_sc:
                best_sc = sc_ref
                best_a = a_ref.copy()

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _coordinate_polish(a_init, deadline, max_rounds=500):
    """Coordinate-wise polish targeting the maximum peak."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c_max_prev = float(_full_convolve(a, a).max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Compute gradient of c[peak_idx] w.r.t. a
        # grad[j] = 2*a[peak_idx - j] if 0 <= peak_idx - j < n, with diagonal correction
        grad = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                if 2 * j == peak_idx:
                    grad[j] = a[idx]
                else:
                    grad[j] = 2.0 * a[idx]

        # Try adjusting each coordinate in a targeted way
        # Reduce coordinates that contribute most to the peak
        top_contrib = np.argsort(grad)[::-1][:10]
        improved = False
        for j in top_contrib:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try reducing
            for frac in [0.5, 0.3, 0.1, 0.05]:
                a_try = a.copy()
                a_try[j] = old_val * (1.0 - frac)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-13:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try the gradient direction
            grad_t = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_t)
            if gnorm < 1e-14:
                break
            step = 0.05 / (1.0 + 0.01 * rnd)
            a_try = a - step * grad_t / gnorm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-13:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                    improved = True

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max_prev) < 1e-14 and not improved:
            break
        c_max_prev = float(c_new.max())

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Load parent, evaluate.
    2. L-BFGS with aggressive log-sum-exp continuation.
    3. Active-set Newton with tight tolerance.
    4. Peak shaving.
    5. Mirror descent.
    6. Coordinate polish.
    7. Try different n values.
    8. Local search.
    9. Final rounds of active-set + peak shave.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_parent = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
                update_best(parent, sc_parent, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS continuation from parent ===
    print(f"\n[Phase 1] L-BFGS continuation (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs_continuation(parent, deadline - 600, min_time_remaining=400.0, aggressive=True)
        update_best(a1, _score(a1), "lbfgs_main")

    # === Phase 2: Active-set Newton (tight) ===
    print(f"\n[Phase 2] Active-set Newton tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a2, sc2 = _active_set_newton(best_seq, deadline - 300, max_iter=300, tol_active=0.0003)
        update_best(a2, sc2, "as_newton_tight")

    # === Phase 3: Peak shaving ===
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a3, sc3 = _peak_shave(best_seq, min(deadline - 100, time.monotonic() + 120), max_rounds=150)
        update_best(a3, sc3, "peak_shave")

    # === Phase 4: Mirror descent ===
    print(f"\n[Phase 4] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a4, sc4 = _mirror_descent(best_seq, min(deadline - 80, time.monotonic() + 80),
                                  max_steps=3000, lr_init=0.03)
        update_best(a4, sc4, "mirror_desc")

    # === Phase 5: Coordinate polish ===
    print(f"\n[Phase 5] Coordinate polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a5, sc5 = _coordinate_polish(best_seq, min(deadline - 50, time.monotonic() + 60), max_rounds=300)
        update_best(a5, sc5, "coord_polish")

    # === Phase 6: Active-set Newton again (even tighter) ===
    print(f"\n[Phase 6] Active-set Newton very tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a6, sc6 = _active_set_newton(best_seq, min(deadline - 40, time.monotonic() + 50),
                                      max_iter=200, tol_active=0.0001)
        update_best(a6, sc6, "as_very_tight")

    # === Phase 7: Peak shave again ===
    print(f"\n[Phase 7] Peak shave round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a7, sc7 = _peak_shave(best_seq, min(deadline - 30, time.monotonic() + 40), max_rounds=100)
        update_best(a7, sc7, "peak_shave2")

    # === Phase 8: Try different n ===
    print(f"\n[Phase 8] Different n exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for n_try in [8192, 32768]:
            if time.monotonic() > deadline - 40:
                break
            a_interp = _interpolate_to_n(best_seq, n_try)
            sc_interp = _score(a_interp)
            print(f"  n={n_try}: interp score={sc_interp:.10f}")
            if sc_interp < best_score + 0.05:
                a_opt = _run_lbfgs_continuation(a_interp, min(deadline - 30, time.monotonic() + 40),
                                                 min_time_remaining=20.0, aggressive=False)
                update_best(a_opt, _score(a_opt), f"n{n_try}_lbfgs")
                if time.monotonic() < deadline - 20:
                    a_as, sc_as = _active_set_newton(a_opt, min(deadline - 10, time.monotonic() + 20),
                                                     max_iter=100, tol_active=0.0005)
                    update_best(a_as, sc_as, f"n{n_try}_as")

    # === Phase 9: Local search ===
    print(f"\n[Phase 9] Local search (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a9, sc9 = _local_search(best_seq, min(deadline - 10, time.monotonic() + 25),
                                n_perturb=50, sigma=0.02)
        update_best(a9, sc9, "local_search")

    # === Phase 10: Final refinement cycle ===
    print(f"\n[Phase 10] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a10, sc10 = _active_set_newton(best_seq, min(deadline - 5, time.monotonic() + remaining - 10),
                                           max_iter=200, tol_active=0.0001)
            update_best(a10, sc10, "final_as")
        if time.monotonic() < deadline - 5:
            a11, sc11 = _peak_shave(best_seq, deadline - 3, max_rounds=100)
            update_best(a11, sc11, "final_peak")
        if time.monotonic() < deadline - 3:
            a12, sc12 = _coordinate_polish(best_seq, deadline - 2, max_rounds=200)
            update_best(a12, sc12, "final_coord")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
