            lr = max(0.001, lr * 0.98)

        if iteration % 500 == 0:
            print(f"    PEQ iter {iteration}: score={best_sc:.10f}, lr={lr:.6f}, t={time.monotonic() - _GLOBAL_START:.1f}s")

    return best_a, _score(best_a)


def _coord_descent_fine(a_init, deadline, min_time_remaining=30.0):
    """Randomized coordinate descent on the simplex for fine-tuning."""
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    c = _full_convolve(a, a)
    c_max = float(np.max(c))
    beta = c_max * 0.01

    no_improve = 0
    max_stall = 5000
    iterations = 0

    while time.monotonic() + min_time_remaining < deadline:
        iterations += 1

        if iterations % 1000 == 0:
            c = _full_convolve(a, a)
            c_max = float(np.max(c))
            current_sc = 2.0 * n * c_max
            if current_sc < best_sc - 1e-15:
                best_sc = current_sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1000
                if no_improve >= max_stall:
                    break
            beta = max(c_max * 0.001, beta * 0.9)
            print(f"    CD iter {iterations}: score={best_sc:.10f}, beta={beta:.4e}, t={time.monotonic() - _GLOBAL_START:.1f}s")

        # Select random coordinate pair (j, k)
        j = rng.integers(0, n)
        k = rng.integers(0, n)
        while k == j:
            k = rng.integers(0, n)

        a_j = a[j]
        a_k = a[k]

        if a_j < 1e-10 and a_k < 1e-10:
            continue

        # Compute delta_c for moving mass from k to j (t = amount moved)
        # a[j] -> a[j] + t, a[k] -> a[k] - t
        # delta_c[m] = t * (a[m-j] + a[j-m] - a[m-k] - a[k-m]) with bounds

        # Compute d[m] = (a[m-j] + a[j-m]) - (a[m-k] + a[k-m])
        d = np.zeros(2 * n - 1)
        # Contribution from a[j] (positive)
        for m in range(2 * n - 1):
            val = 0.0
            if 0 <= m - j < n:
                val += a[m - j]
            if 0 <= j - m < n:
                # This would be a[j-m] but j-m < 0 means m > j, and j-m >= 0 means m <= j
                # Actually j-m is only valid if 0 <= j-m < n, i.e., -n+1 <= m-j... 
                # Let me just check: a[j-m] is valid if 0 <= j-m < n, i.e., j-n+1 <= m <= j
                pass
        # This O(n) loop per coordinate is too slow. Let me use a different approach.

        # Instead, compute the gradient at the current point and take a step.
        # Skip this approach and use a simpler perturbation.

        # Simple perturbation: try moving a small amount from k to j
        amounts = [0.001 * a_j, -0.001 * a_k, 0.005 * a_j, -0.005 * a_k,
                   0.01 * a_j, -0.01 * a_k]

        best_delta = 0.0
        for delta in amounts:
            new_j = a_j + delta
            new_k = a_k - delta
            if new_j < 0 or new_k < 0:
                continue
            a[j] = new_j
            a[k] = new_k
            c_tmp = _full_convolve(a, a)
            sc_tmp = float(np.max(c_tmp))
            a[j] = a_j
            a[k] = a_k
            if sc_tmp < c_max * (1 - 1e-10):
                best_delta = delta
                c_max_try = sc_tmp

        if best_delta != 0:
            a[j] = a_j + best_delta
            a[k] = a_k - best_delta
            s = float(np.sum(a))
            a = a / s

    return best_a, _score(best_a)


def _active_set_polish(a_init, deadline, min_time_remaining=30.0):
    """Active-set gradient polish targeting the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.03
    tol_frac = 0.005
    no_improve = 0
    max_stall = 300

    for iteration in range(100000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(np.max(c))
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.05, lr * 1.01)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(0.00005, tol_frac * 0.8)
                lr *= 0.85
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(0.00005, tol_frac * 0.5)
                lr = 0.01
                no_improve = 0
                if tol_frac < 0.0001 and lr < 0.005:
                    break

        # Find active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(np.sum(active))
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute gradient: reduce active peaks
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1: 2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        alpha = min(lr * c_max, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(np.max(c_try))
            if sc_try * 2.0 * n < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1

    return best_a, _score(best_a)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / np.sum(a)
    elif kind == "cos_mod":
        if rng is None:
            freq = 5.0
        else:
            freq = rng.uniform(3, 20)
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * freq * t / n)
        a = np.maximum(a, 1e-10)
        return a / np.sum(a)
    elif kind == "periodic":
        if rng is None:
            f1, f2 = 13.0, 7.0
            a1, a2 = 0.3, 0.15
        else:
            f1 = rng.uniform(5, 30)
            f2 = rng.uniform(3, 20)
            a1 = rng.uniform(0.1, 0.4)
            a2 = rng.uniform(0.05, 0.2)
        t = np.arange(n)
        a = 1.0 + a1 * np.sin(2.0 * np.pi * f1 * t / n) + a2 * np.cos(2.0 * np.pi * f2 * t / n)
        a = np.maximum(a, 1e-10)
        return a / np.sum(a)
    elif kind == "spectral":
        # Try to make a flat power spectrum
        if rng is None:
            phases = np.random.default_rng(0).standard_normal(n)
        else:
            phases = rng.standard_normal(n)
        a = np.abs(phases)
        a = a + 0.5 * np.ones(n)
        return a / np.sum(a)
    else:
        return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    """Main search function: multi-phase optimization with L-BFGS, peak
    equalization, active-set polish, and multi-n exploration."""
    global _GLOBAL_START
    _GLOBAL_START = time.monotonic()
    t_start = _GLOBAL_START
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(np.sum(parent_raw))
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS on parent ===
    print(f"\n[P1] L-BFGS on parent n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = min(deadline - 1200, t_start + 250)
    a1, sc1 = _run_lbfgs(best_seq, p1_deadline, min_time_remaining=30.0)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1

    # === Phase 2: Peak equalization on parent ===
    print(f"\n[P2] Peak equalization n={len(best_seq)} (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 1000:
        p2_deadline = min(deadline - 1000, time.monotonic() + 150)
        a2, sc2 = _peak_equalization(best_seq, p2_deadline, min_time_remaining=30.0)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peq] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set polish on parent ===
    print(f"\n[P3] Active-set polish n={len(best_seq)} (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        p3_deadline = min(deadline - 800, time.monotonic() + 100)
        a3, sc3 = _active_set_polish(best_seq, p3_deadline, min_time_remaining=30.0)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Try larger n (2x) ===
    n_large = n_parent * 2
    print(f"\n[P4] Upsample to n={n_large} (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_large = _resample(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  Init score at n={n_large}: {sc_large_init:.10f}")

        p4_deadline = min(deadline - 200, time.monotonic() + 500)
        a4, sc4 = _run_lbfgs(a_large, p4_deadline, min_time_remaining=60.0)

            best_seq = a4.copy()

        # Peak equalization on large

    # === Phase 6: L-BFGS again with very small beta (exploitation) ===
    print(f"\n[P6] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        p6_deadline = min(deadline - 30, time.monotonic() + 100)
        n_cur = len(best_seq)
        c_cur = _full_convolve(best_seq, best_seq)
        max_c = float(np.max(c_cur))
        # Use very small beta for precise peak targeting
        beta_small = max_c * 0.0001
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - np.mean(z)
        func = _LseFunc(beta_small, n_cur)
        remaining = p6_deadline - time.monotonic()

        try:

            a6 = _softmax(result.x)

            if sc6 < best_score:

    # === Phase 7: Final active-set polish ===
