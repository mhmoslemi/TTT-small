Optimized step function search using smoothed-max objective with
direct simplex optimization, L-BFGS refinement, active-set equalization,
and multi-start strategy targeting n=8192.
"""
import numpy as np
import time
from scipy.optimize import minimize as sp_minimize


def _next_pw2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pw2(out_n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _fft_conv(a, a)
    return 2.0 * n * float(c.max())


def _score(a):
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    return _score_norm(a / s)

    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj_grad(z, T, max_active=50):
    """Objective and gradient in log-space (z -> a = softmax(z) -> smoothed max of conv(a,a))."""
    n = len(z)
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = c.max()
    c_shift = (c - c_max) / T
    e = np.exp(c_shift)
    s = e.sum()

        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        f_val = c_max

        w = e / s
        f_val = c_max + T * np.log(s)

    g_a = np.zeros(n)

        if len(active) > max_active:

            active = active[top]

            hi = min(k, n - 1)

    W = float(np.dot(a, g_a))
    g_z = a * (g_a - W)
    
    return f_val, g_z

        v = b2 * v + (1 - b2) * g_z ** 2
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip step
        step_norm = np.linalg.norm(step)
        if step_norm > 2.0:
            step *= 2.0 / step_norm
        
        z = z - step
        z = np.clip(z, -100, 100)
        z -= z.mean()
        
        if t % eval_every == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if sc < best_sc:
                best_sc = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 20:
                lr_mult *= 0.5
                no_improve = 0
                if lr_mult < 0.05:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z -= z.mean()
                    z += 0.005 * np.random.randn(n)
                    z -= z.mean()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    lr_mult = 0.3
    
    return _softmax(z), best_a, best_sc


def _active_equalize(a, deadline, tol=0.002, max_iter=500):
    """Try to equalize top convolution peaks."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv(a, a)
        c_max = float(c.max())
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        
        if len(active) < 2:
            break
        if len(active) > 20:
            active = active[np.argsort(c[active])[::-1][:20]]
        
        # Build linear system to equalize peaks
        K = len(active)
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # We want to find d such that M @ d = -c[active] + c_max (equalize to c_max)
        # With constraint sum(d) = 0 (preserve sum)
        # Minimum norm: d = M^T (M M^T + lambda*I)^{-1} (c_max - c[active])
        # Plus the sum constraint
        
        b = c_max - c[active]
        # Add regularization
        lam = 0.01
        G = M @ M.T + lam * np.eye(K)
        try:
            d = M.T @ np.linalg.solve(G, b)
        except:
            d = np.linalg.lstsq(M, b, rcond=None)[0]
        
        # Enforce sum constraint
        d = d - d.mean()
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d /= d_norm
        
        # Line search
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_sc * (1 - 1e-10):
                best_sc = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_sc


def _subgrad_polish(a, deadline, max_iter=20000, top_k=5, lr0=0.003):
    """Subgradient method for the raw max."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    lr = lr0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv(a, a)
        c_max = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.9995:
                active.append(idx)
        
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project to tangent space of simplex
        g = g - np.dot(a, g) * a  # approximate projection for log-space
        # Actually for direct space: g - mean(g) * ones... no
        # For simplex constraint, the projection of gradient is g - (g . 1/n) * 1
        # But we also have non-negativity. Let's just use g - a*(g.a)/... 
        # Simpler: subtract the mean to preserve sum
        g = g - g.mean()
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        a_new = a - lr * g
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc = _score_norm(a_new)
        if sc < best_sc * (1 - 1e-12):
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9998
        else:
            lr *= 0.97
            if lr < 1e-9:
                break
    
    return best_a, best_sc


def _init_trials(n, rng):
    """Generate various initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular
    j = np.arange(n)
    tri = 1.0 - 2.0 * np.abs(j - (n-1)/2.0) / (n-1)
    tri = np.maximum(tri, 0.1)
    inits.append(tri / tri.sum())
    
    # Gaussian (various widths)
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        sig = sigma_frac * n
        gauss = np.exp(-((j - n/2)**2) / (2*sig**2))
        inits.append(gauss / gauss.sum())
    
    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        pl = (j + 1) ** (-alpha)
        inits.append(pl / pl.sum())
    
    # Cosine
    cos_w = 0.5 * (1 + np.cos(2*np.pi*j/n))
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Bimodal
    for sep in [0.2, 0.3, 0.4]:
        bim = np.exp(-((j - n*(0.5-sep/2))**2)/(n*0.05)) + np.exp(-((j - n*(0.5+sep/2))**2)/(n*0.05))
        inits.append(bim / bim.sum())
    
    # Random Dirichlet
    for alpha in [0.1, 0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.full(n, alpha))
        inits.append(r)
    
    # Super-Gaussian (flatter top)
    for p in [4, 8, 16]:
        sg = np.cos(np.pi * j / n) ** p if n > 1 else np.ones(1)
        sg = np.maximum(sg, 0.001)
        inits.append(sg / sg.sum())
    
    return inits

        inits.insert(0, parent)
