"""
Improved multi-strategy optimizer for minimizing step function autocorrelation peak.
Key improvements: finer LSE beta schedule, very large power-sum p, Adam optimizer,
adaptive Newton, spectral-inspired perturbations, and better time allocation.
"""
import numpy as np
import time
from scipy.optimize import minimize


# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# Smoothing objectives and gradients
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    sum_cp = np.sum(np.power(c_safe, p))
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _adam_lse(a_init, deadline, min_remaining=30.0, n_stages=20):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Finer beta schedule: 20 stages from base down to very small
    ratios = [1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.055, 0.038,
              0.025, 0.016, 0.010, 0.006, 0.004, 0.0025, 0.0015, 0.0008,
              0.0004, 0.0002]
    betas = base_beta * np.array(ratios[:n_stages])

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 8192.0)
        max_steps = min(8000, max(200, int(remaining / (est_time * 10))))

        m = np.zeros(n)
        v = np.zeros(n)
        step_count = 0
        lr_base = 0.01 * max(1.0, n / 8192.0)

        for step in range(max_steps):
            if time.monotonic() + min_remaining > deadline:
                break
            if step % 25 == 0 and remaining < 5:
                break

            grad_z = _lse_gradient(z, beta, n)
            step_count += 1

            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** step_count)
            v_hat = v / (1 - beta2 ** step_count)

            # Adaptive learning rate
            lr = lr_base / (1.0 + 0.001 * step)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()

            if step % 500 == 499:
                a_check = _softmax(z)
                sc_check = _score(a_check)
                if sc_check < current_sc - 1e-15:
                    current_a = a_check
                    current_sc = sc_check
                    z = np.log(np.maximum(current_a, 1e-30))
                    z = z - z.mean()

        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_new
            z = np.log(np.maximum(current_a, 1e-30))
            z = z - z.mean()

    return current_a, current_sc

# Subgradient polish (improved with adaptive tolerance)
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps=8000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    tol_current = tol_frac

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_current)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                # Tighten tolerance as we improve
                tol_current = max(0.0003, tol_current * 0.8)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3
                    tol_current = max(0.0003, tol_frac * 0.5)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_asymmetric(a, rng, strength=0.1):
    """Create an asymmetric perturbation to break symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    # Add a linear ramp in log-space to break symmetry
    ramp = np.linspace(-1, 1, n)
    z = z + strength * ramp * rng.standard_normal()
    z = z + 0.01 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_lowpass(a, rng, sigma=0.05):
    """Perturb only low-frequency components."""
    n = len(a)
    A = np.fft.rfft(a)
    # Only perturb first n//4 frequencies
    n_low = max(2, n // 4)
    A_low = A[:n_low].copy()
    A_low += sigma * A_low * rng.standard_normal(n_low)
    A_new = A.copy()
    A_new[:n_low] = A_low
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "cos4":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(8.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.4, 1.0, 0.3)
        return a / a.sum()
    else:
        return np.ones(n) / n


# ============================================================
# Variance minimization (secondary objective)
# ============================================================

def _variance_obj(a):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    return float(np.mean((c - mean_c) ** 2))


def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    return g - np.mean(g)


def _variance_warm_start(a_init, deadline, n_steps=300):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.003

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad = _variance_grad(a, n)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        v_new = _variance_obj(a_new)
        v_old = _variance_obj(a)
        if v_new < v_old:
            a = a_new
            lr = min(lr * 1.02, 0.01)
        else:
            lr *= 0.85
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

# Main optimization function
# ============================================================

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining:
    - L-BFGS with fine LSE beta schedule (20 stages)
    - Adam optimizer with LSE and power-sum objectives
    - Power-sum L-BFGS up to p=4096
    - Newton active-set refinement
    - Subgradient polish with adaptive tolerance
    - Diverse restarts with asymmetric perturbations
    - Multi-n exploration
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        n_parent = 16384
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (50s) ===
    print(f"\n[Phase 1] Diverse exploration (t={time.monotonic()-t_start:.1f}s)")
    expl_deadline = min(deadline, t_start + 50)
    n_expl = max(2048, n_parent // 8) if n_parent > 4096 else 2048
    for kind in ["uniform", "tapered", "cos2", "wheatstone", "gauss",
                 "bimodal", "ramp", "cos4", "plateau"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # Quick LSE on best
    if _best_seq is not None and n_parent > 0 and time.monotonic() < deadline - 600:
        a_up = _resize_sequence(_best_seq, n_parent) if len(_best_seq) != n_parent else _best_seq.copy()
        a_opt, sc_opt = _lbfgs_lse(a_up, deadline, min_remaining=500, n_stages=8)
        upd(a_opt, sc_opt, "expl_lbfgs")

    # === Phase 2: Main L-BFGS LSE with fine schedule (350s) ===
    print(f"\n[Phase 2] Main L-BFGS LSE (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=350, n_stages=18)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 3: Adam LSE (NEW, 200s) ===
    print(f"\n[Phase 3] Adam LSE (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a_adam, sc_adam = _adam_lse(_best_seq.copy(), deadline, min_remaining=200, n_stages=15)
        upd(a_adam, sc_adam, "adam_lse")

    # === Phase 4: Power-sum L-BFGS up to p=4096 (150s) ===
    print(f"\n[Phase 4] Power-sum L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 180:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=150)
        upd(a_ps, sc_ps, "lbfgs_power_sum")

    # === Phase 5: Adam Power-Sum (NEW, 100s) ===
    print(f"\n[Phase 5] Adam Power-Sum (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 130:
        a_ap, sc_ap = _adam_power_sum(_best_seq.copy(), deadline, min_remaining=100)
        upd(a_ap, sc_ap, "adam_power_sum")

    # === Phase 6: Newton active-set (150s) ===
    print(f"\n[Phase 6] Newton active-set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 130:
        n_dead = min(deadline - 110, time.monotonic() + 150)
        a_nw, sc_nw = _newton_active_set(_best_seq, n_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 7: Subgradient polish (120s) ===
    print(f"\n[Phase 7] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        sg_dead = min(deadline - 80, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=8000, lr_init=0.008, tol_frac=0.001)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 8: Newton + SG alternating (120s) ===
    print(f"\n[Phase 8] Newton + SG alternating (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_cur = _best_seq.copy()
        for round_i in range(4):
            if time.monotonic() > deadline - 80:
                break
            nw_dead = min(deadline - 80, time.monotonic() + 30)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.001)
            if sc_nw2 < _score(a_cur) - 1e-14:
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")

            sg_dead2 = min(deadline - 80, time.monotonic() + 25)
            a_sg2, sc_sg2 = _subgrad_polish(a_cur, sg_dead2, n_steps=3000, lr_init=0.005, tol_frac=0.0008)
            if sc_sg2 < _score(a_cur) - 1e-14:
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 9: Diverse restarts (150s) ===
    print(f"\n[Phase 9] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)
            elif i == 2:
                p_start = _perturb_block_swap(_best_seq, rng, n_swaps=4)
            elif i == 3:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.08)
            elif i == 4:
                p_start = _perturb_asymmetric(_best_seq, rng, strength=0.08)
            elif i == 5:
                p_start = _perturb_lowpass(_best_seq, rng, sigma=0.05)
            elif i == 6:
                p_start = _perturb_log(_best_seq, 0.05, rng)
            else:
                p_start = _perturb_asymmetric(_best_seq, rng, strength=0.15)

            # Use Adam for restarts (faster convergence from perturbed start)
            a_r, sc_r = _adam_lse(p_start, deadline, min_remaining=30, n_stages=8)
            upd(a_r, sc_r, f"restart_adam_{i}")

            # Newton polish
            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 30:
                nw_dead = min(deadline - 25, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 10: Multi-n exploration (150s) ===
    print(f"\n[Phase 10] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 24576, 32768]
    if n_parent not in n_candidates and n_parent >= 1024:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, deadline, min_remaining=40, n_stages=10)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        if _score(a_opt_n) < _best_score * 1.005 and time.monotonic() < deadline - 40:
            nw_dead = min(deadline - 35, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=60, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 11: Final aggressive polish (remaining time) ===
    print(f"\n[Phase 11] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()

        if remaining > 10:
            # Ultra-low beta LSE via Adam
            beta_final = mb * 0.00003
            max_steps = min(5000, max(200, int(remaining / 0.006)))

            m = np.zeros(n_fin)
            v = np.zeros(n_fin)
            step_count = 0
            beta1_adam = 0.9
            beta2_adam = 0.999
            eps_adam = 1e-8
            lr_final = 0.003 * max(1.0, n_fin / 8192.0)

            for step in range(max_steps):
                if time.monotonic() > deadline - 3:
                    break
                grad_z = _lse_gradient(z_fin, beta_final, n_fin)
                if not np.all(np.isfinite(grad_z)):
                    break
                step_count += 1
                m = beta1_adam * m + (1 - beta1_adam) * grad_z
                v = beta2_adam * v + (1 - beta2_adam) * grad_z ** 2
                m_hat = m / (1 - beta1_adam ** step_count)
                v_hat = v / (1 - beta2_adam ** step_count)
                lr = lr_final / (1.0 + 0.001 * step)
                z_fin = z_fin - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
                z_fin = np.clip(z_fin, -80.0, 80.0)
                z_fin = z_fin - z_fin.mean()

                if step % 500 == 499:
                    a_check = _softmax(z_fin)
                    sc_check = _score(a_check)
                    if sc_check < _best_score - 1e-14:
                        _best_seq = a_check.copy()
                        _best_score = sc_check
                        print(f"    Final Adam improved: {_best_score:.10f} at step {step}")

            a_fin = _softmax(z_fin)
            sc_fin = _score(a_fin)
            upd(a_fin, sc_fin, "final_ultra_adam")

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=60, tol_frac=0.0008)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=5000, lr_init=0.004, tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            if np.isfinite(eval_score):
                print(f"FINAL (evaluated): score={eval_score:.10f}, n={len(final)}, "
                      f"time={time.monotonic() - t_start:.1f}s")
            else:
                print(f"WARNING: evaluate_sequence returned non-finite, using internal score")
        except Exception:
            pass
        sc_final = _score(final)
        print(f"FINAL (internal): score={sc_final:.10f}, n={len(final)}")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
