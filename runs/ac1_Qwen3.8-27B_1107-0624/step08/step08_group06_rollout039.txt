def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))

    return e / s


def _hankel_matvec(q_vec, v, n):
    """Compute (Q v) where Q[i,j] = q_vec[i+j] (Hankel matrix action)."""
    full_len = 2 * n - 1
    q_full = q_vec if len(q_vec) == full_len else np.pad(q_vec, (0, full_len - len(q_vec)))
    out = np.zeros(n)
    for k in range(full_len):
        if abs(q_full[k]) < 1e-30:
            continue
        # (Q v)[i] += q[k] * v[k-i] for valid i
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        j_lo = k - i_hi
        j_hi = k - i_lo
        if i_lo <= i_hi and j_lo <= j_hi:
            out[i_lo:i_hi + 1] += q_full[k] * v[j_lo:j_hi + 1]
    return out

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not cond.any():
        return np.ones(n) / n
    i = np.max(np.where(cond)[0])
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)


def _cg_solve_hankel(q_vec, rhs, n, max_iter=200, tol=1e-10):
    """Solve Q x = rhs where Q is the Hankel matrix defined by q_vec."""
    def matvec(v):
        return _hankel_matvec_fast(q_vec, v, n)
    
    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    
    for _ in range(max_iter):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        if pAp < 0:
            # Q is not PD, use a different approach
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        if not np.isfinite(rs_new):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _inner_qp(q_vec, n, max_proj_iter=50):
    """Solve min a^T Q a s.t. a in simplex, where Q is Hankel defined by q_vec."""
    rhs = np.ones(n)
    x = _cg_solve_hankel(q_vec, rhs, n, max_iter=100)
    
    if x.min() >= -1e-8 and np.all(np.isfinite(x)):
        a = np.maximum(x, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
            return a
    
    # Fallback: projected gradient
    a = np.ones(n) / n
    for _ in range(max_proj_iter):
        grad = 2.0 * _hankel_matvec_fast(q_vec, a, n)
        step = 1.0 / (np.linalg.norm(grad) + 1e-10)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        # Check convergence
        if np.linalg.norm(a_new - a) < 1e-12:
            break
        a = a_new
    return a


def _mw_optimize(a_init, n, deadline, max_outer=5000):
    """Multiplicative weights saddle-point optimization."""
    t0 = time.monotonic()
    full_len = 2 * n - 1
    q = np.zeros(full_len)
    # Initialize q concentrated near the center (peak of uniform autocorrelation)
    center = n - 1
    spread = max(5, n // 100)
    k_idx = np.arange(max(0, center - spread), min(full_len, center + spread + 1))
    q[k_idx] = np.exp(-0.5 * ((k_idx - center) / (spread / 3)) ** 2)
    q = q / q.sum()
    
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    eta = 0.5  # MW temperature
    n_iter = 0
    
    while time.monotonic() < deadline and n_iter < max_outer:
        # Compute f_k(a) = conv(a,a)[k]
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(np.dot(q, c))
        
        # Update q: softmax with temperature eta
        # q_k <- exp(eta * c_k / c_scale) / Z
        c_scale = max(c_max, 1e-15)
        exponents = eta * c / c_scale
        exponents -= exponents.max()
        q_new = np.exp(exponents)
        q_sum = q_new.sum()
        if q_sum < 1e-300 or not np.isfinite(q_sum):
            q_new = np.zeros_like(q)
            q_new[int(np.argmax(c))] = 1.0
        else:
            q_new = q_new / q_sum
        
        # Blend with old q for stability
        alpha_blend = min(1.0, 0.3 + 0.7 * n_iter / 100)
        q = alpha_blend * q_new + (1 - alpha_blend) * q
        q = q / q.sum()
        
        # Solve inner QP: min a^T Q a s.t. a in simplex
        a_new = _inner_qp(q, n)
        
        # Check if a_new is good
        if a_new is not None and np.all(np.isfinite(a_new)):
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            # Blend for stability
            blend = min(1.0, 0.5 + 0.5 * n_iter / 50)
            a = blend * a_new + (1 - blend) * a
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
        
        n_iter += 1
        if n_iter % 500 == 0:
            sc_check = _score(best_a)
            print(f"    MW iter {n_iter}: best_score={sc_check:.10f}, c_max={c_max:.6f}, "
                  f"q_support={int((q > 1e-10).sum())}, t={time.monotonic()-t0:.1f}s")
    
    # Return the best found
    final_sc = _score(best_a)
    if final_sc < best_score:
        best_score = final_sc
    return best_a, best_score, n_iter

    h = h_full[n - 1:2 * n - 1]
    
    # The softmax gradient: dz_j of the objective
    # obj = 2n * (c_max + beta * log(sum exp((c-c_max)/beta)))
    # d(obj)/dz_j = 2n * (w @ d_c/dz_j)
    # d_c[k]/dz_j = 2*a[j]*a[k-j] (using d(a[j])/dz_j = a[j](delta_jk - a[k]))
    # This gets complicated. Let me use the chain rule more carefully.
    
    # The standard approach: if we parameterize a = softmax(z), then
    # d/dz_j [f(a)] = sum_i (df/da[i]) * (da[i]/dz_j)
    # da[i]/dz_j = a[i](delta_ij - a[j])
    # So d/dz_j f = a[j] * (sum_i df/da[i] * a[i] / a[j]) ... 
    # Actually: sum_i (df/da[i]) * a[i](delta_ij - a[j])
    # = a[j] * (df/da[j]) - a[j] * sum_i (df/da[i]) * a[i]
    
    df_da = 4.0 * n * h  # 2n * 2 * h
    
    grad_z = a * (df_da - np.dot(a, df_da))
    
    return grad_z


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation on log-sum-exp smoothed objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                _lsobj,
                z,
                args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-13,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_shaving(a, deadline, n_iters=200):
    """Targeted peak reduction: reduce coordinates contributing most to peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        n_peaks = max(3, min(len(c), 20))
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        
        a_new = a.copy()
        for k_pos in top_peaks:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            # Find the pair with largest product
            max_prod = 0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                j = k_pos - i
                prod = a_new[i] * a_new[j]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                # Reduce both proportionally
                reduction = 0.99
                a_new[max_i] *= reduction
                if j != max_i:
                    a_new[j] *= reduction
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller reduction
            a_try = a.copy()
            for k_pos in top_peaks[:5]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    j = k_pos - i
                    prod = a_try[i] * a_try[j]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    a_try[max_i] *= 0.999
                    if j != max_i:
                        a_try[j] *= 0.999
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
            else:
                break
    
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = max(1, n // 4)
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "noise":
        rng = np.random.default_rng(12345)
        a = np.abs(rng.standard_normal(n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _active_set_newton(a_init, deadline, max_iter=300, tol_list=None):
    """Active-set Newton method."""
    if tol_list is None:
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Try multiple tolerance levels
        best_dir = None
        best_t = 0
        best_dir_score = np.inf
        
        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[int(np.argmax(c))] = True
            m = int(active.sum())
            if m < 2 or m > 8 * n:
                if m > 8 * n:
                    top_idx = np.argsort(c)[::-1][:8 * n]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                continue
            
            A_vec = active.astype(np.float64)
            
            # Gradient: (2/m) * correlation of a with A_vec
            flip_a = a[::-1]
            out_len = n + len(A_vec) - 1
            N = _next_pow2(out_len)
            fa = np.fft.rfft(flip_a, N)
            fA = np.fft.rfft(A_vec, N)
            g = np.fft.irfft(fa * fA, N)[:out_len]
            g = g[n - 1:2 * n - 1]
            g = (2.0 / m) * g
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            
            # Hessian approximation: (2/m) * correlation operator + mu * I
            mu = 1e-6 * (c_max / max(m, 1)) + 1e-8
            A_vec_fixed = A_vec
            m_fixed = m
            mu_fixed = mu
            
            def hess_vp(v):
                flip_v = v[::-1]
                out_l = n + len(A_vec_fixed) - 1
                N_l = _next_pow2(out_l)
                fv = np.fft.rfft(flip_v, N_l)
                fA_f = np.fft.rfft(A_vec_fixed, N_l)
                r = np.fft.irfft(fv * fA_f, N_l)[:out_l]
                r = r[n - 1:2 * n - 1]
                return (2.0 / m_fixed) * r + mu_fixed * v
            
            # CG for H * u = g_t
            u = _cg_solve(hess_vp, g_t, max_iter=150)
            w = _cg_solve(hess_vp, np.ones(n), max_iter=150)
            w_sum = float(w.sum())
            
            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w
            
            dx = dx - np.mean(dx)
            
            # Line search
            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)
            
            t = min(1.0, t_max)
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            
            if sc_try < best_dir_score:
                best_dir_score = sc_try
                best_dir = dx
                best_t = t
        
        if best_dir is None:
            # Gradient fallback
            g_all = _full_convolve(a[::-1], np.ones(2 * n - 1))
            g_all = g_all[n - 1:2 * n - 1]
            g_t = g_all - np.mean(g_all)
            if np.linalg.norm(g_t) < 1e-14:
                break
            t = 0.05
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                if _score(a_try) < best_sc - 1e-13:
                    a = a_try
                    break
                t *= 0.5
            else:
                break
        
        # Full line search
        f_old = c_max
        accepted = False
        t = best_t
        for ls in range(40):
            a_try = a + t * best_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            f_try = float(_full_convolve(a_try, a_try).max())
            if f_try <= f_old - 1e-12 * t or ls >= 30:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5
        
        if not accepted:
            break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        
        if abs(float(_full_convolve(a, a).max()) - f_old) < 1e-13:
            break
    
    return best_a, best_sc


def _cg_solve(matvec, rhs, max_iter=200, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_iter):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        if pAp < 0:
            break
        alpha = rs / pAp
        if not np.isfinite(alpha):
            break
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if not np.isfinite(rs_new):
            break
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Multiplicative Weights saddle-point method (primary)
    2. L-BFGS polish on log-sum-exp
    3. Active-set Newton
    4. Peak shaving
    5. Multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # ============================================================
    # PHASE 1: MW Method (primary optimizer) on parent n
    # ============================================================
    print(f"\n[Phase 1] MW method on n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    mw_deadline = min(deadline - 600, t_start + 400)
    a_mw, sc_mw, n_mw_iter = _mw_optimize(parent, n_parent, mw_deadline, max_outer=5000)
    update_best(a_mw, sc_mw, f"mw_{n_parent}")

    # ============================================================
    # PHASE 2: L-BFGS polish
    # ============================================================
    print(f"\n[Phase 2] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        lbfgs_dl = min(deadline - 400, time.monotonic() + 250)
        a_l = _run_lbfgs(best_seq, lbfgs_dl, min_time_remaining=300)
        update_best(a_l, _score(a_l), "lbfgs_polish_1")

    # ============================================================
    # PHASE 3: Active-set Newton
    # ============================================================
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        as_dl = min(deadline - 300, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=200,
                                         tol_list=[0.005, 0.002, 0.001, 0.0005, 0.0002])
        update_best(a_as, sc_as, "as_newton_1")
        
        # Another round with tighter tolerances
        if time.monotonic() < deadline - 200:
            as_dl2 = min(deadline - 200, time.monotonic() + 80)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=150,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001])
            update_best(a_as2, sc_as2, "as_newton_2")

    # ============================================================
    # PHASE 4: Peak shaving
    # ============================================================
    print(f"\n[Phase 4] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        ps_dl = min(deadline - 200, time.monotonic() + 60)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_dl, n_iters=100)
        update_best(a_ps, sc_ps, "peak_shaving")

    # ============================================================
    # PHASE 5: Second L-BFGS round
    # ============================================================
    print(f"\n[Phase 5] Second L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        lbfgs_dl2 = min(deadline - 150, time.monotonic() + 100)
        a_l2 = _run_lbfgs(best_seq, lbfgs_dl2, min_time_remaining=100)
        update_best(a_l2, _score(a_l2), "lbfgs_polish_2")

    # ============================================================
    # PHASE 6: Multi-n exploration
    # ============================================================
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_parent:
            continue  # Already done
        
        # Try a few initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_to_n(parent, n_try))
        inits.append(_make_profile(n_try, "uniform"))
        inits.append(_make_profile(n_try, "ramp"))
        inits.append(_make_profile(n_try, "noise"))
        
        best_n_init = None
        best_n_init_sc = np.inf
        for init in inits:
            sc = _score(init)
            if sc < best_n_init_sc:
                best_n_init_sc = sc
                best_n_init = init
        
        # MW on the best init
        if time.monotonic() < deadline - 40:
            mw_dl = min(deadline - 40, time.monotonic() + 80)
            a_mw_n, sc_mw_n, _ = _mw_optimize(best_n_init, n_try, mw_dl, max_outer=2000)
            update_best(a_mw_n, sc_mw_n, f"mw_n{n_try}")
        
        # L-BFGS polish
        if best_seq is not None and len(best_seq) == n_try and time.monotonic() < deadline - 20:
            a_ln = _run_lbfgs(best_seq, min(deadline - 15, time.monotonic() + 30), min_time_remaining=15)
            update_best(a_ln, _score(a_ln), f"lbfgs_n{n_try}")

    # ============================================================
    # PHASE 7: Diverse restarts + final polish
    # ============================================================
    print(f"\n[Phase 7] Restarts and final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        # Small perturbations
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, sigma, rng)
            if time.monotonic() < deadline - 5:
                a_r = _run_lbfgs(p_start, min(deadline - 3, time.monotonic() + 15), min_time_remaining=5)
                update_best(a_r, _score(a_r), f"restart_s{sigma}")

    # ============================================================
    # PHASE 8: Final tight optimization
    # ============================================================
    print(f"\n[Phase 8] Final optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tight MW
        if time.monotonic() < deadline - 5:
            mw_final_dl = min(deadline - 4, time.monotonic() + 20)
            a_mw_f, sc_mw_f, _ = _mw_optimize(best_seq, len(best_seq), mw_final_dl, max_outer=1000)
            update_best(a_mw_f, sc_mw_f, "mw_final")
        
        # Tight Newton
        if time.monotonic() < deadline - 3:
            a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 2, time.monotonic() + 15),
                                               max_iter=100, tol_list=[0.0005, 0.0002, 0.0001, 0.00005])
            update_best(a_fin, sc_fin, "newton_final")
        
        # Very tight L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_final = mb * 0.00005
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter = min(1000, max(30, int(remaining / 0.05)))
                try:
                    result = minimize(
                        _lsobj, z, args=(beta_final, n),
                        jac=_lsgrad, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 80}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "lbfgs_tiny_final")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
