def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() == 0:
        return np.ones(n) / n
    # Algorithm from Duchi et al. 2012
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Optimized step function search using:
1. Correct analytical gradients (dc[k]/da[m] = 2*a[k-m])
2. Projected gradient with momentum + adaptive line search
3. Aggressive active-set equalization via linear systems
4. Multiple initializations (parent, structured, Fourier-based)
5. Multi-n exploration and final coordinate descent polish

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1. Returns 2*n*max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_active(a, c, active_lags):
    """Compute gradient of max(c) using weighted active lags.
    dc[k]/da[m] = 2*a[k-m] for 0 <= k-m < n.
    """
    n = len(a)
    g = np.zeros(n)
    total_w = 0.0
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
            total_w += 1.0
    if total_w > 0:
        g /= total_w
    return g


def _get_active_lags(c, n_active, tol_frac=0.99):
    """Get active lags: those within tol_frac of max, up to n_active."""
    c_max = c.max()
    threshold = c_max * tol_frac
    active = np.where(c >= threshold)[0]
    if len(active) > n_active:
        # Take the top n_active
        top = np.argsort(c[active])[::-1][:n_active]
        active = active[top]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    return active


def _project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _adam_logit(a_init, deadline, T_start, T_end, lr_start, lr_end,
                max_iter=200000, eval_interval=50, active_k=8,
                backtrack_interval=100):
    """Adam optimizer in logit space with smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    last_improve = 0
    lr_mult = 1.0
    max_patience = 12

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.7) * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        # Get active lags with temperature-based selection
        c_max = float(c.max())
        tol = max(1 - T / c_max, 0.5)
        active = _get_active_lags(c, active_k, tol)

        g = _gradient_active(a, c, active)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Center gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 2.0:
            update = update * (2.0 / un)
        z_new = z - update
        z_new = np.clip(z_new, -70.0, 70.0)
        z_new = z_new - z_new.mean()

        if t % backtrack_interval == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z.copy()
                m_buf *= 0.3
                v_buf *= 0.3
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve = t
            else:
                patience += 1

            if patience >= max_patience:
                lr_mult *= 0.6
                patience = 0
                if lr_mult < 0.05:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.2

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _momentum_projected(a_init, deadline, max_iter=100000,
                        lr_start=0.01, lr_end=0.0001, momentum=0.9):
    """Projected gradient with momentum on the simplex."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_start
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, (it + 1) / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac

        c = _fft_convolve(a, a)
        active = _get_active_lags(c, 8, 0.99)
        g = _gradient_active(a, c, active)

        # Project gradient onto tangent space (sum(g_tangent) = 0)
        g_tangent = g - g.mean()

        v = momentum * v - lr * g_tangent
        a_new = a + v
        a_new = _project_simplex(a_new)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            a = a_new  # Still move (momentum helps escape)
            patience += 1
            if patience > 50:
                v *= 0.5
                patience = 0

    return best_a, best_score


def _active_set_solve(a_init, deadline, max_iter=5000, tol=0.001):
    """Equalize top peaks by solving linear least-squares."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Find active peaks within tolerance
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            active = np.array([np.argmax(c)])
        if len(active) > 20:
            top = np.argsort(c[active])[::-1][:20]
            active = active[top]

        K = len(active)
        # Build matrix: M[i, j] = 2*a[active[i] - j] for valid active[i]-j
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        # We want to find d such that M @ d reduces all active peaks equally
        # and sum(d) = 0 (to preserve normalization)
        # Minimize ||M @ d + 1||^2 + lambda * ||d||^2 subject to sum(d) = 0
        # Using Lagrange: (M^T M + lambda*I - uu^T/n) d = -M^T 1
        # where u = ones(n)

        lam = 0.01
        G = M.T @ M + lam * np.eye(n)  # This is n x n, too big!
        # Instead, use the normal equations in reduced form:
        # d = M^T @ (M @ M^T + lam*I)^{-1} @ (-1) + correction for sum constraint

        # Since K << n, use: d = M^T @ w where w solves (M M^T + lam I) w = -M 1 - nu * 1_K
        # The sum constraint: sum(d) = 1^T M^T w = (M 1)^T w = 0

        M1 = M @ np.ones(n)  # sum of each row of M
        # Solve: (M M^T + lam I) w + nu * 1_K = -M1
        # Constraint: M1^T w = 0
        # Augmented system:
        # [M M^T + lam I, 1] [w]   [-M1]
        # [M1^T,           0] [nu] = [  0 ]

        A_sys = np.zeros((K + 1, K + 1))
        A_sys[:K, :K] = M @ M.T + lam * np.eye(K)
        A_sys[:K, K] = 1.0
        A_sys[K, :K] = M1
        b_sys = np.zeros(K + 1)
        b_sys[:K] = -M1

        try:
            sol = np.linalg.solve(A_sys + 1e-12 * np.eye(K + 1), b_sys)
        except np.linalg.LinAlgError:
            break

        w = sol[:K]
        d = M.T @ w

        # Ensure sum(d) = 0
        d -= d.mean()

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Try step sizes
        improved = False
        for alpha in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = a + alpha * d
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            # Try with larger tol
            if tol > 0.0001:
                tol *= 0.5
            else:
                # Small perturbation to escape
                a += 0.001 * np.random.randn(n)
                a = np.maximum(a, 0)
                a /= a.sum()
                if time.monotonic() > deadline - 5:
                    break

    return best_a, best_score

def _subgradient_polish(a_init, deadline, max_iter=50000, top_k=5, lr_start=0.003):
    """Subgradient method with decreasing step size."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_start
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        active = _get_active_lags(c, top_k, 0.999)
        g = _gradient_active(a, c, active)

        # Project to tangent space
        g_t = g - g.mean()
        gn = np.linalg.norm(g_t)
        if gn < 1e-15:
            break

        step = lr * g_t / gn
        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr *= 0.9998
        else:
            patience += 1
            lr *= 0.99
            if lr < 1e-10:
                break
            a = a_new  # Still take the step for exploration

        if patience > 100:
            # Reset with perturbation
            a = best_a.copy()
            a += 0.002 * np.random.randn(n)
            a = np.maximum(a, 0)
            a /= a.sum()
            lr = lr_start * 0.5
            patience = 0

    return best_a, best_score


def _fourier_init(n, rng):
    """Initialize using random Fourier phases."""
    N = _next_pow2(2 * n - 1)
    # Random amplitudes with decay
    freqs = np.arange(N // 2 + 1)
    amps = np.exp(-0.1 * freqs / (N // 2)) * (0.5 + 0.5 * rng.random(N // 2 + 1))
    phases = 2 * np.pi * rng.random(N // 2 + 1)
    A = amps * np.exp(1j * phases)
    # Make real by symmetry
    if N // 2 + 1 > 1:
        A_full = np.zeros(N, dtype=complex)
        A_full[:N // 2 + 1] = A
        A_full[N // 2 + 1:] = np.conj(A[1:N // 2][::-1])
    else:
        A_full = A.astype(complex)
    a = np.fft.irfft(A_full, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _structured_init(n, rng):
    """Try several structured initializations."""
    inits = []
    # Gaussian
    x = np.arange(n)
    sigma = n / 4
    a = np.exp(-0.5 * ((x - n // 2) / sigma) ** 2)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    # Cosine window
    a = np.cos(np.pi * x / n) ** 2
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    # Bimodal
    a = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    inits.append(a / a.sum())
    # Random smooth
    f = np.fft.rfft(rng.standard_normal(n), _next_pow2(2 * n))
    f *= np.exp(-0.5 * (np.arange(len(f)) / (n // 4)) ** 2)
    a = np.fft.irfft(f, _next_pow2(2 * n))[:n]
    a = np.maximum(a, 0)
    if a.sum() > 0:
        inits.append(a / a.sum())
    else:
        inits.append(np.ones(n) / n)
    return inits

        a_opt, sc = _adam_logit(
            best_a, sub_dl,
            T_start=max_cb * 0.001, T_end=max_cb * 0.00005,
            lr_start=0.008, lr_end=0.0005,
            max_iter=500000, eval_interval=40, active_k=12
        )
        full_sc = _score(a_opt)
        print(f"  Phase 4: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # Phase 5: Subgradient polish (1500s to 1650s)
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Subgradient polish")
        a_sg, sc_sg = _subgradient_polish(best_a, phase5_end, max_iter=100000,
                                          top_k=6, lr_start=0.002)
        full_sc = _score(a_sg)
        print(f"  Phase 5: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_sg.copy()

    # Phase 6: Final active set + coordinate descent (1650s to deadline-10)
    phase6_end = max(deadline - 10, time.monotonic() + 10)
    if time.monotonic() < phase6_end:
        print(f"Phase 6: Final polish")
        # One more active set round
        sub_dl = min(phase6_end - 20, time.monotonic() + 40.0)
        a_as, sc_as = _active_set_solve(best_a, sub_dl, max_iter=2000, tol=0.0003)
        full_sc = _score(a_as)
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_as.copy()
            print(f"  Final AS: score={full_sc:.10f}")

        # Coordinate descent
        if time.monotonic() < phase6_end - 5:
            a_cd, sc_cd = _coord_descent(best_a, phase6_end - 5, max_rounds=100)
            full_sc = _score(a_cd)
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_cd.copy()
                print(f"  Final CD: score={full_sc:.10f}")

        # Tiny subgradient
        if time.monotonic() < phase6_end - 3:
            a_sg2, sc_sg2 = _subgradient_polish(best_a, phase6_end - 3,
                                                max_iter=20000, top_k=3, lr_start=0.0005)
            full_sc = _score(a_sg2)
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_sg2.copy()
                print(f"  Final SG: score={full_sc:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
