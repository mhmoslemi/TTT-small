import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _hankel_matvec_precond(v, W, n, diag_inv):
    """Hankel matvec with diagonal preconditioner applied to input."""
    return _hankel_matvec(v * diag_inv, W, n)


def _cg_solve_precond(W, n, rhs, max_iter=800, tol=1e-14, diag_inv=None):
    """Preconditioned CG for the Hankel system Hx=b where H[i,j]=W[i+j]."""
    if diag_inv is None:
        diag_inv = np.ones(n)
    x = np.zeros(n)
    r = rhs.copy()
    # Apply preconditioner: z = M^{-1} r
    z = r * diag_inv
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-40:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-40 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rz_new_val = float(r_new @ r_new)
        if not np.isfinite(rz_new_val) or rz_new_val > rz0 * 10:
            break
        if rz_new_val < tol * tol * rz0:
            x = x_new
            break
        z_new = r_new * diag_inv
        rz_new = float(r_new @ z_new)
        if abs(rz_new) < 1e-40:
            x = x_new
            break
        beta_cg = rz_new / rz
        p = z_new + beta_cg * p
        x = x_new
        r = r_new
        rz = rz_new
    return x


def _compute_diag_precond(W, n):
    """Compute diagonal of the Hankel matrix H[i,j] = W[i+j] for preconditioning."""
    diag = W[:n].copy()
    # Ensure positivity
    diag = np.maximum(diag, 1e-20)
    return 1.0 / np.sqrt(diag)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGrad:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):

def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):

    # Initial beta based on spread of convolution

    idx = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    beta = max(gap * 0.3, max_c * 0.002)
    min_beta = max_c * 1e-10

    while beta > min_beta:

        remaining = deadline - min_time_remaining - time.monotonic()

        scale = max(1.0, n / 8192.0)

        try:

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.9
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        fade = 0.15
        mask = (t < fade) | (t > 1 - fade)
        a[mask] = 0.3
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick profile exploration
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    profile_kinds = ["tapered", "cos", "gaussian", "flat_top", "asym_taper"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for pk in profile_kinds:
        if time.monotonic() > deadline - 600:
            break
        a_p = _make_profile(n_parent, pk)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
    print(f"  Best profile: {best_profile_sc:.10f}")

    # Quick Newton on best profile
    if time.monotonic() < deadline - 500:
        a_q, sc_q = _newton_active_polish(best_profile_a, min(deadline - 500, time.monotonic() + 90), max_rounds=60)
        update_best(a_q, sc_q, "profile_newton")
        if sc_q < best_profile_sc:
            best_profile_a = a_q
            best_profile_sc = sc_q

    # Phase 2: Main Newton optimization from parent
    print(f"\n[Phase 2] Main Newton (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else parent
    t_p2_end = min(deadline - 400, t_start + 500)
    a_main, sc_main = _newton_active_polish(start_a, t_p2_end, max_rounds=100, verbose=True)
    update_best(a_main, sc_main, "newton_main")

    # Phase 3: L-BFGS refinement
    print(f"\n[Phase 3] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_lb = _run_adaptive_lbfgs(best_seq, min(deadline - 300, time.monotonic() + 300), min_time_remaining=250.0)
        sc_lb = _score(a_lb)
        update_best(a_lb, sc_lb, "lbfgs")

    # Phase 4: Newton after L-BFGS
    print(f"\n[Phase 4] Newton after L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_n2, sc_n2 = _newton_active_polish(best_seq, min(deadline - 200, time.monotonic() + 180), max_rounds=80, verbose=True)
        update_best(a_n2, sc_n2, "newton_2")

    # Phase 5: Narrow band polish
    print(f"\n[Phase 5] Narrow band polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_nb, sc_nb = _narrow_band_polish(best_seq, min(deadline - 100, time.monotonic() + 80), max_rounds=100)
        update_best(a_nb, sc_nb, "narrow_band")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 24576, 12288]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        a_init = _resample(best_seq, n_try)
        # Quick Newton
        if time.monotonic() < deadline - 40:
            a_opt, sc_opt = _newton_active_polish(a_init, min(deadline - 40, time.monotonic() + 60), max_rounds=40)
            update_best(a_opt, sc_opt, f"n_{n_try}")
        # L-BFGS if still time
        if time.monotonic() < deadline - 30:
            a_lb2 = _run_adaptive_lbfgs(best_seq if len(best_seq) != n_try else a_init,
                                         min(deadline - 30, time.monotonic() + 40), min_time_remaining=25.0)
            if len(a_lb2) == n_try:
                sc_lb2 = _score(a_lb2)
                update_best(a_lb2, sc_lb2, f"n_{n_try}_lb")
                if sc_lb2 < best_score + 0.001:
                    a_np, sc_np = _narrow_band_polish(a_lb2, deadline - 20, max_rounds=30)
                    update_best(a_np, sc_np, f"n_{n_try}_polish")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        # Try perturbed versions
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 20:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _newton_active_polish(a_pert, min(deadline - 15, time.monotonic() + 30), max_rounds=30)
            update_best(a_r, sc_r, f"perturb_{sigma}")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _narrow_band_polish(best_seq, deadline - 2, max_rounds=100)
        update_best(a_fin, sc_fin, "final_polish")

    # Phase 9: One more Newton with widest active set (for different geometry)
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        # Use a very wide active set
        W_wide = np.zeros(2 * n - 1)
        thresh = np.sort(c_best)[::-1][min(n, len(c_best) - 1)]
        W_wide[c_best >= thresh] = 1.0
        w_sum = W_wide.sum()
        if w_sum > 0:
            W_wide /= w_sum
            g_wide = _hankel_matvec(best_seq, W_wide, n)
            g_t = g_wide - np.mean(g_wide)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-14:
                diag_inv = _compute_diag_precond(W_wide, n)
                Hg = _cg_solve_precond(W_wide, n, g_wide, max_iter=300, tol=1e-13, diag_inv=diag_inv)
                H1 = _cg_solve_precond(W_wide, n, np.ones(n), max_iter=300, tol=1e-13, diag_inv=diag_inv)
                H1n = float(np.linalg.norm(H1))
                if H1n > 1e-20:
                    lam = float(np.ones(n) @ Hg) / H1n
                    dx = lam * H1 - Hg
                    dx = dx - np.mean(dx)
                    # Try a small step
                    for alpha_try in [0.01, 0.005, 0.001]:
                        a_try = best_seq + alpha_try * dx
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        sc_try = _score(a_try)
                        if sc_try < best_score:
                            update_best(a_try, sc_try, "wide_active")
                            break

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
