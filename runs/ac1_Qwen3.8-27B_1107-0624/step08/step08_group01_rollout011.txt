"""
Optimized step-function search using: (1) LSE + L-BFGS with aggressive annealing,
(2) Adam optimizer in log-domain for different landscape navigation,
(3) Palindromic constraint for reduced search space, (4) Active-set polishing,
(5) Multi-n exploration up to 24576, (6) Diverse restarts with power-mean smooth max.
Targets score below 1.5027.

import numpy as np
import time
import math
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        w = e / s
        lse_val = c_max + beta * np.log(s)
    score_val = 2.0 * n * lse_val
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = 2.0 * wc[n - 1:2 * n - 1]
    W = float(np.dot(a, h))
    grad = 2.0 * n * a * (h - W)
    return score_val, grad


def _lse_obj_fn(z, beta, n):
    val, _ = _lse_obj_grad(z, beta, n)
    return val


def _lse_grad_fn(z, beta, n):
    _, g = _lse_obj_grad(z, beta, n)
    return g


def _power_mean_obj_grad(z, p, n):
    """Power mean smooth max: (sum c_k^p)^(1/p) for large p approximates max."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    pm = float(np.sum(c ** p)) ** (1.0 / p)
    # Gradient: d/dz_j = (2n/p) * (sum c^p)^{1/p - 1} * sum_k p*c_k^{p-1} * dc_k/dz_j
    # = 2n * (sum c^p)^{1/p - 1} * sum_k c_k^{p-1} * dc_k/dz_j
    factor = 2.0 * n * (np.sum(c ** p)) ** (1.0 / p - 1.0)
    w = (c ** (p - 1)) / (c ** (p - 1)).sum()  # normalize weights
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = 2.0 * wc[n - 1:2 * n - 1]
    W = float(np.dot(a, h))
    grad = factor * (np.sum(c ** p)) ** (1.0 - 1.0 / p) * a * (h - W)
    # Simplify: factor * (sum c^p)^{1-1/p} = 2n * (sum c^p)^{1/p - 1} * (sum c^p)^{1 - 1/p} = 2n
    grad = 2.0 * n * a * (h - W)
    score_val = float(2.0 * n * pm)
    return score_val, grad


def _run_lbfgs_anneal(a_init, deadline, min_remaining=15.0, beta_schedule=None):
    """L-BFGS with beta annealing for LSE objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    if beta_schedule is None:
        beta_schedule = base_beta * np.array(
            [1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005,
             0.0025, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
             0.000025, 0.00001, 0.000005]
        )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3.0:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_iter_time * 20))))

        def obj(z_):
            return _lse_obj_fn(z_, beta, n)

        def grad(z_):
            return _lse_grad_fn(z_, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _run_adam_anneal(a_init, deadline, min_remaining=15.0):
    """Adam optimizer with LSE objective in log-domain."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

        [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in beta_schedule:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3.0:
            break

        # Adam parameters
        lr = 0.01
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0

        # Nesterov momentum on top of Adam
        z_nest = z.copy()

        max_steps = min(500, max(20, int(remaining / (0.005 * max(1.0, n / 8192.0)))))
        best_z = z.copy()
        best_obj = _lse_obj_fn(z, beta, n)

        for step in range(max_steps):
            if time.monotonic() + min_remaining > deadline:
                break
            t += 1
            # Nesterov lookahead
            z_look = z_nest + beta1 * m
            obj_val, g = _lse_obj_grad(z_look, beta, n)

            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z_nest = z

            if obj_val < best_obj:
                best_obj = obj_val
                best_z = z.copy()

            # Adaptive LR decay
            lr = 0.01 / (1 + 0.01 * t)

        a_new = _softmax(best_z)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()

def _active_set_polish(a_init, deadline, top_k=5, tol_start=0.004, tol_end=0.00001, lr_start=0.02):
    """Active-set projected gradient with top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Select active peaks
        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 5:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 2.0 / (gnorm + 1e-15))
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(tol_end, tol * 0.6)
                lr *= 0.5
            if no_improve >= 200:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.4)
                lr = lr_start * 0.3
                no_improve = 0
                if no_improve > 500:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.97)

    return best_a, _score(best_a)

            break

        def obj_half(zh):
            ah = _softmax(zh)
            # Build full palindromic sequence
            if n % 2 == 0:
                a_full = np.concatenate([ah, ah[::-1]])
            else:
                a_full = np.concatenate([ah, ah[-2::-1]])
            a_full = a_full / a_full.sum()
            c = _full_convolve(a_full, a_full)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s_e))

        def grad_half(zh):
            ah = _softmax(zh)
            if n % 2 == 0:
                a_full = np.concatenate([ah, ah[::-1]])
            else:
                a_full = np.concatenate([ah, ah[-2::-1]])
            a_full = a_full / a_full.sum()
            c = _full_convolve(a_full, a_full)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e
            a_rev = a_full[::-1]
            wc = _full_convolve(w, a_rev)
            h = 2.0 * wc[n - 1:2 * n - 1]
            W = float(np.dot(a_full, h))
            grad_full = 2.0 * n * a_full * (h - W)
            # Project back to half
            if n % 2 == 0:
                grad_half_out = grad_full[:half] + grad_full[half:][::-1]
            else:
                grad_half_out = grad_full[:half].copy()
                # The middle element appears once
                grad_half_out[-1] = grad_full[half - 1]
                # Add contributions from the mirrored part
                grad_half_out[:-1] += grad_full[half:][::-1][:-1] * 0  # already handled
            return grad_half_out

        zh = np.log(np.maximum(a_half, 1e-30))
        zh = zh - zh.mean()

        max_iter = min(2000, max(20, int((deadline - time.monotonic() - 10.0) / 0.02)))
        try:
            result = minimize(obj_half, zh, jac=grad_half, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-13})
            ah_new = _softmax(result.x)
            if n % 2 == 0:
                a_new = np.concatenate([ah_new, ah_new[::-1]])
            else:
                a_new = np.concatenate([ah_new, ah_new[-2::-1]])
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    a_half = a_new[:half].copy()
        except Exception:
            pass

    return current_a, _score(current_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 23.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.arange(1, n + 1)
        a = 1.0 / np.sqrt(t.astype(float))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _peak_shave_refine(a, deadline, n_trials=200, rng=None):
    """Targeted peak reduction by redistributing mass."""
    if rng is None:
        rng = np.random.default_rng(777)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo >= k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            continue

        # Reduce top contributors
        n_reduce = max(1, int(len(contrib) * rng.uniform(0.02, 0.1)))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * rng.uniform(0.05, 0.15)
            a_try[idx] -= red
            reduction += red

        # Distribute to low-contribution elements
        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
        for idx in top_idx:
            j_val = k_star - idx
            if 0 <= j_val < n:
                low_mask[j_val] = False

        n_low = low_mask.sum()
        if n_low == 0:
            continue

        a_try[low_mask] += reduction / n_low
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer targeting score < 1.5027."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    def _try_update(a_cand, label):
        nonlocal best_a, best_score
        if len(a_cand) == 0:
            return
        sc = _score(a_cand)
        if sc < best_score - 1e-14:
            best_a = a_cand.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: L-BFGS aggressive on parent ===
    print(f"\n[P1] L-BFGS aggressive parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200.0, t_start + 400.0)
    a1 = _run_lbfgs_anneal(best_a, p1_deadline, min_remaining=20.0)
    _try_update(a1, "lbfgs_p1")

    # === Phase 2: Adam anneal ===
    print(f"\n[P2] Adam anneal (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 1000.0, t_start + 600.0)
    a2 = _run_adam_anneal(best_a, p2_deadline, min_remaining=20.0)
    _try_update(a2, "adam_p2")

    # === Phase 3: Active-set polish ===
    print(f"\n[P3] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 800.0, t_start + 800.0)
    a3, sc3 = _active_set_polish(best_a, p3_deadline, top_k=5, tol_start=0.003,
                                  tol_end=0.00002, lr_start=0.02)
    _try_update(a3, "polish_p3")

    # === Phase 4: Peak shaving ===
    print(f"\n[P4] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 600.0, t_start + 950.0)
    a4, sc4 = _peak_shave_refine(best_a, p4_deadline, n_trials=300, rng=rng)
    _try_update(a4, "peak_shave_p4")

    # Re-optimize after peak shaving
    if _score(best_a) < best_score + 0.002 and time.monotonic() < deadline - 500:
        p4b_deadline = min(deadline - 500.0, t_start + 1050.0)
        a4b = _run_lbfgs_anneal(best_a, p4b_deadline, min_remaining=20.0)
        _try_update(a4b, "lbfgs_post_shave")

    # === Phase 5: Palindromic optimization ===
    print(f"\n[P5] Palindromic optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p5_deadline = min(deadline - 400.0, t_start + 1200.0)
        # Make initial guess palindromic
        n_cur = len(best_a)
        half = (n_cur + 1) // 2
        a_palin_init = best_a.copy()
        a_palin_init[half:] = a_palin_init[:half][::-1] if n_cur % 2 == 0 else a_palin_init[:half-1][::-1]
        a_palin_init = a_palin_init / a_palin_init.sum()
        a5, sc5 = _palindromic_optimize(a_palin_init, p5_deadline)
        _try_update(a5, "palindromic_p5")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [16384, 20480, 24576]
    n_cur = len(best_a)
    for n_try in sorted(n_targets):
        if time.monotonic() > deadline - 150.0:
            break
        if n_try == n_cur:
            continue
        if n_try > 32768:
            continue
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  Resampled to n={n_try}, init score={sc_init:.10f}")
        remaining_for_n = min(200.0, deadline - time.monotonic() - 80.0)
        if remaining_for_n < 20:
            continue
        a_opt = _run_lbfgs_anneal(a_init, time.monotonic() + remaining_for_n * 0.6, min_remaining=10.0)
        _try_update(a_opt, f"multi_n_{n_try}")
        if time.monotonic() < deadline - 60.0:
            a_pol, sc_pol = _active_set_polish(a_opt, time.monotonic() + min(60.0, deadline - time.monotonic() - 30.0),
                                                top_k=3, tol_start=0.002, tol_end=0.0001, lr_start=0.012)
            _try_update(a_pol, f"multi_n_{n_try}_polish")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80.0:
        n_best = len(best_a)
        restarts = []
        # Perturbations
        for sigma in [0.01, 0.03, 0.08]:
            z = np.log(np.maximum(best_a, 1e-30))
            z = z + sigma * rng.standard_normal(n_best)
            a_r = _softmax(z)
            restarts.append((f"pert_{sigma}", a_r))
        # Different profiles
        for kind in ["tapered", "gaussian", "cosine_mod", "bimodal"]:
            a_r = _make_profile(n_best, kind)
            restarts.append((f"profile_{kind}", a_r))

        for label, a_r in restarts:
            if time.monotonic() > deadline - 40.0:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 10.0:
                break
            a_opt = _run_lbfgs_anneal(a_r, time.monotonic() + min(60.0, remaining * 0.7), min_remaining=8.0)
            _try_update(a_opt, f"restart_{label}")

    # === Phase 8: Final ultra-fine polishing ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 3.0,
                                            tol_start=0.0003, tol_end=0.000005,
                                            lr_start=0.006, top_k=3)
        _try_update(a_fin, "final_polish")

        # Peak shave at the end
        if time.monotonic() < deadline - 3.0:
            a_ps, sc_ps = _peak_shave_refine(best_a, deadline - 2.0, n_trials=50, rng=rng)
            _try_update(a_ps, "final_peak_shave")

        # One final tiny L-BFGS
        remaining = deadline - 2.0 - time.monotonic()
        if remaining > 5.0:
            n = len(best_a)
            c_best = _full_convolve(best_a, best_a)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()

            def obj_fin(z_):
                return _lse_obj_fn(z_, beta_fin, n)

            def grad_fin(z_):
                return _lse_grad_fn(z_, beta_fin, n)

            max_iter = min(500, max(10, int(remaining / 0.05)))
            try:
                result = minimize(obj_fin, z, jac=grad_fin, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 50})
                a_fin2 = _softmax(result.x)
                a_fin2 = np.maximum(a_fin2, 0.0)
                s_fin2 = a_fin2.sum()
                if s_fin2 > 1e-10:
                    a_fin2 = a_fin2 / s_fin2
                    _try_update(a_fin2, "final_lbfgs")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
