        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient (average over active)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            g[lo:hi+1] += 2*a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Polyak step
        # alpha = (f - f_star) / ||g||^2
        g_proj = g - a * (g @ a)  # project tangent to simplex
        g_norm_sq = (g_proj @ g_proj)
        if g_norm_sq < 1e-30:
            break
        alpha = (f - f_star) / g_norm_sq
        alpha = min(alpha, lr)  # cap the step
        
        a_new = a - alpha * g_proj
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        f_new = objective(a_new)
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.95

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    """Returns 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())

def _score_full(seq):
    """Full score as in evaluate_sequence."""
    n = len(seq)
    s = sum(seq)
    if s < 0.01:
        return float('inf')
    c = _fft_convolve(np.array(seq), np.array(seq))
    return 2.0 * n * float(c.max()) / (s * s)

def _smoothed_gradient(a, c, T, top_k=10):
    n = len(a)
    c_max = c.max()
    e = np.exp((c - c_max) / T)
    e_sum = e.sum()
    if e_sum < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / e_sum
    top_idx = np.argsort(p)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        if p[k] < 1e-15:
            break
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return grad

"""
Optimized step function search using smoothed gradient descent with adaptive
temperature, active-set peak equalization, and multi-scale n exploration.
Focuses on minimizing 2*n*max(convolve(a,a)) subject to sum(a)=1, a>=0.
"""
import numpy as np
import time
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _smoothed_gradient(a, c, T, top_k=12):
    """Gradient of T*log(sum(exp(c/T))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    # Clip to avoid underflow issues
    shifted = np.clip(shifted, -500, 0)
    e = np.exp(shifted)
    e_sum = e.sum()
    if e_sum < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / e_sum
    # Only use top peaks for efficiency
    top_idx = np.argsort(p)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        wk = p[k]
        if wk < 1e-15:
            break
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * wk * a[k - hi:k - lo + 1][::-1]
    return grad


def _optimize_smoothed(a_init, deadline, T_ratio_start=0.01, T_ratio_end=0.0002,
                       alpha_start=0.008, top_k=10, max_iter=500000):
    """Smoothed gradient descent on the probability simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a /= s

    c = _fft_convolve(a, a)
    c_max = float(c.max())
    best_a = a.copy()
    best_f = 2.0 * n * c_max

    alpha = alpha_start
    no_improve = 0
    patience_limit = 80

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        f = 2.0 * n * c_max

        frac = min(1.0, it / float(max_iter))
        T = c_max * T_ratio_start * (T_ratio_end / T_ratio_start) ** (frac ** 0.6)
        T = max(T, c_max * 1e-7)

        g = _smoothed_gradient(a, c, T, top_k=top_k)

        # Project gradient to tangent space of simplex
        g_center = g - a * np.dot(g, a)

        # Step
        a_new = a - alpha * g_center
        a_new = np.maximum(0, a_new)
        sn = a_new.sum()
        if sn < 1e-15:
            alpha *= 0.5
            continue
        a_new /= sn

        c_new = _fft_convolve(a_new, a_new)
        f_new = 2.0 * n * float(c_new.max())

        if f_new < best_f * (1.0 - 1e-13):
            best_f = f_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly increase alpha on improvement
            alpha = min(alpha * 1.005, alpha_start)
        else:
            no_improve += 1
            alpha *= 0.985
            if no_improve >= patience_limit:
                # Perturbation restart
                noise_scale = 0.003 + 0.002 * (no_improve / patience_limit)
                a = best_a + noise_scale * np.random.randn(n)
                a = np.maximum(0, a)
                sn = a.sum()
                if sn > 1e-15:
                    a /= sn
                else:
                    a = best_a.copy()
                alpha = alpha_start * 0.6
                no_improve = 0

        if alpha < 1e-12:
            break

    # Final check
    sc = _score_norm(best_a)
    if sc < best_f:
        best_f = sc
    return best_a, best_f


def _active_set_equalize(a, deadline, max_iter=2000, tol_frac=0.001, max_peaks=15):
    """Active-set peak equalization via minimum-norm least squares."""
    n = len(a)
    a = a.copy()
    a = np.maximum(0, a)
    a /= a.sum()
    best_a = a.copy()
    best_f = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        tol = c_max * (1.0 - tol_frac)
        active = np.where(c >= tol)[0]
        if len(active) < 2:
            break
        if len(active) > max_peaks:
            active = active[np.argsort(c[active])[::-1][:max_peaks]]

        K = len(active)
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        M_full = np.vstack([M, np.ones(n)])
        b = np.concatenate([-np.ones(K), [0.0]])

        G = M_full @ M_full.T
        try:
            G_reg = G + 1e-12 * np.eye(K + 1)
            G_inv = np.linalg.inv(G_reg)
            d = M_full.T @ (G_inv @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm

        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            f_cand = _score_norm(a_cand)
            if f_cand < best_f * (1.0 - 1e-12):
                best_f = f_cand
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            # Try with slightly wider tolerance
            if tol_frac > 0.0001:
                tol_frac *= 0.5
            else:
                break

    return best_a, best_f


def _coordinate_descent(a, deadline, max_rounds=200):
    """Coordinate descent on elements affecting the peak most."""
    n = len(a)
    a = a.copy()
    a = np.maximum(0, a)
    a /= a.sum()
    best_a = a.copy()
    best_f = _score_norm(a)
    c = _fft_convolve(best_a, best_a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        improved = False
        # Try modifying contributing elements
        step_size = max(1, (hi - lo + 1) // 20)
        for j in range(lo, hi + 1, step_size):
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                f_cand = 2.0 * n * float(c_cand.max())
                if f_cand < best_f * (1.0 - 1e-12):
                    best_f = f_cand
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try all coordinates sparsely
            any_improve = False
            for j in range(0, n, max(1, n // 100)):
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.5, 0.75, 1.25, 1.5, 2.0]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    f_cand = 2.0 * n * float(c_cand.max())
                    if f_cand < best_f * (1.0 - 1e-12):
                        best_f = f_cand
                        best_a = a_cand.copy()
                        c = c_cand
                        any_improve = True
                        improved = True
                        break
                if improved:
                    break
            if not any_improve:
                break

    return best_a, best_f


def _subgradient_polish(a, deadline, max_iter=30000, top_k=5):
    """Direct subgradient with Polyak-like step."""
    n = len(a)
    a = a.copy()
    a = np.maximum(0, a)
    a /= a.sum()
    best_a = a.copy()
    best_f = _score_norm(a)

    # Lower bound: 2n/(2n-1) for sum=1
    f_lower = 2.0 * n / (2.0 * n - 1)
    lr = 0.005

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        f = 2.0 * n * c_max

        # Active set
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.9995:
                active.append(idx)

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        g_center = g - a * np.dot(g, a)
        g_norm_sq = np.dot(g_center, g_center)
        if g_norm_sq < 1e-20:
            break

        # Polyak step (capped)
        gap = f - f_lower
        polyak_alpha = gap / g_norm_sq
        alpha = min(lr, polyak_alpha)
        alpha = max(alpha, 1e-10)

        a_new = a - alpha * g_center
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_new /= s

        f_new = _score_norm(a_new)
        if f_new < best_f * (1.0 - 1e-13):
            best_f = f_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.01)
        else:
            lr *= 0.99
            if lr < 1e-10:
                break

    return best_a, best_f


def _try_n_value(n, parent, seed, deadline):
    """Initialize and optimize for a given n."""
    rng = np.random.default_rng(seed % (2**63 - 1))

    # Initialize
    if parent is not None and len(parent) == n:
        a = parent.copy().astype(np.float64)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s > 0.01:
            a /= s
        else:
            a = np.ones(n) / n
    elif parent is not None and len(parent) > 0:
        # Interpolate or resample parent to size n
        m = len(parent)
        if m > n:
            # Downsample
            indices = np.linspace(0, m - 1, n).astype(int)
            a = parent[indices].copy().astype(np.float64)
        else:
            # Upsample
            x_old = np.linspace(0, 1, m)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, parent).astype(np.float64)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s > 0.01:
            a /= s
        else:
            a = np.ones(n) / n
    else:
        # No parent: use structured initialization
        a = np.ones(n) / n
        # Add some structure: raised cosine modulation
        t = np.arange(n)
        a = a * (1.0 + 0.3 * np.cos(2.0 * np.pi * t / n))
        a = np.maximum(0, a)
        a /= a.sum()

    # Main optimization
    budget_remaining = deadline - time.monotonic()
    if budget_remaining < 30:
        return a, _score_norm(a)

    # Phase 1: Smoothed gradient (70% of budget)
    phase1_dl = time.monotonic() + budget_remaining * 0.5
    a_opt, sc = _optimize_smoothed(a, phase1_dl,
                                   T_ratio_start=0.01, T_ratio_end=0.0003,
                                   alpha_start=0.01, top_k=10,
                                   max_iter=300000)
    print(f"    n={n} Phase1 smoothed: {sc:.10f}")

    # Phase 2: Subgradient polish (15% of budget)
    phase2_dl = min(deadline, time.monotonic() + budget_remaining * 0.1)
    if time.monotonic() < phase2_dl:
        a_sg, sc_sg = _subgradient_polish(a_opt, phase2_dl, max_iter=20000, top_k=4)
        if sc_sg < sc:
            a_opt, sc = a_sg, sc_sg
            print(f"    n={n} Phase2 subgrad: {sc:.10f}")

    # Phase 3: Active set equalization (10% of budget)
    phase3_dl = min(deadline, time.monotonic() + budget_remaining * 0.08)
    if time.monotonic() < phase3_dl:
        a_as, sc_as = _active_set_equalize(a_opt, phase3_dl, max_iter=1000,
                                           tol_frac=0.001, max_peaks=12)
        if sc_as < sc:
            a_opt, sc = a_as, sc_as
            print(f"    n={n} Phase3 active-set: {sc:.10f}")

    # Phase 4: Coordinate descent (5% of budget)
    phase4_dl = min(deadline, time.monotonic() + budget_remaining * 0.05)
    if time.monotonic() < phase4_dl:
        a_cd, sc_cd = _coordinate_descent(a_opt, phase4_dl, max_rounds=100)
        if sc_cd < sc:
            a_opt, sc = a_cd, sc_cd
            print(f"    n={n} Phase4 coord: {sc:.10f}")

    return a_opt, sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: smoothed gradient descent, subgradient polish,
    active-set equalization, coordinate descent, and random restarts.
    Tries multiple sequence lengths to find the best.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() < 0.01:
            parent = None
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 8192
    print(f"Parent: n={n_parent}, sum={parent.sum():.4f}" if parent is not None else "No parent")

    best_overall = None
    best_score = np.inf

    # Strategy 1: Optimize from parent at its native size (50% of time)
    time_for_main = (deadline - t_start) * 0.45
    main_dl = t_start + time_for_main
    print(f"Strategy 1: Main optimization at n={n_parent}, budget={time_for_main:.0f}s")
    a_main, sc_main = _try_n_value(n_parent, parent, seed, main_dl)
    full_sc = _score_norm(a_main)
    print(f"  Result: n={n_parent}, score={full_sc:.10f}")
    if full_sc < best_score:
        best_score = full_sc
        best_overall = a_main.copy()

    # Strategy 2: Try different n values (30% of time)
    alt_ns = []
    if n_parent != 4096:
        alt_ns.append(4096)
    if n_parent != 16384:
        alt_ns.append(16384)
    if n_parent != 12288:
        alt_ns.append(12288)

    remaining = deadline - time.monotonic()
    if remaining > 60 and len(alt_ns) > 0:
        time_per_alt = max(30.0, remaining * 0.3 / len(alt_ns))
        print(f"Strategy 2: Alternative n values {alt_ns}")
        for i, n_alt in enumerate(alt_ns):
            sub_dl = min(deadline - 30, time.monotonic() + time_per_alt)
            if time.monotonic() >= sub_dl - 10:
                break
            print(f"  Trying n={n_alt}...")
            a_alt, sc_alt = _try_n_value(n_alt, parent, seed + 100 + i, sub_dl)
            full_sc_alt = _score_norm(a_alt)
            print(f"  n={n_alt}: score={full_sc_alt:.10f}")
            if full_sc_alt < best_score:
                best_score = full_sc_alt
                best_overall = a_alt.copy()

    # Strategy 3: Random restarts from best (15% of time)
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_overall is not None:
        n_restarts = min(5, int(remaining // 25))
        time_per_restart = max(10.0, remaining * 0.12 / max(1, n_restarts))
        print(f"Strategy 3: {n_restarts} random restarts")
        for r in range(n_restarts):
            sub_dl = min(deadline - 20, time.monotonic() + time_per_restart)
            if time.monotonic() >= sub_dl - 5:
                break
            sigma = 0.003 + 0.002 * r
            n_r = len(best_overall)
            a_pert = best_overall + sigma * rng.standard_normal(n_r)
            a_pert = np.maximum(0, a_pert)
            a_pert /= a_pert.sum()
            a_r, sc_r = _optimize_smoothed(a_pert, sub_dl,
                                           T_ratio_start=0.005, T_ratio_end=0.0002,
                                           alpha_start=0.006, top_k=8,
                                           max_iter=50000)
            full_sc_r = _score_norm(a_r)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_sc_r:.10f}")
            if full_sc_r < best_score:
                best_score = full_sc_r
                best_overall = a_r.copy()

    # Strategy 4: Final polish with active-set + subgradient (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_overall is not None:
        print(f"Strategy 4: Final polish, remaining={remaining:.0f}s")
        # Active set
        as_dl = min(deadline - 10, time.monotonic() + remaining * 0.5)
        a_as, sc_as = _active_set_equalize(best_overall, as_dl, max_iter=2000,
                                           tol_frac=0.0005, max_peaks=20)
        if sc_as < best_score:
            best_score = sc_as
            best_overall = a_as.copy()
            print(f"  Active-set polish: {sc_as:.10f}")

        # Subgradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            sg_dl = min(deadline - 5, time.monotonic() + remaining2 * 0.7)
            a_sg, sc_sg = _subgradient_polish(best_overall, sg_dl,
                                              max_iter=10000, top_k=3)
            if sc_sg < best_score:
                best_score = sc_sg
                best_overall = a_sg.copy()
                print(f"  Subgradient final: {sc_sg:.10f}")

        # Coordinate descent
        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            cd_dl = min(deadline - 3, time.monotonic() + remaining3 * 0.7)
            a_cd, sc_cd = _coordinate_descent(best_overall, cd_dl, max_rounds=100)
            if sc_cd < best_score:
                best_score = sc_cd
                best_overall = a_cd.copy()
                print(f"  Coordinate final: {sc_cd:.10f}")

    # Final validation
    final = np.clip(best_overall, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
        s = 256 * 0.01

    # Normalize to get the actual score
    final_norm = final / s  # sum = 1
    sc_final = _score_norm(final_norm)

    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic() - t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
