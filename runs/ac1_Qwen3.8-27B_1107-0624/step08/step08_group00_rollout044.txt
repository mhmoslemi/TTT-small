        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def _kkt_active_set_direction(a, c, active_indices, n):
    """Compute KKT-optimal subgradient direction for active peak set.
    Returns direction d (already projected to sum(d)=0) and the weights lambda."""
    m = len(active_indices)
    if m == 0:
        return None, None

    # Build B matrix: B[i, t] = a[k_t - i] with zero padding
    # For peak at position k, grad c_k(a)[i] = a[k-i] if 0 <= k-i < n, else 0
    B = np.zeros((n, m))
    for t, k in enumerate(active_indices):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                B[i, t] = a[j]

    # Center B columns (remove mean) for the QP
    B_centered = B - B.mean(axis=0, keepdims=True)

    # Solve: min lambda^T (B_c^T B_c) lambda s.t. lambda >= 0, sum(lambda) = 1
    # Using projected gradient method (fast for small m)
    M = B_centered.T @ B_centered  # m x m

    # Initialize: uniform
    lam = np.ones(m) / m

    # Projected gradient iterations
    lr_kkt = 1.0 / (np.linalg.norm(M, 2) + 1e-30)
    for _ in range(200):
        grad = 2.0 * M @ lam
        lam_new = lam - lr_kkt * grad
        # Project onto simplex
        lam_new = _project_simplex(lam_new)
        if np.linalg.norm(lam_new - lam) < 1e-12:
            lam = lam_new
            break
        lam = lam_new

    # Direction: d = -(B @ lam) + mean(B @ lam) * 1
    g = B @ lam  # the optimal subgradient
    d = -(g - g.mean())

    return d, lam


def _kkt_phase(a_init, deadline, min_time_remaining=30.0):
    """KKT active-set Newton method: iteratively find KKT-optimal direction."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.005
    min_tol_frac = 0.0001
    lr = 0.08
    no_improve = 0
    max_stall = 200
    stall_count = 0
    momentum = 0.0
    v = np.zeros(n)  # momentum accumulator

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            momentum = 0.9
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
                tol_frac = max(min_tol_frac, tol_frac * 0.8)

        if no_improve >= max_stall:
            a = best_a.copy()
            tol_frac = max(min_tol_frac, tol_frac * 0.5)
            lr = 0.08
            no_improve = 0
            stall_count += 1
            if stall_count > 10:
                break

        # Select active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active < 3:
            idx = np.argsort(c)[::-1][:max(3, n // 100)]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = len(idx)

        # Cap active set size for efficiency
        if n_active > 50:
            top_idx = np.argsort(c)[::-1][:50]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 50

        active_indices = np.where(active)[0]

        # Compute KKT direction
        d, lam = _kkt_active_set_direction(a, c, active_indices, n)
        if d is None or np.linalg.norm(d) < 1e-15:
            no_improve += 5
            continue

        # Nesterov-style momentum
        v = momentum * v - lr * d
        a_trial = a + v
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            v *= 0.5
            continue
        a_trial = a_trial / s_trial

        # Backtracking line search
        improved = False
        alpha = 1.0
        for ls_iter in range(50):
            a_try = a + alpha * v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 2
            alpha = 0.01
            a_try = a + alpha * v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc:
                    a = a_try
                    improved = True

        # Adapt parameters
        if no_improve == 0:
            tol_frac = max(min_tol_frac, tol_frac * 0.97)
            lr = min(0.15, lr * 1.03)
        elif no_improve > 50:
            tol_frac = max(min_tol_frac, tol_frac * 0.9)
            lr *= 0.85

    return best_a, _score(best_a)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive decreasing beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # Geometric sequence of betas
    betas = []
    b = base_beta
    for i in range(25):
        betas.append(b)
        b *= 0.6
    betas = np.array(betas)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 15))))

        def obj(z_):
            return _smoobj(z_, beta, n)

        def grad(z_):
            return _smograd(z_, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coordinate_descent(a_init, deadline, min_time_remaining=20.0):
    """Coordinate descent: pick the coordinate that most reduces the top peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    top_peak_idx = int(np.argmax(c))

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        top_peak_idx = int(np.argmax(c))

        # Find which coordinate to modify
        # Reducing a[i] by delta changes c[k] by -delta*(a[k-i] + a[k-n+i] ... ) approximately
        # For the top peak at k*, reducing a[i] where a[k*-i] is large helps most
        i_lo = max(0, top_peak_idx - (n - 1))
        i_hi = min(n - 1, top_peak_idx)

        # Score each coordinate by how much it contributes to the top peaks
        # Get top 5 peaks
        peak_order = np.argsort(c)[::-1][:min(10, 2 * n - 1)]

        contributions = np.zeros(n)
        for pk in peak_order:
            pk_lo = max(0, pk - (n - 1))
            pk_hi = min(n - 1, pk)
            for i in range(pk_lo, pk_hi + 1):
                j = pk - i
                contributions[i] += c[pk] * a[j]

        # Try reducing the top contributing coordinates
        n_try = max(5, n // 200)
        top_coords = np.argsort(contributions)[::-1][:n_try]

        improved = False
        for coord in top_coords:
            if time.monotonic() > deadline - min_time_remaining:
                break
            # Try reducing a[coord] and increasing others proportionally
            for frac in [0.01, 0.02, 0.05, 0.005]:
                delta = a[coord] * frac
                if delta < 1e-15:
                    continue
                a_try = a.copy()
                a_try[coord] -= delta
                a_try = np.maximum(a_try, 0.0)
                # Redistribute to maintain sum
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    current_sc = sc_try
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing low coordinates
            bottom_coords = np.argsort(a)[:n_try]
            for coord in bottom_coords[:n_try // 2]:
                for frac in [0.01, 0.02]:
                    delta = a.mean() * frac
                    a_try = a.copy()
                    a_try[coord] += delta
                    # Reduce highest to maintain sum
                    max_idx = int(np.argmax(a_try))
                    a_try[max_idx] -= delta
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-14:
                        a = a_try
                        current_sc = sc_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "antiperiodic":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [perturb_s{sigma}] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 10:
                a_kkt, sc_kkt = _kkt_phase(best_seq, min(deadline - 8, time.monotonic() + 10), min_time_remaining=5)
                if sc_kkt < best_score:
                    best_seq = a_kkt.copy()
                    best_score = sc_kkt
                    print(f"  NEW BEST: {best_score:.10f} [perturb_kkt_s{sigma}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final KKT + L-BFGS ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        a_fin, sc_fin = _kkt_phase(best_seq, deadline - 3, min_time_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_kkt] t={time.monotonic()-t_start:.1f}s")

        remaining2 = deadline - 3 - time.monotonic()
        if remaining2 > 8:
            a_fin2 = _run_lbfgs(best_seq, deadline - 2, min_time_remaining=1)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Verify with evaluate_sequence
        val = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence: {val:.10f}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
