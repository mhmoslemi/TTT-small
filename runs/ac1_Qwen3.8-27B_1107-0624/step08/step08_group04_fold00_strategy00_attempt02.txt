        a = a / a.sum()  # ensure normalization
        conv = full_convolve(a, a)
        return 2 * n * conv.max() / (a.sum() ** 2)
    
    try:
        result = minimize(objective_vector, np.log(seq), method='CG', options={'maxiter': 500, 'disp': True})
        # Verify the improvement with exact eval
        candidate = result.x
        cand_seq = np.exp(candidate) / np.exp(candidate).sum()
        score = evaluate_sequence(np.maximum(cand_seq, 1e-10))
        if np.isfinite(score) and score < evaluate_sequence(seq) - 1e-6:
            return cand_seq, score
        return None, evaluate_sequence(seq)  # no improvement
    except Exception as e:
        print(f"Optimizer failed: {str(e)}, continuing with current solution")
        return None, evaluate_sequence(seq)

def build_candidate_from_restart(old_seq, new_len=None):
    n_old = len(old_seq)
    if new_len is None:
        new_len = n_old
    if n_old == 0:
        return np.ones(new_len) / new_len
    # Smoothly reshape the sequence to the new length
    if new_len >= n_old:
        start = int(np.floor(n_old * (new_len/n_old - 1)) * 0.5)
        # Use a simple geometric growth
        growth = np.sqrt(new_len / n_old)
        base = start * growth
        arr = np.interp(np.linspace(0,1,new_len), np.linspace(0,1,n_old), np.maximum(old_seq, 1e-10) * growth)
    else:
        # Reduce the sequence length
        # Take the most significant components
        arr = np.maximum(old_seq[:new_len], 1e-4)
        
    return arr / arr.sum()
