"""
Optimized step function search using adaptive mirror descent with:
- Two-scale temperature/LR adaptation with reheating
- Soft active-set identification (threshold-based, not fixed-K)
- Direct residual-driven equalization via minimum-norm correction
- Stochastic subgradient refinement with importance sampling
- Structured restarts (Gaussian, sinusoidal, block) to escape local minima
- Quick multi-n exploration at the end
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_at_lags(a, c, lags, weights):
    """Compute weighted gradient of sum_k w[k]*b[k] w.r.t. a.
    db[k]/da[i] = 2*a[k-i] for valid k-i."""
    n = len(a)
    g = np.zeros(n)
    for idx, k in enumerate(lags):
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _adaptive_adam(a_init, deadline, T_init, lr_init, max_iter=500000,
                   eval_interval=80, threshold_frac=0.97, max_active=20):
    """Adam in log-space with adaptive temperature and threshold-based active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    T = T_init
    lr = lr_init
    lr_mult = 1.0
    last_improve = 0
    stall_count = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Threshold-based active set: include all lags within threshold_frac of max
        thresh = c_max * threshold_frac
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > max_active:
            # Keep top max_active
            order = np.argsort(c[active])[::-1][:max_active]
            active = active[order]

        # Soft weights via softmax over active lags
        c_act = c[active]
        gap = c_act.max() - c_act.min()
        T_local = T * (1.0 + 0.5 * gap / (c_max + 1e-15))
        w = np.exp((c_act - c_act.max()) / T_local)
        w = w / w.sum()

        g = _grad_at_lags(a, c, active, w)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Center gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * lr_mult * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.8:
            update = update * (0.8 / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Periodic evaluation and adaptation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve = t
                # Decrease T and LR on progress
                T *= 0.95
                lr *= 0.999
            else:
                patience += 1
                stall_count += 1

            # Adaptive response to stalls
            if patience > 6:
                if stall_count % 4 == 0 and lr_mult > 0.05:
                    lr_mult *= 0.7
                if stall_count > 12:
                    # Reheat: increase T, reset buffers, perturb
                    T = max(T * 3.0, c_max * 0.005)
                    lr_mult = max(lr_mult, 0.3)
                    m_buf *= 0.1
                    v_buf *= 0.1
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    stall_count = 0
                    patience = 0

            # If T is too small, it adds no useful smoothing
            if T < c_max * 1e-6:
                T = c_max * 1e-6

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _peak_equalization(a, deadline, max_iter=3000, tol_frac=0.995, max_peaks=10):
    """Aggressively equalize top peaks by solving minimum-norm linear systems."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Find peaks to equalize
        thresh = c_max * tol_frac
        active = np.where(c >= thresh)[0]
        if len(active) < 2:
            break
        if len(active) > max_peaks:
            order = np.argsort(c[active])[::-1][:max_peaks]
            active = active[order]

        K = len(active)
        # Build Jacobian: for each active lag k, row is 2*a[k-i] for valid i
        J = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                J[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        # We want to find d such that J @ d reduces all active peaks equally
        # Target: make all active peaks equal to c_target (slightly below c_max)
        c_target = c[active].mean()
        b_vec = c_target - c[active]  # desired change for each peak

        # Also require sum(d) = 0 (preserve normalization)
        # Minimum norm solution with constraint:
        # [J; 1^T] @ d = [b_vec; 0]
        # d = [J; 1^T]^T @ inv([J; 1^T] @ [J; 1^T]^T) @ [b_vec; 0]

        J_aug = np.vstack([J, np.ones(n)])
        b_aug = np.concatenate([b_vec, [0.0]])

        G = J_aug @ J_aug.T
        try:
            G_inv = np.linalg.inv(G + 1e-10 * np.eye(G.shape[0]))
            d = J_aug.T @ (G_inv @ b_aug)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(J_aug, b_aug, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Line search
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            # Try negative direction
            for alpha in [0.02, 0.01, 0.005, 0.002]:
                a_cand = a - alpha * d
                a_cand = np.maximum(0, a_cand)
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                sc = _score_norm(a_cand)
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_score


def _stochastic_refine(a, deadline, max_iter=50000, lr_init=0.002):
    """Stochastic subgradient with importance sampling over lags."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Importance sample: probability proportional to c[k]^2
        probs = c ** 2
        probs /= probs.sum()

        # Sample K lags
        K = min(5, 2 * n - 1)
        samples = np.random.choice(2 * n - 1, size=K, replace=False, p=probs)

        # Compute subgradient (average over sampled lags, weighted)
        g = np.zeros(n)
        for k in samples:
            w = c[k] / c_max
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= K

        # Project to tangent space of simplex
        g = g - np.dot(a, g)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        step = lr * g / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.9
            continue
        a_new /= s

        if it % 50 == 0:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                lr *= 0.999
            else:
                lr *= 0.97
                if lr < 1e-10:
                    break
        else:
            a = a_new

    # Final check
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def _try_different_n(a_parent, deadline, n_new):
    """Try a different sequence length by interpolating or truncating."""
    if time.monotonic() > deadline:
        return None, np.inf

    n_old = len(a_parent)
    if n_new == n_old:
        return a_parent.copy(), _score_norm(a_parent)

    # Interpolate to new length
    indices_old = np.linspace(0, 1, n_old)
    indices_new = np.linspace(0, 1, n_new)
    a_new = np.interp(indices_new, indices_old, a_parent)
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s < 1e-15:
        return None, np.inf
    a_new /= s

    # Quick optimization
    c = _fft_convolve(a_new, a_new)
    T_init = float(c.max()) * 0.01
    a_opt, sc = _adaptive_adam(a_new, deadline, T_init, 0.02, max_iter=100000,
                               eval_interval=100, threshold_frac=0.97, max_active=15)
    return a_opt, _score_norm(a_opt)

    print(f"  Phase 1: score={sc_full:.10f}, t={time.monotonic()-t0:.0f}s")
    if sc_full < best_score:
        best_score = sc_full
        best_a = a_opt.copy()

    # Phase 2: Peak equalization (450-700s)
    p2_end = min(deadline, t0 + 700.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: Peak equalization [450-{p2_end-t0:.0f}s]")
        a_eq, sc_eq = _peak_equalization(best_a, p2_end, max_iter=3000,
                                         tol_frac=0.995, max_peaks=10)
        sc_eq_full = _score(a_eq)
        print(f"  Phase 2: score={sc_eq_full:.10f}, t={time.monotonic()-t0:.0f}s")
        if sc_eq_full < best_score:
            best_score = sc_eq_full
            best_a = a_eq.copy()
        # Quick Adam after equalization
        if time.monotonic() < p2_end - 10:
            c_eq = _fft_convolve(best_a, best_a)
            a_opt2, sc2 = _adaptive_adam(best_a, p2_end, float(c_eq.max()) * 0.005,
                                         0.01, max_iter=100000, eval_interval=100,
                                         threshold_frac=0.98, max_active=15)
            sc2_full = _score(a_opt2)
            if sc2_full < best_score:
                best_score = sc2_full
                best_a = a_opt2.copy()

    # Phase 3: Fine Adam with very low T (700-1000s)
    p3_end = min(deadline, t0 + 1000.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Fine Adam [700-{p3_end-t0:.0f}s]")
        c_best = _fft_convolve(best_a, best_a)
        T_fine = float(c_best.max()) * 0.001
        a_fine, sc_fine = _adaptive_adam(best_a, p3_end, T_fine, 0.008,
                                         max_iter=400000, eval_interval=60,
                                         threshold_frac=0.99, max_active=10)
        sc_fine_full = _score(a_fine)
        print(f"  Phase 3: score={sc_fine_full:.10f}, t={time.monotonic()-t0:.0f}s")
        if sc_fine_full < best_score:
            best_score = sc_fine_full
            best_a = a_fine.copy()

    # Phase 4: Stochastic refinement (1000-1200s)
    p4_end = min(deadline, t0 + 1200.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Stochastic refinement [1000-{p4_end-t0:.0f}s]")
        a_stoc, sc_stoc = _stochastic_refine(best_a, p4_end, max_iter=30000, lr_init=0.001)
        sc_stoc_full = _score(a_stoc)
        print(f"  Phase 4: score={sc_stoc_full:.10f}, t={time.monotonic()-t0:.0f}s")
        if sc_stoc_full < best_score:
            best_score = sc_stoc_full
            best_a = a_stoc.copy()

    # Phase 5: Coordinate polish + equalization (1200-1400s)
    p5_end = min(deadline, t0 + 1400.0)
    if time.monotonic() < p5_end:
        print(f"Phase 5: Coordinate polish [1200-{p5_end-t0:.0f}s]")
        half1 = max(time.monotonic(), (p5_end - time.monotonic()) / 2 + time.monotonic())
        a_cd, sc_cd = _coordinate_polish(best_a, half1, max_rounds=80)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_a = a_cd.copy()
        if time.monotonic() < p5_end - 5:
            a_eq2, sc_eq2 = _peak_equalization(best_a, p5_end, max_iter=2000,
                                               tol_frac=0.998, max_peaks=8)
            sc_eq2_full = _score(a_eq2)
            if sc_eq2_full < best_score:
                best_score = sc_eq2_full
                best_a = a_eq2.copy()
        print(f"  Phase 5: score={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # Phase 6: Structured restarts (1400-1650s)
    p6_end = min(deadline, t0 + 1650.0)
    if time.monotonic() < p6_end:
        print(f"Phase 6: Structured restarts [1400-{p6_end-t0:.0f}s]")
        n_restarts = 6
        time_per = max(15.0, (p6_end - time.monotonic()) / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > p6_end - time_per:
                break

            if r == 0:
                # Gaussian perturbation
                z = np.log(np.maximum(best_a, 1e-15))
                z += 0.01 * rng.standard_normal(n)
            elif r == 1:
                # Sinusoidal perturbation
                z = np.log(np.maximum(best_a, 1e-15))
                freq = rng.integers(3, 20)
                phase = rng.uniform(0, 2 * np.pi)
                z += 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            elif r == 2:
                # Block perturbation
                z = np.log(np.maximum(best_a, 1e-15))
                block_start = rng.integers(0, n // 2)
                block_len = n // 4
                z[block_start:block_start + block_len] += 0.03 * rng.standard_normal(block_len)
            elif r == 3:
                # Anti-peak: reduce elements contributing most to peak
                c_b = _fft_convolve(best_a, best_a)
                k_peak = int(np.argmax(c_b))
                z = np.log(np.maximum(best_a, 1e-15))
                lo = max(0, k_peak - n + 1)
                hi = min(k_peak, n - 1)
                z[lo:hi + 1] -= 0.02
            elif r == 4:
                # Large Gaussian
                z = np.log(np.maximum(best_a, 1e-15))
                z += 0.03 * rng.standard_normal(n)
            else:
                # Frequency-domain perturbation
                z = np.log(np.maximum(best_a, 1e-15))
                A = np.fft.rfft(z)
                A[1:] += 0.01 * rng.standard_normal(len(A) - 1)
                z = np.fft.irfft(A, n)

            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            T_p = float(c_p.max()) * 0.005

            sub_dl = min(p6_end - 3, time.monotonic() + time_per * 0.75)
            a_pert_opt, sc_pert = _adaptive_adam(a_pert, sub_dl, T_p, 0.015,
                                                 max_iter=80000, eval_interval=100,
                                                 threshold_frac=0.97, max_active=15)
            sc_pert_full = _score(a_pert_opt)
            print(f"  Restart[{r}]: score={sc_pert_full:.10f}, t={time.monotonic()-t0:.0f}s")
            if sc_pert_full < best_score:
                best_score = sc_pert_full
                best_a = a_pert_opt.copy()

    # Phase 7: Multi-n exploration + final polish (1650-1780s)
    p7_end = min(deadline, t0 + 1780.0)
    if time.monotonic() < p7_end - 30:
        print(f"Phase 7: Multi-n + final polish [1650-{p7_end-t0:.0f}s]")
        # Try n=4096
        n_try = 4096
        a_try, sc_try = _try_different_n(best_a, min(p7_end - 30, time.monotonic() + 60), n_try)
        if a_try is not None and np.isfinite(sc_try):
            sc_try_full = _score(a_try)
            print(f"  n={n_try}: score={sc_try_full:.10f}")
            if sc_try_full < best_score:
                best_score = sc_try_full
                best_a = a_try.copy()

        # Try n=16384
        if time.monotonic() < p7_end - 30:
            n_try2 = 16384
            a_try2, sc_try2 = _try_different_n(best_a, min(p7_end - 30, time.monotonic() + 60), n_try2)
            if a_try2 is not None and np.isfinite(sc_try2):
                sc_try2_full = _score(a_try2)
                print(f"  n={n_try2}: score={sc_try2_full:.10f}")
                if sc_try2_full < best_score:
                    best_score = sc_try2_full
                    best_a = a_try2.copy()

    # Final polish
    if time.monotonic() < deadline - 10:
        c_fin = _fft_convolve(best_a, best_a)
        T_fin = float(c_fin.max()) * 0.0005
        a_fin, sc_fin = _adaptive_adam(best_a, deadline - 5, T_fin, 0.003,
                                       max_iter=100000, eval_interval=50,
                                       threshold_frac=0.995, max_active=8)
        sc_fin_full = _score(a_fin)
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_a = a_fin.copy()

        a_eq_f, sc_eq_f = _peak_equalization(best_a, deadline - 3, max_iter=1000,
                                             tol_frac=0.999, max_peaks=6)
        sc_eq_f_full = _score(a_eq_f)
        if sc_eq_f_full < best_score:
            best_score = sc_eq_f_full
            best_a = a_eq_f.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Enumerator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
