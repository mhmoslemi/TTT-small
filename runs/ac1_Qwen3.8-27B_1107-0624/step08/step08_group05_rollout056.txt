def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _variance_objective(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    mean_c = float(c.mean())
    return float((c - mean_c) @ (c - mean_c))


def _variance_gradient(z, n):

    c = _full_convolve(a, a)
    mean_c = float(c.mean())
    centered = c - mean_c
    c_rev = centered[::-1]
    conv = _full_convolve(a, c_rev)
    h = conv[2 * n - 2:3 * n - 2]
    grad_a = 2.0 * h
    return a * (grad_a - np.dot(a, grad_a))


def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _weighted_peak_gradient(a, c, n, top_k=10):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    indices = np.argsort(c)[::-1][:top_k]
    peak_vals = c[indices]
    weights = peak_vals / peak_vals.sum()
    w = np.zeros_like(c)
    for i, idx in enumerate(indices):
        w[idx] = weights[i]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300, tol_frac=0.002):
    n = len(a_init)

        active_f = active.astype(np.float64)
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

                break

                break
            a_new = a

        sc = _score(a)

        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))

        a_v2, sc_v2 = _lbfgs_variance(_best_seq.copy(), p3_dead, min_remaining=10)
        upd(a_v2, sc_v2, "var_polish")

    # Phase 4: Newton active-set (250s)
    print("\n[Phase 4] Newton active-set")
    if time.monotonic() < deadline - 300:
        p4_dead = min(deadline - 300, time.monotonic() + 250)
        a_nw, sc_nw = _newton_active_set(_best_seq, p4_dead, max_iter=500, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # Phase 5: Subgradient polish (200s)
    print("\n[Phase 5] Subgradient polish")
    if time.monotonic() < deadline - 200:
        p5_dead = min(deadline - 200, time.monotonic() + 200)
        a_sg, sc_sg = _subgrad_polish(_best_seq, p5_dead, n_steps=15000, lr_init=0.008, tol_frac=0.001)
        upd(a_sg, sc_sg, "sg_polish")

    # Phase 6: Alternate Newton + Subgrad (150s)
    print("\n[Phase 6] Newton + SG alternate")
    if time.monotonic() < deadline - 120:
        a_cur = _best_seq.copy()
        for rd in range(4):
            if time.monotonic() > deadline - 100:
                break
            nw_dead = min(deadline - 100, time.monotonic() + 30)
            a_tmp, sc_tmp = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.002)
            if sc_tmp < _score(a_cur):
                a_cur = a_tmp
            upd(a_cur, _score(a_cur), f"alt_nw_{rd}")

            sg_dead = min(deadline - 100, time.monotonic() + 20)
            a_tmp2, sc_tmp2 = _subgrad_polish(a_cur, sg_dead, n_steps=3000, lr_init=0.005, tol_frac=0.001)
            if sc_tmp2 < _score(a_cur):
                a_cur = a_tmp2
            upd(a_cur, _score(a_cur), f"alt_sg_{rd}")

    # Phase 7: Coordinate descent polish (100s)
    print("\n[Phase 7] Coordinate descent")
    if time.monotonic() < deadline - 60:
        cd_dead = min(deadline - 60, time.monotonic() + 100)
        a_cd, sc_cd = _coord_descent_polish(_best_seq, cd_dead, n_sweeps=3)
        upd(a_cd, sc_cd, "coord_desc")

    # Phase 8: Multi-n exploration (200s)
    print("\n[Phase 8] Multi-n exploration")
    if time.monotonic() < deadline - 60:
        n_candidates = [16384, 20480, 24576, 32768]
        if n_parent not in n_candidates and 4096 <= n_parent <= 65536:
            n_candidates.append(n_parent)
        n_candidates = sorted(set(n_candidates))

        for n_try in n_candidates:
            if time.monotonic() > deadline - 40:
                break
            if len(_best_seq) == n_try:
                a_init_n = _best_seq.copy()
            else:
                a_init_n = _resize_sequence(_best_seq, n_try)

            sc_init_n = _score(a_init_n)
            if sc_init_n < _best_score:
                upd(a_init_n, sc_init_n, f"n_{n_try}_resize")

            # Quick LSE
            lse_dead = min(deadline - 30, time.monotonic() + 60)
            a_opt_n, sc_opt_n = _lbfgs_lse(a_init_n, lse_dead, min_remaining=20, n_stages=8)
            upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

            # Quick Newton
            if _score(a_opt_n) < _best_score * 1.005 and time.monotonic() < deadline - 20:
                nw_dead = min(deadline - 15, time.monotonic() + 30)
                a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=100, tol_frac=0.002)
                upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # Phase 9: Diverse restarts (100s)
    print("\n[Phase 9] Diverse restarts")
    if time.monotonic() < deadline - 30:
        for i in range(4):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(_best_seq, rng, max_amp=0.2)
            elif i == 2:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.05)
            else:
                p_start = _perturb_log(_best_seq, 0.04, rng)

            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=15, n_stages=8)
            upd(a_r, sc_r, f"restart_{i}")

            if sc_r < _best_score * 1.003 and time.monotonic() < deadline - 10:
                nw_dead = min(deadline - 5, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{i}_nw")

    # Phase 10: Final ultra-precision Newton (30s)
    print("\n[Phase 10] Final Newton")
    if _best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_set(_best_seq, deadline - 2, max_iter=200, tol_frac=0.0005)
        upd(a_fin, sc_fin, "final_newton")

    # Final ultra-LSE with very low beta
    if _best_seq is not None and time.monotonic() < deadline - 2:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 2 - time.monotonic()
        if remaining > 3:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                upd(a_fin2, sc_fin2, "final_ultra_lse")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print("FALLBACK uniform")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
