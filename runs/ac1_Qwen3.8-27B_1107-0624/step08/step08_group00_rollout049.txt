Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive L-BFGS with log-sum-exp smoothing, Nesterov-accelerated
projected subgradient, and adaptive active-set identification.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# Smoothing-based objective and gradient for L-BFGS
def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive L-BFGS with log-sum-exp smoothing, Nesterov-accelerated
projected subgradient with adaptive active sets, asymmetric perturbations

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=100.0):
    """Aggressive multi-stage L-BFGS with many beta levels."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 2.0, max_c * 0.01)

    # More aggressive: start from 2x base and go to 1e-6 of base
    n_stages = 25
    betas = base_beta * np.geomspace(1.0, 1e-6, n_stages)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 15))))

        prob = _SmoProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 5 == 0:
                    print(f"    L-BFGS stage {i}/{n_stages}: beta={beta:.2e}, score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _nesterov_polish(a_init, deadline, tol_frac_start=0.01,
                     tol_frac_end=0.00003, lr_init=0.08, momentum=0.9):
    """Nesterov-accelerated projected subgradient with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Nesterov: compute gradient at lookahead point
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        s_look = a_look.sum()
        if s_look > 1e-15:
            a_look = a_look / s_look
            c_look = _full_convolve(a_look, a_look)
            c_max_look = float(c_look.max())
            tol_look = tol_frac * c_max_look
            active_look = c_look >= (c_max_look - tol_look)
            n_active_look = max(1, int(active_look.sum()))
            w_look = np.zeros_like(c_look)
            w_look[active_look] = 1.0 / n_active_look
            a_rev_look = a_look[::-1]
            wc_look = _full_convolve(w_look, a_rev_look)
            h_look = wc_look[n - 1:2 * n - 1]
            grad_look = 2.0 * h_look
            grad_look_t = grad_look - np.mean(grad_look)
            gnorm_look = np.linalg.norm(grad_look_t)
            if gnorm_look > 1e-16:
                grad_t = grad_look_t
                gnorm = gnorm_look

        alpha = min(lr, 1.0 / (gnorm + 1e-15))

        # Line search with Nesterov step
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                v = a_try - a
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            v *= 0.5  # decay momentum on failure
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.5
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _asymmetric_perturb(a, strength, rng):
    """Apply asymmetric perturbation to break symmetry."""
    n = len(a)
    # Create a ramp-like perturbation
    t = np.linspace(0, 1, n)
    # Asymmetric: stronger on one side
    asym = strength * (t - 0.5)
    # Add some noise
    noise = strength * 0.1 * rng.standard_normal(n)
    z = np.log(np.maximum(a, 1e-30))
    z = z + asym + noise
    z = z - z.mean()
    return _softmax(z)


def _projected_gradient_simple(a_init, deadline, lr=0.05):
    """Simple projected gradient (no Nesterov) as fallback."""

    a = a_init.copy()

    best_a = a.copy()

    tol_frac = 0.005
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = max(1, int(active.sum()))
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]

            s_try = a_try.sum()
            if s_try < 1e-15:

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_initial(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.3) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer combining aggressive L-BFGS, Nesterov-accelerated
    projected subgradient, asymmetric perturbations, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_initial(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Aggressive L-BFGS on parent ===
    print(f"\n[P1] Aggressive L-BFGS on parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 800, t_start + 500)
    a_p1 = _run_lbfgs_aggressive(parent, p1_deadline, min_time_remaining=200.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  L-BFGS: score={sc_p1:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Nesterov polish ===
    print(f"\n[P2] Nesterov polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 500, t_start + 800)
    a_p2, sc_p2 = _nesterov_polish(best_seq, p2_deadline,
                                    tol_frac_start=0.008, tol_frac_end=0.00005,
                                    lr_init=0.06, momentum=0.85)
    if sc_p2 < best_score:
        best_seq = a_p2.copy()
        best_score = sc_p2
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [nesterov] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Nesterov: score={sc_p2:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Asymmetric perturbation + re-optimize ===
    print(f"\n[P3] Asymmetric perturbation (t={time.monotonic()-t_start:.1f}s)")
    for asym_strength in [0.05, 0.02, 0.01]:
        if time.monotonic() > deadline - 300:
            break
        a_asym = _asymmetric_perturb(best_seq, asym_strength, rng)
        sc_asym = _score(a_asym)
        if sc_asym < best_score:
            # Re-optimize from this point
            p3_dl = min(deadline - 250, time.monotonic() + 120)
            a_p3 = _run_lbfgs_aggressive(a_asym, p3_dl, min_time_remaining=100.0)
            sc_p3 = _score(a_p3)
            if sc_p3 < best_score:
                best_seq = a_p3.copy()
                best_score = sc_p3
                print(f"  NEW BEST: {best_score:.10f} [asym_{asym_strength}] t={time.monotonic()-t_start:.1f}s")
            # Nesterov polish
            if time.monotonic() < deadline - 200:
                p3b_dl = min(deadline - 180, time.monotonic() + 60)
                a_p3b, sc_p3b = _nesterov_polish(best_seq, p3b_dl,
                                                  tol_frac_start=0.003, tol_frac_end=0.00003,
                                                  lr_init=0.03, momentum=0.8)
                if sc_p3b < best_score:
                    best_seq = a_p3b.copy()
                    best_score = sc_p3b
                    print(f"  NEW BEST: {best_score:.10f} [asym_polish] t={time.monotonic()-t_start:.1f}s")
            break  # Only try the first successful asymmetry

    # === Phase 4: Random restarts ===
    print(f"\n[P4] Random restarts (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(5):
        if time.monotonic() > deadline - 200:
            break
        sigma = 0.02 * (1.0 - trial * 0.15)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z + sigma * rng.standard_normal(len(best_seq))
        z = z - z.mean()
        a_restart = _softmax(z)
        # Quick L-BFGS
        p4_dl = min(deadline - 150, time.monotonic() + 60)
        a_r = _run_lbfgs_aggressive(a_restart, p4_dl, min_time_remaining=60.0)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_seq = a_r.copy()
            best_score = sc_r
            print(f"  NEW BEST: {best_score:.10f} [restart_{trial}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  restart {trial}: score={sc_r:.10f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_parent]
    for n_try in [16384, 8192, 32768, 12288]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # L-BFGS
        p5_dl = min(deadline - 80, time.monotonic() + 80)
        a_opt = _run_lbfgs_aggressive(a_init, p5_dl, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs] t={time.monotonic()-t_start:.1f}s")

        # Quick Nesterov
        if time.monotonic() < deadline - 60:
            p5b_dl = min(deadline - 55, time.monotonic() + 40)
            a_opt2, sc_opt2 = _nesterov_polish(a_opt, p5b_dl,
                                               tol_frac_start=0.003, tol_frac_end=0.0001,
                                               lr_init=0.02, momentum=0.8)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_nest] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Structured initializations at different n ===
    print(f"\n[P6] Structured inits (t={time.monotonic()-t_start:.1f}s)")
    for n_try in [n_parent, 16384]:
        if time.monotonic() > deadline - 80:
            break
        for kind in ["tapered", "asym_taper", "exp_taper"]:
            if time.monotonic() > deadline - 60:
                break
            a_struct = _make_initial(n_try, kind)
            sc_struct = _score(a_struct)
            # Quick optimize
            p6_dl = min(deadline - 50, time.monotonic() + 40)
            a_s = _run_lbfgs_aggressive(a_struct, p6_dl, min_time_remaining=40.0)
            sc_s = _score(a_s)
            if sc_s < best_score:
                best_seq = a_s.copy()
                best_score = sc_s
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}, {kind}) [struct] t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  {kind} n={n_try}: score={sc_s:.10f}")

    # === Phase 7: Final deep Nesterov ===
    print(f"\n[P7] Final deep Nesterov (t={time.monotonic()-t_start:.1f}s)")
    p7_dl = deadline - 20
    a_fin, sc_fin = _nesterov_polish(best_seq, p7_dl,
                                     tol_frac_start=0.002, tol_frac_end=0.00001,
                                     lr_init=0.03, momentum=0.9)
    if sc_fin < best_score:
        best_seq = a_fin.copy()
        best_score = sc_fin
        print(f"  NEW BEST: {best_score:.10f} [final_nest] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Tiny final L-BFGS ===
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        print(f"\n[P8] Final tiny L-BFGS (remaining={remaining:.1f}s)")
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.0001
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _SmoProb(beta_fin, n)
        max_iter = min(3000, max(10, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 80})
            a_fin2 = _softmax(result.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_initial(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
