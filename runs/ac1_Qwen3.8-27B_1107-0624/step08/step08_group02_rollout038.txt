    return a


def _dp_optimize(a_init, n, deadline, max_rounds=500):
    """(d,p) alternating method with adaptive active set."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        t = float(c.max())

        # Adaptive tolerance: start wide, tighten as we converge
        if rnd < 5:
            tol_frac = 0.1
        elif rnd < 20:
            tol_frac = 0.05
        elif rnd < 50:
            tol_frac = 0.02
        elif rnd < 100:
            tol_frac = 0.01
        elif rnd < 200:
            tol_frac = 0.003
        else:
            tol_frac = 0.001

        active = c >= (t - tol_frac * t)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        # If active set is very large, subset to top contributors
        if n_active > max(200, n // 4):
            thresh = np.sort(c)[::-1][min(n_active, max(200, n // 4)) - 1]
            active = c >= thresh
            n_active = int(active.sum())

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        # Primal step: solve QP with warm start
        a_new = _solve_qp_inexact(w, n, a, max_cg=60, reg=1e-8)

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 80 and rnd > 30:
            # Diversify: add noise and reset
            a = a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_sc


def _newton_polish(a_init, deadline, max_rounds=80):
    """Newton method on the active-set KKT conditions."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with tight tolerance
        for tol_frac in [0.001, 0.005, 0.01, 0.03, 0.1]:
            active = c >= (c_max - tol_frac * c_max)
            if active.sum() >= 3:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        n_active = int(active.sum())
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        # Compute gradient of the active-set objective
        # grad_j = 2 * sum_k w[k] * (a[k-j] + a[j-k]) - mu
        Ma = _matvec_Mw(a, w, n)
        mu = float(np.sum(Ma)) / n
        grad = Ma - mu

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12 * n:
            break

        # Newton step: solve M_w dx = -grad with constraint 1^T dx = 0
        rhs = -grad
        rhs = rhs - np.mean(rhs)

        M1 = _cg_solve(w, n, rhs, x0=np.zeros(n), max_iter=200, reg=1e-10)
        M1 = M1 - np.mean(M1)

        # Line search
        step_done = False
        alpha = 1.0
        for ls in range(60):
            a_try = a + alpha * M1
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                alpha *= 0.5
                if alpha < 1e-15:
                    break
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-14:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * grad / max(gnorm, 1e-20)
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        if not step_done:
            break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS on smoothed max problem with adaptive beta."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    c_s4 = np.sort(c_cur)[::-1]
    gap = c_s4[0] - c_s4[min(50, len(c_s4) - 1)]
    beta = max(gap * 0.3, max_c * 0.001)
    min_beta = max_c * 1e-9

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        max_iter = min(5000, max(20, int(remaining / (0.005 * scale * 30))))

        try:
            result = minimize(
                lambda zz: _smoobj(zz, beta, n),
                z,
                jac=lambda zz: _smograd(zz, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_s4_new = np.sort(c_new)[::-1]
        new_gap = c_s4_new[0] - c_s4_new[min(20, len(c_s4_new) - 1)]

        if new_gap < 1e-7 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.4
        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "u_shaped":
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t)
    elif kind == "u_shaped_strong":
        a = 1.0 + 0.8 * np.cos(2 * np.pi * t)
    elif kind == "multi_harm":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t) \
            + 0.05 * np.cos(6 * np.pi * t)
    elif kind == "asym":
        a = np.sin(np.pi * t ** 0.6) ** 0.7
    elif kind == "bimodal":
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _dual_descent_step(a, w, eta, w_old):
    """Subgradient dual update."""
    c = _full_convolve(a, a)
    w_new = w_old + eta * c
    w_new = np.maximum(w_new, 0)
    s = w_new.sum()
    if s < 1e-30:
        return w_old
    return w_new / s


def propose_candidate(seed=42, budget_s=1800):
    """Main search function: combines (d,p) method, L-BFGS, and Newton polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick profile exploration
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["tapered", "u_shaped", "u_shaped_strong", "multi_harm", "cos", "bimodal", "asym"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 800:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p.copy()
    print(f"  Best profile score: {best_profile_sc:.10f}")

    # Phase 2: (d,p) alternating method on best candidates
    print(f"\n[Phase 2] (d,p) method (t={time.monotonic() - t_start:.1f}s)")
    # Start from parent
    if time.monotonic() < deadline - 400:
        a_dp1, sc_dp1 = _dp_optimize(parent, len(parent), min(deadline - 300, t_start + 200),
                                      max_rounds=300)
        update_best(a_dp1, sc_dp1, "dp_parent")

    # Start from best profile
    if time.monotonic() < deadline - 300:
        a_dp2, sc_dp2 = _dp_optimize(best_profile_a, len(best_profile_a),
                                      min(deadline - 200, t_start + 400),
                                      max_rounds=300)
        update_best(a_dp2, sc_dp2, "dp_profile")

    # Phase 3: L-BFGS diversification
    print(f"\n[Phase 3] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_lb = _run_lbfgs(best_seq, min(deadline - 150, t_start + 600), min_time_remaining=100)
        update_best(a_lb, _score(a_lb), "lbfgs_main")

    # Newton polish after L-BFGS
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_np1, sc_np1 = _newton_polish(best_seq, min(deadline - 80, t_start + 700), max_rounds=60)
        update_best(a_np1, sc_np1, "newton_1")

    # Phase 4: More (d,p) from current best
    print(f"\n[Phase 4] (d,p) round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_dp3, sc_dp3 = _dp_optimize(best_seq, len(best_seq),
                                      min(deadline - 60, t_start + 900),
                                      max_rounds=400)
        update_best(a_dp3, sc_dp3, "dp_round2")

    # Phase 5: Try different n values
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = []
    n_base = len(best_seq) if best_seq is not None else n_parent
    for factor in [0.5, 2.0]:
        n_try = int(n_base * factor)
        n_try = max(2048, min(65536, n_try))
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    # Add some fixed candidates
    for n_try in [8192, 32768]:
        if n_try not in n_candidates and n_try != n_base:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        print(f"  Trying n={n_try}")
        a_init = _resample(best_seq if best_seq is not None else parent, n_try)
        # Quick (d,p)
        a_opt, sc_opt = _dp_optimize(a_init, n_try, min(deadline - 40, t_start + 100 + n_try / 4096),
                                      max_rounds=150)
        update_best(a_opt, sc_opt, f"dp_n{n_try}")
        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 30:
            a_np2, sc_np2 = _newton_polish(a_opt, min(deadline - 20, t_start + 120 + n_try / 4096),
                                            max_rounds=40)
            update_best(a_np2, sc_np2, f"np_n{n_try}")

    # Phase 6: Diversification with perturbation + (d,p)
    print(f"\n[Phase 6] Diversification (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 40:
                break
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
            sc_pert = _score(a_pert)
            if sc_pert < best_score + 0.01:
                a_opt, sc_opt = _dp_optimize(a_pert, n, min(deadline - 20, t_start + 1400),
                                              max_rounds=200)
                update_best(a_opt, sc_opt, f"perturb_s{sigma}")

    # Phase 7: Final Newton polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 2, max_rounds=100)
        update_best(a_fin, sc_fin, "final_newton")

        # One more (d,p) with very tight active set
        if time.monotonic() < deadline - 3:
            a_dp_final, sc_dp_final = _dp_optimize(best_seq, len(best_seq), deadline - 1,
                                                    max_rounds=100)
            update_best(a_dp_final, sc_dp_final, "final_dp")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
