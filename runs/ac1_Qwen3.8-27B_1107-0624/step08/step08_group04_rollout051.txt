Hankel-QP optimizer with softmax weight continuation, FFT-accelerated CG,
and multi-scale active-set refinement for minimizing peak autocorrelation.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z, beta):
    """Softmax with temperature beta: w_k = exp(c_k/beta) / sum exp(c_j/beta)"""
    z_shifted = z / beta - z.max() / beta
    e = np.exp(np.clip(z_shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(z)
        w[int(np.argmax(z))] = 1.0
        return w
    return e / s

    """Compute H_w * v where (H_w)_{i,j} = w_{i+j}, w has length 2n-1, v has length n."""
    V = v[::-1]
    out_len = len(V) + len(w) - 1  # n + (2n-1) - 1 = 3n-2
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n-1:2*n-1]

def _cg_hankel(w, n, rhs, max_iter=2000, tol=1e-12):
    """Solve H_w x = rhs using CG with FFT-based Hankel matvec."""
    x = np.zeros(n)
    r = rhs.copy().astype(np.float64)
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_mv(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs_new < tol * tol * rs0:
            break
        beta_cg = rs_new / rs
        p = r + beta_cg * p
        rs = rs_new
    return x

def _solve_qp_full(w, n, a_init, max_cg_iter=3000):
    """Solve min a^T H_w a s.t. sum(a)=1, a>=0 assuming full support."""
    # a = H_w^{-1} 1 / (1^T H_w^{-1} 1)
    one = np.ones(n)

    oney = float(np.sum(y))
    if abs(oney) < 1e-20 or not np.isfinite(oney):
        return a_init.copy()
    a = y / oney

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return a_init.copy()
        a = a / s
    return a

MM (Majorization-Minimization) optimizer with QP subproblems, adaptive softmax/Hankel weights,
FFT-accelerated conjugate gradient, multi-scale tolerance refinement, and multi-n exploration
for minimizing peak-to-sum autocorrelation ratio.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_mv(v, w, n):
    """Compute H_w * v where (H_w)_{i,j} = w_{i+j}, w has length 2n-1, v has length n."""
    V = v[::-1]
    out_len = len(V) + len(w) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_hankel(w, n, rhs, max_iter=5000, tol=1e-13, restart=500):
    """Solve H_w x = rhs using CG with FFT-based Hankel matvec, with periodic restarts."""
    x = np.zeros(n)
    r = rhs.copy().astype(np.float64)
    rs0 = float(r @ r)
    if rs0 < 1e-30:
        return x
    for outer in range(max(1, max_iter // restart)):
        remaining = max(1, max_iter - outer * restart)
        p = r.copy()
        rs = float(r @ r)
        if rs < tol * tol * rs0:
            break
        for i in range(min(restart, remaining)):
            Hp = _hankel_mv(p, w, n)
            pHp = float(p @ Hp)
            if abs(pHp) < 1e-30 or not np.isfinite(pHp):
                break
            alpha = rs / pHp
            x = x + alpha * p
            r = r - alpha * Hp
            rs_new = float(r @ r)
            if not np.isfinite(rs_new):
                break
            if rs_new < tol * tol * rs0:
                break
            beta_cg = rs_new / rs
            p = r + beta_cg * p
            rs = rs_new
    return x


def _solve_qp_full_support(w, n, max_cg_iter=5000):
    """Solve min a^T H_w a s.t. sum(a)=1 assuming full support. Returns (a, success)."""
    one = np.ones(n)
    y = _cg_hankel(w, n, one, max_iter=max_cg_iter)
    oney = float(np.sum(y))
    if abs(oney) < 1e-20 or not np.isfinite(oney):
        return None, False
    a = y / oney
    if np.min(a) < -1e-12:
        return a, False
    return a, True


def _solve_qp_active_set(w, n, a_init, max_outer=100, max_cg_iter=3000):
    """Solve min a^T H_w a s.t. sum(a)=1, a>=0 using active-set with full-support QP."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_mv(a, w, n))

    support = a > 1e-14
    for outer in range(max_outer):
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            n_sup = n

        if n_sup == n:
            a_new, ok = _solve_qp_full_support(w, n, max_cg_iter)
            if not ok:
                if a_new is not None:
                    a_new = np.maximum(a_new, 0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a_new = a_new / s_new
                    else:
                        break
                else:
                    break
        else:
            # Solve on reduced support using CG on submatrix
            # Extract submatrix operation via masking
            idx = np.where(support)[0]
            n_sub = len(idx)
            if n_sub < 2:
                a_new = a.copy()
            else:
                # Build the subproblem: min a_sub^T H_sub a_sub s.t. 1^T a_sub = 1
                # H_sub[i,j] = w[idx[i]+idx[j]]
                # Use CG with a custom matvec
                w_sub = w  # we still use the full w, just access different entries
                one_sub = np.ones(n_sub)

                def _sub_mv(v_sub):
                    # Compute H_sub * v_sub
                    v_full = np.zeros(n)
                    v_full[idx] = v_sub
                    result_full = _hankel_mv(v_full, w, n)
                    return result_full[idx]

                # CG on subproblem
                x_sub = np.zeros(n_sub)
                r_sub = one_sub.copy()
                p_sub = r_sub.copy()
                rs = float(r_sub @ r_sub)
                rs0 = rs
                for i in range(max_cg_iter):
                    Hp_sub = _sub_mv(p_sub)
                    pHp = float(p_sub @ Hp_sub)
                    if abs(pHp) < 1e-30 or not np.isfinite(pHp):
                        break
                    alpha = rs / pHp
                    x_sub = x_sub + alpha * p_sub
                    r_sub = r_sub - alpha * Hp_sub
                    rs_new = float(r_sub @ r_sub)
                    if not np.isfinite(rs_new):
                        break
                    if rs_new < 1e-26 * rs0:
                        break
                    beta_cg = rs_new / rs
                    p_sub = r_sub + beta_cg * p_sub
                    rs = rs_new

                oney_sub = float(np.sum(x_sub))
                if abs(oney_sub) < 1e-20:
                    break
                a_new = np.zeros(n)
                a_new[idx] = x_sub / oney_sub
                # Check KKT: for i not in support, need (H_w a)_i >= (H_w a)_support_mean
                grad_full = _hankel_mv(a_new, w, n)
                lam = float(np.mean(grad_full[idx]))
                if np.any(grad_full[~support] < lam - 1e-10):
                    # Some inactive constraints violated - add them to support
                    violated = (~support) & (grad_full < lam - 1e-10)
                    support[violated] = True
                    a_new = np.maximum(a_new, 0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a_new = a_new / s_new
                    continue

        val_new = float(a_new @ _hankel_mv(a_new, w, n))
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-15:
            best_val = val_new
            best_a = a_new.copy()

        # Check if we should update support
        grad = _hankel_mv(a_new, w, n)
        lam = float(np.mean(grad[support]))
        # Entries that are 0 but want to be positive
        inactive = ~support
        if inactive.any():
            max_violation = float(np.max(lam - grad[inactive]))
            if max_violation > 1e-8 * (float(np.max(np.abs(grad))) + 1e-15):
                # Add the most violated entry
                worst = np.where(inactive)[0][np.argmax(lam - grad[inactive])]
                support[worst] = True
                a_new[worst] = 1e-12
                s_fix = a_new.sum()
                if s_fix > 1e-15:
                    a_new = a_new / s_fix
                a = a_new
                continue
        # Entries in support that should be 0
        if np.any(a_new[support] < 1e-14):
            to_remove = support & (a_new < 1e-14)
            if to_remove.any():
                # Remove the most negative
                worst = np.where(to_remove)[0][np.argmin(a_new[to_remove])]
                support[worst] = False
                a_new[worst] = 0.0
                a = a_new
                continue

        a = a_new

        # Check convergence
        if support.sum() > 0:
            lag = float(np.mean(grad[support]))
            resid = float(np.max(np.abs(grad[support] - lag)))
            if (~support).any():
                resid = max(resid, float(np.max(lag - grad[~support])))
            rel_resid = resid / (float(np.max(np.abs(grad))) + 1e-15)
            if rel_resid < 1e-12:
                break

    return best_a


def _mm_optimize(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """
    MM algorithm with adaptive weight schedule.
    Alternates between: (1) compute autocorrelation, (2) choose weights, (3) solve QP.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Two-phase weight strategy:
    # Phase A: Softmax weights with decreasing temperature
    # Phase B: Hard active-set with decreasing tolerance

    # Phase A parameters
    temp_schedule = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                     0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                     0.000001]
    iters_per_temp = 3

    phaseA_deadline = deadline - min_time_remaining * 0.6

    if verbose:
        print(f"  MM Phase A (softmax weights), n={n}")

    for ti, temp in enumerate(temp_schedule):
        if time.monotonic() > phaseA_deadline:
            break
        for it in range(iters_per_temp):
            if time.monotonic() > phaseA_deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            if c_max < 1e-15:
                break

            # Softmax weights
            z = c / (temp * c_max + 1e-30)
            z = z - z.max()
            e = np.exp(np.clip(z, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            a_new = _solve_qp_active_set(w, n, a, max_outer=60, max_cg_iter=3000)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new
            if verbose and it == iters_per_temp - 1:
                print(f"    temp={temp:.6f}: score={best_sc:.10f}, t={time.monotonic()-t_start_global:.0f}s")

    # Phase B: Hard active-set with decreasing tolerance
    if verbose:
        print(f"  MM Phase B (hard active-set), n={n}")

    tol_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                    0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    iters_per_tol = 3

    for ti, tol in enumerate(tol_schedule):
        if time.monotonic() > deadline - min_time_remaining:
            break
        for it in range(iters_per_tol):
            if time.monotonic() > deadline - min_time_remaining:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            if c_max < 1e-15:
                break

            # Hard active set
            active = c >= c_max * (1.0 - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            w = np.zeros_like(c)
            w[active] = 1.0 / active.sum()

            a_new = _solve_qp_active_set(w, n, a, max_outer=80, max_cg_iter=5000)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new
            if verbose and it == iters_per_tol - 1:
                print(f"    tol={tol:.8f}: score={best_sc:.10f}, active={int(active.sum())}, t={time.monotonic()-t_start_global:.0f}s")

    # Final check
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _lbfgs_optimize(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing, adaptive beta continuation."""
    n = len(a_init)

    def _make_obj_grad(beta):
        def obj(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w_vec = np.zeros_like(c)
                w_vec[int(np.argmax(c))] = 1.0
            else:
                w_vec = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w_vec, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w_vec, c))
            return 4.0 * n * a * (h - W_val)

        return obj, grad

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Adaptive beta: start wide, narrow down
    c_init = _full_convolve(current_a, current_a)
    c_max_init = float(c_init.max())
    gap = c_max_init - float(c_init.mean())
    base_beta = max(gap * 0.1, c_max_init * 0.005)

    ratios = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        obj, grad = _make_obj_grad(beta)
        max_iter = min(2000, max(30, int(remaining / 0.1)))
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 50, 'maxls': 100})
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(np.clip(z - z.max(), -700, 0))
    return e / e.sum()

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining MM-QP, L-BFGS smoothing, and multi-n exploration.
    Key innovation: proper majorization-minimization with exact QP subproblems
    solved via FFT-accelerated CG on Hankel systems, with adaptive weight scheduling.
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start_global:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS warmup (get to a good basin) ===
    print(f"\n[Phase 1] L-BFGS warmup (t={time.monotonic() - t_start_global:.1f}s)")
    p1_deadline = min(deadline - 900.0, t_start_global + 300.0)
    if time.monotonic() < p1_deadline:
        a1 = _lbfgs_optimize(parent, deadline, min_time_remaining=900.0)
        update_best(a1, _score(a1), "lbfgs_warmup")

    # === Phase 2: MM with QP (main optimization) ===
    if best_seq is not None:
        print(f"\n[Phase 2] MM-QP main (t={time.monotonic() - t_start_global:.1f}s)")
        p2_deadline = min(deadline - 400.0, t_start_global + 1000.0)
        a2, sc2 = _mm_optimize(best_seq, p2_deadline, min_time_remaining=400.0, verbose=True)
        update_best(a2, sc2, "mm_main")

    # === Phase 3: L-BFGS refinement on MM result ===
    if best_seq is not None and time.monotonic() < deadline - 300.0:
        print(f"\n[Phase 3] L-BFGS refinement (t={time.monotonic() - t_start_global:.1f}s)")
        p3_deadline = min(deadline - 300.0, t_start_global + 1300.0)
        a3 = _lbfgs_optimize(best_seq, p3_deadline, min_time_remaining=300.0)
        update_best(a3, _score(a3), "lbfgs_refine")

    # === Phase 4: MM with very tight weights (polish) ===
    if best_seq is not None and time.monotonic() < deadline - 200.0:
        print(f"\n[Phase 4] MM polish tight (t={time.monotonic() - t_start_global:.1f}s)")
        p4_deadline = min(deadline - 200.0, t_start_global + 1500.0)
        a4, sc4 = _mm_optimize(best_seq, p4_deadline, min_time_remaining=200.0, verbose=True)
        update_best(a4, sc4, "mm_polish")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start_global:.1f}s)")
    n_candidates = [32768, 8192, 4096]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120.0:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        if best_seq is not None:
            a_init = _resample(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")
        # Quick L-BFGS
        a_opt = _lbfgs_optimize(a_init, deadline, min_time_remaining=100.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"lbfgs_n{n_try}")

        # Quick MM
        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 80.0:
            a_mm, sc_mm = _mm_optimize(a_opt, deadline, min_time_remaining=80.0, verbose=False)
            update_best(a_mm, sc_mm, f"mm_n{n_try}")

    # === Phase 6: Diverse restarts ===
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60.0:
        n_best = len(best_seq)
        restarts = [
            ("perturb_small", _perturb(best_seq, 0.03, rng)),
            ("perturb_med", _perturb(best_seq, 0.08, rng)),
            ("tapered_fresh", _make_profile(n_best, "tapered")),
            ("bimodal_fresh", _make_profile(n_best, "bimodal")),
        ]
        for label, p_start in restarts:
            if time.monotonic() > deadline - 25.0:
                break
            a_r = _lbfgs_optimize(p_start, deadline, min_time_remaining=25.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")
            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 15.0:
                a_r2, sc_r2 = _mm_optimize(a_r, deadline, min_time_remaining=15.0, verbose=False)
                update_best(a_r2, sc_r2, f"restart_{label}_mm")

    # === Phase 7: Final ultra-tight MM polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5.0:
        a_fin, sc_fin = _mm_optimize(best_seq, deadline - 3.0, min_time_remaining=3.0, verbose=False)
        update_best(a_fin, sc_fin, "final_mm")

        # One more L-BFGS with ultra-tight beta
        if time.monotonic() < deadline - 3.0:
            a_fin2 = _lbfgs_optimize(best_seq, deadline - 2.0, min_time_remaining=2.0)
            update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start_global:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
