"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp L-BFGS, peak-chasing via direct gradient,
coordinate descent, and multi-n/multi-start exploration.
Key innovations: (1) FFT-based gradient of weighted sum of convolution values,
(2) adaptive peak concentration (sharpening weights on top peaks),
(3) coordinate descent for final polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_softmax(z):
    """Stable log-softmax."""
    z = z - np.max(z)
    log_s = np.log(np.sum(np.exp(z)))
    return z - log_s


def _conv_grad_weighted(a, w):
    """Compute grad_a sum_k w[k] * conv(a,a)[k].
    w: shape (2n-1,)
    Returns: shape (n,)
    """
    n = len(a)
    a_flipped = a[::-1]
    corr = _full_convolve(w, a_flipped)
    h = corr[n - 1:n - 1 + n]
    return 2.0 * h


def _lse_objective(z, beta, n):
    """Log-sum-exp smoothed objective in log-parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad_z(z, beta, n):
    """Gradient of LSE objective w.r.t. z (log-parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        # Fallback: point mass on argmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    h = _conv_grad_weighted(a, w)
    da = 2.0 * n * h  # d(f)/d(a_j)
    
    # Chain rule: d(f)/d(z_k) = a_k * (sum_j da_j * a_j - da_k)
    G = float(np.dot(da, a))
    dz = a * (G - da)
    return dz


class _LSEProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    
    def __call__(self, z):
        return _lse_objective(z, self.beta, self.n)
    
    def grad(self, z):
        return _lse_grad_z(z, self.beta, self.n)


def _run_adaptive_lbfgs(a_init, deadline, min_remaining=30.0):
    """Run L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    # Initial beta estimate from peak distribution
    c = _full_convolve(a_init, a_init)
    c_sorted = np.sort(c)[::-1]
    c_max = c_sorted[0]
    # Use the gap between top few peaks
    k = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[k]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.005)
    
    # Adaptive beta schedule: start wide, narrow down
    n_phases = 8
    betas = base_beta * np.array([1.0] + [0.5**i for i in range(1, n_phases)])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        # Estimate iterations
        iter_time = 0.005 * (n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (iter_time * 15))))
        
        prob = _LSEProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.2e} failed: {ex}")
    
    return current_a, current_sc


def _peak_chase(a_init, deadline, min_remaining=30.0):
    """Directly chase and reduce the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Parameters
    lr = 0.01
    n_top = 1  # number of peaks to focus on
    patience = 200
    no_improve = 0
    
    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Select top peaks
        top_indices = np.argsort(c)[::-1][:n_top]
        
        # Compute gradient for the worst peak
        w = np.zeros_like(c)
        w[top_indices] = 1.0 / n_top
        h = _conv_grad_weighted(a, w)
        grad = 2.0 * n * h
        
        # Project gradient to maintain sum constraint (remove mean)
        grad_proj = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_proj)
        
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                current_sc = sc_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
                n_top = min(n_top * 2, max(5, n // 100))
            if no_improve >= patience:
                # Reset
                a = best_a.copy()
                lr = 0.01
                n_top = 1
                no_improve = 0
                patience = min(patience * 0.7, 20)
                if patience < 10:
                    break
        else:
            # Slowly increase focus
            if no_improve == 0 and n_top < max(3, n // 500):
                n_top += 1
    
    return best_a, _score(best_a)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) FFT-based adaptive log-sum-exp L-BFGS, (2) peak-chasing via
direct gradient on top peaks, (3) perturbation-based restarts to escape
local minima, (4) coordinate descent polish, (5) multi-n exploration.
Key innovations: continuous beta annealing, frequency-domain initialization,
and hybrid smooth/focused optimization alternation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_grad_weighted(a, w):
    """Compute grad_a sum_k w[k] * conv(a,a)[k]. w shape (2n-1,), returns shape (n,)."""
    n = len(a)
    a_flipped = a[::-1]
    corr = _full_convolve(w, a_flipped)
    h = corr[n - 1:n - 1 + n]
    return 2.0 * h


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad_z(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _conv_grad_weighted(a, w)
    da = 2.0 * n * h
    G = float(np.dot(da, a))
    dz = a * (G - da)
    return dz


class _LSEProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_objective(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad_z(z, self.beta, self.n)


def _run_adaptive_lbfgs(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    c = _full_convolve(a_init, a_init)
    c_sorted = np.sort(c)[::-1]
    c_max = float(c_sorted[0])
    k = min(100, len(c_sorted) - 1)
    gap = c_max - float(c_sorted[k])
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.01)

    # Continuous geometric beta schedule
    n_phases = 12
    betas = base_beta * (0.45 ** np.arange(n_phases))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    stall = 0

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        iter_time = 0.004 * (n / 4096.0)
        max_iter = min(8000, max(100, int(remaining / (iter_time * 18))))

        prob = _LSEProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall = 0
                print(f"    L-BFGS beta={beta:.3e}: score={sc_new:.10f}")
            else:
                stall += 1
                if stall >= 4:
                    break
        except Exception:
            stall += 1
            if stall >= 4:
                break

    return current_a, current_sc


def _peak_chase(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.05
    n_top = 1
    patience = 300
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        top_indices = np.argsort(c)[::-1][:n_top]
        w = np.zeros_like(c)
        w[top_indices] = 1.0 / n_top
        h = _conv_grad_weighted(a, w)
        grad = 2.0 * n * h
        grad_proj = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_proj))

        if gnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                current_sc = sc_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.75
                n_top = min(n_top * 2, max(3, n // 200))
            if no_improve >= patience:
                a = best_a.copy()
                lr = 0.05
                n_top = 1
                no_improve = 0
                stall_count += 1
                patience = max(50, int(patience * 0.7))
                if stall_count > 10:
                    break
        else:
            if no_improve == 0 and n_top < max(5, n // 500):
                n_top += 1
            lr = min(0.1, lr * 1.01)

    return best_a, _score(best_a)


def _coord_descent_topk(a_init, deadline, min_remaining=30.0, top_k_frac=0.1):
    """Coordinate descent on top-K largest elements only."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()

    max_stall = 3

    while time.monotonic() + min_remaining < deadline:
        # Get top-k indices by value
        top_idx = np.argsort(a)[::-1][:k]
        np.random.seed(int(time.monotonic() * 1000) % 10000)
        rng.shuffle(top_idx)

        any_improvement = False
        for j in top_idx:
            if time.monotonic() + min_remaining > deadline:
                break

            a_old = a[j]
            if a_old < 1e-10:
                continue

            # Try several candidate values
            candidates = np.array([a_old * 0.85, a_old * 0.6, a_old * 0.3,
                                   a_old * 1.15, a_old * 1.3, 0.0])

            c_cur = _full_convolve(a, a)
            cur_max = float(c_cur.max())
            best_val = a_old
            best_val_max = cur_max

            s_rest = a.sum() - a_old

            for cv in candidates:
                if cv < 0:
                    cv = 0.0
                a[j] = cv
                s_new = s_rest + cv
                if s_new < 1e-10:
                    a[j] = a_old
                    continue
                a_try = a.copy()
                a_try[j] = cv
                a_try = a_try / s_new
                c_try = _full_convolve(a_try, a_try)
                sc_try = float(c_try.max())
                if sc_try < best_val_max:
                    best_val_max = sc_try
                    best_val = cv

            if best_val != a_old:
                a[j] = best_val
                a = a / a.sum()
                any_improvement = True
            else:
                a[j] = a_old

        if any_improvement:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
            no_full_sweep_no_improve = 0
        else:

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "freq_flat":
        # Try to create a sequence with flat frequency response
        # Start with random phase, take real part of IDFT
        N = n
        phase = np.random.default_rng(777).standard_normal(N // 2 + 1)
        # Build Hermitian symmetric spectrum with unit magnitude
        spec = np.exp(1j * phase)
        # Ensure real output: spec[k] = conj(spec[N-k])
        spec_full = np.zeros(N, dtype=complex)
        spec_full[:N // 2 + 1] = spec
        if N % 2 == 0:
            spec_full[N // 2:] = np.conj(spec_full[1:N // 2][::-1])
        else:
            spec_full[(N // 2 + 1):] = np.conj(spec_full[1:N // 2 + 1][::-1])
        a = np.real(np.fft.ifft(spec_full))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage(a, deadline):
    """Try zeroing out small elements to see if score improves."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(a_try)

    return best_a, best_sc

                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
