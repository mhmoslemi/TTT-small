"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a))/(sum(a))^2.
Combines: (1) Gerchberg-Saxton phase retrieval, (2) L-BFGS with smooth max,
(3) Preconditioned CG active-set QP, (4) Flattening polish, (5) Multi-n search.
Key novel element: GS iterations exploit the Wiener-Khinchin duality between
flat autocorrelation and flat Fourier magnitude.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _pcg_solve(W, n, rhs, max_iter=2000, tol=1e-12, precond=True, restart=100):
    """Preconditioned CG with restarts for Hankel matrix solve."""
    x = np.zeros(n)
    r = rhs.copy()
    if precond:
        diag = W[0:2*n-1:2]
        diag = np.maximum(diag, 1e-30)
        z = r / diag
    else:
        z = r.copy()
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        if precond:
            z_new = r / diag
        else:
            z_new = r.copy()
        rz_new = float(r @ z_new)
        if not np.isfinite(rz_new) or abs(rz_new) < tol * tol * rz0:
            break
        if (i + 1) % restart == 0:
            p = z_new.copy()
        else:
            if abs(rz) < 1e-30:
                p = z_new.copy()
            else:
                p = z_new + (rz_new / rz) * p
        rz = rz_new
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.001)
    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.00025, 0.000125,
                           0.0000625, 0.00003, 0.000015, 0.0000075,
                           0.000004, 0.000002, 0.000001, 0.0000005,
                           0.00000025, 0.0000001, 0.00000005, 0.00000002])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005])
    betas = base_beta * ratios
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_iter_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 15))))
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _gerchberg_saxton(a_init, deadline, max_iters=5000, damping=1.0, soft_param=0.0):
    """
    Gerchberg-Saxton iterations exploiting Wiener-Khinchin:
    flat autocorrelation <-> constant |FFT| magnitude.
    Uses soft/hard projection with optional damping.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    target_mag = 1.0 / np.sqrt(2 * n - 1)
    N = _next_pow2(2 * n - 1)
    patience = 0
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mag[mag < 1e-30] = 1e-30
        phase = A / mag
        A_target = target_mag * phase
        a_new = np.fft.irfft(A_target, N)[:n]
        if soft_param > 0:
            a_neg = np.minimum(a_new, 0)
            a_new = a - soft_param * a_neg
            a_new = np.maximum(a_new, 0)
        else:
            a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_damped = (1 - damping) * a + damping * a_new
        a_damped = np.maximum(a_damped, 0)
        s_d = a_damped.sum()
        if s_d < 1e-15:
            a_damped = a_new
        else:
            a_damped = a_damped / s_d
        a = a_damped
        if i % 100 == 0 and i > 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    break
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _gs_with_errors(a_init, deadline, max_iters=3000):
    """
    Error-reduction variant of GS: uses gradient-like corrections
    instead of hard projection. Often converges to better minima.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    target_c = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    gamma_freq = 0.9
    gamma_time = 0.5
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N)
        C = np.abs(A) ** 2
        C_full = np.zeros(2 * n - 1)
        C_full[:len(C)] = C[:len(C)]
        if len(C) > n - 1:
            C_full[n - 1:] = C[n - 1:len(C)]
        err_freq = C_full - target_c
        err_freq_conj = np.zeros(N // 2 + 1, dtype=complex)
        err_freq_conj[:n] = err_freq[:min(n, len(err_freq))]
        mag = np.abs(A)
        mag[mag < 1e-30] = 1e-30
        A_new = A - gamma_freq * err_freq_conj * np.conj(A) / (2 * mag)
        a_new = np.fft.irfft(A_new, N)[:n]
        err_time = np.minimum(a_new, 0)
        a_new = a_new - gamma_time * err_time
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if i % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _qp_solve(W, n, a_init, max_outer=200, cg_max_iter=3000):
    """Active-set QP solver using preconditioned CG on the support."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n) / 2.0)
    for outer in range(max_outer):
        support = a > 1e-12
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            n_sup = n
        if n_sup == n or n_sup > 0.9 * n:
            y = _pcg_solve(W, n, np.ones(n), max_iter=cg_max_iter, tol=1e-14)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                break
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            grad = _hankel_matvec(a, W, n)
            lam = grad[support].mean()
            step = 0.5 / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.2)
            a_new = a - step * (grad - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        val_new = float(a_new @ _hankel_matvec(a_new, W, n) / 2.0)
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new
        grad = _hankel_matvec(a, W, n)
        support = a > 1e-12
        if support.any():
            lam = grad[support].mean()
            resid = np.max(np.abs(grad[support] - lam))
            off_s = ~support
            if off_s.any():
                resid = max(resid, np.max(lam - grad[off_s]))
        else:
            resid = np.inf
        if resid < 1e-12 * (np.max(np.abs(grad)) + 1e-15):
            break
    return best_a


def _active_set_polish(a_init, deadline, max_rounds=150, top_k_list=None):
    """
    Progressive active-set QP with top-k selection.
    Tries multiple k values and tightens over time.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    if top_k_list is None:
        top_k_list = [10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 4
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active
        a_qp = _qp_solve(W, n, a, max_outer=80, cg_max_iter=2000)
        sc_qp = _score(a_qp)
        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-15:
                Hg = _pcg_solve(W, n, g, max_iter=1500, tol=1e-13)
                H1 = _pcg_solve(W, n, np.ones(n), max_iter=1500, tol=1e-13)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20 and np.isfinite(H1_norm):
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t / (gn + 1e-30)
                alpha = 1.0
                for ls in range(50):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    c_max_try = float(c_try.max())
                    if c_max_try < c_max:
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()
                        break
                    alpha *= 0.5
                    if alpha < 1e-15:
                        break
        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=300):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(c.sum() / len(c))
        excess = np.maximum(c - c_mean, 0)
        excess_max = excess.max()
        if excess_max < 1e-16:
            break
        excess = excess / excess_max
        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]
        step = 0.05 / (1.0 + 0.005 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a + 0.3 * (a_new - a)
            a = np.maximum(a, 0)
            a = a / a.sum()
    return best_a, best_sc


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.02, 0.05, 0.10, 0.15, 0.25]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "zadoff_inspired":
        t = np.linspace(0, 1, n)
        # Inspired by Zadoff-Chu: varying density
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t * 3)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "sqrt_taper":
        t = np.linspace(0, 1, n)
        a = np.sqrt(np.sin(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _gs_multi_strategy(a_init, deadline, min_time_remaining=60):
    """Run multiple GS variants and return the best."""
    best_a = a_init.copy()
    best_sc = _score(a_init)
    strategies = [
        ("gs_hard", lambda: _gerchberg_saxton(a_init, min(deadline - min_time_remaining, time.monotonic() + 120),
                                               max_iters=5000, damping=1.0, soft_param=0.0)),
        ("gs_soft", lambda: _gerchberg_saxton(a_init, min(deadline - min_time_remaining, time.monotonic() + 120),
                                              max_iters=5000, damping=0.8, soft_param=0.1)),
        ("gs_damped", lambda: _gerchberg_saxton(a_init, min(deadline - min_time_remaining, time.monotonic() + 120),
                                                max_iters=3000, damping=0.5, soft_param=0.0)),
        ("gs_err", lambda: _gs_with_errors(a_init, min(deadline - min_time_remaining, time.monotonic() + 120),
                                           max_iters=3000)),
    ]
    for name, fn in strategies:
        if time.monotonic() + min_time_remaining > deadline:
            break
        try:
            a_try, sc_try = fn()
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                print(f"    GS [{name}]: {sc_try:.10f}")
        except Exception as e:
            print(f"    GS [{name}] failed: {e}")
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining phase retrieval, L-BFGS,
    active-set QP, and flattening for minimizing autocorrelation peak.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS from parent ===
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs_1")

    # === Phase 2: Gerchberg-Saxton (novel) ===
    print(f"\n[Phase 2] GS phase retrieval (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_gs, sc_gs = _gs_multi_strategy(best_seq, deadline, min_time_remaining=400)
        update_best(a_gs, sc_gs, "gs_1")

    # === Phase 3: Active-set QP ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_qp, sc_qp = _active_set_polish(best_seq, min(deadline - 400, time.monotonic() + 250),
                                         max_rounds=100)
        update_best(a_qp, sc_qp, "qp_1")

    # === Phase 4: Flatten + re-LBFGS ===
    print(f"\n[Phase 4] Flatten+LBFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_fl, sc_fl = _flatten_polish(best_seq, min(deadline - 300, time.monotonic() + 80))
        update_best(a_fl, sc_fl, "flatten_1")
        if time.monotonic() < deadline - 250:
            a_re = _run_lbfgs(a_fl, deadline, min_time_remaining=200, aggressive=True)
            update_best(a_re, _score(a_re), "lbfgs_2")

    # === Phase 5: GS from current best (second pass) ===
    print(f"\n[Phase 5] GS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_gs2, sc_gs2 = _gs_multi_strategy(best_seq, deadline, min_time_remaining=250)
        update_best(a_gs2, sc_gs2, "gs_2")

    # === Phase 6: Active-set QP round 2 (tighter) ===
    print(f"\n[Phase 6] QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_qp2, sc_qp2 = _active_set_polish(best_seq, min(deadline - 200, time.monotonic() + 180),
                                           max_rounds=120)
        update_best(a_qp2, sc_qp2, "qp_2")

    # === Phase 7: Shrinkage ===
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sh, sc_sh = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=150)
        update_best(a_sh, sc_sh, "shrink")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_init_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt_n = _run_lbfgs(a_init_n, deadline, min_time_remaining=80, aggressive=True)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")
        if time.monotonic() < deadline - 60:
            a_gs_n, sc_gs_n = _gerchberg_saxton(a_opt_n, min(deadline - 50, time.monotonic() + 80),
                                                 max_iters=3000)
            update_best(a_gs_n, sc_gs_n, f"n_{n_try}_gs")

    # === Phase 9: Diverse restarts ===
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_r = len(best_seq)
        restart_opts = [
            ("perturb", lambda: _perturb(best_seq, 0.03, rng)),
            ("zadoff", lambda: _make_profile(n_r, "zadoff_inspired")),
            ("sqrt_taper", lambda: _make_profile(n_r, "sqrt_taper")),
            ("centered", lambda: _make_profile(n_r, "centered")),
        ]
        for name, gen_fn in restart_opts[:3]:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20)
            update_best(a_r, _score(a_r), f"restart_{name}")
            if time.monotonic() < deadline - 20:
                a_gs_r, sc_gs_r = _gerchberg_saxton(a_r, min(deadline - 15, time.monotonic() + 30),
                                                     max_iters=2000)
                update_best(a_gs_r, sc_gs_r, f"restart_{name}_gs")

    # === Phase 10: Final ultra-tight polish ===
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(
            best_seq, deadline - 3, max_rounds=80,
            top_k_list=[50, 100, 200, 500, 1000])
        update_best(a_fin, sc_fin, "final_qp")
        if time.monotonic() < deadline - 3:
            a_fl_f, sc_fl_f = _flatten_polish(best_seq, deadline - 2, max_iters=200)
            update_best(a_fl_f, sc_fl_f, "final_flatten")
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 2e-8
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(3000, max(100, int(remaining / 0.008)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_final_lb = _softmax(result.x)
                    update_best(a_final_lb, _score(a_final_lb), "final_lbfgs")
                except Exception:
                    pass

    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"total_time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
