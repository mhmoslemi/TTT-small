Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using multi-scale warm-starting,
improved Newton active-set method, chipping heuristic, and diverse perturbation ensemble.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):

        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp

        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:

    return x

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: try progressively tighter tolerances
        active = None
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 3e-4, 1e-3, 3e-3]:
            tol = tol_frac * c_max
            candidate = c >= (c_max - tol)
            n_act = int(candidate.sum())
            if n_act >= 1 and n_act <= max(1, n // 4):
                active = candidate
                break
        if active is None or active.sum() == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True

        n_active = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with multiple strategies
        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_done:
            # Try sub-gradient with varying magnitudes
            for trial in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _chipping_heuristic(a_init, deadline, max_iters=5000, n_hot=5):
    """Iteratively reduce elements that contribute most to convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr_base = 0.01
    patience = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find the top-n_hot positions contributing to the peak
        sorted_idx = np.argsort(c)[::-1]
        hot_positions = sorted_idx[:n_hot]

        # Compute blame: for each hot position k, blame on a[i] is a[k-i]
        blame = np.zeros(n)
        for k in hot_positions:
            # c[k] = sum_i a[i]*a[k-i], blame on a[i] is a[k-i]
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    blame[i] += a[i] * a[j]

        # Normalize blame
        blame_sum = blame.sum()
        if blame_sum < 1e-30:
            break
        blame = blame / blame_sum

        # Step size with decay
        lr = lr_base / (1.0 + 0.001 * it)

        # Reduce a proportionally to blame
        a_try = a * (1.0 - lr * blame * n / max(blame.sum() * n, 1.0))
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            lr *= 0.5
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-15:
            a = a_try
            best_sc = sc_try
            best_a = a_try.copy()
            patience = 0
        else:
            patience += 1
            lr_base *= 0.95
            if patience > 100:
                break

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 5e-10
    prev_sc = current_a  # placeholder
    prev_sc = current_sc
    stagnation = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(10000, max(20, int(remaining / (est_time_per_iter * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-19, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-17:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnation = 0
            else:
                stagnation += 1
        except Exception:
            stagnation += 1

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        if new_gap < 1e-9 * c_max_new and stagnation == 0:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.001)

        # If stuck, try a perturbation
        if stagnation > 3:
            z = z + 0.01 * np.random.standard_normal(n)
            z = z - z.mean()
            stagnation = 0

    return current_a


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.9
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.03
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.02
        return a / a.sum()
    elif kind == "step_like":
        a = np.ones(n)
        third = n // 3
        two_third = 2 * n // 3
        a[:third] = 0.5
        a[third:two_third] = 1.0
        a[two_third:] = 0.5
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hamming":
        t = np.linspace(0, 1, n)
        a = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "poly_taper":
        t = np.linspace(0, 1, n)
        a = t * (1 - t) * (4 * t * (1 - t) + 0.3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # =========================================================
    # Phase 1: Multi-scale warm-start ladder (n: 1024 -> 16384)
    # =========================================================
    print(f"\n[Phase 1] Multi-scale warm-start (t={time.monotonic() - t_start:.1f}s)")
    ladder_ns = [1024, 2048, 4096, 8192, 12288, 16384, 20480]
    ladder_a = None

    for n_try in ladder_ns:
        if time.monotonic() > deadline - 600:
            break
        if ladder_a is None:
            a_init = _make_profile(n_try, "tapered")
        else:
            a_init = _resample(ladder_a, n_try)

        # Quick optimization
        t_alloc = min(60.0, (deadline - 600 - time.monotonic()) / max(1, len([x for x in ladder_ns if x >= n_try])) * 2.0)
        a_pat = _run_adaptive_lbfgs(a_init, min(time.monotonic() + t_alloc, deadline - 600),
                                     min_time_remaining=30.0)
        sc_pat = _score(a_pat)
        print(f"  n={n_try}: score={sc_pat:.10f} (t={time.monotonic() - t_start:.1f}s)")
        update_best(a_pat, sc_pat, f"ladder_n{n_try}")
        ladder_a = a_pat

        # Quick Newton polish
        if time.monotonic() < deadline - 580:
            a_np, sc_np = _newton_active_polish(a_pat, min(time.monotonic() + 20, deadline - 580), max_rounds=30)
            if sc_np < sc_pat:
                update_best(a_np, sc_np, f"ladder_np_n{n_try}")
                ladder_a = a_np

    # =========================================================
    # Phase 2: Diverse profile exploration at best n
    # =========================================================
    print(f"\n[Phase 2] Profile diversity (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(ladder_a) if ladder_a is not None else 8192
    profiles = ["tapered", "cos", "asym_taper", "raised_cos", "hamming", "blackman",
                "poly_taper", "bimodal", "left_heavy", "exp_decay"]

    profile_candidates = []
    for prof in profiles:
        if time.monotonic() > deadline - 500:
            break
        a_p = _make_profile(n_best, prof)
        # Quick L-BFGS (short)
        a_opt = _run_adaptive_lbfgs(a_p, min(time.monotonic() + 25, deadline - 500),
                                     min_time_remaining=200.0)
        sc_opt = _score(a_opt)
        print(f"  Profile '{prof}': score={sc_opt:.10f}")
        profile_candidates.append((sc_opt, a_opt, prof))
        update_best(a_opt, sc_opt, f"prof_{prof}")

    # Sort and keep top 4
    profile_candidates.sort(key=lambda x: x[0])
    top_profiles = profile_candidates[:4]

    # =========================================================
    # Phase 3: Chipping heuristic from best candidates
    # =========================================================
    print(f"\n[Phase 3] Chipping heuristic (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        # Chipping from the best
        a_chip, sc_chip = _chipping_heuristic(
            best_seq,
            min(time.monotonic() + 80, deadline - 400),
            max_iters=3000, n_hot=3
        )
        update_best(a_chip, sc_chip, "chipping_1")

        # Chipping from a perturbed version
        a_pert = _perturb(best_seq, 0.1, rng)
        a_chip2, sc_chip2 = _chipping_heuristic(
            a_pert,
            min(time.monotonic() + 60, deadline - 400),
            max_iters=2000, n_hot=5
        )
        update_best(a_chip2, sc_chip2, "chipping_2")

    # =========================================================
    # Phase 4: Focused L-BFGS + Newton cycling on top candidates
    # =========================================================
    print(f"\n[Phase 4] Focused optimization (t={time.monotonic() - t_start:.1f}s)")
    candidates = []

    # From ladder
    if ladder_a is not None:
        candidates.append(("ladder", ladder_a.copy()))
    # From profiles
    for sc_c, a_c, name in top_profiles[:2]:
        candidates.append((f"prof_{name}", a_c.copy()))
    # From parent
    if parent is not None:
        candidates.append(("parent", _resample(parent, n_best)))

    # Remove duplicates (by score proximity)
    seen_scores = []
    unique_candidates = []
    for label, a_c in candidates:
        sc_c = _score(a_c)
        if not any(abs(sc_c - ss) < 0.001 for ss in seen_scores):
            unique_candidates.append((label, a_c, sc_c))
            seen_scores.append(sc_c)

    for label, a_c, sc_c in unique_candidates[:4]:
        if time.monotonic() > deadline - 200:
            break
        # Long L-BFGS
        t_alloc = min(120.0, (deadline - 200 - time.monotonic()) / max(1, len(unique_candidates) - 1))
        a_opt = _run_adaptive_lbfgs(a_c, min(time.monotonic() + t_alloc, deadline - 200),
                                     min_time_remaining=100.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"focus_{label}")

        # Newton polish
        if time.monotonic() < deadline - 100:
            a_np, sc_np = _newton_active_polish(
                a_opt, min(time.monotonic() + 40, deadline - 100), max_rounds=80
            )
            update_best(a_np, sc_np, f"focus_np_{label}")

    # =========================================================
    # Phase 5: Perturbation ensemble + re-optimization
    # =========================================================
    print(f"\n[Phase 5] Perturbation ensemble (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_perturb_trials = 6
        for i in range(n_perturb_trials):
            if time.monotonic() > deadline - 100:
                break
            sigma = 0.02 + 0.03 * i
            a_pert = _perturb(best_seq, sigma, rng)
            # Short L-BFGS
            t_alloc = min(30.0, (deadline - 100 - time.monotonic()) / max(1, n_perturb_trials - i))
            a_opt = _run_adaptive_lbfgs(a_pert, min(time.monotonic() + t_alloc, deadline - 100),
                                         min_time_remaining=50.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                update_best(a_opt, sc_opt, f"perturb_{i}")
                # Newton if improved
                if time.monotonic() < deadline - 80:
                    a_np, sc_np = _newton_active_polish(
                        a_opt, min(time.monotonic() + 20, deadline - 80), max_rounds=40
                    )
                    update_best(a_np, sc_np, f"perturb_np_{i}")

    # =========================================================
    # Phase 6: Multi-n final exploration
    # =========================================================
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n_best_cur = len(best_seq)
        n_candidates = []
        # Try nearby sizes
        for factor in [0.75, 1.25, 1.5, 2.0]:
            n_try = int(n_best_cur * factor)
            if 2048 <= n_try <= 40000:
                n_candidates.append(n_try)
        # Also try specific sizes
        n_candidates.extend([12288, 16384, 20480, 24576, 30000])
        n_candidates = list(set(n_candidates))
        n_candidates.sort()

        for n_try in n_candidates[:5]:
            if time.monotonic() > deadline - 80:
                break
            a_init = _resample(best_seq, n_try)
            t_alloc = min(40.0, (deadline - 80 - time.monotonic()) / 2.0)
            a_opt = _run_adaptive_lbfgs(a_init, min(time.monotonic() + t_alloc, deadline - 80),
                                         min_time_remaining=40.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"multin_{n_try}")

            if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 40:
                a_np, sc_np = _newton_active_polish(
                    a_opt, min(time.monotonic() + 25, deadline - 40), max_rounds=50
                )
                update_best(a_np, sc_np, f"multin_np_{n_try}")

    # =========================================================
    # Phase 7: Final ultra-fine polish
    # =========================================================
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Newton
        a_fin, sc_fin = _newton_active_polish(best_seq, min(time.monotonic() + 60, deadline - 5),
                                               max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # L-BFGS at tiny beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-11
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(50, int(remaining / 0.005)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-17,
                                 'maxcor': 120, 'maxls': 500}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Final Newton
        if best_seq is not None and time.monotonic() < deadline - 1:
            a_veryfin, sc_veryfin = _newton_active_polish(best_seq, deadline - 0.5, max_rounds=80)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
