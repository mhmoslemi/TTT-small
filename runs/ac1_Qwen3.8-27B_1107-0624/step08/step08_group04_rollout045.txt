and diverse profile initialization. Uses FFT-accelerated convolution and

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _eff_grad(a, W):
    """Gradient of sum_k W[k]*c[k] w.r.t. a, where c = conv(a,a).
    Returns array of length n."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 - 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "centered_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t + 0.1
        return a / a.sum()
    elif kind == "squared_ramp":
        t = np.linspace(0, 1, n)
        a = t ** 2 + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _adam_logspace(a_init, deadline, min_time=5.0, max_steps=200000,
                   beta_schedule=None, lr0=0.5, beta1=0.9, beta2=0.999):
    """Adam optimization in log-space with smoothed max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    m = np.zeros(n)
    v = np.zeros(n)

    if beta_schedule is None:
        c0 = _full_convolve(a_init, a_init)
        c_max0 = float(c0.max())
        gap = c_max0 - np.sort(c0)[::-1][min(20, len(c0) - 1)]
        base_beta = max(gap * 0.5, c_max0 * 0.005)
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
                  0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                  0.000001, 0.0000005, 0.0000002, 0.000001]
        beta_schedule = [base_beta * r for r in ratios]

    best_a = a_init.copy()
    best_sc = _score(a_init)
    t0 = time.monotonic()

    for beta in beta_schedule:
        if time.monotonic() + min_time > deadline:
            break
        step = 1

        while step <= max_steps:
            if time.monotonic() + min_time > deadline:
                break

            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            W = float(np.dot(w, c))
            grad = _eff_grad(a, w)
            grad = grad - np.mean(grad)  # center for sum-invariance

            step += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)

            lr = lr0 / np.sqrt(step + 1)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            z = z - z.mean()

                    best_a = a_check.copy()

    a_final = _softmax(z)

    if sc_final < best_sc:

        best_a = a_final.copy()
    return best_a, best_sc


def _projected_gradient_topk(a_init, deadline, max_steps=10000,
                             k_frac=0.02, lr0=0.3):
    """Projected gradient descent on the top-k autocorrelation sum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    k = max(3, int(n * k_frac))

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        # Get top-k indices
        topk_idx = np.argpartition(c, -k)[-k:]
        W = np.zeros_like(c)
        W[topk_idx] = 1.0 / k

        grad = _eff_grad(a, W)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        lr = lr0 / (1.0 + 0.001 * step)
        # Backtracking line search
        alpha = lr
        a_old_cmax = float(c.max())
        for _ in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < a_old_cmax:
                a = a_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break
        else:
            a = a - alpha * grad
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s

    return best_a, best_sc

def _high_freq_smoothing(a, sigma_param=1.0):
    """Apply a gentle smoothing to reduce high-frequency oscillations."""
    n = len(a)
    kernel_size = max(3, int(n * 0.001))
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (sigma_param * n * 0.001) ** 2)
    kernel = kernel / kernel.sum()
    # Use FFT for efficient convolution
    N = _next_pow2(n + kernel_size - 1)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    k_pad = np.zeros(N)
    mid = kernel_size // 2
    k_pad[-mid:-1] = kernel[mid + 1:] if mid + 1 < kernel_size else kernel[mid + 1:]
    k_pad[N - mid - 1:N - mid - 1 + (kernel_size - mid - 1)] = kernel[:mid]
    k_pad[N - mid - 1] += kernel[mid]
    result = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(k_pad), N)[:n]
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        return a
    return result / s

    if time.monotonic() < deadline - 600:
        a_main, sc_main = _iterative_refine(
            parent, min(deadline - 600, time.monotonic() + 600), min_time=120)
        update_best(a_main, sc_main, "main_refine")

    # Phase 2: More aggressive Adam from current best
    print(f"\n[Phase 2] Aggressive Adam (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_aggr, sc_aggr = _adam_logspace(
            best_seq, min(deadline - 400, time.monotonic() + 400),
            min_time=80, max_steps=300000, lr0=1.5)
        update_best(a_aggr, sc_aggr, "aggr_adam")

        # Follow up with top-k
        if time.monotonic() < deadline - 300:
            a_tk, sc_tk = _projected_gradient_topk(
                a_aggr, min(deadline - 300, time.monotonic() + 200),
                k_frac=0.005, lr0=0.8)
            update_best(a_tk, sc_tk, "aggr_topk")

    # Phase 3: Active-set refinement on best
    print(f"\n[Phase 3] Active-set refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_as, sc_as = _active_set_qp_refine(
            best_seq, min(deadline - 250, time.monotonic() + 250),
            max_rounds=300)
        update_best(a_as, sc_as, "active_set")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_options = [8192, 16384, 32768]
    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue
        a_init_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")

        if time.monotonic() < deadline - 100:
            a_opt, sc_opt = _adam_logspace(
                a_init_n, min(deadline - 100, time.monotonic() + 150),
                min_time=30, max_steps=50000, lr0=1.0)
            update_best(a_opt, sc_opt, f"n_{n_try}_adam")

        if time.monotonic() < deadline - 60 and sc_opt < best_score + 0.01:
            a_ref, sc_ref = _active_set_qp_refine(
                a_opt, min(deadline - 60, time.monotonic() + 60), max_rounds=80)
            update_best(a_ref, sc_ref, f"n_{n_try}_refine")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_r = len(best_seq)
        profiles = ["tapered", "cos", "raised_cos", "bimodal", "centered_gauss", "ramp"]
        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 40:
                break
            if i < 2:
                p_start = _perturb(best_seq, 0.03 + 0.02 * i, rng)
            else:
                p_start = _make_profile(n_r, prof)

            a_r, sc_r = _adam_logspace(
                p_start, min(deadline - 40, time.monotonic() + 80),
                min_time=20, max_steps=30000, lr0=0.8)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 20:
                a_rp, sc_rp = _active_set_qp_refine(
                    a_r, min(deadline - 20, time.monotonic() + 30), max_rounds=50)
                update_best(a_rp, sc_rp, f"restart_{i}_refine")

    # Phase 6: Final ultra-tight polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active set
        a_ft, sc_ft = _active_set_qp_refine(
            best_seq, deadline - 3,
            tol_schedule=[0.00001, 0.000005, 0.000002, 0.000001,
                          0.0000005, 0.0000002, 0.0000001, 0.00000005,
                          0.00000002, 0.00000001])
        update_best(a_ft, sc_ft, "final_tight")

        # Top-k with very small k
        if time.monotonic() < deadline - 5:
            a_fk, sc_fk = _projected_gradient_topk(
                best_seq, deadline - 3, k_frac=0.001, lr0=0.3, max_steps=5000)
            update_best(a_fk, sc_fk, "final_topk")

        # One more Adam with very small beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            tiny_beta = mb * 1e-6
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            step = 1000  # start with some momentum warmup
            remaining = deadline - 3 - time.monotonic()
            max_steps = min(50000, int(remaining / 0.01))
            for s in range(max_steps):
                if time.monotonic() > deadline - 3:
                    break
                a_s = _softmax(z)
                c_s = _full_convolve(a_s, a_s)
                c_max_s = float(c_s.max())
                shifted = (c_s - c_max_s) / tiny_beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_s)
                    w[int(np.argmax(c_s))] = 1.0
                else:
                    w = e / s_e
                grad = _eff_grad(a_s, w)
                grad = grad - np.mean(grad)
                step += 1
                m = 0.9 * m + 0.1 * grad
                v = 0.999 * v + 0.001 * grad ** 2
                m_hat = m / (1 - 0.9 ** step)
                v_hat = v / (1 - 0.999 ** step)
                lr = 0.5 / np.sqrt(step + 1)
                z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
                z = z - z.mean()
                if s % 50 == 0:
                    a_check = _softmax(z)
                    sc_check = _score(a_check)
                    if sc_check < best_score:
                        best_score = sc_check
                        best_seq = a_check.copy()

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
