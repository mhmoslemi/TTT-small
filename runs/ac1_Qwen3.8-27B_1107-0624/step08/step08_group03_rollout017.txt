import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of mean active correlation
        g = (2.0 / m) * _corr_vec(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        # Regularization
        mu = 1e-7 * (c_max / max(m, 1)) + 1e-9

        def hess_matvec(v):
            return (2.0 / m) * _corr_vec(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=400):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            rs_init = rs
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * np.sqrt(rs_init):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search with non-negativity constraint
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-12:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback: steepest descent
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14 * c_max:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap_idx = min(30, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[gap_idx]
    base_beta = max(gap * 0.4, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                                      0.015625, 0.0078, 0.0039, 0.00195,
                                      0.00098, 0.00049, 0.000245, 0.00012,
                                      0.00006, 0.00003, 0.000015, 0.000007,
                                      0.000003, 0.000001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        if n <= 4096:
            est_time = 0.003
        elif n <= 8192:
            est_time = 0.006
        elif n <= 16384:
            est_time = 0.012
        else:
            est_time = 0.025 * (n / 32768.0)

        max_iter_val = min(8000, max(50, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, float(beta), int(n)),
                z,
                jac=lambda z_local: _lsgrad(z_local, float(beta), int(n)),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter_val,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_peel(a_init, deadline, max_rounds=200):
    """Iteratively target the top peaks and reduce them."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive number of peaks to target
        n_peaks = min(20 + rnd, 50)
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[top_peaks]
        # Only target peaks that are close to the max
        threshold = c_max * (1.0 - 0.005 * (1 + rnd * 0.1))
        active_peaks = top_peaks[c[top_peaks] >= threshold]
        if len(active_peaks) == 0:
            active_peaks = top_peaks[:5]

        w = np.zeros(2 * n - 1)
        for k in active_peaks:
            w[k] = 1.0 / len(active_peaks)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        grad = 2.0 * a * (h - W)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        f_old = c_max
        step = 0.5 / (1.0 + 0.05 * rnd)
        for ls in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14 * c_max:
            break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 0.5 + 1.5 * t ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, active-set Newton,
    peak peeling, and subgradient polishing with multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"prev={prev:.10f} dlta={prev-best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform", rng)
        print(f"No parent, uniform n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS on parent ===
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        lbfgs_deadline = deadline - 700
        a_lbfgs = _run_lbfgs(parent, lbfgs_deadline, min_time_remaining=200, aggressive=True)
        sc_lbfgs = _score(a_lbfgs)
        print(f"  L-BFGS result: {sc_lbfgs:.10f}")
        update_best(a_lbfgs, sc_lbfgs, "lbfgs")

    # === Phase 2: Active-set Newton (tight) ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_deadline = min(deadline - 400, time.monotonic() + 500)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=400,
                                         tol_active_start=0.003, tol_active_end=0.00005)
        update_best(a_as, sc_as, "active_set")

    # === Phase 3: Peak peeling ===
    print(f"\n[Phase 3] Peak peeling (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        pp_deadline = min(deadline - 250, time.monotonic() + 200)
        a_pp, sc_pp = _peak_peel(best_seq, pp_deadline, max_rounds=150)
        update_best(a_pp, sc_pp, "peak_peel")

    # === Phase 4: Another L-BFGS pass ===
    print(f"\n[Phase 4] L-BFGS second pass (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        lbfgs2_deadline = min(deadline - 200, time.monotonic() + 250)
        a_l2 = _run_lbfgs(best_seq, lbfgs2_deadline, min_time_remaining=100, aggressive=True)
        update_best(a_l2, _score(a_l2), "lbfgs2")

    # === Phase 5: Active-set Newton (very tight) ===
    print(f"\n[Phase 5] Active-set very tight (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        as2_deadline = min(deadline - 100, time.monotonic() + 150)
        a_as2, sc_as2 = _active_set_newton(best_seq, as2_deadline, max_iter=500,
                                           tol_active_start=0.001, tol_active_end=0.00001)
        update_best(a_as2, sc_as2, "as_tight")

    # === Phase 6: Try different n values ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    for n_try in [8192, 32768]:
        if time.monotonic() > deadline - 120:
            break
        if best_seq is not None:
            a_interp = _interpolate_to_n(best_seq, n_try)
        else:
            a_interp = _make_profile(n_try, "uniform", rng)
        sc_interp = _score(a_interp)
        print(f"  n={n_try} interpolated: {sc_interp:.10f}")

        if sc_interp < best_score + 0.05:
            rem = deadline - 60
            a_opt = _run_lbfgs(a_interp, min(deadline - 40, time.monotonic() + 120),
                               min_time_remaining=20, aggressive=True)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n{n_try}_lbfgs")

            if time.monotonic() < deadline - 30 and sc_opt < best_score + 0.01:
                a_as3, sc_as3 = _active_set_newton(a_opt, min(deadline - 10, time.monotonic() + 80),
                                                   max_iter=200, tol_active_start=0.002,
                                                   tol_active_end=0.0001)
                update_best(a_as3, sc_as3, f"n{n_try}_as")

    # === Phase 7: Subgradient polish ===
    print(f"\n[Phase 7] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 5, time.monotonic() + 60),
                                      n_max_steps=3000, lr_init=0.015)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 8: Final active-set ===
    print(f"\n[Phase 8] Final active-set (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1, max_iter=200,
                                           tol_active_start=0.0005, tol_active_end=0.000005)
        update_best(a_fin, sc_fin, "final_as")

    # === Final output ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
