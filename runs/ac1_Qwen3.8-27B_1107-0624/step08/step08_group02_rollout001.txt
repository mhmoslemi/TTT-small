        beta_cg = rz_new / rz
        p = z_new + beta_cg * p
        rz = rz_new
        if rz < tol * tol * max(rz0, 1e-30):
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _newton_polish(a_init, deadline, max_rounds=200):
    """Newton's method with active set and preconditioned CG."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Try multiple active set tolerances
        best_step_sc = c_max
        best_step_a = None
        for tol_frac in [1e-8, 5e-7, 1e-6, 5e-6, 1e-5, 5e-5, 2e-4, 1e-3]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active < 2:
                active[int(np.argmax(c))] = True
                # Also include a few neighbors
                idx = int(np.argmax(c))
                for d in [1, 2, 3]:
                    if idx + d < len(c):
                        active[idx + d] = True
                    if idx - d >= 0:
                        active[idx - d] = True
                n_active = int(active.sum())
            if n_active == 0:
                continue

            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / n_active

            g = _hankel_matvec(a, W, n)
            g_mean = np.mean(g)
            g_t = g - g_mean
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-14 * n:
                if best_step_a is None:
                    best_step_a = a.copy()
                    best_step_sc = c_max
                break

            # Solve H*dx = -g_t with constraint sum(dx) = 0
            Hg = _cg_solve_precond(W, n, g_t, max_iter=500)
            H1 = _cg_solve_precond(W, n, np.ones(n), max_iter=500)
            H1_norm = float(np.linalg.norm(H1))
            if H1_norm < 1e-20 or not np.isfinite(H1_norm):
                dx = -g_t / (gn + 1e-30)
            else:
                lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
                dx = -Hg + lam * H1
                dx = dx - np.mean(dx)
                dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    dx = -g_t / (gn + 1e-30)

            # Try multiple step sizes (non-monotone)
            for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < best_step_sc:
                    best_step_sc = c_max_try
                    best_step_a = a_try.copy()

            if best_step_a is not None and best_step_sc < c_max * (1 - 1e-10):
                break

        if best_step_a is not None and best_step_sc < c_max - 1e-15:
            a = best_step_a.copy()
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Subgradient fallback
            c = _full_convolve(a, a)
            c_max = float(c.max())
            tol = 0.001 * c_max
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / active.sum()
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-14:
                for trial in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                    a_try = a - trial * g_t / gn
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max - 1e-15:
                        a = a_try
                        sc = _score(a)
                        if sc < best_sc - 1e-14:
                            best_sc = sc
                            best_a = a.copy()
                            no_improve = 0
                        break

        if no_improve > 30:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Adaptive L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 5e-10
    prev_sc = current_sc
    stall_count = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.004 * scale
        max_iter = min(10000, max(50, int(remaining / est_time_per_iter * 0.8)))

        try:
            def obj_fn(zv):
                return _smoobj(zv, beta, n)

            def grad_fn(zv):
                return _smograd(zv, beta, n)

            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
        except Exception:
            stall_count += 1

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        idx2 = min(30, len(c_sorted_new) - 1)
        new_gap = c_sorted_new[0] - c_sorted_new[idx2]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-14:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        elif sc_improved < 1e-16:
            beta *= 0.4
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.0003)
        if stall_count > 5:
            break

    return current_a


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.0 + 0.01
        return a / a.sum()
    elif kind == "step_like":
        a = np.ones(n)
        half = n // 2
        a[:half] = 1.0
        a[half:] = 0.3
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "wider_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.05)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.015

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _cycle_optimize(a_init, deadline, min_time_remaining=60.0):
    """Alternate L-BFGS bursts and Newton polishing."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    cycle_time = min(45.0, (deadline - min_time_remaining - time.monotonic()) / 5.0)
    if cycle_time < 5:
        return current_a

    for cycle in range(8):
        if time.monotonic() + min_time_remaining > deadline:
            break
        t_cycle_start = time.monotonic()
        t_cycle_end = t_cycle_start + cycle_time

        if time.monotonic() < t_cycle_end - 3:
            a_lb = _run_lbfgs(current_a, min(t_cycle_end - 3, deadline - min_time_remaining),
                              min_time_remaining=15.0)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb

        if time.monotonic() + 8 < min(t_cycle_end, deadline - min_time_remaining):
            a_np, sc_np = _newton_polish(
                current_a,
                min(t_cycle_end, deadline - min_time_remaining),
                max_rounds=50
            )
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np

        cycle_time = min(cycle_time * 1.15, 100.0)

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick L-BFGS polish from parent
    print(f"\n[Phase 1] Quick L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_quick = _run_lbfgs(parent, min(deadline - 550, time.monotonic() + 100),
                             min_time_remaining=500.0)
        update_best(a_quick, None, "quick_lbfgs")

    # Phase 2: Newton polish on current best
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_np, sc_np = _newton_polish(best_seq, min(deadline - 450, time.monotonic() + 150),
                                     max_rounds=100)
        update_best(a_np, sc_np, "newton_1")

    # Phase 3: Multi-scale approach
    print(f"\n[Phase 3] Multi-scale (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        # Downsample to smaller n, optimize, upsample
        for n_small in [2048, 4096]:
            if time.monotonic() > deadline - 350:
                break
            a_small = _resample(best_seq, n_small)
            a_small_opt = _run_lbfgs(a_small, min(deadline - 300, time.monotonic() + 80),
                                     min_time_remaining=200.0)
            update_best(a_small_opt, None, f"small_n{n_small}")
            if a_small_opt is not None and _score(a_small_opt) < best_score + 0.01:
                a_up = _resample(a_small_opt, len(best_seq))
                a_up_opt = _run_lbfgs(a_up, min(deadline - 250, time.monotonic() + 60),
                                      min_time_remaining=150.0)
                update_best(a_up_opt, None, f"upsample_{n_small}")
                if _score(a_up_opt if a_up_opt is not None else a_up) < best_score + 0.005:
                    a_np2, sc_np2 = _newton_polish(a_up_opt, min(deadline - 200, time.monotonic() + 40),
                                                   max_rounds=50)
                    update_best(a_np2, sc_np2, f"upsample_{n_small}_np")

    # Phase 4: Cycle optimization on best
    print(f"\n[Phase 4] Cycle optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_cyc = _cycle_optimize(best_seq, deadline, min_time_remaining=200.0)
        update_best(a_cyc, None, "cycle")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else n_parent
    restart_kinds = ["gaussian", "wider_gauss", "asym_taper", "cos", "bimodal"]
    for i, rk in enumerate(restart_kinds):
        if time.monotonic() > deadline - 100:
            break
        if i < 2:
            a_start = _perturb(best_seq, 0.03 + 0.03 * i, rng)
        else:
            a_start = _make_profile(n_best, rk)

        remaining_for_this = min(60.0, deadline - 100 - time.monotonic())
        if remaining_for_this < 20:
            break
        a_r = _run_lbfgs(a_start, t_start + 1800 - 100 - (deadline - t_start - 1800 + time.monotonic() - t_start),
                         min_time_remaining=80.0)
        # Simpler: just use deadline - 100
        a_r = _run_lbfgs(a_start, deadline - 100 - i * 10, min_time_remaining=60.0)
        update_best(a_r, None, f"restart_{rk}")

        if time.monotonic() < deadline - 80:
            a_rp, sc_rp = _newton_polish(a_r, min(deadline - 70, time.monotonic() + 25),
                                         max_rounds=30)
            update_best(a_rp, sc_rp, f"restart_{rk}_np")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else 8192
    n_candidates = [16384, 20480, 12288]
    if n_best in n_candidates:
        n_candidates.remove(n_best)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        a_init = _resample(best_seq, n_try)
        remaining = min(deadline - 50 - time.monotonic(), 80.0)
        if remaining < 15:
            break
        a_opt = _run_lbfgs(a_init, t_start + budget_s - 40, min_time_remaining=30.0)
        update_best(a_opt, None, f"n_{n_try}")

        if time.monotonic() < deadline - 25:
            a_np3, sc_np3 = _newton_polish(a_opt, min(deadline - 20, time.monotonic() + 20),
                                           max_rounds=30)
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # Phase 7: Intensive final Newton
    print(f"\n[Phase 7] Final intensive Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _newton_polish(best_seq, min(deadline - 5, time.monotonic() + 100),
                                       max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

    # Phase 8: One more L-BFGS at very small beta
    print(f"\n[Phase 8] Ultra-fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 1e-10
        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            max_iter = min(5000, max(50, int(remaining / 0.005)))
            try:
                def obj_ultra(zv):
                    return _smoobj(zv, beta_fin, n)

                def grad_ultra(zv):
                    return _smograd(zv, beta_fin, n)

                result = minimize(
                    obj_ultra, z, jac=grad_ultra,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                             'maxcor': 120, 'maxls': 500}
                )
                a_flb = _softmax(result.x)
                update_best(a_flb, None, "ultra_lbfgs")
            except Exception:
                pass

    # Phase 9: Very final Newton
    if best_seq is not None and time.monotonic() < deadline - 1:
        a_veryfin, sc_veryfin = _newton_polish(best_seq, deadline - 0.5, max_rounds=50)
        update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
