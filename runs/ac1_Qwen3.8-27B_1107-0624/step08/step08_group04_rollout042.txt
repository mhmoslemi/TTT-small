def _full_convolve(a,b):

    if la==0 or lb==0: return np.zeros(max(0,la+lb-1))

```python
"""
Optimize a non-negative sequence for
    2*n*max(conv(a,a)) / (sum(a)**2)
using:
  - FFT-based smooth surrogates (log-sum-exp, p-norm, hinge),
  - L-BFGS in log-softmax parameter space,
  - a formal-polynomial square-root projection that enforces capped prefix
    convolution coefficients,
  - LSQR-based active-set/KKT refinements,
  - flatten polish and multi-resolution resampling.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.sparse.linalg import LinearOperator, lsqr

# ---------------------------------------------------------------
# FFT helpers
# ---------------------------------------------------------------

_N_CACHE = {}


def _get_next_pow2(k):
    k = int(max(2, k))
    val = _N_CACHE.get(k)
    if val is not None:
        return val
    p = 1
    while p < k:
        p <<= 1
    _N_CACHE[k] = p
    return p


def _full_convolve(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.zeros(max(0, la + lb - 1), dtype=float)
    out_n = la + lb - 1
    N = _get_next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    a = a / s
    c = _full_convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def _softmax(z):
    z = np.asarray(z, dtype=float).ravel()
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


# ---------------------------------------------------------------
# Global objective/gradient state for scipy
# ---------------------------------------------------------------

_OBJ = {
    'n': 0,
    'type': 0,          # 0 log-sum-exp, 1 p-norm, 2 hinge
    'beta': 1.0,
    'p': 8.0,
    'T': 0.0,
    'scale': 1.0,
}


def _global_obj(z):
    n = int(_OBJ['n'])
    if n <= 0:
        return np.inf
    a = _softmax(z)
    c = _full_convolve(a, a)
    typ = int(_OBJ['type'])
    scale = float(_OBJ['scale'])

    if typ == 0:
        cmax = float(np.max(c))
        if cmax <= 0.0:
            return 0.0
        beta = max(float(_OBJ['beta']), 1e-300)
        shifted = (c - cmax) / beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        s = float(e.sum())
        if s < 1e-300:
            J = cmax
        else:
            J = cmax + beta * np.log(s)
        return float(scale * 2.0 * n * J)

    if typ == 1:
        m = float(np.max(c))
        if m <= 0.0:
            return 0.0
        p = max(2.0, float(_OBJ['p']))
        d = c / m
        dpm1 = np.power(d, p - 1.0)
        dp = dpm1 * d
        S = float(dp.sum())
        if S < 1e-300 or not np.isfinite(S):
            S = 1.0
        J = m * (S ** (1.0 / p))
        return float(scale * 2.0 * n * J)

    if typ == 2:
        T = float(_OBJ['T'])
        w = np.maximum(c - T, 0.0)
        return float(scale * 0.5 * float(np.dot(w, w)))

    return np.inf


def _global_grad(z):
    n = int(_OBJ['n'])
    if n <= 0:
        return np.zeros(0)
    a = _softmax(z)
    c = _full_convolve(a, a)
    typ = int(_OBJ['type'])
    scale = float(_OBJ['scale'])

    if typ == 0:
        cmax = float(np.max(c))
        if cmax <= 0.0:
            return np.zeros(n)
        beta = max(float(_OBJ['beta']), 1e-300)
        shifted = (c - cmax) / beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

    elif typ == 1:
        m = float(np.max(c))
        if m <= 0.0:
            return np.zeros(n)
        p = max(2.0, float(_OBJ['p']))
        d = c / m
        dpm1 = np.power(d, p - 1.0)
        dp = dpm1 * d
        S = float(dp.sum())
        if S < 1e-300 or not np.isfinite(S):
            S = 1.0
        factor = S ** (1.0 / p - 1.0)
        w = factor * dpm1

    elif typ == 2:
        T = float(_OBJ['T'])
        w = np.maximum(c - T, 0.0)

    else:
        return np.zeros(n)

    if not np.all(np.isfinite(w)):
        w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    if float(np.sum(np.abs(w))) < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    grad *= scale

    if not np.all(np.isfinite(grad)):
        idx = int(np.argmax(c))
        w2 = np.zeros_like(c)
        w2[idx] = 1.0
        wc2 = _full_convolve(w2, a_rev)
        h2 = wc2[n - 1:2 * n - 1]
        W2 = float(c[idx])
        grad = 4.0 * n * a * (h2 - W2)
        grad *= scale
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
    return grad


def _lbfgs_pass(a, end_time, mode, param, maxiter, maxcor=20, label=""):
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n == 0 or maxiter <= 0:
        return a, _score(a)
    if end_time - time.monotonic() < 2.0:
        return a, _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()

    _OBJ['n'] = n
    if mode == 'log':
        _OBJ['type'] = 0
        _OBJ['beta'] = max(float(param), 1e-300)
        _OBJ['scale'] = 1.0
    elif mode == 'pnorm':
        _OBJ['type'] = 1
        _OBJ['p'] = float(max(2.0, param))
        _OBJ['scale'] = 1.0
    elif mode == 'hinge':
        _OBJ['type'] = 2
        _OBJ['T'] = float(param)
        _OBJ['scale'] = float(n)
    else:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(best_a)

    maxcor = max(5, min(30, int(maxcor)))
    try:
        res = minimize(
            _global_obj,
            z,
            jac=_global_grad,
            method='L-BFGS-B',
            options={
                'maxiter': int(maxiter),
                'ftol': 1e-13,
                'gtol': 1e-10,
                'maxcor': maxcor,
                'maxls': 12,
            }
        )
        if res is not None and res.x is not None:
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new < best_sc - 1e-15:
                best_a = a_new
                best_sc = sc_new
    except Exception:
        pass

    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    t = np.linspace(0.0, 1.0, n)
    if kind == "uniform":
        a = np.ones(n, dtype=float)
    elif kind == "linear":
        a = 0.25 + 0.75 * t
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-8)
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-8)
    elif kind == "centered":
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 1e-4
    elif kind == "geo_inc":
        a = np.exp(2.2 * t)
    elif kind == "geo_dec":
        a = np.exp(-2.2 * t)
    else:
        a = np.ones(n, dtype=float)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-30:
        return np.ones(n, dtype=float) / n
    return a / s


def _resample(a, n_new):
    a = np.asarray(a, dtype=float).ravel()
    n_old = len(a)
    if n_old == 0:
        return np.ones(n_new, dtype=float) / max(1, n_new)
    if n_old == n_new:
        b = a.copy()
    else:
        x_old = np.linspace(0.0, 1.0, n_old)
        x_new = np.linspace(0.0, 1.0, n_new)
        b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-30:
        return np.ones(n_new, dtype=float) / max(1, n_new)
    return b / s


def _perturb(a, sigma, rng):
    a = np.asarray(a, dtype=float).ravel()
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(z))
    z -= z.mean()
    return _softmax(z)


def _measure_conv_time(a):
    t0 = time.monotonic()
    _full_convolve(a, a)
    dt = time.monotonic() - t0
    return max(1e-5, dt)


def _maxiter_for(available, conv_time, safety=28.0, cap=3000, floor=10):
    if available <= 0.0:
        return 0
    est = max(1e-6, float(conv_time)) * float(safety)
    mi = int(available / est)
    mi = max(int(floor), mi)
    return min(int(cap), mi)


# ---------------------------------------------------------------
# Formal polynomial square root, used for capped-prefix projection
# ---------------------------------------------------------------


def _poly_conv(a, b, limit):
    a = np.asarray(a, dtype=float).ravel()
    b = np.asarray(b, dtype=float).ravel()
    if len(a) == 0 or len(b) == 0:
        return np.zeros(limit, dtype=float)
    if len(a) * len(b) < 20000:
        out = np.convolve(a, b)[:limit]
        if out.shape[0] < limit:
            out = np.pad(out, (0, limit - out.shape[0]), mode='constant')
        return out
    full = len(a) + len(b) - 1
    N = _get_next_pow2(full)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    out = np.fft.irfft(fa * fb, N)[:limit]
    if out.shape[0] < limit:
        out = np.pad(out, (0, limit - out.shape[0]), mode='constant')
    return out


def _poly_sqrt(Q, N):
    """
    Compute the formal power-series square root A[0:N] such that
    A*A = Q[0:N] modulo x^N and A[0] >= 0.
    Returns None if it becomes numerically bad.
    """
    Q = np.asarray(Q, dtype=float).ravel()
    if len(Q) < N:
        Q = np.pad(Q, (0, N - len(Q)), mode='constant')
    else:
        Q = Q[:N]

    if not np.all(np.isfinite(Q)):
        Q = np.nan_to_num(Q, nan=0.0, posinf=0.0, neginf=0.0)

    q0 = float(Q[0])
    if q0 <= 1e-300 or not np.isfinite(q0):
        return None

    Qs = Q / q0

    A = np.array([1.0], dtype=float)
    B = np.array([1.0], dtype=float)
    m = 1

    while m < N:
        m2 = min(2 * m, N)

        A_pad = np.zeros(m2, dtype=float)
        A_pad[:m] = A
        B_pad = np.zeros(m2, dtype=float)
        B_pad[:m] = B

        B2 = _poly_conv(B_pad, B_pad, m2)
        AB2_old = _poly_conv(A_pad, B2, m2)
        B_old2 = 2.0 * B_pad - AB2_old

        Q_cur = Qs[:m2]
        QB = _poly_conv(Q_cur, B_old2, m2)
        A_new = 0.5 * (A_pad + QB)

        AB2_new = _poly_conv(A_new, B2, m2)
        B_new = 2.0 * B_pad - AB2_new

        if not (np.all(np.isfinite(A_new)) and np.all(np.isfinite(B_new))):
            return None

        A = A_new
        B = B_new
        m = m2

    return A


def _factor_project_candidates(a, T, mix_candidates=(1.0,)):
    """
    Project by hard-capping the first n convolution coefficients at T,
    taking the formal square root, clipping to non-negativity, and keeping
    the best blend with the original oriented a.
    """
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n < 8:
        return a.copy(), _score(a)

    best_a = a.copy()
    best_sc = _score(best_a)

    work = a.copy()
    if work[0] < 1e-8:
        work[0] += 1e-7
        work /= work.sum()

    c = _full_convolve(work, work)
    Q = c[:n].copy()
    if T is None or T <= 0.0:
        T = float(np.max(c))
    Q = np.minimum(Q, T)

    if not np.all(np.isfinite(Q)):
        Q = np.nan_to_num(Q, nan=0.0, posinf=T, neginf=0.0)

    q0 = float(Q[0])
    Qmax = float(np.max(Q))
    if Qmax <= 1e-300:
        return best_a, best_sc
    if q0 <= max(1e-12 * Qmax, 1e-16):
        q0 = max(1e-10 * Qmax, 1e-14)
        Q[0] = q0
    if q0 <= 1e-300:
        return best_a, best_sc

    Qs = Q / q0
    A_scaled = _poly_sqrt(Qs, n)
    if A_scaled is None:
        return best_a, best_sc

    A = A_scaled * np.sqrt(q0)
    A = np.nan_to_num(A, nan=0.0, posinf=0.0, neginf=0.0)
    A_pos = np.maximum(A, 0.0)
    s = float(A_pos.sum())
    if s <= 1e-12:
        return best_a, best_sc
    root = A_pos / s

    for mix in mix_candidates:
        if mix >= 1.0:
            cand = root.copy()
        else:
            cand = (1.0 - mix) * work + mix * root
        cand = np.maximum(cand, 0.0)
        sc = float(cand.sum())
        if sc <= 1e-12:
            continue
        cand = cand / sc
        val = _score(cand)
        if np.isfinite(val) and val < best_sc:
            best_a = cand
            best_sc = val

    return best_a, best_sc


def _factor_refine(a, end_time, conv_time, target_score=1.500):
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n < 64 or end_time - time.monotonic() < 10.0:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(best_a)
    T_floor = max(1.02 / float(2 * n - 1), target_score / (2.0 * n))

    current_M = best_sc / (2.0 * n)
    if current_M <= T_floor * 1.001:
        return best_a, best_sc

    # Use the orientation with the larger left endpoint as base.
    base = best_a if best_a[0] >= best_a[-1] else best_a[::-1]

    max_rounds = 4
    ratios = [0.998, 0.995, 0.990, 0.980, 0.965, 0.945]
    mix_candidates = (1.0, 0.5)

    for _ in range(max_rounds):
        if time.monotonic() > end_time - 6.0:
            break
        current_M = best_sc / (2.0 * n)
        improved = False
        for ratio in ratios:
            if time.monotonic() > end_time - 4.0:
                break
            T = max(float(ratio) * current_M, T_floor)
            if T >= current_M * 0.999:
                T = current_M * 0.997
            T = min(T, current_M - 1e-18)
            if T <= T_floor * 0.999:
                break

            for orient in (base, base[::-1]):
                if time.monotonic() > end_time - 3.0:
                    break
                cand, sc = _factor_project_candidates(orient, T, mix_candidates)
                if np.isfinite(sc) and sc < best_sc - 1e-15:
                    best_a = cand
                    best_sc = sc
                    base = best_a if best_a[0] >= best_a[-1] else best_a[::-1]
                    improved = True

        if not improved:
            break

    return best_a, best_sc


# ---------------------------------------------------------------
# Active-set / LSQR KKT refinement
# ---------------------------------------------------------------

_ASTATE = {'n': 0, 'w': None}


def _hankel_matvec(x):
    n = int(_ASTATE['n'])
    w = _ASTATE['w']
    if n <= 0 or w is None:
        return np.zeros(n, dtype=float)
    v = x[::-1]
    return 2.0 * _full_convolve(v, w)[n - 1:2 * n - 1]


def _active_solve_candidates(a, w, available_time, conv_time, damp=1e-3):
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n < 16 or available_time <= 0.0 or w is None:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(best_a)

    w = np.asarray(w, dtype=float).ravel()
    if len(w) != 2 * n - 1:
        return best_a, best_sc

    iter_lim = int(max(12, min(90, available_time / max(1e-5, conv_time * 10.0))))
    _ASTATE['n'] = n
    _ASTATE['w'] = w
    Aop = LinearOperator((n, n), matvec=_hankel_matvec)
    b = np.ones(n, dtype=float)

    try:
        out = lsqr(Aop, b, damp=float(damp), atol=1e-7, btol=1e-7, iter_lim=iter_lim)
        x = np.asarray(out[0], dtype=float)
    except Exception:
        _ASTATE['w'] = None
        return best_a, best_sc

    _ASTATE['w'] = None
    if not np.all(np.isfinite(x)):
        return best_a, best_sc

    candidates = []
    xp = np.maximum(x, 0.0)
    sp = float(xp.sum())
    if sp > 1e-10:
        candidates.append(xp / sp)
    xa = np.abs(x)
    sa = float(xa.sum())
    if sa > 1e-10:
        candidates.append(xa / sa)

    if not candidates:
        return best_a, best_sc

    for cand_pos in candidates:
        for mix in (1.0, 0.6, 0.3):
            cand = (1.0 - mix) * a + mix * cand_pos
            cand = np.maximum(cand, 0.0)
            sc = float(cand.sum())
            if sc <= 1e-12:
                continue
            cand = cand / sc
            val = _score(cand)
            if np.isfinite(val) and val < best_sc:
                best_a = cand
                best_sc = val

    return best_a, best_sc


def _active_lsqr_refine(a, end_time, conv_time, max_rounds=4):
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n < 1024 or end_time - time.monotonic() < 60.0:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(best_a)

    for _ in range(max_rounds):
        if time.monotonic() > end_time - 30.0:
            break
        c = _full_convolve(best_a, best_a)
        M = float(np.max(c))
        if M <= 1e-300:
            break

        c_len = len(c)
        k = max(16, min(c_len, int(0.02 * n) + 32))
        if k >= c_len:
            thr = 0.0
            active = np.ones(c_len, dtype=bool)
        else:
            idx = max(0, c_len - k)
            thr = float(np.partition(c, idx)[idx])
            active = c >= thr
        active_count = max(1, int(active.sum()))
        w_active = np.zeros(c_len, dtype=float)
        w_active[active] = 1.0 / active_count
        w1 = 0.90 * w_active + 0.10 / float(c_len)

        theta = max(M * 0.002, 1e-300)
        z = (c - M) / theta
        e = np.exp(np.clip(z, -60.0, 0.0))
        s = float(e.sum())
        if s <= 1e-300:
            w_soft = np.zeros(c_len, dtype=float)
            w_soft[int(np.argmax(c))] = 1.0
        else:
            w_soft = e / s
        w2 = 0.78 * w_soft + 0.22 / float(c_len)

        solve_budget = min(40.0, max(8.0, 0.20 * (end_time - time.monotonic())))
        for w in (w1, w2):
            if time.monotonic() > end_time - 15.0:
                break
            cand, sc = _active_solve_candidates(best_a, w, solve_budget, conv_time)
            if np.isfinite(sc) and sc < best_sc - 1e-15:
                best_a = cand
                best_sc = sc

    return best_a, best_sc


# ---------------------------------------------------------------
# Flatten polish
# ---------------------------------------------------------------


def _flatten_polish(a, end_time, max_iters=250):
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n == 0 or end_time - time.monotonic() < 5.0:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(best_a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > end_time - 1.0:
            break
        c = _full_convolve(a, a)
        cmax = float(np.max(c))
        cmean = float(np.sum(c)) / float(len(c))
        excess = np.maximum(c - cmean, 0.0)
        emax = float(np.max(excess))
        if emax <= 0.0:
            break

        a_rev = a[::-1]
        wc = _full_convolve(excess, a_rev)
        grad = wc[n - 1:2 * n - 1]
        gn = float(np.max(np.abs(grad)))
        if gn <= 0.0:
            break

        step = 0.08 / (1.0 + 0.03 * it)
        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        s = float(a_try.sum())
        if s <= 1e-12:
            break
        a_try /= s

        sc_try = _score(a_try)
        if np.isfinite(sc_try) and sc_try < best_sc - 1e-15:
            best_a = a_try
            best_sc = sc_try
            a = a_try
            stall = 0
        else:
            a_half = 0.5 * a + 0.5 * a_try
            a_half = np.maximum(a_half, 0.0)
            sh = float(a_half.sum())
            if sh > 1e-12:
                a_half /= sh
                sc_half = _score(a_half)
                if np.isfinite(sc_half) and sc_half < best_sc - 1e-15:
                    best_a = a_half
                    best_sc = sc_half
                    a = a_half
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

        if stall > 45:
            break

    return best_a, best_sc


# ---------------------------------------------------------------
# Composite optimizer
# ---------------------------------------------------------------


def _optimize_sequence(a, conv_time, end_time, target_score=1.500):
    a = np.asarray(a, dtype=float).ravel()
    n = len(a)
    if n == 0 or end_time - time.monotonic() < 20.0:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(best_a)
    start = time.monotonic()
    total = max(20.0, end_time - start)

    # 1. Log-sum-exp continuation.
    end1 = start + 0.32 * total
    M = best_sc / (2.0 * n)
    log_params = [0.020 * M, 0.004 * M, 0.0008 * M, 0.00016 * M, 0.00003 * M]
    for i, beta in enumerate(log_params):
        run_end = start + ((i + 1) / float(len(log_params))) * (end1 - start)
        mi = _maxiter_for(run_end - time.monotonic(), conv_time, safety=30, cap=2500, floor=15)
        if mi > 0:
            best_a, best_sc = _lbfgs_pass(
                best_a, run_end, 'log', beta, mi,
                maxcor=min(25, max(10, int(400000 / max(1, n)))),
                label=f"log_{i}"
            )

    # 2. p-norm continuation.
    end2 = end1 + 0.25 * total
    p_params = [6.0, 12.0, 24.0, 48.0]
    M = best_sc / (2.0 * n)
    for i, p in enumerate(p_params):
        run_end = end1 + ((i + 1) / float(len(p_params))) * (end2 - end1)
        mi = _maxiter_for(run_end - time.monotonic(), conv_time, safety=30, cap=2500, floor=15)
        if mi > 0:
            best_a, best_sc = _lbfgs_pass(
                best_a, run_end, 'pnorm', p, mi,
                maxcor=min(25, max(10, int(400000 / max(1, n)))),
                label=f"pnorm_{i}"
            )

    # 3. Hinge target reduction.
    end3 = end2 + 0.25 * total
    M = best_sc / (2.0 * n)
    T_target = target_score / (2.0 * n)
    hinge_params = [
        max(M * 0.997, T_target * 1.03),
        max(M * 0.990, T_target * 1.015),
        max(M * 0.978, T_target * 1.005),
        max(M * 0.960, T_target),
        T_target,
    ]
    for i, T in enumerate(hinge_params):
        run_end = end2 + ((i + 1) / float(len(hinge_params))) * (end3 - end2)
        mi = _maxiter_for(run_end - time.monotonic(), conv_time, safety=35, cap=2000, floor=15)
        if mi > 0:
            best_a, best_sc = _lbfgs_pass(
                best_a, run_end, 'hinge', T, mi,
                maxcor=min(25, max(10, int(400000 / max(1, n)))),
                label=f"hinge_{i}"
            )

    # 4. Formal square-root capped prefix projection.
    end4 = end3 + 0.10 * total
    if end4 - time.monotonic() > 15.0:
        best_a, best_sc = _factor_refine(best_a, end4, conv_time, target_score)

    # 5. Active-set / LSQR polish.
    end5 = end4 + 0.05 * total
    if end5 - time.monotonic() > 40.0:
        best_a, best_sc = _active_lsqr_refine(best_a, end5, conv_time, max_rounds=3)

    # 6. Final flatten and a short ultra-tight log pass.
    if end_time - time.monotonic() > 20.0:
        best_a, best_sc = _flatten_polish(best_a, end_time - 5.0, max_iters=180)

    if end_time - time.monotonic() > 25.0:
        M = best_sc / (2.0 * n)
        mi = _maxiter_for(end_time - time.monotonic() - 2.0, conv_time, safety=35, cap=1200, floor=10)
        if mi > 0:
            best_a, best_sc = _lbfgs_pass(
                best_a, end_time - 2.0, 'log', max(1e-30, 0.00004 * M), mi,
                maxcor=min(20, max(8, int(300000 / max(1, n)))),
                label="final_log"
            )

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    safe_seq = None
    safe_score = np.inf

    def update(cand, sc, label=""):
        nonlocal best_seq, best_score, safe_seq, safe_score
        cand = np.asarray(cand, dtype=float).ravel()
        cand = np.clip(cand, 0.0, 1000.0)
        s = float(cand.sum())
        if cand.size == 0 or s < 0.01:
            return
        cand = cand / s
        if sc is None:
            sc = _score(cand)
        if not np.isfinite(sc):
            return
        if sc < best_score:
            prev = best_score
            best_seq = cand.copy()
            best_score = float(sc)
            print(f"NEW BEST: {best_score:.10f} n={len(cand)} [{label}] "
                  f"improve={prev - best_score:.3e} t={time.monotonic() - t_start:.1f}s")
        if len(cand) <= 32768 and sc < safe_score:
            safe_seq = cand.copy()
            safe_score = float(sc)

    # Load parent if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=float).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if p.size > 0 and s > 0.01:
            p = p / s
            parent = p
    except Exception:
        parent = None

    if parent is None:
        parent = _make_profile(8192, "tapered")
        print("No parent; using tapered profile n=8192")

    update(parent, _score(parent), "parent")
    best_seq = parent.copy()
    best_score = _score(parent)

    # Phase A: optimize at 16384 (or upsample/reoptimize small parents).
    try:
        if len(best_seq) <= 16384 and deadline - time.monotonic() > 550:
            nA = 16384
            initA = best_seq if len(best_seq) == nA else _resample(best_seq, nA)
            ctA = _measure_conv_time(initA)
            allocA = min(300.0, max(90.0, 0.20 * (deadline - time.monotonic())))
            endA = time.monotonic() + allocA
            print(f"\nPhase A: n={nA}, alloc={allocA:.0f}s, ct={ctA:.4f}s")
            candA, scA = _optimize_sequence(initA, ctA, endA, target_score=1.500)
            update(candA, scA, "phaseA_16384")
    except Exception as e:
        print("Phase A error:", e)

    # Phase B: n=32768.
    try:
        if deadline - time.monotonic() > 650:
            nB = 32768
            initB = _resample(best_seq, nB)
            ctB = _measure_conv_time(initB)
            allocB = min(650.0, max(160.0, 0.35 * (deadline - time.monotonic())))
            endB = time.monotonic() + allocB
            if endB > deadline - 50:
                endB = deadline - 50
            print(f"\nPhase B: n={nB}, alloc={endB - time.monotonic():.0f}s, ct={ctB:.4f}s")
            if endB - time.monotonic() > 60:
                candB, scB = _optimize_sequence(initB, ctB, endB, target_score=1.500)
                update(candB, scB, "phaseB_32768")
    except Exception as e:
        print("Phase B error:", e)

    # Phase C: n=65536 if enough time remains.
    try:
        if deadline - time.monotonic() > 650:
            nC = 65536
            initC = _resample(best_seq, nC)
            ctC = _measure_conv_time(initC)
            remaining = deadline - time.monotonic()
            if remaining > 500 and ctC < 0.12:
                endC = deadline - 30.0
                print(f"\nPhase C: n={nC}, remaining={remaining:.0f}s, ct={ctC:.4f}s")
                candC, scC = _optimize_sequence(initC, ctC, endC, target_score=1.500)
                update(candC, scC, "phaseC_65536")
            else:
                print("\nPhase C skipped: not enough time or conv too slow")
    except Exception as e:
        print("Phase C error:", e)

    # Final polish on the current best, but do not blow the deadline.
    try:
        if best_seq is not None and deadline - time.monotonic() > 120:
            nF = len(best_seq)
            ctF = _measure_conv_time(best_seq)
            allocF = min(180.0, max(50.0, deadline - time.monotonic() - 25.0))
            endF = time.monotonic() + allocF
            print(f"\nFinal polish: n={nF}, alloc={allocF:.0f}s, ct={ctF:.4f}s")
            candF, scF = _optimize_sequence(best_seq, ctF, endF, target_score=1.500)
            update(candF, scF, "final_polish")
    except Exception as e:
        print("Final polish error:", e)

    # Final selection. Prefer score; but if a very safe sequence is very close,
    # keep the better score to maximize the reported bound.
    if best_seq is None:
        fb = _make_profile(8192, "tapered")
        update(fb, _score(fb), "fallback")
        fb = np.asarray(best_seq, dtype=float).ravel()
        print(f"FALLBACK: score={best_score:.10f}")
        return [float(x) for x in fb]

    final = np.asarray(best_seq, dtype=float).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(8192, "uniform")
    final = final / final.sum()

    sc_final = _score(final)
    print(f"\n{'=' * 60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
    return [float(x) for x in final]
```
