```python
"""
Optimized step function search using cutting-plane Frank-Wolfe method (LP-based),
combined with log-space Adam, active-set equalization, and coordinate descent.
Key innovation: exploits the bilinear structure of convolution to find
improvement directions via linear programming on the active constraint set.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _frank_wolfe_lp(a, deadline, max_iter=300, eps=0.005, max_active=50):
    """Frank-Wolfe method using LP on active constraints to find directions."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_t = float(_fft_convolve(a, a).max())
    no_improve_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        t = float(c.max())

        if t < best_t - 1e-12:
            best_t = t
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > 50:
            break

        # Find active set
        active = np.where(c >= t * (1.0 - eps))[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > max_active:
            order = np.argsort(c[active])[::-1]
            active = active[order[:max_active]]

        K = len(active)

        # Build constraint matrix A (K x n)
        A_mat = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A_mat[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        # Solve LP: max sum(b) s.t. A_mat @ b <= t * 1, b >= 0
        try:
            c_obj = -np.ones(n)
            res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(K, t),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 15.0})
        except Exception:
            break

        if not res.success:
            break

        b_star = res.x
        sum_b = float(b_star.sum())

        if sum_b <= 1.0 + 1e-6:
            # Try slightly relaxed tolerance
            break

        # Rescale to get direction with sum=1
        g = b_star / sum_b

        # Line search: find best alpha
        best_alpha = 0.0
        best_new_t = t
        alphas = np.unique(np.concatenate([
            np.linspace(0.005, 0.3, 30),
            np.linspace(0.3, 0.7, 20),
            np.linspace(0.7, 1.0, 10)
        ]))
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g
            a_trial = np.maximum(0, a_trial)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            c_trial = _fft_convolve(a_trial, a_trial)
            new_t = float(c_trial.max())
            if new_t < best_new_t:
                best_new_t = new_t
                best_alpha = alpha

        if best_alpha < 1e-6:
            break

        # Update
        a = (1.0 - best_alpha) * a + best_alpha * g
        a = np.maximum(0, a)
        a /= a.sum()

        if it % 20 == 0:
            print(f"  FW[{it}]: t={t:.10f}, best={best_t:.10f}, alpha={best_alpha:.4f}, sum_b={sum_b:.6f}")

    return best_a, float(_fft_convolve(best_a, best_a).max())


def _frank_wolfe_multi_restart(a_init, deadline, n_restarts=3, max_active=40):
    """Run Frank-Wolfe from multiple perturbations of the initial point."""
    n = len(a_init)
    best_a = a_init.copy()
    best_t = float(_fft_convolve(a_init, a_init).max())
    rng = np.random.default_rng(12345)

    time_per = max(30.0, (deadline - time.monotonic()) / n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline - 30:
            break
        sub_dl = min(deadline - 10, time.monotonic() + time_per)

        if r == 0:
            a_start = a_init.copy()
        else:
            sigma = 0.003 + 0.002 * r
            z = np.log(np.maximum(a_init, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_start = _softmax_stable(z)

        a_opt, t_opt = _frank_wolfe_lp(a_start, sub_dl, max_iter=200,
                                        eps=0.005, max_active=max_active)
        if t_opt < best_t:
            best_t = t_opt
            best_a = a_opt.copy()
            print(f"  FW multi[{r}]: t={t_opt:.10f} (improved!)")

    return best_a, best_t


def _adam_optimize(a_init, deadline, T_start=1e-5, T_end=1e-7,
                   lr_start=0.02, lr_end=0.002, max_iter=500000,
                   eval_interval=50, active_k=5, backtrack_interval=150):
    """Adam optimizer in log-space with adaptive temperature and active-set awareness."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    last_improve = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8) * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Active-set gradient
        tol = c_max * 0.995
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > active_k:
            top_idx = np.argsort(c[active])[::-1][:active_k]
            active = active[top_idx]

        # Compute gradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        # Center gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps_adam)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if t % backtrack_interval == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z.copy()
                m_buf *= 0.5
                v_buf *= 0.5
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve = t
            else:
                patience += 1

            if patience > 10 and t - last_improve > 500:
                lr_mult *= 0.6
                patience = 0
                if lr_mult < 0.02:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.3

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=50000, top_k=5, initial_lr=0.003):
    """Direct subgradient method on the max convolution peak."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.998:
                active.append(idx)

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        g_tangent = g - float(np.dot(a, g)) * a
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break

        step = lr * g_tangent / g_norm
        a_new = np.maximum(0, a - step)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr *= 0.999
        else:
            lr *= 0.97
            if lr < 1e-9:
                break

    return best_a, _score_norm(best_a)


def _active_set_equalization(a, deadline, max_iter=3000, tol=0.002):
    """Equalize top peaks via minimum-norm linear system."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 12:
            sorted_a = active[np.argsort(c[active])[::-1][:12]]
            active = sorted_a

        K = len(active)
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        M_full = np.vstack([M, np.ones(n)])
        b_vec = np.concatenate([-np.ones(K), [0.0]])

        G = M_full @ M_full.T
        try:
            G_inv = np.linalg.inv(G + 1e-12 * np.eye(G.shape[0]))
            d = M_full.T @ (G_inv @ b_vec)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b_vec, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_cand = np.maximum(0, a + alpha * d)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            break

    return best_a, best_score

                        best_a = a_cand.copy()
                        any_imp = True
                        improved = True
                        break
                if improved:
                    break
            if not any_imp:
                break

    return best_a, best_score


def _build_constraint_matrix(a, active, n):
    """Build the K x n constraint matrix for active lags."""
    K = len(active)
    A_mat = np.zeros((K, n))
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A_mat[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A_mat


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. LP-based Frank-Wolfe (cutting plane) for large principled steps
    2. Log-space Adam for fine tuning
    3. Active-set equalization + coordinate descent for polish
    4. Random restarts to escape local minima
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Get initial solution
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Frank-Wolfe LP (0-400s)
    p1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: Frank-Wolfe LP ({p1_end - t_start:.0f}s budget)")
    a_fw, t_fw = _frank_wolfe_multi_restart(a, p1_end, n_restarts=2, max_active=50)
    sc_fw = _score(a_fw)
    print(f"Phase 1 done: score={sc_fw:.10f}, t={time.monotonic()-t_start:.0f}s")
    if sc_fw < best_score:
        best_score = sc_fw
        best_a = a_fw.copy()

    # Phase 2: Adam optimization (400-750s)
    p2_end = min(deadline, t_start + 750.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: Adam optimization")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_adam, sc_adam = _adam_optimize(
            a_init=best_a, deadline=p2_end,
            T_start=max_cb * 0.005, T_end=max_cb * 0.0001,
            lr_start=0.03, lr_end=0.003,
            max_iter=400000, eval_interval=50,
            active_k=5, backtrack_interval=100
        )
        full_sc = _score(a_adam)
        print(f"Phase 2 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_adam.copy()

    # Phase 3: Second Frank-Wolfe round (750-1050s)
    p3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Second Frank-Wolfe round")
        a_fw2, t_fw2 = _frank_wolfe_multi_restart(best_a, p3_end, n_restarts=3, max_active=60)
        sc_fw2 = _score(a_fw2)
        print(f"Phase 3 done: score={sc_fw2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_fw2 < best_score:
            best_score = sc_fw2
            best_a = a_fw2.copy()

    # Phase 4: Active-set equalization + coordinate descent (1050-1350s)
    p4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Active-set + coordinate descent")
        a_as, sc_as = _active_set_equalization(best_a, min(p4_end - 120, t_start + 1200),
                                               max_iter=5000, tol=0.001)
        full_sc_as = _score(a_as)
        print(f"  Active-set: score={full_sc_as:.10f}")
        if full_sc_as < best_score:
            best_score = full_sc_as
            best_a = a_as.copy()

        if time.monotonic() < p4_end - 30:
            a_cd, sc_cd = _coordinate_descent_polish(best_a, p4_end - 10, max_rounds=300)
            full_sc_cd = _score(a_cd)
            print(f"  Coord descent: score={full_sc_cd:.10f}")
            if full_sc_cd < best_score:
                best_score = full_sc_cd
                best_a = a_cd.copy()

    # Phase 5: Adam with lower temperature (1350-1550s)
    p5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p5_end:
        print(f"Phase 5: Refined Adam (low T)")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_adam2, sc_adam2 = _adam_optimize(
            a_init=best_a, deadline=p5_end,
            T_start=max_cb * 0.001, T_end=max_cb * 0.00001,
            lr_start=0.015, lr_end=0.001,
            max_iter=300000, eval_interval=40,
            active_k=4, backtrack_interval=80
        )
        full_sc2 = _score(a_adam2)
        print(f"Phase 5 done: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_adam2.copy()

    # Phase 6: Random restarts with quick optimization (1550-1730s)
    p6_end = min(deadline, t_start + 1730.0)
    if time.monotonic() < p6_end:
        print(f"Phase 6: Random restarts")
        n_restarts = 4
        time_per = max(25.0, (p6_end - time.monotonic()) / n_restarts)
        for r in range(n_restarts):
            if time.monotonic() > p6_end - time_per * 0.5:
                break
            sigma = 0.003 + 0.002 * r
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax_stable(z)

            sub_dl = min(p6_end - 5, time.monotonic() + time_per * 0.7)

            # Quick Frank-Wolfe on perturbed
            a_pert_opt, t_pert = _frank_wolfe_lp(a_pert, sub_dl, max_iter=100,
                                                  eps=0.005, max_active=40)
            # Quick Adam polish
            if time.monotonic() < sub_dl - 5:
                c_p = _fft_convolve(a_pert_opt, a_pert_opt)
                a_pert_opt2, _ = _adam_optimize(
                    a_init=a_pert_opt,
                    deadline=min(p6_end - 3, time.monotonic() + 20),
                    T_start=float(c_p.max()) * 0.001,
                    T_end=float(c_p.max()) * 0.00001,
                    lr_start=0.01, lr_end=0.001,
                    max_iter=50000, eval_interval=50, active_k=4
                )
                sc_pert2 = _score(a_pert_opt2)
                if sc_pert2 < _score(a_pert_opt):
                    a_pert_opt = a_pert_opt2

            full_sc_pert = _score(a_pert_opt)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_sc_pert:.10f}")
            if full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()

    # Phase 7: Final ultra-fine polish (1730-deadline)
    if time.monotonic() < deadline - 15:
        print(f"Phase 7: Final polish")
        a_sg, sc_sg = _subgradient_polish(best_a, deadline - 10,
                                           max_iter=30000, top_k=3, initial_lr=0.001)
        full_sc_sg = _score(a_sg)
        if full_sc_sg < best_score:
            best_score = full_sc_sg
            best_a = a_sg.copy()

        if time.monotonic() < deadline - 5:
            a_cd_f, _ = _coordinate_descent_polish(best_a, deadline - 3, max_rounds=50)
            full_sc_cd = _score(a_cd_f)
            if full_sc_cd < best_score:
                best_score = full_sc_cd
                best_a = a_cd_f.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) * 0.01

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
