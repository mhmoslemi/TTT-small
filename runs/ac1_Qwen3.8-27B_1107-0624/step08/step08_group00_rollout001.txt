def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

def _smoothed_obj_grad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

def _qp_solve(a, active_indices, target, n):
    """Solve QP to reduce peaks at active_indices to target level.
    min ||d||^2 s.t. conv(a+d, a+d)[k] <= target for k in active, sum(d)=0, a+d>=0"""
    c = _full_convolve(a, a)
    m = len(active_indices)
    
    # Build constraint matrix
    # conv(a+d, a+d)[k] = conv(a,a)[k] + 2*conv(a,d)[k] + conv(d,d)[k]
    # Linearize: conv(a+d, a+d)[k] ≈ conv(a,a)[k] + 2*conv(a,d)[k]
    # conv(a,d)[k] = sum_i a[i] * d[k-i]
    
    B = np.zeros((m, n))
    rhs = np.zeros(m)
    
    for row, k in enumerate(active_indices):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        j_lo = k - i_hi
        j_hi = k - i_lo
        a_vals = a[i_lo:i_hi + 1][::-1]
        B[row, j_lo:j_hi + 1] = a_vals
        rhs[row] = (target - c[k]) / 2.0
    
    # Use simple projected gradient on the QP (avoid cvxpy for speed)
    # min ||d||^2 s.t. B@d <= rhs, sum(d) = 0, a+d >= 0
    
    # Actually, let me use scipy's minimize for this.
    # The QP: min 0.5 * ||d||^2 s.t. B@d - rhs <= 0, sum(d) = 0, -d <= a
    
    from scipy.optimize import minimize as sp_minimize
    
    def qp_obj(d):
        return 0.5 * np.dot(d, d)
    
    def qp_grad(d):
        return d
    
    cons = [
        {'type': 'eq', 'fun': lambda d: np.sum(d), 'jac': lambda d: np.ones(n)},
    ]
    # Add inequality constraints for the active peaks
    # B@d <= rhs  =>  -(B@d) + rhs >= 0
    cons.append({
        'type': 'ineq',
        'fun': lambda d: rhs - B @ d,
        'jac': lambda d: -B
    })
    
    bounds = [(-a[i], np.inf) for i in range(n)]
    
    d0 = np.zeros(n)
    result = sp_minimize(qp_obj, d0, jac=qp_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 200, 'ftol': 1e-12})
    
    if result.success or result.fun < 1e-10:
        return result.x
    return None
