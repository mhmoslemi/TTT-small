import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

            # Reset tolerance if we improved significantly
            if best_sc - sc > 1e-10:
                tol_idx = max(0, tol_idx - 1)
        else:
            no_improve += 1

        if tol_idx >= len(tol_fracs) - 1 and no_improve > 8:
            break
        if no_improve > 20:
            break

    return best_a, best_sc


def _run_lbfgs_phase(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.004 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-7 * c_max_new:
            beta *= 0.2
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.5

        beta = min(beta, c_max_new * 0.001)
        if sc_improved < 1e-16:
            beta *= 0.3

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.5 + 0.005
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "wide_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "optimal_shape":
        # Based on asymptotic analysis: sin^{4/3} profile minimizes peak
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** (4.0 / 3.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _coordinate_descent(a_init, deadline, max_sweeps=50):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    idx_order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Randomize order each sweep
        rng.shuffle(idx_order)
        improved = False
        for j in idx_order:
            if time.monotonic() > deadline:
                break
            a_j_old = a[j]
            # Try a few multiplicative factors
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                a_try = a.copy()
                a_try[j] = a_j_old * factor
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try
                    improved = True
                    break
        if not improved and sweep > 5:
            # Try smaller perturbations
            for j in idx_order[:min(n, 2000)]:
                if time.monotonic() > deadline:
                    break
                a_j_old = a[j]
                for factor in [0.95, 1.05, 0.98, 1.02]:
                    a_try = a.copy()
                    a_try[j] = a_j_old * factor
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-15:
                        a = a_try
                        best_a = a.copy()
                        best_sc = sc_try
                        improved = True
                        break
            if not improved:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "optimal_shape")
        print(f"No parent, using optimal_shape n={n_parent}")
        update_best(parent, _score(parent), "init")

    target_n = n_parent if n_parent >= 8192 else 16384

    # Phase 1: Multi-resolution optimization from parent
    print(f"\n[Phase 1] Multi-resolution L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    mr_n_values = [target_n // 4, target_n // 2, target_n]
    mr_n_values = [xn for xn in mr_n_values if xn >= 512]
    if len(mr_n_values) < 3:
        mr_n_values = [2048, 4096, 8192, 16384]
        mr_n_values = [xn for xn in mr_n_values if xn <= target_n]
        if not mr_n_values:
            mr_n_values = [target_n]

    # Ensure we end at target_n
    if mr_n_values[-1] != target_n:
        mr_n_values.append(target_n)
    # Remove duplicates while preserving order
    seen = set()
    mr_n_unique = []
    for xn in mr_n_values:
        if xn not in seen:
            seen.add(xn)
            mr_n_unique.append(xn)
    mr_n_values = mr_n_unique

    current_a = parent.copy()
    for i, xn in enumerate(mr_n_values):
        if time.monotonic() > deadline - 200:
            break
        if len(current_a) != xn:
            current_a = _resample(current_a, xn)
        alloc = min(120.0, (deadline - 200 - time.monotonic()) / max(1, len(mr_n_values) - i))
        time_budget = time.monotonic() + alloc
        print(f"  MR stage: n={xn}, budget={alloc:.1f}s")
        current_a = _run_lbfgs_phase(current_a, min(time_budget, deadline - 150), min_time_remaining=30.0)
        sc = _score(current_a)
        update_best(current_a, sc, f"MR_n{xn}")

    # Phase 2: Active-set Newton polish with continuation
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        np_deadline = min(deadline - 50, time.monotonic() + 180)
        a_np, sc_np = _newton_active_polish(best_seq, np_deadline, max_rounds=200)
        update_best(a_np, sc_np, "newton_1")

    # Phase 3: L-BFGS from Newton result
    print(f"\n[Phase 3] Post-Newton L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_lb = _run_lbfgs_phase(best_seq, min(deadline - 30, time.monotonic() + 120), min_time_remaining=10.0)
        update_best(a_lb, _score(a_lb), "post_newton_lbfgs")

    # Phase 4: Another Newton round
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_np2, sc_np2 = _newton_active_polish(best_seq, min(deadline - 20, time.monotonic() + 60), max_rounds=100)
        update_best(a_np2, sc_np2, "newton_2")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    restart_configs = [
        ("optimal_shape", target_n, 0.0),
        ("tapered", target_n, 0.05),
        ("asym_taper", target_n, 0.05),
        ("gaussian", target_n, 0.0),
        ("wide_gauss", target_n, 0.0),
        ("left_heavy", target_n, 0.03),
        ("bimodal", target_n, 0.0),
    ]

    for i, (prof_kind, n_try, sigma) in enumerate(restart_configs):
        if time.monotonic() > deadline - 30:
            break
        remaining_for_restart = deadline - 30 - time.monotonic()
        if remaining_for_restart < 20:
            break
        alloc = min(40.0, remaining_for_restart / (len(restart_configs) - i))
        if alloc < 10:
            break

        if sigma > 0 and best_seq is not None:
            a_start = _perturb(best_seq, sigma, rng)
        else:
            a_start = _make_profile(n_try, prof_kind)

        a_opt = _run_lbfgs_phase(a_start, min(time.monotonic() + alloc, deadline - 20), min_time_remaining=5.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"restart_{prof_kind}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 15:
            a_rp, sc_rp = _newton_active_polish(a_opt, min(deadline - 8, time.monotonic() + 20), max_rounds=50)
            update_best(a_rp, sc_rp, f"restart_{prof_kind}_np")

    # Phase 6: Try different n values
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n_candidates = [32768, 8192, 4096]
        # Add target_n if not already there
        for nc in n_candidates:
            if time.monotonic() > deadline - 25:
                break
            if nc == len(best_seq):
                continue
            alloc = min(30.0, (deadline - 25 - time.monotonic()) / 2.0)
            if alloc < 10:
                break
            a_init = _resample(best_seq, nc)
            a_opt = _run_lbfgs_phase(a_init, min(time.monotonic() + alloc, deadline - 15), min_time_remaining=5.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{nc}")
            if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_np3, sc_np3 = _newton_active_polish(a_opt, min(deadline - 5, time.monotonic() + 15), max_rounds=40)
                update_best(a_np3, sc_np3, f"n_{nc}_np")

    # Phase 7: Coordinate descent polish
    print(f"\n[Phase 7] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        cd_deadline = min(deadline - 5, time.monotonic() + 30)
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_deadline, max_sweeps=30)
        update_best(a_cd, sc_cd, "coord_desc")

    # Phase 8: Final ultra-fine Newton
    print(f"\n[Phase 8] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=100)
        update_best(a_fin, sc_fin, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "optimal_shape")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
