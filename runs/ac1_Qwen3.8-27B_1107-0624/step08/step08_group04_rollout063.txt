        for i in range(n):
            hi = min(i + n - 1, 2 * n - 2)
            row_sums[i] = w_cumsum[hi + 1] - w_cumsum[i]
        row_sums = np.maximum(row_sums, 1e-15)
        precon = 1.0 / row_sums
    else:
        precon = np.ones(n)
    z = r * precon
    p = z.copy()
    rz = float(r @ z)
    rhs_norm_sq = float(rhs @ rhs)
    if rz < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, w, n) + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        r_ns = float(r @ r)
        if r_ns < tol * tol * max(1.0, rhs_norm_sq):
            break
        z_new = r * precon
        rz_new = float(r @ z_new)
        if abs(rz_new) < 1e-30:
            break
        beta_cg = rz_new / rz
        p = z_new + beta_cg * p
        rz = rz_new
    return x


def _newton_optimize(a_init, deadline, min_time_remaining=60.0, max_rounds=300):
    """Active-set Newton with adaptive tolerance and exponential weighting."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                    0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                    0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001]
    rounds_per_tol = 4

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - min_time_remaining:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        tol_idx = min(rnd // rounds_per_tol, len(tol_schedule) - 1)
        tol = tol_schedule[tol_idx]
        threshold = c_max * (1.0 - tol)
        active = c >= threshold
        if not active.any():
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        # Exponential weighting: emphasize the highest constraints
        beta_w = max(c_max * 0.001, 1e-10)
        w = np.zeros(2 * n - 1)
        w[active] = np.exp(-(c[active] - c_max) / beta_w)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w[active] = 1.0 / n_active

        # Gradient
        a_rev = a[::-1]
        wa = _full_convolve(w, a_rev)
        g = 2.0 * wa[n - 1:2 * n - 1]
        g_mean = g.mean()
        g_cent = g - g_mean
        g_norm = float(np.linalg.norm(g_cent))
        if g_norm < 1e-15 * c_max:
            break

        # Newton step via PCG
        dp = _pcg_solve(w, n, -g_cent, max_iter=300, tol=1e-10, reg=c_max * 1e-8)
        dp = dp - dp.mean()

        # Check directional derivative
        dir_deriv = float(g_cent @ dp)
        if dir_deriv > -1e-15 * g_norm:
            # Fallback to gradient descent
            dp = -g_cent / (g_norm + 1e-30)
            dir_deriv = -g_norm

        # Line search with backtracking
        step = 1.0
        found = False
        c_max_old = c_max
        for ls in range(40):
            a_try = a + step * dp
            if np.any(a_try < -1e-12):
                step *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_old - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                found = True
                break
            step *= 0.5
            if step < 1e-16:
                break

        if not found:
            # Smaller gradient step
            step2 = 0.05 / (g_norm + 1e-15)
            a_try = a - step2 * g_cent
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_old:
                a = a_try
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS on smoothed max (soft-max) objective in log domain."""
    from scipy.optimize import minimize
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(a)
    z = z - z.mean()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for beta_ratio in ratios:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * beta_ratio

        a_curr = a.copy()
        z_curr = np.log(np.maximum(a_curr, 1e-30))
        z_curr = z_curr - z_curr.mean()

        def obj_fun(zl):
            al = np.exp(zl - zl.max())
            al = al / al.sum()
            ca = _full_convolve(al, al)
            cm = float(ca.max())
            shifted = (ca - cm) / beta
            e = np.exp(np.clip(shifted, -500, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s))

        def grad_fun(zl):
            al = np.exp(zl - zl.max())
            al = al / al.sum()
            ca = _full_convolve(al, al)
            cm = float(ca.max())
            shifted = (ca - cm) / beta
            e = np.exp(np.clip(shifted, -500, 0))
            s = e.sum()
            if s < 1e-300:
                wv = np.zeros_like(ca)
                wv[int(np.argmax(ca))] = 1.0
            else:
                wv = e / s
            al_rev = al[::-1]
            wc = _full_convolve(wv, al_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(wv, ca))
            return 4.0 * n * al * (h - W_val)

        max_iter = min(3000, max(100, int(remaining / 0.05)))
        try:
            result = minimize(obj_fun, z_curr, jac=grad_fun, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxcor': 50})
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
        except Exception:
            pass

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t * t * 2.0)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(12345)
        freqs = rng.integers(1, 20, size=20)
        phases = rng.uniform(0, 2 * np.pi, size=20)
        amps = rng.uniform(0.1, 1.0, size=20)
        a = np.ones(n)
        for fi, ph, am in zip(freqs, phases, amps):
            a += am * np.cos(2 * np.pi * fi * t + ph)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()


def _coordinate_polish(a_init, deadline, max_passes=3):
    """Try single-coordinate adjustments to reduce the peak."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for p in range(max_passes):
        if time.monotonic() > deadline - 10:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find the argmax
        k_max = int(np.argmax(c))
        # For coordinate i, the contribution to c[k_max] is 2*a[i]*a[k_max-i] (if valid)
        # Try reducing coordinates that contribute most to c[k_max]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contributions[i] = 2.0 * a[i] * a[j]
            else:
                contributions[i] = 0.0
        # Try zeroing the top few contributors
        top_k = np.argsort(contributions)[::-1]
        for idx in top_k[:5]:
            if contributions[idx] < 1e-15:
                break
            a_try = a.copy()
            a_try[idx] *= 0.5
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()

    return best_a, best_sc
