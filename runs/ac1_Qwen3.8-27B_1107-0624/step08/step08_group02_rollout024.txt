Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp smoothing with L-BFGS, iterative active-set Newton
with KKT residual verification, and direct KKT linear solve for final polish.
Improvements over previous: tighter active set refinement, better line search,
direct KKT solve, more aggressive beta schedule, multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_grad(w, a, n):
    """Compute gradient of weighted average of convolution values.
    (H_w a)_j = sum_k w_k * d(conv(a,a)[k])/da_j
    = sum_k w_k * [a[k-j] + a[j-k]] (with bounds)
    This equals 2 * conv(w, a_reversed)[n-1 : 2n-1] (with appropriate handling)
    Actually let's compute it directly via the convolution identity.
    """
    a_rev = a[::-1]
    conv_w = _full_convolve(w, a_rev)
    # The gradient for position j is related to the convolution at offset n-1+j
    grad = 2.0 * conv_w[n - 1:2 * n - 1]
    return grad


def _cg_solve_hankel(w, n, rhs, max_iter=600, tol=1e-14):
    """Solve H_w x = rhs using CG, where H_w is the Hankel-like operator."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_grad(w, p, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.2:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = _hankel_grad(w, a, n)
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (g - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=200):
    """Improved Newton active set method with iterative refinement."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: start tight, expand if needed
        active_found = False
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 3e-3, 1e-2]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 1:
                active_found = True
                break
        if not active_found:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_grad(W, a, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Solve H @ dx = -g_t with constraint sum(dx) = 0
        Hg = _cg_solve_hankel(W, n, g, max_iter=600)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Backtracking line search with sufficient decrease
        alpha = 1.0
        step_done = False
        c_target = c_max * (1.0 - 0.01)  # want at least 1% relative decrease initially
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo-like condition
            if c_max_try < c_max - 0.001 * alpha * gn / n:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Try sub-gradient at various scales
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _direct_kkt_polish(a_init, deadline, max_iters=50):
    """Direct KKT solve: iteratively refine active set and solve the linear system."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify active set with very tight tolerance
        tol = 1e-10 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 1:
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())

        # Identify support (where a > threshold)
        support = a > 1e-12
        n_sup = int(support.sum())
        if n_sup < 1:
            break

        # Build and solve the KKT system on the support
        # H_w a = -lambda * 1 (on support), sum(a) = 1
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_act

        # Get the submatrix H_w restricted to support
        # We solve: H_w_full * a = -lambda * 1, sum(a) = 1, a >= 0
        # Using CG to compute H_w @ v for any v
        idx_sup = np.where(support)[0]

        # Use CG to solve the augmented system
        # [H_w  1] [a]   [0]
        # [1^T  0] [lam] = [1]
        # We'll use a projection approach:
        # Step 1: Find x such that H_w x = 1 (on support, approximately)
        # Step 2: The solution is a proportional to something related to H_w^{-1}

        # Simpler approach: solve H_w a = c0 * 1 sum(a) = 1 using 2x2 reduced system
        # Let a = alpha * H_w^{-1} 1 + beta * 1 (in the subspace)
        h1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=800)
        h1_h1 = float(np.dot(h1, _hankel_grad(W, h1, n)))
        h1_ones = float(np.dot(h1, np.ones(n)))
        ones_h1 = h1_ones
        ones_ones = float(n)

        # We want: H_w a = -lambda * 1, sum(a) = 1
        # a = x * h1 + y * ones
        # H_w (x*h1 + y*ones) = x*H_w h1 + y*H_w ones
        # This doesn't directly give us -lambda*1 unless H_w h1 is proportional to 1
        # Better: solve directly
        # For the full system (not just on support), after solving, clip to 0 and renormalize

        Hg = _cg_solve_hankel(W, n, np.ones(n), max_iter=800)
        # Now H_w @ Hg ≈ ones
        # We want a such that H_w a = c * 1 and sum(a) = 1
        # a = H_w^{-1} * c * 1 = c * Hg (approximately)
        # sum(Hg) = c * sum(a) = c, so c = 1/sum(Hg)
        sum_Hg = float(np.sum(Hg))
        if abs(sum_Hg) < 1e-20:
            break
        a_candidate = Hg / sum_Hg
        a_candidate = np.maximum(a_candidate, 0)
        s_cand = a_candidate.sum()
        if s_cand < 1e-15:
            break
        a_candidate = a_candidate / s_cand

        # Check if this is better
        c_cand = _full_convolve(a_candidate, a_candidate)
        c_max_cand = float(c_cand.max())

        # Also try a blend
        best_blend_a = a
        best_blend_sc = _score(a)
        for t in [0.0, 0.1, 0.25, 0.5, 0.75, 1.0]:
            a_blend = (1 - t) * a + t * a_candidate
            a_blend = np.maximum(a_blend, 0)
            s_b = a_blend.sum()
            if s_b < 1e-15:
                continue
            a_blend = a_blend / s_b
            sc_b = _score(a_blend)
            if sc_b < best_blend_sc:
                best_blend_sc = sc_b
                best_blend_a = a_blend.copy()

        if best_blend_sc < best_sc - 1e-15:
            a = best_blend_a
            best_a = a.copy()
            best_sc = best_blend_sc
        else:
            # Converged
            c_final = _full_convolve(a, a)
            # Check KKT residual
            g = _hankel_grad(W, a, n)
            g_t = g - np.mean(g)
            if np.linalg.norm(g_t) < 1e-12:
                break

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.0005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-12
    prev_sc = current_sc
    stall_count = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.002 * scale
        max_iter = min(15000, max(50, int(remaining / (est_iter_time * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-17:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
        except Exception:
            stall_count += 1

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-7 * c_max_new:
            beta *= 0.15
        elif stall_count < 2:
            beta *= 0.3
        else:
            beta *= 0.5
            # If stalled, try a small perturbation
            z = z + 0.01 * np.random.randn(n)
            z = z - z.mean()

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _run_cycle_optimize(a_init, deadline, min_time_remaining=60.0):
    """Alternate between L-BFGS bursts and Newton polish with direct KKT."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(a_init)

    cycle_time = min(45.0, (deadline - min_time_remaining - time.monotonic()) / 8.0)
    if cycle_time < 5:
        return current_a

    for cycle in range(10):
        if time.monotonic() + min_time_remaining > deadline:
            break
        t_cycle_start = time.monotonic()
        t_cycle_end = t_cycle_start + cycle_time

        # L-BFGS burst
        if time.monotonic() < t_cycle_end - 5:
            a_lb = _run_adaptive_lbfgs(current_a, min(t_cycle_end - 5, deadline - min_time_remaining),
                                       min_time_remaining=15.0)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb

        # Newton polish
        if time.monotonic() + 10 < min(t_cycle_end, deadline - min_time_remaining):
            a_np, sc_np = _newton_active_polish(
                current_a,
                min(t_cycle_end - 5, deadline - min_time_remaining),
                max_rounds=50
            )
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np

        # Direct KKT polish
        if time.monotonic() + 5 < min(t_cycle_end, deadline - min_time_remaining):
            a_kkt, sc_kkt = _direct_kkt_polish(
                current_a,
                min(t_cycle_end, deadline - min_time_remaining),
                max_iters=30
            )
            if sc_kkt < current_sc:
                current_a = a_kkt
                current_sc = sc_kkt

        cycle_time *= 1.3
        if cycle_time > 100:
            cycle_time = 100

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.5) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * t * (1 - t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with moderate tolerance
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        g = _hankel_grad(w, a, n)
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (g - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.05)
            else:
                patience += 1
                lr *= 0.7
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.015

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick L-BFGS from parent (use most of the time here)
    print(f"\n[Phase 1] Main L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_adaptive_lbfgs(parent, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Newton polish
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 200),
                                             max_rounds=150)
        update_best(a_np, sc_np, "newton_1")

    # Phase 3: Direct KKT polish
    print(f"\n[Phase 3] Direct KKT (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_kkt, sc_kkt = _direct_kkt_polish(best_seq, min(deadline - 300, time.monotonic() + 150),
                                            max_iters=80)
        update_best(a_kkt, sc_kkt, "kkt_1")

    # Phase 4: Second L-BFGS pass (with perturbation to escape)
    print(f"\n[Phase 4] Second L-BFGS with perturbation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_pert = _perturb(best_seq, 0.03, rng)
        a_l2 = _run_adaptive_lbfgs(a_pert, deadline, min_time_remaining=200.0)
        sc_l2 = _score(a_l2)
        update_best(a_l2, sc_l2, "lbfgs_2")

        if sc_l2 < best_score + 0.001:
            a_np2, sc_np2 = _newton_active_polish(a_l2, min(deadline - 150, time.monotonic() + 80),
                                                   max_rounds=80)
            update_best(a_np2, sc_np2, "newton_2")
            a_kkt2, sc_kkt2 = _direct_kkt_polish(a_np2, min(deadline - 100, time.monotonic() + 50),
                                                  max_iters=40)
            update_best(a_kkt2, sc_kkt2, "kkt_2")

    # Phase 5: Subgradient diversification
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 150, time.monotonic() + 80))
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.002:
            a_np3, sc_np3 = _newton_active_polish(a_sg, min(deadline - 80, time.monotonic() + 40),
                                                   max_rounds=60)
            update_best(a_np3, sc_np3, "newton_3")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192, 4096, 65536]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt_n = _run_adaptive_lbfgs(a_init_n, deadline, min_time_remaining=40.0)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")

        if sc_opt_n < best_score + 0.005 and time.monotonic() < deadline - 30:
            a_np_n, sc_np_n = _newton_active_polish(a_opt_n, min(deadline - 20, time.monotonic() + 20),
                                                     max_rounds=40)
            update_best(a_np_n, sc_np_n, f"n_{n_try}_np")
            a_kkt_n, sc_kkt_n = _direct_kkt_polish(a_np_n, deadline - 10, max_iters=20)
            update_best(a_kkt_n, sc_kkt_n, f"n_{n_try}_kkt")

    # Phase 7: Cycle optimization with diverse starts
    print(f"\n[Phase 7] Cycle optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_cyc = _run_cycle_optimize(best_seq, deadline, min_time_remaining=25.0)
        update_best(a_cyc, _score(a_cyc), "cycle")

    # Phase 8: Final intensive polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Multiple rounds of Newton + KKT
        for round_idx in range(5):
            if time.monotonic() > deadline - 2:
                break
            a_fin, sc_fin = _newton_active_polish(best_seq, min(deadline - 2, time.monotonic() + 10),
                                                   max_rounds=50)
            if sc_fin < best_score:
                update_best(a_fin, sc_fin, f"final_np_{round_idx}")
            a_kkt_f, sc_kkt_f = _direct_kkt_polish(best_seq, min(deadline - 1, time.monotonic() + 5),
                                                    max_iters=20)
            if sc_kkt_f < best_score:
                update_best(a_kkt_f, sc_kkt_f, f"final_kkt_{round_idx}")

        # One last L-BFGS at ultra-small beta
        if best_seq is not None and time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-11
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(50, int(remaining / 0.005)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    sc_flb = _score(a_flb)
                    update_best(a_flb, sc_flb, "final_lbfgs")

                    # Immediate Newton after
                    if sc_flb < best_score + 0.001 and time.monotonic() < deadline - 1:
                        a_veryfin, sc_veryfin = _newton_active_polish(a_flb, deadline - 0.5, max_rounds=30)
                        update_best(a_veryfin, sc_veryfin, "very_final_np")
                        a_verykkt, sc_verykkt = _direct_kkt_polish(a_veryfin, deadline - 0.1, max_iters=10)
                        update_best(a_verykkt, sc_verykkt, "very_final_kkt")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
