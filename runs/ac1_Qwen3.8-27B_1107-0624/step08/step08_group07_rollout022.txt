"""
Step function optimization using spectral-domain projected gradient,
log-domain Adam with active sets, and multi-strategy search.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1 (or we just compute 2*n*max_conv)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _spectral_gradient_step(a, alpha, n_active=1, T=0.01):
    """One step of spectral-domain projected gradient.
    
    Parameters:
    - a: input sequence (sum=1, non-negative)
    - alpha: step size
    - n_active: number of active peaks to equalize
    - T: temperature for active peak weighting
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    
    A = np.fft.fft(a, N)
    C = np.abs(A)**2  # Power spectrum (real, even)
    c = np.fft.ifft(C, N).real  # Autocorrelation (real, even)
    
    # Find active peaks
    c_max = float(c.max())
    if n_active == 1:
        k_star = int(np.argmax(c))
        active_weights = np.array([c_star])
        active_lags = np.array([k_star])
    else:
        # Use softmax weighting over top peaks
        c_sorted_idx = np.argsort(c)[::-1][:n_active]
        c_vals = c[c_sorted_idx]
        w = np.exp((c_vals - c_vals.max()) / (T + 1e-30))
        w = w / w.sum()
        active_lags = c_sorted_idx
        active_weights = w
    
    # Compute gradient in spectral domain
    # grad_C[k] = sum_j w_j * cos(2*pi * k_j * k / N) / N
    k_indices = np.arange(N)
    grad_C = np.zeros(N)
    for j in range(len(active_lags)):
        grad_C += active_weights[j] * np.cos(2.0 * np.pi * active_lags[j] * k_indices / N)
    grad_C /= N
    
    # Update C
    C_new = C - alpha * grad_C
    C_new = np.maximum(0, C_new)  # Project to C >= 0
    
    # Ensure C is even (real signal): C[k] = C[N-k]
    C_new = (C_new + C_new[::-1]) / 2.0
    C_new = np.maximum(0, C_new)
    
    # Recover a: keep phase, update magnitude
    phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
    A_new = np.sqrt(C_new) * phase
    # Ensure A_new has the right symmetry for real signal: A_new[k] = conj(A_new[N-k])
    A_new = (A_new + np.conj(A_new[::-1])) / 2.0
    
    a_new_padded = np.fft.ifft(A_new, N).real
    a_new = a_new_padded[:n]
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a  # Fallback
    a_new /= s
    return a_new

def _spectral_optimize(a, deadline, max_iter=100000, alpha_start=0.1, alpha_end=0.001,
                        n_active_start=1, n_active_end=5, T=0.01):
    """Optimize using spectral-domain projected gradient."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        n_active = int(n_active_start + (n_active_end - n_active_start) * frac)
        
        a_new = _spectral_gradient_step(a, alpha, n_active=n_active, T=T)
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        
        if patience > 200:
            # Reduce alpha
            alpha_start *= 0.5
            alpha_end *= 0.5
            patience = 0
            if alpha_start < 1e-10:
                break
    
    return best_a, best_score

def _subgradient_refine(a, deadline, max_iter=20000, top_k=3, lr_start=0.005):
    """Subgradient method with backtracking line search."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
        
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project to tangent space (preserve sum)
        g_mean = g.mean()
        g = g - g_mean
        
        step = lr * g
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_new /= s
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_a, best_score

"""

1. Log-domain Adam with smoothed and active-set gradients
2. Subgradient method with backtracking line search and top-k equalization  
3. Simulated annealing with peak-targeted moves
4. Spectral-domain gradient with magnitude modification

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex: a >= 0, sum(a) = 1."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _compute_gradient_active(a, c, active_lags):
    """Compute gradient of max(c) at active lags w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active_lags)
    return g


def _compute_gradient_smoothed(a, c, T, top_m=20):
    """Compute smoothed gradient using softmax over top lags."""
    n = len(a)
    idx = np.argsort(c)[::-1][:top_m]
    c_top = c[idx]
    w = np.exp((c_top - c_top[0]) / (T + 1e-30))
    w = w / w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += w[i] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _adam_log_domain(a_init, deadline, T_start, T_end, lr_start, lr_end,
                     max_iter, eval_interval=50, active_k=5,
                     backtrack_interval=150, top_m=20):
    """Log-domain Adam with adaptive temperature and active-set awareness."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    last_best_t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.7) * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Choose gradient method based on temperature
        if T < c_max * 0.005:
            tol = c_max * (1.0 - 0.002)
            active = np.where(c >= tol)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])
            elif len(active) > active_k:
                top_idx = np.argsort(c[active])[::-1][:active_k]
                active = active[top_idx]
            g = _compute_gradient_active(a, c, active)
        else:
            g = _compute_gradient_smoothed(a, c, T, top_m=top_m)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Log-space gradient with centering
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 1.0:
            update = update / un
        z = z - update
        z = np.clip(z, -70.0, 70.0)
        z = z - z.mean()

        # Backtracking
        if t % backtrack_interval == 0:
            a_cand = _softmax_stable(z)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_best_t = t
            else:
                patience += 1

            if patience > 12 and (t - last_best_t) > 200:
                lr_mult *= 0.7
                patience = 0
                if lr_mult < 0.02:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.004 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.3

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _subgradient_linesearch(a, deadline, max_iter=30000, top_k=5,
                            lr_start=0.01, lr_min=1e-12):
    """Subgradient with backtracking line search and top-k equalization."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_start
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        active = [int(sorted_idx[0])]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.998:
                active.append(int(idx))

        g = _compute_gradient_active(a, c, active)
        # Project to preserve sum (tangent space)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Backtracking line search
        found = False
        step_lr = lr
        for _ in range(10):
            a_new = a - step_lr * g / g_norm
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                step_lr *= 0.5
                continue
            a_new /= s
            sc_new = _score_norm(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                lr = min(lr * 1.02, lr_start)
                stall_count = 0
                found = True
                break
            else:
                step_lr *= 0.5

        if not found:
            lr *= 0.85
            stall_count += 1
            if stall_count > 200 or lr < lr_min:
                break

    return best_a, best_score


def _simulated_annealing(a, deadline, max_iter=50000, T_start=0.01,
                         T_end=0.0001, n_sub=10, pert_scale=0.02):
    """Simulated annealing with peak-targeted perturbation moves."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max_iter)
        T = T_start * (T_end / T_start) ** frac

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Generate perturbation targeting the peak
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        if lo > hi:
            lo, hi = 0, n - 1

        # Pick random indices to move mass
        candidates = np.arange(lo, hi + 1)
        if len(candidates) < 2:
            candidates = np.arange(n)

        for _ in range(n_sub):
            j1 = int(rng.integers(len(candidates)))
            j2 = int(rng.integers(len(candidates)))
            if j1 == j2:
                continue
            i1 = candidates[j1]
            i2 = candidates[j2]
            if a[i1] < 1e-15:
                continue

            # Move mass from i1 to i2 (or vice versa)
            delta = pert_scale * a[i1] * rng.uniform(0.1, 1.0)
            sign = 1.0 if rng.random() < 0.5 else -1.0

            a_pert = a.copy()
            a_pert[i1] -= sign * delta
            a_pert[i2] += sign * delta
            a_pert = np.maximum(0, a_pert)
            s = a_pert.sum()
            if s < 1e-15:
                continue
            a_pert /= s

            sc_pert = _score_norm(a_pert)
            delta_sc = sc_pert - _score_norm(a)

            if delta_sc < 0 or rng.random() < np.exp(-delta_sc / (T + 1e-30)):
                a = a_pert
                sc_a = _score_norm(a)
                if sc_a < best_score:
                    best_score = sc_a
                    best_a = a.copy()

    return best_a, best_score


def _spectral_magnitude_step(a, alpha, n_active=3, T_smooth=0.01):
    """Spectral-domain step: modify |FFT(a)|^2 to flatten the autocorrelation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)

    A = np.fft.fft(a, N)
    C = np.abs(A) ** 2
    c = np.fft.ifft(C, N).real

    # Find active peaks with softmax weighting
    c_max = float(c.max())
    idx = np.argsort(c)[::-1][:n_active]
    c_top = c[idx]
    w = np.exp((c_top - c_top[0]) / (T_smooth + 1e-30))
    w = w / w.sum()

    # Gradient of max(c) w.r.t. C (in spectral domain)
    k_indices = np.arange(N)
    grad_C = np.zeros(N)
    for j in range(len(idx)):
        grad_C += w[j] * np.cos(2.0 * np.pi * int(idx[j]) * k_indices / N)
    grad_C /= N

    # Update
    C_new = C - alpha * grad_C
    C_new = np.maximum(0, C_new)
    # Enforce Hermitian symmetry
    C_new = (C_new + C_new[np.mod(-k_indices, N)]) / 2.0
    C_new = np.maximum(0, C_new)

    # Recover a
    mag_new = np.sqrt(C_new)
    phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
    A_new = mag_new * phase
    # Enforce conjugate symmetry
    A_new = (A_new + np.conj(A_new[np.mod(-k_indices, N)])) / 2.0

    a_new = np.fft.ifft(A_new, N).real[:n]
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def _spectral_optimize(a, deadline, alpha_start=0.02, alpha_end=0.0005,
                       max_iter=30000, n_active=3, T_smooth=0.01):
    """Iterative spectral-domain optimization."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    stall = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iter))
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        a_new = _spectral_magnitude_step(a, alpha, n_active=n_active,
                                         T_smooth=T_smooth)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 100:
                alpha_start *= 0.5
                alpha_end *= 0.5
                stall = 0
                if alpha_start < 1e-12:
                    break

    return best_a, best_score


def _coordinate_polish(a, deadline, max_rounds=100):
    """Quick coordinate descent on elements contributing to the peak."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()

    for rnd in range(max_rounds):

        c = _fft_convolve(best_a, best_a)
        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        improved = False
        # Try modifying a few random contributing elements
        candidates = np.random.choice(np.arange(lo, hi + 1),
                                      size=min(20, hi - lo + 1),
                                      replace=False)
        for j in candidates:
            if best_a[j] < 1e-15:
                continue
            for scale in [0.8, 0.9, 1.1, 1.2]:
