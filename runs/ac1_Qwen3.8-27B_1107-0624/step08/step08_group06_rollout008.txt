"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Peak equalization via linear system solve
2. Adam optimizer with log-sum-exp smoothing
3. Multiplicative weights (mirror descent)
4. Adaptive beta scheduling
5. Multi-n exploration with interpolation
6. Final active-set Newton polish
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

            i_lo2 = max(0, k)

def _adam_optimize(a_init, deadline, max_steps=50000):
    """Adam optimizer with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    # Adaptive beta: start large, decrease
    c_init = _full_convolve(a_init, a_init)
    c_max_init = float(c_init.max())
    c_sorted = np.sort(c_init)[::-1]
    gap = c_sorted[min(10, len(c_sorted)-1)] - c_sorted[min(100, len(c_sorted)-1)]
    base_beta = max(c_max_init * 0.05, abs(gap) * 2, c_max_init * 0.01)
    
    m_v = np.zeros_like(z)  # First moment
    v_v = np.zeros_like(z)  # Second moment
    beta1 = 0.8
    beta2 = 0.999
    lr = 0.01
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Adaptive beta
        beta = base_beta * (0.95 ** (step // 100))
        beta = max(beta, c_max_init * 0.0001)
        
        # Compute gradient
        obj = _lsobj(z, beta, n)
        grad = _lsgrad(z, beta, n)
        
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**(step+1))
        v_hat = v_v / (1 - beta2**(step+1))
        
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z_new = z_new - z_new.mean()  # Preserve normalization in log-space
        
        a_new_conv = _softmax(z_new)
        
        # Check improvement
        if step % 50 == 0:
            sc = _score(a_new_conv)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_new_conv.copy()
        
        z = z_new
    
    sc_final = _score(_softmax(z))
    if sc_final < best_sc:
        best_a = _softmax(z)
        best_sc = sc_final
    return best_a

`dx = (J^T J + lam * 1*1^T)^{-1} J^T r` (approximately, for small lam)

The `lam * 1*1^T` term enforces the zero-sum constraint. But since `J^T J` already has the null space containing `1` (the all-ones vector), we can instead just solve `J @ dx ≈ r` and then project out the mean: `dx = dx - mean(dx)`.

Actually, that's not quite right either. The minimum-norm solution to `J @ dx = r` (ignoring the constraint) is `dx = J^+ r` where `J^+` is the Moore-Penrose pseudoinverse. The constraint `sum(dx) = 0` means we want the component of `dx` that's orthogonal to `1`.

For practical purposes, using a small `lam` in `J^T J + lam * I` and then subtracting the mean should work well.

Let me also think about efficiency. For `n = 16384`:
- Each `_full_convolve` call requires an FFT of size ~32768, which takes about 0.1ms
- In the CG loop, each iteration requires 2 `_full_convolve` calls for `J@v` and 2 for `J^T@(J@v)`, so 4 FFTs total, about 0.4ms
- With 100 CG iterations, that's about 40ms per outer iteration
- With 30 outer iterations, that's about 1.2 seconds total

That's efficient enough!

Now let me also think about the **Adam optimizer** approach. The main advantage of Adam over L-BFGS is:
1. It doesn't need to store and update the Hessian approximation
2. It's more robust to noise in the gradient
3. It can handle the rapidly changing landscape as beta decreases better

The disadvantage is that it's a first-order method and might converge slower near the optimum. So the strategy should be:
- Use Adam for the bulk of the optimization (when beta is moderate)
- Switch to L-BFGS or Newton for the final polish (when beta is very small)

Let me now write the complete solution:

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Peak equalization via FFT-accelerated CG on normal equations
2. Adam optimizer with log-sum-exp smoothing and adaptive beta
3. Multiplicative weights (mirror descent) for robustness
4. Multi-n exploration with interpolation
5. Final active-set Newton polish
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _peak_equalize_cg(a, n_peaks=80, max_outer=40, max_cg=80, lam=1e-8):
    """Peak equalization using CG on (J^T J + lam*I) dx = J^T r."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > _peak_eq_deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        n_top = min(n_peaks, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        top_vals = c[top_idx].copy()
        
        # Target: mean of top values
        target = np.mean(top_vals)
        r_vec = target - top_vals
        
        # Precompute FFT of a and a_rev
        N_fft = _next_pow2(3 * n - 2)  # Need this for _full_convolve of length-(2n-1) and length-n
        Fa = np.fft.rfft(a, N_fft)
        Farev = np.fft.rfft(a[::-1], N_fft)
        
        def compute_Jv(v):
            """Compute J @ v efficiently."""
            Fv = np.fft.rfft(v, N_fft)
            conv_av = np.fft.irfft(Fa * Fv, N_fft)  # a * v, valid length 2n-1
            conv_varev = np.fft.irfft(Fv * Farev, N_fft)  # v * a_rev, valid length 2n-1
            return conv_av[top_idx] + conv_varev[top_idx + n - 1]
        
        def compute_JtJv(v):
            """Compute (J^T J + lam*I) @ v."""
            Jv = compute_Jv(v)
            # J^T @ Jv: place Jv at top_idx positions in a length-(2n-1) vector
            w = np.zeros(2 * n - 1)
            w[top_idx] = Jv
            # J^T @ w = convolve(w, a_rev)[n-1:n-1+n] + convolve(w, a)[:n]
            Fw = np.fft.rfft(w, N_fft)
            conv_w_arev = np.fft.irfft(Fw * Farev, N_fft)
            conv_w_a = np.fft.irfft(Fw * Fa, N_fft)
            result = conv_w_arev[n-1:2*n-1] + conv_w_a[:n]
            return result + lam * v
        
        # Compute rhs = J^T @ r_vec
        w_r = np.zeros(2 * n - 1)
        w_r[top_idx] = r_vec
        Fwr = np.fft.rfft(w_r, N_fft)
        conv_wr_arev = np.fft.irfft(Fwr * Farev, N_fft)
        conv_wr_a = np.fft.irfft(Fwr * Fa, N_fft)
        rhs = conv_wr_arev[n-1:2*n-1] + conv_wr_a[:n]
        
        # CG solver
        x = np.zeros(n)
        r_cg = rhs.copy()
        p = rhs.copy()
        rs = float(np.dot(r_cg, r_cg))
        rhs_norm = float(np.linalg.norm(rhs))
        
        if rhs_norm < 1e-15:
            continue
        
        for cg_iter in range(max_cg):
            Ap = compute_JtJv(p)
            pAp = float(np.dot(p, Ap))
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_cg -= alpha * Ap
            rs_new = float(np.dot(r_cg, r_cg))
            if np.sqrt(rs_new) < 1e-10 * (1.0 + rhs_norm):
                break
            p = r_cg + (rs_new / rs) * p
            rs = rs_new
        
        # Enforce zero sum
        x = x - np.mean(x)
        
        # Line search
        t = 1.0
        accepted = False
        for ls in range(50):
            a_new = a + t * x
            a_new = np.maximum(a_new, 0)
            s_new = float(a_new.sum())
            if s_new < 1e-15:
                t *= 0.5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-13:
                a = a_new.copy()
                best_a = a_new.copy()
                best_sc = sc_new
                accepted = True
                break
            t *= 0.5
        
        if not accepted:
            # Try smaller step
            t = 0.1
            for ls in range(20):
                a_new = a + t * x
                a_new = np.maximum(a_new, 0)
                s_new = float(a_new.sum())
                if s_new < 1e-15:
                    t *= 0.5
                    continue
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-14:
                    a = a_new.copy()
                    best_a = a_new.copy()
                    best_sc = sc_new
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break
    
    return best_a, best_sc


def _adam_optimize(a_init, deadline, max_steps=30000):
    """Adam optimizer with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    c_init = _full_convolve(a_init, a_init)
    c_max_init = float(c_init.max())
    c_sorted = np.sort(c_init)[::-1]
    # Adaptive base beta based on peak structure
    n_check = min(50, len(c_sorted))
    gap = c_sorted[0] - c_sorted[n_check - 1]
    base_beta = max(c_max_init * 0.03, gap * 1.5, c_max_init * 0.005)
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    lr = 0.05
    eps = 1e-8
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Adaptive beta: decrease over time
        progress = step / max(max_steps, 1)
        beta = base_beta * (1.0 - 0.9 * progress)
        beta = max(beta, c_max_init * 0.0005)
        
        # Compute gradient
        grad = _lsgrad(z, beta, n)
        
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * (grad * grad)
        m_hat = m_v / (1.0 - beta1 ** (step + 1))
        v_hat = v_v / (1.0 - beta2 ** (step + 1))
        
        # Scale learning rate based on gradient magnitude
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            lr_eff = lr * c_max_init / gnorm
        else:
            lr_eff = lr
        
        z = z - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()
        
        # Periodic check
        if step % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final
        best_sc = sc_final
    return best_a, best_sc


def _multiplicative_weights(a_init, deadline, n_steps=10000):
    """Mirror descent with negative entropy (multiplicative weights)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = 0.1
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Softmax weights on peaks
        tau = c_max * 0.01
        shifted = (c - c_max) / tau
        w = np.exp(np.clip(shifted, -700, 0))
        w = w / w.sum()
        
        # Gradient direction: 2 * (w conv a_rev) [n-1:2n-1]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)
        
        # Project gradient (zero mean)
        grad = grad - np.mean(grad)
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        # Adaptive eta
        if step % 500 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                eta *= 1.1  # Increase step
            else:
                eta *= 0.9  # Decrease step
            eta = np.clip(eta, 0.01, 1.0)
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a
        best_sc = sc_final
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(4000, max(30, int(remaining / (est_time_per_iter * 5))))
        
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-13,
                    'maxcor': 50,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = max(1, n // 5)
        a[:block_size] = 1.0
        a[-(block_size):] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n
