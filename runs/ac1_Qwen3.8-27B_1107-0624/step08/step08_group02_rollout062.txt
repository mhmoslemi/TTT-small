"""
Optimized minimizer for 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements:
- Arcsine-based initialization (theoretically near-optimal continuous limit)
- Variance-aware objective to encourage flat convolution
- Direct KKT system solve via structured CG
- Multi-eps arcsine exploration
- Refined active set Newton with adaptive tolerance and complementarity checks
- Better beta continuation schedule

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute 2 * (W_reversed * v)[n-1:2n-1] which gives the gradient structure."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    """Solve H_W x = rhs where H_W is the Hankel operator defined by W."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            alpha *= 0.5
            x_new = x + alpha * p
            r_new = r - alpha * Hp
            rs_new = float(r_new @ r_new)
            if not np.isfinite(rs_new):
                break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n, var_weight=0.0):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
    if var_weight > 0:
        c_mean = c.sum() / len(c)
        var_c = float(np.mean((c - c_mean) ** 2))
        obj += var_weight * var_c / (c_max * c_max + 1e-30)
    return obj


def _smograd(z, beta, n, var_weight=0.0):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    if var_weight > 0:
        c_mean = c.sum() / len(c)
        dc_mean = w * 0  # derivative of mean is 0 w.r.t. individual c_k in this param
        dc_var = 2.0 * (c - c_mean) / len(c)
        wc_var = _full_convolve(dc_var, a_rev)
        h_var = wc_var[n - 1:2 * n - 1]
        c_max_sq = c_max * c_max + 1e-30
        grad += var_weight * 4.0 * n * a * h_var / c_max_sq
    return grad

def _newton_active_polish(a_init, deadline, max_rounds=200, verbose=False):
    """Active set Newton method for the KKT conditions."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set selection
        active = None
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 3e-3, 1e-2]:
            tol = tol_frac * c_max
            candidate = c >= (c_max - tol)
            if int(candidate.sum()) >= 2:
                active = candidate
                break
        if active is None or int(active.sum()) < 1:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[max(0, idx-1):idx+2] = True
            active = c >= (c_max - 0.01 * c_max)
            if int(active.sum()) < 1:
                active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute gradient: g[m] = 2 * sum_{k in S} (1/|S|) * a_{k-m}
        g = _hankel_matvec(a, W, n)
        g_mean = np.mean(g)
        g_t = g - g_mean
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * max(1, n):
            break

        # Solve the KKT system: H_W * dx = -g_t, with constraint sum(dx) = 0
        # Use augmented system: [H  1; 1^T  0] [dx; lam] = [-g; 0]
        # But we solve H*dx = -g_t + lam*1, then determine lam from sum(dx)=0
        rhs_neg = -g_t
        Hg = _cg_solve(W, n, rhs_neg, max_iter=800, tol=1e-15)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=800, tol=1e-15)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = g_t / max(gn, 1e-30)
            dx = -dx
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = Hg - lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search with exact merit
        alpha = 1.0
        step_done = False
        c_max_before = c_max
        for ls in range(60):
            a_try = a + alpha * dx
            n_neg = int((a_try < 0).sum())
            if n_neg > 0:
                # Backtracking to find feasible alpha
                alpha *= 0.5
                if alpha < 1e-15:
                    break
                continue
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo condition
            if c_max_try <= c_max_before - 0.001 * alpha * gn / max(n, 1) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.7
            if alpha < 1e-15:
                break

        if not step_done:
            # Try smaller fixed steps
            for trial_alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a + trial_alpha * dx
                if np.any(a_try < 0):
                    a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max_before - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            if verbose:
                print(f"    Newton round {rnd}: score={sc:.12f}, active={n_active}")
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break
        if step_done and gn < 1e-10 * n:
            break

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0, var_weight=0.01):
    """Adaptive log-sum-exp smoothing with L-BFGS."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    z = np.log(np.maximum(current_a, 1e-300))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.005 * scale
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n, var_weight)
            grad_fn = _SmoGradAdapter(beta, n, var_weight)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-300))
                z = z - z.mean()
        except Exception as e:
            if verbose if 'verbose' in dir() else False:
                print(f"  L-BFGS error: {e}")

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-7 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        elif new_gap < 1e-3 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _make_arcsine(n, eps=0.5):
    """Arcsine-like profile with tunable boundary cutoff."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(np.maximum(i + eps, 0.01) * np.maximum(n - 1 - i + eps, 0.01))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_beta_dist(n, alpha=0.5, beta=0.5):
    """Beta distribution profile."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = t ** (alpha - 1) * (1 - t) ** (beta - 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.7):
    t = np.linspace(0.01, 0.99, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asym_taper(n):
    t = np.linspace(0.01, 0.99, n)
    a = np.sin(np.pi * t ** 0.7) ** 0.8
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-300))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _run_cycle_optimize(a_init, deadline, min_time_remaining=60.0):
    """Alternate L-BFGS bursts with Newton polish."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    cycle_time = min(45.0, max(10.0, (deadline - min_time_remaining - time.monotonic()) / 8.0))

    for cycle in range(8):
        if time.monotonic() + min_time_remaining > deadline:
            break
        t_start_cycle = time.monotonic()

        # L-BFGS burst
        if time.monotonic() < deadline - min_time_remaining - 10:
            lb_deadline = min(t_start_cycle + cycle_time * 0.6, deadline - min_time_remaining)
            a_lb = _run_adaptive_lbfgs(current_a, lb_deadline, min_time_remaining=5.0,
                                        var_weight=0.005)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb

        # Newton polish
        if time.monotonic() < deadline - min_time_remaining - 5:
            np_deadline = min(t_start_cycle + cycle_time, deadline - min_time_remaining)
            a_np, sc_np = _newton_active_polish(current_a, np_deadline, max_rounds=30)
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np

        cycle_time *= 1.15

    return current_a

def _subgrad_polish(a_init, deadline, max_steps=2000):
    """Subgradient method with active set for robustness."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        step_size = lr / (1.0 + 0.001 * i)
        z = z - step_size * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.03)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-300))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.008

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search function. Minimizes 2*n*max(conv(a,a))/(sum(a))^2."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_arcsine(n_parent, eps=1.0)
        par_sc = _score(parent)
        print(f"No parent, using arcsine n={n_parent}, score={par_sc:.10f}")
        update_best(parent, par_sc, "arcsine_init")

    n_use = n_parent
    start_a = parent.copy()

    # Phase 1: Diverse initialization exploration (fast)
    print(f"\n[Phase 1] Initialization exploration (t={time.monotonic()-t_start:.1f}s)")
    candidates = []

    # Arcsine with various eps
    for eps in [0.5, 1.0, 2.0, 5.0, 10.0]:
        a_cand = _make_arcsine(n_use, eps=eps)
        sc_cand = _score(a_cand)
        candidates.append((sc_cand, a_cand, f"arcsine_eps{eps}"))
    
    # Beta distributions
    for alpha_beta in [(0.3, 0.3), (0.5, 0.5), (0.7, 0.7), (0.4, 0.6)]:
        a_cand = _make_beta_dist(n_use, alpha=alpha_beta[0], beta=alpha_beta[1])
        sc_cand = _score(a_cand)
        candidates.append((sc_cand, a_cand, f"beta_{alpha_beta}"))

    # Tapered
    for pw in [0.5, 0.7, 1.0, 1.5]:
        a_cand = _make_tapered(n_use, power=pw)
        sc_cand = _score(a_cand)
        candidates.append((sc_cand, a_cand, f"taper_{pw}"))

    # Asymmetric
    a_cand = _make_asym_taper(n_use)
    sc_cand = _score(a_cand)
    candidates.append((sc_cand, a_cand, "asym_taper"))

    # Parent perturbations
    for sig in [0.02, 0.05, 0.1]:
        a_cand = _perturb(parent, sig, rng)
        sc_cand = _score(a_cand)
        candidates.append((sc_cand, a_cand, f"perturb_{sig}"))

    # Sort and print top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Top 5 initializations:")
    for sc, a, label in candidates[:5]:
        print(f"    {label}: {sc:.10f}")

    # Quick L-BFGS on top 3 candidates
    quick_deadline = t_start + 200
    for i, (sc_c, a_c, label) in enumerate(candidates[:4]):
        if time.monotonic() > quick_deadline:
            break
        a_opt = _run_adaptive_lbfgs(a_c, min(quick_deadline, time.monotonic() + 40),
                                      min_time_remaining=50.0, var_weight=0.02)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"quick_{label}")

    # Phase 2: Main optimization from best
    print(f"\n[Phase 2] Main L-BFGS optimization (t={time.monotonic()-t_start:.1f}s)")
    start_a = best_seq.copy() if best_seq is not None else candidates[0][1]
    main_deadline = t_start + 800
    a_main = _run_adaptive_lbfgs(start_a, min(main_deadline, deadline),
                                   min_time_remaining=400.0, var_weight=0.01)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 3: Cycle optimization
    print(f"\n[Phase 3] Cycle optimization (t={time.monotonic()-t_start:.1f}s)")
    cycle_deadline = t_start + 1300
    a_cyc = _run_cycle_optimize(best_seq, min(cycle_deadline, deadline),
                                 min_time_remaining=300.0)
    update_best(a_cyc, _score(a_cyc), "cycle")

    # Phase 4: Newton polish
    print(f"\n[Phase 4] Newton active set polish (t={time.monotonic()-t_start:.1f}s)")
    np_deadline = t_start + 1600
    a_np, sc_np = _newton_active_polish(best_seq, min(np_deadline, deadline),
                                         max_rounds=200, verbose=True)
    update_best(a_np, sc_np, "newton_main")

    # Phase 5: Subgradient diversification
    print(f"\n[Phase 5] Subgradient + restart (t={time.monotonic()-t_start:.1f}s)")
    sg_deadline = t_start + 1700
    a_sg, sc_sg = _subgrad_polish(best_seq, min(sg_deadline, deadline), max_steps=1500)
    update_best(a_sg, sc_sg, "subgrad")

    if sc_sg < best_score + 0.005 and time.monotonic() < deadline - 50:
        a_np2, sc_np2 = _newton_active_polish(a_sg, min(deadline - 20, time.monotonic() + 40),
                                               max_rounds=50)
        update_best(a_np2, sc_np2, "newton_sg")

    # Phase 6: Try different n values
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_trials = [8192, 32768]
    for n_try in n_trials:
        if time.monotonic() > deadline - 40:
            break
        a_init = _resample(best_seq, n_try)
        # Quick optimize
        a_opt = _run_adaptive_lbfgs(a_init, min(deadline - 20, time.monotonic() + 30),
                                      min_time_remaining=10.0, var_weight=0.01)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 15:
            a_np3, sc_np3 = _newton_active_polish(a_opt, min(deadline - 5, time.monotonic() + 15),
                                                   max_rounds=30)
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # Phase 7: Final ultra-precise Newton
    print(f"\n[Phase 7] Final ultra-precise Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        # First, one more L-BFGS at tiny beta to get close to the true max
        n_fin = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        if time.monotonic() < deadline - 10:
            z = np.log(np.maximum(best_seq, 1e-300))
            z = z - z.mean()
            beta_fin = mb * 5e-9
            remaining = deadline - 15 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.02)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n_fin, 0.0)
                    grad_fn = _SmoGradAdapter(beta_fin, n_fin, 0.0)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Final Newton with very tight tolerances
        if time.monotonic() < deadline - 3:
            a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=300,
                                                   verbose=True)
            update_best(a_fin, sc_fin, "final_newton")

    # Final validation and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_arcsine(256, eps=1.0)
        sc_final = _score(final)
        # Verify with evaluate_sequence
        eval_sc = evaluate_sequence([float(x) for x in final])
        print(f"FINAL: internal_score={sc_final:.10f}, eval_score={eval_sc:.10f}, "
              f"n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_arcsine(8192, eps=1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
