    """Compute the balanced multi-peak gradient."""
    n = len(a)
    K = len(peak_indices)
    
    if K == 1:
        g = _grad_peak_fast(a, peak_indices[0])
        g = g - g.mean()
        return g

        G[:, idx] = _grad_peak_fast(a, s)
    
    # Project each column to sum=0
    G = G - G.mean(axis=0, keepdims=True)

    # This is: minimize w^T (G^T G) w s.t. 1^T w = 1, w >= 0
    # KKT: 2 G^T G w + lambda * 1 + mu = 0, mu >= 0, mu * w = 0
    # For simplicity, use unweighted (equal weights) as a starting point

    from scipy.optimize import minimize as _min
    def qf(w):
        return float(w @ (G.T @ G) @ w)
    def qf_grad(w):
        return 2 * (G.T @ G) @ w
    
    w0 = np.ones(K) / K
    constraints = [{'type': 'eq', 'fun': lambda w: w.sum() - 1}]
    bounds = [(0, 1)] * K
    
    try:

                   options={'maxiter': 100, 'ftol': 1e-12})

            w = res.x
        else:
            w = w0
    except:
        w = w0
    
    grad = G @ w
    return grad

    """
    Compute a balanced gradient that reduces multiple peaks simultaneously.
    Uses KKT-based weighting: find w minimizing ||sum w_s * grad_s||^2
    subject to sum(w) = 1, w >= 0, where grads are projected to sum=0.
    """

    K = len(peak_indices)

    if K == 1:
        g = _grad_peak_fast(a, peak_indices[0])
        return g - g.mean()

    G = np.zeros((n, K))

        A = g1n * g1n - 2.0 * dot + g2n * g2n

            top_idx = np.argsort(c)[::-1][:max_active]
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[top_idx] = True
            n_active = max_active

        peak_indices = np.where(active_mask)[0]
        peak_indices = [int(x) for x in peak_indices]

        # Compute balanced gradient
        grad = _balanced_multi_peak_grad(a, peak_indices)
        grad = grad - grad.mean()  # Project to sum=0

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Update momentum
        momentum = momentum_beta * momentum - lr * grad / gnorm
        momentum = momentum - momentum.mean()

        # Backtracking line search
        a_try = a + momentum
        a_try = np.maximum(a_try, 0.0)
        s_try = float(a_try.sum())
        improved = False

        step = 1.0
        for _ in range(40):
            a_try = a + step * momentum
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try without momentum (simple gradient step)
            step = lr
            for _ in range(30):
                a_try = a - step * grad / gnorm
                a_try = np.maximum(a_try, 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-10:
                    step *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    momentum = np.zeros(n)
                    improved = True
                    break
                step *= 0.5

            if not improved:
                # Stuck: adjust parameters
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
                momentum *= 0.3
                if no_improve >= max_stall:
                    a = best_a.copy()
                    momentum = np.zeros(n)
                    tol = max(tol_end, tol * 0.4)
                    lr = lr_start * 0.3
                    stall_count += 1
                    no_improve = 0
                    if stall_count > 15:
                        break
        else:
            if no_improve == 0:
                tol = max(tol_end, tol * 0.97)
                lr = min(lr_start, lr * 1.02)

    return best_a, best_sc


def _lse_softmax_obj(z, beta, n):
    a = z - np.max(z)
    e = np.exp(a)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n
    a = e / s
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e2 = np.exp(np.clip(shifted, -700, 0))
    s2 = e2.sum()
    if s2 < 1e-300 or not np.isfinite(s2):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s2))


def _lse_softmax_grad(z, beta, n):
    a = z - np.max(z)
    e = np.exp(a)
    s = e.sum()
    if s < 1e-300:
        return np.zeros_like(z)
    a = e / s
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e2 = np.exp(np.clip(shifted, -700, 0))
    s2 = e2.sum()
    if s2 < 1e-300 or not np.isfinite(s2):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e2 / s2
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, deadline, min_remaining=15.0):
    """L-BFGS with log-sum-exp softening and aggressive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 8192.0)
        est_time = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_time * 20))))

        def obj(z):
            return _lse_softmax_obj(z, beta, n)

        def grad(z):
            return _lse_softmax_grad(z, beta, n)

        try:
            result = _scipy_minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 60, 'maxls': 40}
            )
            z_new = result.x
            a_new = z_new - np.max(z_new)
            e = np.exp(a_new)
            s = e.sum()
            if s < 1e-300:
                continue
            a_new = e / s
            a_new = np.maximum(a_new, 0.0)
            s2 = a_new.sum()
            if s2 < 1e-15:
                continue
            a_new = a_new / s2
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_equalize(a, deadline, n_rounds=200):
    """
    Direct peak equalization: identify the top contributing pairs to the max peak
    and redistribute mass to flatten.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find top contributing pairs to c[k_star]
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            break

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            break

        # Identify top contributors (fraction of total)
        n_top = max(2, int(len(contrib) * 0.1))
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_i = i_range[top_local]
        top_j = j_range[top_local]

        # Amount to redistribute
        frac = rng.uniform(0.02, 0.08)
        a_try = a.copy()
        reduction = 0.0
        for ii in top_i:
            red = a_try[ii] * frac
            a_try[ii] -= red
            reduction += red
        # Also reduce the j-side if different
        for jj in top_j:
            if jj not in top_i:
                red = a_try[jj] * frac * 0.5
                a_try[jj] -= red
                reduction += red

        # Distribute to low-contributing positions
        low_mask = np.ones(n, dtype=bool)
        for ii in top_i:
            low_mask[ii] = False
        for jj in top_j:
            low_mask[jj] = False
        # Also exclude positions that are already very small
        low_mask &= (a > a.mean() * 0.3)

        if low_mask.sum() < 10:
            low_mask = np.ones(n, dtype=bool)
            for ii in np.concatenate([top_i, top_j]):
                low_mask[ii] = False
            if low_mask.sum() < 5:
                continue

        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = float(a_try.sum())
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.abs(t - 0.5)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n, dtype=float)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.arange(n, dtype=float)
        a = 1.0 / (1.0 + np.exp(-8.0 * (t - n * 0.5) / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return a.copy()
    return e / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: L-BFGS soft-max on parent
    P2: Multi-peak active-set gradient with momentum
    P3: Peak equalization (direct mass redistribution)
    P4: Follow-up L-BFGS + polish
    P5: Multi-n exploration
    P6: Diverse restarts
    P7: Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent: uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS on parent ===
    print(f"\n[P1] L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p1_end = min(deadline - 700.0, t_start + 250.0)
    if time.monotonic() < p1_end - 10:
        a1 = _lbfgs_run(best_a, p1_end, min_remaining=15.0)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_a = a1.copy()
            best_score = sc1
            print(f"  P1: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")
        else:
            print(f"  P1: no improvement, still {best_score:.10f}")

    # === Phase 2: Multi-peak active-set descent ===
    print(f"\n[P2] Multi-peak descent (t={time.monotonic()-t_start:.1f}s)")
    p2_end = min(deadline - 450.0, t_start + 550.0)
    a2, sc2 = _active_set_descent(best_a, p2_end, tol_start=0.005, tol_end=0.00001,
                                   lr_start=0.04, max_active=10, momentum_beta=0.3)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  P2: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 3: Peak equalization ===
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    p3_end = min(deadline - 350.0, t_start + 700.0)
    a3, sc3 = _peak_equalize(best_a, p3_end, n_rounds=200)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  P3: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # Follow-up optimization after peak equalization
    if time.monotonic() < deadline - 250.0:
        p3b_end = min(deadline - 250.0, t_start + 800.0)
        a3b = _lbfgs_run(best_a, p3b_end, min_remaining=15.0)
        sc3b = _score(a3b)
        if sc3b < best_score:
            best_a = a3b.copy()
            best_score = sc3b
            print(f"  P3b L-BFGS: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")
        # Quick active-set polish
        if time.monotonic() < deadline - 200.0:
            p3c_end = min(deadline - 200.0, t_start + 870.0)
            a3c, sc3c = _active_set_descent(best_a, p3c_end, tol_start=0.002, tol_end=0.000005,
                                             lr_start=0.02, max_active=8, momentum_beta=0.2)
            if sc3c < best_score:
                best_a = a3c.copy()
                best_score = sc3c
                print(f"  P3c polish: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_targets = [int(n_best * 1.25), int(n_best * 0.75), 16384]
    n_targets = [x for x in n_targets if 2048 <= x <= 32768 and x != n_best]
    n_targets = sorted(set(n_targets))[:3]

    for n_try in n_targets:
        if time.monotonic() > deadline - 120.0:
            break
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        remaining = deadline - time.monotonic() - 80.0
        if remaining < 30:
            continue
        phase_end = time.monotonic() + min(remaining * 0.6, 120.0)
        a_opt = _lbfgs_run(a_init, phase_end, min_remaining=10.0)
        sc_opt = _score(a_opt)
        if time.monotonic() < deadline - 40.0:
            a_opt2, sc_opt2 = _active_set_descent(a_opt, deadline - 40.0,
                                                   tol_start=0.002, tol_end=0.00001,
                                                   lr_start=0.02, max_active=8)
            if sc_opt2 < sc_opt:
                a_opt = a_opt2
                sc_opt = sc_opt2
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST n={n_try}: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 5: Diverse restarts ===
    print(f"\n[P5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60.0:
        n_r = len(best_a)
        restarts = [
            ("pert_small", _perturb(best_a, 0.015, rng)),
            ("pert_med", _perturb(best_a, 0.04, rng)),
            ("tapered", _make_profile(n_r, "tapered")),
            ("edge_heavy", _make_profile(n_r, "edge_heavy")),
            ("cos_mod", _make_profile(n_r, "cos_mod")),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 20:
                break
            phase_end = time.monotonic() + min(remaining * 0.5, 60.0)
            a_opt = _lbfgs_run(a_r, phase_end, min_remaining=5.0)
            sc_opt = _score(a_opt)
            if time.monotonic() < deadline - 25.0:
                a_opt2, sc_opt2 = _active_set_descent(a_opt, deadline - 25.0,
                                                       tol_start=0.001, tol_end=0.00001,
                                                       lr_start=0.01, max_active=5)
                if sc_opt2 < sc_opt:
                    a_opt = a_opt2
                    sc_opt = sc_opt2
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  Restart '{label}': {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 6: Final ultra-fine polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_descent(best_a, deadline - 2.0,
                                             tol_start=0.0002, tol_end=0.000001,
                                             lr_start=0.005, max_active=5, momentum_beta=0.1)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  Final polish: {best_score:.10f}")

        # One more peak equalize pass
        if time.monotonic() < deadline - 2.0:
            a_ps, sc_ps = _peak_equalize(best_a, deadline - 1.0, n_rounds=30)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  Final peak eq: {best_score:.10f}")

        # Tiny L-BFGS
        if time.monotonic() < deadline - 3.0:
            remaining = deadline - 3.0 - time.monotonic()
            if remaining > 5.0:
                n = len(best_a)
                c_best = _full_convolve(best_a, best_a)
                mb = float(c_best.max())
                beta_fin = mb * 0.00005
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()

                def obj_final(z):
                    return _lse_softmax_obj(z, beta_fin, n)

                def grad_final(z):
                    return _lse_softmax_grad(z, beta_fin, n)

                max_iter = min(500, max(10, int(remaining / 0.05)))
                try:
                    result = _scipy_minimize(obj_final, z, jac=grad_final, method='L-BFGS-B',
                                             options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 40})
                    a_fin2 = result.x
                    a_fin2 = a_fin2 - np.max(a_fin2)
                    e = np.exp(a_fin2)
                    s = e.sum()
                    if s > 1e-300:
                        a_fin2 = e / s
                        a_fin2 = np.maximum(a_fin2, 0.0)
                        s2 = a_fin2.sum()
                        if s2 > 1e-10:
                            a_fin2 = a_fin2 / s2
                            sc_fin2 = _score(a_fin2)
                            if sc_fin2 < best_score:
                                best_a = a_fin2.copy()
                                best_score = sc_fin2
                                print(f"  Final LBFGS: {best_score:.10f}")
                except Exception:
                    pass

    # Final validation and return
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
