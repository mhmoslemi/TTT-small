def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _hankel_row_sums(W, n):
    """Compute row sums of Hankel matrix H with (H)_{ij} = W[i+j]."""
    # Row i has entries W[i], W[i+1], ..., W[i+n-1]
    # row_sum_i = sum_{j=0}^{n-1} W[i+j] = prefix_sum_W[i+n] - prefix_sum_W[i]
    W_ext = np.zeros(2 * n)
    W_ext[:len(W)] = W
    ps = np.cumsum(W_ext)
    row_sums = ps[n:2 * n] - ps[:n]
    return row_sums


def _pcg_solve(W, n, rhs, x0=None, max_iter=2000, tol=1e-14, reg=0.0):
    """Preconditioned Conjugate Gradient solve for (H + reg*I)x = rhs."""

    else:
        x = x0.copy()

    rz0 = rz

def _cg_solve(W, n, rhs, x0=None, max_iter=2000, tol=1e-14, reg=1e-12):

    rs0 = rs

        return x
    for i in range(max_iter):

        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp) or pHp < 0:

        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)

        x = x_new
        r = r_new
        rs = rs_new

        # Also ensure top-n_top are included
        top_idx = np.argsort(c)[-n_top:]
        active[top_idx] = True
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve QP
        a_qp = _solve_qp_active(W, n, a, max_outer=100, max_cg_iter=1500)
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-15:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Damped update
            alpha = 0.5
            a_try = (1 - alpha) * a + alpha * a_qp
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st > 1e-15:
                a_try = a_try / st
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                a = a_try

        # Periodic report
        if (rnd + 1) % 50 == 0:
            print(f"    IR round {rnd+1}/{total_rounds}: score={best_sc:.10f}, "
                  f"n_active={n_active}, tol={tol:.2e}")

    return best_a, best_sc


def _smoobj(z, beta, n, top_k=100):
    """Smoothed objective: 2n * (c_max + beta * log(sum exp((c-c_max)/beta))) with top-k focus."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    """Gradient of smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _lbfgs_optimize(a_init, deadline, min_time=30.0, beta_schedule=None):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.001)

    if beta_schedule is None:
        ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                  0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    else:
        ratios = beta_schedule

    betas = [base_beta * r for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        # Estimate max iterations based on time
        t_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(30, int(remaining / (t_per_iter * 5))))

        try:
            res = minimize(
                lambda zl: _smoobj(zl, beta, n),
                z,
                jac=lambda zl: _smograd(zl, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 100}
            )
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 - 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "prolate":
        # Approximate prolate spheroidal sequence (concentrated in time and freq)
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.15 * 0.15))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin_pow":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    """Resample sequence to new length via interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _kkt_residual(a, W, n):
    """Compute KKT residual for the QP."""
    grad = _hankel_matvec(a, W, n)
    support = a > 1e-10
    if support.sum() < 2:
        return np.inf
    lam = np.mean(grad[support])
    resid_on = np.max(np.abs(grad[support] - lam))
    off = ~support
    if off.any():
        resid_off = np.max(np.maximum(0, lam - grad[off]))
    else:
        resid_off = 0.0
    return max(resid_on, resid_off)


def _direct_simplex_step(a, W, n, gamma, max_inner=50):
    """Direct projected gradient step on the simplex."""
    grad = 2.0 * _hankel_matvec(a, W, n)
    # Backtracking line search
    obj_cur = float(a @ _hankel_matvec(a, W, n))
    for ls in range(max_inner):
        a_try = a - gamma * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            gamma *= 0.5
            continue
        a_try = a_try / s
        # Check sum constraint more carefully: project onto {a>=0, sum=1}
        # The above is not exact simplex projection, but a good approximation
        obj_try = float(a_try @ _hankel_matvec(a_try, W, n))
        if obj_try < obj_cur:
            return a_try
        gamma *= 0.5
    return a


def _simplex_project(v):
    """Exact Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not cond.any():
        return np.ones(n) / n
    i = np.where(cond)[0][-1]
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)
