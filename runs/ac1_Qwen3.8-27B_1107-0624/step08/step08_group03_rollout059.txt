3. L-BFGS with log-sum-exp smoothing for coarse optimization
4. High-precision active-set Newton with FFT-based Hessian matvec
5. Subgradient polishing for final precision
6. Adaptive active-set management with proper KKT verification
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[50]
    elif len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001,
                                      0.0005, 0.0002, 0.0001, 0.00005,
                                      0.00002, 0.00001, 0.000005])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        if n <= 4096:
            est_time_per_iter = 0.002
        elif n <= 8192:
            est_time_per_iter = 0.005
        elif n <= 16384:
            est_time_per_iter = 0.012
        elif n <= 32768:
            est_time_per_iter = 0.04
        else:
            est_time_per_iter = 0.08

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 12))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_newton_v2(a_init, deadline, max_iter=500, tol_active=0.001, verbose=False):
    """High-precision active-set Newton with FFT-based Hessian matvec."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > int(2.5 * n):
            top_idx = np.argsort(c)[::-1][:int(2.5 * n)]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = int(active.sum())

        # Build lambda vector (uniform weights on active set for initial estimate)
        lam = np.zeros(2 * n - 1)
        lam[active] = 1.0 / n_active

        # Hessian matvec using FFT: (Hv)[j] = 2 * sum_k lam[k] * v[k-j]
        # This is 2 * cross_correlation(lam, v)
        N_fft = _next_pow2(2 * n)
        flam = np.fft.rfft(lam, N_fft)

        def hess_matvec(v):
            fv = np.fft.rfft(v, N_fft)
            # cross-correlation: IFFT(conj(flam) * fv) gives (lam ⊛ v)[j] = sum_k lam[k]*v[k-j]
            corr = np.fft.irfft(np.conj(flam) * fv, N_fft)
            return 2.0 * corr[:n]

        # Gradient of sum_k lam[k]*conv(a,a)[k] w.r.t. a = 2 * sum_k lam[k]*a[k-j]
        fa = np.fft.rfft(a, N_fft)
        grad = np.fft.irfft(np.conj(flam) * fa, N_fft)[:n] * 2.0

        # Project gradient onto the subspace sum(dx) = 0 (for maintaining sum=1)
        g_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # CG solve with regularization for stability
        mu = 1e-8 * c_max / max(n_active, 1)

        def reg_hess_matvec(v):
            return hess_matvec(v) + mu * v

        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - reg_hess_matvec(x)
            p = r.copy()
            rs = float(np.dot(r, r))
            for cg_it in range(max_cg):
                Ap = reg_hess_matvec(p)
                pAp = float(np.dot(p, Ap))
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = float(np.dot(r_new, r_new))
                if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n) / n)
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam_val = float(u.sum()) / sum_w
            dx = -u + lam_val * w
            dx = dx - np.mean(dx)

        # Line search with non-negativity constraint
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            # Wolfe-like condition
            if f_try < f_old - 1e-13 * t * gnorm:
                a = a_try
                accepted = True
                break
            elif f_try <= f_old + 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Fallback: steepest descent
            t = 0.01
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            if verbose:
                print(f"    AS iter {it}: score={sc:.12f}, t={t:.6f}, n_act={n_active}")

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _subgrad_polish_v2(a_init, deadline, n_max_steps=10000, lr_init=0.02):
    """High-precision subgradient method with adaptive learning rate and top-k tracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_lr = lr_init

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive top-k based on how spread out the peaks are
        tol_frac = 0.005
        active = c >= (c_max - tol_frac * c_max)
        n_active = max(1, int(active.sum()))
        # Cap the active set size
        if n_active > 100:
            top_idx = np.argsort(c)[::-1][:100]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 100

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0001 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Evaluate periodically
        check_interval = max(1, n_max_steps // 500)
        if i % check_interval == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, best_lr)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = best_lr * 0.5
                    best_lr = lr

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None, alpha=0.5):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
        return a / a.sum()
    elif kind == "u_shape_strong":
        t = np.linspace(0, 1, n)
        a = 1.0 + alpha * (2.0 * t - 1.0) ** 4
        return a / a.sum()
    elif kind == "cosine_u":
        t = np.linspace(0, 1, n)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "bimodal_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.1) ** 2) / 0.04) + np.exp(-((t - 0.9) ** 2) / 0.04) + 0.1
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.standard_normal(n)
        k = max(2, n // 80)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent_peaks(a_init, deadline, max_rounds=200):
    """Directly target the top peaks using coordinate descent on the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find the single worst peak
        k_peak = int(np.argmax(c))

        # Gradient of conv(a,a)[k_peak] w.r.t. a
        # grad[j] = 2*a[k_peak - j] if 0 <= k_peak - j < n
        grad = np.zeros(n)
        lo = max(0, k_peak - (n - 1))
        hi = min(n - 1, k_peak)
        for j in range(lo, hi + 1):
            idx = k_peak - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        # Project to maintain sum=1
        grad = grad - np.mean(grad)

        # Line search: reduce the peak
        f_old = c_max
        step = 0.05 / (1.0 + 0.01 * rnd)
        for ls in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5
        else:
            break

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _multi_peak_chase(a_init, deadline, max_rounds=300):
    """Chase the top few peaks simultaneously with adaptive weights."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    n_top = min(5, 2 * n - 1)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Get top peaks with weights proportional to excess
        top_idx = np.argsort(c)[::-1][:n_top]
        # Weight by how much above the median each peak is
        c_med = np.median(c)
        weights = np.maximum(c[top_idx] - c_med, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(n_top)
        weights = weights / weights.sum()

        # Combined gradient
        grad = np.zeros(n)
        for ii, k_peak in enumerate(top_idx):
            lo = max(0, k_peak - (n - 1))
            hi = min(n - 1, k_peak)
            for j in range(lo, hi + 1):
                idx = k_peak - j
                if 0 <= idx < n:
                    grad[j] += weights[ii] * 2.0 * a[idx]

        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        f_old = c_max
        step = 0.03 / (1.0 + 0.005 * rnd)
        for ls in range(30):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer combining L-BFGS, active-set Newton, and subgradient methods.
    Explores multiple n values and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "u_shape", alpha=0.3)
        print(f"No parent, using U-shape n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS refinement on parent ===
    print(f"\n[Phase 1] L-BFGS on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_p1 = _run_lbfgs(parent, deadline - 400, min_time_remaining=300.0, aggressive=True)
        update_best(a_p1, _score(a_p1), "lbfgs_parent")

    # === Phase 2: Try larger n (32768) ===
    print(f"\n[Phase 2] Larger n exploration (t={time.monotonic()-t_start:.1f}s)")
    for n_try in [2 * n_parent, 32768]:
        if time.monotonic() > deadline - 200:
            break
        if n_try <= n_parent:
            continue
        print(f"  Trying n={n_try}")
        a_large_init = _interpolate_to_n(best_seq if best_seq is not None else parent, n_try)
        sc_init = _score(a_large_init)
        print(f"    init score: {sc_init:.10f}")

        remaining_for_lbfgs = min(200, deadline - 100 - time.monotonic())
        if remaining_for_lbfgs > 30:
            a_large = _run_lbfgs(a_large_init, t_start + time.monotonic() - t_start + remaining_for_lbfgs,
                                 min_time_remaining=20.0, aggressive=True)
            update_best(a_large, _score(a_large), f"lbfgs_n{n_try}")

    # === Phase 3: U-shaped initializations at parent n ===
    print(f"\n[Phase 3] U-shaped initializations (t={time.monotonic()-t_start:.1f}s)")
    u_profiles = [
        ("u_03", "u_shape", 0.3),
        ("u_05", "u_shape", 0.5),
        ("u_08", "u_shape", 0.8),
        ("u4_03", "u_shape_strong", 0.3),
        ("u4_06", "u_shape_strong", 0.6),
        ("cos_03", "cosine_u", 0.3),
        ("bimodal", "bimodal_gauss", 1.0),
    ]
    for label, kind, alpha in u_profiles:
        if time.monotonic() > deadline - 80:
            break
        a_u = _make_profile(n_parent, kind, rng, alpha)
        sc_u = _score(a_u)
        if sc_u < best_score + 0.05:
            print(f"  {label}: init score={sc_u:.10f} -> optimizing")
            a_u_opt = _run_lbfgs(a_u, deadline, min_time_remaining=30.0, aggressive=True)
            update_best(a_u_opt, _score(a_u_opt), f"opt_{label}")
        else:
            print(f"  {label}: init score={sc_u:.10f} (skipped, too far)")

    # === Phase 4: Active-set Newton (main refinement) ===
    print(f"\n[Phase 4] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        # Medium tolerance first
        a_as1, sc_as1 = _active_set_newton_v2(best_seq, min(deadline - 40, time.monotonic() + 60),
                                              max_iter=200, tol_active=0.002)
        update_best(a_as1, sc_as1, "as_tight_medium")

        # Tight tolerance
        if time.monotonic() < deadline - 30:
            a_as2, sc_as2 = _active_set_newton_v2(best_seq, min(deadline - 20, time.monotonic() + 40),
                                                  max_iter=300, tol_active=0.0005)
            update_best(a_as2, sc_as2, "as_tight")

        # Ultra-tight
        if time.monotonic() < deadline - 15:
            a_as3, sc_as3 = _active_set_newton_v2(best_seq, min(deadline - 10, time.monotonic() + 20),
                                                  max_iter=200, tol_active=0.0001)
            update_best(a_as3, sc_as3, "as_ultra")

    # === Phase 5: Multi-peak chase ===
    print(f"\n[Phase 5] Multi-peak chase (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_mp, sc_mp = _multi_peak_chase(best_seq, min(deadline - 20, time.monotonic() + 30), max_rounds=200)
        update_best(a_mp, sc_mp, "multi_peak")

    # === Phase 6: Subgradient polish ===
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sg, sc_sg = _subgrad_polish_v2(best_seq, min(deadline - 10, time.monotonic() + 40),
                                         n_max_steps=5000, lr_init=0.01)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 7: Final active-set on best ===
    print(f"\n[Phase 7] Final active-set (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_newton_v2(best_seq, deadline - 2, max_iter=300, tol_active=0.0001)
        update_best(a_fin, sc_fin, "final_as")

    # === Phase 8: Coordinate descent on peaks ===
    print(f"\n[Phase 8] Coordinate descent peaks (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_cd, sc_cd = _coordinate_descent_peaks(best_seq, deadline - 1, max_rounds=100)
        update_best(a_cd, sc_cd, "coord_desc")

    # === Phase 9: Diverse restarts ===
    print(f"\n[Phase 9] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        for sigma in [0.01, 0.02, 0.05]:
            if time.monotonic() > deadline - 5:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            if sc_start < best_score + 0.005:
                a_r = _run_lbfgs(p_start, deadline, min_time_remaining=3.0)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"restart_{sigma}")
                if time.monotonic() < deadline - 3:
                    a_r2, sc_r2 = _active_set_newton_v2(a_r, deadline - 1, max_iter=50, tol_active=0.001)
                    update_best(a_r2, sc_r2, f"restart_{sigma}_as")

    # === Phase 10: Final subgrad + AS ===
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_f1, sc_f1 = _subgrad_polish_v2(best_seq, deadline - 1, n_max_steps=3000, lr_init=0.005)
        update_best(a_f1, sc_f1, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(1024, "u_shape", alpha=0.5)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
