import numpy as np

from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_matvec(v, A_vec, n, m, mu):
    """Compute Hessian-vector product for active-set subproblem."""
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fv = np.fft.rfft(v[::-1], N)
    fA = np.fft.rfft(A_vec, N)
    res = np.fft.irfft(fv * fA, N)[:out_len]
    return (2.0 / m) * res[n - 1:2 * n - 1] + mu * v


def _cg_solve(rhs, A_vec, n, m, mu, max_cg=300):
    """Conjugate gradient solve for (H+mu*I) x = rhs."""
    x = np.zeros(n)
    Ap0 = _hess_matvec(x, A_vec, n, m, mu)
    r = rhs - Ap0
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = _hess_matvec(p, A_vec, n, m, mu)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < 1e-12 * (1 + rhs_norm):
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)

        if gnorm < 1e-14:

            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * t * gnorm or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break

        c_new = _full_convolve(a, a)

def _run_adam(a_init, deadline, lr_init=0.01, max_steps=50000):
    """Adam optimizer in log-space with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    # Schedule: start with larger beta, decrease over time
    beta_schedule_steps = max_steps // 5
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    step_counter = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Adaptive beta: decrease over time
        frac = step / max_steps
        if frac < 0.2:
            beta = base_beta
        elif frac < 0.4:
            beta = base_beta * 0.2
        elif frac < 0.6:
            beta = base_beta * 0.05
        elif frac < 0.8:
            beta = base_beta * 0.01
        else:
            beta = base_beta * 0.002

        a = _softmax(z)
        grad = _lsgrad(z, beta, n)
        step_counter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step_counter)
        v_hat = v / (1 - beta2 ** step_counter)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Periodic evaluation
        if step % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < current_sc - 1e-14:
                current_a = a_check.copy()
                current_sc = sc
            else:
                lr *= 0.95
                if lr < 1e-6:
                    z = np.log(np.maximum(current_a, 1e-30))
                    z = z - z.mean()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < current_sc:
        current_a = a_final
        current_sc = sc_final
    return current_a


def _build_conv_matrix(a, n):
    """Build the convolution matrix A where A[k,j] = a[k-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        length = min(n, 2 * n - 1 - j)
        A[start:start + length, j] = a[:length]
    return A


def _lp_best_response(a, deadline, max_time_per_lp=10.0):
    """Solve the LP: minimize t s.t. conv(a,b)[k] <= t, sum(b)=1, b>=0."""
    n = len(a)
    # Build constraint matrix
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        length = min(n, 2 * n - 1 - j)
        A_mat[start:start + length, j] = a[:length]

    # LP: min t s.t. A_mat @ b - t <= 0, sum(b) = 1, b >= 0, t >= 0
    # Variables: [b (n), t (1)]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t

    # A_ub @ x <= b_ub: A_mat @ b - t <= 0
    A_ub = np.hstack([A_mat, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # A_eq @ x = b_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b_opt = res.x[:n]
            b_opt = np.maximum(b_opt, 0)
            s = b_opt.sum()
            if s > 1e-15:
                b_opt = b_opt / s
                return b_opt, res.x[-1]
    except Exception:
        pass

    return None, None


def _lp_alternate(a_init, deadline, n_lp, max_rounds=200):
    """LP-based alternating scheme for finding good sequences."""
    # Start from a_init, truncated or interpolated to n_lp
    if len(a_init) == n_lp:
        a = a_init.copy()
    else:
        x_old = np.linspace(0, 1, len(a_init))
        x_new = np.linspace(0, 1, n_lp)
        a = np.interp(x_new, x_old, a_init)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n_lp) / n_lp
        else:
            a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    prev_t = None

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b_res, t_val = _lp_best_response(a, deadline)
        if b_res is None:
            # LP failed, try a gradient step instead
            break

        # Damped update
        alpha = 0.5
        a_new = (1 - alpha) * a + alpha * b_res
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        # Check convergence
        if prev_t is not None and abs(t_val - prev_t) < 1e-10:
            break
        prev_t = t_val

        a = a_new

    # Also polish with active-set Newton
    a_polished, sc_polished = _active_set_newton(best_a, min(deadline, time.monotonic() + 30), max_iter=50, tol_active=0.001)
    if sc_polished < best_sc:
        best_a = a_polished
        best_sc = sc_polished

    return best_a, best_sc


def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "ramp_steep":
        t = np.linspace(0, 1, n)
        a = 0.1 + 5.0 * t
        return a / a.sum()
    elif kind == "plateau_low_high":
        a = np.ones(n)
        a[n // 2:] *= 2.0
        return a / a.sum()
    elif kind == "plateau_high_low":
        a = np.ones(n)
        a[:n // 2] *= 2.0
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic_ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + 2.0 * t ** 2
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.25) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 0.5 + 3.0 * (2 * t - 1) ** 2
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * np.exp(-((t - 0.3) ** 2) / 0.05)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.standard_normal(n)
        k = max(2, n // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n

    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.88
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _peak_refine(a_init, deadline, max_rounds=50):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        n_peaks = min(10, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]

        w = np.zeros(2 * n - 1)
        for k in top_peaks:
            w[k] = 1.0 / n_peaks

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)

        f_old = c_max
        for ls in range(30):
            step = 0.1 / (1.0 + 0.01 * rnd)
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _try_reverse_shift(a, deadline):
    """Try reversing and shifting the sequence to find better configurations."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    # Try reverse
    a_rev = a[::-1].copy()
    sc_rev = _score(a_rev)
    if sc_rev < best_sc:
        best_a = a_rev
        best_sc = sc_rev

    # Try a few circular shifts (zero-pad at ends)
    for shift in [1, 2, 3, 5, 10]:
        if shift >= n:
            break
        a_shift = np.zeros(n)
        a_shift[shift:] = a[:-shift]
        s = a_shift.sum()
        if s < 1e-15:
            continue
        a_shift = a_shift / s
        sc_sh = _score(a_shift)
        if sc_sh < best_sc:
            best_a = a_shift
            best_sc = sc_sh

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, Adam, active-set Newton,
    LP-based alternating, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except NameError:
        pass
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS from parent
    print(f"\n[Phase 1] Aggressive L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_main = _run_lbfgs(parent, deadline - 500, min_time_remaining=300.0, aggressive=True)
        update_best(a_main, _score(a_main), "lbfgs_parent")

    # Phase 2: Adam optimization
    print(f"\n[Phase 2] Adam optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_adam = _run_adam(best_seq, min(deadline - 150, time.monotonic() + 120), lr_init=0.005, max_steps=30000)
        update_best(a_adam, _score(a_adam), "adam_1")

    # Phase 3: Active-set Newton
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 60, time.monotonic() + 60), max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "active_set")
        if time.monotonic() < deadline - 30:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 15, time.monotonic() + 30), max_iter=100, tol_active=0.0003)
            update_best(a_as2, sc_as2, "active_set_tight")

    # Phase 4: LP-based alternating for smaller n
    print(f"\n[Phase 4] LP-based alternating (t={time.monotonic() - t_start:.1f}s)")
    for n_lp in [1024, 2048]:
        if time.monotonic() > deadline - 80:
            break
        print(f"  Trying n_lp={n_lp}...")
        a_lp, sc_lp = _lp_alternate(best_seq, min(deadline - 40, time.monotonic() + 60), n_lp, max_rounds=50)
        update_best(a_lp, sc_lp, f"lp_n{n_lp}")

        # Upsample to parent size and polish
        if a_lp is not None and sc_lp < best_score + 0.05 and time.monotonic() < deadline - 60:
            a_up = _interpolate_to_n(a_lp, n_parent)
            a_up_opt = _run_lbfgs(a_up, deadline, min_time_remaining=30.0, aggressive=True)
            update_best(a_up_opt, _score(a_up_opt), f"lp_up_n{n_lp}")

    # Phase 5: Structured initializations
    print(f"\n[Phase 5] Structured initializations (t={time.monotonic() - t_start:.1f}s)")
    structured_profiles = ["ramp_steep", "u_shape", "asym_gauss", "exp_ramp", "quadratic_ramp"]
    for kind in structured_profiles:
        if time.monotonic() > deadline - 60:
            break
        a_struct = _make_profile(n_parent, kind, rng)
        sc_struct = _score(a_struct)
        print(f"  {kind}: score={sc_struct:.10f}")
        if sc_struct < best_score + 0.1:
            a_opt = _run_lbfgs(a_struct, deadline, min_time_remaining=30.0, aggressive=True)
            update_best(a_opt, _score(a_opt), f"struct_{kind}")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        restart_configs = [0.01, 0.03, 0.05, 0.10, 0.15, 0.25, 0.50]
        for sigma in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0, aggressive=(sigma > 0.1))
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_s{sigma}")

            # Quick active-set polish on good restarts
            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 20), max_iter=50, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_s{sigma}_as")

    # Phase 7: Peak refinement
    print(f"\n[Phase 7] Peak refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_pt, sc_pt = _peak_refine(best_seq, min(deadline - 15, time.monotonic() + 30), max_rounds=40)
        update_best(a_pt, sc_pt, "peak_refine")

    # Phase 8: Try reverse/shift
    print(f"\n[Phase 8] Reverse/shift (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_rs, sc_rs = _try_reverse_shift(best_seq, deadline)
        update_best(a_rs, sc_rs, "reverse_shift")

    # Phase 9: Final subgradient polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 3, n_max_steps=3000, lr_init=0.002)
        update_best(a_fin, sc_fin, "final_sg")

    # Phase 10: Final active-set with very tight tolerance
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin2, sc_fin2 = _active_set_newton(best_seq, deadline - 1, max_iter=100, tol_active=0.0001)
        update_best(a_fin2, sc_fin2, "final_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
