def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _convolve_grad(a, v):
    """
    Compute gradient of sum_k v_k * c_k where c = a * a (linear convolution).
    Returns a vector of length len(a).
    Uses the identity: grad_k = (v_flip * a)[k + 2n - 2]
    where v_flip = v[::-1] and * is linear convolution.
    """
    n = len(a)
    # v has length 2n-1
    v_flip = v[::-1]  # length 2n-1
    # Linear convolution of v_flip (length 2n-1) and a (length n) has length 3n-2
    N = _next_pow2(3 * n)
    fv = np.fft.rfft(v_flip, N)
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fv * fa, N)
    # We want elements at indices 2n-2 through 3n-3
    return conv[2*n-2 : 3*n-2]

def _irls_grad(a, p, eps):
    """
    Compute gradient of sum_k w_k * c_k^2 where w_k = 1/(c_k + eps)^p and c = a * a.
    This equals 2 * sum_k w_k * c_k * d(c_k)/d(a)
    = 2 * convolve_grad(a, w * c)

    n = len(a)
    c = _full_convolve(a, a)
    w = 1.0 / (c + eps) ** p
    v = w * c  # = 1/(c + eps)^(p-1)
    grad = 2.0 * _convolve_grad(a, v)
    return grad, c

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    grad_a = _convolve_grad(a, w)
    # Chain rule through softmax
    return a * (grad_a - np.dot(a, grad_a))

    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)

    c = _full_convolve(a, a)

    cp = np.power(c_safe, p)

    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0)
    return a * (factor * grad_a - np.dot(a, factor * grad_a))
