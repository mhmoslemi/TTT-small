7. **Projected subgradient/gradient method**:

"""
KKT fixed-point iteration with annealing, combined with L-BFGS smoothing,
for minimizing the peak autocorrelation of non-negative sequences.
Uses FFT-accelerated Hankel matvec and CG for the KKT linear system.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]

def _hankel_matvec_reg(v, W, n, reg):
    return _hankel_matvec(v, W, n) + reg * v

def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-12, reg=0.0):
    """Solve (H_W + reg*I) * x = rhs using CG with FFT Hankel matvec."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        if reg > 0:
            Hp = _hankel_matvec(p, W, n) + reg * p
        else:
            Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            # Reset (restart CG)
            x = x_new
            r = r_new
            p = r.copy()
            rs = float(r @ r)
            if rs < tol * tol * rs0:
                break
            continue
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _kkt_fixed_point(a_init, deadline, max_rounds=200, beta_start=None, beta_end=None, 
                      damping_start=0.3, damping_end=1.0, reg=1e-10):
    """
    KKT fixed-point iteration with annealing.
    At each step: compute softmax weights from autocorrelation, solve H_W * x = 1, update a.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Initial beta
    c = _full_convolve(a, a)
    c_range = c.max() - np.percentile(c, 50)
    if beta_start is None:
        beta_start = c_range * 2.0
    if beta_end is None:
        beta_end = c_range * 0.001
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Annealing schedule (geometric)
        t_frac = rnd / max(1, max_rounds - 1)
        log_beta = np.log(beta_start) + t_frac * (np.log(beta_end) - np.log(beta_start))
        beta = np.exp(log_beta)
        
        # Compute softmax weights
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e
        
        # Solve H_W * x = 1
        rhs = np.ones(n)
        x = _cg_solve_hankel(w, n, rhs, max_iter=300, tol=1e-10, reg=reg)
        
        # Normalize and project
        x = np.maximum(x, 0)
        s_x = x.sum()
        if s_x < 1e-15:
            break
        x = x / s_x
        
        # Damped update
        alpha = damping_start + (damping_end - damping_start) * t_frac
        a_new = (1 - alpha) * a + alpha * x
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
              0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
              0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
              0.000001, 0.0000005, 0.0000002, 0.0000001]
    betas = [base_beta * r for r in ratios]
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))
        
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _flatten_polish(a_init, deadline, max_iters=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_target = np.sum(c) / len(c)
        c_max = float(c.max())
        
        excess = np.maximum(c - c_target, 0)
        excess_max = excess.max()
        if excess_max < 1e-15:
            break
        
        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]
        
        step = 0.1 / (1.0 + 0.01 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a + 0.5 * (a_new - a)
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new

**KKT fixed-point, refined**:
At the optimum, there exist weights w ≥ 0, sum(w) = 1, supported on argmax(c) = {m : c_m = max(c)}, such that:
2 * sum_m w_m * a_{m-i} = λ  for all i with a_i > 0
2 * sum_m w_m * a_{m-i} ≤ λ  for all i with a_i = 0

If a has full support (all a_i > 0), this becomes:
sum_m w_m * a_{m-i} = λ/2  for all i = 0, ..., n-1

This is: (W * a) = (λ/2) * 1, where W is the weight vector and * denotes the convolution (with the specific indexing).

In matrix form: H_W * a = (λ/2) * 1, where (H_W)_{i,j} = w_{i+j}.

Solving: a = (λ/2) * H_W^{-1} * 1, and λ is determined by sum(a) = 1.

So: a = H_W^{-1} * 1 / (1^T H_W^{-1} * 1)

The algorithm:
1. Start with initial a (e.g., tapered).
2. Compute c = a * a.
3. Compute w = softmax(c / β) (smoothed indicator of argmax).
4. Solve H_W * x = 1 (where H_W is the Hankel matrix with entries w_{i+j}).
5. Set a_new = max(0, x) / sum(max(0, x)).
6. Damped update: a = (1-α) * a + α * a_new.
7. Decrease β.
8. Repeat.

This should converge to a local optimum of the original problem (as β → 0, the smoothed problem approaches the original).

"""
Multi-phase optimizer for minimizing peak autocorrelation of non-negative sequences.
Combines: (1) KKT fixed-point iteration with annealing (exploiting the fact that at
optimum, a satisfies a Hankel linear system), (2) L-BFGS with log-sum-exp smoothing,
(3) flattening heuristic, (4) multi-n exploration, (5) local search with random perturbations.
Uses FFT-accelerated Hankel matvec and CG for the KKT linear system.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_func(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-12, reg=0.0):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            x = x_new
            r = r_new
            p = r.copy()
            rs = float(r @ r)
            if rs < tol * tol * rs0:
                break
            continue
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _kkt_fixed_point(a_init, deadline, max_rounds=200, beta_start=None, beta_end=None,
                     damping_start=0.3, damping_end=1.0, reg=1e-10):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_range = c.max() - np.percentile(c, 50)
    if c_range < 1e-15:
        c_range = c.max() * 0.1 + 1e-15
    if beta_start is None:
        beta_start = c_range * 2.0
    if beta_end is None:
        beta_end = c_range * 0.001
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        t_frac = rnd / max(1, max_rounds - 1)
        log_beta = np.log(beta_start) + t_frac * (np.log(beta_end) - np.log(beta_start))
        beta = np.exp(log_beta)
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        x = _cg_solve_hankel(w, n, rhs, max_iter=300, tol=1e-10, reg=reg)
        x = np.maximum(x, 0)
        s_x = x.sum()
        if s_x < 1e-15:
            break
        x = x / s_x
        alpha = damping_start + (damping_end - damping_start) * t_frac
        a_new = (1 - alpha) * a + alpha * x
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _smoothed_objective(z, beta, n):
    a = _softmax_func(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_gradient(z, beta, n):
    a = _softmax_func(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)
