"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-stage L-BFGS with log-sum-exp on log-space parameters (n=32768 primary)
2. Kaiser-window + interpolated parent initialization
3. Adaptive beta schedule with gradient-norm monitoring
4. Targeted peak deflation via responsibility-weighted reduction
5. Projected subgradient with Nesterov acceleration for final polish
6. Multi-n comparison (16384, 32768)
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import i0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _kaiser_window(n, alpha):
    t = np.linspace(-1, 1, n)
    w = i0(alpha * np.sqrt(np.maximum(0, 1 - t**2))) / i0(alpha)
    return np.maximum(w, 1e-10)


def _make_init(n, parent, mode="blend"):
    if mode == "kaiser":
        a = _kaiser_window(n, 12.0)
    elif mode == "uniform":
        a = np.ones(n)
    elif mode == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
        a = np.maximum(a, 1e-10)
    elif mode == "block":
        a = np.zeros(n)
        bs = n // 3
        a[:bs] = 1.0
        a[2 * bs:] = 1.0
        a = np.maximum(a, 1e-10)
    else:
        a = np.ones(n)
    if parent is not None:
        n_old = len(parent)
        x_old = np.linspace(0, 1, n_old)
        x_new = np.linspace(0, 1, n)
        a_interp = np.interp(x_new, x_old, parent)
        a_interp = np.maximum(a_interp, 1e-10)
        a_interp = a_interp / a_interp.sum()
        a = a / a.sum()
        if mode == "blend":
            a = 0.5 * a + 0.5 * a_interp
        else:
            a = a * a_interp
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_stage(a_init, beta, n, max_iter, deadline):
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    def obj(z_local):
        return _lsobj(z_local, beta, n)

    def grad(z_local):
        return _lsgrad(z_local, beta, n)

    est_time_per_iter = 0.003 * max(1, n / 2048.0)
    remaining = deadline - time.monotonic()
    actual_max_iter = min(max_iter, max(20, int(remaining / (est_time_per_iter * 8))))

    try:
        result = minimize(
            obj, z, jac=grad, method='L-BFGS-B',
            options={'maxiter': actual_max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxcor': 50, 'maxls': 50}
        )
        a_new = _softmax(result.x)
        return a_new
    except Exception:
        return a_init


def _run_multi_stage_lbfgs(a_init, deadline, n, time_budget):
    a = a_init.copy()
    c = _full_convolve(a, a)
    max_c = float(c.max())

    # Beta schedule: start smooth, end sharp
    beta_start = 0.08 * max_c
    beta_end = 0.00005 * max_c
    n_stages = 8
    betas = beta_start * (beta_end / beta_start) ** (np.arange(n_stages + 1) / n_stages)

    sc = _score(a)
    best_a = a.copy()
    best_sc = sc

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        stage_time = time_budget / n_stages
        stage_deadline = time.monotonic() + stage_time
        max_iter = max(100, min(2000, int(stage_time / (0.02 * n / 16384))))

        a = _run_lbfgs_stage(a, beta, n, max_iter, stage_deadline)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        print(f"    Stage {i+1}/{n_stages}: beta={beta:.2e}, score={sc:.8f}")

    return best_a, best_sc


def _peak_deflation(a, deadline, n_rounds=30):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        target = 0.95 * c_max

        # Find positions above target
        above = np.where(c > target)[0]
        if len(above) == 0:
            break

        # Compute responsibility of each element
        resp = np.zeros(n)
        for k_pos in above:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            excess = c[k_pos] - target
            for i in range(i_lo, i_hi + 1):
                j = k_pos - i
                if 0 <= i < n and 0 <= j < n:
                    resp[i] += excess * a[j] / (c[k_pos] + 1e-30)
                    resp[j] += excess * a[i] / (c[k_pos] + 1e-30)

        max_resp = resp.max()
        if max_resp < 1e-30:
            break

        # Apply reduction
        eta = 0.02
        a_new = a * (1.0 - eta * resp / max_resp)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller eta
            a_small = a * (1.0 - eta * 0.1 * resp / max_resp)
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-12:
                    best_sc = sc_small
                    best_a = a_small.copy()
                    a = a_small
                else:
                    break

    return best_a, best_sc


def _nesterov_projected(a_init, deadline, n_steps=5000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_sc = _score(a)

    c_lb = 1.0 / (2 * n - 1)  # Lower bound on max_c (with sum=1)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Subgradient: average of active peaks
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient in a-space: 4n * h where h[i] = sum_j w[i+j]*a[j]
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * n * h

        # Subtract mean for sum constraint
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak step size
        gap = c_max - c_lb
        step_size = gap / (gnorm * gnorm + 1e-30)
        step_size = min(step_size, 0.1 / gnorm)

        a_new = y - step_size * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a_new.copy()
        a = a_new
        t_k = t_new

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a

            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax(z)
            beta_p = 0.002 * float(_full_convolve(a_pert, a_pert).max())
            a_pert_opt = _run_lbfgs_stage(a_pert, beta_p, n_best, 500, phase4_deadline)
            sc_pert = _score(a_pert_opt)
            update_best(a_pert_opt, sc_pert, f"perturb_{sigma}")

            if sc_pert < best_score + 0.01 and time.monotonic() < phase4_deadline - 10:
                a_pert_def, sc_pert_def = _peak_deflation(a_pert_opt, phase4_deadline, n_rounds=10)
                update_best(a_pert_def, sc_pert_def, f"perturb_{sigma}_def")

    # === Phase 5: Final polish ===
    print(f"\n[Phase 5] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        max_c_fin = float(c_fin.max())

        # Very small beta L-BFGS
        beta_very_small = max_c_fin * 0.00003
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            a_fin = _run_lbfgs_stage(best_seq, beta_very_small, n_fin,
                                     max(50, int(remaining / 0.05)), deadline - 5)
            update_best(a_fin, _score(a_fin), "final_lbfgs")

        # Final deflation
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _peak_deflation(best_seq, deadline - 2, n_rounds=10)
            update_best(a_fin2, sc_fin2, "final_deflate")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
