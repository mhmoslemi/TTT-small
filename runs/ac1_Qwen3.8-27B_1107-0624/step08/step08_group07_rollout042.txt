            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.7) * lr_mult

        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _fft_convolve(a, a)

        if T > T_start * 0.05:
            g = _grad_smooth(a, c, T, top_k)
        else:
            g = _grad_active(a, c, tol)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 1.5:
            update *= 1.5 / un
        z_new = z - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if t % backtrack_interval == 0:
            a_cand = np.exp(z_new - z_new.max())
            a_cand = a_cand / a_cand.sum()
            sc_cand = _score_norm(a_cand)
            if np.isfinite(best_score) and sc_cand > best_score * 1.008:
                z_new = z.copy()
                m_buf *= 0.3
                v_buf *= 0.3
        z = z_new

        if t % eval_interval == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve = t
            else:
                patience += 1

        if patience > patience_limit and t - last_improve > eval_interval * 5:
            lr_mult *= 0.7
            patience = 0
            if lr_mult < 0.05:
                if best_a is not None:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                lr_mult = 0.3
                last_improve = t

    if best_a is None:
        best_a = np.exp(z - z.max())
        best_a = best_a / best_a.sum()
        best_score = _score_norm(best_a)
    return best_a, best_score


def _active_set_polish(a0, deadline, max_iter=5000, tol=0.003, max_active=12):
    """Equalize active peaks using minimum-norm direction with line search."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    stall = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_max = float(c.max())
        active = np.where(c >= c_max * (1.0 - tol))[0]
        if len(active) < 2:
            break
        if len(active) > max_active:
            top = np.argsort(c[active])[::-1][:max_active]
            active = active[top]
        K = len(active)
        M = np.zeros((K + 1, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        M[K, :] = 1.0
        b = np.concatenate([-np.ones(K), [0.0]])
        G = M @ M.T
        try:
            G_inv = np.linalg.inv(G + 1e-10 * np.eye(G.shape[0]))
            d = M.T @ (G_inv @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M, b, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            sc_sum = a_cand.sum()
            if sc_sum < 1e-15:
                continue
            a_cand = a_cand / sc_sum
            sc = _score_norm(a_cand)
            if sc < best_score * (1.0 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                c = _fft_convolve(a, a)
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 20:
                break

    return best_a, best_score


def _nesterov_simplex(a0, deadline, max_iter=30000, lr_init=0.003, momentum=0.9,
                      top_k=5, tol=0.999):
    """Nesterov momentum directly on the simplex."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_init
    c = _fft_convolve(a, a)
    stall = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        if lr > lr_init * 0.01:
            T = c_max * 0.005
            g = _grad_smooth(a, c, T, top_k)
        else:
            g = _grad_active(a, c, tol)

        g_tan = g - g.mean()
        v = momentum * v - lr * g_tan
        a_new = _project_simplex(a + v)
        c_new = _fft_convolve(a_new, a_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            v = momentum * v
            stall = 0
        else:
            a = a_new
            c = c_new
            v = momentum * v
            lr *= 0.98
            stall += 1

        if stall > 50:
            break

    return best_a, best_score


def _coord_descent(a0, deadline, max_rounds=100):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        improved = False

        # Try the most impactful coordinates first
        contrib = np.zeros(hi - lo + 1)
        for j in range(lo, hi + 1):
            kj = k_max - j
            if 0 <= kj < n:
                contrib[j - lo] = a[j] * a[kj]

        order = np.argsort(contrib)[::-1]
        for idx in order[:20]:
            j = lo + idx
            old_val = a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.6, 0.8, 1.2, 1.5]:
                a_cand = a.copy()
                a_cand[j] = old_val * scale
                sc_sum = a_cand.sum()
                if sc_sum < 1e-15:
                    continue
                a_cand /= sc_sum
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1.0 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try broader search
            step = max(1, n // 500)
            any_imp = False
            for j in range(0, n, step):
                if a[j] < 1e-12:
                    continue
                for scale in [0.5, 1.5]:
                    a_cand = a.copy()
                    a_cand[j] *= scale
                    sc_sum = a_cand.sum()
                    if sc_sum < 1e-15:
                        continue
                    a_cand /= sc_sum
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
                        c = c_cand
                        any_imp = True
                        improved = True
                        break
                if improved:
                    break
            if not any_imp:
                break

    return best_a, best_score


def _try_n_value(n, a_parent, deadline, rng):
    """Optimize for a specific n value, using parent as init if possible."""
    if a_parent is not None and len(a_parent) == n:
        a_init = a_parent.copy()
    elif a_parent is not None:
        # Resize parent to n
        a_init = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a_parent)), a_parent)
    else:
        a_init = np.ones(n) / n

    a_init = np.maximum(a_init, 0.0)
    s = a_init.sum()
    if s > 0:
        a_init = a_init / s

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    # Quick Adam optimization
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    sub_dl = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.7))
    a_opt, sc = _adam_logspace(z, sub_dl, T_start=0.02, T_end=0.001,
                               lr_start=0.03, lr_end=0.005, max_iter=50000,
                               eval_interval=100, top_k=5, tol=0.999)
    if sc < best_score:
        best_score = sc
        best_a = a_opt

    # Quick active set polish
    sub_dl2 = min(deadline, time.monotonic() + max(3.0, (deadline - time.monotonic()) * 0.5))
    a_pol, sc_pol = _active_set_polish(best_a, sub_dl2, max_iter=1000, tol=0.005)
    if sc_pol < best_score:
        best_score = sc_pol
        best_a = a_pol

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Quick exploration of n values
    2. Main Adam optimization on best n
    3. Active-set equalization
    4. Nesterov momentum polish
    5. Coordinate descent
    6. Random restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    a_parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_parent = parent / parent.sum()
    except Exception:
        pass

    if a_parent is None:
        n_default = 8192
        a_parent = np.ones(n_default) / n_default
        print("Using uniform initialization, n=8192")

    n_parent = len(a_parent)
    print(f"Parent: n={n_parent}, score={_score(a_parent):.10f}")

    # Phase 0: Quick exploration of nearby n values (30s)
    phase0_end = min(deadline, t_start + 30.0)
    n_candidates = [n_parent]
    for dn in [-1024, 1024, -2048, 2048, -512, 512]:
        nc = n_parent + dn
        if nc >= 256 and nc <= 32768:
            n_candidates.append(nc)
    n_candidates = list(set(n_candidates))[:4]

    best_overall_a = a_parent.copy()
    best_overall_score = _score(a_parent)
    best_overall_n = n_parent

    print(f"Phase 0: Exploring n values: {n_candidates}")
    for nc in n_candidates:
        if time.monotonic() > phase0_end - 5:
            break
        sub_dl = min(phase0_end, time.monotonic() + 7.0)
        a_try, sc_try = _try_n_value(nc, a_parent, sub_dl, rng)
        full_sc = _score(a_try)
        print(f"  n={nc}: score={full_sc:.10f}")
        if full_sc < best_overall_score:
            best_overall_score = full_sc
            best_overall_a = a_try.copy()
            best_overall_n = nc

    n_main = best_overall_n
    a_main = best_overall_a.copy()
    if a_parent is not None and n_main == n_parent:
        a_main = a_parent.copy()
    print(f"Main n={n_main}, score={_score(a_main):.10f}")

    # Phase 1: Main Adam optimization (60% of remaining time)
    phase1_time = 0.60 * (deadline - time.monotonic())
    phase1_end = time.monotonic() + phase1_time
    print(f"Phase 1: Main Adam, time={phase1_time:.0f}s")

    z = np.log(np.maximum(a_main, 1e-15))
    z = z - z.mean()

    # Run multiple Adam passes with decreasing T
    n_passes = 3
    time_per_pass = phase1_time / n_passes

    for p in range(n_passes):
        sub_dl = time.monotonic() + time_per_pass
        T0 = 0.015 * (0.5 ** p)
        T1 = 0.0003 * (0.5 ** p)
        lr0 = 0.035 * (0.6 ** p)
        lr1 = 0.003 * (0.6 ** p)
        top_k = 5 + p

        a_opt, sc = _adam_logspace(z, sub_dl, T_start=T0, T_end=T1,
                                   lr_start=lr0, lr_end=lr1,
                                   max_iter=100000, eval_interval=80,
                                   top_k=top_k, tol=0.999 - 0.001 * p)
        full_sc = _score(a_opt)
        print(f"  Pass {p}: T0={T0:.5f}, score={full_sc:.10f}")
        if full_sc < best_overall_score:
            best_overall_score = full_sc
            best_overall_a = a_opt.copy()
            z = np.log(np.maximum(a_opt, 1e-15))
            z = z - z.mean()
        else:
            z = np.log(np.maximum(a_opt, 1e-15))
            z = z - z.mean()

    print(f"Phase 1 done: score={best_overall_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Active-set equalization (15% of remaining time)
    phase2_time = 0.15 * (deadline - time.monotonic())
    phase2_end = time.monotonic() + phase2_time
    print(f"Phase 2: Active-set equalization, time={phase2_time:.0f}s")

    a_as, sc_as = _active_set_polish(best_overall_a, phase2_end,
                                     max_iter=5000, tol=0.002, max_active=12)
    full_sc_as = _score(a_as)
    print(f"  Active-set: score={full_sc_as:.10f}")
    if full_sc_as < best_overall_score:
        best_overall_score = full_sc_as
        best_overall_a = a_as.copy()

    # Nesterov on simplex
    phase2b_end = time.monotonic() + 0.5 * phase2_time
    if time.monotonic() < phase2b_end:
        a_ne, sc_ne = _nesterov_simplex(best_overall_a, phase2b_end,
                                        max_iter=20000, lr_init=0.002,
                                        momentum=0.9, top_k=4, tol=0.9995)
        full_sc_ne = _score(a_ne)
        print(f"  Nesterov: score={full_sc_ne:.10f}")
        if full_sc_ne < best_overall_score:
            best_overall_score = full_sc_ne
            best_overall_a = a_ne.copy()

    print(f"Phase 2 done: score={best_overall_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: More Adam with perturbation (15%)
    phase3_time = 0.15 * (deadline - time.monotonic())
    phase3_end = time.monotonic() + phase3_time
    print(f"Phase 3: Perturbed Adam, time={phase3_time:.0f}s")

    n_restarts = 4
    time_per_r = phase3_time / n_restarts
    for r in range(n_restarts):
        if time.monotonic() > phase3_end - 5:
            break
        sigma = 0.003 + 0.002 * r
        z_pert = np.log(np.maximum(best_overall_a, 1e-15))
        z_pert = z_pert + sigma * rng.standard_normal(len(z_pert))
        z_pert = z_pert - z_pert.mean()
        a_pert = np.exp(z_pert - z_pert.max())
        a_pert = a_pert / a_pert.sum()

        sub_dl = time.monotonic() + time_per_r * 0.8
        a_r, sc_r = _adam_logspace(z_pert, sub_dl, T_start=0.005, T_end=0.0002,
                                   lr_start=0.02, lr_end=0.002,
                                   max_iter=50000, eval_interval=100,
                                   top_k=5, tol=0.9995)
        full_sc_r = _score(a_r)
        print(f"  Restart {r} (sigma={sigma:.4f}): score={full_sc_r:.10f}")
        if full_sc_r < best_overall_score:
            best_overall_score = full_sc_r
            best_overall_a = a_r.copy()

    print(f"Phase 3 done: score={best_overall_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 4: Final polish - active set + coordinate descent (10%)
    phase4_time = 0.10 * (deadline - time.monotonic())
    phase4_end = time.monotonic() + phase4_time
    print(f"Phase 4: Final polish, time={phase4_time:.0f}s")

    # Active set with very tight tolerance
    a_final_as, sc_final_as = _active_set_polish(best_overall_a,
                                                 time.monotonic() + 0.4 * phase4_time,
                                                 max_iter=3000, tol=0.001, max_active=8)
    full_sc_fa = _score(a_final_as)
    print(f"  Fine active-set: score={full_sc_fa:.10f}")
    if full_sc_fa < best_overall_score:
        best_overall_score = full_sc_fa
        best_overall_a = a_final_as.copy()

    # Coordinate descent
    a_cd, sc_cd = _coord_descent(best_overall_a,
                                 time.monotonic() + 0.4 * phase4_time,
                                 max_rounds=50)
    full_sc_cd = _score(a_cd)
    print(f"  Coord descent: score={full_sc_cd:.10f}")
    if full_sc_cd < best_overall_score:
        best_overall_score = full_sc_cd
        best_overall_a = a_cd.copy()

    # One more active set pass
    if time.monotonic() < phase4_end - 5:
        a_as2, sc_as2 = _active_set_polish(best_overall_a, phase4_end,
                                           max_iter=2000, tol=0.001, max_active=8)
        full_sc_a2 = _score(a_as2)
        if full_sc_a2 < best_overall_score:
            best_overall_score = full_sc_a2
            best_overall_a = a_as2.copy()

    # Phase 5: Final check and validation
    final = np.clip(best_overall_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01

    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
