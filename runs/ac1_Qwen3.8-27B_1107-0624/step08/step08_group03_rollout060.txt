"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-scale optimization with L-BFGS log-sum-exp, active-set Newton,
and coordinate descent. Tries multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    """Compute the correlation needed for active-set gradient."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_max = float(c.max())

            return (2.0 / m) * result + mu * v

        # CG to solve H_system * dx = -g_t
        # where H_system[i,j] = (2/m) * A_vec[i+j] + mu * delta_ij
        x = np.zeros(n)
        r = -g_t - _hankel_mv(x) + g_t  # = -_hankel_mv(0) = mu*0 = 0... 
        # Wait, r should be b - A*0 = b = -g_t
        r = -g_t
        p = r.copy()
        rs = np.dot(r, r)
        
        for cg_it in range(200):
            Ap = _hankel_mv(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x_new = x + alpha * p
            r_new = -g_t - _hankel_mv(x_new) + (0 if False else 0)
            # Actually r_new = b - A*x_new = -g_t - _hankel_mv(x_new)
            # But _hankel_mv already includes the +mu*v term
            # So A_system * x = (2/m)*H*x + mu*x
            # r = b - A_system * x = -g_t - (2/m)*H*x - mu*x
            # _hankel_mv(x) = (2/m)*H*x + mu*x
            # So r_new = -g_t - _hankel_mv(x_new)

            rs_new = np.dot(r_new, r_new)

                x = x_new
                break
            x = x_new
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        # Handle the sum constraint: dx should have zero mean (to preserve sum(a)=1)
        # Actually, we need: sum(a + t*dx) = 1, so we need sum(dx) = 0
        # Use a projection: dx = x - mean(x)
        dx = x - np.mean(x)
        
        # But wait, we also need to handle the constraint properly.
        # The KKT system is:
        #   H_system * dx + lam * 1 = -g_t
        #   1^T dx = 0
        # We solved H_system * dx = -g_t (ignoring the constraint)
        # Then projected to zero-mean. This is an approximation.
        # For better accuracy, we should solve the full KKT system.
        
        # For now, use the projected direction.
        
        # Line search
        f_old = c_max
        accepted = False
        t_step = min(1.0, 1.0 / (1.0 + 0.1 * np.linalg.norm(dx)))

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, aggressive=False):
    """Log-sum-exp smoothed L-BFGS optimization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - min(c_sorted[min(20, len(c_sorted)-1)], max_c)
    else:
        gap = max_c
    base_beta = max(gap * 0.3, max_c * 0.01)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03,
                                      0.015, 0.007, 0.0035, 0.0018,
                                      0.0009, 0.0005, 0.00025, 0.00012,
                                      0.00006])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001,
                                      0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30)).astype(np.float64)
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations based on available time
        if n <= 4096:
            est_time_per_iter = 0.002
        elif n <= 8192:
            est_time_per_iter = 0.004
        elif n <= 16384:
            est_time_per_iter = 0.01
        elif n <= 32768:
            est_time_per_iter = 0.025
        else:
            est_time_per_iter = 0.05

        max_iter = min(8000, max(20, int(remaining / (est_time_per_iter * 15))))

        def obj_fn(z_local):
            return _lse_objective(z_local, beta, n)

        def grad_fn(z_local):
            return _lse_gradient(z_local, beta, n)

        try:
            result = minimize(
                obj_fn,
                z,
                jac=grad_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30)).astype(np.float64)
                z = z - z.mean()
            else:
                # If no improvement, keep z as is
                pass
        except Exception:
            pass

    return current_a


def _coord_descent_polish(a_init, deadline, max_rounds=200):
    """Coordinate descent on individual entries to polish the solution."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find the lag with maximum convolution
        k_max = int(np.argmax(c))
        
        # The value at k_max is sum_j a_j * a_{k_max - j}
        # To reduce it, we should decrease the products a_j * a_{k_max-j}
        
        # Gradient at the peak:
        # d(c[k_max])/d(a_i) = a[k_max - i] (for valid indices)
        # So decreasing a_i by epsilon reduces c[k_max] by epsilon * a[k_max-i]
        
        # Best coordinate to decrease: the one that reduces c[k_max] the most
        # per unit decrease in a_i, while not increasing other peaks too much.
        
        # Simple approach: try decreasing the largest a_i that contributes to the peak
        # and increasing a_j that doesn't contribute much.
        
        # Compute the gradient of c_max w.r.t. a
        w = np.zeros(2 * n - 1)
        # Focus on top few peaks
        n_top = min(5, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        peak_threshold = c[top_idx[-1]] * 0.999
        for k in top_idx:
            if c[k] >= peak_threshold:
                w[k] = 1.0 / n_top
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n -
