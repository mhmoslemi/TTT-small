"""
Convex optimization for minimum peak autocorrelation step functions.
Uses FFT-based gradient/Hessian computation, Adam for global phase,
and active-set Newton for local polish.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _autocorr(a):
    return _fft_convolve(a, a)

def _corr_weighted(x, p):
    """Compute sum_k p[k] * (x[m-k] + x[m+k]) for m = 0..n-1.
    x has length n, p has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(x)
    N = _next_pow2(2*n - 1 + n - 1)
    P = np.fft.rfft(p, N)
    X = np.fft.rfft(x, N)
    # Part 1: conv(p, x)[0:n]
    v1 = np.fft.irfft(P * X, N)[:n]
    # Part 2: cross-correlation
    p_flip = p[::-1]
    PF = np.fft.rfft(p_flip, N)
    conv_fa = np.fft.irfft(PF * X, N)
    offset = 2*n - 2
    v2 = conv_fa[offset:offset + n]
    return v1 + v2

def _grad_smoothed(a, c, T):
    """Compute gradient of T*log(sum exp(c[k]/T)) w.r.t. a using FFT."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    total = e.sum()
    if total < 1e-300 or not np.isfinite(total):
        p = np.zeros(2*n - 1)
        p[np.argmax(c)] = 1.0
    else:
        p = e / total
    return _corr_weighted(a, p)

def _grad_exact(a, c, active):
    """Compute gradient of max(c[k] for k in active) w.r.t. a.
    Uses uniform weights over the active set.
    """
    n = len(a)
    K = len(active)
    p = np.zeros(2*n - 1)
    for k in active:
        p[k] = 1.0 / K
    return _corr_weighted(a, p)

def _hess_vec(x, p):
    """Compute H @ x where H = sum_k p[k] * H_k (the linearized Hessian).
    Same formula as _corr_weighted but with x instead of a.
    """
    return _corr_weighted(x, p)

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1: 2n * max(autocorr)"""
    n = len(a)
    c = _autocorr(a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, eval_interval=50):
    """Adam optimizer in log-space with FFT-based gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    last_improve = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8) * lr_mult
        
        a = _softmax_stable(z)
        c = _autocorr(a)
        g = _grad_smoothed(a, c, T)
        
        # Center gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        unorm = np.linalg.norm(update)
        if unorm > 1.0:
            update /= unorm
        z = z - update
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve = t
            else:
                patience += 1
            if patience > 15 and t - last_improve > 500:
                lr_mult *= 0.7
                patience = 0
                if lr_mult < 0.05:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.3
    
    a_final = _softmax_stable(z)
    sc = _score_norm(a_final)
    if sc < best_score:

        best_a = a_final.copy()

def _projected_gradient(a, deadline, max_iter=50000, tol_gap=1e-10):
    """Projected gradient method on the simplex with backtracking."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr(a)
        c_max = float(c.max())
        
        # Find active peaks (within relative tolerance of max)
        tol = c_max * 0.999
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute exact gradient
        g = _grad_exact(a, c, active)
        
        # Project onto tangent space (sum constraint)
        g_t = g - np.mean(g)
        
        # Backtracking line search
        alpha = 0.1
        for _ in range(30):
            a_new = a - alpha * g_t
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_new /= s
            sc = _score_norm(a_new)
            if sc <= best_score + 1e-15:
                break
            alpha *= 0.5
        else:
            break
        
        if not np.isfinite(sc) or sc > best_score * (1 + 1e-10):
            # Try smaller alpha
            alpha *= 0.1
            if alpha < 1e-15:
                break
            a_new = a - alpha * g_t
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new /= s
            sc = _score_norm(a_new)
            if not np.isfinite(sc) or sc >= best_score:
                break
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        
        # Check convergence
        g_norm = np.linalg.norm(g_t)
        if g_norm < tol_gap:
            break
    
    return best_a, best_score


def _newton_polish(a, deadline, max_iter=5000, cg_tol=1e-10):
    """Newton-like method with FFT-based Hessian and constraint handling."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr(a)
        c_max = float(c.max())
        
        # Active set
        tol = c_max * 0.9995
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 30:
            idx = np.argsort(c[active])[::-1][:30]
            active = active[idx]
        
        K = len(active)
        p = np.zeros(2*n - 1)
        for k in active:
            p[k] = 1.0 / K
        
        # Gradient
        g = _corr_weighted(a, p)
        
        # Project gradient onto tangent space
        g_t = g - np.mean(g)
        
        # Newton direction: solve (H + reg*I) @ d = -g_t in tangent space
        # Use CG with the constraint 1^T d = 0
        # Simpler: solve H @ d = -g, then project
        reg = 1e-6 * c_max
        
        # CG to solve (H + reg*I) @ d = -g_t
        def hvp(v):
            return _hess_vec(v, p) + reg * v
        
        # Initial guess
        d = np.zeros(n)
        r = -g_t - hvp(d)  # = -g_t since d=0
        p_cg = r.copy()
        rr = float(r @ r)
        
        for _ in range(min(50, n)):  # At most 50 CG iterations
            Ap = hvp(p_cg)
            pAp = float(p_cg @ Ap)
            if abs(pAp) < 1e-30:
                break
            alpha_cg = rr / pAp
            d += alpha_cg * p_cg
            r -= alpha_cg * Ap
            rr_new = float(r @ r)
            if np.sqrt(rr_new) < cg_tol * np.sqrt(rr + 1e-30):
                rr = rr_new
                break
            p_cg = r + (rr_new / rr) * p_cg
            rr = rr_new
        
        # Project d onto tangent space
        d = d - np.mean(d)
        
        # Also ensure d respects non-negativity (simple check)
        # If a + d has negative components, scale down

"""
Convex-structured optimization for minimum peak autocorrelation step functions.
Uses FFT-based gradient/Hessian computation, Adam for global convergence,
and a linearized active-set LP with optimal step selection for local polish.
Exploits the fact that near the optimum, the problem reduces to a sequence
of small LPs over the active peak set.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _autocorr(a):
    return _fft_convolve(a, a)


def _corr_weighted(x, p):
    """Compute sum_k p[k] * (x[m-k] + x[m+k]) for m=0..n-1 via FFT.
    x has length n, p has length 2n-1. Returns vector of length n.
    """
    n = len(x)
    if n == 0:
        return np.zeros(0)

    P = np.fft.rfft(p, N)
    X = np.fft.rfft(x, N)
    v1 = np.fft.irfft(P * X, N)[:n]
    p_flip = p[::-1]
    PF = np.fft.rfft(p_flip, N)
    conv_fa = np.fft.irfft(PF * X, N)
    offset = 2 * n - 2
    v2 = conv_fa[offset:offset + n]
    return v1 + v2


def _grad_smoothed(a, c, T):
    """Gradient of T*log(sum exp(c[k]/T)) w.r.t. a, via FFT."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    total = e.sum()
    if total < 1e-300 or not np.isfinite(total):
        p = np.zeros(2 * n - 1)
        p[np.argmax(c)] = 1.0
    else:
        p = e / total
    return _corr_weighted(a, p)


def _grad_active(a, active):
    """Exact gradient of max(c[k] for k in active) w.r.t. a, uniform weights."""
    n = len(a)
    K = len(active)
    p = np.zeros(2 * n - 1)
    for k in active:
        p[k] = 1.0 / K
    return _corr_weighted(a, p)


def _compute_rates(a, d, active):
    """Compute rate_k = grad_k . d for each active peak k.
    rate_k = sum_j a[j]*d[j-k] + sum_j a[j]*d[j+k] (valid indices).
    """
    n = len(a)
    rates = np.zeros(len(active))
    for i, k in enumerate(active):
        if k == 0:
            rates[i] = 2.0 * np.dot(a, d)
        else:
            rates[i] = np.dot(a[k:], d[:n - k]) + np.dot(a[:n - k], d[k:])
    return rates


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _autocorr(a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, eval_interval=50):
    """Adam in log-space with FFT-based gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    last_improve = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8) * lr_mult

        a = _softmax_stable(z)
        c = _autocorr(a)
        g = _grad_smoothed(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        unorm = np.linalg.norm(update)
        if unorm > 1.0:
            update /= unorm
        z = z - update
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve = t
            else:
                patience += 1
            if patience > 20 and t - last_improve > 800:
                lr_mult *= 0.6
                patience = 0
                if lr_mult < 0.04:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.003 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.2

    a_final = _softmax_stable(z)
    sc = _score_norm(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score


def _active_set_lp_step(a, deadline, max_iter=100000, tol_relax=0.9999):
    """Local optimization using linearized active-set LP with optimal step.
    At each iteration:
    1. Identify active peaks
    2. Compute gradient (tangent-projected)
    3. Find optimal step size that minimizes max of active peaks (1D piecewise linear)
    4. Take the step
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _autocorr(a)
        c_max = float(c.max())

        # Active set: peaks within relative tolerance
        tol_val = c_max * tol_relax
        active = np.where(c >= tol_val)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 40:
            idx = np.argsort(c[active])[::-1][:40]
            active = active[idx]

        # Compute gradient using active set
        g = _grad_active(a, active)
        # Project onto tangent space (sum constraint)
        g_t = g - np.mean(g)
        g_norm = np.linalg.norm(g_t)
        if g_norm < 1e-14:
            break

        # Direction: negative gradient (projected)
        d = -g_t / g_norm

        # Compute rates for active peaks
        rates = _compute_rates(a, d, active)
        c_active = c[active]

        # Find max alpha before non-negativity violation
        neg_mask = d < -1e-15
        if np.any(neg_mask):
            max_alpha = float(np.min(a[neg_mask] / (-d[neg_mask])))
        else:
            max_alpha = np.inf
        max_alpha = min(max_alpha, 1.0)

        # Minimize max_i (c_active[i] + alpha * rates[i]) over alpha in [0, max_alpha]
        # This is a 1D piecewise-linear convex function
        # The minimum is at a breakpoint or at a boundary
        best_alpha = 0.0
        best_val = float(c_active.max())

        # Candidate points: 0, max_alpha, and all breakpoints
        candidates = [0.0, max_alpha * 0.01, max_alpha * 0.1, max_alpha * 0.5, max_alpha]
        K = len(active)
        for i_idx in range(K):
            for j_idx in range(i_idx + 1, K):
                dr = rates[i_idx] - rates[j_idx]
                if abs(dr) > 1e-15:
                    bp = (c_active[j_idx] - c_active[i_idx]) / dr
                    if 0 <= bp <= max_alpha:
                        candidates.append(bp)

        for alpha in candidates:
            vals = c_active + alpha * rates
            v = float(vals.max())
            if v < best_val - 1e-15:
                best_val = v
                best_alpha = alpha

        if best_alpha < 1e-15:
            # No improvement possible with this direction
            # Try a smaller active set or gradient direction only
            break

        # Take the step
        a_new = a + best_alpha * d
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new

        sc_new = _score_norm(a_new)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Step didn't help; try gradient direction with small step
            for alpha_try in [0.001, 0.0005, 0.0001]:
                a_try = a - alpha_try * g_t / g_norm
                a_try = np.maximum(0, a_try)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try /= s_try
                sc_try = _score_norm(a_try)
                if np.isfinite(sc_try) and sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                break

    return best_a, best_score


def _newton_cg_step(a, p, reg, max_cg=30):
    """Compute Newton direction using CG with FFT-based Hessian-vector product.
    Solves (H + reg*I) @ d = -g in the tangent space (1^T d = 0).
    """
    n = len(a)
    g = _corr_weighted(a, p)
    g_t = g - np.mean(g)

    def hvp(v):
        return _corr_weighted(v, p) + reg * v

    # CG in the subspace 1^T d = 0
    d = np.zeros(n)
    r = -(g_t) - (hvp(d) - np.mean(hvp(d)))  # residual in tangent space
    # Since d=0: r = -g_t
    r = -g_t
    p_vec = r.copy()
    rr = float(r @ r)
    if rr < 1e-30:
        return np.zeros(n)

    for _ in range(max_cg):
        Ap = hvp(p_vec)
        Ap_t = Ap - np.mean(Ap)  # project to tangent space
        pAp = float(p_vec @ Ap_t)
        if abs(pAp) < 1e-30:
            break
        alpha_cg = rr / pAp
        d += alpha_cg * p_vec
        r = r - alpha_cg * Ap_t
        rr_new = float(r @ r)
        if np.sqrt(rr_new) < 1e-10 * np.sqrt(rr + 1e-30):
            break
        p_vec = r + (rr_new / rr) * p_vec
        rr = rr_new

    # Project to tangent space
    d = d - np.mean(d)
    return d


def _newton_polish(a, deadline, max_iter=5000):
    """Newton-like method with FFT Hessian and active set."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _autocorr(a)
        c_max = float(c.max())

        tol = c_max * 0.9999
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 25:
            idx = np.argsort(c[active])[::-1][:25]
            active = active[idx]

        K = len(active)
        p = np.zeros(2 * n - 1)
        for k in active:
            p[k] = 1.0 / K

        reg = 1e-8 * c_max + 1e-10
        d = _newton_cg_step(a, p, reg, max_cg=20)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm

        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
            a_new = a + alpha * d
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

            for alpha in [0.5, 0.25, 0.1, 0.05, 0.01]:
