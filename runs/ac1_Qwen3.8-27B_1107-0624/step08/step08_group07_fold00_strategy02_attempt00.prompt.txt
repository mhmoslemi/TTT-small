def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _subgradient_polish(a, deadline, max_iter=50000, top_k=3, initial_lr=0.005):
    """Direct subgradient method on the max convolution peak."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Get top-k peaks (equalized)
        sorted_idx = np.argsort(c)[::-1]
        # Only include peaks that are very close to max
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
        
        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project gradient to the tangent space of the sum constraint
        # (subtract the mean to ensure the step preserves the sum)
        g_tangent = g - np.sum(g) * a  # approximate projection
        
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break
        
        # Take step
        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9995
        else:
            lr *= 0.98
            if lr < 1e-8:
                break
    
    return best_a, _score_norm(best_a)


def _active_set_equalization(a, deadline, max_iter=5000, tol=0.001):
    """Try to equalize the top peaks by solving a linear system."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 15:
            sorted_a = active[np.argsort(c[active])[::-1][:15]]
            active = sorted_a
        
        # Build the linear system:
        # For each active peak k: sum_j M[k,j] * d[j] = -1
        # where M[k,j] = 2*a[k-j] if 0 <= k-j < n, else 0
        # Plus constraint: sum_j d[j] = 0
        # Solve minimum-norm: d = M^T (M M^T)^{-1} b
        
        K = len(active)
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Add sum constraint
        M_full = np.vstack([M, np.ones(n)])
        b = np.concatenate([-np.ones(K), [0.0]])
        
        # Minimum norm solution: d = M_full^T @ pinv(M_full @ M_full^T) @ b
        # But M_full is (K+1) x n with n >> K+1, so:
        # d = M_full^T @ inv(M_full @ M_full^T) @ b
        G = M_full @ M_full.T  # (K+1) x (K+1)
        try:
            G_inv = np.linalg.inv(G + 1e-12 * np.eye(G.shape[0]))
            d = M_full.T @ (G_inv @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b, rcond=None)[0]
        
        # Scale the step
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Try multiple step sizes
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            # Try small random perturbation in the null space
            break
    
    return best_a, best_score
