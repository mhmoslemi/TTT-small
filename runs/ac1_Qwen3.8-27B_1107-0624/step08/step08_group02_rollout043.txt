```python
"""
Optimized sequence design using QP-weighted Newton method with alternating weight optimization.
Key innovation: Solve a small QP at each iteration to find optimal weights on the active set,
which minimizes gradient variance and leads to more accurate Newton steps.
Combined with a linear-system-based "dual" solve for direct a-update.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, w, n):
    """Compute H*v where H[i,j] = w[i+j]. Returns 2*H*v for consistency."""
    V = v[::-1]
    out_len = len(V) + len(w) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(w, n, rhs, max_iter=500, tol=1e-13):
    """Solve H(w)*x = rhs using CG, where H is Hankel with H[i,j]=w[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _compute_active_set(c, tol_frac=0.001, max_active=500):
    """Find active set: positions where c[k] >= c_max * (1 - tol_frac)."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.array([argmax]), np.array([1.0])
    threshold = c_max * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) > max_active:
        # Keep only the top max_active
        c_at_active = c[active]
        idx = np.argsort(c_at_active)[::-1][:max_active]
        active = active[idx]
    return active


def _build_P_matrix(a, active_indices):
    """Build matrix P where P[i, idx] = a[k - i] if valid."""
    n = len(a)
    A = len(active_indices)
    P = np.zeros((n, A))
    for idx in range(A):
        k = active_indices[idx]
        i_max = min(n - 1, k)
        i_min = max(0, k - n + 1)
        if i_max >= i_min:
            P[i_min:i_max + 1, idx] = a[k:i_min - 1:-1]
    return P


def _solve_qp_weights(P, n):
    """Solve min w^T Q^T Q w s.t. w>=0, sum(w)=1 where Q = P - 1*(S/n)^T."""
    A = P.shape[1]
    if A == 1:
        return np.array([1.0])
    if A == 2:
        # Simple case: minimize variance with 2 weights
        col_var = np.var(P, axis=0)
        w1 = col_var[1] / (col_var[0] + col_var[1] + 1e-30)
        w1 = np.clip(w1, 0.01, 0.99)
        return np.array([1.0 - w1, w1])
    
    S = P.sum(axis=0)  # column sums
    Q = P - np.outer(np.ones(n), S / n)
    # Compute A_mat = Q^T @ Q efficiently
    A_mat = Q.T @ Q
    # Symmetrize
    A_mat = (A_mat + A_mat.T) / 2.0
    
    n_a = len(A_mat)
    w0 = np.ones(n_a) / n_a
    
    def obj(w):
        return float(w @ A_mat @ w)
    
    def grad(w):
        return 2.0 * (A_mat @ w)
    
    constraints = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0,
                    'jac': lambda w: np.ones(n_a)}]
    bounds = [(0.0, 1.0)] * n_a
    
    try:
        result = minimize(obj, w0, jac=grad, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 500, 'ftol': 1e-18})
        w = result.x
        w = np.maximum(w, 0)
        w = w / w.sum()
        return w
    except Exception:
        return np.ones(n_a) / n_a


def _newton_step(a, W_full, n):
    """Compute Newton step given weight vector W_full (length 2n-1)."""
    g = _hankel_matvec(a, W_full, n)
    g_mean = g.mean()
    g_c = g - g_mean
    
    Hg = _cg_solve_hankel(W_full, n, g, max_iter=500)
    H1 = _cg_solve_hankel(W_full, n, np.ones(n), max_iter=500)
    
    num = float(np.ones(n) @ Hg)
    den = float(np.ones(n) @ H1)
    if abs(den) < 1e-25:
        # Fallback: steepest descent
        gn = np.linalg.norm(g_c)
        if gn < 1e-25:
            return None
        delta = -g_c / gn
    else:
        nu = num / den
        delta = -Hg + nu * H1
        delta = delta - delta.mean()
    
    delta_norm = np.linalg.norm(delta)
    if delta_norm < 1e-25 or not np.isfinite(delta_norm):
        return None
    
    return delta


def _line_search(a, delta, c_max_current, max_tries=60):
    """Aggressive line search: find alpha such that max conv decreases."""
    n = len(a)
    alpha = 1.0
    for _ in range(max_tries):
        a_try = a + alpha * delta
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            alpha *= 0.5
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        c_max_try = float(c_try.max())
        if c_max_try < c_max_current - 1e-15:
            return a_try, alpha
        alpha *= 0.7
        if alpha < 1e-18:
            break
    return None, 0.0


def _subgradient_direction(a, c, active, n):
    """Compute subgradient direction (for fallback)."""
    W = np.zeros(2 * n - 1)
    W[active] = 1.0 / len(active)
    g = _hankel_matvec(a, W, n)
    g_c = g - g.mean()
    gn = np.linalg.norm(g_c)
    if gn < 1e-25:
        return None
    return -g_c / gn


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _lbfgs_phase(a_init, deadline, min_time=30.0):
    """Run adaptive-beta L-BFGS on log-scale parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.3, max_c * 0.002)
    min_beta = max_c * 1e-10

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(6000, max(20, int(remaining / (est_time_per_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-8 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.15
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.0005)

    return current_a


def _qps_newton_loop(a_init, deadline, min_time=60.0):
    """Main QP-Newton alternating optimization loop."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    tol_frac = 0.005  # active set tolerance (start loose)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    iteration = 0
    while time.monotonic() + min_time < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Compute active set
        active = _compute_active_set(c, tol_frac, max_active=300)
        n_active = len(active)

        # Compute weights
        if n_active >= 3 and n_active <= 300:
            P = _build_P_matrix(a, active)
            w_active = _solve_qp_weights(P, n)
        else:
            w_active = np.ones(n_active) / n_active

        # Build full weight vector
        W_full = np.zeros(2 * n - 1)
        W_full[active] = w_active

        # Newton step
        delta = _newton_step(a, W_full, n)

        if delta is None:
            # Fallback to subgradient
            delta = _subgradient_direction(a, c, active, n)
            if delta is None:
                break
            alpha = 0.01
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try.max() < c_max:
                    a = a_try
                    continue
            break

        # Line search
        a_new, alpha = _line_search(a, delta, c_max)

        if a_new is None:
            # Try smaller fixed steps
            found = False
            for trial_alpha in [0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
                a_try = a + trial_alpha * delta
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try.max() < c_max * (1 - 1e-8):
                    a = a_try
                    found = True
                    break
            if not found:
                no_improve += 1
                if no_improve > 10:
                    break
                # Try subgradient fallback
                delta_sg = _subgradient_direction(a, c, active, n)
                if delta_sg is not None:
                    for sa in [0.005, 0.002, 0.001]:
                        a_try = a + sa * delta_sg
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        c_try = _full_convolve(a_try, a_try)
                        if c_try.max() < c_max:
                            a = a_try
                            break
                continue
        else:
            a = a_new

        # Update best
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            # Tighten active set tolerance as we converge
            if no_improve == 0 and tol_frac > 0.001:
                tol_frac *= 0.8
        else:
            no_improve += 1
            if no_improve > 20:
                # Loosen active set and try again
                tol_frac = min(tol_frac * 2.0, 0.01)
                if tol_frac >= 0.01:
                    break

        # KKT violation check
        g = _hankel_matvec(a, W_full, n)
        g_c = g - g.mean()
        kkt_viol = float(np.max(np.abs(g_c))) / (float(c.max()) + 1e-30)
        if kkt_viol < 1e-10:
            break

    return best_a, best_sc


def _final_newton_polish(a_init, deadline, max_rounds=200):
    """Ultra-fine Newton polish with very tight active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Very tight active set
        best_tol = 0.0001
        for tol_frac in [0.000005, 0.00001, 0.00005, 0.0001, 0.0005, 0.001, 0.005]:
            tol = tol_frac * c_max
            active = np.where(c >= c_max - tol)[0]
            if 1 <= len(active) <= 200:
                best_tol = tol_frac
                break
        else:
            active = np.where(c >= c_max * (1 - 0.005))[0]
            if len(active) > 200:
                c_at = c[active]
                idx = np.argsort(c_at)[::-1][:200]
                active = active[idx]
            best_tol = 0.005

        n_active = len(active)

        # QP weights if possible
        if n_active >= 3 and n_active <= 200:
            P = _build_P_matrix(a, active)
            w_active = _solve_qp_weights(P, n)
        else:
            w_active = np.ones(n_active) / n_active

        W_full = np.zeros(2 * n - 1)
        W_full[active] = w_active

        delta = _newton_step(a, W_full, n)
        if delta is None:
            break

        # Very aggressive line search
        a_new, alpha = _line_search(a, delta, c_max, max_tries=80)

        if a_new is not None:
            a = a_new
        else:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_sc


def _dual_solve(a_init, deadline):
    """Solve the linear system H(w)*a = mu*1, 1^T*a=1 directly (dual approach)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = _score(a)

    for it in range(50):
        if time.monotonic() + 10 > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = _compute_active_set(c, 0.002, max_active=300)

        if len(active) >= 3 and len(active) <= 300:
            P = _build_P_matrix(a, active)
            w_active = _solve_qp_weights(P, n)
        else:
            w_active = np.ones(len(active)) / len(active)

        W_full = np.zeros(2 * n - 1)
        W_full[active] = w_active

        # Solve the augmented system:
        # [H(w)  -1] [a_new] = [0]
        # [1^T    0] [mu]    = [1]
        # Using block elimination:
        # H(w) * a_new = mu * 1
        # mu = (1^T * H(w) * a_new) / n
        # So: H(w) * a_new = (1^T * H(w) * a_new) / n * 1
        # This means H(w) * a_new must be a constant vector.
        # Equivalently: (H(w) - H(w)*1/n * 1^T) * a_new = 0
        # Which means a_new is in the null space of the "centered" Hankel matrix.
        # This is tricky. Instead, let's solve the augmented system with CG.
        
        # Augmented system matrix M:
        # M @ [v; nu] = [H(w)*v - nu*1; 1^T*v]
        # We want M @ [a_new; mu] = [0; 1]
        
        # Use CG on the augmented system
        N_aug = n + 1
        x_aug = np.zeros(N_aug)
        x_aug[:n] = a  # initial guess
        r_aug = np.zeros(N_aug)
        r_aug[:n] = 0  # will compute below
        # M @ x_aug = [H(w)*a - mu*1; 1^T*a] = [H(w)*a - 0; 1] (since a is normalized)
        # We want this to equal [0; 1], so r_aug = [0 - H(w)*a; 1 - 1] = [-H(w)*a; 0]
        H_a = _hankel_matvec(a, W_full, n)  # = 2 * H * a
        r_aug[:n] = -H_a  # residual for first n components
        r_aug[n] = 0.0  # residual for last component (1^T*a - 1 = 0)
        
        p_aug = r_aug.copy()
        rs = float(r_aug @ r_aug)
        if rs < 1e-30:
            a_new = a.copy()
        else:
            for k in range(300):
                # M @ p_aug
                Mp = np.zeros(N_aug)
                Hp = _hankel_matvec(p_aug[:n], W_full, n)
                Mp[:n] = Hp - p_aug[n] * np.ones(n)
                Mp[n] = float(np.ones(n) @ p_aug[:n])
                
                pMp = float(p_aug @ Mp)
                if abs(pMp) < 1e-30:
                    break
                alpha_cg = rs / pMp
                x_aug = x_aug + alpha_cg * p_aug
                r_aug = r_aug - alpha_cg * Mp
                rs_new = float(r_aug @ r_aug)
                if rs_new >= rs * 1.1 or not np.isfinite(rs_new):
                    break
                p_aug = (rs_new / rs) * p_aug + r_aug
                rs = rs_new
                if rs < 1e-25:
                    break
        
        a_new = np.maximum(x_aug[:n], 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            for t in [0.5, 0.3, 0.1]:
                a_try = (1 - t) * a + t * a_new
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
```
