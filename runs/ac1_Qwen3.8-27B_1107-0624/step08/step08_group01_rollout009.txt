def spectral_flatten(a, n_iter, N):
    a = np.array(a, dtype=float)
    n = len(a)
    for it in range(n_iter):
        # Compute FFT
        A = np.fft.rfft(a, N)
        # Flatten magnitude
        mag = np.abs(A)
        target_mag = np.ones_like(mag) * (mag.sum() / len(mag))
        # Preserve phase
        phase = np.angle(A)
        # New a
        a_new = np.fft.irfft(target_mag * np.exp(1j * phase), N)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        a = a_new
    return a

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)

def _momentum_active_set(a_init, deadline, lr=0.01, momentum_decay=0.9,
                         tol_start=0.003, tol_end=0.0001):
    """Momentum-based active-set gradient descent on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    tol = tol_start
    no_improve = 0
    stall_breaks = 0

    while time.monotonic() < deadline - 1.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Adaptive active set: all peaks within tol of max
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Use top-k if too many active
        max_active = max(10, n // 100)
        if n_active > max_active:
            top_idx = np.argsort(c)[::-1][:max_active]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = max_active

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)

        if gnorm < 1e-16:
            break

        # Momentum update
        v = momentum_decay * v + (1.0 - momentum_decay) * grad_centered
        alpha = lr / (np.linalg.norm(v) + 1e-15)

        # Backtracking line search
        improved = False
        for _ in range(60):

            a_try = np.maximum(a_try, 0.0)

            if s_try < 1e-10:

            if sc_try < current_sc - 1e-14:

def _peak_balance(a_init, deadline, n_trials=200):
    """Surgical peak balancing: reduce top contributors to dominant peaks and
    redistribute to low-impact positions."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

            break

        c = _full_convolve(a, a)

        # Find top 2-3 peaks
        top_k = min(3, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        # For each peak, compute contribution
        for k_star in peak_indices:
            k_lo = max(0, k_star - n + 1)
            k_hi = min(n - 1, k_star)
            if k_lo > k_hi:
                continue

            i_range = np.arange(k_lo, k_hi + 1)
            j_range = k_star - i_range
            contrib = a[i_range] * a[j_range]
            total = contrib.sum()
            if total < 1e-15:
                continue

            # Reduce top contributors
            n_reduce = max(1, int(len(contrib) * 0.05))
            top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

            a_try = a.copy()
            reduction = 0.0
            for idx in top_idx:
                red = a_try[idx] * 0.15
                a_try[idx] -= red
                reduction += red

            # Distribute to positions with low contribution to ALL top peaks
            low_mask = np.ones(n, dtype=bool)
            for pk in peak_indices:
                lo = max(0, pk - n + 1)
                hi = min(n - 1, pk)
                if lo <= hi:
                    ir = np.arange(lo, hi + 1)
                    jr = pk - ir
                    c_pk = a[ir] * a[jr]
                    top_c = c_pk[np.argsort(c_pk)[::-1][:max(1, len(c_pk)//5)]]
                    for idx in ir[np.argsort(c_pk)[::-1][:max(1, len(c_pk)//5)]]:
                        low_mask[idx] = False
                        jv = pk - idx
                        if 0 <= jv < n:
                            low_mask[jv] = False

            n_low = low_mask.sum()
            if n_low == 0:

                for idx in top_idx:
                    low_mask[idx] = False
                n_low = low_mask.sum()
                if n_low == 0:

            a_try = np.maximum(a_try, 0.0)

            if s_try < 1e-10:

            a_try = a_try / s_try

            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:

                best_a = a_try.copy()
                a = a_try

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 23.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(123)
        a = rng.uniform(0.5, 1.5, n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=10)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _optimize_one_n(a_init, deadline, label=""):
    """Run the full optimization pipeline for a single n."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    t0 = time.monotonic()

    # Step 1: LSE-LBFGS
    rem = deadline - time.monotonic()
    if rem > 60:
        a1 = _run_lbfgs_adaptive(best_a, min(deadline, time.monotonic() + rem * 0.45),
                                  min_remaining=30)
        sc1 = _score(a1)
        if sc1 < best_sc:
            best_a = a1.copy()
            best_sc = sc1
            print(f"    [{label}] LSE-LBFGS: {best_sc:.10f} (n={n}) t={time.monotonic()-t0:.1f}s")

    # Step 2: Momentum active set
    rem = deadline - time.monotonic()
    if rem > 40:
        a2, sc2 = _momentum_active_set(best_a, min(deadline, time.monotonic() + rem * 0.5),
                                        lr=0.015, tol_start=0.004, tol_end=0.0001)
        if sc2 < best_sc:
            best_a = a2.copy()
            best_sc = sc2
            print(f"    [{label}] Momentum AS: {best_sc:.10f} (n={n}) t={time.monotonic()-t0:.1f}s")

    # Step 3: Peak balance
    rem = deadline - time.monotonic()
    if rem > 20:
        a3, sc3 = _peak_balance(best_a, min(deadline, time.monotonic() + rem * 0.4),
                                 n_trials=100)
        if sc3 < best_sc:
            best_a = a3.copy()
            best_sc = sc3
            print(f"    [{label}] Peak balance: {best_sc:.10f} (n={n}) t={time.monotonic()-t0:.1f}s")

    # Step 4: Quick LSE polish
    rem = deadline - time.monotonic()
    if rem > 30:
        a4 = _run_lbfgs_adaptive(best_a, min(deadline - 5, time.monotonic() + rem * 0.6),
                                  min_remaining=10)
        sc4 = _score(a4)
        if sc4 < best_sc:
            best_a = a4.copy()
            best_sc = sc4
            print(f"    [{label}] Final LSE: {best_sc:.10f} (n={n}) t={time.monotonic()-t0:.1f}s")

    # Step 5: Final momentum polish
    rem = deadline - time.monotonic()
    if rem > 15:
        a5, sc5 = _momentum_active_set(best_a, min(deadline - 2, time.monotonic() + rem * 0.7),
                                        lr=0.005, momentum_decay=0.95,
                                        tol_start=0.001, tol_end=0.00005)
        if sc5 < best_sc:
            best_a = a5.copy()
            best_sc = sc5
            print(f"    [{label}] Final momentum: {best_sc:.10f} (n={n}) t={time.monotonic()-t0:.1f}s")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search: multi-n, multi-init, combined optimization pipeline."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    best_a_global = None
    best_sc_global = np.inf

    if parent is not None:
        best_a_global = parent.copy()
        best_sc_global = _score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_sc_global:.10f}")
    else:
        n_parent = 12288
        best_a_global = _make_profile(n_parent, "uniform")
        best_sc_global = _score(best_a_global)
        print(f"No parent, uniform n={n_parent}, score={best_sc_global:.10f}")

    # === Phase 1: Optimize parent n ===
    print(f"\n[Phase 1] Optimize parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200.0, t_start + 400.0)
    if p1_deadline > t_start + 60:
        a1, sc1 = _optimize_one_n(best_a_global, p1_deadline, label=f"parent_n{n_parent}")
        if sc1 < best_sc_global:
            best_a_global = a1
            best_sc_global = sc1
            print(f"  [P1] NEW BEST: {best_sc_global:.10f} (n={n_parent})")

    # === Phase 2: Spectral init + optimize ===
    print(f"\n[Phase 2] Spectral init (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 800.0, t_start + 600.0)
    if p2_deadline > time.monotonic() + 30:
        a_spec, sc_spec = _spectral_init(n_parent, n_iters=150, seed=42)
        print(f"  Spectral init score: {sc_spec:.10f} (n={n_parent})")
        if sc_spec < best_sc_global:
            best_a_global = a_spec
            best_sc_global = sc_spec

        # Also optimize the spectral init
        if p2_deadline > time.monotonic() + 30:
            sub_deadline = min(p2_deadline, time.monotonic() + 150.0)
            a2, sc2 = _optimize_one_n(a_spec, sub_deadline, label="spectral")
            if sc2 < best_sc_global:
                best_a_global = a2
                best_sc_global = sc2
                print(f"  [P2] NEW BEST: {best_sc_global:.10f} (n={n_parent})")

    # === Phase 3: Multi-n exploration ===
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([8192, 10000, 12288, 16384, 20000, 24576]))
    # Remove if too close to what we already have
    n_best_current = len(best_a_global)
    n_to_try = [n for n in n_candidates if abs(n - n_best_current) > 500]
    n_to_try.sort(key=lambda x: abs(x - n_best_current))

    # Allocate time based on remaining budget
    remaining = deadline - time.monotonic() - 200.0  # save 200s for restarts + final
    if remaining < 60:
        n_to_try = []

    per_n_time = min(200.0, remaining / max(1, len(n_to_try))) if n_to_try else 0

    for n_try in n_to_try[:3]:  # Try at most 3 different n values
        if time.monotonic() > deadline - 200.0:
            break
        a_init = _resample(best_a_global, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}, budget={per_n_time:.0f}s")

        sub_deadline = time.monotonic() + per_n_time
        a_opt, sc_opt = _optimize_one_n(a_init, sub_deadline, label=f"n{n_try}")
        if sc_opt < best_sc_global:
            best_a_global = a_opt
            best_sc_global = sc_opt
            print(f"  [P3] NEW BEST: {best_sc_global:.10f} (n={n_try}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Diverse restarts at best n ===
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100.0:
        n_best = len(best_a_global)
        restarts = []

        # Perturbed versions
        for sigma in [0.02, 0.05, 0.1]:
            restarts.append((f"pert_{sigma}", _perturb(best_a_global, sigma, rng)))

        # Different profiles
        restarts.append(("tapered", _make_profile(n_best, "tapered")))
        restarts.append(("cos_mod", _make_profile(n_best, "cosine_mod")))
        restarts.append(("random_smooth", _make_profile(n_best, "random_smooth")))

        # Spectral with different seed
        if time.monotonic() < deadline - 80.0:
            a_s2, sc_s2 = _spectral_init(n_best, n_iters=100, seed=999)
            if sc_s2 < best_sc_global:
                restarts.insert(0, ("spectral2", a_s2))

        per_restart = min(80.0, (deadline - time.monotonic() - 60.0) / max(1, len(restarts)))
        for label, a_r in restarts:
            if time.monotonic() > deadline - 60.0:
                break
            if per_restart < 15:
                break
            a_opt, sc_opt = _optimize_one_n(a_r, time.monotonic() + per_restart, label=f"rst_{label}")
            if sc_opt < best_sc_global:
                best_a_global = a_opt
                best_sc_global = sc_opt
                print(f"  [P4] NEW BEST: {best_sc_global:.10f} (n={n_best}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final ultra-fine polish ===
    print(f"\n[Phase 5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic() - 3.0
    if remaining > 20:
        n = len(best_a_global)

        # Very tight momentum
        sub_dl = min(deadline - 10, time.monotonic() + remaining * 0.4)
        a_f1, sc_f1 = _momentum_active_set(best_a_global, sub_dl,
                                            lr=0.003, momentum_decay=0.97,
                                            tol_start=0.0008, tol_end=0.00003)
        if sc_f1 < best_sc_global:
            best_a_global = a_f1
            best_sc_global = sc_f1
            print(f"  [P5] NEW BEST: {best_sc_global:.10f} [tight_momentum]")

        # Peak balance
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 10:
            sub_dl = min(deadline - 5, time.monotonic() + remaining * 0.5)
            a_f2, sc_f2 = _peak_balance(best_a_global, sub_dl, n_trials=80)
            if sc_f2 < best_sc_global:
                best_a_global = a_f2
                best_sc_global = sc_f2
                print(f"  [P5] NEW BEST: {best_sc_global:.10f} [peak_balance]")

        # Tiny LSE
        remaining = deadline - time.monotonic() - 2.0
        if remaining > 10:
            c_best = _full_convolve(best_a_global, best_a_global)
            mb = float(c_best.max())
            beta_fin = mb * 0.00003
            z = np.log(np.maximum(best_a_global, 1e-30))
            z = z - z.mean()
            prob = _LseProblem(beta_fin, n)
            max_iter = min(2000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 60})
                a_fin = _softmax(result.x)
                a_fin = np.maximum(a_fin, 0.0)
                s_fin = a_fin.sum()
                if s_fin > 1e-10:
                    a_fin = a_fin / s_fin
                    sc_fin = _score(a_fin)
                    if sc_fin < best_sc_global:
                        best_a_global = a_fin
                        best_sc_global = sc_fin
                        print(f"  [P5] NEW BEST: {best_sc_global:.10f} [final_lse]")
            except Exception:
                pass

        # Last peak balance
        remaining = deadline - time.monotonic() - 1.0
        if remaining > 5:
            a_pb, sc_pb = _peak_balance(best_a_global, deadline - 0.5, n_trials=40)
            if sc_pb < best_sc_global:
                best_a_global = a_pb
                best_sc_global = sc_pb
                print(f"  [P5] NEW BEST: {best_sc_global:.10f} [final_peak]")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a_global, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    # Verify with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence score: {eval_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence failed: {e}")

    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
