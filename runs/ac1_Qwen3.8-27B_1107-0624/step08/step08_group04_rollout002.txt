1. Extend $v$ to the full length by zeroing out non-support entries
2. Compute $H \hat{v}$ using the FFT trick
3. Extract the support entries

This gives us a matrix-vector product with $H_{SS}$ in O(n log n) time (where n is the full length). CG can then be applied to this reduced system.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

    return _hankel_matvec(v_full, W, n)[support]

def _dense_hankel_solve(W, n, support, rhs):
    """Dense solve of H_SS * x = rhs where H_SS is the support submatrix."""
    idx = support
    k = len(idx)
    if k == 0:
        return np.array([])
    # Form H_SS: (H_SS)_{pq} = 2 * W[idx[p] + idx[q]]
    i_idx = idx[:, None]
    j_idx = idx[None, :]
    H_ss = 2.0 * W[i_idx + j_idx]
    # Solve using Cholesky (H_ss is SPD if W is from a valid autocorrelation window)
    try:
        L = np.linalg.cholesky(H_ss + 1e-15 * np.eye(k))
        y = np.linalg.solve(L, rhs)
        x = np.linalg.solve(L.T, y)
        return x
    except np.linalg.LinAlgError:
        # Fallback to pinv
        x, _, _, _ = np.linalg.lstsq(H_ss, rhs, rcond=None)
        return x

        else:
            resid = np.inf
        
        if resid < 1e-10 * (np.max(np.abs(grad)) + 1e-15):
            break
    
    return best_a


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs_deep(a_init, deadline, min_remaining=60.0):
    """Deep L-BFGS optimization with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.001)

    # Very aggressive beta schedule
    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625,
                       0.0078, 0.0039, 0.00195, 0.00098, 0.00049,
                       0.00024, 0.00012, 0.00006, 0.00003, 0.000015,
                       0.000007, 0.0000035, 0.000001, 0.0000005,
                       0.0000002, 0.0000001, 0.00000005])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(100, int(remaining / (est_time * 5))))
        
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _active_set_optimize(a_init, deadline, min_remaining=30.0):
    """Progressive active-set QP optimization with tightening tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    # Progressive tolerance schedule
    tols = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
            0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
            0.0000005, 0.0000002, 0.0000001]
    
    rounds_per_tol = 15
    
    for tol_idx, tol in enumerate(tols):
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        
        for rnd in range(rounds_per_tol):
            if time.monotonic() > deadline - min_remaining:
                break
            
            c = _full_convolve(a, a)
            c_max = float(c.max())
            if c_max < 1e-30:
                break
            
            # Build active set
            active = c >= (c_max - tol * c_max)
            n_act = int(active.sum())
            if n_act == 0:
                active[int(np.argmax(c))] = True
                n_act = 1
            
            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / n_act
            
            # Solve QP
            a_qp = _solve_qp(W, n, a, tol_active=tol, max_outer=100)
            sc_qp = _score(a_qp)
            
            if sc_qp < best_sc - 1e-15:
                best_sc = sc_qp
                best_a = a_qp.copy()
                a = a_qp.copy()
            else:
                # Damped step
                a_try = 0.5 * a + 0.5 * a_qp
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
            
            # Check convergence
            c_check = _full_convolve(a, a)
            flatness = float(c_check.max() / (c_check.sum() / len(c_check)))
            if flatness < 1.0 + tol * 0.1:
                break
    
    return best_a, best_sc


def _peak_shave(a_init, deadline, min_remaining=20.0, max_iters=500):
    """Direct peak shaving: iteratively reduce the largest autocorrelation peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - min_remaining:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(c.sum() / len(c))
        
        if c_max <= c_mean * (1 + 1e-8):
            break
        
        # Identify peaks to shave
        threshold = c_max - 0.001 * c_max
        peaks = c >= threshold
        n_peaks = int(peaks.sum())
        
        # Gradient of sum of peak values
        W = np.zeros(2 * n - 1)
        W[peaks] = 1.0 / n_peaks
        
        # grad_a = H * a where H_{ij} = 2*W[i+j]
        grad = _hankel_matvec(a, W, n)
        
        # Normalized gradient direction (project onto sum-preserving subspace)
        lam = float(grad.mean())
        grad_proj = grad - lam
        
        # Line search
        a_old = a.copy()
        alpha = 0.5
        improved = False
        for ls in range(30):
            a_new = a - alpha * grad_proj
            a_new = np.maximum(a_new, 0)
            s_new = float(a_new.sum())
            if s_new < 1e-10:
                alpha *= 0.5
                continue
            a_new = a_new / s_new
            c_new = _full_convolve(a_new, a_new)
            if float(c_new.max()) < c_max - 1e-12:
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if not improved:
            # Try smaller step
            alpha = 1e-4
            a_new = a - alpha * grad_proj
            a_new = np.maximum(a_new, 0)
            s_new = float(a_new.sum())
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc = _score(a_new)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    a = a_new
        
        # Adaptive step size reduction
        if it % 50 == 49:
            alpha *= 0.8
    
    return best_a, best_sc


def _make_profile(n, kind="sin_taper"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "broad_gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2) + 0.1
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        # Flat top with smooth edges
        edge = 0.1
        a = np.ones(n)
        for i in range(n):
            if t[i] < edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * t[i] / edge))
            elif t[i] > 1 - edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * (1 - t[i]) / edge))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Initialize from parent or good profile
    2. L-BFGS with log-sum-exp smoothing
    3. Active-set QP with progressive tightening
    4. Peak shaving
    5. Repeat 2-4 for convergence
    6. Multi-n exploration
    7. Final ultra-tight polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "sin_taper")
        print(f"No parent, using sin_taper n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs_deep(parent, deadline, min_remaining=400.0)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: Active-set QP
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    a2, sc2 = _active_set_optimize(best_seq, deadline, min_remaining=300.0)
    update_best(a2, sc2, "qp_1")

    # Phase 3: Another L-BFGS pass
    print(f"\n[Phase 3] L-BFGS pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a3 = _run_lbfgs_deep(best_seq, deadline, min_remaining=200.0)
        update_best(a3, _score(a3), "lbfgs_2")

    # Phase 4: Active-set QP round 2 (tighter)
    print(f"\n[Phase 4] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    a4, sc4 = _active_set_optimize(best_seq, deadline, min_remaining=150.0)
    update_best(a4, sc4, "qp_2")

    # Phase 5: Peak shaving
    print(f"\n[Phase 5] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a5, sc5 = _peak_shave(best_seq, min(deadline - 80, time.monotonic() + 80),
                              min_remaining=80.0, max_iters=300)
        update_best(a5, sc5, "peak_shave")

    # Phase 6: Final QP polish (ultra-tight)
    print(f"\n[Phase 6] Final QP (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        a6, sc6 = _active_set_optimize(best_seq, deadline, min_remaining=30.0)
        update_best(a6, sc6, "qp_final")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 40:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "sin_taper")
        a_opt = _run_lbfgs_deep(a_init, deadline, min_remaining=30.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        
        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 20:
            a_np, sc_np = _active_set_optimize(a_opt, deadline, min_remaining=15.0)
            update_best(a_np, sc_np, f"n_{n_try}_qp")

    # Phase 8: Try different profiles at parent n
    print(f"\n[Phase 8] Profile variations (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        for prof in ["flat_top", "broad_gaussian", "cos_taper"]:
            if time.monotonic() > deadline - 20:
                break
            a_prof = _resample(best_seq, n_parent)  # Start from best
            # Mix with profile
            p_base = _make_profile(n_parent, prof)
            a_mix = 0.7 * a_prof + 0.3 * p_base
            a_mix = np.maximum(a_mix, 0)
            a_mix = a_mix / a_mix.sum()
            a_opt = _run_lbfgs_deep(a_mix, deadline, min_remaining=15.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"prof_{prof}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "sin_taper")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
