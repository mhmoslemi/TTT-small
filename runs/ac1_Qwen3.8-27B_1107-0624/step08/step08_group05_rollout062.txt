"""
Multi-strategy optimizer combining Gerchberg-Saxton (HIO) flat-spectrum iteration,

import numpy as np

from scipy.optimize import minimize

_best_seq = None
_best_score = np.inf


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)

        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)

    else:
        return np.ones(n) / n


def _gs_hio_iterate(a_init, deadline, n_steps=200, relaxation=0.7, rho_hio=1.3,
                    target_sharpness=1.0, check_interval=10):
    """Gerchberg-Saxton with HIO for flat anti-correlation target."""
    global _best_seq, _best_score
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    L = 2 * n - 1

    # Compute target: flat in time domain -> specific spectrum
    c_target = np.zeros(N)
    c_target[:L] = 1.0 / L
    C_target = np.fft.fft(c_target)
    X_target_mag = np.sqrt(np.abs(C_target))
    X_target_mag = np.maximum(X_target_mag, 0)

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    if best_sc < _best_score:
        _best_score = best_sc
        _best_seq = best_a.copy()

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        x = np.zeros(N)
        x[:n] = a
        X = np.fft.fft(x)

        X_mag = np.abs(X)
        X_mag_new = (1.0 - relaxation) * X_mag + relaxation * X_target_mag
        X_phases = np.angle(X)
        X_new = X_mag_new * np.exp(1j * X_phases)

        x_new = np.fft.ifft(X_new)
        a_raw = np.real(x_new[:n])

        # HIO projection onto non-negativity
        a_gs = np.maximum(a_raw, 0)
        if rho_hio != 1.0:
            bad = a_raw < 0
            a_hio = a_gs.copy()
            a_hio[bad] = a_raw[bad] * (1.0 - rho_hio)
            # Ensure positivity after HIO
            a_hio = np.maximum(a_hio, 0)
            a_new = a_hio
        else:
            a_new = a_gs

        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        if step % check_interval == 0 or step == n_steps - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                if best_sc < _best_score:
                    _best_score = best_sc
                    _best_seq = best_a.copy()

    return best_a, best_sc

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _hess_matvec(v, active, n, N=None):
    if N is None:
        N = _next_pow2(2 * n - 1)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002, mu=1e-8):
    global _best_seq, _best_score
    n = len(a_init)

    if s < 1e-15:

        if time.monotonic() > deadline:
            break

        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)
        grad = _hess_matvec(a, active_f, n, N)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        def _hv(v):
            return _hess_matvec(v, active_f, n, N) + mu * v

        dx = dx - np.mean(dx)

        alpha = 1.0

            c_try = _full_convolve(a_try, a_try)

                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:

            best_a = a.copy()
            if best_sc < _best_score:

                _best_seq = best_a.copy()

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06,
                                  0.03, 0.015, 0.007, 0.003, 0.001])
    betas = betas[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-13, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if current_sc < _best_score:
                    _best_score = current_sc
                    _best_seq = current_a.copy()
        except Exception:
            pass

    return current_a, current_sc


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    global _best_seq, _best_score
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                if best_sc < _best_score:
                    _best_score = best_sc
                    _best_seq = best_a.copy()
            else:
                patience += 1
                lr *= 0.93
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer: GS/HIO flat-spectrum iteration + active-set Newton
    + LSE L-BFGS + multi-n exploration + diversified restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: GS/HIO Flat-Spectrum (600s) ===
    print(f"\n[Phase 1] GS/HIO flat-spectrum iteration")
    gs_deadline = min(deadline, t_start + 600)

    # Try multiple initializations with GS
    init_profiles = ["uniform", "tapered", "cos2", "raised_cos", "gauss"]
    for kind in init_profiles:
        if time.monotonic() > gs_deadline - 60:
            break
        n_try = n_parent if n_parent > 0 else 16384
        a_init = _make_profile(n_try, kind)
        a_gs, sc_gs = _gs_progressive(a_init, gs_deadline - 30, max_steps=300)
        upd(a_gs, sc_gs, f"gs_{kind}")

    # Also run GS from parent if available
    if parent is not None and time.monotonic() < gs_deadline - 60:
        a_gs_p = _gs_hio_iterate(parent, gs_deadline - 20, n_steps=200,
                                  relaxation=0.7, rho_hio=1.3)
        upd(a_gs_p[0], a_gs_p[1], "gs_parent")

    # === Phase 2: Newton active-set polish (300s) ===
    print(f"\n[Phase 2] Newton active-set polish")
    if _best_seq is not None and time.monotonic() < deadline - 300:
        nw_dead = min(deadline - 280, time.monotonic() + 300)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=300,
                                          tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 3: LSE L-BFGS (300s) ===
    print(f"\n[Phase 3] LSE L-BFGS")
    if _best_seq is not None and time.monotonic() < deadline - 280:
        a_lse, sc_lse = _lbfgs_lse(_best_seq, deadline, min_remaining=250, n_stages=10)
        upd(a_lse, sc_lse, "lbfgs_lse")

    # === Phase 4: Alternating GS + Newton (200s) ===
    print(f"\n[Phase 4] Alternating GS + Newton")
    if _best_seq is not None and time.monotonic() < deadline - 180:
        a_cur = _best_seq.copy()
        for rnd in range(4):
            if time.monotonic() > deadline - 160:
                break
            # GS step
            gs_dead = min(deadline - 160, time.monotonic() + 30)
            a_gs2, _ = _gs_hio_iterate(a_cur, gs_dead, n_steps=50,
                                        relaxation=0.8, rho_hio=1.4)
            if _score(a_gs2) < _score(a_cur) - 1e-8:
                a_cur = a_gs2
            upd(a_cur, _score(a_cur), f"alt_gs_{rnd}")
            # Newton step
            nw_dead2 = min(deadline - 160, time.monotonic() + 30)
            a_nw2, _ = _newton_active_set(a_cur, nw_dead2, max_iter=80,
                                           tol_frac=0.002)
            if _score(a_nw2) < _score(a_cur) - 1e-8:
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_nw_{rnd}")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n[Phase 5] Multi-n exploration")
    if _best_seq is not None and time.monotonic() < deadline - 160:
        n_candidates = [8192, 16384, 32768]
        if n_parent not in n_candidates and n_parent >= 4096:
            n_candidates.append(n_parent)
        n_candidates = sorted(set(n_candidates))

        for n_try in n_candidates:
            if time.monotonic() > deadline - 120:
                break
            if len(_best_seq) == n_try:
                a_init = _best_seq.copy()
            else:
                a_init = _resize_sequence(_best_seq, n_try)

            # Quick GS
            a_ms, sc_ms = _gs_hio_iterate(a_init, min(deadline - 120,
                                                       time.monotonic() + 60),
                                           n_steps=100, relaxation=0.7, rho_hio=1.3)
            upd(a_ms, sc_ms, f"mn_{n_try}_gs")

            # Quick Newton
            if time.monotonic() < deadline - 100:
                a_ms2, sc_ms2 = _newton_active_set(a_ms, min(deadline - 100,
                                                              time.monotonic() + 40),
                                                    max_iter=60, tol_frac=0.002)
                upd(a_ms2, sc_ms2, f"mn_{n_try}_nw")

    # === Phase 6: Diversified restarts (200s) ===
    print(f"\n[Phase 6] Diversified restarts")
    if _best_seq is not None and time.monotonic() < deadline - 100:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)
            elif i == 2:
                p_start = _perturb_block_swap(_best_seq, rng, n_swaps=4)
            elif i == 3:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.08)
            elif i == 4:
                p_start = _perturb_log(_best_seq, 0.05, rng)
            elif i == 5:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=16, max_amp=0.5)
            elif i == 6:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.15)
            else:
                p_start = _perturb_log(_best_seq, 0.03, rng)

            # Quick GS from perturbed
            a_r, sc_r = _gs_hio_iterate(p_start, min(deadline - 50,
                                                      time.monotonic() + 30),
                                         n_steps=80, relaxation=0.8, rho_hio=1.3)
            upd(a_r, sc_r, f"restart_gs_{i}")

            # Newton polish
            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 30:
                a_r2, sc_r2 = _newton_active_set(a_r, min(deadline - 30,
                                                           time.monotonic() + 20),
                                                  max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_nw_{i}")

    # === Phase 7: Final aggressive polish (remaining time) ===
    print(f"\n[Phase 7] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        a_fin, sc_fin = _newton_active_set(_best_seq, deadline - 3,
                                            max_iter=100, tol_frac=0.0005)
        upd(a_fin, sc_fin, "final_newton")

        # Subgradient with very small tolerance
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(_best_seq, deadline - 2,
                                               n_steps=3000, lr_init=0.003,
                                               tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_sg")

        # One more very low beta LSE
        if time.monotonic() < deadline - 5:
            n_fin = len(_best_seq)
            c_fin = _full_convolve(_best_seq, _best_seq)
            mb = float(c_fin.max())
            z_fin = np.log(np.maximum(_best_seq, 1e-30))
            z_fin = z_fin - z_fin.mean()
            remaining = deadline - 5 - time.monotonic()
            if remaining > 5:
                beta_final = mb * 0.0001
                max_iter = min(1500, max(50, int(remaining / 0.005)))
                try:
                    result = minimize(
                        _lse_objective, z_fin, args=(beta_final, n_fin),
                        jac=_lse_gradient, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16,
                                 'gtol': 1e-14, 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    sc_fin3 = _score(a_fin3)
                    upd(a_fin3, sc_fin3, "final_ultra_lse")
                except Exception:
                    pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
