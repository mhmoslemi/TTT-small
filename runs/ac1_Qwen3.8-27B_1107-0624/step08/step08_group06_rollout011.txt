Optimizer using adaptive re-weighted descent, multi-scale initialization,

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr_a_A(a, A_vec, n):
    """Compute the 'gradient' component: correlation of a with A_vec."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]

    rho = ind[cond][-1]

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    # More betas, going to much smaller values
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                                  0.015625, 0.0078125, 0.00390625, 0.001953125,
                                  0.0009765625, 0.00048828125, 0.000244140625,
                                  0.0001220703125, 0.00006103515625,
                                  0.000030517578125, 0.0000152587890625,
                                  0.000007629394531])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-13,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _active_set_newton_v2(a_init, deadline, max_iter=300):
    """
    Active-set Newton method that directly enforces KKT conditions.
    At the optimum, for the active set S, the gradient should be constant
    (equal to the Lagrange multiplier) on the support of a.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Determine active set: top values within tolerance of max
        # Use multiple tolerance levels
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 8 * n:
                continue
            
            A_vec = active.astype(np.float64) / m
            
            # Compute gradient: 2 * corr(a, A_vec)
            g = 2.0 * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            
            # Newton direction: solve (H + mu*I) * dx = -g_t with sum constraint
            mu = 1e-6 / m + 1e-10
            
            def hess_vp(v):
                return 2.0 * _corr_a_A(v, A_vec, n) + mu * v
            
            # Use conjugate gradient
            x = np.zeros(n)
            r = -g_t
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(150):
                Ap = hess_vp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p
                r = r - alpha * Ap
                rs_new = np.dot(r, r)
                if np.sqrt(rs_new) < 1e-12 * np.sqrt(np.dot(-g_t, -g_t)) + 1e-15:
                    break
                p = r + (rs_new / rs) * p
                rs = rs_new
            
            # Handle the sum constraint: dx should have zero sum
            # Project: dx -= mean(dx)
            dx = x - np.mean(x)
            
            # Line search
            f_old = c_max
            t = 1.0
            for ls in range(40):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t:
                    improvement = f_old - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            # Try gradient step
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            t = 0.05
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_a_new = a_try.copy()
                        best_improvement = 1
                        break
                t *= 0.5
        
        if best_a_new is None:
            break
        
        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break
    
    return best_a, best_sc

def _aggressive_deflation(a, deadline, n_attempts=50):
    """
    More aggressive deflation: reduce multiple peaks simultaneously
    using a global optimization step.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find all peaks above threshold
        threshold = c_max * 0.995
        peak_positions = np.where(c >= threshold)[0]
        
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(c)])
        
        # For each peak, compute the "deflation direction"
        # We want to reduce a[i] for the i that contributes most to each peak
        deflation = np.zeros(n)
        for k_pos in peak_positions:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            # Contribution of a[i] to c[k_pos] is a[i] * a[k_pos - i]
            # We want to reduce the product, so reduce the larger factor
            for i in range(i_lo, i_hi + 1):
                j = k_pos - i
                if 0 <= j < n:
                    # Gradient of c[k_pos] w.r.t. a[i] is a[j]
                    quotient = a[j] / (a[i] + 1e-30)
                    deflation[i] += quotient
        
        # Normalize deflation
        deflation = deflation / (np.max(np.abs(deflation)) + 1e-30)
        
        # Apply deflation with various strengths
        for strength in [0.001, 0.005, 0.01, 0.02, 0.05]:
            a_try = a * (1.0 - strength * deflation)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                break
    
    return best_a, best_sc

def _multiscale_init(parent, n_target):
    """
    Multi-scale initialization: downsample parent, optimize coarsely, upsample.
    """
    n_parent = len(parent)
    
    if n_target == n_parent:
        return parent.copy()
    
    # Downsample to n_target/4, optimize, upsample to n_target
    n_coarse = max(64, n_target // 4)
    
    x_old = np.linspace(0, 1, n_parent)
    x_coarse = np.linspace(0, 1, n_coarse)
    a_coarse = np.interp(x_coarse, x_old, parent)
    a_coarse = np.maximum(a_coarse, 0)
    s = a_coarse.sum()
    if s > 1e-15:
        a_coarse = a_coarse / s
    
    # Interpolate to target
    x_new = np.linspace(0, 1, n_target)
    a_fine = np.interp(x_new, x_coarse, a_coarse)
    a_fine = np.maximum(a_fine, 0)
    s = a_fine.sum()
    if s > 1e-15:
        a_fine = a_fine / s
    
    return a_fine

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2*block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        return a
    elif kind == "block3":
        a = np.zeros(n)
        block = n // 5
        a[:block] = 1.0
        a[2*block:3*block] = 1.0
        a[4*block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        return a
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)


def _lsobj_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_factory(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=200, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _newton_step(a, c, c_max, tol_frac):
    """Compute one Newton step given active set defined by tol_frac."""
    n = len(a)
    active = c >= (c_max - tol_frac * c_max)
    if not active.any():
        active[np.argmax(c)] = True
    m = int(active.sum())
    if m < 2 or m > 10 * n:
        return None, None
    
    A_vec = active.astype(np.float64) / m
    g = 2.0 * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None, None
    
    mu = 1e-7 / m + 1e-10
    
    # Use a closure-free approach for the matvec
    A_vec_copy = A_vec
    n_copy = n
    mu_copy = mu
    
    def _hess_vp(v):
        return 2.0 * _corr_a_A(v, A_vec_copy, n_copy) + mu_copy * v
    
    u = _cg_solve(_hess_vp, g_t, max_cg=200)
    w = _cg_solve(_hess_vp, np.ones(n), max_cg=200)
    w_sum = float(w.sum())
    
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w
    
    dx = dx - np.mean(dx)
    return dx, A_vec


def _active_set_two_phase(a_init, deadline, max_outer=100):
    """
    Two-phase active-set Newton:
    Phase 1: Coarse active set (loose tolerance) to get close
    Phase 2: Fine active set (tight tolerance) for precise KKT
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Phase 1: Try loose tolerances
        # Phase 2: Try tight tolerances
        if outer < max_outer // 2:
            tol_list = [0.02, 0.01, 0.005, 0.002, 0.001]
        else:
            tol_list = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        
        best_improvement = 0
        best_a_new = None
        
        for tol_frac in tol_list:
            dx, A_vec = _newton_step(a, c, c_max, tol_frac)
            if dx is None:
                continue
            
            # Line search
            t = 1.0
            for ls in range(50):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < c_max - 1e-14 * t:
                    improvement = c_max - f_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_a_new = a_try.copy()
                    break
                t *= 0.5
        
        if best_a_new is None:
            # Gradient fallback
            g_all = 2.0 * _corr_a_A(a, np.ones(2 * n - 1) / (2 * n - 1), n)
            g_t = g_all - np.mean(g_t) if hasattr(g_all, 'det') else g_all - np.mean(g_all)
            t = 0.1
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_a_new = a_try.copy()
                        best_improvement = 1
                        break
                t *= 0.5
        
        if best_a_new is None:
            break
        
        a = best_a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-15:
            break
    
    return best_a, best_sc


def _multi_peak_deflation(a, deadline, n_attempts=80):
    """
    Simultaneous multi-peak deflation.
    Computes a global deflation direction that reduces all significant peaks at once.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find all peaks that are within 99% of max
        threshold = c_max * 0.99
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        
        if n_peaks == 0:
            peak_mask[np.argmax(c)] = True
            n_peaks = 1
        
        # Compute weighted deflation: for each peak position k,
        # the gradient of c[k] w.r.t. a is 2 * corr(e_k, a) restricted to valid range
        # Simplification: deflation[i] = sum_{k in peaks} w[k] * a[k-i] (where valid)
        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks
        
        # The gradient of sum_k w[k] * c[k] w.r.t. a is 2 * corr(a, w)
        grad = 2.0 * _corr_a_A(a, w, n)
        g_t = grad - np.mean(grad)
        
        # Apply deflation with various strengths
        applied = False
        for strength in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]:
            a_try = a - strength * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                applied = True
                break
        
        if not applied:
            # Try smaller perturbations
            rng_local = np.random.default_rng(attempt)
            for _ in range(5):
                a_try = a * (1.0 + 0.01 * rng_local.standard_normal(n))
                a_try = _project_simplex(a_try)
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-13:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    applied = True
                    break
            if not applied:
                break
    
    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):
    """

    Goes to much smaller beta values than before.
    """

    base_beta = max(gap * 0.3, max_c * 0.002)
    
    # Very aggressive schedule: go to beta = max_c * 1e-7
    n_betas = 25
    log_betas = np.linspace(np.log(base_beta), np.max([np.log(max_c * 1e-7), -30]), n_betas)
    betas = np.exp(log_betas)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

                method='L-BFGS-B',

        except Exception as ex:

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2 * block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        return a
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_seq(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refinement_loop(a, deadline, max_rounds=10):
    """
    Iterate between Newton and deflation for refinement.
    """

    best_a = a.copy()

    for rnd in range(max_rounds):

            a = a_newton

            a = best_a

            a = a_def

        if abs(sc_current - best_sc) < 1e-15:
            break

        update_best(a_pol, _score(a_pol), "lbfgs_polish")
    
    # === Phase 6: Try n=32768 ===
    print(f"\n[Phase 6] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        n_large = 32768
        a_large = _interpolate_seq(best_seq, n_large) if best_seq is not None else _make_profile(n_large, "uniform")
        sc_large_init = _score(a_large)
        print(f"  n={n_large} init score: {sc_large_init:.12f}")
        
        # Quick L-BFGS on large n
        lbfgs3_dl = min(deadline - 100, time.monotonic() + 80)
        a_large_opt = _lbfgs_continuation(a_large, lbfgs3_dl, min_time_remaining=100.0)
        sc_large = _score(a_large_opt)
        update_best(a_large_opt, sc_large, "n_32768")
    
    # === Phase 7: Diversified restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for sigma, label in [(0.005, "tiny"), (0.02, "small"), (0.05, "med"), (0.1, "large")]:
            if time.monotonic() > deadline - 50:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _lbfgs_continuation(p_start, deadline, min_time_remaining=40.0)
            update_best(a_r, _score(a_r), f"restart_{label}")
    
    # === Phase 8: Final polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Final refinement
        ref_final = min(deadline - 10, time.monotonic() + 50)
        a_fin, sc_fin = _refinement_loop(best_seq, ref_final, max_rounds=5)
        update_best(a_fin, sc_fin, "final_ref")
        
        # Very small beta L-BFGS
        if time.monotonic() < deadline - 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 5e-8
            remaining = deadline - 8 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(2000, max(50, int(remaining / 0.06)))
                try:
                    def _obj_final(z_val):
                        return _lsobj_factory(z_val, beta_final, n)
                    
                    def _grad_final(z_val):
                        return _lsgrad_factory(z_val, beta_final, n)
                    
                    result = minimize(
                        _obj_final,
                        z,
                        jac=_grad_final,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
