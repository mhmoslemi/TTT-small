"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive log-sum-exp L-BFGS with continuation
2. Peak-equalization Newton via CG with FFT mat-vec
3. Multi-scale n exploration (4096, 8192, 16384)
4. Bump-shaped parameterization for diverse initializations
5. Peak peeling coordinate descent for final polish
6. Population-based restarts with varied perturbations

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    """Compute the correlation: result[j] = sum_k A_vec[k] * a[|k-j|] (with bounds)."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _grad_active(a, w, n):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, np.zeros(2*n-1)))  # placeholder
    # Actually we need: grad[j] = 2*sum_k w[k]*a[k-j] (approximately)
    # which is 2 * (w flipped) convolved with a, evaluated at positions 0..n-1
    # Let's use the _corr_a_A approach
    h2 = _corr_a_A(a, w, n)
    return 2.0 * h2


def _hess_matvec_active(v, w, n):
    """Hessian-vector product for sum_k w[k]*conv(a,a)[k]."""
    return 2.0 * _corr_a_A(v, w, n)


def _peak_equalize_newton(a_init, deadline, max_iter=400, tol_peak=0.0005, max_active=None):
    """
    Peak-equalization Newton method.
    Finds direction that equalizes top peaks while keeping sum(a)=1 and a>=0.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if max_active is None:
        max_active = min(4 * n, 2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_peak * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > max_active:
            top_idx = np.argsort(c)[::-1][:max_active]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = max_active

        # Equal weights on active peaks
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (with bounds)
        g = 2.0 * _corr_a_A(a, w, n)

        # Center the gradient (to maintain sum constraint)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        # Hessian: H[v] = 2 * (w flipped) * v (correlation)
        # Add regularization for stability
        mu = 1e-8 * c_max / max(n_active, 1) + 1e-10

        def hess_matvec(v):
            return 2.0 * _corr_a_A(v, w, n) + mu * v

        # CG solve: H * dx = -g_t (with sum(dx) = 0 constraint via projection)
        # First solve H * u = g_t
        x = np.zeros(n)
        r = g_t - hess_matvec(x)
        p = r.copy()
        rs = float(np.dot(r, r))

        max_cg = min(300, max(50, n // 10))
        for cg_it in range(max_cg):
            Ap = hess_matvec(p)
            pAp = float(np.dot(p, Ap))
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r = r - alpha * Ap
            rs_new = float(np.dot(r, r))
            if np.sqrt(rs_new) < 1e-12 * (1.0 + float(np.linalg.norm(g_t))):
                break
            p = r + (rs_new / rs) * p
            r = r
            rs = rs_new

        u = x  # H^{-1} * g_t

        # Solve for sum constraint: H * wv = 1
        x2 = np.zeros(n)
        r2 = np.ones(n) - hess_matvec(x2)
        p2 = r2.copy()
        rs2 = float(np.dot(r2, r2))
        for cg_it in range(max_cg):
            Ap2 = hess_matvec(p2)
            pAp2 = float(np.dot(p2, Ap2))
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            x2 += alpha2 * p2
            r2 = r2 - alpha2 * Ap2
            rs2_new = float(np.dot(r2, r2))
            if np.sqrt(rs2_new) < 1e-12:
                break
            p2 = r2 + (rs2_new / rs2) * p2
            rs2 = rs2_new

        wv = x2
        sum_wv = float(wv.sum())
        if abs(sum_wv) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_wv
            dx = -u + lam * wv
            dx = dx - np.mean(dx)

        # Line search with backtracking
        f_old = c_max
        accepted = False
        t = 1.0
        # Enforce non-negativity
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
            t = min(t, t_max)

        for ls in range(50):
            if t < 1e-15:
                break
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())

            if f_try < f_old - 1e-13 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Gradient step fallback
            t = 0.05
            for ls in range(30):
                if t < 1e-15:
                    break
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _peak_peel(a_init, deadline, max_rounds=200):
    """
    Peak peeling: iteratively find the highest peak and take a step to reduce it.
    Simpler than full Newton but can be effective for final polishing.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks (use a window around the max)
        n_peaks = min(20, len(c))
        idx = np.argsort(c)[::-1][:n_peaks]
        # Weight by proximity to max
        weights = np.zeros(2 * n - 1)
        for i, k in enumerate(idx):
            weights[k] = np.exp(-0.5 * (c[k] - c_max) / (c_max * 0.01 + 1e-30))
        weights /= weights.sum()

        # Gradient direction to reduce these peaks
        g = 2.0 * _corr_a_A(a, weights, n)
        g = g - g.mean()  # maintain sum constraint
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = 0.1 / (1.0 + 0.005 * rnd)
        for ls in range(20):
            a_try = a - step * g / gnorm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        if abs(float(c.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _run_ls_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with adaptive log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        n_betas = 20
    else:
        n_betas = 14

    # Use geometric schedule with faster initial decrease
    betas = base_beta * (0.6 ** np.arange(n_betas, dtype=np.float64))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Estimate time per iteration
        if n <= 4096:
            est_time_per_iter = 0.004
        elif n <= 8192:
            est_time_per_iter = 0.008
        elif n <= 16384:
            est_time_per_iter = 0.02
        else:
            est_time_per_iter = 0.04 * (n / 32768.0)

        max_iter = min(8000, max(50, int(remaining / est_time_per_iter)))

        # For early betas (large beta), use fewer iterations
        if i < n_betas // 4:
            max_iter = min(max_iter, 500)
        elif i < n_betas // 2:
            max_iter = min(max_iter, 2000)

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                    'iprint': -1,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 4 == 0:
                    print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception:
            pass

    return current_a


def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.04):
    """Mirror descent in log-space with top-k peak weighting."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-k peaks with exponential weighting
        k = min(50, len(c) // 20)
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros(2 * n - 1)
        for idx in top_k:
            w[idx] = np.exp(-(c.max() - c[idx]) / (c_max * 0.005 + 1e-30))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        g = 2.0 * _corr_a_A(a, w, n)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * g / max(gnorm, 1e-30) * np.sqrt(n)
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _bump_init(n, center=0.5, width=0.3, asym=0.0):
    """Create a bump-shaped initialization."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - center) / width) ** 2)
    if asym != 0.0:
        a *= np.exp(asym * (t - center))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent_polish(a_init, deadline, max_rounds=500):
    """
    Coordinate descent: for each position, find the optimal value
    that minimizes the max convolution while keeping sum = 1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Precompute which convolution indices are affected by each a[i]
    # a[i] affects conv[k] for k in [i, i+n-1] (specifically conv[i+j] for j in [0,n-1] where a[i+j] exists)
    # Actually a[i] affects conv[k] where k-i is in [0, n-1], i.e., k in [i, i+n-1]
    # and also where i is in [0, n-1] and k-i in [0, n-1]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        # Randomize order for diversity
        if rnd % 10 == 0:
            order = rng_cd.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() > deadline:
                return best_a, best_sc

            # Conv indices affected by a[i]: k such that 0 <= k-i < n and 0 <= i < n
            # conv[k] includes term a[i]*a[k-i]
            # So k ranges from i to i+n-1 (but bounded by 0 to 2n-2)
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            # Current max
            c = _full_convolve(a, a)
            c_max = c.max()

            # Try to reduce a[i] slightly
            # New contribution to conv[k] from a[i]: a[i]*a[k-i]
            # If we change a[i] to a[i]+delta, conv[k] changes by delta*a[k-i]
            # We want to find delta that minimizes max of affected conv values

            # Simple approach: binary search on a[i]
            old_val = a[i]
            # Try reducing
            for frac in [0.5, 0.7, 0.85, 0.95]:
                a[i] = old_val * frac
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    break
                a_norm = a / s_new
                c_new = _full_convolve(a_norm, a_norm)
                if c_new.max() < c_max - 1e-15:
                    a = a_norm
                    best_sc_check = _score(a)
                    if best_sc_check < best_sc:
                        best_sc = best_sc_check
                        best_a = a.copy()
                    improved = True
                    break
            a[i] = old_val  # need to restore if we didn't update

            if not improved and rnd > 100:
                # Try increasing some other coordinate
                break

        if not improved:
            break

    return best_a, best_sc


# Global for coordinate descent
rng_cd = np.random.default_rng(12345)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining:
    - Adaptive L-BFGS with log-sum-exp continuation
    - Peak-equalization Newton (CG with FFT)
    - Peak peeling coordinate descent
    - Mirror descent
    - Multi-n exploration
    - Diverse restarts
    """
    global rng_cd
    rng_cd = np.random.default_rng(int(seed) + 777)

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick peak-equalization on parent ===
    print(f"\n[Phase 1] Peak-equalize parent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_pe, sc_pe = _peak_equalize_newton(parent, deadline - 600, max_iter=200, tol_peak=0.001)
        update_best(a_pe, sc_pe, "peak_eq_parent")

    # === Phase 2: L-BFGS on parent ===
    print(f"\n[Phase 2] L-BFGS on parent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_lb = _run_ls_lbfgs(best_seq, deadline - 400, min_time_remaining=200.0, aggressive=True)
        update_best(a_lb, _score(a_lb), "lbfgs_parent")

    # === Phase 3: Peak-equalize after L-BFGS ===
    print(f"\n[Phase 3] Peak-equalize after L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_pe2, sc_pe2 = _peak_equalize_newton(best_seq, deadline - 200, max_iter=300, tol_peak=0.0005)
        update_best(a_pe2, sc_pe2, "peak_eq_after_lbfgs")

    # === Phase 4: Try n=8192 ===
    print(f"\n[Phase 4] n=8192 exploration (t={time.monotonic()-t_start:.1f}s)")
    n_mid = 8192
    if time.monotonic() < deadline - 200:
        if n_parent != n_mid:
            a_mid_init = _interpolate_to_n(best_seq, n_mid)
        else:
            a_mid_init = best_seq.copy()
        sc_mid = _score(a_mid_init)
        print(f"  n={n_mid} init score: {sc_mid:.10f}")

        a_mid = _run_ls_lbfgs(a_mid_init, deadline - 150, min_time_remaining=60.0, aggressive=True)
        update_best(a_mid, _score(a_mid), f"lbfgs_n{n_mid}")

        a_mid_pe, sc_mid_pe = _peak_equalize_newton(a_mid, deadline - 100, max_iter=200, tol_peak=0.0005)
        update_best(a_mid_pe, sc_mid_pe, f"peak_eq_n{n_mid}")

    # === Phase 5: Try n=4096 (easier to optimize, might find better local min) ===
    print(f"\n[Phase 5] n=4096 exploration (t={time.monotonic()-t_start:.1f}s)")
    n_small = 4096
    if time.monotonic() < deadline - 150:
        a_small_init = _interpolate_to_n(best_seq, n_small)
        sc_small = _score(a_small_init)
        print(f"  n={n_small} init score: {sc_small:.10f}")

        a_small = _run_ls_lbfgs(a_small_init, deadline - 120, min_time_remaining=40.0, aggressive=True)
        update_best(a_small, _score(a_small), f"lbfgs_n{n_small}")

        a_small_pe, sc_small_pe = _peak_equalize_newton(a_small, deadline - 80, max_iter=200, tol_peak=0.0004)
        update_best(a_small_pe, sc_small_pe, f"peak_eq_n{n_small}")

    # === Phase 6: Bump initialization on n=16384 ===
    print(f"\n[Phase 6] Bump inits (t={time.monotonic()-t_start:.1f}s)")
    n_large = max(n_parent, 16384)
    if time.monotonic() < deadline - 150:
        bump_configs = [
            (0.5, 0.4, 0.0),
            (0.5, 0.3, 0.5),
            (0.4, 0.35, 0.3),
            (0.6, 0.35, -0.3),
            (0.5, 0.5, 0.0),
        ]
        for center, width, asym in bump_configs:
            if time.monotonic() > deadline - 80:
                break
            a_bump = _bump_init(n_large, center, width, asym)
            sc_bump = _score(a_bump)
            if sc_bump < best_score + 0.3:
                a_bump_opt = _run_ls_lbfgs(a_bump, deadline - 60, min_time_remaining=30.0, aggressive=True)
                update_best(a_bump_opt, _score(a_bump_opt), f"bump_{center}_{width}")

    # === Phase 7: Peak peeling on best ===
    print(f"\n[Phase 7] Peak peeling (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_pp, sc_pp = _peak_peel(best_seq, min(deadline - 40, time.monotonic() + 60), max_rounds=100)
        update_best(a_pp, sc_pp, "peak_peel")

    # === Phase 8: Mirror descent ===
    print(f"\n[Phase 8] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_md, sc_md = _mirror_descent(best_seq, min(deadline - 30, time.monotonic() + 50), max_steps=3000, lr_init=0.03)
        update_best(a_md, sc_md, "mirror_descent")

    # === Phase 9: Peak-equalize with tighter tolerance ===
    print(f"\n[Phase 9] Tight peak-equalize (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_te, sc_te = _peak_equalize_newton(best_seq, deadline - 20, max_iter=300, tol_peak=0.0002, max_active=8 * len(best_seq))
        update_best(a_te, sc_te, "peak_eq_tight")

    # === Phase 10: Diverse restarts ===
    print(f"\n[Phase 10] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("sigma_001", 0.005),
            ("sigma_005", 0.02),
            ("sigma_010", 0.05),
            ("sigma_020", 0.10),
            ("sigma_050", 0.20),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            # Quick optimization
            a_r = _run_ls_lbfgs(p_start, deadline, min_time_remaining=3.0, aggressive=False)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _peak_equalize_newton(a_r, deadline - 2, max_iter=50, tol_peak=0.0005)
                update_best(a_r2, sc_r2, f"restart_{label}_pe")

    # === Phase 11: Peak peeling final ===
    print(f"\n[Phase 11] Final peak peeling (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fp, sc_fp = _peak_peel(best_seq, deadline - 2, max_rounds=100)
        update_best(a_fp, sc_fp, "final_peel")

    # === Phase 12: Very tight peak-equalize ===
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_ft, sc_ft = _peak_equalize_newton(best_seq, deadline - 1, max_iter=100, tol_peak=0.0001, max_active=6 * len(best_seq))
        update_best(a_ft, sc_ft, "final_tight_pe")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
