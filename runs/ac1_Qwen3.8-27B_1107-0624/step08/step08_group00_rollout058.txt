Let me analyze the problem carefully.

We need to minimize: `2 * n * max(conv(a,a)) / sum(a)^2`

Where `conv(a,a)` is the full convolution of `a` with itself.

Key observations:
1. The score is scale-invariant (if we multiply `a` by a constant, the score doesn't change since both numerator and denominator scale the same way).
2. The score is `2n * max_b / sum_a^2` where `max_b = max(conv(a,a))`.

Let me think about what makes this small:
- We want `max_b` to be small relative to `sum_a^2 / (2n)`
- `sum_a^2` is maximized when mass is concentrated, but `max_b` is also large then
- We want the convolution to be as flat as possible (low peak-to-average ratio)

The theoretical minimum: if `conv(a,a)` were perfectly flat, then `max_b = sum_a^2 / (2n-1)` (approximately, since sum of all convolution values = sum_a^2). So the minimum score would be approximately `2n / (2n-1) ≈ 1`.

So we're looking for sequences where the autocorrelation is as flat as possible. This is related to Barker sequences, low-PAPR sequences, etc.

"""
Game-theoretic QP + Multiplicative Weights + Active Set Polish
for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Key innovation: Formulates the problem as a zero-sum game between
the peak selection (row player, distribution over convolution positions)
and the sequence (column player). Solves the inner QP via CG with
FFT-based mat-vec for the Hankel matrix, and uses multiplicative weights
for the outer dual update. This avoids non-convexity issues and provides
a principled optimization path.
"""
import numpy as np
import time
from scipy.optimize import minimize

    import cvxpy as cp

except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(p, v, n):
    """Compute Q(p) * v where Q(p)[i,j] = p[i+j], p has length 2n-1, v has length n."""
    v_rev = v[::-1]
    c = _full_convolve(v_rev, p)
    return c[n - 1:2 * n - 1]


def _solve_hankel_cg(p, n, x0, max_iter=200, tol=1e-10, reg=1e-12):
    """Solve (Q(p) + reg*I) x = 1 using CG with warm start."""
    x = x0.copy().astype(np.float64)

    def matvec(v):
        return _hankel_matvec(p, v, n) + reg * v

    b = np.ones(n)
    r = b - matvec(x)
    p_vec = r.copy()
    rs = r.dot(r)
    b_norm = np.linalg.norm(b)
    
    for i in range(max_iter):
        Ap = matvec(p_vec)
        pAp = p_vec.dot(Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x_new = x + alpha * p_vec
        r_new = r - alpha * Ap
        rs_new = r_new.dot(r_new)
        if rs_new < tol * rs or rs_new < tol * b_norm * b_norm:
            x = x_new
            break
        x = x_new
        r = r_new
        p_vec = r + (rs_new / rs) * p_vec
        rs = rs_new

    return x


def _game_qp_optimize(a_init, deadline, min_remaining=30.0, cg_iters=100, verbose=True):
    """
    Game-theoretic QP optimization.
    Outer: multiplicative weights on p (distribution over convolution positions)
    Inner: CG solve of Q(p) x = 1 (the dual of the QP min a^T Q(p) a s.t. sum a = 1)
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Initialize p from current peak distribution
    c = _full_convolve(a, a)
    c_max = c.max()
    beta_init = max(c_max * 0.1, 1e-6)
    p = np.exp(c / beta_init)
    p = p / p.sum()

    eta = 0.5  # MW step for p
    eta_decay = 0.9995
    reg = 1e-10
    stall_count = 0
    max_stall = 50

    for it in range(100000):
        if time.monotonic() + min_remaining > deadline:
            break

        # Inner QP solve: Q(p) x = 1
        x = _solve_hankel_cg(p, n, a, max_iter=cg_iters, reg=reg)
        a_qp = np.maximum(x, 0.0)
        s_qp = a_qp.sum()
        if s_qp < 1e-15:
            p = np.ones(2 * n - 1) / (2 * n - 1)
            continue
        a_qp = a_qp / s_qp

        # Evaluate
        c = _full_convolve(a_qp, a_qp)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_qp.copy()
            stall_count = 0
            if verbose and it % 100 == 0:
                print(f"    GQP iter {it}: score={best_sc:.10f}, p_entropy={-np.sum(p*np.log(p+1e-300)):.3f}")
        else:
            stall_count += 1
            if stall_count >= max_stall:
                # Reset p to be more uniform to escape local optimum
                p = 0.5 * p + 0.5 * np.ones(2 * n - 1) / (2 * n - 1)
                p = p / p.sum()
                stall_count = 0
                max_stall = min(200, max_stall + 50)
                if max_stall > 200:
                    break

        # Update p via multiplicative weights
        c_shifted = c - c_max
        p = p * np.exp(eta * c_shifted / max(c_max, 1e-15))
        p = p / p.sum()
        eta *= eta_decay

        a = a_qp

    return best_a, best_sc


def _multiplicative_weights(a_init, deadline, min_remaining=20.0, verbose=True):
    """
    Fast multiplicative weights (mirror descent on simplex).
    O(n log n) per iteration via FFT-based gradient computation.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Adaptive parameters
    beta = best_sc / (2.0 * n) * 0.05  # smoothing temperature
    beta_min = best_sc / (2.0 * n) * 0.002
    eta = 0.05  # learning rate
    eta_min = 0.0001
    no_improve = 0
    max_no_improve = 200

    t_start = time.monotonic()
    it = 0

    while time.monotonic() + min_remaining < deadline:
        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    MW iter {it}: score={best_sc:.10f}, beta={beta:.2e}, eta={eta:.2e}")
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Compute gradient: g[j] = 2 * sum_k p[k] * a[k-j]
        # Use softmax weights on c
        z = c / beta
        z = z - z.max()
        e = np.exp(-z)  # invert: weight valleys more
        p = e / e.sum()
        
        # g[j] = 2 * sum_k p[k] * a[max(0,min(n-1,k-j))]
        # = 2 * conv(p, a_rev)[n-1:2n-1]
        a_rev = a[::-1]
        pc = _full_convolve(p, a_rev)
        g = 2.0 * pc[n - 1:2 * n - 1]

        # Center g to maintain sum approximately
        g_centered = g - g.mean()

        # Multiplicative update
        a = a * np.exp(-eta * g_centered)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        # Anneal
        eta = max(eta_min, eta * 0.9995)
        beta = max(beta_min, beta * 0.999)

    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00003, lr_init=0.05, verbose=True):
    """Active-set gradient polish: focus on top peaks, reduce them."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.02)

    return best_a, _score(best_a)


def _lbfgs_smooth(a_init, deadline, min_remaining=30.0, verbose=True):
    """L-BFGS with log-sum-exp smoothing of the max."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_time_remaining_safe(deadline, min_remaining) - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 10))))

        def obj(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            return 4.0 * n * a * (h - W)

        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-13, 'maxcor': 50, 'maxls': 30})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a, current_sc


def min_time_remaining_safe(deadline, min_rem):
    return min_rem


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective n."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _perturb_and_optimize(a_base, n_perturb, deadline, rng, min_rem=10.0):
    """Perturb the base solution and re-optimize."""
    n = len(a_base)
    # Log-space perturbation
    sigma = n_perturb
    z = np.log(np.maximum(a_base, 1e-20))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    a_pert = _softmax(z)
    return a_pert

            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # =====================
    # Phase 5: Second round of GQP with tightened params
    # =====================
    print(f"\n[P5] GQP round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_qp2 = min(len(best_seq), 4096)
        a_init2 = _resample(best_seq, n_qp2)
        qp2_deadline = min(deadline - 50, time.monotonic() + 100)
        a_qp2, sc_qp2 = _game_qp_optimize(a_init2, qp2_deadline, min_remaining=30.0,
                                           cg_iters=150, verbose=True)
        if sc_qp2 < best_score:
            # Resample back to full size
            a_qp2_full = _resample(a_qp2, len(best_seq))
            sc_qpf = _score(a_qp2_full)
            if sc_qpf < best_score:
                best_seq = a_qp2_full.copy()
                best_score = sc_qpf
                print(f"  NEW BEST: {best_score:.10f} [GQP2] t={time.monotonic()-t_start:.1f}s")

    # =====================
    # Phase 6: Shrinkage
    # =====================
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 40)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")

    # =====================
    # Phase 7: Multi-n exploration
    # =====================
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    current_n = len(best_seq)
    n_tries = [current_n // 2, current_n * 2, 2048, 4096, 8192, 16384]
    n_tries = sorted(set([x for x in n_tries if 512 <= x <= 32768 and x != current_n]))

    for n_try in n_tries[:4]:
        if time.monotonic() > deadline - 80:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_seq = a_r.copy()
            best_score = sc_r
            print(f"  Resampled n={n_try}: {sc_r:.10f} [NEW BEST]")
        # Quick MW polish at this n
        if time.monotonic() < deadline - 40:
            a_mw2, sc_mw2 = _multiplicative_weights(a_r, min(deadline - 30, time.monotonic() + 60),
                                                     min_remaining=15.0, verbose=False)
            if sc_mw2 < best_score:
                best_seq = a_mw2.copy()
                best_score = sc_mw2
                print(f"  MW n={n_try}: {sc_mw2:.10f} [NEW BEST]")

    # =====================
    # Phase 8: Final ultra-fine polish
    # =====================
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                            tol_frac_start=0.001, tol_frac_end=0.00001,
                                            lr_init=0.005, verbose=False)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  FINAL POLISH: {sc_fin:.10f} [NEW BEST]")

        # Tiny MW
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            a_fin2, sc_fin2 = _multiplicative_weights(best_seq, deadline - 2,
                                                       min_remaining=2.0, verbose=False)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  FINAL MW: {sc_fin2:.10f} [NEW BEST]")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
