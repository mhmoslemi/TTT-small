    """Conjugate gradient solver for the regularized active set problem."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)

        alpha = rs / pAp

        rs_new = np.dot(r, r)

Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
1. Start from parent, normalize, optional symmetrization
2. Log-sum-exp smoothed L-BFGS with adaptive beta continuation (full schedule)
3. Active-set Newton with adaptive tolerance for precise local optimization
4. Peak shaving - targeted reduction of top convolution peaks via iterative redistribution
5. Multi-n exploration (16384, 24576, 32768) with interpolation
6. Randomized restarts with log-space perturbations
7. Final ultra-fine polish with smallest beta and tightest Newton

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsobj(z, beta, n, do_sym):
    a = _softmax(z)
    if do_sym:
        a = _symmetrize(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n, do_sym):
    a = _softmax(z)
    if do_sym:
        a = _symmetrize(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if do_sym:
        grad = _symmetrize(grad)
    return grad


def _hessian_vec(v, A_vec, m, mu, n):
    return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v


def _cg_solve(A_vec, m, mu, n, rhs, max_cg=200, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = _hessian_vec(p, A_vec, m, mu, n)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _peak_shave(a_init, deadline, n_attempts, do_sym):
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())

        top_k = np.argsort(c)[::-1]
        n_top = max(5, min(len(c), int(0.08 * len(c))))
        top_positions = top_k[:min(n_top, 30)]

        a_new = best_a.copy()
        reduction = 0.997

        for k_pos in top_positions:
            i_lo = max(0, int(k_pos) - n + 1)
            i_hi = min(int(k_pos), n - 1)
            if i_lo > i_hi:
                continue
            prods = a_new[i_lo:i_hi + 1] * a_new[i_hi - i_lo:]
            max_idx = int(np.argmax(prods))
            i = i_lo + max_idx
            j = int(k_pos) - i
            if 0 <= i < n and 0 <= j < n:
                if a_new[i] >= a_new[j]:
                    a_new[i] *= reduction
                else:
                    a_new[j] *= reduction

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        if do_sym:
            a_new = _symmetrize(a_new)
            a_new = _normalize(a_new)

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            break

    return best_a, best_sc


def _fista_step(a_init, deadline, n_steps, do_sym):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)

    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad = 2.0 * _corr_a_A(y, w, n)
        if do_sym:
            grad = _symmetrize(grad)
        g_t = grad - np.mean(grad)

        step_size = 0.05 / (1.0 + 0.02 * step)
        a_new = y - step_size * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        if do_sym:
            a_new = _symmetrize(a_new)
            a_new = _normalize(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        if do_sym:
            y = _symmetrize(y)
            y = _normalize(y)
        a = a_new
        t_mom = t_new

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = n // 4
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _run_lbfgs(a_init, deadline, min_time_remaining, do_sym):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    if base_beta < max_c * 0.001:
        base_beta = max_c * 0.001

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                                  0.015625, 0.0078, 0.0039, 0.002, 0.001,
                                  0.0005, 0.00025, 0.000125, 0.0000625,
                                  0.000031, 0.000015])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.005 * max(1.0, n / 16384.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        obj_fn = partial(_lsobj, beta=beta, n=n, do_sym=do_sym)
        grad_fn = partial(_lsgrad, beta=beta, n=n, do_sym=do_sym)

        try:
            result = minimize(
                obj_fn,
                z,
                jac=grad_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            if do_sym:
                a_new = _symmetrize(a_new)
            a_new = _normalize(a_new)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i >= len(betas) - 3:
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

        ps_dl = min(deadline - 300, time.monotonic() + 80)
        a_ps, sc_ps = _peak_shave(best_seq, ps_dl, n_attempts=50, do_sym=True)
        update_best(a_ps, sc_ps, "peak_shave_sym")

        if sc_ps < best_score:
            a_ps2 = _run_lbfgs(a_ps, deadline, min_time_remaining=200.0, do_sym=True)
            update_best(a_ps2, _score(a_ps2), "lbfgs_post_shave")

    # Phase 5: FISTA (momentum-based, good for escaping shallow local minima)
    print(f"\n[Phase 5] FISTA (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        fista_dl = min(deadline - 250, time.monotonic() + 80)
        a_fista = _fista_step(best_seq, fista_dl, n_steps=15000, do_sym=True)
        update_best(a_fista, _score(a_fista), "fista")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 24576, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent and abs(n_try - 16384) < 100:
            continue

        # Interpolate parent to new n
        a_init_n = _interpolate_to_n(working, n_try)
        sc_init = _score(a_init_n)

        # Also try structured profiles
        a_block = _make_profile(n_try, "cos2")
        sc_block = _score(a_block)
        a_taper = _make_profile(n_try, "tapered")
        sc_taper = _score(a_taper)

        # Pick best initialization
        candidates = [(a_init_n, sc_init, "interp"),
                      (a_block, sc_block, "cos2"),
                      (a_taper, sc_taper, "taper")]
        candidates.sort(key=lambda x: x[1])
        best_init, best_init_sc, best_init_label = candidates[0]

        a_opt = _run_lbfgs(best_init, deadline, min_time_remaining=100.0, do_sym=True)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_{best_init_label}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 40, time.monotonic() + 50),
                                                  max_iter=150,
                                                  tol_list=[0.002, 0.001, 0.0005, 0.0002],
                                                  do_sym=True)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_configs = [
            ("perturb_001", 0.005),
            ("perturb_005", 0.02),
            ("perturb_01", 0.05),
            ("perturb_05", 0.15),
            ("perturb_1", 0.3),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0, do_sym=True)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 10, time.monotonic() + 25),
                                                  max_iter=80,
                                                  tol_list=[0.001, 0.0005, 0.0002],
                                                  do_sym=True)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Tight active-set Newton
        as_final_dl = min(deadline - 3, time.monotonic() + 50)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final_dl, max_iter=300,
                                            tol_list=[0.0005, 0.0003, 0.0001, 0.00005, 0.00002, 0.00001],
                                            do_sym=True)
        update_best(a_fin, sc_fin, "final_as_sym")

        # Peak shaving on result
        if time.monotonic() < deadline - 3:
            a_ps3, sc_ps3 = _peak_shave(best_seq, min(deadline - 3, time.monotonic() + 20),
                                         n_attempts=30, do_sym=True)
            update_best(a_ps3, sc_ps3, "final_shave")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 6 - time.monotonic()
            if remaining > 5:
                max_iter = min(3000, max(100, int(remaining / 0.06)))
                obj_fn = partial(_lsobj, beta=beta_final, n=n, do_sym=True)
                grad_fn = partial(_lsgrad, beta=beta_final, n=n, do_sym=True)
                try:
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 80, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    a_fin3 = _symmetrize(a_fin3)
                    a_fin3 = _normalize(a_fin3)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # One more Newton pass on absolute best
        if time.monotonic() < deadline - 3 and best_seq is not None:
            a_final2, sc_final2 = _active_set_newton(best_seq, min(deadline - 1, time.monotonic() + 20),
                                                      max_iter=100,
                                                      tol_list=[0.0001, 0.00005, 0.00002],
                                                      do_sym=False)
            update_best(a_final2, sc_final2, "final_as_nosym")

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        # Also check symmetrized version
        final_sym = _symmetrize(final)
        final_sym = _normalize(final_sym)
        sc_final_sym = _score(final_sym)
        if sc_final_sym < sc_final:
            final = final_sym
            sc_final = sc_final_sym
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
