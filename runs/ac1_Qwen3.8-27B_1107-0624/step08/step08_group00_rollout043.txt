"""
Optimized convex solver for min 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: (1) corrected gradient with 1/beta factor,
(2) aggressive multi-stage beta annealing, (3) subgradient polish with Polyak
step size and line search, (4) momentum-based subgradient for faster convergence,
(5) multi-n exploration with resampling.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return (4.0 * n / beta) * a * (h - W)


class _SmoothProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.25) ** 2) + np.exp(-10.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(current_a, current_a)
    c_max = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.01)

    # Multi-stage beta: start large (smooth), end very small (sharp)
    n_stages = 20
    betas = base_beta * (0.5 ** np.arange(n_stages))
    # Add extra fine stages at the end
    extra = betas[-1] * (0.5 ** np.arange(6))
    betas = np.concatenate([betas, extra])

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.0015 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(50, int(remaining / (est_time * 8))))

        prob = _SmoothProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 60,
                    'maxfun': 30000,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"    LBFGS stage {i}/{len(betas)} beta={beta:.2e}: score={sc_new:.10f} nit={result.nit}")
            else:
                if i % 5 == 0:
                    print(f"    LBFGS stage {i}/{len(betas)}: no improvement (best={current_sc:.10f})")
        except Exception as e:
            if i % 5 == 0:
                print(f"    LBFGS stage {i}: error {e}")

    return current_a


def _subgradient_polish(a_init, deadline, min_time_remaining=10.0):
    """Subgradient method with Polyak step, momentum, and adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c = _full_convolve(a, a)
    c_max = float(c.max())
    upper_bound = c_max
    best_a = a.copy()
    best_val = c_max

    eps = c_max * 0.05
    no_improve = 0
    max_no_improve = 500
    momentum = 0.5
    g_mom = np.zeros(n)
    iteration = 0

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())

        if c_max < best_val - 1e-14:
            best_val = c_max
            best_a = a.copy()
            no_improve = 0
            upper_bound = c_max

        tol = eps * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size for efficiency
        if n_active > max(10, int(0.05 * n)):
            c_active = c[active]
            thresh = np.percentile(c_active, 100 - 100.0 * max(10, int(0.05 * n)) / n_active)
            active = active & (c >= thresh)
            n_active = int(active.sum())

        # Compute subgradient
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h
        g = g - g.mean()

        # Momentum
        g_mom = momentum * g_mom + (1 - momentum) * g

        g_norm_sq = float(g_mom @ g_mom)
        if g_norm_sq < 1e-25:
            break

        # Polyak step size
        gap = upper_bound - c_max
        if gap < 1e-12:
            upper_bound = c_max + max(c_max * 1e-8, 1e-15)
            gap = upper_bound - c_max
        eta = gap / (g_norm_sq + 1e-20)
        g_norm = np.sqrt(g_norm_sq)
        eta = min(eta, 0.05 / g_norm)

        # Line search with backtracking
        improved = False
        for _ in range(30):
            a_try = a - eta * g_mom
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                eta *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(c_try.max())
            if sc_try < c_max * (1.0 - 1e-12):
                a = a_try
                improved = True
                break
            eta *= 0.5
            if eta < 1e-20:
                break

        if improved:
            if no_improve == 0 and eps > c_max * 0.005:
                eps = max(c_max * 0.005, eps * 0.97)
            momentum = min(0.8, momentum * 1.01)
        else:
            no_improve += 1
            eps = max(c_max * 0.002, eps * 0.95)
            momentum = max(0.3, momentum * 0.98)
            if no_improve >= max_no_improve:
                a = best_a.copy()
                g_mom = np.zeros(n)
                eps = max(c_max * 0.001, eps * 0.5)
                no_improve = 0
                momentum = 0.5
                if eps < c_max * 0.0005:
                    break

        if iteration % 5000 == 0:
            print(f"    SubGrad iter {iteration}: val={c_max:.10f}, best={best_val:.10f}, eps={eps:.2e}, mom={momentum:.3f}")

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a, _score(best_a)


def _direct_peak_reduction(a_init, deadline, min_time_remaining=10.0):
    """Directly reduce the dominant peak using its gradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < best_sc / (2.0 * n) - 1e-15:
            best_sc = 2.0 * n * c_max
            best_a = a.copy()

        # Find the dominant peak
        k_star = int(np.argmax(c))

        # Gradient direction to reduce this peak
        # d conv(a,a)[k_star]/da[j] = 2*a[k_star - j] (for valid k_star-j)
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        g = np.zeros(n)
        # For j in [k_star - i_hi, k_star - i_lo]:
        j_lo = k_star - i_hi
        j_hi = k_star - i_lo
        g[j_lo:j_hi + 1] = 2.0 * a[i_lo:i_hi + 1][::-1]
        # Fix the diagonal if k_star is even
        if k_star % 2 == 0:
            g[k_star // 2] = a[k_star // 2]  # Only one term, not two

        g = g - g.mean()  # Maintain sum constraint

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Step with line search
        eta = 0.01 / g_norm
        improved = False
        for _ in range(40):
            a_try = a - eta * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                eta *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1 - 1e-12):
                a = a_try
                improved = True
                break
            eta *= 0.5
            if eta < 1e-20:
                break

        if not improved:
            break

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a, _score(best_a)


def _coordinate_descent_polish(a_init, deadline, min_time_remaining=10.0):
    """Coordinate-wise optimization: adjust each element to reduce the peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Pre-compute which elements affect each position
    # For position j, changing a[j] affects conv[k] for k in [j, j+n-1]
    # d conv[k]/da[j] = a[k-j] + (1 if k-j != j else 0) * a[j] ... 
    # Actually d conv[k]/da[j] = a[k-j] + a[j-(k-j)] = a[k-j] + a[2j-k] (with bounds)

    # Strategy: repeatedly find the peak, then adjust the element that most
    # reduces it while minimally increasing other values.

    max_rounds = 50
    for round_num in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < best_sc / (2.0 * n) - 1e-15:
            best_sc = 2.0 * n * c_max
            best_a = a.copy()

        # Find top few peaks
        top_k = 3
        c_idx = np.argsort(c)[::-1][:top_k]

        for k_star in c_idx:
            if time.monotonic() + min_time_remaining > deadline:
                break

            # For this peak, find the element to adjust
            # d conv[k_star]/da[j] = 2*a[k_star-j] (or a[j] if k_star=2j)
            i_lo = max(0, k_star - (n - 1))
            i_hi = min(n - 1, k_star)
            j_lo = k_star - i_hi
            j_hi = k_star - i_lo

            if j_lo > j_hi:
                continue

            # We want to decrease a[j] where a[k_star-j] is large
            # Try decreasing the element with largest a[k_star-j]
            contrib = a[i_lo:i_hi + 1]
            j_adjust = j_lo + int(np.argmax(contrib))

            # Try decreasing a[j_adjust] slightly
            delta = a[j_adjust] * 0.01
            a_try = a.copy()
            a_try[j_adjust] -= delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(c_try.max())
            if sc_try < c_max * (1 - 1e-12):
                a = a_try

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2n*max(conv(a,a))/sum(a)^2 using:
    1. L-BFGS with corrected gradient and aggressive beta annealing
    2. Subgradient polish with momentum and Polyak step
    3. Direct peak reduction
    4. Multi-n exploration
    5. Final fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS (main optimization) ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 500, t_start + 750)
    a_main = _run_lbfgs(best_seq, p1_deadline, min_time_remaining=450.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  L-BFGS: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Subgradient polish ===
    print(f"\n[P2] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 350, time.monotonic() + 250)
    a_sub, sc_sub = _subgradient_polish(best_seq, p2_deadline, min_time_remaining=100.0)
    if sc_sub < best_score:
        best_seq = a_sub.copy()
        best_score = sc_sub
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [subgrad] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS round 2 (fine-tuning) ===
    print(f"\n[P3] L-BFGS fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p3_deadline = min(deadline - 200, time.monotonic() + 150)
        a_r2 = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=180.0)
        sc_r2 = _score(a_r2)
        if sc_r2 < best_score:
            best_seq = a_r2.copy()
            best_score = sc_r2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Subgradient polish 2 ===
    print(f"\n[P4] Subgradient polish 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p4_deadline = min(deadline - 100, time.monotonic() + 100)
        a_sub2, sc_sub2 = _subgradient_polish(best_seq, p4_deadline, min_time_remaining=80.0)
        if sc_sub2 < best_score:
            best_seq = a_sub2.copy()
            best_score = sc_sub2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [subgrad2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Direct peak reduction ===
    print(f"\n[P5] Direct peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        p5_deadline = min(deadline - 50, time.monotonic() + 40)
        a_peak, sc_peak = _direct_peak_reduction(best_seq, p5_deadline, min_time_remaining=30.0)
        if sc_peak < best_score:
            best_seq = a_peak.copy()
            best_score = sc_peak
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peak_red] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [24576, 12288, 8192, 32768]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        # Estimate if this n could be better
        # Score scales roughly as 2n * (flatness), so larger n allows flatter convolution
        # but has more degrees of freedom to optimize
        print(f"  Trying n={n_try}...")
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        if sc_init < best_score:
            best_seq = a_init.copy()
            best_score = sc_init
            print(f"  BEST from resample: {best_score:.10f} (n={n_try}) t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < deadline - 40:
            a_opt = _run_lbfgs(a_init, min(deadline - 30, time.monotonic() + 30), min_time_remaining=25.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _subgradient_polish(best_seq, deadline - 2, min_time_remaining=2.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
