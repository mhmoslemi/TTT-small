def _mp2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fc(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _mp2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _sc(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fc(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _sm(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gr(ad, w):
    ar = ad[::-1]

    return wc[n - 1:2 * n - 1]


def _lse_obj(z, beta, n):

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm

    c = _fc(a, a)
